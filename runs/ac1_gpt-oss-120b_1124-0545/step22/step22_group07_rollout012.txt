"""
Propose a step‑function (non‑negative heights) that minimises ``evaluate_sequence``.

Algorithm
---------
1.  Build a large, diverse pool of analytic and random seeds:
    * uniform, geometric, symmetric‑geometric, power‑law, beta, logistic,
      gamma and piecewise‑linear families.
    * Dirichlet draws with a wide range of concentration parameters.
    * The previously found best sequence (if any) and a few permutations of it.
    All seeds are ℓ₁‑normalised (sum = 1).

2.  Evaluate the exact surrogate objective (the same as
    ``evaluate_sequence`` after ℓ₁‑normalisation):
       S(p) = 2·n·max(conv(p,p)).
   Keep the best few seeds (``top_k``).

3.  For each of the top seeds run a *two‑stage* optimisation:

   a) **Smoothed Adam** – minimise a log‑sum‑exp smooth approximation of the
      max‑convolution.  The temperature ``β`` is increased gradually and
      80 000 inner iterations are performed for each temperature.  Adam is
      used in the log‑softmax (probability) space.

   b) **Exact exponentiated‑gradient (EG)** – mirror‑descent on the true
      max‑convolution.  The step‑size decays as ``η₀ / t^{0.6}`` (slightly
      slower than 1/√t), which gives a more accurate final polishing.

   The better of the two refinements becomes the candidate for that seed.

4.  Run a *long* EG polishing on the current global best using the remaining
    time budget.

5.  Finish with a short permutation‑annealing (swap moves only) to shave off
    any remaining excess.

All heavy work is performed with FFT‑based convolution (``O(n log n)``).  The
search respects the 1000 s budget, prints timing / objective information,
stores the final sequence in ``height_sequence_1`` and returns it as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real, length = n+m‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate (identical to the true objective after ℓ₁‑normalisation)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) for a probability vector (sum = 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑like density on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_decay(n: int, k: float) -> np.ndarray:
    """Monotonic logistic (sigmoid) decay."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp(k * (i - centre)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gamma_shape(n: int, shape: float, rate: float) -> np.ndarray:
    """Gamma‑like density: i^{shape‑1}·exp(-rate·i).  i starts at 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1.0)) * np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the exact max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 1_800_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (a_i = exp(log_a_i) / Σ).
    Step size ≈ η0 / t^{0.6} (a little slower decay than 1/√t).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)          # avoid log(0)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient w.r.t. the active entry
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # slower decay gives a bit more movement at large t
        eta = eta0 / (it ** 0.6 + 1e-9)
        log_a -= eta * g
        a = np.exp(log_a - np.max(log_a))   # numerical stability
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Minimise the log‑sum‑exp smoothed max of conv(p,p) using Adam.
    The temperature β is increased gradually; larger β makes the smooth
    approximation tighter.
    """
    if beta_schedule is None:
        # start very smooth, finish extremely sharp
        beta_schedule = [5, 20, 80, 320, 1280, 5120, 20_480,
                         81_920, 327_680, 1_310_720]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.15                                 # relatively aggressive
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                     # soft‑max weights

            # gradient w.r.t a (see derivation in the description)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule to log‑softmax space
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            log_a = np.clip(log_a, -30.0, 30.0)        # avoid overflow

            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # allocate roughly equal shares of the remaining budget to each seed
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # keep ~80 % of the remaining budget for optimisation,
        # split evenly among the still‑to‑process seeds.
        share = remaining * 0.80 / max(1, (top_k - idx))
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated time ≈ {share:.1f}s")

        # ---- smoothed Adam optimisation --------------------------------
        seq_adam, val_adam = _adam_smooth_opt(seed_vec,
                                              sub_deadline,
                                              iters_per_beta=70_000)

        # ---- EG polishing (uses whatever time is left) -----------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam,
                                               sub_deadline,
                                               max_iters=2_000_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick the better of the two
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Final long EG polishing (uses all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        # keep a short safety margin for the final permutation tweak
        final_eg_deadline = time.time() + max(5.0, remaining * 0.85)
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 final_eg_deadline,
                                                 max_iters=3_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Very short permutation tweak (a few seconds) – may shave a bit
    # --------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 20.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=80_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
