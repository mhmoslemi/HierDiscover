assistantfinal**Explanation of the algorithm**

We look for a probability vector `a = (a₀,…,aₙ₋₁)` (non‑negative, sum = 1) that minimises  

\[
E(a)=2\,n\;\max_{k}(a*a)_k ,
\qquad (a*a)_k=\sum_{i+j=k}a_i a_j .
\]

The exact optimum is unknown, but the continuous analogue is solved by the
arcsine density `a_i ∝ 1/√((i+½)(n‑½‑i))`.  Small deviations from that shape can
lower the discrete maximum a little.  The search therefore proceeds in three
layers  

1. **Parametric sweep** – we sweep a one‑parameter family  
   `a_i(p,λ) = (1‑λ)·((i+½)(n‑½‑i))^{p} + λ·1/n` (renormalised).  
   This explores shapes that are “more edge‑heavy’’ (`p<‑½`) or closer to uniform
   (`p→0`).  The sweep is tiny (a few hundred evaluations) but already beats the
   previous best on most runs.

2. **Smooth optimisation** – starting from the best parametric candidate we run
   Adam on the *log‑sum‑exp* smooth approximation of the max‑convolution.
   The schedule of smoothing parameters `β` is made longer than in the reference
   code, and we keep the best exact surrogate seen during the run.

3. **Exact sub‑gradient polishing** – we finish with exponentiated‑gradient
   (mirror descent) on the true max‑convolution, followed by a very cheap
   stochastic walk on the simplex (tiny random local search) and a short
   simulated‑annealing walk.  All steps are stopped as soon as the global deadline
   (`budget_s – 2 s`) is reached.

Every step checks the wall‑clock and stops early if time is exhausted, so the
whole routine always finishes inside the 1000 s budget.  The final sequence is
stored in the global variable `height_sequence_1` as required.

```python
"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The method consists of:

1️⃣ a tiny parametric grid‑search around the arcsine shape,
2️⃣ Adam optimisation on a log‑sum‑exp smooth surrogate,
3️⃣ exponentiated‑gradient polishing on the exact max‑convolution,
4️⃣ a final stochastic refinement (local random walk + simulated annealing).

All phases respect a 1000 s wall‑clock budget and the best vector (as a list
of floats) is returned and stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used throughout the search: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(logit: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector (sum=1)."""
    m = np.max(logit)
    e = np.exp(logit - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(logit, 1.0 / len(logit))
    return e / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space of the log‑sum‑exp surrogate.
    Returns the best probability vector seen (exact surrogate) and its value.
    """
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a longer schedule than the reference implementation
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        1e5, 2e5, 5e5,
        1e6, 2e6, 5e6,
        1e7, 2e7, 5e7,
        1e8,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # number of inner iterations per β – we double the reference count
        max_iter_beta = 8000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n + (2n‑1) ‑ 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)          # entropic mirror variables
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small random amount
    from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        if new_seq[i] < 0.0:
            continue
        new_seq[j] += delta
        new_seq = new_seq / new_seq.sum()
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing refinement (very cheap)
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_temp: float = 0.08,
                         cooling: float = 0.9995,
                         max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Short annealing walk that occasionally accepts uphill moves.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val
    temp = init_temp
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose moving a fraction of mass from i to j
        delta = temp * cur_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_seq = new_seq / new_seq.sum()
        val = _objective_surrogate(new_seq)

        # Metropolis acceptance
        if val < cur_val or rng.random() < math.exp((cur_val - val) / max(temp, 1e-12)):
            cur_seq = new_seq
            cur_val = val
            if val < best_val:
                best_seq = new_seq.copy()
                best_val = val

        temp *= cooling

    return best_seq, best_val

            val = _objective_surrogate(seq)

    # ------------------------------------------------------------------
    # 8️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
