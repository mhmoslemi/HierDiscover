"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  The optimiser combines:

1.  A fast multi‑start L‑BFGS phase on a log‑sum‑exp surrogate of the
    max‑convolution.  The surrogate temperature (beta) is increased
    gradually; each beta receives many L‑BFGS iterations so that the
    optimiser can converge tightly.

2.  A final mirror‑descent (exponentiated‑gradient) polishing on the
    exact max‑convolution objective.  The step‑size uses a larger
    initial value (eta0=20) and decays as 1/√t.

3.  A few random‑restart refinements if time permits.

All helper functions are defined at module level, use only numpy (and
SciPy if available) and respect the 1000 s wall‑time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – used only for the L‑BFGS smoothing phase
# ----------------------------------------------------------------------
try:                                 # pragma: no cover
    import scipy.optimize as _opt
except Exception:                   # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numerical utilities (FFT‑based full convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Softmax utilities – we work in log‑space to stay on the simplex
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact value for a probability vector)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Families of candidate initial sequences (all normalised to sum=1)
# ----------------------------------------------------------------------
def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    """a_i ∝ exp(-(i)^2/(2σ²)).  Symmetric about 0, truncated to length n."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(- (i ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Smoothed max‑convolution objective (log‑sum‑exp) – used by L‑BFGS
# ----------------------------------------------------------------------
def _smooth_obj_and_grad(v_local: np.ndarray,
                         n: int,
                         beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (objective, gradient) of the smoothed objective:
        2·n·(max_c + log Σ exp(beta·(c−max_c))/beta)
    where c = conv(a, a) and a = softmax(v_local).
    """
    a = _softmax(v_local)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # shift for numerical stability
    shift = max_c
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    # gradient w.r.t. a
    w = exp_beta / sum_exp                    # weights for each conv entry
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    # chain rule: ∂obj/∂v = a * (grad_a − (grad_a·a))
    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max‑conv objective
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 20.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # work in log‑space (softmax) to enforce positivity and sum=1
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("input must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – fall back to uniform
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
