def _lp_step(f, M):
    n = len(f)
    # build sparse A
    rows = []
    cols = []
    data = []
    for i in range(n):
        # column i
        # rows k = i + np.arange(n)
        rows_i = i + np.arange(n)   # shape (n,)
        cols_i = np.full(n, i, dtype=int)
        data_i = f  # f values length n
        rows.extend(rows_i.tolist())
        cols.extend(cols_i.tolist())
        data.extend(data_i.tolist())
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g
```

def _conv_fft(x, y):

def _lp_step(f, M, max_time):
    n = len(f)
    # Build sparse matrix A of shape (2n-1, n)
    rows = np.empty(n*n, dtype=np.int32)
    cols = np.empty(n*n, dtype=np.int32)
    data = np.empty(n*n, dtype=np.float64)
    idx = 0
    for i in range(n):
        rows[idx:idx+n] = i + np.arange(n, dtype=np.int32)
        cols[idx:idx+n] = i
        data[idx:idx+n] = f[i]
        idx += n
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    bounds = [(0.0, None)] * n
    # Solve LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': max_time})
    if not res.success:
        return None
    return res.x
```

"""
Optimiser for the step‑function bound.

The evaluation to minimise is

    2·n·max(conv(a,a)) / (sum(a))² ,   a_i ≥ 0 .

Because the expression is invariant under scaling we work with ℓ₁‑normalised
vectors (sum = 1) and minimise the surrogate `2·n·max(conv(a,a))`.

The optimiser combines three ideas:

1️⃣  **Analytic seeds** – a wide family (uniform, triangular, sine windows,
    Gaussians, exponentials, power‑laws, beta‑shapes, Dirichlet draws).  The
    best few are kept.

2️⃣  **LP‐based boosting** – given the current vector `f` we solve the linear
    programme

        maximise   Σ g_j
        subject to  conv(f,g) ≤ M   (M = max(conv(f,f)))
                  g ≥ 0

    (the convolution is linear in the unknown `g`).  Any feasible `g` with
    Σ g > 1 can be mixed with `f` to strictly lower the maximum autocorrelation.
    The optimal mixing weight `t` is found by a cheap 1‑D line search.

3️⃣  **Exponentiated‑gradient polishing** – a mirror‑descent method on the
    exact sub‑gradient of the active convolution entry (the one attaining the
    maximum).  This is run repeatedly after every LP boost and finally for the
    remaining time budget.

Random perturbations are tried when time permits to escape shallow minima.
The whole search stops before the budget expires and returns the best
sequence found (as a plain Python list of floats).
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc). Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Simple simplex projection (used after random perturbations)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser with exact sub‑gradient
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (guaranteed feasibility).
    Returns (best_seq, best_surrogate).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑parameters (softmax will enforce sum=1 and non‑negativity)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))                     # active index

        # exact sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]                  # off‑diagonal counted twice
        # diagonal already counted twice; need only a[i] (since product a_i^2)
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        # mirror‑descent update in log‑space
        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)               # <g, a>
        grad_log = a * (g - dot)         # ∇_log objective
        log_a = log_a - eta * grad_log
        a = np.exp(log_a)
        a /= a.sum()                     # re‑normalise (softmax)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming boost: maximise Σ g_j subject to conv(f,g) ≤ M
# ----------------------------------------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j   s.t.   conv(f,g) ≤ M, g ≥ 0.
    Returns the optimal g (may be None on failure).
    """
    n = len(f)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = i + np.arange(n, dtype=np.int32)  # rows i … i+n-1
        cols[pos:pos + n] = i                               # column i
        data[pos:pos + n] = f[i]                            # coefficient f_i
        pos += n
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)   # maximise Σ g ⇒ minimise −Σ g
    bounds = [(0.0, None)] * n
    # HiGHS (default) handles sparse matrices efficiently.
    try:
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception:   # pragma: no cover
        return None
    if not res.success:
        return None
    return res.x


def _best_mix(f: np.ndarray,
              g: np.ndarray,
              t_vals: np.ndarray) -> tuple[float, float]:
    """
    Given two normalised vectors f and g, search a mixing coefficient t∈t_vals
    that minimises max(conv((1−t)f + t g, (1−t)f + t g)).
    Returns (best_t, best_max_conv).
    """
    conv_ff = _conv_fft(f, f)
    conv_gg = _conv_fft(g, g)
    conv_fg = _conv_fft(f, g)

    best_t = 0.0
    best_max = np.max(conv_ff)   # corresponds to t = 0

    for t in t_vals:
        # quadratic combination of the three convolutions
        comb = ((1 - t) ** 2) * conv_ff + 2 * t * (1 - t) * conv_fg + (t ** 2) * conv_gg
        cur_max = np.max(comb)
        if cur_max < best_max - 1e-14:   # tolerance to avoid noise
            best_max = cur_max
            best_t = t
    return best_t, best_max

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine builds analytic seeds, refines them with a linear‑programming
    boost and exponentiated‑gradient polishing, and finally returns the best
    sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin for final eval

    # --------------------------------------------------------------
    # Problem size – 1500 gives a reasonable trade‑off between quality
    # and speed (the asymptotic limit does not depend on n).
    # --------------------------------------------------------------
    n = 1500

    # --------------------------------------------------------------
    # Load a previous solution if it exists (helpful for continuation runs)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:   # pragma: no cover
            prev_arr = None

    # --------------------------------------------------------------
    # 1. Build a pool of analytical seeds
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows – moderate sweep
    for gamma in np.linspace(0.2, 4.0, 25):
        seeds.append(_sine_window(n, gamma))

    # Gaussian – sweep sigma fraction
    for sigma_frac in np.linspace(0.05, 0.30, 15):
        seeds.append(_gaussian(n, sigma_frac))

    # shifted Gaussians – sweep centre and width
    for mu_frac in np.linspace(0.15, 0.85, 8):
        for sigma_frac in np.linspace(0.05, 0.25, 8):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # exponential decays – log‑space sweep
    for rate in np.logspace(-5, 0.5, 30):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law sweeps
    for p in np.linspace(0.0, 6.0, 20):
        seeds.append(_power(n, p))

    # beta‑shape grid
    alphas = np.geomspace(0.1, 10.0, 8)
    betas  = np.geomspace(0.1, 10.0, 8)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # random Dirichlet draws
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for conc in conc_vals:
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:

        vec = np.maximum(vec, 0.0)

        if s == 0.0:

        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()

    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
