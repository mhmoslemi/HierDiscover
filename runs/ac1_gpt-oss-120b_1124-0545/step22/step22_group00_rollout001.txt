"""
Optimiser for the step‑function upper bound.

The search proceeds in three main stages:

1. **Diverse seeding** – a rich library of analytic families (U‑shape,
   reversed triangular, symmetric power, Beta with <1 parameters, exponential
   decays, mixtures) plus random Dirichlet draws and an up‑sampled copy of
   the previous best solution (if any).  All candidates are ℓ₁‑normalised
   and ranked by the fast surrogate 2·n·max(conv(a,a)) (the true score when
   sum(a)=1).

2. **Multi‑start refinement** – each of the best seeds is locally improved
   by three complementary optimisers:
      * Exponentiated‑gradient (mirror–descent on the exact max‑convolution
        sub‑gradient).
      * Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate with a
        continuation schedule over the smoothing parameter β.
      * Simulated‑annealing with two move types (global Gaussian perturbation
        and targeted mass‑transfer between two coordinates).

   The best of the three refinements is taken as the seed’s local optimum,
   and the global best is updated whenever an improvement is found.

3. **Final polishing** – a long Adam run (high‑β continuation) followed by
   a high‑budget SA run refines the global champion.

The routine respects a hard time budget (default 1000 s) and stores the
best sequence in the global variable ``height_sequence_2`` for possible
re‑use in subsequent calls.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact max for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a, a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _rev_triangular(n: int) -> np.ndarray:
    """U‑shaped (low centre, high edges)."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.abs(i - c) / c
    a = np.clip(a, 0.0, 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_power(n: int, p: float) -> np.ndarray:
    """Symmetric power law a_i ∝ (i+1)^p + (n-i)^p (U‑shape for p>1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_u_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution (α<1 or β<1 → U‑shape)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _upsample(vec: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation up‑sampling to new length."""
    old_n = len(vec)
    if old_n == new_n:
        return vec.copy()
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    up = np.interp(new_x, old_x, vec)
    s = up.sum()
    return up / s if s > 0 else _uniform(new_n)


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set (averaged)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Averaged sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)   # length 2n‑1
    count = active.sum()
    if count == 0:
        return np.zeros_like(a)
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)                     # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / count
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the exact max‑convolution (full active set)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        grad = _subgradient_max_conv(a)
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)          # chain rule for log‑softmax
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   beta_schedule=None,
                   iters_per_beta: int = 20_000,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) approximation of max(conv)."""
    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0,
            5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0,
        ]

    v = np.log(start_seq + 1e-30)          # log‑softmax variables
    best_seq = _softmax(v)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)
        m = np.zeros_like(v)
        v_hat = np.zeros_like(v)
        t = 0

        while t < iters_per_beta and time.time() < deadline:
            t += 1
            a = _softmax(v)
            conv = _conv_fft(a, a)
            max_c = conv.max()
            shift = max_c
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            w = exp_beta / sum_exp                 # softmax weights over conv entries

            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)            # chain‑rule for log‑softmax

            # Adam update with bias correction
            m = beta1 * m + (1 - beta1) * grad_v
            v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat_corr = v_hat / (1 - beta2 ** t)

            step = lr * m_hat / (np.sqrt(v_hat_corr) + eps)
            v -= step

            # evaluate surrogate (exact max) after the step
            a_new = _softmax(v)
            cur_val = _objective_surrogate(a_new)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a_new.copy()

        # end of this beta level
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing (Gaussian + mass‑transfer moves)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """SA with two move types: global Gaussian perturbation and targeted mass transfer."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # Linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        if rng.random() < 0.5:
            # Gaussian perturbation of all coordinates
            ns = a + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                attempts += 1
                continue
            ns /= s
        else:
            # Targeted mass‑transfer between two indices
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                attempts += 1
                continue
            # delta is a small random fraction of a[i]
            delta = sigma * rng.random() * a[i]
            ns = a.copy()
            ns[i] -= delta
            ns[j] += delta
            # numerical safety
            if ns[i] < 0.0:
                ns[i] = 0.0
            s = ns.sum()
            if s == 0.0:
                attempts += 1
                continue
            ns /= s

        val = _objective_surrogate(ns)
        delta_obj = val - cur_val
        if delta_obj < 0.0 or rng.random() < math.exp(-delta_obj / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimizing the provided
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and stores the best found sequence in ``height_sequence_2``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 3.0            # safety margin

    # ------------------------------------------------------------------
    # Determine problem size – re‑use previous length if reasonable.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_2")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim == 1 and prev_arr.size > 0 and np.sum(prev_arr) > 0:
                prev_len = prev_arr.shape[0]
                # keep the same length, but also try an up‑sampled version
                n = prev_len
                prev_norm = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 5000                     # default length – a bit larger than before
        print(f"\n[Info] No previous solution – using n={n}")
    else:
        print(f"\n[Info] Re‑using previous solution of length n={n}")

    # ------------------------------------------------------------------
    # 1. Broad seeding – generate a diverse pool of candidates.
    # ------------------------------------------------------------------
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_rev_triangular(n))
    for p in (0.5, 1.0, 2.0, 3.0, 4.5):
        seeds.append(_sym_power(n, p))

    # Beta U‑shape grid (both α,β in [0.2,0.8] produce U‑shape)
    alphas = np.linspace(0.2, 0.8, 7)
    betas = np.linspace(0.2, 0.8, 7)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_u_shape(n, a, b))

    # exponential decays (both sides)
    rates = np.logspace(-4, -0.2, 24)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))

    # mixtures with uniform
    for r in rates[::6]:
        exp_l = _exp_decay_left(n, r)
        exp_r = _exp_decay_right(n, r)
        for alpha in (0.2, 0.5, 0.8):
            seeds.append(_mix_shapes(_uniform(n), exp_l, alpha))
            seeds.append(_mix_shapes(_uniform(n), exp_r, alpha))

    # random Dirichlet draws – explore sparse and dense regimes
    for conc in (0.001, 0.005, 0.01, 0.05, 0.2):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # up‑sampled copy of previous best (if any)
    if prev_arr is not None:
        seeds.append(prev_norm)                         # original size
        if n != prev_len:
            up = _upsample(prev_norm, n)
            seeds.append(up)

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds.
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] top surrogate values:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best.
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start local refinement.
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a generous slice of the remaining budget
        remaining = deadline - time.time()
        # guarantee at least a few seconds even for many seeds
        sub_deadline = time.time() + max(4.0, remaining * 0.12 / max(1, (top_k - si)))

        # ---- (a) Exponentiated‑gradient (full active set) ----
        seq_eg, val_eg = _exp_gradient_opt_full(seed_vec.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=250_000,
                                                eta0=7.0)

        # ---- (b) Adam on smooth‑max ----
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_optimize(seq_eg,
                                                n,
                                                sub_deadline,
                                                beta_schedule=[0.5, 1.0, 5.0, 20.0, 80.0,
                                                               200.0, 500.0, 1_000.0,
                                                               2_000.0, 5_000.0,
                                                               20_000.0, 50_000.0,
                                                               100_000.0, 200_000.0,
                                                               500_000.0],
                                                iters_per_beta=15_000)
        else:
            seq_adam, val_adam = seq_eg, val_eg

        # ---- (c) Simulated annealing ----
        if time.time() < sub_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_adam,
                                                  n,
                                                  sub_deadline,
                                                  rng,
                                                  max_attempts=180_000,
                                                  start_sigma=0.0025,
                                                  T_start=0.012,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_adam, val_adam

        # pick the best among all refinements for this seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_adam, val_adam),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive Adam polishing (smooth‑max) ---")
        seq_final_adam, val_final_adam = _adam_optimize(
            global_best_seq,
            n,
            deadline,
            beta_schedule=[0.5, 1.0, 5.0, 20.0, 80.0, 200.0, 500.0,
                           1_000.0, 2_000.0, 5_000.0, 20_000.0,
                           50_000.0, 100_000.0, 200_000.0,
                           500_000.0, 1_000_000.0],
            iters_per_beta=30_000,
            lr=0.07)
        if val_final_adam < global_best_val:
            global_best_seq = seq_final_adam
            global_best_val = val_final_adam
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing (high‑budget) ---")
        seq_final_sa, val_final_sa = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            rng,
            max_attempts=400_000,
            start_sigma=9e-4,
            T_start=4e-3,
            T_end=1e-9)
        if val_final_sa < global_best_val:
            global_best_seq = seq_final_sa
            global_best_val = val_final_sa
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function (optional).
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[Warning] evaluate_sequence raised an exception:", exc)

    # store for possible future runs
    globals()["height_sequence_2"] = global_best_seq.tolist()
    return global_best_seq.tolist()
