"""
Propose a step‑function (non‑negative heights) that minimises
``evaluate_sequence``.  The search combines a dense grid of analytic
parameterisations (Beta‑shaped, exponential and power‑law families) with
a long projected mirror‑descent (exponentiated‑gradient) refinement.
The grid is searched first – the best few candidates are then polished
by sub‑gradient descent on the exact max‑convolution (the same objective
as ``evaluate_sequence`` when the sequence is ℓ₁‑normalised).  The
optimizer runs in the log‑softmax (probability) space and projects back
onto the simplex after each update.  Time is split dynamically between
candidates; the remaining budget is spent on a final long polish of the
global best.  The resulting sequence is stored in the global variable
``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, length = n+m‑1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate (the true objective when the sum of the vector is 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector (sum=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (i starts at 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """a_i ∝ i^{α‑1} (n‑i+1)^{β‑1}, i = 1..n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (sub‑gradient on the exact max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (a_i = exp(log_a_i) / Σ).
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    a = a / a.sum()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)  # stay in the interior of the simplex

    it = 0
    while time.time() < deadline:
        it += 1
        # --- compute convolution and active shift ---
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # --- sub‑gradient of the active entry (see comment in analysis) ---
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # --- step‑size schedule (decaying) ---
        eta = eta0 / math.sqrt(it)

        # --- mirror‑descent update (log‑space) ---
        log_a -= eta * g
        # keep logs bounded – prevents overflow/underflow
        log_a = np.clip(log_a, -30.0, 30.0)

        # --- back to probability simplex ---
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        # --- keep track of the best exact objective ---
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    """

    # --------------------------------------------------------------

                                             eta0=6.0)

        print(f"[Seed {idx+1}] best surrogate after EG = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4.  Final long polishing of the global best (use all remaining time)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 eta0=5.0)

            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # 5.  Evaluate with the provided function and store the solution.

              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
