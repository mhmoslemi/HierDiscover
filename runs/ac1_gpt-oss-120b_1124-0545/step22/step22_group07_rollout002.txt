"""
Propose a step‑function (non‑negative heights) that minimises ``evaluate_sequence``.

Algorithm (≈ 1000 s budget)
-----------------------------------------------------------------
1.  Build a diverse pool of analytic and random seeds:
    * Uniform, Hann, Hamming, Blackman, Gaussian, symmetric exponential,
      power‑law, Beta‑shaped, Dirichlet draws and the previous best.
2.  Keep the top few seeds according to the exact surrogate
    ``2·n·max(conv(a,a))`` (the objective with ``sum(a)=1``).
3.  For each top seed run a *smooth* Adam optimiser on a log‑sum‑exp
    approximation of the max–convolution.  The temperature β is
    increased gradually; each β runs many (≈60 k) inner iterations.
4.  On the same time slice polish the result with a long
    exponentiated‑gradient (mirror‑descent) run on the *exact* max.
5.  After all seeds, spend the remaining budget on a final long EG polish.
6.  Finish with a short permutation‑annealing tweak (swap moves only).
7.  Evaluate the final vector with ``evaluate_sequence`` and store it in the
    global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real inputs)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate objective (assumes sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a))   (a is already normalised)."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic / random families (always return a normalised probability vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    a /= a.sum()
    return a


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    a /= a.sum()
    return a


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    a /= a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a /= a.sum()
    return a


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    w /= w.sum()
    return w


def _gaussian_window(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    w /= w.sum()
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser on the *exact* max‑convolution.
# ----------------------------------------------------------------------
def _exp_gradient_opt(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 1_800_000,
    eta0: float = 5.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (a_i = exp(log_a_i) / Σ).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # initialise log‑probabilities (add tiny epsilon for safety)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution a*a and location of the current max
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (exact max)
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # learning‑rate schedule (AdaGrad‑like)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        # project back to the simplex via log‑softmax
        a = np.exp(log_a - np.max(log_a))
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(
    start_seq: np.ndarray,
    deadline: float,
    beta_schedule=None,
    iters_per_beta: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv) using Adam.
    Objective: 2·n·M_β where M_β = log‑sum‑exp_β(conv).
    """
    if beta_schedule is None:
        # start very smooth, end very sharp
        beta_schedule = [5, 20, 80, 320, 1280, 5120, 20_000, 80_000]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)          # stay in log‑softmax space

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                                 # base learning‑rate factor
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                # soft‑max weights

            # gradient w.r.t a  (2 * (rev_a * w) restricted to length n)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)           # length = 3n‑2
            grad_a = 2.0 * full[n - 1 : n - 1 + n]

            # chain rule to log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded
            log_a = np.clip(log_a, -30.0, 30.0)

            # track best exact objective
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(
    start_seq: np.ndarray,
    deadline: float,
    max_iters: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The algorithm builds a rich pool of analytic / random seeds,
    refines the best few with a smoothed Adam optimiser,
    follows up with a long exponentiated‑gradient polish,
    and finishes with a short permutation‑annealing tweak.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good speed / quality trade‑off.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Helper to normalise any vector to a probability distribution.
    # ------------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s <= 0.0:
            return _uniform(n)
        return v / s

    # ------------------------------------------------------------------
    # 1. Construct a modest but diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = [
        _uniform(n),
        _hann_window(n),
        _hamming_window(n),
        _blackman_window(n),
        _gaussian_window(n, sigma_factor=0.2),
        _sym_exp_decay(n, rate=0.0018),
        _power(n, p=0.8),
        _power(n, p=1.2),
    ]

    # a few Dirichlet draws with various concentration parameters
    for conc in (0.05, 0.2, 0.6, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Beta‑shaped families
    for _ in range(6):
        a = rng.uniform(0.3, 2.5)
        b = rng.uniform(0.3, 2.5)
        seeds.append(_beta_shape(n, a, b))

    # previous global solution, if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                seeds.append(_normalise(arr))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best few (top_k)
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Main refinement loop (Adam‑smooth + EG polish)
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a shrinking fraction of the remaining budget for this seed
        remaining = deadline - time.time()
        share = remaining * 0.15          # roughly 15 % of what is left
        sub_deadline = time.time() + share
        print(f"\n[Seed {idx+1}] allocated time ≈ {share:.1f}s")

        # ---- smoothed Adam optimisation --------------------------------
        seq_adam, val_adam = _adam_smooth_opt(
            seed_vec,
            sub_deadline,
            beta_schedule=[5, 20, 80, 320, 1280, 5120, 20_000, 80_000],
            iters_per_beta=60_000,
        )

        # ---- short EG run on the same slice (if time left) -------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(
                seq_adam,
                sub_deadline,
                max_iters=2_000_000,
                eta0=7.0,
            )
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick the better of the two refinements
        if val_eg < val_adam:
            local_seq, local_val = seq_eg, val_eg
        else:
            local_seq, local_val = seq_adam, val_adam

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (uses all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(
            global_best_seq,
            deadline,
            max_iters=5_000_000,
            eta0=6.0,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation tweak (a few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        perm_deadline = time.time() + 30.0
        seq_last, val_last = _perm_anneal(
            global_best_seq.copy(),
            perm_deadline,
            max_iters=80_000,
        )
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
