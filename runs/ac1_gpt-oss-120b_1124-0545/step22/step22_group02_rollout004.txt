def propose_candidate(seed=123, budget_s=1000):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0

    # Determine n based on budget: we attempt n=8000 if possible.
    n = 8000

    # Load previous best if exist and upsample
    prev = globals().get('height_sequence_1')
    prev_vec = None
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0:
            # upscale if needed
            if arr.size != n:
                x_old = np.arange(arr.size)
                x_new = np.linspace(0, arr.size-1, n)
                prev_vec = np.interp(x_new, x_old, arr)
                prev_vec = np.maximum(prev_vec, 0.0)
                s = prev_vec.sum()
                if s > 0:
                    prev_vec /= s
            else:
                prev_vec = arr.copy()
                prev_vec = prev_vec / prev_vec.sum()
    # Build seeds (use many analytic and random)
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    # add DPSS multiple
    for nw in [2.5,3.0,3.5,4.0,4.5,5.0,6.0]:
         seeds.append(_dpss_window(n, nw))
    # add gaussian
    for sigma_frac in np.linspace(0.05, 0.3, 6):
         seeds.append(_gaussian(n, sigma_frac))
    # Kaiser windows
    for beta in [2,4,6,8]:
         seeds.append(_kaiser_window(n,beta))
    # exponential decays
    for rate in np.logspace(-5,-0.5,12):
         seeds.append(_exp_decay(n, rate))
    # power law
    for p in np.linspace(0,4,9):
         seeds.append(_power(n,p))
    # beta shape
    for a in [0.5,1,2,5]:
         for b in [0.5,1,2,5]:
              seeds.append(_beta_shape(n,a,b))
    # random Dirichlet
    for conc in [0.001,0.01,0.05,0.1,0.5]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    # include reversed of each
    seeds_rev = [s[::-1] for s in seeds]
    seeds.extend(seeds_rev)
    # add previous upsampled if available
    if prev_vec is not None:
        seeds.append(prev_vec)
        # perturbed versions around prev
        for _ in range(3):
            pert = prev_vec + 0.001 * rng.normal(size=n)
            pert = np.maximum(pert,0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert/s)

    # Evaluate seeds quickly using surrogate objective.
    evals = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec,0)
        s = vec.sum()
        if s == 0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evals.append((vec,val))
    evals.sort(key=lambda x:x[1])
    top_k = min(6, len(evals))
    top_seeds = [e[0] for e in evals[:top_k]]
    # Print
    print(f"Top {top_k} seeds: {[v for _,v in evals[:top_k]]}")

    # Global best
    best_seq = top_seeds[0].copy()
    best_val = _objective_surrogate(best_seq)
    print("Initial best surrogate", best_val)

    # Multi-start refinement for each top seed
    for i, seed_vec in enumerate(top_seeds):
        # compute share of remaining time, e.g., 15% each
        remaining = deadline - time.time()
        if remaining <= 0: break
        share = remaining * (0.15 / max(1, (len(top_seeds)-i)))
        sub_deadline = time.time() + share

        # EG multi-active
        seq_eg, val_eg = _exp_gradient_opt_multi(seed_vec.copy(), n, sub_deadline,
                                                max_iters=2_000_000, eta0=6.0, tol=1e-9)

        # L-BFGS smooth, if still time
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n, sub_deadline,
                                               beta_schedule=None,
                                               max_iter_per_beta=2500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # flattening
        if time.time() < sub_deadline:
            seq_flat, val_flat = _flatten_gradient_opt(seq_lbfgs, n, sub_deadline,
                                                      max_iters=500_000, eta0=5.0)
        else:
            seq_flat, val_flat = seq_lbfgs, val_lbfgs

        # SA polish

Advanced step‑function optimiser – version 2

* Uses a larger problem size (n≈8000) to exploit extra degrees of freedom.
* Rich seed library (analytic windows, DPSS, random Dirichlet, up‑sampled
  previous best, and their reversals).
* Multi‑start mirror‑descent (exponentiated‑gradient) on the exact max‑
  convolution, with an active‑set sub‑gradient.
* Smooth log‑sum‑exp max optimisation with L‑BFGS‑B (“β‑continuation”).
* Quadratic flattening phase that directly minimises the variance of the
  whole convolution.
* A short simulated‑annealing polish.
* Optional linear‑programming step (SciPy ‘highs’) that solves
  max‑convolution constraints on the active set and mixes the result
  back into the iterate.
* All computation respects a 1000 s wall‑clock budget; the best sequence
  found is stored in the global variable ``height_sequence_1``.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (sum(seq) == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:

def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy i0 is unavailable."""
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)

    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))

    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:

    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:

    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b

def _dpss_window(n: int, nw: float) -> np.ndarray:

# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 500_000,
                            eta0: float = 6.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (multiplicative mirror descent) that minimises
    the exact max‑convolution.  The sub‑gradient is built from all indices
    whose value lies within ``tol`` relative distance of the current maximum.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (averaged over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # momentum (Adam‑style without bias correction)
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient descent that flattens the convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300_000,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the loss Σ_k (conv_k - avg)².
    The aim is to flatten the whole convolution.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full linear convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum == 1
        diff = conv - avg

        rev_a = a[::-1]
        # gradient of Σ diff² w.r.t a:
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # conversion to log‑space (mirror‑descent)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise a smooth approximation of the max‑convolution using L‑BFGS‑B.
    Contrary to the pure EG step, this works on a differentiable log‑sum‑exp
    surrogate controlled by a temperature parameter β.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1e3, 2e3, 5e3,
            2e4, 5e4, 1e5, 2e5, 5e5, 1e6,
        ]

    v = np.log(start_vec + 1e-30)          # log‑softmax parameters

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp                 # weights over conv entries
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)            # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for log‑softmax (mirror‑descent)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 100_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional linear‑programming improvement (active‑set LP)
# ----------------------------------------------------------------------
def _lp_improvement(f: np.ndarray,
                    M: float,
                    tol: float = 1e-5,
                    max_active: int = 500) -> np.ndarray | None:
    """
    Solve  max Σ b_j  subject to  f * b ≤ M  (element‑wise)  and  b ≥ 0.
    Only constraints whose current left‑hand side is within ``tol``·M
    of the bound are kept (active set).  Returns a normalised vector b
    (ℓ₁‑norm = 1) or None if the LP fails.
    """
    if _opt is None or _sp is None:
        return None
    n = len(f)
    conv = _conv_fft(f, f)
    active = np.where(conv >= M - tol * M)[0]

    # keep the number of active constraints reasonable
    if active.size > max_active:
        # keep the most violated (largest conv) ones
        idx_sorted = np.argsort(-conv[active])  # descending
        active = active[idx_sorted[:max_active]]

    rows = []
    cols = []
    data = []

    for row_idx, i in enumerate(active):
        # coefficients for b_j are f_{i-j} when 0 ≤ i‑j < n
        start_j = max(0, i - (n - 1))
        end_j = min(n - 1, i)
        js = np.arange(start_j, end_j + 1, dtype=np.int64)
        coeff = f[i - js]
        rows.extend([row_idx] * len(js))
        cols.extend(js.tolist())
        data.extend(coeff.tolist())

    A = _sp.csr_matrix((data, (rows, cols)), shape=(len(active), n), dtype=np.float64)
    c = -np.ones(n)                     # maximise Σ b → minimise -Σ b
    b_ub = np.full(len(active), M, dtype=np.float64)

    try:
        res = _opt.linprog(c,
                           A_ub=A,
                           b_ub=b_ub,
                           bounds=[(0, None)] * n,
                           method='highs',
                           options={'presolve': True, 'disp': False})
        if not res.success:
            return None
        b = res.x
        if b.sum() == 0.0:
            return None
        return b / b.sum()
    except Exception:   # pragma: no cover
        return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget,
    stores the best found sequence in the global variable
    ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin of a couple of seconds
    deadline = start_time + budget_s - 2.0

    # --------------------------------------------------------------
    # Problem size – larger than before to increase expressive power
    # --------------------------------------------------------------
    n = 8000                     # feel free to lower this if time is scarce

    # --------------------------------------------------------------
    # Load a previous solution (if present) and up‑sample it
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    prev_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                if arr.size != n:
                    # linear interpolation to the new length
                    old_x = np.arange(arr.size, dtype=np.float64)
                    new_x = np.linspace(0, arr.size - 1, n)
                    up = np.interp(new_x, old_x, arr)
                else:
                    up = arr.copy()
                up = np.maximum(up, 0.0)
                s = up.sum()
                if s > 0:
                    prev_vec = up / s
        except Exception:
            prev_vec = None

    # --------------------------------------------------------------
    # 1. Build a rich seed set
    # --------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    # deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    # Kaiser windows
    for beta in [0.0, 2.0, 4.0, 6.0, 8.0, 12.0]:
        seeds.append(_kaiser_window(n, beta))

    # Gaussian shapes
    for sigma_frac in np.linspace(0.05, 0.35, 7):
        seeds.append(_gaussian(n, sigma_frac))

    # DPSS windows – the proven work‑horse
    if _sigwin is not None:
        for nw in [2.5, 3.0, 3.5, 4.0, 4.5, 5.0, 5.5]:
            seeds.append(_dpss_window(n, nw))

    # Exponential decays (log‑spaced)
    for rate in np.logspace(-5, -0.2, 20):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails
    for p in np.linspace(0.0, 6.0, 13):
        seeds.append(_power(n, p))

    # Beta‑distribution shapes
    for alpha in [0.1, 0.5, 1.0, 2.0, 5.0, 8.0]:
        for beta in [0.1, 0.5, 1.0, 2.0, 5.0, 8.0]:
            seeds.append(_beta_shape(n, alpha, beta))

    # Logistic profiles
    for scale in [0.5, 1.0, 2.0, 5.0]:
        for mid in [n * 0.25, n * 0.5, n * 0.75]:
            seeds.append(_logistic_shape(n, scale, mid))

    # Random Dirichlet draws – wide range of concentrations
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of uniform with exponential
    exp_rates = np.logspace(-5, -0.5, 8)
    for r in exp_rates:
        exp_seq = _exp_decay(n, r)
        for alpha in (0.05, 0.15, 0.30, 0.45):
            seeds.append(_mix_shapes(_uniform(n), exp_seq, alpha))

    # Reversed copies of everything (break symmetry)
    seeds += [s[::-1].copy() for s in list(seeds)]

    # Include the up‑sampled previous best (if any) and perturbations
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(4):
            pert = prev_vec + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        total = vec.sum()
        if total == 0.0:
            continue
        vec = vec / total
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining_total = deadline - time.time()
        # allocate a modest fraction of the remaining budget per seed
        share = remaining_total * (0.12 / max(1, (top_k - si)))
        sub_deadline = time.time() + share

        # ---- a) exponentiated‑gradient (multi‑active) ----------
        seq_eg, val_eg = _exp_gradient_opt_multi(seed_vec.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=2_000_000,
                                                eta0=8.0,
                                                tol=1e-9)

        # ---- b) L‑BFGS smooth optimisation ----------
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                               n,
                                               sub_deadline,
                                               beta_schedule=None,
                                               max_iter_per_beta=2500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs = seq_eg
            val_lbfgs = val_eg

        # ---- c) flattening gradient descent ----------
        if time.time() < sub_deadline:
            seq_flat, val_flat = _flatten_gradient_opt(seq_lbfgs,
                                                       n,
                                                       sub_deadline,
                                                       max_iters=500_000,
                                                       eta0=5.0)
        else:
            seq_flat, val_flat = seq_lbfgs, val_lbfgs

        # ---- d) simulated annealing polish ----------
        hill_deadline = sub_deadline - 0.2
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_flat,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=70_000,
                                                  start_sigma=0.0015,
                                                  T_start=0.012,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_flat, val_flat

        # ---- e) optional LP‑based improvement (if SciPy available) ----
        if time.time() < sub_deadline and _opt is not None:
            M = _conv_fft(seq_sa, seq_sa).max()
            g_lp = _lp_improvement(seq_sa, M, tol=1e-5, max_active=300)
            if g_lp is not None:
                t_mix = 0.25
                seq_lp = (1.0 - t_mix) * seq_sa + t_mix * g_lp
                seq_lp = seq_lp / seq_lp.sum()
                val_lp = _objective_surrogate(seq_lp)
            else:
                seq_lp, val_lp = seq_sa, val_sa
        else:
            seq_lp, val_lp = seq_sa, val_sa

        # Pick the best local candidate
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_lbfgs, val_lbfgs),
            (seq_flat, val_flat),
            (seq_sa, val_sa),
            (seq_lp, val_lp),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si + 1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long polishing (heavy EG + flattening)
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final EG polishing (multi‑active) ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=5_000_000,
                                                       eta0=6.0,
                                                       tol=1e-9)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final flattening stage ---")
        final_seq2, final_val2 = _flatten_gradient_opt(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=1_200_000,
                                                       eta0=4.0)
        if final_val2 < global_best_val:
            global_best_seq = final_seq2
            global_best_val = final_val2
            print(f"[Final flatten] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                         n,
                                                         deadline,
                                                         rng,
                                                         max_attempts=120_000,
                                                         start_sigma=0.0008,
                                                         T_start=5e-3,
                                                         T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
