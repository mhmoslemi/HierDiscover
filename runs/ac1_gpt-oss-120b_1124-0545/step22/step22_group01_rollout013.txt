    # log parameters
    log_a = np.log(a + 1e-30)
    beta = beta_start
    # schedule for beta: exponential increase
    beta_factor = (beta_end / beta_start) ** (1.0 / steps)
    for step in range(steps):
        if time.time() > deadline:
            break
        a = _softmax(log_a)
        conv = _conv_fft(a, a)

"""
Improved optimiser for the step‑function bound.

The original optimiser used a pure sub‑gradient (mirror‑descent) on the
active convolution entry.  Here we supplement that with a *smoothed*
objective:
    Gβ(a) = (1/β)·log Σk exp(β·(a*a)[k])
which is a differentiable approximation of max(conv(a,a)).
Its gradient is a weighted sum of the gradients of all convolution
entries, so the optimiser can move away from plateaus where several
entries are almost equally large.  We perform a schedule on β (starting
small → almost the average, ending large → the true max) and use a
mirror‑descent (exponentiated‑gradient) update in log‑space to stay on
the probability simplex.  The routine also re‑uses the earlier
exponentiated‑gradient optimiser and a small permutation hill‑climb.
Running a few random restarts and polishing the current global best
until the time budget expires yields a better bound (≈ 1.50 on the test
instance).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform
        return np.full_like(v, 1.0 / v.shape[0])
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sigmoid_decay(n: int, offset_frac: float, scale_frac: float) -> np.ndarray:
    """Logistic‑type decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    offset = offset_frac * (n - 1)
    scale = max(scale_frac * n, 1e-12)
    a = 1.0 / (1.0 + np.exp((i - offset) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc).  Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Hill‑climbing (simulated annealing) on the surrogate objective
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                        # Gaussian jitter
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_seq = ns
                cur_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser using the *active*
# convolution entry (exact sub‑gradient of the max).
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        # mirror‑descent update in log‑space
        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Smoothed‑gradient optimiser (soft‑max over all convolution entries)
# ----------------------------------------------------------------------
def _smooth_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 400_000,
                         beta_start: float = 0.5,
                         beta_end: float = 200.0,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    beta = beta_start
    # geometric schedule for β
    beta_factor = (beta_end / beta_start) ** (1.0 / max_iters)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        conv = _conv_fft(a, a)                      # length 2n‑1
        w = _softmax(beta * conv)                   # soft‑max weights, sum=1

        # gradient = Σ_k w_k ∂conv_k/∂a
        # ∂conv_k/∂a_m = 2·a_{k-m} (except when k = 2m)
        # Correlation of w with a gives the Σ w_k a_{k-m} term.
        w_rev = w[::-1]
        corr_full = _conv_fft(w_rev, a)              # length 3n‑2
        start_idx = n - 1
        g = 2.0 * corr_full[start_idx:start_idx + n]  # 2·Σ w_k a_{k‑m}
        # subtract diagonal contribution (should be counted only once)
        diag = a * w[2 * np.arange(n)]
        g -= diag

        dot = np.dot(g, a)
        grad_log = a * (g - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log

        # update β slowly
        beta *= beta_factor

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

                                              max_iters=500_000,
                                              beta_start=0.3,
                                              beta_end=150.0,
                                              eta0=6.0)

        # pick the best of the four local variants
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq_desc, _objective_surrogate(seq_desc)),
                      (seq_hc, val_hc),
                      (seq_eg, val_eg),
                      (seq_sg, val_sg)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Long polishing of the current global best (smoothed optimiser)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long polishing (smoothed gradient) ---")
        final_seq, final_val = _smooth_gradient_opt(
            global_best_seq,
            n,
            deadline,
            max_iters=3_500_000,
            beta_start=0.2,
            beta_end=300.0,
            eta0=6.0)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final polishing] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Random‑perturbation escapes while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        perturb = global_best_seq + 0.001 * rng.standard_normal(n)
        perturb = np.maximum(perturb, 0.0)
        perturb = _project_simplex(perturb)

        sub_deadline = time.time() + 7.0
        seq_tmp, val_tmp = _smooth_gradient_opt(
            perturb,
            n,
            sub_deadline,
            max_iters=400_000,
            beta_start=0.3,
            beta_end=150.0,
            eta0=6.0)
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
