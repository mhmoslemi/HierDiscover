def _exp_gradient_opt_momentum(start_seq, n, deadline, max_iters=..., eta0=6.0, beta=0.9):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = conv.argmax()
        # gradient as before
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]
        # mirror-descent gradient in log space
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        # momentum
        m = beta * m + (1 - beta) * grad_log
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * m
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val
```

"""
Improved optimiser for the step‑function problem.

Main ideas
----------
* Use a larger search space (n = 8000) – more degrees of freedom reduce the
  theoretical lower bound.
* Create a very rich pool of analytical and random seeds (uniform, triangular,
  sine‑windows, Gaussians, exponentials, power‑laws, beta‑shapes and many
  Dirichlet draws with a wide range of concentrations).  The best few seeds
  are kept for intensive local optimisation.
* Local optimisation = mirror‑descent (exponentiated‑gradient) on the exact
  sub‑gradient of the active convolution entry, enhanced with a Nesterov‑type
  momentum term.  The update works entirely in log‑space, guaranteeing a
  feasible probability vector after each step.
* After a short refinement of every top seed, a long polishing run is performed
  on the current global best.  Random perturbations with a short optimisation
  “escape” are repeatedly tried while time permits.
* All heavy lifting (convolution) is performed with an FFT of length
  2ⁿ⁺¹, which is the same for any n ≤ 8000, keeping the per‑iteration cost
  low.  The routine respects the 1000 s budget (a small safety margin is
  reserved for the final call to evaluate_sequence).

The function ``propose_candidate`` follows the scheme above and returns a list
of non‑negative heights that should improve the known bound (≈ 1.5085) and aim
for the target 1.503.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Returns 2·n·max(conv(seq, seq)) for a vector whose sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc). Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Simplex projection (used for random perturbations)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection needs a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Mirror‑descent optimiser (exponentiated gradient) with momentum
# ----------------------------------------------------------------------
def _exp_gradient_opt_momentum(start_seq: np.ndarray,
                               n: int,
                               deadline: float,
                               max_iters: int = 2_000_000,
                               eta0: float = 6.0,
                               beta: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Log‑space is used (softmax) to keep the sequence on the simplex.
    Nesterov‑type momentum (parameter ``beta``) speeds up convergence.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)          # momentum accumulator

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # --- exact sub‑gradient of max(conv) w.r.t. a ---
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        g[idx_i] = 2.0 * a[rev_idx]          # off‑diagonal contributions

        # diagonal entry counted only once
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        # mirror‑descent gradient in log‑space
        dot = np.dot(g, a)                    # <g, a>
        grad_log = a * (g - dot)              # ∇_log objective

        # momentum update
        m = beta * m + (1.0 - beta) * grad_log

        # diminishing step size
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * m
        a = _softmax(log_a)                   # guarantees a ≥ 0, Σa = 1

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation hill‑climb (lightweight escape manoeuvre)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Randomly swaps two entries or adds a tiny Gaussian perturbation.
    Accepts worsening moves according to a simple annealing schedule.
    """
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                        # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds a wide pool of analytical and
    random seeds, refines the best few with momentum‑accelerated mirror descent,
    and finally spends the remaining budget polishing the current global best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation
    deadline = start_time + budget_s - 3.0

    # Problem size – 8000 gives a good trade‑off between expressivity
    # and per‑iteration cost (FFT length = 16384 for all n ≤ 8000).
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Load a previous solution if present – useful for continuation runs.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n:
                prev_arr = arr / arr.sum()
            else:
                # interpolate to the current size
                x_old = np.linspace(0, 1, arr.size)
                x_new = np.linspace(0, 1, n)
                prev_arr = np.interp(x_new, x_old, arr)
                prev_arr = prev_arr / prev_arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a large pool of analytical and random seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows – a dense sweep over gamma
    for gamma in np.linspace(0.2, 4.0, 48):
        seeds.append(_sine_window(n, gamma))

    # Gaussians – sweep sigma fraction
    for sigma_frac in np.linspace(0.05, 0.35, 22):
        seeds.append(_gaussian(n, sigma_frac))

    # shifted Gaussians – sweep centre and width
    for mu_frac in np.linspace(0.15, 0.85, 10):
        for sigma_frac in np.linspace(0.05, 0.30, 8):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # exponential decays – log‑space sweep
    for rate in np.logspace(-5, 0.5, 55):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law sweeps
    for p in np.linspace(0.0, 6.0, 28):
        seeds.append(_power(n, p))

    # beta‑shape grid
    alphas = np.geomspace(0.1, 12.0, 11)
    betas  = np.geomspace(0.1, 12.0, 11)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # random Dirichlet draws with a very wide range of concentration values
    conc_vals = [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(14):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(7):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2️⃣  Quick surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Refine each top seed
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # give roughly 10 % of the remaining budget to this seed,
        # but never less than 12 s (ensures a few hundred thousand iterations)
        share = max(12.0, remaining * 0.10 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # a) short permutation hill‑climb (helps escape a flat region)

                                     sub_deadline,
                                     rng,
                                     max_iters=120_000)

        variants = [seed_vec, np.sort(seed_vec)[::-1], seq_hc]
        local_best_seq = None
        local_best_val = np.inf

        for vi, var in enumerate(variants):
            if time.time() >= sub_deadline:
                break
            # split remaining time equally among remaining variants
            time_left = sub_deadline - time.time()
            var_share = time_left / (len(variants) - vi)
            var_deadline = time.time() + var_share

            seq_opt, val_opt = _exp_gradient_opt_momentum(var, n,
                                                          var_deadline,
                                                          max_iters=1_200_000,
                                                          eta0=7.0,
                                                          beta=0.92)

    # ------------------------------------------------------------------

                                                         n,
                                                         deadline,
                                                         max_iters=4_500_000,
                                                         eta0=7.5,
                                                         beta=0.94)
