    K = len(idx)
    logw = np.log(weights + 1e-30)
    best_weights = weights.copy()
    best_val = objective(weights)  # compute using conv_max * 2n
    best_seq = np.zeros(n)
    best_seq[idx] = best_weights

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute conv and max index
        max_val, max_idx = compute_conv_max_weights(weights, idx, n) # we need both max and index.
        # gradient
        index_to_w = {i: w for i, w in zip(idx, weights)}
        g = np.zeros(K)
        for p, i in enumerate(idx):
            rev = max_idx - i
            w_rev = index_to_w.get(rev, 0.0)
            g[p] = 2.0 * w_rev

        eta = eta0 / math.sqrt(it)
        logw = logw - eta * g
        # Softmax normalization across K entries
        logw = logw - np.max(logw)  # for numeric stability
        w = np.exp(logw)

        weights = w

        cur_val = 2 * n * max_val  # because sum=1
        if cur_val < best_val:
            best_val = cur_val
            best_weights = weights.copy()
    # Build full sequence
    seq = np.zeros(n)

    return seq, best_val

def compute_max_conv(idx, w, n):
    sum_indices = idx[:, None] + idx[None, :]  # shape (m,m)
    prod = w[:, None] * w[None, :]  # shape (m,m)
    # ordered contributions: off-diagonals each count twice
    # We'll first multiply by 2
    ordered = 2 * prod
    # But diagonal should only be once, not twice
    np.fill_diagonal(ordered, prod.diagonal())
    conv = np.zeros(2*n-1, dtype=np.float64)
    flat_sums = sum_indices.ravel()
    flat_values = ordered.ravel()
    np.add.at(conv, flat_sums, flat_values)
    max_val = conv.max()
    return max_val
```

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    n = 4000

    best_seq = None
    best_val = float('inf')

    # random search
    while time.time() < deadline - 30:  # keep time for polishing
        K = rng.integers(70, 301)  # random size between 70 and 300
        idx = rng.choice(n, size=K, replace=False)
        idx.sort()  # maybe sort for consistency
        w = np.full(K, 1.0 / K, dtype=np.float64)

        max_val = _max_conv_val(idx, w, n)  # compute max conv (max entry)
        obj = 2.0 * n * max_val

        if obj < best_val:
            best_val = obj
            seq = np.zeros(n, dtype=np.float64)
            seq[idx] = w
            best_seq = seq
            print(f"New best {obj:.6f} with K={K}")

    # polishing via exponentiated gradient using previous best as start
    if best_seq is not None and time.time() < deadline:
        print("\n--- Polishing best candidate ---")
        seq, val = _exp_gradient_opt(best_seq, n, deadline,
                                     max_iters=5_000_000,
                                     eta0=5.0)
        if val < best_val:
            best_val = val
            best_seq = seq
            print(f"Polished improved {best_val:.6f}")

    # final permutation hill climb
    if best_seq is not None and time.time() + 10.0 < deadline:
        tmp_seq, tmp_val = _perm_hill_climb(best_seq.copy(), n,
                                            deadline,

        if tmp_val < best_val:

    # evaluate true objective

        print(f"True evaluate_sequence = {true_val:.6f}")
    except Exception as exc:

    globals()["height_sequence_1"] = best_seq.tolist()

def _max_conv_val(indices: np.ndarray, weights: np.ndarray, n: int) -> float:
    # compute ordered contributions conv via O(K^2)
    K = len(indices)
    conv = np.zeros(2 * n - 1, dtype=np.float64)
    # self terms
    for i in range(K):
        idx_i = indices[i]
        wi = weights[i]
        conv[2 * idx_i] += wi * wi
    # off-diagonal pairs
    for i in range(K):
        idx_i = indices[i]
        wi = weights[i]
        for j in range(i + 1, K):
            idx_j = indices[j]
            wj = weights[j]
            v = 2.0 * wi * wj
            conv[idx_i + idx_j] += v
    return conv.max()
```

def build_low_collision_set(n, target_size, max_per_sum=3, rng):
    counts = np.zeros(2*n-1, dtype=np.int16)
    S = []
    attempts = 0
    while len(S) < target_size and attempts < target_size * 10:
        i = rng.integers(0, n)
        # compute sums with existing S
        new_counts = counts.copy()
        conflict = False
        # self sum
        s = 2*i
        if new_counts[s] + 1 > max_per_sum:
            conflict = True
        else:
            new_counts[s] += 1
        for j in S:
            s = i + j
            # ordered count increments 2 for unordered pair, but we store unordered
            if new_counts[s] + 1 > max_per_sum:
                conflict = True
                break
            new_counts[s] += 1
        if not conflict:
            # accept
            S.append(i)
            counts = new_counts
        attempts += 1
    return np.array(sorted(S), dtype=int)
```

    order = rng.permutation(n)
    counts = np.zeros(2*n-1, dtype=np.int16)

def _max_conv_val(idx, w, n):
    # compute conv via outer addition; use np.add.at for aggregation
    idx = np.asarray(idx, dtype=np.int64)
    w = np.asarray(w, dtype=np.float64)
    sum_idx = idx[:, None] + idx[None, :]
    prod = w[:, None] * w[None, :]
    # ordered contributions
    ordered = 2.0 * prod
    np.fill_diagonal(ordered, prod.diagonal())
    conv = np.zeros(2 * n - 1, dtype=np.float64)
    np.add.at(conv, sum_idx.ravel(), ordered.ravel())
    return conv.max()
```

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5

    n = 4000

    best_seq = None
    best_val = float('inf')
    best_set = None

    # Phase 1: construct low-collision sparse sets
    # allocate budget: 30% of time
    phase1_deadline = start_time + int(0.3 * budget_s)
    limit = 2  # try limit 2 first
    attempts = 0

    while time.time() < phase1_deadline:
        attempts += 1
        # randomly shuffle indices
        idx_order = rng.permutation(n)
        counts = np.zeros(2*n - 1, dtype=np.int16)
        S = []
        for i in idx_order:
            # check if we can add i
            s_self = 2 * i
            if counts[s_self] + 1 > limit:
                continue
            conflict = False
            for j in S:
                s = i + j
                if counts[s] + 1 > limit:
                    conflict = True
                    break
            if conflict:
                continue
            # accept i
            S.append(i)
            counts[s_self] += 1
            for j in S[:-1]:
                s = i + j
                counts[s] += 1
        m = len(S)
        if m == 0:
            continue
        w = np.full(m, 1.0 / m, dtype=np.float64)
        max_conv = _max_conv_val(np.array(S, dtype=np.int64), w, n)
        obj = 2.0 * n * max_conv
        if obj < best_val:
            best_val = obj
            best_seq = np.zeros(n, dtype=np.float64)
            best_seq[S] = w
            best_set = np.array(S, dtype=np.int64)
            print(f"[Attempt {attempts}] new best obj {obj:.6f} with m={m} limit={limit}")
        # optionally break if extremely good
        if best_val < 1.45:
            break

    # Phase 2: polish via exponentiated gradient
    if best_seq is not None and time.time() < deadline:
        print("\n--- Polish candidate with EG ---")
        cur_seq, cur_val = _exp_gradient_opt(best_seq, n, deadline,
                                             max_iters=5_000_000,
                                             eta0=10.0)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = cur_seq
            print(f"[Polish] improved obj {cur_val:.6f}")
    else:
        # fallback: if no best_seq found, use uniform or previous
        if best_seq is None:
            prev = globals().get("height_sequence_1")
            if prev is not None:
                best_seq = np.asarray(prev, dtype=np.float64)
                best_val = evaluate_sequence(best_seq.tolist())
            else:
                # uniform distribution for fallback (objective 2)
                best_seq = np.full(n, 1.0 / n)
                best_val = evaluate_sequence(best_seq.tolist())

    # Phase 3: final permutation tweak (optional)
    if time.time() + 10.0 < deadline:
        short_deadline = time.time() + 10.0
        seq_last, val_last = _perm_hill_climb(best_seq.copy(), n,
                                              short_deadline,
                                              rng,
                                              max_iters=80_000)
        if val_last < best_val:
            best_seq = seq_last
            best_val = val_last
            print(f"[Final permutation] improved obj {best_val:.6f}")

    # Evaluate true objective
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\nFinal true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:
        print("Error evaluating sequence:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
A fast constructive / local‑search optimiser for the max–convolution problem.

Goal
----
Find a non‑negative step‑function `a` (length n=4000) that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)
The objective is homogeneous, so we may restrict to sum(a)=1.
The best known value is ≈1.5085; the target is 1.503.

Idea
----
*   Build a *low‑collision* support set S ⊂ {0,…,n‑1}.  
    For each unordered pair (i,j) with i≤j the sum s=i+j is recorded.
    We enforce that no sum is used more than `limit` times (limit=2 or 3).
    This guarantees that each convolution entry receives at most a few
    products, keeping the maximum small.
*   Use a uniform weight 1/|S| on that support – this already yields a
    good upper bound because the contribution of any entry is at most
    (2·limit)/|S|².
*   Run a short exponentiated‑gradient (mirror‑descent) refinement
    starting from that sparse vector; this step can improve the bound
    further while staying within the time budget.
*   Optionally finish with a tiny permutation hill‑climb.

The whole routine runs under the supplied 1000 s budget, stores the
best sequence in the global variable ``height_sequence_1`` and returns it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: full convolution via FFT (used in the EG polishing stage)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact max‑convolution for a *sparse* vector (O(m²) with m=|S|).
# ----------------------------------------------------------------------
def _max_conv_val(idx: np.ndarray, w: np.ndarray, n: int) -> float:
    """
    Compute max(conv(a,a)) for a vector a that is non‑zero only at positions
    ``idx`` with corresponding weights ``w`` (sum(w)=1).  The convolution is
    evaluated exactly using the ordered pair formulation.
    """
    idx = np.asarray(idx, dtype=np.int64)
    w = np.asarray(w, dtype=np.float64)
    # pairwise sums (matrix) and ordered products (matrix)
    sum_idx = idx[:, None] + idx[None, :]               # shape (m,m)
    prod = w[:, None] * w[None, :]                     # shape (m,m)
    ordered = 2.0 * prod
    np.fill_diagonal(ordered, prod.diagonal())        # diagonal entries count once
    conv = np.zeros(2 * n - 1, dtype=np.float64)
    np.add.at(conv, sum_idx.ravel(), ordered.ravel())
    return float(conv.max())


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (mirror‑descent) on the exact objective.
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 5_000_000,
                     eta0: float = 10.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Returns the best sequence seen and the corresponding objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * np.max(_conv_fft(a, a))

    log_a = np.log(a + 1e-30)          # work in the log‑simplex

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active convolution entry (ordered version)
        g = np.zeros_like(a)
        rev_idx = max_idx - np.arange(n)
        mask = (rev_idx >= 0) & (rev_idx < n)
        g[mask] = 2.0 * a[rev_idx[mask]]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = 2.0 * n * np.max(_conv_fft(a, a))
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation / annealing hill‑climb (optional final tweak)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps positions or adds small Gaussian noise.
    Accepts worsening moves with a simple annealing schedule.
    """
    cur_seq = start_seq.copy()
    cur_val = 2.0 * n * np.max(_conv_fft(cur_seq, cur_seq))
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                # swap two entries
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                   # small Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = 2.0 * n * np.max(_conv_fft(ns, ns))

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            if T > 0.0:
                prob = math.exp(-(val - cur_val) / T)
                if rng.random() < prob:
                    cur_seq = ns
                    cur_val = val

    return best_seq, best_val

# Main entry point – a constructive + polishing optimiser
# ----------------------------------------------------------------------

    """

    deadline = start_time + budget_s - 5.0      # a small safety margin

    n = 4000

    best_seq = None
    best_val = float('inf')
    best_set = None

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    phase1_deadline = start_time + int(0.30 * budget_s)   # ~30 % of budget
    attempts = 0
    # we try two limits; 2 gives tighter collision bound, 3 is more permissive
    limits_to_try = [2, 3]

    while time.time() < phase1_deadline:
        attempts += 1
        limit = int(rng.choice(limits_to_try))
        # random order of indices
        order = rng.permutation(n)
        # counts of unordered pairs for each possible sum (0…2n‑2)
        counts = np.zeros(2 * n - 1, dtype=np.int16)
        S = []                     # the support set
        for i in order:
