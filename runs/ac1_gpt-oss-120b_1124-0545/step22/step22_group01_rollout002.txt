"""
Propose a non‑negative step‑function (vector of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a))² .

The algorithm is a hybrid of:

1️⃣  Analytic seed generation (uniform, triangular, sine‑windows,
    Gaussians, exponentials, power‑laws, beta‑shapes, random Dirichlet).
    All seeds are ℓ₁‑normalised (sum = 1).

2️⃣  Quick surrogate evaluation (exact for sum = 1) to keep the best few seeds.

3️⃣  For each top seed a smooth‑max surrogate is minimised with an
    Adam‑type mirror‑descent on the **softmax** parametrisation.
    The smooth‑max uses a temperature β and we employ a continuation
    schedule β = 10 → 30 → 100 → 300 → 1000.  The gradient of the
    surrogate is computed analytically using two FFT‑convolutions per
    iteration (O(n log n)).

4️⃣  The best sequence from the above is polished with a longer
    Adam run (β = 1000) and occasional random perturbations to escape
    shallow minima.

All heavy work is done with FFT‑based convolutions; the routine respects
the 1000 s budget, prints progress, stores the best solution in the
global variable ``height_sequence_1`` for possible continuation runs and
finally returns the best list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector with sum = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet(conc,…,conc) vector (strictly positive)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    y_init: np.ndarray,
    n: int,
    deadline: float,
    beta_schedule: list[float],
    max_iters_per_beta: int = 30_000,
    lr: float = 0.8,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the smooth‑max surrogate.
    The variable is `y` such that a = softmax(y) (automatically satisfies
    a ≥ 0 and sum(a)=1).  A continuation over increasing β improves
    approximation of the true max‑conv objective.
    Returns the best sequence (in simplex) seen and its exact surrogate value.
    """
    y = y_init.copy()
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    t = 0
    best_seq = _softmax(y)
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        # limited inner loop – stop early if time runs out
        while time.time() < deadline and t < max_iters_per_beta:
            t += 1

            # a = softmax(y)
            a = _softmax(y)

            # ----- compute convolution a*a (linear) -----
            conv_a = _conv_fft(a, a)            # length 2n‑1
            max_c = np.max(conv_a)

            # ----- smooth‑max weights -----
            # p_k ∝ exp(β (conv_a[k]‑max_c)), stable
            exp_term = np.exp(beta * (conv_a - max_c))
            p = exp_term / exp_term.sum()       # still length 2n‑1, sums to 1

            # ----- compute g = (p * a)_i = Σ_k p_k a_{k‑i}  -----
            # g_i = Σ_{j=0}^{n‑1} p_{i+j} a_j  → convolution of p with a_rev
            a_rev = a[::-1]
            conv_pa = _conv_fft(p, a_rev)       # length 3n‑2
            start = n - 1
            g = conv_pa[start : start + n]      # size n

            # ----- gradient w.r.t. a -----
            grad_a = g + a                      # ∂f/∂a_i

            # ----- gradient w.r.t. y (softmax Jacobian) -----
            inner = np.dot(a, grad_a)           # scalar ⟨a,∇_a⟩
            grad_y = a * (grad_a - inner)       # shape (n,)

            # ----- Adam update -----
            m = beta1 * m + (1 - beta1) * grad_y
            v = beta2 * v + (1 - beta2) * (grad_y ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr / (np.sqrt(v_hat) + eps)
            y = y - step * m_hat

            # ----- occasional exact evaluation (every 1000 iters) -----
            if t % 1000 == 0:
                a_now = _softmax(y)
                val = _objective_surrogate(a_now)
                if val < best_val:
                    best_val = val
                    best_seq = a_now.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function of length n that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin for the final evaluation
    deadline = start_time + budget_s - 2.0

    # Problem dimension – a good trade‑off between expressivity and speed
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (helps continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a diverse pool of analytical & random seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine window sweep (moderate size)
    for gamma in np.linspace(0.3, 3.5, 30):
        seeds.append(_sine_window(n, gamma))

    # Gaussian sweep
    for sigma_frac in np.linspace(0.07, 0.30, 18):
        seeds.append(_gaussian(n, sigma_frac))

    # shifted Gaussians
    for mu_frac in np.linspace(0.20, 0.80, 9):
        for sigma_frac in np.linspace(0.08, 0.25, 6):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # exponential decays (log‑space)
    for rate in np.logspace(-4.5, 0.4, 20):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law sweep
    for p in np.linspace(0.0, 5.0, 12):
        seeds.append(_power(n, p))

    # beta‑shape grid (coarse)
    alphas = np.geomspace(0.2, 8.0, 8)
    betas_ = np.geomspace(0.2, 8.0, 8)
    for a in alphas:
        for b_ in betas_:
            seeds.append(_beta_shape(n, a, b_))

    # random Dirichlet draws
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(8):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(5):
            noisy = prev_arr + 0.0025 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s                         # ℓ₁‑normalise
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed with Adam on a smooth‑max surrogate
    # ------------------------------------------------------------------
    beta_schedule = [10.0, 30.0, 100.0, 300.0, 1000.0]

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate roughly equal time to each remaining seed (minimum 12 s)
        share = max(12.0, remaining * 0.9 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] allocation ≈ {share:.1f}s")

        # initialise y from the seed (use log‑softmax parametrisation)
        y0 = np.log(seed_vec + 1e-30)

        seq_opt, val_opt = _adam_optimize(
            y0,
            n,
            sub_deadline,
            beta_schedule,
            max_iters_per_beta=40_000,
            lr=0.8,
        )

        print(f"[Seed {si+1}] best surrogate after opt = {val_opt:.6f}")

        if val_opt < global_best_val:
            global_best_seq = seq_opt
            global_best_val = val_opt
            print(f"  *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing on global best ---")
        y_start = np.log(global_best_seq + 1e-30)
        final_seq, final_val = _adam_optimize(
            y_start,
            n,
            deadline,
            beta_schedule,
            max_iters_per_beta=120_000,
            lr=0.6,
        )
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Polish] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Random‑perturbation escape loop (while time permits)
    # ------------------------------------------------------------------
    while time.time() < deadline - 4.0:   # reserve a few seconds for final eval
        # generate a mildly perturbed copy
        pert = global_best_seq + 0.001 * rng.standard_normal(n)
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum()

        y_pert = np.log(pert + 1e-30)
        sub_deadline = time.time() + 6.0
        seq_tmp, val_tmp = _adam_optimize(
            y_pert,
            n,
            sub_deadline,
            [300.0, 1000.0],
            max_iters_per_beta=20_000,
            lr=0.5,
        )
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
