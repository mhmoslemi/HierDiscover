def _upsample(seq, new_n):
    old_n = len(seq)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0)
    new_seq /= new_seq.sum()
    return new_seq

"""
Multi‑resolution optimisation for the convolution‑max bound.

The algorithm proceeds in three stages:

1. **Coarse search** – on a modest size (n≈1024) we generate a large pool
   of candidates (analytic families + many random Dirichlet draws).  All are
   ℓ₁‑normalised and evaluated with the fast surrogate
   ``2·n·max(conv(a,a))``.  The best few are kept.

2. **Local refinement** – each top candidate is improved by
   exponentiated‑gradient (mirror‑descent on the exact sub‑gradient of
   ``max(conv)``) and a short L‑BFGS pass on the smooth‑max surrogate.
   The best refined small‑size vector is up‑sampled to a larger size
   (n≈4096) while preserving the shape.

3. **Large‑scale polishing** – on the up‑sampled vector we run another
   exponentiated‑gradient phase, a more aggressive L‑BFGS continuation,
   and finally a high‑budget simulated‑annealing sweep.  The final
   sequence is returned and stored in ``height_sequence_optim`` for future
   runs.

All steps respect the global 1000 s budget; time is split dynamically
between stages.  The routine calls the provided ``evaluate_sequence`` only
once, at the very end.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ------------------------------------------------------------
# Basic numeric utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ------------------------------------------------------------
# Surrogate objective (exact max for ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a, a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set (averaged)
# ------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Return the (averaged) sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    count = active.sum()
    if count == 0:
        return np.zeros_like(a)

    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)  # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / count
    return grad


# ------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution,
    using the full active set.
    Returns the best sequence and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        grad_a = _subgradient_max_conv(a)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        return start_vec                     # SciPy not available

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0, 500_000.0,
            1_000_000.0, 2_000_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)               # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ------------------------------------------------------------
# Upsampling helper – linear interpolation onto a larger grid
# ------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate `seq` onto `new_n` points and renormalise."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    up = np.interp(new_x, old_x, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        return _uniform(new_n)
    return up / s

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the time budget, stores the
    best sequence in the global variable ``height_sequence_optim`` and returns it.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # tiny safety margin

    # ----------------------------------------------------------------
    # Stage 1 – coarse search on a modest size
    # ----------------------------------------------------------------
    n_small = 1024
    seeds_small = []

    # deterministic analytic families
    seeds_small.append(_uniform(n_small))
    seeds_small.append(_triangular(n_small))
    seeds_small.append(_gaussian(n_small, sigma_frac=0.10))
    seeds_small.append(_gaussian(n_small, sigma_frac=0.20))
    seeds_small.append(_exp_decay_left(n_small, rate=1e-3))
    seeds_small.append(_exp_decay_right(n_small, rate=1e-3))
    seeds_small.append(_power(n_small, p=0.5))

    # beta‑skewed grid
    alphas = np.linspace(0.5, 5.0, 9)
    betas = np.linspace(0.5, 5.0, 9)
    for a in alphas:
        for b in betas:
            seeds_small.append(_beta_shape(n_small, a, b))

    # exponential decays on a log‑grid
    rates = np.logspace(-4, -0.3, 13)
    for r in rates:
        seeds_small.append(_exp_decay_left(n_small, r))
        seeds_small.append(_exp_decay_right(n_small, r))

    # power‑law sweep
    for p in np.linspace(0.0, 4.0, 9):
        seeds_small.append(_power(n_small, p))

    # many random Dirichlet draws (different concentrations)
    for _ in range(2500):
        conc = 10 ** rng.uniform(-4, 0)   # between 1e‑4 and 1
        seeds_small.append(_random_dirichlet(n_small, conc, rng))

    # reuse previous solution (if any) – down‑sample / up‑sample to n_small
    prev = globals().get("height_sequence_optim")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape[0] > 0 and arr.sum() > 0.0:
                if arr.shape[0] != n_small:
                    arr = _upsample(arr, n_small)
                seeds_small.append(arr / arr.sum())
        except Exception:
            pass

    # ----------------------------------------------------------------
    # Evaluate all seeds (surrogate only)
    # ----------------------------------------------------------------
    evaluated_small = []
    for vec in seeds_small:
        if time.time() > deadline:
            break
        v = np.maximum(vec, 0.0)
        s = v.sum()
        if s == 0.0:
            continue
        v = v / s
        val = _objective_surrogate(v)
        evaluated_small.append((v, val))

    evaluated_small.sort(key=lambda x: x[1])
    top_k_small = min(10, len(evaluated_small))
    top_small = [vec for vec, _ in evaluated_small[:top_k_small]]

    print("\n[Coarse] top surrogate values (small n):")
    for i, (_, v) in enumerate(evaluated_small[:top_k_small]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ----------------------------------------------------------------
    # Stage 2 – refinement on small size
    # ----------------------------------------------------------------
    best_small_seq = None
    best_small_val = float('inf')
    for idx, seed_vec in enumerate(top_small):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget
        remaining = deadline - time.time()
        # give each seed roughly equal remaining time, but cap per‑seed
        per_seed = min(remaining * 0.5 / max(1, top_k_small - idx),
                       30.0)   # at most 30 s per seed
        sub_deadline = time.time() + per_seed

        # (a) Exponentiated‑gradient
        seq_eg, val_eg = _exp_gradient_opt_full(seed_vec.copy(),
                                                n_small,
                                                sub_deadline,
                                                max_iters=200_000,
                                                eta0=8.0)

        # (b) L‑BFGS smoothing (short continuation)
        if time.time() < sub_deadline:
            seq_lb = _lbfgs_smooth_optimize(seq_eg,
                                            n_small,
                                            sub_deadline,
                                            beta_schedule=[0.5, 1.0, 5.0, 20.0,
                                                           80.0, 200.0, 500.0,
                                                           1_000.0, 2_000.0,
                                                           5_000.0, 20_000.0])
            val_lb = _objective_surrogate(seq_lb)
        else:
            seq_lb, val_lb = seq_eg, val_eg

        # pick best of the two refinements
        if val_eg < val_lb:
            cand_seq, cand_val = seq_eg, val_eg
        else:
            cand_seq, cand_val = seq_lb, val_lb

        print(f"[Refine {idx+1}/{top_k_small}] best surrogate = {cand_val:.6f}")

        if cand_val < best_small_val:
            best_small_val = cand_val
            best_small_seq = cand_seq

    # fallback in case everything timed‑out
    if best_small_seq is None and evaluated_small:
        best_small_seq = evaluated_small[0][0]
        best_small_val = evaluated_small[0][1]

    # ----------------------------------------------------------------
    # Stage 3 – up‑sample to a larger size and polish
    # ----------------------------------------------------------------
    n_large = 4096
    seq_large = _upsample(best_small_seq, n_large)

    # (a) Exponentiated‑gradient on the large problem
    if time.time() < deadline:
        sub_deadline = time.time() + (deadline - time.time()) * 0.45
        seq_eg_large, val_eg_large = _exp_gradient_opt_full(seq_large.copy(),
                                                            n_large,
                                                            sub_deadline,
                                                            max_iters=250_000,
                                                            eta0=6.0)
        if val_eg_large < _objective_surrogate(seq_large):
            best_seq = seq_eg_large
            best_val = val_eg_large
        else:
            best_seq = seq_large
            best_val = _objective_surrogate(seq_large)
    else:
        best_seq = seq_large
        best_val = _objective_surrogate(seq_large)

    # (b) L‑BFGS continuation on the large scale
    if time.time() < deadline:
        seq_lb_large = _lbfgs_smooth_optimize(best_seq,
                                              n_large,
                                              deadline,
                                              beta_schedule=[0.5, 1.0, 5.0, 20.0, 80.0,
                                                             200.0, 500.0, 1_000.0,
                                                             2_000.0, 5_000.0, 20_000.0,
                                                             50_000.0, 100_000.0, 200_000.0,
                                                             500_000.0, 1_000_000.0])
        val_lb_large = _objective_surrogate(seq_lb_large)
        if val_lb_large < best_val:
            best_seq = seq_lb_large
            best_val = val_lb_large
            print(f"[L‑BFGS large] improved to {best_val:.6f}")

    # (c) Final simulated‑annealing polishing
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(best_seq,
                                              n_large,
                                              deadline,
                                              rng,
                                              max_attempts=300_000,
                                              start_sigma=9e-4,
                                              T_start=4e-3,
                                              T_end=1e-9)
        if val_sa < best_val:
            best_seq = seq_sa
            best_val = val_sa
            print(f"[Final SA] improved to {best_val:.6f}")

    # ----------------------------------------------------------------
    # Official evaluation via the provided function
    # ----------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for later runs
    globals()["height_sequence_optim"] = best_seq.tolist()
    return best_seq.tolist()
