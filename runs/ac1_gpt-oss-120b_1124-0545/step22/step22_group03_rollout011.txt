def _smooth_exp_gradient(start_seq, deadline, eta0=5.0, beta0=0.5, beta_max=30.0, max_iters=int(1e7)):
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    N = _next_pow2(3*n - 2) # size for FFT

    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break

        # compute conv a*a
        F_a = np.fft.rfft(a, n=N)
        L = 2*n - 1
        conv = np.fft.irfft(F_a * F_a, n=N)[:L]

        # schedule beta (increase linearly)
        beta = beta0 + (beta_max - beta0) * min(1.0, it / max_iters)

        # compute softmax probabilities across shifts
        # we want p = softmax(beta * conv). Use stable.
        maxc = np.max(beta * conv)
        exps = np.exp(beta * conv - maxc)
        p = exps / exps.sum()

        # gradient term = 4 * n * term1 where term1 = conv(a, p_rev) slice reversed

"""
Propose a step‑function (non‑negative height vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv denotes the linear convolution.  The objective is scale‑invariant,
so we restrict the search to probability vectors (sum = 1).

Algorithm
---------
1. Initialise from a stored solution if present, otherwise a uniform
   probability vector of length 4000.

2. **Phase 1 – Smoothed exponentiated‑gradient descent**  
   Optimise a smooth approximation of the max‑convolution:
        fβ(a) = 2·n·(1/β)·log Σ_k exp(β·(a*a)_k) .
   β grows linearly from a small value to a larger one, so the early
   iterations see a smooth landscape and later iterations focus on the true
   maximum.  The optimisation is performed in log‑probability space; after
   each gradient step a soft‑max projection restores the simplex constraint.

3. **Phase 2 – Hard sub‑gradient descent**  
   Continue optimisation using the classic single‑active‑shift sub‑gradient
   (the worst convolution shift) as in the reference implementation.

4. **Phase 3 – Simulated‑annealing fine‑tuning**  
   Tiny random mass transfers between coordinates accept worsening moves
   according to an exponential cooling schedule.

All phases respect the overall 1000 s budget.  The best vector found is
stored in the global name ``height_sequence_1`` and returned as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Numeric utilities (FFT‑based convolution, stable softmax, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(x: np.ndarray, y: np.ndarray, N: int) -> np.ndarray:
    """Full linear convolution of ``x`` and ``y`` using a fixed FFT size ``N``."""
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)
    return conv[: x.size + y.size - 1]


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return e / s


def _surrogate_obj(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    L = 2 * n - 1
    N = _next_pow2(L)
    conv = _conv_full_fft(a, a, N)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Phase 1 – Smoothed exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             deadline: float,
                             eta0: float = 1.0,
                             beta_start: float = 0.5,
                             beta_end: float = 30.0,
                             max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the smooth log‑sum‑exp
    approximation of the max‑convolution.
    Returns the best probability vector encountered and its surrogate value.
    """
    n = start_seq.size
    a = start_seq.copy()
    # initialise log‑weights safely
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    L = 2 * n - 1                     # length of a*a convolution
    N = _next_pow2(3 * n - 2)         # size sufficient for both convolutions

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current probability vector
        a = _softmax(log_a)

        # FFT of a (re‑used for both convolutions)
        a_fft = np.fft.rfft(a, n=N)

        # ---- (a * a) -------------------------------------------------
        conv = np.fft.irfft(a_fft * a_fft, n=N)[:L]

        # schedule for β (smoothness parameter)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / max_iters)

        # soft‑max weights over the shifts
        exp_arg = beta * conv
        max_exp = np.max(exp_arg)
        w = np.exp(exp_arg - max_exp)
        p = w / w.sum()                     # probabilities ∈ Δ_{2n‑1}

        # ---- gradient of the smoothed objective ----------------------
        p_rev = p[::-1]                     # reversed weights
        p_rev_fft = np.fft.rfft(p_rev, n=N)

        # convolution a * p_rev (full length = n + L - 1 = 3n - 2)
        conv_grad_full = np.fft.irfft(a_fft * p_rev_fft, n=N)[: n + L - 1]

        # term1_i = Σ_j a_j p_{i+j}  (slice and reverse)
        term1 = conv_grad_full[n - 1 : 2 * n - 1][::-1]

        # gradient of 2·n·log‑sum‑exp = 4·n·term1
        grad = 4.0 * n * term1

        # learning‑rate schedule (decays as 1/√t)
        eta = eta0 / math.sqrt(it)

        # mirror‑descent step in log‑space
        log_a -= eta * grad

        # keep the best solution seen so far (evaluated on the exact max)
        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 2 – Classical single‑active‑shift exponentiated‑gradient descent
# ----------------------------------------------------------------------
def _exp_gradient_opt_long(start_seq: np.ndarray,
                           deadline: float,
                           eta0: float = 5.0,
                           max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *exact* max‑convolution (single active shift).
    Returns best sequence (ℓ₁‑normalised) and its surrogate value.
    """
    n = start_seq.size
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    L = 2 * n - 1
    N = _next_pow2(L)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(log_a)

        # worst shift (index of maximal convolution entry)
        a_fft = np.fft.rfft(a, n=N)
        conv = np.fft.irfft(a_fft * a_fft, n=N)[:L]
        k = int(conv.argmax())

        # gradient of the active shift (see derivation in analysis)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        g = np.zeros_like(a)
        g[idx] = a[k - idx]          # vectorised contribution

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g

        cur_val = _surrogate_obj(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 3 – Simulated‑annealing fine‑tuning
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 600_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.003) -> tuple[np.ndarray, float]:
    """
    Tiny mass transfers between random indices with a linear cooling schedule.
    Returns the best sequence observed.
    """
    n = start_seq.size
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
