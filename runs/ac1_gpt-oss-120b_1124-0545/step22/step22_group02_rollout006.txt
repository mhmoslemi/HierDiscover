def propose_candidate(seed=123, budget_s=1000):
    import numpy, time, math, etc.
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2

    n = 4000  # Might try 5000? Actually bigger n increases conv size; but we may stay with 4000.

    # Get previous best if any
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,):
            prev_arr = arr.copy()
        # else ignore.

    # Build seed sets
    seeds = []
    # Add uniform, exponential families, etc.
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_exp_decay(n, 0.00001)) # extremely slow decay.
    # we can add more dynamic seeds.
    # Random Dirichlet seeds
    for conc in [0.001,0.005,0.01,0.02,0.05,0.1,0.2,0.5]:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Add previous best
    if prev_arr is not None:
        seeds.append(prev_arr/prev_arr.sum())

    # Evaluate seeds
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        # Ensure non-neg, sum>0
        vec = np.maximum(vec,0.0)
        tot = vec.sum()
        if tot < 1e-12:
            continue
        vec = vec/tot
        val = _objective_surrogate(vec)
        evaluated.append((vec,val))

    # Sort and select top_k = maybe 6
    evaluated.sort(key=lambda x:x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec,_ in evaluated[:top_k]]
    print('Initial best surrogate', evaluated[0][1] if evaluated else None)

    # Global best initialization
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)

    # For each seed, refine
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # compute allocated time portion; we can compute remaining and distribute.
        remaining = deadline - time.time()
        # allocate maybe 0.15 of remaining per seed.
        sub_deadline = time.time() + remaining * 0.15

        # Stage 1: L-BFGS smoothing
        seq1 = _lbfgs_smooth_optimize(seed_vec, n, sub_deadline,
                                      beta_schedule=[0.5,1,2,5,10,20,40,80,200,500,1000,2000,5000,10000],
                                      max_iter_per_beta=3000)
        val1 = _objective_surrogate(seq1)

        # Stage 2: EG refinement
        seq2, val2 = _exp_gradient_opt_multi(seq1, n, sub_deadline,
                                             max_iters=300_000, eta0=8.0, tol=1e-9)

        # Stage 3: Flatten
        seq3, val3 = _flatten_gradient_opt(seq2, n, sub_deadline,
                                           max_iters=200_000, eta0=5.0)

        # Stage 4: SA
        seq4, val4 = _simulated_annealing(seq3, n, sub_deadline, rng,
                                          max_attempts=80_000, start_sigma=0.0015,
                                          T_start=0.012, T_end=5e-9)

        # Pick best among stage results
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq1, val1), (seq2, val2), (seq3, val3), (seq4, val4)]
        best_seq, best_val = min(candidates, key=lambda x:x[1])
        print(f"Seed {idx+1}/{top_k} local best = {best_val:.6f}")

        if best_val < global_best_val:
            global_best_val = best_val
            global_best_seq = best_seq.copy()
            print(f"  New global best = {global_best_val:.6f}")

    # Final polishing with EG over remaining time
    if time.time() < deadline:
        seq_final, val_final = _exp_gradient_opt_multi(global_best_seq, n, deadline,
                                                      max_iters=2_000_000,
                                                      eta0=6.0, tol=1e-9)
        if val_final < global_best_val:
            global_best_val = val_final
            global_best_seq = seq_final.copy()
            print(f"Final EG improved to {global_best_val:.6f}")

    # Final evaluation
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"Finished: Surrogate {global_best_val:.6f}, true {true_val:.6f}")
    except Exception as e:
        print("Evaluation error:", e)
        true_val = None

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()

"""
Improved step‑function optimiser

The routine builds a diverse seed library (analytic windows, random
Dirichlet draws, the previous best solution) and evaluates a cheap
surrogate 2·n·max(conv) (the sequence is ℓ₁‑normalised).  The best few
seeds are refined by a pipeline:

1. Smooth “max’’ optimisation with L‑BFGS‑B (log‑sum‑exp
   continuation).
2. Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
   using a multi‑active set and Adam‑style momentum.
3. A quadratic “flatten’’ phase that drives the whole convolution
   toward a constant.
4. A short simulated‑annealing polish.

All stages respect the global deadline (default 1000 s).  The final
vector is stored in the global ``height_sequence_1`` and returned as a
plain list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used only for the L‑BFGS stage)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (sum(seq) == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate*i) , i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution shape on the discrete set 1…n (α,β>0)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 6.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Uses all indices within a relative tolerance of the current maximum.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)          # Adam‑style momentum
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices that are almost maximal
        if max_val == 0:
            active = np.arange(len(conv))
        else:
            active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # Adam‑like momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient‑descent that flattens the convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 200_000,
                          eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on Σ_k (conv_k – avg)².
    Drives the whole convolution towards a constant (the average).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # number of entries in the full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum==1
        diff = conv - avg

        rev_a = a[::-1]
        # gradient of Σ diff² w.r.t a (full convolution, then slice)
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent (log‑space) conversion
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS optimiser (smooth max via log‑sum‑exp)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2000) -> np.ndarray:
    """
    Optimise a smooth approximation of the max‑convolution with L‑BFGS‑B.
    Works in log‑softmax space so that the iterate stays on the simplex.
    """
    if _opt is None:
        # SciPy not available – fall back to the start vector.
        return start_vec

    if beta_schedule is None:
        # A wide continuation from a very smooth approximation to a very sharp one.
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 10_000.0, 20_000.0,
            50_000.0, 100_000.0, 200_000.0, 500_000.0,
            1_000_000.0
        ]

    # log‑softmax parameters
    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()                 # numerical stabiliser
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # weights for the smooth max
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
