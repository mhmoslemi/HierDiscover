def _adam_optimize(start_seq, n, deadline, max_iters=200_000, eta=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for t in range(1, max_iters+1):
        if time.time()>deadline: break
        grad_a = _subgradient_max_conv(a)
        grad_log = a * (grad_a - np.dot(grad_a, a))
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val

def _coordinate_sa(start_seq, n, deadline, rng, max_attempts=300_000, start_sigma=0.001, T_start=0.005, T_end=1e-9):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0
    # Determine average magnitude for sigma scaling
    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)
        # pick source i with a[i] > 0
        src = rng.choice(np.where(a > 1e-12)[0])
        dst = rng.integers(0, n)
        if src == dst:
            continue
        # sample amount
        delta = sigma * rng.normal()
        # clamp
        eps = np.clip(delta, -a[dst], a[src])
        if np.abs(eps) < 1e-15:
            attempts += 1
            continue
        a[src] -= eps
        a[dst] += eps
        # ensure non-negative
        # recompute objective
        val = _objective_surrogate(a)
        delta_val = val - cur_val
        if delta_val < 0.0 or rng.random() < math.exp(-delta_val / (T + 1e-12)):
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        else:
            # revert
            a[src] += eps
            a[dst] -= eps
        attempts += 1
    return best_seq, best_val
```

"""
Optimiser for the step‑function bound.

The routine builds a large but tractable search space (default length 8000)
and explores it with several complementary optimisation stages:

1. **Rich seeding** – analytic families (uniform, triangular, Gaussian,
   exponentials, beta, power‑law) plus specialised “block” families that
   concentrate mass in a few separated intervals.  Random Dirichlet draws
   and perturbations of the previous best solution are also added.

2. **Continuous optimisation** – a mirror‑descent (exponentiated‑gradient)
   pass on the exact max‑convolution, a second pass with the Adam optimiser,
   and a smooth‑max L‑BFGS refinement (log‑sum‑exp surrogate).

3. **Stochastic refinement** – two simulated‑annealing stages:
   the classic full‑vector Gaussian perturbation and a lightweight
   coordinate‑wise mass‑transfer SA that moves a tiny amount of probability
   from one index to another.

4. **Final polishing** – a long Adam run, a final L‑BFGS pass and an
   intensive coordinate SA.

All steps respect a hard 1000 s budget; the best sequence found is stored
in the global variable ``height_sequence_1`` and returned as a plain list.
The surrogate objective is
    2·n·max(conv(a,a))      (a is ℓ₁‑normalised)
which equals the true ``evaluate_sequence`` value.
"""

import time
import math
import numpy as np

# optional SciPy import (used for L‑BFGS)
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (exact surrogate, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2 · n · max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – each returns an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_block_uniform(n: int, total_frac: float) -> np.ndarray:
    """
    Two equal blocks at the ends, zero middle.
    ``total_frac`` is the fraction of the whole vector occupied by both blocks.
    """
    total_len = max(2, int(round(total_frac * n)))
    L = total_len // 2                # length of each block
    if L == 0:
        L = 1
    seq = np.zeros(n, dtype=np.float64)
    h = 1.0 / (2 * L)                  # each entry weight
    seq[:L] = h
    seq[-L:] = h
    return seq


def _k_block_uniform(n: int, k: int, block_frac: float) -> np.ndarray:
    """
    ``k`` equal blocks placed with equal gaps (deterministic spacing).
    ``block_frac`` is the fraction of the whole vector occupied by a single block.
    """
    L = max(1, int(round(block_frac * n)))
    if k * L > n:
        L = n // k
    seq = np.zeros(n, dtype=np.float64)
    total_len = k * L
    gap = (n - total_len) // (k + 1)
    pos = gap
    h = 1.0 / (k * L)
    for _ in range(k):
        seq[pos:pos + L] = h
        pos += L + gap
    return seq


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set (averaged)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Return the (averaged) sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    cnt = active.sum()
    if cnt == 0:
        return np.zeros_like(a)
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)             # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / cnt    # averaged gradient
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser on the exact max
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution,
    using the full active set.
    Returns the best sequence and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        grad_a = _subgradient_max_conv(a)   # shape (n,)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)       # chain‑rule

        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser (mirror‑descent with momentum) on the exact max
# ----------------------------------------------------------------------
def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   eta: float = 5.0,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """
    Adam on the exact max‑convolution, parameters tuned for the simplex
    (updates are performed in log‑softmax space).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        grad_a = _subgradient_max_conv(a)
        grad_log = a * (grad_a - np.dot(grad_a, a))

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        return start_vec                     # SciPy not available

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0, 500_000.0,
            1_000_000.0
        ]

    v = np.log(start_vec + 1e-30)               # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing (full‑vector Gaussian perturbations)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9):
    """
    Classic SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Coordinate‑wise simulated annealing (tiny mass transfers)
# ----------------------------------------------------------------------
def _coordinate_sa(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   rng: np.random.Generator,
                   max_attempts: int = 250_000,
                   start_sigma: float = 0.001,
                   T_start: float = 0.008,
                   T_end: float = 1e-9):
    """
    Move a small amount of probability from one index to another.
    Extremely cheap per iteration, allowing many more SA steps.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    # pre‑select indices that currently carry mass (to avoid useless moves)
    nonzero_idx = np.flatnonzero(a > 1e-15)

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        if len(nonzero_idx) == 0:
            break
        i = rng.choice(nonzero_idx)          # source (has some mass)
        j = rng.integers(0, n)               # destination (any index)
        if i == j:
            attempts += 1
            continue

        # sample transfer amount (Gaussian, then clipped)
        delta = sigma * rng.normal()
        eps = np.clip(delta, -a[j], a[i])     # keep both entries non‑negative
        if abs(eps) < 1e-15:
            attempts += 1
            continue

        a[i] -= eps
        a[j] += eps

        # keep the list of non‑zero indices up‑to‑date (cheap heuristic)
        if a[i] <= 1e-15:
            nonzero_idx = nonzero_idx[nonzero_idx != i]
        if a[j] > 1e-15 and j not in nonzero_idx:
            nonzero_idx = np.append(nonzero_idx, j)

        val = _objective_surrogate(a)
        delta_val = val - cur_val
        if delta_val < 0.0 or rng.random() < math.exp(-delta_val / (T + 1e-12)):
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = a.copy()
        else:
            # revert
            a[i] += eps
            a[j] -= eps
            # fix the index lists after revert
            if a[i] > 1e-15 and i not in nonzero_idx:
                nonzero_idx = np.append(nonzero_idx, i)
            if a[j] <= 1e-15 and j in nonzero_idx:
                nonzero_idx = nonzero_idx[nonzero_idx != j]

        attempts += 1

    return best_seq, best_val

# Main entry point

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in
    the global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # tiny safety margin

    # --------------------------------------------------------------
    # Determine problem size – reuse and possibly enlarge previous solution.
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.ndim != 1 or prev_arr.size == 0:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 8000
        print(f"\n[Info] No previous solution – using n={n}")
        base_seq = None
    else:
        prev_n = prev_arr.shape[0]
        # attempt to enlarge modestly (helps the asymptotic constant)
        n = min(max(prev_n * 2, 8000), 12000)
        print(f"\n[Info] Re‑using previous solution of length {prev_n}, expanding to n={n}")
        # linear interpolation to the new length

        x_new = np.linspace(0.0, 1.0, n)
        base_seq = np.interp(x_new, x_old, prev_arr)
        base_seq = np.maximum(base_seq, 0.0)
        s = base_seq.sum()
        if s > 0.0:
            base_seq /= s
        else:
            base_seq = _uniform(n)

    # --------------------------------------------------------------
    # 1. Broad seeding (analytic families + blocks + random draws)
    # --------------------------------------------------------------
    seeds = []

    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))

            seeds.append(_k_block_uniform(n, k, block_frac))

        if s == 0.0:

        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] best surrogate values:")

        print(f"  #{i+1:2d}: {v:.6f}")

    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start local refinement on top seeds
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        sub_deadline = time.time() + max(5.0,
                                         remaining * 0.12 / max(1, (top_k - si)))

        # (a) exponentiated‑gradient (full active set)

                                                n,
                                                sub_deadline,
                                                max_iters=400_000,
                                                eta0=7.0)

        # (b) Adam optimisation

            seq_adam, val_adam = _adam_optimize(seq_eg,
                                                n,

                                               n,
                                               sub_deadline,
                                               max_iter_per_beta=3000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:

                                                  n,

        else:

            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_adam, val_adam),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa),
            (seq_coord, val_coord)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final intensive polishing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final Adam polishing ---")
        final_seq, final_val = _adam_optimize(global_best_seq,
                                              n,
                                              deadline,
                                              max_iters=1_200_000,
                                              eta=4.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final L‑BFGS polishing ---")
        final_seq2 = _lbfgs_smooth_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            max_iter_per_beta=4000)
        final_val2 = _objective_surrogate(final_seq2)
        if final_val2 < global_best_val:
            global_best_seq = final_seq2
            global_best_val = final_val2
            print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final coordinate SA polishing ---")
        final_seq3, final_val3 = _coordinate_sa(global_best_seq,
                                                n,
                                                deadline,
                                                rng,
                                                max_attempts=400_000,
                                                start_sigma=5e-4,
                                                T_start=3e-3,
                                                T_end=1e-9)
        if final_val3 < global_best_val:
            global_best_seq = final_seq3
            global_best_val = final_val3
            print(f"[Final Coord SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
