The objective  

\[
F(a)=\frac{2\,n\;\max_k\;(a*a)[k]}{(\sum_i a_i)^2}
\]

is homogeneous, therefore we may restrict ourselves to vectors `a` with
\(\sum_i a_i = 1\) and minimise  

\[
G(a)=2n\;\max_k\;(a*a)[k].
\]

`G` is a non‑smooth function.  The code below

1. builds a very diverse set of **seed vectors** (analytic windows,
   random Dirichlet samples, mixtures, permutations of the current best,
   …);
2. evaluates the cheap surrogate `G` for all seeds and keeps the best
   *k* of them;
3. refines every selected seed with three complementary local
   optimisers  

   * **Exponentiated‑gradient (mirror descent)** – works directly on the
     exact non‑smooth objective using the full active set;
   * **Adam on a smooth‑max surrogate** – a first‑order optimiser that
     follows a differentiable approximation of `max`;
   * **Simulated‑annealing** – a stochastic hill‑climber that can escape
     shallow local minima.
4. performs a **final intensive polishing** of the global best by
   long runs of the two gradient methods followed by a high‑budget
   simulated‑annealing pass.

All work respects the 1000 s‐budget.  The implementation is completely
self‑contained; only `numpy`, `scipy` (optional) and the provided
`evaluate_sequence` are required.

```python
"""
Proposes a step‑function (non‑negative height vector) that minimises
evaluate_sequence().  The optimiser combines a large pool of analytic
initialisers, exponentiated‑gradient mirror descent, Adam on a smooth‑max
surrogate and simulated‑annealing.  It respects the 1000 s time budget
and stores the best vector in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (used for L‑BFGS)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

# ----------------------------------------------------------------------
# FFT utilities (convolution via real FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (exact max, assuming sum(a)=1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a, a))   with   sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    return a / a.sum()


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    return a / a.sum()


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    return a / a.sum()


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    return mixed / mixed.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# Additional families ----------------------------------------------------
def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – uses scipy.special.i0 if available."""
    try:
        from scipy.special import i0
    except Exception:  # pragma: no cover
        # fallback: use a simple approximation (cosine window)
        return _hamming_window(n)
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    arg = beta * np.sqrt(1.0 - ((i - alpha) / alpha) ** 2)
    w = i0(arg) / i0(beta)
    return w / w.sum()


def _tukey_window(n: int, alpha: float = 0.5) -> np.ndarray:
    """Tukey (tapered cosine) window."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.empty_like(i)
    # left taper
    left = i < alpha / 2
    a[left] = 0.5 * (1 + np.cos(np.pi * (2 * i[left] / alpha - 1)))
    # middle flat part
    middle = (i >= alpha / 2) & (i <= 1 - alpha / 2)
    a[middle] = 1.0
    # right taper
    right = i > 1 - alpha / 2
    a[right] = 0.5 * (1 + np.cos(np.pi * (2 * i[right] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set (averaged)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Averaged sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # 2n‑1
    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)             # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / active.sum()
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0):
    """
    Mirror‑descent (exponentiated‑gradient) minimising the exact
    max‑convolution.  Works in log‑softmax space; always returns a
    ℓ₁‑normalised vector.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        grad_a = _subgradient_max_conv(a)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)        # chain rule (∂/∂log a)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_opt_smooth(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 2500,
                     lr: float = 0.12):
    """
    Adam optimisation of the log‑softmax variables with the smooth‑max
    surrogate  fβ(a)=2·n·(shift+log(∑exp(β·(conv−shift)))/β).
    """
    if beta_schedule is None:
        beta_schedule = [0.5, 1.0, 5.0, 20.0, 80.0,
                         200.0, 500.0, 1_000.0, 2_000.0,
                         5_000.0, 20_000.0, 50_000.0,
                         100_000.0, 200_000.0, 500_000.0,
                         1_000_000.0]

    v = np.log(start_seq + 1e-30)               # log‑softmax variables
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    t = 0
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        beta_val = float(beta)
        for _ in range(iters_per_beta):
            if time.time() > deadline:
                return best_seq
            a = _softmax(v)

            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # smooth‑max value (not needed for the optimiser itself)
            # obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp                     # weight vector
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]        # ∂fβ/∂a

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)                # chain‑rule

            # Adam update
            t += 1
            m = 0.9 * m + 0.1 * grad_v
            v_hat = 0.999 * v_hat + 0.001 * (grad_v ** 2)
            m_hat = m / (1.0 - 0.9 ** t)
            v_hat_hat = v_hat / (1.0 - 0.999 ** t)
            v -= lr * m_hat / (np.sqrt(v_hat_hat) + 1e-9)

            cur_val = _objective_surrogate(_softmax(v))
            if cur_val < best_val:
                best_val = cur_val
                best_seq = _softmax(v).copy()
    return best_seq


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

                n = arr.shape[0]
                prev_arr = arr / arr.sum()
            else:
                prev_arr = None
                n = 4000
        except Exception:
            prev_arr = None
            n = 4000
    else:

                                                       1_000.0, 2_000.0,

                                        iters_per_beta=1800,
                                        lr=0.13)
            val_adam = _objective_surrogate(seq_adam)
        else:
```
