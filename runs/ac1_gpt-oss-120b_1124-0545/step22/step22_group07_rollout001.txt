"""
Propose a step‑function (non‑negative heights) that minimises ``evaluate_sequence``.
The search consists of three stages:

1. **Diverse seed generation** – many analytic families (uniform, exponential,
   symmetric exponential, Gaussian, power‑law, beta, Dirichlet, random sparse,
   piecewise‑linear) with a wide sweep of parameters plus the previous best.

2. **Refinement of the top seeds** – each top seed is processed with a
   smoothed Adam optimiser (log‑softmax representation, log‑sum‑exp smoothing)
   using a long schedule of inverse‑temperature values, followed by an exact
   exponentiated‑gradient (EG) polish on the true max‑convolution.

3. **Final polishing** – a long EG run on the current global best and a very
   short permutation‑annealing tweak.

All work respects the global time budget (default 1000 s).  The final
sequence is stored in ``height_sequence_1`` and returned as a plain ``list``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real, full length)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact surrogate objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """2·n·max(conv(seq, seq)) – the same value that ``evaluate_sequence``
    returns when ``seq`` sums to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries of equal weight (k≥1)."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=min(k, n), replace=False)
    a[idx] = 1.0 / len(idx)
    return a


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the exact max‑convolution.
    Operates in log‑softmax (probability) space.
    """
    if beta_schedule is None:
        # start smooth, end very sharp – geometric growth
        beta_schedule = [5, 20, 80, 320, 1280, 5120,
                         20480, 81920, 327680, 1_310_720]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)          # initialise log‑softmax

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.13                                 # learning‑rate factor
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)                     # for numerical stability
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                    # soft‑max weights

            # gradient wrt a  (see derivation in the original code)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient wrt log‑softmax parameters
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded to avoid overflow/underflow
            log_a = np.clip(log_a, -50.0, 50.0)

            # exact surrogate value for the current iterate
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient optimiser (sub‑gradient of the true max)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active (max) diagonal
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size schedule η_t = η0 / sqrt(t)
        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a - np.max(log_a))      # keep numerically stable
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very short permutation‑annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Simple permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val

    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # allocate ≈ 70 % of the remaining budget to per‑seed refinement
    remaining_total = deadline - time.time()
    refinement_budget = remaining_total * 0.70
    per_seed_budget = refinement_budget / top_k if top_k > 0 else 0.0

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_seed_budget
        print(f"\n[Seed {idx+1}] time slice ≈ {per_seed_budget:.1f}s")

        # ---- smoothed Adam optimisation --------------------------------
        seq_adam, val_adam = _adam_smooth_opt(seed_vec,
                                              sub_deadline,
                                              iters_per_beta=100_000)

        # ---- exact EG polish (if time left) ----------------------------
        if time.time() < sub_deadline:
            seq_eg, val_eg = _exp_gradient_opt(seq_adam,
                                               sub_deadline,
                                               max_iters=1_200_000,
                                               eta0=6.0)
        else:
            seq_eg, val_eg = seq_adam, val_adam

        # pick the better of the two refinements
        if val_adam < val_eg:
            local_seq, local_val = seq_adam, val_adam
        else:
            local_seq, local_val = seq_eg, val_eg

        print(f"[Seed {idx+1}] local best surrogate = {local_val:.6f}")

        if local_val < global_best_val:
            global_best_seq = local_seq
            global_best_val = local_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Final long EG polishing (uses the rest of the budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=3_000_000,
                                                 eta0=5.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Very short permutation‑annealing tweak (a few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 20.0
        seq_last, val_last = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=70_000)
        if val_last < global_best_val:
            global_best_seq = seq_last
            global_best_val = val_last
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
