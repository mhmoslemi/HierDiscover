"""
Propose a better height sequence.

Algorithm
---------
1. Initialise from the previously stored best sequence (``height_sequence_1``)
   or from a uniform distribution.  The vector is ℓ₁‑normalised to 1.

2. **Smoothed‑gradient descent**  
   Minimise the smooth approximation of  max(conv(a,a)) :

        φ(a) = log‑sum‑exp(α·conv(a,a)) / α

   The gradient of φ is obtained analytically via a convolution,
   and each step is projected onto the probability simplex.  The
   smoother parameter α is large (≈5000) so φ is very close to the
   true max.  A decaying learning‑rate (η₀ / √t) is used.  The loop
   runs until the allocated time is exhausted, keeping the best
   vector seen.

3. **Simulated‑annealing refinement** – tiny random mass transfers
   between two coordinates, with an exponential‐cooling schedule.
   This fine‑tunes the solution found by the gradient phase.

The two phases together respect the 1000 s budget.  The best sequence
found is stored globally as ``height_sequence_1`` and returned as a
plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via a real FFT.
    Returns a 1‑D array of length len(x)+len(y)-1.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Projection onto the probability simplex (sum = 1, non‑negative)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x ≥ 0, sum(x)=1}.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive, return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed‑gradient descent on φ(a) = log‑sum‑exp(α·conv(a,a))/α
# ----------------------------------------------------------------------
def _smooth_gradient_descent(start_seq: np.ndarray,
                             deadline: float,
                             alpha: float = 5000.0,
                             eta0: float = 0.25,
                             max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Perform projected gradient descent on the smoothed objective.
    Returns the best sequence (ℓ₁‑normalised) encountered and its
    surrogate value 2·n·max(conv).
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    # surrogate = 2·n·max(conv) because sum(a)=1
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)

    # pre‑allocate buffers to avoid repeated allocations where possible
    rev_a = a[::-1]

    for it in range(1, max_iters + 1):
        now = time.time()
        if now > deadline:
            break

        # ----- compute convolution and smooth weights -----
        conv = _conv_fft(a, a)                 # length 2n‑1
        max_conv = np.max(conv)

        # smooth max: w_i ∝ exp(α·(conv_i - max_conv))
        # subtraction of max_conv improves numerical stability
        exp_terms = np.exp(alpha * (conv - max_conv))
        w = exp_terms / exp_terms.sum()        # sum(w)=1, length 2n‑1

        # ----- gradient of φ -----
        # g = 2 * (w * a_rev) convolution, slice n entries
        rev_a = a[::-1]
        conv_w_rev = _conv_fft(w, rev_a)       # length 3n‑2
        grad = 2.0 * conv_w_rev[n - 1: 2 * n - 1]   # shape (n,)

        # ----- update with decaying step size -----
        eta = eta0 / math.sqrt(it)
        a_next = a - eta * grad

        # ----- project back onto simplex -----
        a = _simplex_projection(a_next)

        # ----- keep the best -----
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional progress output (every 2000 iterations)
        if it % 2000 == 0:
            print(f"[GD] iter={it:,}  surrogate={cur_val:.6f}  best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 1_000_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simple SA: transfer a small amount of probability mass between two random
    indices.  Temperature and perturbation size decay linearly.
    Returns the best sequence seen.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    # surrogate = 2·n·max(conv) because sum=1
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * np.max(conv)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)   # proportionally shrink

        # pick donor index that currently has some mass
        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        # amount to move: a tiny fraction of a[i]
        amt = sigma * rng.random() * a[i]      # 0 ≤ amt ≤ sigma·a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        # evaluate surrogate of candidate
        conv_ns = _conv_fft(ns, ns)
        val = 2.0 * n * np.max(conv_ns)
        delta = val - cur_val

        # Metropolis acceptance
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

        # occasional logging
        if attempts % 100_000 == 0:
            print(f"[SA] attempts={attempts:,}  cur={cur_val:.6f}  best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (height vector) that minimises
    ``evaluate_sequence``.  The routine loads any previous solution stored
    in the global variable ``height_sequence_1``, runs a smoothed‑gradient
    descent phase followed by simulated‑annealing refinement, stores the
    best vector globally and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # --------------------------------------------------------------
    # Initialise – use previously discovered best if available
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                raise ValueError
            s = arr.sum()
            if s <= 0.0:
                raise ValueError
            seq0 = arr / s                 # ℓ₁‑normalise to sum = 1
        except Exception:
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        # default to a uniform vector of length 4000
        n = 4000
        seq0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = seq0.shape[0]

    print("\n=== Optimisation started — n =", n, "=== ")
    init_conv = _conv_fft(seq0, seq0)
    init_val = 2.0 * n * np.max(init_conv)
    print(f"Initial surrogate value: {init_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – smoothed gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    gd_deadline = time.time() + max(5.0, 0.65 * remaining)   # ~65 % of remaining time
    print("\n--- Phase 1: Smoothed gradient descent ---")
    seq_gd, val_gd = _smooth_gradient_descent(seq0, gd_deadline,
                                              alpha=5000.0,
                                              eta0=0.25,
                                              max_iters=10_000_000)
    print(f"[GD] best surrogate = {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – simulated‑annealing refinement
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        sa_deadline = time.time() + remaining
        print("\n--- Phase 2: Simulated‑annealing refinement ---")
        seq_sa, val_sa = _simulated_annealing_refine(seq_gd, sa_deadline,
                                                     rng,
                                                     max_attempts=1_000_000,
                                                     T_start=0.02,
                                                     T_end=1e-8,
                                                     sigma_start=0.02)
        print(f"[SA] best surrogate = {val_sa:.6f}")
    else:
        seq_sa, val_sa = seq_gd, val_gd

    # --------------------------------------------------------------
    # Choose the overall best
    # --------------------------------------------------------------
    if val_sa < val_gd:
        best_seq = seq_sa
        best_val = val_sa
    else:
        best_seq = seq_gd
        best_val = val_gd

    print(f"\n=== Final surrogate after polishing = {best_val:.6f} ===")

    # --------------------------------------------------------------
    # Evaluate with the provided ``evaluate_sequence`` (may differ slightly)
    # --------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"True evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for reuse by subsequent runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
