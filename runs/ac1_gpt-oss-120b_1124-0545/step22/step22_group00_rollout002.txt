"""
Step‑function upper‑bound optimiser.

The optimiser generates a large, diverse set of candidate
distributions (analytic families, random Dirichlet draws,
permutations and mixtures).  All candidates are ℓ₁‑normalised, then the
cheap surrogate 2·n·max(conv(a,a)) is evaluated.  The best few candidates
are refined by three complementary local optimisers:

* exponentiated‑gradient (mirror‑descent on the exact sub‑gradient);
* Adam (mirror‑descent with adaptive moments);
* simulated‑annealing (large‑scale stochastic search).

A final high‑budget polishing pass combines the two gradient methods
and a long SA run.  The routine respects a 1000 s wall‑time budget,
stores the best sequence in ``height_sequence_1`` and returns it as a
plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when ∑a = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set (averaged)
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Return the (averaged) sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    count = active.sum()
    if count == 0:
        return np.zeros_like(a)

    rev_a = a[::-1]
    full = _conv_fft(active, rev_a)             # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / count   # averaged gradient
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 250_000,
                           eta0: float = 8.0):
    """Mirror‑descent on the exact max‑convolution (full active set)."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        grad_a = _subgradient_max_conv(a)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)          # chain‑rule (∇_log a)

        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser (mirror‑descent in log‑space)
# ----------------------------------------------------------------------
def _adam_opt_full(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 250_000,
                   eta: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8):
    """Adam‑style optimisation of the exact max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1_pow = beta1
    beta2_pow = beta2

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        grad_a = _subgradient_max_conv(a)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        log_a -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # update bias‑correction factors
        beta1_pow *= beta1
        beta2_pow *= beta2

        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimiser (log‑softmax space)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 2500):
    """Smooth‑max L‑BFGS (log‑softmax variables)."""
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:  # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 250_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-9):
    """Gaussian‑perturbation SA with a linear temperature schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation – a diverse palette of analytic & random families
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator, prev_seq: np.ndarray | None):
    """Return a list of candidate probability vectors."""
    seeds = []

    # Deterministic analytic families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    for gamma in (0.5, 1.0, 2.0, 3.0, 5.0):
        seeds.append(_sine_window(n, gamma=gamma))

    for sigma_frac in (0.05, 0.10, 0.15, 0.20, 0.30):
        seeds.append(_gaussian(n, sigma_frac=sigma_frac))

    for p in (0.0, 0.5, 1.0, 2.0, 3.0, 4.5, 6.0):
        seeds.append(_power(n, p))

    # Beta families – random sampling of (α,β) in log‑space
    for _ in range(30):
        loga = rng.uniform(-3.0, 2.0)   # α ∈ [10⁻³, 10²]
        logb = rng.uniform(-3.0, 2.0)
        alpha = 10.0 ** loga
        beta = 10.0 ** logb
        seeds.append(_beta_shape(n, alpha, beta))

    # Exponential decays (left & right) with many rates
    for rate in np.logspace(-4, -0.2, 40):
        seeds.append(_exp_decay_left(n, rate))
        seeds.append(_exp_decay_right(n, rate))

    # Mixtures of simple shapes -------------------------------------------------------
    for _ in range(20):
        a = _uniform(n)
        b = _exp_decay_left(n, rng.uniform(1e-4, 1e-1))
        w = rng.random()
        seeds.append(_mix_shapes(a, b, w))

        a = _uniform(n)
        b = _exp_decay_right(n, rng.uniform(1e-4, 1e-1))
        w = rng.random()
        seeds.append(_mix_shapes(a, b, w))

    # Random Dirichlet draws (covering sparse → dense regimes) ----------------------
    for conc in (0.001, 0.005, 0.01, 0.05, 0.2, 1.0, 2.0):
        for _ in range(6):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Permuted Dirichlet draws – adds combinatorial diversity -------------------------
    for _ in range(20):
        base = _random_dirichlet(n, rng.uniform(0.01, 0.5), rng)
        seeds.append(base)                               # ordered version
        perm = rng.permutation(base)                     # shuffled version
        seeds.append(perm)

    # Perturb previous best (if any) --------------------------------------------------
    if prev_seq is not None and prev_seq.sum() > 0.0:
        base = prev_seq.copy()
        seeds.append(base)
        for _ in range(10):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0.0:
                seeds.append(pert / pert.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – reuse previous solution if present.
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            n = arr.shape[0]
            if n > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is None:
        n = 5000                        # a slightly larger default size
        print(f"\n[Info] No previous solution – using n={n}")
    else:
        print(f"\n[Info] Re‑using previous solution of length n={n}")

    # ------------------------------------------------------------------
    # 1. Broad seeding
    # ------------------------------------------------------------------
    seeds = _generate_seeds(n, rng, prev_arr)

    # ------------------------------------------------------------------
    # 2. Fast surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds generated before deadline.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(30, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] top surrogate values (sorted):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Multi‑start local refinement on the top seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a slice of the remaining budget (roughly equal share)
        remaining = deadline - time.time()
        per_seed_budget = max(4.0, remaining / (top_k - si) * 0.7)
        sub_deadline = time.time() + per_seed_budget

        # -------- (a) exponentiated‑gradient (full active set) -------------
        seq_eg, val_eg = _exp_gradient_opt_full(seed_vec.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=300_000,
                                                eta0=9.0)

        # -------- (b) Adam optimiser -------------------------------------
        if time.time() < sub_deadline:
            seq_adam, val_adam = _adam_opt_full(seq_eg,
                                                n,
                                                sub_deadline,
                                                max_iters=250_000,
                                                eta=0.6)
        else:
            seq_adam, val_adam = seq_eg, val_eg

        # -------- (c) L‑BFGS smoothing ------------------------------------
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_adam,
                                               n,
                                               sub_deadline,
                                               beta_schedule=[
                                                   0.5, 1.0, 5.0, 20.0, 80.0,
                                                   200.0, 500.0, 1_000.0,
                                                   2_000.0, 5_000.0,
                                                   20_000.0, 50_000.0,
                                                   100_000.0, 200_000.0,
                                                   500_000.0, 1_000_000.0],
                                               max_iter_per_beta=2000)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_adam, val_adam

        # -------- (d) Simulated annealing ---------------------------------
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=180_000,
                                                  start_sigma=0.0025,
                                                  T_start=0.012,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # pick the best among all refinements
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_adam, val_adam),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa)
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final polishing – intensive EG + Adam + SA
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final intensive EG polishing (full‑active) ---")
        final_seq, final_val = _exp_gradient_opt_full(global_best_seq,
                                                      n,
                                                      deadline,
                                                      max_iters=2_500_000,
                                                      eta0=7.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final Adam polishing ---")
        final_seq2, final_val2 = _adam_opt_full(global_best_seq,
                                                n,
                                                deadline,
                                                max_iters=1_500_000,
                                                eta=0.5)
        if final_val2 < global_best_val:
            global_best_seq = final_seq2
            global_best_val = final_val2
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final SA polishing (high‑budget) ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          deadline,
                                                          rng,
                                                          max_attempts=350_000,
                                                          start_sigma=9e-4,
                                                          T_start=4e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate = {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
