    else:

    alpha_end = 200.0

    it = 0

    while True:

        alpha = alpha0 * (1 - frac) + alpha_end * frac

        maxc = conv.max()

        rev_a = a[::-1]

"""
Propose a (potentially) better step‑function height vector.

The algorithm works on a probability vector (sum = 1) and minimises the
surrogate objective

    S(a) = 2·n·max(conv(a, a))

which, after scaling the final vector to ℓ₁‑norm √(2 n), equals the value
returned by ``evaluate_sequence``.  Two phases are performed:

1. **Smoothed exponentiated‑gradient descent**.
   The true max in the objective is replaced by a soft‑max with a
   temperature α that grows from a low to a high value.  The resulting
   (sub‑)gradient is a weighted cross‑correlation which can be obtained
   with three FFT‑based convolutions per iteration.  The update is carried
   out in the log‑domain and the iterate is projected onto the simplex
   by a soft‑max.  A decaying step size (η ∝ 1/√t) and a slowly increasing
   α allow the method to escape plateaus and converge to a lower bound
   than the simple “single active shift’’ approach used previously.

2. **Simulated‑annealing fine‑tuning**.
   Starting from the best vector found in phase 1, tiny mass‑transfer
   moves (and occasional swaps) are accepted according to a linear
   cooling schedule.  This refines the solution without the cost of the
   full gradient step.

Both phases respect the overall 1000 s budget and the best sequence is
stored globally under the name ``height_sequence_1`` for future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric primitives
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    # s cannot be zero – if it is, fall back to uniform
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Smoothed surrogate objective and its gradient (in log‑space)
# ----------------------------------------------------------------------
def _surrogate_and_grad(a: np.ndarray, alpha: float) -> tuple[float, np.ndarray]:
    """
    Returns (value, gradient) of the smoothed surrogate:
        value = 2·n·max(conv(a,a))   (for reporting)
        gradient = ∂/∂a  (soft‑max_{k} conv_k(a))

    ``a`` is assumed to be a probability vector (sum = 1).
    """
    n = a.shape[0]

    # exact convolution
    conv = _conv_fft(a, a)                     # length 2n‑1
    max_c = conv.max()

    # soft‑max weights over the 2n‑1 shifts
    # use max_c to keep the exponentials well‑scaled
    exp_arg = alpha * (conv - max_c)
    # for very large α the exponentials could under‑flow → protect
    exp_val = np.exp(exp_arg)
    w = exp_val / exp_val.sum()                # weights, sum to 1

    # gradient of the smoothed max:
    #   g_i = 2 * Σ_k w_k * a_{k-i}   (cross‑correlation)
    rev_a = a[::-1]
    cross = _conv_fft(w, rev_a)               # length 3n‑2
    g = 2.0 * cross[n - 1 : n - 1 + n]        # slice the n relevant entries

    # surrogate value for the original (non‑smoothed) objective
    surrogate_val = 2.0 * n * max_c
    return surrogate_val, g


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (tiny mass moves + optional swaps)
# ----------------------------------------------------------------------
def _simulated_annealing(start_a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 300_000,
                         T_start: float = 0.02,
                         T_end: float = 1e-6,
                         sigma_start: float = 0.005) -> tuple[np.ndarray, float]:
    """
    Tiny‑mass transfers (and occasional swaps) on the probability simplex.
    Returns the best vector seen and its surrogate value 2·n·max(conv).
    """
    n = start_a.shape[0]
    a = start_a.copy()
    best_a = a.copy()
    best_val = 2.0 * n * _conv_fft(a, a).max()

    attempts = 0
    while attempts < max_attempts and time.time() < deadline:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        # with small probability try a pure swap (useful for ordering)
        if rng.random() < 0.15:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                attempts += 1
                continue
            ns = a.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:
            # move a tiny amount of probability mass from i → j
            i = rng.integers(n)
            if a[i] <= 1e-15:
                attempts += 1
                continue
            amt = sigma * rng.random() * a[i]          # ≤ σ·a[i]
            if amt <= 0.0:
                attempts += 1
                continue
            j = rng.integers(n)
            if i == j:
                attempts += 1
                continue
            ns = a.copy()
            ns[i] -= amt
            ns[j] += amt

        # evaluate the move (full convolution – still cheap enough)
        cur_val = 2.0 * n * _conv_fft(ns, ns).max()
        delta = cur_val - best_val

        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            if cur_val < best_val:
                best_val = cur_val
                best_a = ns.copy()
        attempts += 1

    return best_a, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a height sequence that minimises ``evaluate_sequence``.
    The function loads any previously computed ``height_sequence_1``,
    runs a smoothed exponentiated‑gradient descent followed by a
    simulated‑annealing polishing phase and finally returns the best
    sequence (as a plain ``list[float]``), scaled to ℓ₁‑norm √(2·n).
    """
    rng = np.random.default_rng(seed)
    t_start = time.time()
    deadline = t_start + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Initialise – reuse previous solution if possible, otherwise uniform
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                raise ValueError
            # rescale to a probability vector (sum = 1)
            a0 = arr / arr.sum()
            n = a0.shape[0]
        except Exception:
            arr = None

    if prev is None or arr is None:
        n = 4000                                   # keep the same length as before
        a0 = np.full(n, 1.0 / n, dtype=np.float64)  # uniform start

    print(f"\n=== Optimisation started – n = {n} ===")
    init_val = 2.0 * n * _conv_fft(a0, a0).max()
    print(f"Initial surrogate value (2·n·max(conv)) = {init_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – smoothed exponentiated‑gradient descent
    # ------------------------------------------------------------------
    max_iters = 500_000                # hard cap; loop also respects time budget
    a = a0.copy()
    log_a = np.log(a + 1e-30)          # safe log‑domain representation

    best_a = a.copy()
    best_surrogate = init_val

    eta0 = 1.0                         # initial step size factor
    alpha_start = 0.5
    alpha_end = 200.0

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # schedule for smoothing temperature α (linear increase)
        frac = it / max_iters
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # compute surrogate value and its gradient
        surrogate, grad = _surrogate_and_grad(a, alpha)

        # step‑size schedule (standard 1/√t decay)
        eta = eta0 / math.sqrt(it)

        # mirror‑descent step in log‑space and project back onto simplex
        log_a -= eta * grad
        a = _softmax(log_a)

        # record best
        if surrogate < best_surrogate:
            best_surrogate = surrogate
            best_a = a.copy()
            print(f"[EG] iter {it:7d} – new best {best_surrogate:.6f}")

        # occasional progress report
        if it % 50_000 == 0:
            elapsed = time.time() - t_start
            print(f"  EG progress – iter {it:,}, best {best_surrogate:.6f}, elapsed {elapsed:.1f}s")

    print(f"\n--- Phase 1 finished: best surrogate = {best_surrogate:.6f} ---")

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:

        polished_a, polished_val = _simulated_annealing(
            best_a,
            sa_deadline,
            rng,
            max_attempts=300_000,
            T_start=0.02,
            T_end=1e-6,
            sigma_start=0.003,
        )
        if polished_val < best_surrogate:
            best_surrogate = polished_val
            best_a = polished_a
            print(f"[SA] new best surrogate {best_surrogate:.6f}")
        else:
            print(f"[SA] best surrogate unchanged ({best_surrogate:.6f})")
    else:
        print("\n--- Phase 2 skipped (time budget exhausted) ---")

    # ------------------------------------------------------------------

    final_seq = np.clip(final_seq, 0.0, 1000.0)
