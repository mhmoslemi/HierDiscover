"""
Advanced optimizer for non‑negative step‑function heights.

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a)²)

which, after normalising the vector to sum = 1, is simply
    2·n·max(conv(a,a)).
The algorithm proceeds in three stages:

1. **Massive seed generation** – a rich library of analytic windows
   (uniform, triangular, sine, cosine, Black‑Man, Hamming, Kaiser,
   Gaussian, power‑law, Beta, logistic, DPSS) together with a very
   large set of random Dirichlet draws and randomly‑parameterised
   windows.  The best *K* seeds are retained using a min‑heap.

2. **Multi‑start local optimisation** – each of the retained seeds is
   refined by
   * exponentiated‑gradient (mirror descent) on the exact max‑convolution
     (active set of constraints),
   * a “flattening” gradient phase that minimises the variance of the
     full convolution,
   * a brief simulated‑annealing polish.
   The best of all runs becomes the global candidate.

3. **Final polishing** – a long EG run, a final flattening phase and a
   short SA polish are performed on the global best until the time
   budget expires.

All iterates stay on the probability simplex via a soft‑max
parameterisation, guaranteeing non‑negative coefficients.
The final sequence is stored in the global variable
``height_sequence_1`` and also returned as a plain list.
"""

import time
import math
import heapq
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Optional scientific imports (used only for a few analytic windows)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
except Exception:   # pragma: no cover
    _opt = None

try:
    from scipy.special import i0 as _i0
except Exception:   # pragma: no cover
    _i0 = None

try:
    from scipy.signal import windows as _sigwin
except Exception:   # pragma: no cover

# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate = 2·n·max(conv(seq,seq))   (assumes sum(seq)=1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy i0 is unavailable."""
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i) , i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    """Sigmoid shape, optionally shifted."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `conc` (scalar)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires scipy.signal."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent on the exact max‑convolution (multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 6.0,
                            tol: float = 1e-8) -> Tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the exact max‑convolution."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # simple momentum (Adam‑like)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices whose value is close to the current maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        # gradient of the max (average over active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient–descent that flattens the convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000,
                         eta0: float = 5.0) -> Tuple[np.ndarray, float]:
    """Mirror‑descent on Σ (conv - avg)² – tries to flatten the autocorrelation."""
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)   # length of full linear convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent step in log‑space
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.003,
                         T_start: float = 0.015,
                         T_end: float = 1e-8) -> Tuple[np.ndarray, float]:
    """Gaussian perturbation SA with a linear cooling schedule."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (ℓ₁‑normalised) that minimises
    ``evaluate_sequence``.  The best vector is stored in the global variable
    ``height_sequence_1`` and also returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Problem size – keep the same length as any previous solution,
    # otherwise fall back to 4000.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            n = int(prev_arr.shape[0])
            if n <= 0:
                raise ValueError
        except Exception:
            n = 4000
            prev_arr = None
    else:
        n = 4000
        prev_arr = None

    # ------------------------------------------------------------------
    # 1.  Build an extensive seed library (analytic + random)
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seed_vectors = []

    # deterministic analytic families
    seed_vectors.append(_uniform(n))
    seed_vectors.append(_triangular(n))
    seed_vectors.append(_sine_window(n, gamma=1.0))
    seed_vectors.append(_blackman_window(n))
    seed_vectors.append(_hamming_window(n))
    seed_vectors.append(_cosine_window(n))

    # Kaiser windows – a sweep of beta values
    for beta in np.linspace(0.0, 12.0, 13):
        seed_vectors.append(_kaiser_window(n, beta))

    # Gaussian windows – sweep of sigma fraction
    for sigma_frac in np.linspace(0.05, 0.5, 10):
        seed_vectors.append(_gaussian(n, sigma_frac))

    # Power‑law tails
    for p in np.linspace(0.0, 6.0, 13):
        seed_vectors.append(_power(n, p))

    # DPSS windows (if available) – a fine grid of NW values
    if _sigwin is not None:
        for nw in np.linspace(2.0, 5.5, 8):
            seed_vectors.append(_dpss_window(n, nw))

    # Random beta‑distribution shapes
    for _ in range(150):
        alpha = 10 ** rng.uniform(-2, 1)     # 0.01 … 10
        beta = 10 ** rng.uniform(-2, 1)
        seed_vectors.append(_beta_shape(n, alpha, beta))

    # Random logistic shapes
    for _ in range(150):
        scale = 10 ** rng.uniform(-0.5, 0.8)   # ~0.3 … 6.3
        midpoint = rng.uniform(0, n)
        seed_vectors.append(_logistic_shape(n, scale, midpoint))

    # Random exponential decays
    for _ in range(200):
        rate = 10 ** rng.uniform(-5, -0.2)    # 1e‑5 … 0.63
        seed_vectors.append(_exp_decay(n, rate))

    # Random mixtures of two analytic shapes
    basic_shapes = [
        _uniform(n),
        _triangular(n),
        _sine_window(n, 1.0),
        _blackman_window(n),
        _hamming_window(n),
        _cosine_window(n),
    ]
    for _ in range(200):
        a, b = rng.choice(basic_shapes, size=2, replace=False)
        alpha = rng.random()
        seed_vectors.append(_mix_shapes(a, b, alpha))

    # Random Dirichlet draws – wide range of concentration parameters
    for _ in range(3000):
        conc = 10 ** rng.uniform(-5, 2)    # 1e‑5 … 100
        seed_vectors.append(_random_dirichlet(n, conc, rng))

    # Include reversed versions of everything gathered so far (helps break symmetry)
    rev_seeds = [s[::-1].copy() for s in seed_vectors]
    seed_vectors.extend(rev_seeds)

    # Add the previous best (if any) and slight perturbations of it
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seed_vectors.append(base)
        for _ in range(6):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seed_vectors.append(pert / s)

    # ------------------------------------------------------------------
    # 2.  Quick surrogate evaluation – keep the K best seeds
    # ------------------------------------------------------------------
    K = 30
    best_heap: List[Tuple[float, np.ndarray]] = []   # max‑heap (store negative surrogate)

    print("\nEvaluating seeds (surrogate)...")
    for vec in seed_vectors:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)

        if len(best_heap) < K:
            heapq.heappush(best_heap, (-val, vec))
        else:
            if -best_heap[0][0] > val:
                heapq.heapreplace(best_heap, (-val, vec))

    # Extract the best K seeds, sorted from best to worst
    best_heap.sort(key=lambda x: -x[0])   # now highest‑negative (i.e. smallest val) first
    top_seeds = [item[1] for item in best_heap]

    print("\nTop seed surrogate values:")
    for i, (_, v) in enumerate(best_heap):
        print(f"  #{i+1}: { -v:.6f}")

    # ------------------------------------------------------------------
    # 3.  Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Multi‑start refinement on the top seeds
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Allocate a time slice for this seed (roughly equal share of remaining budget)
        remaining = deadline - time.time()
        slice_time = remaining * 0.8 / max(1, len(top_seeds) - si)
        sub_deadline = time.time() + slice_time

        # ---- a) exponentiated‑gradient (multi‑active) ----------
        seq_eg, val_eg = _exp_gradient_opt_multi(seed_vec.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=2_500_000,
                                                eta0=8.0,
                                                tol=1e-9)

        # ---- b) flattening phase ----------
        if time.time() < sub_deadline:
            seq_flat, val_flat = _flatten_gradient_opt(seq_eg,
                                                       n,
                                                       sub_deadline,
                                                       max_iters=300_000,
                                                       eta0=5.0)
        else:
            seq_flat, val_flat = seq_eg, val_eg

        # ---- c) short simulated annealing ----------
        hill_deadline = sub_deadline - 0.25
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_flat,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=70_000,
                                                  start_sigma=0.0015,
                                                  T_start=0.012,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_flat, val_flat

        # Choose the best among the candidates for this seed
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_flat, val_flat),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{len(top_seeds)}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq, global_best_val = local_best_seq, local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5.  Final long polishing (heavy EG + flatten + brief SA)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=5_000_000,
                                                       eta0=6.0,
                                                       tol=1e-8)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final flattening stage ---")
        final_seq2, final_val2 = _flatten_gradient_opt(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=1_200_000,
                                                       eta0=4.0)
        if final_val2 < global_best_val:
            global_best_seq, global_best_val = final_seq2, final_val2
            print(f"[Final flatten] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final brief SA polishing ---")
        final_seq3, final_val3 = _simulated_annealing(global_best_seq,
                                                      n,
                                                      deadline,
                                                      rng,
                                                      max_attempts=100_000,
                                                      start_sigma=0.001,
                                                      T_start=5e-3,
                                                      T_end=1e-9)
        if final_val3 < global_best_val:
            global_best_seq, global_best_val = final_seq3, final_val3
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6.  Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
