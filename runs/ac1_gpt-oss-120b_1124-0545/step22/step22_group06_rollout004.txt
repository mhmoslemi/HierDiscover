"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.  The algorithm builds a diverse pool of
parameterised seed distributions (uniform, triangular, Beta and a
family a_i ∝ ((i+½)(N‑½‑i))⁻ᵞ).  The best seed is refined by three
phases:

1. **Exponentiated‑gradient polishing** – mirror‑descent on the exact
   max‑convolution using the sub‑gradient at the active index.
2. **Simulated‑annealing walk** on the simplex – small random mass
   transfers are accepted with a temperature schedule, allowing the
   search to escape shallow local minima.
3. **Final polishing** – a short second pass of (1) to tighten the
   result.

All phases respect the global time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)).  ``seq`` is assumed to sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

# Seed generators (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Symmetric / asymmetric Beta density on a uniform grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    a = ((x + eps) ** (alpha - 1.0)) * ((1.0 - x + eps) ** (beta - 1.0))
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = _uniform(n)
    else:
        a = a / total
    return a


def _gamma_shape(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{-γ}  – a family that contains the arcsine (γ=0.5)."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + 0.5) * (n - 0.5 - i) + 1e-12
    a = denom ** (-gamma)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = _uniform(n)
    else:
        a = a / total
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 1_000_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it + 1)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny simulated‑annealing refinement on the simplex
# ----------------------------------------------------------------------


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         init_temp: float = None,
                         final_temp: float = 1e-8,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Stochastic walk that moves a small random amount of mass from one
    coordinate to another.  Moves that increase the surrogate are accepted
    with probability exp(-Δ / T) where T falls linearly from ``init_temp``
    to ``final_temp``.
    """
    rng = np.random.default_rng()
    cur_seq = start_seq.copy()
    cur_val = _surrogate(cur_seq)

    if init_temp is None:
        init_temp = cur_val * 0.01  # 1 % of the current value

    best_seq = cur_seq.copy()
    best_val = cur_val

    start_time = time.time()
    while time.time() < deadline:
        # linear temperature schedule
        elapsed = time.time() - start_time
        total = deadline - start_time
        T = init_temp * (1.0 - elapsed / total) + final_temp * (elapsed / total)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a transfer proportional to the source mass
        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0.0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue

        new_val = _surrogate(new_seq)
        dE = new_val - cur_val
        if dE < 0.0 or rng.random() < math.exp(-dE / max(T, 1e-12)):
            cur_seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the time budget is stored in the global
    variable ``height_sequence_1`` and returned as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # reserve a small safety margin
    deadline = start_time + budget_s - 2.0

    # problem size (default 4000)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if its length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 0️⃣  Build an expressive seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n)]

    # 0.1  Beta families (α varies, β = α for symmetry)
    for α in np.arange(0.15, 2.05, 0.15):
        seeds.append(_beta_distribution(n, α, α))

    # 0.2  Gamma‑shape family (γ varies)
    for γ in (0.3, 0.4, 0.5, 0.6, 0.8, 1.0):
        seeds.append(_gamma_shape(n, γ))

    # 0.3  Some random Dirichlet draws with various concentrations
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # 0.4  Re‑use the previous best if available
    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣  Evaluate all seeds, keep the best one
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        # ensure proper normalisation (numerical safety)
        vec = np.maximum(0.0, vec)
        total = vec.sum()
        if total == 0.0:
            continue
        vec = vec / total
        val = _surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()

    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Exact exponentiated‑gradient polishing (sub‑gradient of max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Gradient polishing (exponentiated‑gradient) ---")
        polishing_deadline = deadline - 5.0   # keep a few seconds for later phases
        # we let the optimiser run as long as it can (it checks the deadline internally)
        seq_polished, val_polished = _exp_gradient_opt(
            best_seq,
            n,
            polishing_deadline,
            max_iters=5_000_000,
            eta0=5.0
        )
        print(f"[Polish] best surrogate = {val_polished:.6f}")
        if val_polished < best_val:
            best_val, best_seq = val_polished, seq_polished.copy()
            print("   >>> Updated global best (polishing)")

    # ------------------------------------------------------------------
    # 3️⃣  Simulated‑annealing local refinement (tiny random walks)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Simulated annealing refinement ---")
        anneal_deadline = deadline
        seq_anneal, val_anneal = _simulated_annealing(
            best_seq,
            n,
            anneal_deadline,
            init_temp=None,
            final_temp=1e-8,
            sigma=0.02
        )
        print(f"[Anneal] best surrogate = {val_anneal:.6f}")
        if val_anneal < best_val:
            best_val, best_seq = val_anneal, seq_anneal.copy()
            print("   >>> Updated global best (annealing)")

    # ------------------------------------------------------------------
    # 4️⃣  (Optional) A final short polishing pass – often squeezes a
    #     few more 1e‑5 improvements.
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        short_deadline = deadline
        seq_final, val_final = _exp_gradient_opt(
            best_seq,
            n,
            short_deadline,
            max_iters=800_000,
            eta0=2.5
        )
        print(f"[Final polish] surrogate = {val_final:.6f}")
        if val_final < best_val:
            best_val, best_seq = val_final, seq_final.copy()
            print("   >>> Updated global best (final polish)")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
