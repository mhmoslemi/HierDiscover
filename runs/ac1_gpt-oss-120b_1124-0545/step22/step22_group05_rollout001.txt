def _adam_opt(start_seq, n, deadline, max_iters=500_000, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    t = 0
    while t < max_iters and time.time() < deadline:
        t += 1
        # conv and k
        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active lag(s)
        idxs = np.where(conv == max_val)[0]
        k = idxs[np.random.randint(len(idxs))]
        # gradient
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        # Adam update on log_a
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # Softmax to get new a
        a = _softmax(log_a)
        # compute surrogate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # maybe periodic reinitialization of moments?

    return best_seq, best_val

"""
Algorithm:  Hybrid stochastic‑gradient + local search

1. Build a rich pool of deterministic and random “seed” vectors on the simplex
   (uniform, arcsine, triangular, beta‑shapes, geometric decay, linear decay,
   perturbations of the uniform distribution, and a warm‑start if present).

2. Keep the best seed according to the *surrogate* objective
      S(a) = 2·n·max(conv(a,a))      (a is normalised to sum = 1).

3. While time permits, draw many Dirichlet samples with a wide range of
   concentration parameters and keep any improvement.

4. Refine the current best vector with an Adam optimiser that works on the
   log‑softmax parametrisation.  The sub‑gradient of the max‑convolution is
   the sum of the gradients of *all* active lags (i.e. all lags attaining the
   current maximum).  Adam’s adaptive step‑size usually finds a lower
   maximum than plain exponentiated‑gradient.

5. Perform a cheap stochastic local walk on the simplex (move a tiny amount
   of mass from one coordinate to another).  Moves are accepted only if they
   improve the surrogate value.

6. Finally, try a symmetric version of the best vector
   (a ← (a+reverse(a))/2) – the objective is invariant under reversal and
   symmetry often reduces the peak of the autocorrelation.

7. Call ``evaluate_sequence`` on the final candidate and return it.

The whole routine respects the supplied ``budget_s`` (default 1000 s) and
stores the result in the global variable ``height_sequence_1`` for warm‑starts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search.
    Assumes sum(seq) == 1.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc`` (scalar)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _geometric(n: int, rate: float) -> np.ndarray:
    """Geometric‑decay distribution (rate∈(0,1))."""
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else _uniform(n)


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add small Gaussian noise, clip, and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------
def _active_lag_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return the sub‑gradient (in the primal space) of the max‑convolution.
    If several lags attain the maximum, their contributions are summed.
    """
    n = len(a)
    max_val = conv.max()
    active = np.where(conv == max_val)[0]
    g = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] += 2.0 * a[rev_idx]
    return g


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax (mirror‑descent) manifold
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              n: int,
              deadline: float,
              max_iters: int = 2_000_000,
              lr: float = 0.8,
              beta1: float = 0.9,
              beta2: float = 0.999,
              eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam on log‑probabilities.
    Returns the best sequence found (still normalised to sum = 1) and its
    surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                # keep on the simplex via softmax
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        g = _active_lag_gradient(a, conv)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        # bias correction
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # update in log‑space
        log_a = log_a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # occasional re‑normalisation to avoid numerical drift
        if it % 10000 == 0:
            a = a / a.sum()
            log_a = np.log(a + 1e-30)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight stochastic local refinement on the simplex
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Random walk: move a tiny fraction of mass from one index to another.
    Accept only improvements of the surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # propose a relative delta proportional to the mass at i
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val
