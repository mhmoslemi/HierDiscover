```
def _project_simplex(v):
    # projection onto simplex {x >=0, sum x = 1}
    n = len(v)
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho = max i where cond true
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
def _smooth_gradient_opt(start_seq, n, deadline, max_iters=500000, eta0=0.5, tau_start=0.5, tau_factor=1.001, tau_max=500.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    tau = tau_start
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)  # length L = 2n-1
        # smoothing
        max_c = np.max(conv)
        ex = np.exp(tau * (conv - max_c))
        w = ex / np.sum(ex)
        # gradient
        # compute cross-correlation a⋆w = conv(w, a_rev) [slice]
        grad = 2.0 * _conv_fft(w, a[::-1])[n-1:n-1+n]
        # step
        eta = eta0 / np.sqrt(it)
        a = a - eta * grad
        # project onto simplex
        a = _project_simplex(a)
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # increase tau slowly
        tau = min(tau_max, tau * tau_factor)
    return best_seq, best_val
```

The goal is to minimise  

\[
\text{score}(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\]

where `a` is a non‑negative vector (the “step‑function”) and `*` denotes the
ordinary linear convolution.  
The score is **scale invariant** (`a → c·a` leaves the value unchanged) and
the only constraints are `a_i ≥ 0` and `∑ a_i > 0.01`.  
Hence we can restrict ourselves to the simplex  

\[
\Delta_n=\{a\mid a_i\ge 0,\; \sum_i a_i = 1\}.
\]

On \(\Delta_n\) the objective simplifies to  

\[
J(a)=2\,n\;\max_k\;(a*a)_k .
\]

`J` is a non‑convex, non‑smooth function (the `max`).  
The algorithm below consists of three stages

1. **Rich seeding** – many deterministic and random probability vectors
   (uniform, arcsine, Beta‑shapes, Dirichlet draws, mixtures).  
   The best seed is kept as a starting point.
2. **Smoothed mirror‑descent** – we minimise a smooth
   approximation of the `max`  
   \[
   \tilde J_\tau (a)=\frac{2n}{\tau}\log\!\Bigl(\sum_k e^{\tau\,(a*a)_k}\Bigr).
   \]
   Its gradient is  
   \[
   \nabla\tilde J_\tau (a)=2\;
   \bigl(w * a^{\;\text{rev}}\bigr)_{n-1:n-1+n},
   \qquad  
   w_k =\frac{e^{\tau (a*a)_k}}{\sum_j e^{\tau (a*a)_j}} .
   \]
   A decreasing step size `η_t = η₀ / √t` and a projection onto the simplex
   (Euclidean projection) give a robust mirror‑descent scheme.
   The temperature `τ` is increased slowly (`τ←τ·τ_factor`) – a classic
   continuation method that first smooths the landscape and later focuses on
   the true `max`.
3. **Fine‑grained random walk** – tiny mass transfers between random
   coordinates that accept only improving moves.  This cleans up any remaining
   local irregularities.

All computations use FFT‑based full convolutions (`_conv_fft`) which run in
\(O(n\log n)\) time.  With `n=4000` the inner loop is extremely cheap, so we
can afford several hundred thousand optimisation steps well inside the 1000 s
budget.

The final vector is stored in the global variable `height_sequence_1` for
warm‑starting subsequent runs.

```python
"""
Propose a step‑function (non‑negative list of heights) that minimises
``evaluate_sequence``.  The method builds a large, diverse pool of probability
vectors, keeps the best seed, refines it with a smoothed mirror‑descent
optimiser (continuation on the temperature of a soft‑max of the convolution),
and finishes with a cheap stochastic local walk.  It respects a 1000 s
budget and stores the result in ``height_sequence_1`` for warm‑starts.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, soft‑max, simplex projection)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
        2·n·max(conv(seq, seq))   (the sum of seq is assumed to be 1)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators – all return a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) pdf on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet( conc·1⃗ ) sample."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _geometric(n: int, rate: float) -> np.ndarray:
    """Geometric‑decay distribution (rate∈(0,1))."""
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    a_sum = a.sum()
    if a_sum == 0.0:
        return _uniform(n)
    return a / a_sum


def _linear_decreasing(n: int) -> np.ndarray:
    """Deterministic linear decay from left to right."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    return a / a.sum()


def _perturb(seq: np.ndarray, rng: np.random.Generator,
             sigma: float = 0.002) -> np.ndarray:
    """Add tiny Gaussian noise and re‑normalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s == 0.0:
        return seq.copy()
    return noisy / s


def _mixture_beta(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random mixture of two Beta PDFs."""
    a1, b1 = rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0)
    a2, b2 = rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0)
    w = rng.random()
    v1 = _beta_distribution(n, a1, b1)
    v2 = _beta_distribution(n, a2, b2)
    mix = w * v1 + (1.0 - w) * v2
    mix = np.maximum(mix, 0.0)
    return mix / mix.sum()


# ----------------------------------------------------------------------
# Smoothed mirror‑descent optimisation
# ----------------------------------------------------------------------
def _smooth_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 800_000,
                         eta0: float = 0.6,
                         tau_start: float = 0.3,
                         tau_factor: float = 1.001,
                         tau_max: float = 250.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective

        \tilde J_\tau(a)=2n/τ·log(∑_k exp(τ·(a*a)_k))

    ``start_seq`` must already sum to 1.
    Returns the best vector found and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    tau = tau_start

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exact convolution a*a
        conv = _conv_fft(a, a)                     # length 2n‑1

        # soft‑max over the lag values (stable)
        max_c = np.max(conv)
        exp_weights = np.exp(tau * (conv - max_c))
        w = exp_weights / exp_weights.sum()        # probabilities, sum=1

        # gradient = 2 * (w * a_rev)  (cross‑correlation)
        grad = 2.0 * _conv_fft(w, a[::-1])[n - 1:n - 1 + n]

        # decreasing step size
        eta = eta0 / math.sqrt(it)

        # additive update + projection onto the simplex
        a = a - eta * grad
        a = _project_simplex(a)

        # record improvement
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # slowly increase the temperature – continuation scheme
        if tau < tau_max:
            tau = min(tau_max, tau * tau_factor)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 250_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Random walk on the simplex: move a tiny amount of mass from one index
    to another and keep it only if the surrogate improves.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises

    stores the best vector in the global ``height_sequence_1`` variable and
    returns it as a plain list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # problem size (default 4000 – can be overridden)
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # --------------------------------------------------------------
    # warm‑start from a previous solution (if compatible)
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev = arr / arr.sum()
        except Exception:
            prev = None
    else:
        prev = None

    # --------------------------------------------------------------
    # 0️⃣ Build a large, diverse pool of seeds (all sum to 1)
    # --------------------------------------------------------------
    seeds: list[np.ndarray] = [_uniform(n)]

    if prev is not None:
        seeds.append(prev)

    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))
    seeds.append(_linear_decreasing(n))

    # a handful of deterministic beta shapes
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 5.0)]:
        seeds.append(_beta_distribution(n, a, b))

    # geometric decays (rate close to 1)
    for rate in (0.9999, 0.9995, 0.999, 0.998):
        seeds.append(_geometric(n, rate))

    # small perturbations of the uniform vector
    for sigma in (0.01, 0.005):
        seeds.append(_perturb(_uniform(n), rng, sigma=sigma))

    # --------------------------------------------------------------
    # systematic beta-sweep (grid)
    # --------------------------------------------------------------
    alphas = np.logspace(-0.7, 0.7, 12)   # approx 0.2 … 5
    betas = np.logspace(-0.7, 0.7, 12)

            seeds.append(_beta_distribution(n, a, b))

    # --------------------------------------------------------------
    # random Dirichlet draws (various concentrations)
    # --------------------------------------------------------------
    conc_options = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    draws = 0
    draw_limit = 6000
    while draws < draw_limit and time.time() < deadline:
        conc = rng.choice(conc_options)
        seeds.append(_random_dirichlet(n, conc, rng))
        draws += 1

    # --------------------------------------------------------------
    # random mixtures of two beta PDFs – adds extra variety
    # --------------------------------------------------------------
    for _ in range(400):
        if time.time() > deadline:
            break
        seeds.append(_mixture_beta(n, rng))

    # --------------------------------------------------------------
    # 1️⃣ Evaluate all seeds and keep the best surrogate
    # --------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    print("\n--- Initial seed evaluation (surrogate) ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
        if idx % 50 == 0:
            print(f" seed #{idx+1:04d}: surrogate = {val:.6f}")

    print(f"[Init] best surrogate = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Smoothed mirror‑descent refinement (core optimisation)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Smoothed mirror‑descent optimisation ---")
        remaining = deadline - time.time()
        # estimate how many iterations we can afford: roughly 0.00004 s per iter
        # (conservative); cap at a few million.
        max_iters = int(min(4_000_000,
                            max(200_000, remaining / 0.00004)))
        seq_opt, val_opt = _smooth_gradient_opt(best_seq,
                                                n,
                                                deadline,
                                                max_iters=max_iters,
                                                eta0=0.6,
                                                tau_start=0.3,
                                                tau_factor=1.001,
                                                tau_max=250.0)
        print(f"[MGD] best surrogate after MGD = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print("   >>> Global best updated (MGD)")

    # --------------------------------------------------------------
    # 3️⃣ Final cheap random local refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_local, val_local = _local_random_search(best_seq,
                                                    n,
                                                    deadline,
                                                    attempts=250_000,
                                                    sigma=0.02)
        print(f"[Local] best surrogate after local walk = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Global best updated (local)")

    # --------------------------------------------------------------
    # 4️⃣ Official evaluation with the provided ``evaluate_sequence``
    # --------------------------------------------------------------
    true_val = None

        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
```
