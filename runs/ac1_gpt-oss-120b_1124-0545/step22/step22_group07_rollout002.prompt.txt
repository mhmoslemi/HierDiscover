import math

# ----------------------------------------------------------------------
# Basic linear convolution via FFT (real, length = n+m-1)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Surrogate objective (ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – always return an ℓ₁‑normalised vector
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))

    a = np.interp(xs, points, vals)

    s = a.sum()

# Smoothed Adam optimiser (log‑softmax, log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     deadline: float,
                     beta_schedule=None,
                     iters_per_beta: int = 40_000) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of max(conv) using Adam.
    The objective is 2·n·M_β where M_β is the log‑sum‑exp smooth max.
    """
    if beta_schedule is None:
        # start very smooth, finish extremely sharp
        beta_schedule = [10, 50, 200, 800, 3_200, 12_800, 51_200, 200_000]

    n = len(start_seq)
    log_a = np.log(start_seq + 1e-30)          # keep in log‑softmax space

    # Adam state
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.12                                 # initial learning‑rate factor
    t = 0

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() >= deadline:
            break
        # run a fixed number of inner iterations for this beta
        for _ in range(iters_per_beta):
            if time.time() >= deadline:
                break
            t += 1

            # probabilities from log‑softmax
            a = np.exp(log_a - np.max(log_a))
            a /= a.sum()

            # full convolution a*a
            conv = _conv_fft(a, a)

            # smooth max via log‑sum‑exp
            shift = np.max(conv)                     # for numerical stability
            exp_beta = np.exp(beta * (conv - shift))
            sum_exp = np.sum(exp_beta)
            w = exp_beta / sum_exp                    # weights (soft max)

            # gradient w.r.t a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # gradient w.r.t log‑softmax parameters (chain rule)
            grad_log = a * (grad_a - np.dot(grad_a, a))

            # Adam update (bias‑corrected)
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            lr = lr0 / (np.sqrt(v_hat) + eps)

            log_a -= lr * m_hat
            # keep logs bounded to avoid overflow/underflow
            log_a = np.clip(log_a, -30.0, 30.0)

            # keep track of the best *exact* objective seen so far
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val

# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val
