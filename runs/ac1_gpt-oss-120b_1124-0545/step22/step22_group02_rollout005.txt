    best_seq = a.copy()

    no_improve_iters = 0
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        conv = conv_fft(a,a)
        max_val = conv.max()
        # active set near max
        tol = 1e-12

        g = np.zeros_like(a)

            rev_idx = kk - idx
            mask = (rev_idx>=0)&(rev_idx<n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size
        # momentum
        momentum = beta * momentum + (1-beta) * g
        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * momentum
        a = softmax(log_a)
        cur_val = objective(a)

            no_improve_iters = 0

            # restart with larger step size and reset momentum
            log_a = np.log(best_seq + 1e-30)
            a = best_seq.copy()
            momentum = np.zeros_like(a)
            no_improve_iters = 0

"""
Advanced optimizer for the “step‑function” challenge.

The goal is to minimise

    2·n·max(conv(a,a)) / (sum(a)^2)

which, after normalising the vector to ℓ₁‑norm = 1, reduces to

    2·n·max(conv(a,a))

where *conv* is the full linear convolution.
The algorithm proceeds as follows:

1. Build a compact library of analytic seed windows (DPSS, Kaiser, Gaussian,
   triangular, sine, …) and a few random Dirichlet draws.
2. Evaluate every seed with the exact objective (the surrogate coincides with
   ``evaluate_sequence`` after normalisation) and keep the best few.
3. For each top seed run a long **exponentiated‑gradient (EG)** optimisation,
   computing the exact sub‑gradient of the max‑convolution.  The EG loop uses
   momentum (Adam‑style) and automatically restarts with a larger step size
   when progress stalls.
4. A short **flattening** phase minimises the ℓ₂‑variance of the convolution
   (∑(c‑mean)²) which drives the whole autocorrelation towards a constant and
   usually lowers the max entry further.
5. A final **simulated‑annealing** polish (Gaussian perturbations with a
   linear cooling schedule) tries to escape tiny local minima.
6. The best vector seen is stored in the global ``height_sequence_1`` and
   returned as a plain Python list.

All stages respect the wall‑clock budget (default 1000 s).  The implementation
uses only NumPy and, if available, SciPy for a DPSS window; it runs comfortably
under a single CPU core but can be given more time for deeper refinement.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (only for DPSS windows)
# ----------------------------------------------------------------------
try:
    from scipy.signal import windows as _sigwin
except Exception:  # pragma: no cover
    _sigwin = None


# ----------------------------------------------------------------------
# Basic numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (real‑valued inputs)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Exact objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Returns 2·n·max(conv(seq,seq)) for a ℓ₁‑normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy is unavailable."""
    if _sigwin is None:
        return _uniform(n)
    from scipy.special import i0 as _i0  # local import; safe even if SciPy unavailable
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires SciPy."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max‑convolution, multi‑active)
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5_000_000,
                            eta0: float = 8.0,
                            tol: float = 1e-12,
                            restart_patience: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Uses a momentum term (Adam‑style) and automatically restarts with a fresh
    step size when no improvement is seen for *restart_patience* iterations.
    Returns the best encountered sequence and its objective value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)            # stay in log‑softmax space
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective(a)

    it_without_improve = 0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # indices whose value is within *tol* relative to the maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over the active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        if active.size:
            for kk in active:
                rev_idx = kk - idx
                mask = (rev_idx >= 0) & (rev_idx < n)
                g[mask] += a[rev_idx[mask]]
            g = 2.0 * g / active.size

        # Adam‑like momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)        # classic 1/√t schedule

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            it_without_improve = 0
        else:
            it_without_improve += 1

        # restart if we are stuck for a long time
        if it_without_improve >= restart_patience:
            # go back to the current best and drop momentum / step size
            a = best_seq.copy()
            log_a = np.log(a + 1e-30)
            momentum = np.zeros_like(a)
            it_without_improve = 0

    return best_seq, best_val


# ----------------------------------------------------------------------
# Flattening optimiser – drives the convolution towards a constant
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_500_000,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the quadratic loss Σ_k (conv_k−mean)².
    The aim is to make the full convolution as flat as possible,
    which in practice reduces its maximum.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective(a)

    denom = float(2 * n - 1)               # length of the full convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom      # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)     # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # projection onto the simplex via log‑space (mirror‑descent)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement – occasional uphill moves
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 150_000,
                         start_sigma: float = 0.0015,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Classic SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before *deadline*.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – orchestrates seeding, optimisation and polishing
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, stores the best found sequence in the
    global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between expressivity and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # 0. Warm start – previous best (if any) or a good analytic seed
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape != (n,):
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    if prev_arr is not None:
        base_seq = prev_arr / prev_arr.sum()
    else:
        # DPSS with a modest bandwidth is a strong deterministic seed
        base_seq = _dpss_window(n, nw=3.0)

    # ------------------------------------------------------------------
    # 1. Build a modest seed library (analytic families + random draws)
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_blackman_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_cosine_window(n))

    # a few Kaiser windows
    for beta in (0.0, 2.0, 4.0, 6.0, 8.0):
        seeds.append(_kaiser_window(n, beta))

    # Gaussian windows with several widths
    for sigma_frac in (0.07, 0.14, 0.21, 0.28):
        seeds.append(_gaussian(n, sigma_frac))

    # DPSS windows – different time–bandwidth products
    if _sigwin is not None:
        for nw in (2.5, 3.0, 3.5, 4.0):
            seeds.append(_dpss_window(n, nw))

    # Random Dirichlet draws – widely varying concentration
    for conc in (0.001, 0.01, 0.1, 0.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Add a few perturbed copies of the current best (to break possible symmetry)
    seeds.append(base_seq)
    for _ in range(4):
        pert = base_seq + 0.003 * rng.normal(size=n)
        pert = np.maximum(pert, 0.0)
        s = pert.sum()
        if s > 0.0:
            seeds.append(pert / s)

    # ------------------------------------------------------------------
    # 2. Quick evaluation of all seeds, keep the top few
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seeding] best surrogate values from the library:")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"   #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best from the very best seed
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop – alternate EG, flattening and SA
    # ------------------------------------------------------------------
    # The loop will consume the remaining budget.  Each stage receives a
    # fraction of the time that is still left.
    while True:
        time_left = deadline - time.time()
        if time_left < 8.0:                     # not enough time for another stage
            break

        # ---- a) Exponentiated‑gradient (long run) ----
        sub_deadline = time.time() + 0.55 * time_left
        seq_eg, val_eg = _exp_gradient_opt_multi(global_best_seq.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=8_000_000,
                                                eta0=9.0,
                                                tol=1e-12,
                                                restart_patience=250_000)

        if val_eg < global_best_val:
            global_best_seq, global_best_val = seq_eg, val_eg
            print(f"[EG] improved to {global_best_val:.6f}")

        # ---- b) Flattening stage (if time permits) ----
        time_left = deadline - time.time()
        if time_left < 6.0:
            break
        sub_deadline = time.time() + 0.30 * time_left
        seq_flat, val_flat = _flatten_gradient_opt(global_best_seq.copy(),
                                                   n,
                                                   sub_deadline,
                                                   max_iters=2_000_000,
                                                   eta0=5.5)

        if val_flat < global_best_val:
            global_best_seq, global_best_val = seq_flat, val_flat
            print(f"[Flatten] improved to {global_best_val:.6f}")

        # ---- c) Short simulated‑annealing polish ----
        time_left = deadline - time.time()
        if time_left < 4.0:
            break
        sub_deadline = time.time() + 0.40 * time_left
        seq_sa, val_sa = _simulated_annealing(global_best_seq.copy(),
                                              n,
                                              sub_deadline,
                                              rng,
                                              max_attempts=90_000,
                                              start_sigma=0.001,
                                              T_start=0.012,
                                              T_end=1e-9)

        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
