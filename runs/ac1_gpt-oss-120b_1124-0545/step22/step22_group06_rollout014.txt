def upsample(seq, new_len):
    old_len = len(seq)
    # compute cumulative sum positions (midpoints)
    x_old = np.linspace(0, 1, old_len, endpoint=False) + 0.5/old_len
    x_new = np.linspace(0, 1, new_len, endpoint=False) + 0.5/new_len
    # Use np.interp for linear interpolation
    values = np.interp(x_new, x_old, seq)
    # ensure sum =1
    values = values / values.sum()
    return values
```

def _conv_fft(x, y):
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

    v = np.asarray(v, dtype=np.float64)
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0:
        return np.full_like(v, 1.0/len(v))
    return e / s

def _adam_smoothing(start_seq, n, deadline):
    # start_seq assumed sum=1
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    # schedule
    beta_schedule = [1.0, 5.0, 20.0, 80.0, 200.0, 500.0, 1000.0, 2000.0,
                    5000.0, 10000.0, 20000.0, 50000.0, 100000.0, 200000.0,
                    500000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0, 10_000_000.0]
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    for beta in beta_schedule:
        if time.time() > deadline:
            break
        # allocate inner loops. Use a base of 3000, increase for low betas
        # Determine max_iter based on remaining time.
        max_iter = 3000
        # we may increase for later betas
        if beta > 1000:
            max_iter = 2000
        for _ in range(max_iter):
            if time.time() > deadline:
                break
            a = _softmax(logit)
            conv = _conv_fft(a, a)
            max_c = np.max(conv)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / np.sum(exp_beta)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)

            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_logit
            v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)

                    best_seq = a.copy()

def _exp_gradient_opt(start_seq, n, deadline, max_iters=200000, eta0=8.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        # update log_a for involved indices only

def _local_random_search(start_seq, n, deadline, attempts=300000, sigma=0.03):
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)
    cur_seq = best_seq.copy()
    cur_val = best_val
    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i:
            # avoid i==j, ensure j different
            j = (j + 1) % n
        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0:
            continue
        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            cur_seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_seq = new_seq.copy()
                best_val = new_val
    return best_seq, best_val

"""
Propose a candidate step‑function.

The search consists of four phases, each limited by the remaining time:

1. **Diverse seeding** – generate a pool of probability vectors
   (uniform, arcsine‑like, triangular, Dirichlet draws and the previous best
   if available) and keep the one with the smallest exact surrogate
   ``2·n·max(conv)``.

2. **Adam on a smooth max** – optimise the log‑softmax parameters of the
   surrogate log‑sum‑exp of the convolution.  A long schedule of increasing
   smoothing parameters (β) and an Adam optimiser are used.  The best exact
   surrogate seen is retained.

3. **Two deterministic refinements**
   * *L‑BFGS‑B* on the same smooth objective (large β) – a quasi‑Newton step
     that usually improves the result further.
   * *Exact exponentiated‑gradient* (sub‑gradient of the true max) – a
     mirror‑descent step that directly reduces the true ``max(conv)``.

4. **Intensive random local walk** – a simple stochastic move that transfers
   a small amount of mass between two coordinates; improvements are kept.
   The walk runs until the time budget expires.

All intermediate results are evaluated with the exact surrogate
``2·n·max(conv)``; the final sequence is passed to ``evaluate_sequence`` and
stored globally as ``height_sequence_1`` for future runs.

The implementation works with any length ``n`` (default 4000) and respects
the 1000 s budget.
"""

import time
import math
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Global variables used by the L‑BFGS callbacks (no closures)
# ----------------------------------------------------------------------
_lbfgs_n = None
_lbfgs_beta = None


# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    v = np.asarray(v, dtype=np.float64)
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq))."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector that sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2, dtype=np.float64)
        right = left[:-1][::-1]
    a = np.concatenate([left, right])
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp surrogate of 2·n·max(conv).
    Returns the best sequence (probability vector) and its exact surrogate.
    """
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # schedule of smoothing parameters (β)
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1_000.0,
        2_000.0, 5_000.0,
        10_000.0, 20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # number of inner iterations – modestly larger for early β
        max_iter_beta = 4000 if beta < 1000 else 2500
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of the weighted sum of convolutions w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = 3·n‑2
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.12
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep exact best occasionally
            if it % 600 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the active (max) index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for the active index k (only a slice is non‑zero)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        # mirror‑descent step on the log‑space (only the active coordinates)
        eta = eta0 / math.sqrt(it)
        log_a[idx_i] -= eta * (2.0 * a[rev_idx])

        # project back to the simplex via softmax
        a = _softmax(log_a)

        if it % 400 == 0:
            cur_val = _objective_surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 300_000,
                         sigma: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic walk on the simplex: repeatedly move a small random
    amount from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    cur_seq = best_seq.copy()
    cur_val = best_val

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j = (j + 1) % n
        if i == j:
            continue

        delta = sigma * cur_seq[i] * rng.random()
        if delta <= 0.0:
            continue

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue

        new_val = _objective_surrogate(new_seq)
        if new_val < cur_val:
            cur_seq = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_seq = new_seq.copy()
                best_val = new_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# L‑BFGS‑B optimisation of the smooth surrogate
# ----------------------------------------------------------------------


def _lbfgs_val_and_grad(logit: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Smooth surrogate based on log‑sum‑exp with a large β.
    Returns (value, gradient w.r.t. logit).
    """
    n = _lbfgs_n
    beta = _lbfgs_beta

    a = _softmax(logit)
    conv = _conv_fft(a, a)
    max_c = np.max(conv)

    # log‑sum‑exp
    exp_term = np.exp(beta * (conv - max_c))
    Z = np.sum(exp_term)
    val = 2.0 * n * (max_c + np.log(Z) / beta)

    # gradient of the weighted sum
    w = exp_term / Z
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)
    grad_weighted = 2.0 * full[n - 1:n - 1 + n]
    grad_a = grad_weighted * (2.0 * n)

    dot = np.dot(grad_a, a)
    grad_logit = a * (grad_a - dot)
    return val, grad_logit


def _lbfgs_val(logit: np.ndarray) -> float:
    val, _ = _lbfgs_val_and_grad(logit)
    return val


def _lbfgs_grad(logit: np.ndarray) -> np.ndarray:
    _, grad = _lbfgs_val_and_grad(logit)
    return grad


def _lbfgs_opt(start_seq: np.ndarray,
               n: int,
               deadline: float,
               maxfun: int = 200,
               beta: float = 5e5) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the smooth surrogate.
    ``maxfun`` is limited (and therefore the running time) by the caller.
    """
    global _lbfgs_n, _lbfgs_beta
    _lbfgs_n = n
    _lbfgs_beta = beta

    logit0 = np.log(start_seq + 1e-30)

    res = scipy.optimize.minimize(_lbfgs_val,
                                  logit0,
                                  method='L-BFGS-B',
                                  jac=_lbfgs_grad,
                                  options={'maxfun': maxfun, 'disp': False})

    final_logit = res.x
    final_seq = _softmax(final_logit)
    final_val = _objective_surrogate(final_seq)
    return final_seq, final_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The best sequence found within the time budget is stored in the global
    variable ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # problem size – can be overridden via kwargs
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # Load any previously stored solution (if its length matches)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_vec = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and np.sum(arr) > 0.0:
                prev_vec = arr / np.sum(arr)   # normalise to sum = 1
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 0️⃣  Build a diverse seed pool (all vectors sum to 1)
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _arcsine(n), _triangular(n)]
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))
    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 1️⃣  Initial evaluation of seeds, keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / np.sum(vec)
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Adam smoothing (smooth max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.30 * remaining
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 3️⃣  L‑BFGS‑B refinement on the smooth surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        lbfgs_deadline = time.time() + 0.30 * remaining
        # crude estimate: one function evaluation ≈ 1 ms → set maxfun accordingly
        maxfun = max(20, int(0.8 * (lbfgs_deadline - time.time()) / 0.001))
        seq_lbfgs, val_lbfgs = _lbfgs_opt(best_seq, n, lbfgs_deadline,
                                          maxfun=maxfun, beta=5e5)
        print(f"[L‑BFGS] exact surrogate = {val_lbfgs:.6f}")
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs.copy()
            print("   >>> Updated global best (L‑BFGS)")

    # ------------------------------------------------------------------
    # 4️⃣  Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        eg_deadline = time.time() + 0.30 * remaining
        # let the optimiser decide how many iterations fit into the time slice
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n, eg_deadline,
                                          max_iters=300_000, eta0=8.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣  Final tiny random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=300_000,
                                                sigma=0.03)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
