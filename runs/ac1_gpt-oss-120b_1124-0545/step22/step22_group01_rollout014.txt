def upsample(seq, new_n):
    old_n = len(seq)
    xs_old = np.linspace(0, 1, old_n)
    xs_new = np.linspace(0, 1, new_n)
    seq_new = np.interp(xs_new, xs_old, seq)
    seq_new = np.maximum(seq_new, 0.0)
    seq_new = seq_new / seq_new.sum()
    return seq_new
```

def _smooth_exp_gradient_opt(start_seq, n, deadline, tau_schedule, eta0=6.0):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    
    it = 0
    for tau in tau_schedule:
        while True:
            if time.time() > deadline:
                break
            it += 1
            # compute conv a*a
            conv = _conv_fft(a, a)
            max_b = float(np.max(conv))
            cur_val = 2.0 * n * max_b
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
            # compute smoothing gradient
            w = np.exp(conv / tau)  # element-wise
            w_sum = np.sum(w)
            if w_sum == 0.0:
                break
            w_norm = w / w_sum
            # gradient via convolution
            # compute conv_rev = conv of w_norm reversed with a
            grad_full = _conv_fft(w_norm[::-1], a)
            start = len(w_norm) - 1
            grad = 2.0 * grad_full[start:start + n]  # shape (n,)
            # log-space gradient update
            dot = np.dot(grad, a)
            grad_log = a * (grad - dot)
            eta = eta0 / math.sqrt(it)
            log_a = log_a - eta * grad_log
            a = _softmax(log_a)
    return best_seq, best_val
```

"""
Improved optimiser for the step‑function “max‑convolution” problem.

The objective (after ℓ₁‑normalisation) is
        2·n·max(conv(a,a))      with   a_i ≥ 0 ,   Σ a_i = 1 .

The algorithm proceeds in three stages

1️⃣  **Seed generation** – a large, diverse pool of analytic families
    (uniform, triangular, sine‑window, Gaussian, exponential, power‑law,
    beta‑shapes) together with many Dirichlet draws (different concentration
    parameters) and a few noisy copies of a previous solution (if any) is
    created.  All seeds are ℓ₁‑normalised and the best *K* (default 12) are
    retained.

2️⃣  **Smooth exponentiated‑gradient refinement** – each retained seed is
    optimised with an exponentiated‑gradient mirror‑descent that minimises a
    smooth approximation of the max‑convolution:
          F_τ(a) = τ·log Σ_k exp(conv_k/τ) .
    A schedule of decreasing temperatures τ (0.01 … 0.0002) is used; the
    gradient of F_τ is obtained analytically via an extra convolution, which
    can be computed efficiently with FFTs.  Updates are performed in the
    log‑softmax (mirror‑descent) space, guaranteeing feasibility at every
    step.  The best vector seen for the seed is kept.

3️⃣  **Final polishing** – the current global best is polished again with the
    same smooth optimiser (finest τ) and finally with a few seconds of the
    exact‑sub‑gradient optimiser (the method from the previous state‑of‑the‑art
    code).  Random perturbations are tried while time permits to escape possible
    local minima.

All heavy work uses FFT‑based convolution (`_conv_fft`) which makes each
iteration O(n log n).  The routine respects the 1000 s time budget,
stores the final sequence in the global variable ``height_sequence_1`` for
possible continuation runs and returns the list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc). Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Exact sub‑gradient optimiser (same as previous state‑of‑the‑art)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    """Mirror‑descent on the exact max‑convolution (log‑space)."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t a
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        g[idx_i] = 2.0 * a[rev_idx]
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑gradient optimiser (log‑softmax, temperature schedule)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            tau_schedule,
                            eta0: float = 6.0):
    """
    Mirror‑descent using the smoothed objective
        F_τ(a) = τ·log Σ_k exp(conv_k/τ) .
    Operates in log‑softmax space and works for any decreasing
    temperature schedule `tau_schedule` (list of τ values).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    for tau in tau_schedule:
        # allocate a modest share of the remaining time for each τ
        sub_deadline = time.time() + (deadline - time.time()) / (len(tau_schedule) - tau_schedule.index(tau))
        while time.time() < sub_deadline:
            if time.time() > deadline:
                break
            it += 1

            # full convolution a*a
            conv = _conv_fft(a, a)
            max_b = float(np.max(conv))
            cur_val = 2.0 * n * max_b
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

            # ---------- smooth gradient ----------
            # stable soft‑max over conv entries
            c_max = np.max(conv)
            w = np.exp((conv - c_max) / tau)
            w_sum = np.sum(w)
            if w_sum == 0.0:
                # should not happen, but break to avoid NaNs
                break
            w_norm = w / w_sum

            # gradient = 2 * (w_norm reversed) * a   (sliding‑dot product)
            grad_full = _conv_fft(w_norm[::-1], a)
            start = len(w_norm) - 1           # equals 2*n-2
            grad = 2.0 * grad_full[start:start + n]

            # mirror‑descent update in log‑space
            dot = np.dot(grad, a)
            grad_log = a * (grad - dot)
            eta = eta0 / math.sqrt(it)
            log_a = log_a - eta * grad_log
            a = _softmax(log_a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple up‑sampling of a previous solution (linear interpolation)
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    xs_old = np.linspace(0.0, 1.0, old_n)
    xs_new = np.linspace(0.0, 1.0, new_n)
    seq_new = np.interp(xs_new, xs_old, seq)
    seq_new = np.maximum(seq_new, 0.0)
    s = seq_new.sum()
    if s == 0.0:
        return _uniform(new_n)
    return seq_new / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine builds a diverse pool of analytic seeds,
    keeps the best few, refines each with a temperature‑annealed exponentiated‑
    gradient optimiser and finishes with a short exact‑subgradient polish.
    The best sequence is stored globally as ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the final call to evaluate_sequence
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Problem size – a larger n reduces discretisation error and often yields
    # a lower constant (target ≈ 1.5).  6000 is a good compromise.
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Load a previous solution for warm‑starting (if available)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if isinstance(prev, list) and len(prev) > 1:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1:
                prev_arr = _upsample(arr, n)
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a large pool of analytical & random seeds
    # ------------------------------------------------------------------
    seeds = []
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows – dense sweep over gamma
    for gamma in np.linspace(0.2, 4.0, 30):
        seeds.append(_sine_window(n, gamma))

    # Gaussian families
    for sigma_frac in np.linspace(0.05, 0.30, 20):
        seeds.append(_gaussian(n, sigma_frac))
    for mu_frac in np.linspace(0.15, 0.85, 10):
        for sigma_frac in np.linspace(0.05, 0.25, 8):
            seeds.append(_gaussian_shift(n, sigma_frac, mu_frac))

    # exponential decays
    for rate in np.logspace(-5, 0.4, 40):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑law
    for p in np.linspace(0.0, 6.0, 25):
        seeds.append(_power(n, p))

    # beta‑shapes
    alphas = np.geomspace(0.1, 10.0, 10)
    betas = np.geomspace(0.1, 10.0, 10)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # many Dirichlet draws with various concentrations
    conc_vals = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for conc in conc_vals:
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution + a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            noisy = prev_arr + 0.003 * rng.normal(size=n)
            noisy = np.maximum(noisy, 0.0)
            s = noisy.sum()
            if s > 0:
                seeds.append(noisy / s)

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed with the smooth optimiser
    # ------------------------------------------------------------------
    tau_schedule = [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]

    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly equal time for each remaining seed (minimum 12 s)
        share = max(12.0, remaining * 0.9 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # a) deterministic descending order (cheap heuristic)
        seq_desc = np.sort(seed_vec)[::-1]

        # b) smooth optimisation from original seed
        seq_opt1, val_opt1 = _smooth_exp_gradient_opt(seed_vec, n, sub_deadline,
                                                      tau_schedule, eta0=6.0)

        # c) smooth optimisation from descending variant (if time permits)
        if time.time() < sub_deadline:
            seq_opt2, val_opt2 = _smooth_exp_gradient_opt(seq_desc, n, sub_deadline,
                                                          tau_schedule, eta0=6.0)
        else:
            seq_opt2, val_opt2 = None, np.inf

        # pick the better of the two runs
        if val_opt2 is not None and val_opt2 < val_opt1:
            local_best_seq, local_best_val = seq_opt2, val_opt2
        else:
            local_best_seq, local_best_val = seq_opt1, val_opt1

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq, global_best_val = local_best_seq, local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing (smooth optimiser + exact sub‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final polishing (smooth optimiser) ---")
        final_seq, final_val = _smooth_exp_gradient_opt(
            global_best_seq, n, deadline,
            [0.005, 0.002, 0.001, 0.0005, 0.0002],
            eta0=6.0
        )
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Polish] improved to {global_best_val:.6f}")

    # short exact‑subgradient polish (≈10 s)
    if time.time() < deadline - 5.0:
        print("\n--- Exact sub‑gradient polishing (≈10 s) ---")
        sub_deadline = time.time() + 10.0
        seq_es, val_es = _exp_gradient_opt(
            global_best_seq, n, sub_deadline, max_iters=500_000, eta0=6.0
        )
        if val_es < global_best_val:
            global_best_seq, global_best_val = seq_es, val_es
            print(f"[Exact polish] improved to {global_best_val:.6f}")

    # 6. Random‑perturbation escapes while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:

            perturb, n, sub_deadline,
            [0.002, 0.001, 0.0005, 0.0002],

        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
