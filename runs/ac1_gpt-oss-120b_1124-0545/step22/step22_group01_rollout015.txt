"""
Improved step‑function optimiser.

Goal
----
Search a non‑negative sequence `a` (ℓ₁‑normalised, i.e. sum(a)=1) that minimises

    2·n·max(conv(a,a))

where `conv` is the linear convolution.  The true objective used by the
grader is `evaluate_sequence(a) = 2·n·max(conv)/sum(a)**2`; with the
ℓ₁‑normalisation it reduces to the surrogate shown above.

Approach
--------
* Use a *smaller* problem size (n≈2000) – this cuts the cost of each FFT
  dramatically and lets us perform many more optimisation steps in the
  1000 s budget.
* Create a rich pool of analytic seeds (uniform, linear decay, polynomial
  decay, shifted exponentials, logistic, power‑law, β‑shapes and many
  Dirichlet draws).  All seeds are ℓ₁‑normalised.
* Keep the K best seeds (according to the cheap surrogate) and refine each
  one independently:
    – short permutation hill‑climb
    – exponentiated‑gradient (mirror‑descent) on the exact sub‑gradient
      of the active convolution entry.
* After the per‑seed phase we *polish* the current global best for the
  remaining time.
* Finally we try simple convex combinations of the best seed with the other
  top seeds and a few random perturbations – this often escapes shallow
  local minima.

All heavy work (convolution, gradient) uses an FFT implementation.
The routine stores the final vector in the global variable
``height_sequence_1`` so that a later run can continue from this point.

The whole routine respects the 1000 s budget – a safety margin of a few
seconds is kept for the final call to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decay(n: int) -> np.ndarray:
    """Strictly decreasing linear ramp: a_i ∝ (n-i)."""
    a = np.arange(n, 0, -1, dtype=np.float64)  # n, n-1, …, 1
    return a / a.sum()


def _poly_decay(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (1 - i/(n-1))**alpha   (alpha≥0)."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - i / (n - 1)) ** alpha
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _shifted_exp(n: int, rate: float, offset_frac: float) -> np.ndarray:
    """
    a_i ∝ exp(-rate * (i - offset)) for i ≥ offset, otherwise 0.
    offset = int(offset_frac * n)
    """
    offset = int(offset_frac * n)
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i)
    mask = i >= offset
    a[mask] = np.exp(-rate * (i[mask] - offset))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic(n: int, scale: float) -> np.ndarray:
    """a_i ∝ 1/(1 + exp((i-mid)/scale)).  monotone decreasing S‑shape."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp((i - mid) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc).  Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exponentiated‑gradient (mirror‑descent) optimiser with exact sub‑gradient
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (guaranteed feasibility).
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    # log‑parameters (softmax will enforce sum=1 and non‑negativity)
    log_a = np.log(a + 1e-30)

    n = len(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))                     # active convolution index

        # --- exact sub‑gradient of max(conv) w.r.t a ---
        g = np.zeros_like(a)

        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i

        # off‑diagonal entries contribute twice
        g[idx_i] = 2.0 * a[rev_idx]

        # diagonal (i == rev_idx) must be counted only once
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        # mirror‑descent update in log‑space
        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)               # <g, a>
        grad_log = a * (g - dot)         # ∇_log objective
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill‑climbing (simulated annealing)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Local search that swaps two entries or adds a tiny Gaussian perturbation.
    Accepts worsening moves according to a simple annealing schedule.
    Returns the best sequence observed.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:                     # swap move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:                                         # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            # Metropolis acceptance
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds a large pool of analytical seeds,
    keeps the best few, refines each with permutation hill‑climbing and an
    exponentiated‑gradient mirror descent, and finally spends the remaining time
    polishing the global best.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin for final eval

    # ------------------------------------------------------------------
    # Problem size – 2000 is a good speed / expressivity compromise.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Load previous solution if it exists (helpful for continuation runs)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1. Build a large pool of analytical seeds
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_linear_decay(n))
    for alpha in np.linspace(0.0, 6.0, 13):
        seeds.append(_poly_decay(n, alpha))

    # exponential families – dense sweep over rate
    for rate in np.logspace(-5, 0.0, 30):
        seeds.append(_exp_decay(n, rate))
        # shifted exponentials (different offsets)
        for off in (0.1, 0.3, 0.5):
            seeds.append(_shifted_exp(n, rate, off))

    # logistic S‑shapes
    for scale in np.geomspace(0.05 * n, 0.5 * n, 8):
        seeds.append(_logistic(n, scale))

    # power‑law sweeps
    for p in np.linspace(0.0, 6.0, 14):
        seeds.append(_power(n, p))

    # beta‑shape grid
    alphas = np.geomspace(0.1, 10.0, 9)
    betas  = np.geomspace(0.1, 10.0, 9)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # many random Dirichlet draws with a variety of concentration parameters
    conc_vals = [0.01, 0.03, 0.07, 0.15, 0.3, 0.6, 1.0, 2.0]
    for conc in conc_vals:
        for _ in range(12):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution (if any) and a few noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            noisy = prev_arr + 0.004 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seeds (surrogate) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        remaining = deadline - time.time()
        # allocate a slice of the remaining time (at least 12 s)
        share = max(12.0, remaining * 0.08 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}] refinement budget ≈ {share:.1f}s")

        # a) deterministic ordering – cheap heuristic
        seq_desc = np.sort(seed_vec)[::-1]

        # b) short permutation hill‑climb on the original seed
        seq_hc, val_hc = _perm_hill_climb(seed_vec, sub_deadline, rng)

        # c) exponentiated‑gradient polish on three variants
        variants = [seed_vec, seq_desc, seq_hc]
        local_best_seq = None
        local_best_val = np.inf

        for vi, var in enumerate(variants):
            if time.time() >= sub_deadline:
                break
            # split remaining time equally among the remaining variants
            time_left = sub_deadline - time.time()
            var_share = time_left / (len(variants) - vi)
            var_deadline = time.time() + var_share
            seq_opt, val_opt = _exp_gradient_opt(var, var_deadline,
                                                 max_iters=300_000,
                                                 eta0=6.0)
            if val_opt < local_best_val:
                local_best_val = val_opt
                local_best_seq = seq_opt

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long polishing on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long polishing (exponentiated gradient) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                 deadline,
                                                 max_iters=4_000_000,
                                                 eta0=6.0)
        if final_val < global_best_val:
            global_best_seq, global_best_val = final_seq, final_val
            print(f"[Final polishing] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Mixing with other top seeds (convex combinations)
    # ------------------------------------------------------------------
    mix_deadline = time.time() + 15.0   # generous time slice for mixing
    for other in top_seeds[1:]:
        if time.time() > mix_deadline:
            break
        for t in np.linspace(0.05, 0.45, 9):
            cand = (1.0 - t) * global_best_seq + t * other
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s == 0.0:
                continue
            cand = cand / s
            val = _objective_surrogate(cand)
            if val < global_best_val:
                global_best_seq, global_best_val = cand, val
                print(f"[Mix] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Random‑perturbation escapes while time permits
    # ------------------------------------------------------------------
    while time.time() < deadline - 5.0:
        perturb = global_best_seq + 0.001 * rng.standard_normal(n)
        perturb = np.maximum(perturb, 0.0)
        perturb = _project_simplex(perturb)

        sub_deadline = time.time() + 8.0
        seq_tmp, val_tmp = _exp_gradient_opt(perturb,
                                             sub_deadline,
                                             max_iters=300_000,
                                             eta0=6.0)
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 8. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
