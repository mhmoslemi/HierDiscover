"""
Propose a candidate step‑function.

The optimiser works in three stages:

1️⃣  **Rich seed pool** – deterministic (uniform, arcsine, triangular,
    Beta(α,α) for many α) + Dirichlet draws + the previous best solution.

2️⃣  **Heavy Adam smoothing** – optimisation in the log‑softmax space
    of the smooth‐max (log‑sum‑exp) surrogate
    `2·n·max(conv(a,a))`.  A long β‑schedule (up to 5·10⁷) and many
    inner iterations (≈ 150 000 total) are performed, with periodic
    exact‐surrogate checks.  The best exact value seen is retained.

3️⃣  **Exact exponentiated‑gradient polishing** – sub‑gradient descent on
    the true max‑convolution (the same idea as the reference solution
    but with a larger iteration budget and an adaptive step‑size).

4️⃣  **Tiny stochastic walk** – a final local random search on the
    simplex to squeeze any remaining improvement.

The whole routine respects the 1000 s budget, prints progress, stores the
best sequence in the global variable ``height_sequence_1`` and finally
returns it as a plain list of floats.  The returned sequence is
non‑negative and normalised to sum = 1, which is the natural scale for
``evaluate_sequence`` (the objective is homogeneous).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, softmax, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector that sums to 1."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(seq,seq)), assuming sum(seq)==1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    """Flat distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density: a_i ∝ 1/√((i+½)(n‑½‑i))."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-14
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) discretisation on {0,…,n‑1}.
    For α=1 → uniform, α=0.5 → arcsine.
    """
    # Work with coordinates in open interval (0,1)
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    eps = 1e-14
    # beta pdf proportional to x^{α-1} (1-x)^{α-1}
    a = np.power(x + eps, alpha - 1.0) * np.power(1.0 - x + eps, alpha - 1.0)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Sample from Dirichlet( conc,…,conc )."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp smooth approximation
    of  `2·n·max(conv(seq,seq))`.  The function returns the best
    exact surrogate value ever seen (not the smoothed one) together
    with the corresponding sequence.
    """
    n = len(start_seq)

    # initialise logits (log‑softmax space)
    logit = np.log(start_seq + 1e-30)

    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # β schedule – slowly moves from a very smooth approximation to a
    # near‑exact max.  The last value is deliberately huge.
    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20_000.0, 50_000.0,
        100_000.0, 200_000.0, 500_000.0,
        1_000_000.0, 2_000_000.0, 5_000_000.0,
        10_000_000.0, 20_000_000.0, 50_000_000.0,
        100_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # we give more inner steps to the early (…) and later stages
        # (the later stages usually need only a few hundred updates).
        max_iter_beta = 4600 if beta < 5_000 else 3000

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            # current probability vector
            a = _softmax(logit)

            # linear convolution a*a
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max weights (stable)
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a.
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)                # length = n + (2n‑1) – 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]  # derivative of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)          # outer factor 2·n

            # convert to gradient on log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            # bias‑corrected
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.20                # a slightly larger step‑size than before
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # every few hundred updates evaluate the *exact* surrogate
            if it % 600 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final exact check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient on true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                      deadline: float,
                      max_iters: int = 500_000,
                      eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated gradient) on the *true* max‑convolution.
    Returns the best sequence seen and its exact surrogate value.
    """
    n = len(start_seq)
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))                     # active index

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]                 # d/d a_i of conv_k

        eta = eta0 / math.sqrt(it)                  # diminishing step‑size
        log_a = log_a - eta * g
        a = _softmax(log_a)                          # stay on simplex

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny stochastic walk on the simplex
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         deadline: float,
                         attempts: int = 80_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Very cheap random walk: move a tiny random amount from one coordinate
    to another and keep the move if it improves the exact surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    n = len(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises

    start_time = time.time()
    # keep a modest safety margin for bookkeeping
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------

            if arr.size == n and arr.sum() > 0.0:

            else:

            prev_vec = None
    else:
        prev_vec = None

    for conc in (0.05, 0.1, 0.2, 0.5, 1.0):

        if val_eg < best_val:

    # ------------------------------------------------------------------
    # 5️⃣  Final random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:

        # use the rest of the time for a cheap stochastic walk
        seq_loc, val_loc = _local_random_search(best_seq,

        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    true_val = None

    # Store for the next invocation

    return best_seq.tolist()
