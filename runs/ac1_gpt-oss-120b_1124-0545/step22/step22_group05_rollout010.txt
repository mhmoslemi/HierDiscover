"""
Propose a step‑function (non‑negative list of heights) that minimizes the
``evaluate_sequence`` objective.

Algorithm
---------
1. Build a pool of deterministic and random seeds (uniform, arcsine,
   triangular, Dirichlet draws, previous solution, tiny perturbations).
   All seeds are normalised to sum = 1 and the best surrogate value is kept.

2. Run a long mirror‑descent on the *smooth* version of the max‑convolution.
   The max is replaced by a soft‑max with a temperature that is slowly
   reduced (deterministic annealing).  The optimisation works on the
   log‑probabilities (a KL‑mirror) and uses an Adam‑style adaptive step.
   Gradient of the smoothed objective:
        g_i = 2·∑_j a_j·p_{i+j},
   where p is the soft‑max over the full convolution a*a.
   Both the convolution and the cross‑correlation needed for the gradient are
   computed by real FFTs (O(n log n)).

3. Finish with a cheap random local walk that moves a tiny amount of mass
   between two coordinates and accepts improvements.

The whole routine respects the 1000 s time budget and stores the best
sequence in the global ``height_sequence_1`` variable for possible warm‑starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(seq, seq)), assuming sum(seq)=1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Adam‑mirrored descent on the smoothed max‑convolution
# ----------------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta: float = 0.2,
                     temp_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (log‑space) with Adam‑style adaptation.
    Uses a soft‑max approximation of max(conv) with a slowly decreasing temperature.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                  # work in log‑softmax space

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- convolution and temperature ---------------------------------
        conv = _conv_fft(a, a)                 # full convolution, length 2n‑1
        max_conv = conv.max()
        # geometric cooling of the temperature (every 5 000 iters)
        temp = max(1e-12, max_conv * (temp_factor ** ((it - 1) // 5000)))

        # soft‑max over the convolution (stable version)
        exp_terms = np.exp((conv - max_conv) / temp)  # max removed → stable
        p = exp_terms / exp_terms.sum()               # probability on each lag

        # --- gradient of the smoothed max --------------------------------
        # g_i = 2 * sum_j a_j * p_{i+j}
        # compute via convolution of a with reversed p
        p_rev = p[::-1]                         # reversed probability vector
        conv_rev = _conv_fft(a, p_rev)          # length n + (2n‑1) - 1 = 3n‑2
        offset = len(p) - 1                     # = 2n‑2
        g = 2.0 * conv_rev[offset - np.arange(n)]

        # --- Adam update on log‑probabilities ----------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        log_a = log_a - step                     # gradient descent in log‑space

        # project back onto the simplex
        a = _softmax(log_a)

        # --- keep best --------------------------------------------------
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex: move a tiny random amount from one
    coordinate to another and keep the move if it improves the surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    i = 0
    while i < attempts and time.time() < deadline:
        i += 1
        i1, i2 = rng.integers(0, n, size=2)
        if i1 == i2:
            continue
        # move a relative amount from i1 to i2
        delta = sigma * best_seq[i1] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i1] -= delta
        new_seq[i2] += delta
        if new_seq[i1] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val
    return best_seq, best_val

        # allocate roughly half of the remaining budget to this stage
        remaining = deadline - time.time()
        # we let the optimiser run until the deadline; max_iters is a large safety bound
        seq_opt, val_opt = _adam_smooth_opt(best_seq, n,
                                            deadline,
                                            max_iters=2_000_000,
                                            eta=0.2,
                                            temp_factor=0.5)
        print(f"[Adam] surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 4️⃣ Final cheap random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_local, val_local = _local_random_search(best_seq, n,
                                                    deadline,
                                                    attempts=300_000,
                                                    sigma=0.02)
        print(f"[Local] surrogate = {val_local:.6f}")
        if val_local < best_val:
            best_val, best_seq = val_local, seq_local.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣ Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
