"""
Advanced step‑function optimiser (improved)

The optimiser builds a diverse pool of analytic “seed” shapes (uniform,
triangular, windows, DPSS, exponential, Gaussian, beta‑distribution,
etc.) and augments it with a fast random‑parameter search over the same
families.  The best few seeds are refined through a cascade of
optimisation phases:

1. Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution
   using a multi‑active set.  A larger iteration budget and a tighter
   tolerance than the previous baseline are used.

2. Smooth‑max optimisation with an Adam optimiser on a log‑sum‑exp
   surrogate.  The smoothing parameter β is increased on a schedule
   (β≈10⁻¹ → 10⁶) so the algorithm converges to the true L∞‑objective.

3. A “flattening’’ phase that minimises the squared deviation of the
   full autocorrelation from its average (drives the whole convolution
   toward a constant).

4. A short simulated‑annealing polish that can escape shallow local
   minima.

The routine respects the 1000 s wall‑clock budget, stores the best
sequence it has found in the global variable ``height_sequence_1`` and
returns it as a plain list.  The code re‑uses the fast FFT‑based
convolution utilities from the reference implementation and adds a
targeted random‑search over window parameters, which is the main source
of the improvement over the previous state‑of‑the‑art.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s

# ----------------------------------------------------------------------
def _surrogate(seq: np.ndarray, n: int) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (seq summed to 1)."""
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy i0 is unavailable."""
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate*i) , i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    """Sigmoid shape, optionally shifted."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires SciPy."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()

def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 300_000,
                            eta0: float = 6.0,
                            tol: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Uses all indices within a relative tolerance of the current maximum.
    Returns the best sequence encountered and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _surrogate(a, n)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # active set – indices with value close to the current maximum
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient (average over active set)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule=None,
                     inner_iters: int = 2500) -> np.ndarray:
    """
    Adam optimiser on the smooth‐max surrogate (log‑sum‑exp).
    Works in log‑softmax space to stay on the probability simplex.
    """
    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
            50.0, 100.0, 250.0, 500.0,
            1000.0, 2500.0, 5000.0, 1e4,
            2e4, 5e4, 1e5, 2e5, 5e5, 1e6,
        ]

    # initialise log‑softmax parameters
    v = np.log(start_seq + 1e-30)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)
    step = 0

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        for _ in range(inner_iters):
            if time.time() > deadline:
                break
            a = _softmax(v)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # log‑sum‑exp smooth max
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # objective (only needed for diagnostic – not returned)
            # obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient of smooth max w.r.t. a
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain rule back to v (log‑softmax)
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)

            # Adam update
            step += 1
            m = beta1 * m + (1.0 - beta1) * grad_v
            v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
            m_hat = m / (1.0 - beta1 ** step)
            v2_hat = v2 / (1.0 - beta2 ** step)

            # learning‑rate schedule – decrease slowly with steps
            lr = 0.1 / math.sqrt(step)
            v = v - lr * m_hat / (np.sqrt(v2_hat) + eps)

    return _softmax(v)


def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 150_000,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the loss Σ_k (conv_k - avg)².
    The goal is to make the whole convolution as flat as possible.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _surrogate(a, n)

    denom = float(2 * n - 1)          # size of the full convolution

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        # gradient of Σ diff² w.r.t a (two convolutions)
        full = _conv_fft(rev_a, diff)            # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent (log‑space)
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _surrogate(a, n)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 100_000,
                         start_sigma: float = 0.002,
                         T_start: float = 0.01,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a, n)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _surrogate(ns, n)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

def _generate_random_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random family (including mixtures) and return a normalised shape."""
    families = [
        _uniform,
        _triangular,
        _sine_window,
        _blackman_window,
        _hamming_window,
        _cosine_window,
        _random_kaiser,
        _random_gaussian,
        _random_exp_decay,
        _random_beta_shape,
        _random_dpss,
        _random_mix_two,
    ]
    f = rng.choice(families)
    if f in {_uniform, _triangular, _sine_window,
             _blackman_window, _hamming_window, _cosine_window}:
        return f(n)
    elif f in {_random_kaiser, _random_gaussian,
               _random_exp_decay, _random_beta_shape,
               _random_dpss, _random_mix_two}:
        return f(n, rng)  # type: ignore
    else:
        # fallback (should never happen)
        return _uniform(n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget, stores the best sequence found in
    the global variable ``height_sequence_1`` and returns it as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Problem size – take the length of a previous solution if present,
    # otherwise fall back to a moderately large value.
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and previous:
        n = len(previous)
        prev_arr = np.maximum(np.asarray(previous, dtype=np.float64), 0.0)
        if prev_arr.sum() > 0.0:
            prev_arr = prev_arr / prev_arr.sum()
        else:
            prev_arr = None
    else:
        n = 4000
        prev_arr = None

    print(f"\n=== STEP‑FUNCTION OPTIMISER ===\nProblem size n = {n}")

    # --------------------------------------------------------------
    # 1. Construct an initial pool of seeds (analytic families)
    # --------------------------------------------------------------
    print("\n--- Building analytic seed pool ---")
    seeds = [
        _uniform(n),
        _triangular(n),
        _sine_window(n, gamma=1.0),
        _blackman_window(n),
        _hamming_window(n),
        _cosine_window(n),
    ]

    # Kaiser windows (different betas)
    for beta in [0.0, 2.0, 4.0, 6.0, 8.0, 12.0]:
        seeds.append(_kaiser_window(n, beta))

    # Gaussian shapes
    for sigma_frac in np.linspace(0.05, 0.5, 7):
        seeds.append(_gaussian(n, sigma_frac))

    # Exponential decays
    for rate in np.logspace(-5, -0.2, 15):
        seeds.append(_exp_decay(n, rate))

    # Power‑law tails
    for p in np.linspace(0.0, 6.0, 13):
        seeds.append(_power(n, p))

    # Beta‑distribution shapes (coarse grid)
    for a in np.linspace(0.1, 10.0, 5):
        for b in np.linspace(0.1, 10.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # DPSS windows (if available)
    if _sigwin is not None:
        for nw in [2.5, 3.0, 3.5, 4.0, 4.5, 5.0]:
            seeds.append(_dpss_window(n, nw))

    # Include the previous best (if any) and a few perturbed copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(4):
            pert = prev_arr + 0.005 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0.0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Fast random‑parameter search (adds diversity)
    # --------------------------------------------------------------
    print("\n--- Random‑parameter exploration ---")
    random_budget = max(0.0, deadline - time.time() - 180.0)   # reserve ~3 min for refinement
    while time.time() < deadline - 180.0:
        shape = _generate_random_shape(n, rng)
        seeds.append(shape)
    print(f"Total seeds after random search: {len(seeds)}")

    # --------------------------------------------------------------
    # 3. Evaluate all seeds (surrogate) and keep the best few
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _surrogate(vec, n)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 4. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _surrogate(global_best_seq, n)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Multi‑start refinement (EG → Adam → flatten → SA)
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        # Allocate a slice of the remaining budget for this seed
        remaining = deadline - time.time()
        # roughly 12 % of the remaining time per seed (shrinks as we go)
        share = remaining * 0.12 / max(1, (top_k - si))
        sub_deadline = time.time() + share

        print(f"\n[Seed {si+1}/{top_k}] optimisation budget ≈ {share:.1f}s")

        # ---- a) exponentiated‑gradient (multi‑active) ----------
        seq_eg, val_eg = _exp_gradient_opt_multi(
            seed_vec.copy(),
            n,
            sub_deadline,
            max_iters=200_000,
            eta0=8.0,
            tol=1e-10,
        )

        # ---- b) Adam on smooth max ----------
        if time.time() < sub_deadline:
            seq_adam = _adam_smooth_opt(seq_eg, n, sub_deadline,
                                        beta_schedule=None,
                                        inner_iters=2000)
            val_adam = _surrogate(seq_adam, n)
        else:
            seq_adam = seq_eg
            val_adam = val_eg

        # ---- c) flattening gradient ----------
        if time.time() < sub_deadline:
            seq_flat, val_flat = _flatten_gradient_opt(
                seq_adam,
                n,
                sub_deadline,
                max_iters=120_000,
                eta0=5.0,
            )
        else:
            seq_flat, val_flat = seq_adam, val_adam

        # ---- d) simulated annealing ----------
        hill_deadline = sub_deadline - 0.25
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(
                seq_flat,
                n,
                hill_deadline,
                rng,
                max_attempts=60_000,
                start_sigma=0.0015,
                T_start=0.012,
                T_end=5e-9,
            )
        else:
            seq_sa, val_sa = seq_flat, val_flat

        # pick the best among the candidates
        candidates = [
            (seed_vec, _surrogate(seed_vec, n)),
            (seq_eg, val_eg),
            (seq_adam, val_adam),
            (seq_flat, val_flat),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Final heavy polishing on the global best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG polishing (multi‑active) ---")
        final_seq, final_val = _exp_gradient_opt_multi(
            global_best_seq,
            n,
            deadline,
            max_iters=4_000_000,
            eta0=6.0,
            tol=1e-11,
        )
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final flattening stage ---")
        final_seq2, final_val2 = _flatten_gradient_opt(
            global_best_seq,
            n,
            deadline,
            max_iters=800_000,
            eta0=4.0,
        )
        if final_val2 < global_best_val:
            global_best_seq = final_seq2
            global_best_val = final_val2
            print(f"[Final flatten] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final simulated‑annealing polish ---")
        final_sa_seq, final_sa_val = _simulated_annealing(
            global_best_seq,
            n,
            deadline,
            rng,
            max_attempts=120_000,
            start_sigma=0.001,
            T_start=5e-3,
            T_end=1e-9,
        )
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(
            f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
            f"true evaluate_sequence = {true_val:.6f}"
        )
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
