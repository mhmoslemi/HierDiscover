"""
Propose a high‑quality step‑function (height vector) that minimizes

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm uses three stages:

1. **Continuation with smoothed‑max Adam optimisation** –   
   A differentiable surrogate `log‑sum‑exp` of the max‑convolution is
   minimised on the probability simplex by Adam.  The smoothing
   parameter β is increased geometrically so that the optimisation
   gradually focuses on the true maximum.  The gradient of the
   surrogate is obtained analytically via a single FFT‑based
   correlation.

2. **Random‑restart warm‑starts** –  
   Several short Adam runs are started from Dirichlet‑sampled vectors
   (the budget is split between these restarts and a long final run).

3. **Fine‑grained simulated‑annealing** –  
   Tiny mass transfers between random indices are accepted with a
   temperature schedule to escape any remaining shallow minima.

All vectors are kept on the ℓ₁‑simplex (non‑negative, sum = 1), which
makes the evaluation scale‑invariant.  The best sequence found is stored
in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _surrogate_obj(a: np.ndarray) -> float:
    """Exact objective when Σa = 1 (i.e. 2·n·max(conv(a,a)))."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Smoothed‑max gradient (log‑sum‑exp) and its exact max value
# ----------------------------------------------------------------------
def _smooth_max_grad(a: np.ndarray, beta: float):
    """
    Returns (grad, max_conv) where grad = ∇[log‑sum‑exp_β(conv(a,a))]
    and max_conv = max(conv(a,a)).  Both are computed with a single
    FFT‑based correlation.
    """
    n = a.shape[0]
    # exact convolution
    conv = _conv_fft(a, a)               # length 2n‑1
    maxc = np.max(conv)

    # soft‑max weights (stable)
    expw = np.exp(beta * (conv - maxc))
    w = expw / expw.sum()                # w ≥ 0, Σw = 1, length 2n‑1

    # gradient = 2 * cross‑correlation(a, w)
    w_rev = w[::-1]                     # reverse for correlation
    N = _next_pow2(3 * n - 2)            # length needed for full correlation
    fa = np.fft.rfft(a, n=N)
    fw = np.fft.rfft(w_rev, n=N)
    corr_full = np.fft.irfft(fa * fw, n=N)[:3 * n - 2]   # length 3n‑2

    # indices that give Σ_i a_i w_{i+j}
    idx = (2 * n - 2) - np.arange(n)     # L‑1‑j  with L = 2n‑1
    grad = 2.0 * corr_full[idx]
    return grad, maxc


# ----------------------------------------------------------------------
# Adam optimiser on the smoothed‑max surrogate
# ----------------------------------------------------------------------
def _adam_opt(start_seq: np.ndarray,
              deadline: float,
              beta_start: float = 0.5,
              beta_end: float = 200.0,
              lr0: float = 0.2,
              max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed max‑convolution.
    Stops when the time deadline is hit or after `max_iters` iterations.
    Returns the best simplex‑projected vector encountered and its exact
    surrogate value.
    """
    n = start_seq.shape[0]
    a = _proj_simplex(start_seq.astype(np.float64))

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    best_a = a.copy()
    best_val = _surrogate_obj(a)

    beta = beta_start
    beta_mul = (beta_end / beta_start) ** (1.0 / max_iters)  # geometric schedule

    t = 0
    # check‑point interval for time‑budget handling
    check_interval = 1000
    next_check = time.time() + 0.5

    while time.time() < deadline and t < max_iters:
        t += 1
        grad, cur_max = _smooth_max_grad(a, beta)

        # Adam update in the ambient space (before projection)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        eta = lr0 / math.sqrt(t)                     # simple decay
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

        # simplex projection (ensures Σa = 1, a ≥ 0)
        a = _proj_simplex(a)

        # exact surrogate (no smoothing) – cheap because we already have cur_max
        cur_val = float(2.0 * n * cur_max)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()

        # increase smoothing parameter
        beta = min(beta_end, beta * beta_mul)

        # occasional time‑budget check
        if t % check_interval == 0 and time.time() > next_check:
            if time.time() >= deadline:
                break
            next_check = time.time() + 0.2

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny mass moves)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 500_000,
                                T_start: float = 0.05,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a tiny fraction of mass between two random
    coordinates.  Temperature and perturbation size decay linearly.
    Returns the best sequence seen.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_a = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = ns.copy()
        attempts += 1

    return best_a, best_val
