    conv = _conv_fft(a, a)  # length 2n-1

"""
Propose a step‑function (non‑negative list of heights) that minimises the
``evaluate_sequence`` objective.

Strategy
--------
1. Build a diverse set of deterministic seeds (uniform, arcsine,
   triangular, several Beta‑shapes, geometric / linear decays, and a
   “γ‑shape” a_i ∝ ((i+½)(n‑½‑i))^{-γ}) and many convex mixtures of the
   arcsine distribution with the uniform and Beta(2,2) distributions.
   All vectors are normalised to sum = 1.

2. Evaluate the exact surrogate 2·n·max(conv(seq,seq)) for every seed
   (the surrogate is identical to ``evaluate_sequence`` when the sum of
   the sequence equals 1) and keep the best seed.

3. Refine the best seed with a *smooth‑max* gradient descent.
   The true objective ``max(conv)`` is replaced by
       S_eps(a) = eps·log(∑_k exp(conv_k/eps)),
   whose gradient is
       ∇S = 2·(w_rev * a)   with   w_k ∝ exp(conv_k/eps).
   The gradient is computed with FFT‑based convolutions, and a
   multiplicative (exponentiated‑gradient) update is performed on the
   log‑probabilities.  ε is decreased geometrically from 10⁻³ to 10⁻⁷,
   while the step size η_t = η₀/√t (η₀≈8) is used together with a
   momentum term.

4. A short exponentiated‑gradient (sub‑gradient of the *active* lag)
   phase is run to polish the solution.

5. A cheap random‑local‑move walk finishes the search.

The whole routine respects a 1000 s time budget; it stops as soon as the
deadline is reached and returns the best sequence discovered.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax (sum = 1, non‑negative)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)) assuming sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) / (n - 1.0)
    eps = 1e-12
    a = np.power(i + eps, alpha - 1.0) * np.power(1.0 - i + eps, beta - 1.0)
    a_sum = a.sum()
    if a_sum <= eps:
        return _uniform(n)
    return a / a_sum


def _geometric(n: int, rate: float) -> np.ndarray:
    if not (0.0 < rate < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - rate) * np.power(rate, i)
    a_sum = a.sum()
    if a_sum <= 0.0:
        return _uniform(n)
    return a / a_sum


def _linear_decreasing(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = n - i
    return a / a.sum()


def _gamma_shape(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ ((i+½)(n‑½‑i))^{-γ}  (γ≈0.5 reproduces arcsine)."""
    i = np.arange(n, dtype=np.float64)
    term = (i + 0.5) * (n - 0.5 - i)
    term = np.maximum(term, 1e-12)
    a = term ** (-gamma)
    a_sum = a.sum()
    if a_sum <= 0.0:
        return _uniform(n)
    return a / a_sum


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _perturb(seq: np.ndarray, rng: np.random.Generator, sigma: float = 0.002) -> np.ndarray:
    """Add tiny Gaussian noise, clip, and renormalise."""
    noisy = seq + sigma * rng.normal(size=seq.shape)
    noisy = np.maximum(noisy, 0.0)
    s = noisy.sum()
    if s <= 0.0:
        return seq.copy()
    return noisy / s


# ----------------------------------------------------------------------
# Smoothed‑max gradient descent (multiplicative updates)
# ----------------------------------------------------------------------
def _smooth_gradient_descent(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             eta0: float = 8.0,
                             eps_start: float = 1.0e-3,
                             eps_end: float = 1.0e-7,
                             eps_decay: float = 0.9995,
                             momentum: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smooth approximation of the max‑convolution.
    Returns (best_sequence, best_surrogate).
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    grad_mom = np.zeros_like(a)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    it = 0
    eps = eps_start

    while time.time() < deadline:
        it += 1
        # --- compute convolution and smooth weights ---
        conv = _conv_fft(a, a)                 # length 2n‑1
        max_conv = conv.max()
        # stable soft‑max on the convolution
        exp_vals = np.exp((conv - max_conv) / eps)
        w = exp_vals / exp_vals.sum()          # length 2n‑1, sum = 1

        # --- gradient of the smooth max ---
        w_rev = w[::-1]
        grad_full = _conv_fft(w_rev, a)        # length 3n‑2
        idx_start = 2 * n - 2
        grad = 2.0 * grad_full[idx_start - np.arange(n)]

        # momentum on the gradient
        grad_mom = momentum * grad_mom + (1.0 - momentum) * grad

        # step size schedule (η_t = η₀ / √t)
        eta = eta0 / math.sqrt(it)

        # multiplicative update in log‑space
        log_a = log_a - eta * grad_mom
        a = _softmax(log_a)

        # evaluate current surrogate
        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()
            # occasional progress report
            if it % 20000 == 0:
                print(f"[SGD] iter {it:>7d}: surrogate = {cur_val:.6f}")

        # decay epsilon (never go below eps_end)
        eps = max(eps * eps_decay, eps_end)

        # safety exit
        if it >= 5_000_000:          # hard cap to avoid infinite loop
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient (active‑lag) optimisation – same as before
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 3_000_000,
                     eta0: float = 8.0):
    """
    Mirror‑descent on the *exact* max‑convolution using the sub‑gradient
    at the active lag only.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of the active lag
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local search
# ----------------------------------------------------------------------
def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex: move a tiny random amount from one
    coordinate to another and keep the move if it improves the surrogate.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    draws = 0

    while draws < draw_limit and time.time() < deadline:
        conc = rng.choice(conc_options)
        vec = _random_dirichlet(n, conc, rng)
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val, best_seq = val, vec.copy()
            print(f"[Dir] new best surrogate = {best_val:.6f} (draw {draws})")
        draws += 1

    # ------------------------------------------------------------------
    # 3️⃣ Smoothed‑max gradient descent (continuous refinement)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Smoothed‑max gradient descent ---")
        seq_sgd, val_sgd = _smooth_gradient_descent(best_seq, n,
                                                    deadline,
                                                    eta0=8.0,
                                                    eps_start=1.0e-3,
                                                    eps_end=1.0e-7,
                                                    eps_decay=0.9995,
                                                    momentum=0.9)
        print(f"[SGD] best surrogate = {val_sgd:.6f}")
        if val_sgd < best_val:
            best_val, best_seq = val_sgd, seq_sgd.copy()
            print("   >>> Updated global best (SGD)")

    # ------------------------------------------------------------------
    # 4️⃣ Exponentiated‑gradient polishing (active‑lag sub‑gradient)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing ---")
        # allocate remaining budget for this phase
        remaining = deadline - time.time()
        # cap iterations heuristically
        max_iters = int(2_000_000 * (remaining / (budget_s - (time.time() - start_time))))
        max_iters = max(300_000, min(max_iters, 5_000_000))

        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           deadline,
                                           max_iters=max_iters,
                                           eta0=8.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣ Final lightweight random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                deadline,
                                                attempts=200_000,
                                                sigma=0.02)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 6️⃣ Official evaluation with the provided evaluate_sequence()
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible next call
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
