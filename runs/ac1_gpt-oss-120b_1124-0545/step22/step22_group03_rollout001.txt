"""
Algorithm: Smoothed exponentiated‑gradient descent on the max‑convolution
objective, followed by a fine‑grained simulated‑annealing refinement.
The gradient of a smooth approximation of the max (log‑sum‑exp) is
computed analytically using FFT‑based convolutions, allowing all
shifts to influence the update (unlike the previous single‑active‑shift
method).  The optimizer runs on the probability simplex via a
multiplicative update (mirror‑descent with KL divergence).  Early in the
run a large smoothing parameter encourages exploration; it is
exponentially decreased so that the method eventually behaves like
sub‑gradient descent on the true max.  The final SA stage performs tiny
mass transfers between random indices to escape any remaining local
plateaus.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (real FFT, power‑of‑two padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = x.shape[0] + y.shape[0] - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, eps_factor: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Compute the convolution of `a` with itself, a smooth approximation of the
    max (log‑sum‑exp) and its gradient w.r.t. `a`.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing parameter:
        epsilon = eps_factor * max(conv).

    Returns
    -------
    conv : np.ndarray
        Linear convolution a * a (length 2n‑1).
    g   : np.ndarray
        Gradient of the smoothed max (d/d a).
    max_conv : float
        The exact maximum entry of `conv`.
    """
    n = a.shape[0]

    # exact convolution a*a
    conv = _conv_fft(a, a)                     # length 2n-1
    max_conv = float(conv.max())

    # smoothing parameter
    eps = max_conv * eps_factor
    if eps < 1e-12:
        eps = 1e-12

    # stable soft‑max weights for the smoothed max
    scaled = conv / eps
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    w = exp_vals / exp_vals.sum()              # length 2n-1, sums to 1

    # compute T_i = sum_j w_{i+j} a_j   (cross‑correlation)
    a_rev = a[::-1]
    T_full = _conv_fft(w, a_rev)               # length 3n-2
    offset = n - 1
    T = T_full[offset:offset + n]              # size n

    # correction term w_{2i}
    w2 = w[2 * np.arange(n)]

    # gradient of the smoothed max (ignoring the constant 2*n factor)
    g = 2.0 * T - a * w2

    return conv, g, max_conv

# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimizer
# ----------------------------------------------------------------------
def _exp_grad_smooth(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 2.0,
                     eps_factor_start: float = 5.0,
                     eps_factor_decay: float = 0.9995,
                     min_eps_factor: float = 0.01,
                     max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the smoothed max‑convolution.

    Returns the best probability vector encountered and its exact surrogate
    value 2·n·max(conv).
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    # initialise log‑weights (use a small offset to avoid log(0))
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = 2.0 * n * _conv_fft(a, a).max()

    eps_factor = eps_factor_start

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv, g, max_conv = _smooth_max_gradient(a, eps_factor)

        # exact surrogate value (not smoothed)
        cur_val = 2.0 * n * max_conv
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # step‑size schedule (standard 1/√t)
        eta = eta0 / math.sqrt(it)

        # mirror‑descent step in log‑space
        log_a -= eta * g

        # re‑normalise in a numerically safe way
        log_a -= np.max(log_a)          # prevent overflow
        a = np.exp(log_a)
        a /= a.sum()

        # decay the smoothing factor
        eps_factor = max(min_eps_factor, eps_factor * eps_factor_decay)

        # occasional progress report
        if it % 10000 == 0:
            print(f"[EG] iter={it:,} eps_factor={eps_factor:.5f} surrogate={cur_val:.6f}")

    return best_seq, best_val

# ----------------------------------------------------------------------
# Simulated‑annealing refinement (tiny transfers, linear cooling)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 500_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.003) -> tuple[np.ndarray, float]:
    """
    Simple SA: move a small amount of probability mass between two random
    coordinates, accepting uphill moves according to a linear temperature
    schedule.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = 2.0 * n * _conv_fft(a, a).max()
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear interpolation of the temperature
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = sigma_start * (T / T_start)

        # pick a donor with non‑negligible mass
        i = rng.integers(n)
        if a[i] <= 1e-15:
            attempts += 1
            continue

        # propose a transfer amount
        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        # pick a distinct receiver
        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = 2.0 * n * _conv_fft(ns, ns).max()
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

    # ------------------------------------------------------------------
    # Pick the best of the two phases
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # True evaluation (may differ slightly due to rounding)
    # ------------------------------------------------------------------

        true_val = evaluate_sequence(best_seq.tolist())
        print(f"True evaluate_sequence value = {true_val:.6f}")

Algorithm: Smoothed exponentiated‑gradient descent on the max‑convolution
objective, followed by a fine‑grained simulated‑annealing refinement.
The gradient of a smooth approximation of the max (log‑sum‑exp) is
computed analytically using FFT‑based convolutions, allowing all
shifts to influence the update (unlike the previous single‑active‑shift
method).  The optimizer runs on the probability simplex via a
multiplicative update (mirror‑descent with KL divergence).  Early in the
run a large smoothing parameter encourages exploration; it is
exponentially decreased so that the method eventually behaves like a
sub‑gradient descent on the true max.  The final SA stage performs tiny
mass transfers between random indices to escape any remaining local
plateaus.
"""

# Convolution utilities (real FFT, power‑of‑two padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = x.shape[0] + y.shape[0] - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Smoothed max‑convolution gradient
# ----------------------------------------------------------------------
def _smooth_max_gradient(a: np.ndarray, eps_factor: float) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Compute the convolution a*a, a smooth approximation of its maximum,
    and the gradient of that smooth max w.r.t. a.

    Parameters
    ----------
    a : np.ndarray
        Probability vector (non‑negative, sum = 1).
    eps_factor : float
        Multiplicative factor for the smoothing parameter:
        epsilon = eps_factor * max(conv).

    Returns
    -------
    conv : np.ndarray
        Linear convolution a*a (length 2n‑1).
    g : np.ndarray
        Gradient of the smoothed max (ignoring the constant 2·n factor).
    max_conv : float
        Exact maximum entry of `conv`.
    """
    n = a.shape[0]

    # exact convolution
    conv = _conv_fft(a, a)          # length 2n‑1
    max_conv = float(conv.max())

    # smoothing parameter
    eps = max_conv * eps_factor
    eps = max(eps, 1e-12)

    # stable soft‑max weights for the smoothed max
    scaled = conv / eps
    max_scaled = np.max(scaled)
    exp_vals = np.exp(scaled - max_scaled)
    w = exp_vals / exp_vals.sum()   # length 2n‑1, sums to 1

    # T_i = Σ_j w_{i+j} a_j  (cross‑correlation)
    a_rev = a[::-1]
    T_full = _conv_fft(w, a_rev)   # length 3n‑2
    offset = n - 1
    T = T_full[offset:offset + n]  # size n

    # correction term w_{2i}
    w2 = w[2 * np.arange(n)]

    # gradient of the smoothed max (ignore constant 2·n)
    g = 2.0 * T - a * w2
    return conv, g, max_conv

# Smoothed exponentiated‑gradient optimizer
# ----------------------------------------------------------------------
def _exp_grad_smooth(start_seq: np.ndarray,
                     deadline: float,
                     eta0: float = 2.0,
                     eps_factor_start: float = 5.0,
                     eps_factor_decay: float = 0.9995,
                     min_eps_factor: float = 0.01,
                     max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the smoothed max‑convolution.

    Returns the best probability vector encountered and its exact surrogate
    value 2·n·max(conv).
    """
    n = start_seq.shape[0]
    a = start_seq.copy()

    best_seq = a.copy()

    eps_factor = eps_factor_start

            best_val = cur_val
            best_seq = a.copy()

        # step‑size schedule
        eta = eta0 / math.sqrt(it)

        # mirror‑descent step in log‑space
        log_a -= eta * g

        # renormalise safely
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        # decay smoothing factor

# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 500_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-8,
                                sigma_start: float = 0.003) -> tuple[np.ndarray, float]:

    Transfer a tiny amount of probability mass between two random indices,
    accepting uphill moves according to a linear temperature schedule.
    """

    cur_val = best_val

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling

            attempts += 1
            continue

            attempts += 1

            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val
