Advanced step‑function optimizer (re‑engineered)

* Uses a richer set of analytic and random seeds.
* Performs a long exponentiated‑gradient (mirror‑descent) phase with a
  gradually tightening active‑set tolerance.
* Refines the solution with a flattened‑quadratic‑loss phase (mirrored).
* Optimises a smooth approximation of the ∞‑norm (log‑sum‑exp) by
  L‑BFGS‑B with an extended β schedule (up to 1e9) and generous
  iteration limits.
* Finishes with a fine‑grained simulated‑annealing polish.
* All optimisation stays on the probability simplex via a soft‑max
  parametrisation.
* The routine respects a 1000 s wall‑clock budget and stores the best
  sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# Optional scientific imports
# ----------------------------------------------------------------------

    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None

except Exception:  # pragma: no cover
    _i0 = None

try:
    from scipy.signal import windows as _sigwin
except Exception:  # pragma: no cover

# Basic numeric utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution via a real FFT (O(n log n))."""
    n = len(a)
    L = 2 * n - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max → a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (vectors are ℓ₁‑normalised, sum = 1)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq, seq)) (sum(seq) == 1)."""
    n = len(seq)
    conv = _conv_self_fft(seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy i0 is unavailable."""
    if _i0 is None:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    m = (n - 1) / 2.0
    arg = np.sqrt(1.0 - ((i - m) / m) ** 2)
    a = _i0(beta * arg) / _i0(beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shape on the discrete index set 1…n."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate*i) , i=0..n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _logistic_shape(n: int, scale: float, midpoint: float) -> np.ndarray:
    """Sigmoid shape, optionally shifted."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-(i - midpoint) / scale))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector (conc parameter controls concentration)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture a·(1‑α) + b·α, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """Discrete prolate spheroidal (Slepian) window – requires scipy.signal."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)
    a = np.abs(a[0])
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on exact max‑convolution
# ----------------------------------------------------------------------
def _exp_gradient_opt_multi(start_seq: np.ndarray,
                            n: int,
                            deadline: float,
                            max_iters: int = 5_000_000,
                            eta0: float = 10.0,
                            tol_start: float = 1e-5,
                            tol_end: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient with a *tightening* active‑set tolerance.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    # Adam‑style momentum
    momentum = np.zeros_like(a)
    beta_mom = 0.9

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # linear schedule for tolerance
        tol = tol_start + (tol_end - tol_start) * (it / max_iters)

        conv = _conv_self_fft(a)
        max_val = conv.max()
        active = np.where(conv >= max_val - tol * max_val)[0]

        # sub‑gradient of the max (average over active shifts)
        g = np.zeros_like(a)
        idx = np.arange(n, dtype=np.int64)
        for kk in active:
            rev_idx = kk - idx
            mask = (rev_idx >= 0) & (rev_idx < n)
            g[mask] += a[rev_idx[mask]]
        if active.size:
            g = 2.0 * g / active.size

        # momentum update
        momentum = beta_mom * momentum + (1.0 - beta_mom) * g
        eta = eta0 / math.sqrt(it)

        log_a = log_a - eta * momentum
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Gradient‑descent that flattens the convolution (quadratic loss)
# ----------------------------------------------------------------------
def _flatten_gradient_opt(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on Σ (conv_k - avg)^2.
    Drives the whole autocorrelation towards a constant.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    denom = float(2 * n - 1)          # length of full convolution
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_self_fft(a)
        avg = (a.sum() ** 2) / denom   # = 1/denom when sum=1
        diff = conv - avg

        rev_a = a[::-1]
        full = _conv_fft(rev_a, diff)          # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # mirror‑descent conversion
        dot = np.dot(grad_a, a)
        grad_log = a * (grad_a - dot)

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth max via log‑sum‑exp – L‑BFGS‑B optimiser
# ----------------------------------------------------------------------
def _smooth_obj_grad(v: np.ndarray, n: int, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (smooth_obj, grad) where smooth_obj approximates 2·n·max(conv).
    The variable `v` lives in log‑softmax space.
    """
    a = _softmax(v)
    conv = _conv_self_fft(a)
    shift = conv.max()
    exp_beta = np.exp(beta * (conv - shift))
    sum_exp = exp_beta.sum()
    obj = 2.0 * n * (shift + np.log(sum_exp) / beta)

    w = exp_beta / sum_exp
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)          # length = 3n‑2
    grad_a = 2.0 * full[n - 1:n - 1 + n]

    dot = np.dot(grad_a, a)
    grad_v = a * (grad_a - dot)
    return obj, grad_v


class _LBFGSCallback:
    """Terminate L‑BFGS when the global deadline is reached."""
    def __init__(self, deadline: float):
        self.deadline = deadline

    def __call__(self, x):
        if time.time() > self.deadline:
            raise KeyboardInterrupt


def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           iters_per_beta: int = 2500) -> np.ndarray:
    """
    Optimise a smooth max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, gradually tightening β.
    """
    if _opt is None:
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0,
            80.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 20_000.0, 50_000.0,
            100_000.0, 200_000.0, 500_000.0, 1_000_000.0,
            2_000_000.0, 5_000_000.0, 10_000_000.0,
        ]

    v = np.log(start_vec + 1e-30)

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        try:
            res = _opt.minimize(fun=_smooth_obj_grad,
                                x0=v,
                                args=(n, beta_val),
                                jac=True,
                                method='L-BFGS-B',
                                callback=_LBFGSCallback(deadline),
                                options={'maxiter': iters_per_beta,
                                         'disp': False,
                                         'ftol': 1e-13})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (accepts occasional worsenings)
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 200_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

            seeds.append(_mix_shapes(_uniform(n), exp_seq, alpha))

    # Include reversed versions (break symmetry)
    rev_seeds = [s[::-1].copy() for s in list(seeds)]
    seeds.extend(rev_seeds)

    # Perturbations of the previous best (if any)
    if prev_arr is not None and prev_arr.sum() > 0.0:
        base = prev_arr / prev_arr.sum()
        seeds.append(base)
        for _ in range(5):
            pert = base + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            s = pert.sum()
            if s > 0:
                seeds.append(pert / s)

    # --------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # --------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # --------------------------------------------------------------
    # 3. Initialise global best
    # --------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 4. Multi‑start refinement
    # --------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # Allocate a share of the remaining budget
        remaining = deadline - time.time()
        # 0.10 of the remaining time per seed (adjusted later)
        share = remaining * 0.10
        sub_deadline = time.time() + share

        # a) Exponentiated‑gradient (tight active set)
        seq_eg, val_eg = _exp_gradient_opt_multi(seed_vec.copy(),
                                                n,
                                                sub_deadline,
                                                max_iters=2_000_000,
                                                eta0=12.0,
                                                tol_start=1e-4,
                                                tol_end=1e-12)

        # b) Flattening gradient phase
        if time.time() < sub_deadline:
            seq_flat, val_flat = _flatten_gradient_opt(seq_eg,
                                                       n,
                                                       sub_deadline,
                                                       max_iters=800_000,
                                                       eta0=6.0)
        else:
            seq_flat, val_flat = seq_eg, val_eg

        # c) L‑BFGS smoothing (high β)
        if time.time() < sub_deadline:
            seq_lbfgs = _lbfgs_smooth_optimize(seq_flat,
                                               n,
                                               sub_deadline,
                                               beta_schedule=None,
                                               iters_per_beta=2500)
            val_lbfgs = _objective_surrogate(seq_lbfgs)
        else:
            seq_lbfgs, val_lbfgs = seq_flat, val_flat

        # d) Simulated‑annealing polishing
        hill_deadline = sub_deadline - 0.3
        if time.time() < hill_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                                  n,
                                                  hill_deadline,
                                                  rng,
                                                  max_attempts=120_000,
                                                  start_sigma=0.0012,
                                                  T_start=0.009,
                                                  T_end=5e-9)
        else:
            seq_sa, val_sa = seq_lbfgs, val_lbfgs

        # Pick the best candidate from this chain
        candidates = [
            (seed_vec, _objective_surrogate(seed_vec)),
            (seq_eg, val_eg),
            (seq_flat, val_flat),
            (seq_lbfgs, val_lbfgs),
            (seq_sa, val_sa),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}/{top_k}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 5. Final long polishing on the global best
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long EG (tight active set) polishing ---")
        final_seq, final_val = _exp_gradient_opt_multi(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=6_000_000,
                                                       eta0=14.0,
                                                       tol_start=1e-5,
                                                       tol_end=1e-12)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final flattening stage ---")
        final_seq2, final_val2 = _flatten_gradient_opt(global_best_seq,
                                                       n,
                                                       deadline,
                                                       max_iters=1_500_000,
                                                       eta0=7.0)
        if final_val2 < global_best_val:
            global_best_seq = final_seq2
            global_best_val = final_val2
            print(f"[Final flatten] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final L‑BFGS smoothing (ultra‑high β) ---")
        final_seq3 = _lbfgs_smooth_optimize(global_best_seq,
                                            n,
                                            deadline,
                                            beta_schedule=[1e6, 2e6, 5e6, 1e7, 2e7, 5e7, 1e8],
                                            iters_per_beta=3000)
        final_val3 = _objective_surrogate(final_seq3)
        if final_val3 < global_best_val:
            global_best_seq = final_seq3
            global_best_val = final_val3
            print(f"[Final L‑BFGS] improved to {global_best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Final simulated‑annealing polish ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                         n,
                                                         deadline,
                                                         rng,
                                                         max_attempts=150_000,
                                                         start_sigma=0.0009,
                                                         T_start=4e-3,
                                                         T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # --------------------------------------------------------------
    # 6. Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
