def _ga_step(pop, fitness, rng, elite_frac=0.2, mutation_scale=0.1):
    # sort by fitness
    idx = np.argsort(fitness)
    pop_sorted = [pop[i] for i in idx]
    fit_sorted = [fitness[i] for i in idx]
    elite = pop_sorted[:int(elite_frac * len(pop))] # best individuals
    # produce new population
    new_pop = elite.copy()
    while len(new_pop) < len(pop):
        # select parents randomly from elite with probability maybe uniform
        parent1, parent2 = rng.choice(elite, size=2, replace=False)
        # crossover: weighted average of log vectors
        log1 = np.log(parent1 + 1e-30)
        log2 = np.log(parent2 + 1e-30)
        alpha = rng.random()
        child_log = alpha * log1 + (1 - alpha) * log2
        # mutation
        child_log += rng.normal(scale=mutation_scale, size=child_log.shape)
        child = np.exp(child_log - np.max(child_log))
        child = child / child.sum()
        new_pop.append(child)
    # maybe add some random mutants
    return new_pop
```

"""
Propose a step‑function (non‑negative heights) that minimises evaluate_sequence().
The algorithm consists of three stages:

1. **Diverse seeds** – analytic families (uniform, exponential, power, beta,
   piecewise‑linear) together with many Dirichlet draws.  All seeds are normalised
   to sum to 1 (a probability vector).

2. **Projected‑gradient descent (PGD)** – a simple sub‑gradient method on the
   exact max‑convolution.  The sub‑gradient of the active entry is
   `g_i = 2*a_{k-i}` (zero outside the support).  After each step the iterate
   is projected onto the probability simplex (Duchi et al.).  Multiple random
   restarts are performed; the best result of the allotted budget is kept.

3. **Exponential‑gradient polishing + permutation annealing** – a long EG run
   (mirror‑descent in log‑softmax space) refines the best PGD candidate, and a
   short simulated‑annealing permutation tweak may shave a little more.

All heavy work uses an FFT based convolution (O(n log n)).  The function
honours the global time budget (default 1000 s) and stores the final sequence
in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (length = len(x)+len(y)-1)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1):
       2·n·max(conv(seq,seq)).
       This is proportional to the value returned by evaluate_sequence().
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _piecewise_linear(n: int, num_breaks: int, rng: np.random.Generator) -> np.ndarray:
    points = np.sort(rng.choice(n, size=num_breaks, replace=False))
    points = np.concatenate(([0], points, [n - 1]))
    vals = rng.uniform(0.1, 1.0, size=len(points))
    xs = np.arange(n, dtype=np.float64)
    a = np.interp(xs, points, vals)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector v onto the simplex {x ≥ 0, sum x = 1}."""
    # shift by max for numeric stability
    u = np.sort(v)[::-1]                 # descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal – fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1              # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Projected‑gradient descent on the exact max‑convolution.
# ----------------------------------------------------------------------
def _proj_gradient_opt(start_seq: np.ndarray,
                       deadline: float,
                       max_iters: int = 2_000_000,
                       alpha0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient method.
    Operates directly on the probability vector (sum = 1, a_i ≥ 0).
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # convolution and index of the active diagonal (the current max)
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the active entry (see analysis)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size (diminishing)
        step = - (alpha0 / math.sqrt(it)) * g
        a = _proj_simplex(a + step)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient optimiser (exact max, mirror‑descent in log‑softmax)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int = 2_500_000,
                     eta0: float = 5.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Operates in log‑softmax space (a_i = exp(log_a_i) / Σ).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # gradient of the active entry
        n = len(a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a -= eta * g
        a = np.exp(log_a - np.max(log_a))      # improve numerical stability
        a /= a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple permutation‑based simulated annealing (swap moves only)
# ----------------------------------------------------------------------
def _perm_anneal(start_seq: np.ndarray,
                 deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pure permutation optimiser: repeatedly swaps two entries.
    Uses a linear cooling schedule.
    """
    n = len(start_seq)
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = cur_val
    T_end = cur_val * 1e-6

    rng = np.random.default_rng()

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = T_start + (T_end - T_start) * (it / max_iters)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        ns = cur_seq.copy()
        ns[i], ns[j] = ns[j], ns[i]

        val = _objective_surrogate(ns)
        if val < cur_val or (T > 0.0 and rng.random() < math.exp(-(val - cur_val) / T)):
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper: create an “alternating‐extremes” ordering of a given multiset.
# ----------------------------------------------------------------------
def _alternating_extreme_order(vec: np.ndarray) -> np.ndarray:
    """
    Re‑order entries by repeatedly taking the current largest and smallest
    remaining element.  This often decorrelates large values in the convolution.
    """
    idx = np.argsort(vec)
    low = idx.tolist()
    high = low[::-1]
    new_order = []
    l, h = 0, 0
    while l < len(low) and h < len(high):
        new_order.append(high[h])
        h += 1
        if l < len(low):
            new_order.append(low[l])
            l += 1
    return vec[new_order]

    seeds.append(_uniform(n))
    seeds.append(_exp_decay(n, rate=0.0015))
    seeds.append(_exp_decay(n, rate=0.0035))
    seeds.append(_sym_exp_decay(n, rate=0.0025))
    seeds.append(_power(n, p=0.6))
    seeds.append(_power(n, p=1.4))

    # Dirichlet draws covering a broad concentration range
    concentrations = (0.01, 0.03, 0.07, 0.15, 0.3, 0.7, 1.5, 3.0)
    for conc in concentrations:
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random piecewise‑linear approximations
    for nb in (3, 7, 12):
        for _ in range(2):
            seeds.append(_piecewise_linear(n, nb, rng))

    # beta‑shaped families
    for _ in range(10):
        a = rng.uniform(0.2, 2.5)
        b = rng.uniform(0.2, 2.5)
        seeds.append(_beta_shape(n, a, b))

    # previous best (if available) and a few random permutations of it
    if prev_vec is not None:
        seeds.append(prev_vec)
        for _ in range(5):
            seeds.append(rng.permutation(prev_vec))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of the seed pool
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Seed summary (surrogate values) ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Projected‑gradient descent with several random restarts
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Global best surrogate = {global_best_val:.6f}")

    # allocate roughly half of the remaining budget to PGD restarts
    pgd_budget = (deadline - time.time()) * 0.55
    pgd_deadline = time.time() + pgd_budget

    # we will run PGD on each of the top seeds (in order) and also a few
    # random Dirichlet restarts until time runs out.
    restart_counter = 0
    while time.time() < pgd_deadline:
        # choose a start point
        if restart_counter < len(top_seeds):
            start_vec = top_seeds[restart_counter]
        else:
            # extra random restarts – very low concentration Dirichlet to get
            # highly sparse candidates
            conc = rng.uniform(0.005, 0.05)
            start_vec = _random_dirichlet(n, conc, rng)

        # allocate a small slice (≈10 s) for each restart
        slice_time = min(12.0, pgd_deadline - time.time())
        sub_deadline = time.time() + slice_time

        seq_pgd, val_pgd = _proj_gradient_opt(start_vec,
                                              sub_deadline,
                                              max_iters=1_200_000,
                                              alpha0=0.9)

        if val_pgd < global_best_val:
            global_best_seq = seq_pgd
            global_best_val = val_pgd
            print(f"[PGD restart {restart_counter+1}] new best surrogate = {global_best_val:.6f}")

        restart_counter += 1

        # safety break if we have exhausted the slice time budget
        if time.time() >= pgd_deadline:
            break

    print(f"\n[PGD phase] best surrogate after restarts = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponential‑gradient polish (uses the remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        remaining = deadline - time.time()
        print("\n--- Final EG polishing (remaining budget) ---")
        final_seq, final_val = _exp_gradient_opt(global_best_seq,
                                                  deadline,
                                                  max_iters=3_500_000,
                                                  eta0=6.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[EG] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. A deterministic re‑ordering (alternating extremes) – cheap test
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_alt = _alternating_extreme_order(global_best_seq)
        val_alt = _objective_surrogate(seq_alt)
        if val_alt < global_best_val:
            global_best_seq = seq_alt
            global_best_val = val_alt
            print(f"[Alt order] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Very short permutation annealing (few seconds) – may shave a
    #    little more off the surrogate.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        short_deadline = time.time() + 22.0
        seq_perm, val_perm = _perm_anneal(global_best_seq.copy(),
                                          short_deadline,
                                          max_iters=120_000)
        if val_perm < global_best_val:
            global_best_seq = seq_perm
            global_best_val = val_perm
            print(f"[Final perm] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Evaluate with the provided function and store the result
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
