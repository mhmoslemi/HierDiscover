def _adam_smooth_opt(start_seq, n, deadline,
                     beta_schedule=None,
                     max_iters_per_beta=2000,
                     eta0=0.5):
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8

    if beta_schedule is None: 
        beta_schedule = [0.5, 1, 2,5,10,20,50,100,200,500,1000,2000,5000,10000,20000,50000,100000,200000,500000,1e6]
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    t_iter = 0
    for beta in beta_schedule:
        beta_val = float(beta)
        if time.time() > deadline:
            break

        for it in range(1, max_iters_per_beta+1):
            t_iter += 1
            if time.time() > deadline:
                break
            a = _softmax(log_a)
            conv = _conv_fft(a, a)
            shift = conv.max()
            # smooth max via log-sum-exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # objective (not needed)
            # weights for gradient
            w = exp_beta / sum_exp

            rev_a = a[::-1]
            # gradient of smooth max term
            full = _conv_fft(rev_a, w)
            grad_a = 2.0 * full[n-1:n-1+n]

            # Adam update in log-space:
            # gradient w.r.t. log_a: a * (grad_a - dot)
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
            m_hat = m / (1.0 - beta1**t_iter)
            v_hat = v / (1.0 - beta2**t_iter)
            step = eta0 / (np.sqrt(v_hat) + eps)
            log_a = log_a - step * m_hat

            # evaluate surrogate occasionally
            if it % 200 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
    final_a = _softmax(log_a)
    # final check
    cur_val = _objective_surrogate(final_a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = final_a.copy()
    return best_seq, best_val

"""
Advanced step‑function optimizer (alternative approach)

Searches for a non‑negative vector a (∑a=1) that minimizes
    evaluate_sequence(a) = 2·n·max(conv(a,a))
by directly minimizing a smooth approximation of the max‑convolution.
The algorithm works in the log‑softmax (exponentiated‑gradient) space,
uses an Adam/RMSProp style adaptive step size and a schedule of
increasing “β” to turn the smooth log‑sum‑exp approximation into the
true max.  Starting from the best known solution (or a uniform seed) it
performs many cheap FFT‑based gradient steps and finishes with a short
simulated‑annealing polish.

The routine respects a 1000 s wall‑clock budget and stores the best
sequence found in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Optional scientific imports (used only for seeding)
# --------------------------------------------------------------
try:
    import scipy.signal as _sigwin
except Exception:  # pragma: no cover
    _sigwin = None

# --------------------------------------------------------------
# Basic numeric utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (exact, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# --------------------------------------------------------------
# Surrogate objective (sum(a)=1)
# --------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(a,a)) with sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Analytic seeds (all ℓ₁‑normalised)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution density on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _exp_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    if s == 0.0:
        return _uniform(n)
    return a / s


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _dpss_window(n: int, nw: float) -> np.ndarray:
    """DPSS (Slepian) window – falls back to uniform if unavailable."""
    if _sigwin is None:
        return _uniform(n)
    a = _sigwin.dpss(n, nw, Kmax=1, sym=True)[0]
    a = np.abs(a)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# --------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution
# --------------------------------------------------------------
def _adam_smooth_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     beta_schedule=None,
                     inner_iters: int = 2500,
                     eta0: float = 0.7,
                     eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam / RMSProp optimiser on the smooth max‑convolution.
    The loss is 2·n·M_β where M_β = (1/β)·log∑exp(β·conv).
    Returns the best sequence encountered and its surrogate value.
    """
    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
            100.0, 200.0, 500.0, 1_000.0,
            2_000.0, 5_000.0, 10_000.0,
            20_000.0, 50_000.0, 100_000.0,
        ]

    a = np.clip(start_seq, 1e-30, None)         # avoid exact zero for log
    log_a = np.log(a)                           # log‑softmax parameters
    # RMSProp accumulator
    avg_sq = np.zeros_like(log_a)

    best_seq = _softmax(log_a)
    best_val = _objective_surrogate(best_seq)

    itr = 0
    for beta in beta_schedule:
        beta_val = float(beta)
        if time.time() > deadline:
            break
        for _ in range(inner_iters):
            itr += 1
            if time.time() > deadline:
                break

            a = _softmax(log_a)                 # enforce simplex
            conv = _conv_fft(a, a)
            shift = conv.max()                  # for numerical stability

            # smooth‑max via log‑sum‑exp
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            # weights for gradient (soft‑selection of active lags)
            w = exp_beta / sum_exp

            # gradient w.r.t. a of 2·n·M_β (the factor 2·n is constant)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)          # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]  # slice to length n

            # mirror‑descent (log‑softmax) conversion
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # RMSProp‑style adaptive step
            avg_sq = 0.9 * avg_sq + 0.1 * (grad_log ** 2)
            step = eta0 / (np.sqrt(avg_sq) + eps)
            log_a -= step * grad_log

            # occasional bookkeeping
            if itr % 300 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
    # final projection
    final_a = _softmax(log_a)
    cur_val = _objective_surrogate(final_a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = final_a.copy()
    return best_seq, best_val


# --------------------------------------------------------------
# Simple simulated‑annealing polish (accepts occasional worsenings)
# --------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 120_000,
                         start_sigma: float = 0.002,
                         T_start: float = 0.015,
                         T_end: float = 1e-9) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm builds a rich set of
    seed vectors, refines the most promising one with an Adam optimiser
    on a smooth approximation of the max‑convolution, and finally
    performs a short simulated‑annealing polish.
    The best sequence is stored in the global variable
    ``height_sequence_1`` and returned as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # ------------------------------------------------------------------
    # Problem size – larger n gives more flexibility; 4000 is a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        base_seq = np.asarray(previous, dtype=np.float64)
        base_seq = np.maximum(base_seq, 0.0)
        if base_seq.sum() > 0.0:
            base_seq /= base_seq.sum()
    else:
        base_seq = None

    # ------------------------------------------------------------------
    # 1. Build a diverse seed collection
    # ------------------------------------------------------------------
    print(f"\n--- Seeding (n={n}) ---")
    seeds = []

    seeds.append(_uniform(n))

    # Beta‑distribution shapes (including U‑shaped arcsine ≈ α=β=0.5)
    for a in np.linspace(0.1, 5.0, 10):
        for b in np.linspace(0.1, 5.0, 10):
            seeds.append(_beta_shape(n, a, b))

    # Exponential decays (cover a wide range)
    for rate in np.logspace(-5, -0.3, 25):
        seeds.append(_exp_decay(n, rate))

    # DPSS windows (different time‑bandwidth products)
    if _sigwin is not None:
        for nw in [2.5, 3.0, 3.5, 4.0, 4.5, 5.0, 6.0]:
            seeds.append(_dpss_window(n, nw))

    # Random Dirichlet draws (different concentration)
    for conc in [0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Include the previous best (if any) and slight perturbations thereof
    if base_seq is not None:
        seeds.append(base_seq)
        for _ in range(4):
            pert = base_seq + 0.004 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0.0:
                seeds.append(pert / pert.sum())

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation of all seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        tot = vec.sum()
        if tot == 0.0:
            continue
        vec = vec / tot
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\nTop seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed with Adam on smooth max‑convolution
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break

        # allocate a modest share of the remaining budget per seed
        remaining = deadline - time.time()
        share = remaining * 0.18 / max(1, top_k - idx)
        sub_deadline = time.time() + share

        print(f"\n[Seed {idx+1}/{top_k}] start Adam refinement")
        seq_adam, val_adam = _adam_smooth_opt(seed_vec.copy(),
                                              n,
                                              sub_deadline,
                                              inner_iters=3000,
                                              eta0=0.8)

        # short SA polish (if time permits)
        if time.time() < sub_deadline:
            seq_sa, val_sa = _simulated_annealing(seq_adam,
                                                  n,
                                                  sub_deadline,
                                                  rng,
                                                  max_attempts=50_000,
                                                  start_sigma=0.001,
                                                  T_start=0.01,
                                                  T_end=1e-9)
        else:
            seq_sa, val_sa = seq_adam, val_adam

        # pick the best among the candidates for this seed
        candidates = [(seed_vec, _objective_surrogate(seed_vec)),
                      (seq_adam, val_adam),
                      (seq_sa, val_sa)]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {idx+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:
            global_best_seq = local_best_seq
            global_best_val = local_best_val
            print(f"    *** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Final long‑run Adam refinement on the current global best
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final long Adam refinement ---")
        final_seq, final_val = _adam_smooth_opt(global_best_seq,
                                                n,
                                                deadline,
                                                inner_iters=8000,
                                                eta0=0.6)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final Adam] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Short SA polish (last chance)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Final simulated‑annealing polish ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                         n,
                                                         deadline,
                                                         rng,
                                                         max_attempts=120_000,
                                                         start_sigma=0.001,
                                                         T_start=5e-3,
                                                         T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
