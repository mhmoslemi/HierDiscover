"""
Improved search for a step‑function that minimises ``evaluate_sequence``.
The algorithm combines:

1. **Rich seed generation** – a dense grid of power‑law shapes
   a_i ∝ ((i+ε)(n‑ε‑i))^{-p} (including the classic arcsine case)
   together with a few Dirichlet draws and the previous best solution.

2. **Fast surrogate evaluation** – the exact objective 2·n·max(conv(a,a))
   computed by a real FFT based convolution.

3. **Adam optimisation** of the log‑softmax (mirrored) parametrisation,
   using a longer β‑schedule and a higher inner‑iteration budget than the
   reference implementation.

4. **Exact exponentiated‑gradient polishing** – sub‑gradient descent on the
   true max‑convolution, with a larger step‑size schedule.

5. **Convex mixing** – systematic convex combinations of the current best
   candidate with a few diverse seeds (uniform, arcsine, Dirichlet) to
   escape shallow local minima.

6. **Intensive random local search** – a stochastic walk on the simplex that
   first uses a relatively large step size and then narrows down, always
   retaining improvements.

All steps respect the supplied time budget (default 1000 s) and the final
sequence is stored in the global variable ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate 2·n·max(conv(seq,seq)).  `seq` must sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    return a / a.sum()


def _power_shape(n: int, p: float, eps: float = 0.5) -> np.ndarray:
    """a_i ∝ ((i+eps)*(n-eps-i))^{-p}  (symmetric, sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    denom = (i + eps) * (n - eps - i)
    # prevent zero denominator
    denom = np.where(denom <= 0.0, 1e-30, denom)
    a = denom ** (-p)
    return a / a.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser (log‑softmax) on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation in the log‑softmax space.
    Returns the best probability vector seen and its exact surrogate value.
    """
    # initialise logits (add tiny epsilon to avoid log(0))
    logit = np.log(start_seq + 1e-30)
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    # a more gradual β schedule – the last values are the same as before
    beta_schedule = np.geomspace(1.0, 1e7, num=25)

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        # more inner iterations per β (the previous code used 4000)
        max_iter_beta = 6000
        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of Σ w·conv w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)              # length = n+(2n-1)-1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)       # outer factor 2·n

            # transform to gradient on log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 8.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the current active index.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient for active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = eta0 / math.sqrt(it)
        log_a = log_a - eta * g
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light‑weight random local refinement (annealed σ)
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma_start: float = 0.08,
                         sigma_end: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex.  The step size σ linearly anneals
    from ``sigma_start`` to ``sigma_end`` over the allotted attempts.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for a in range(attempts):
        if time.time() > deadline:
            break

        # linear annealing of σ
        sigma = sigma_start - (sigma_start - sigma_end) * (a / attempts)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue

        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue

        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Convex mixing of two simplex vectors
# ----------------------------------------------------------------------


def _mix_sequences(a: np.ndarray, b: np.ndarray,
                   n: int,
                   deadline: float,
                   steps: int = 21) -> tuple[np.ndarray, float]:
    """
    Returns the best convex combination t∈[0,1] of a and b.
    The function evaluates a grid of ``steps`` points.
    """
    best_seq = a.copy()
    best_val = _objective_surrogate(best_seq)

    for t in np.linspace(0.0, 1.0, steps):
        cand = (1.0 - t) * a + t * b
        cand /= cand.sum()
        val = _objective_surrogate(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if time.time() > deadline:
            break
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a list of floats) and stores it in
    the global variable ``height_sequence_1`` for possible warm‑starts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a safety margin (filesystem I/O and printing)
    deadline = start_time + budget_s - 3.0

    # problem size (default 4000)
    n = int(kwargs.get("n", 4000))

    # ------------------------------------------------------------------
    # 0️⃣  Warm‑start from previous run (if any)
    # ------------------------------------------------------------------
    prev_vec = None
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_vec = arr / arr.sum()
        except Exception:
            prev_vec = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a rich seed pool
    # ------------------------------------------------------------------
    seeds = [_uniform(n), _triangular(n)]

    # classic arcsine (p=0.5, eps=0.5)
    seeds.append(_power_shape(n, p=0.5, eps=0.5))

    # grid of power‑law shapes (p in [0.2,0.65])
    for p in np.linspace(0.20, 0.65, 19):
        for eps in (0.2, 0.5, 0.8):
            seeds.append(_power_shape(n, p=float(p), eps=float(eps)))

    # exponential‑decay style (λ) – stored as a normalised vector
    for lam in (0.0005, 0.001, 0.002, 0.004, 0.008):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
        seeds.append(a / a.sum())

    # a few Dirichlet draws
    for conc in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous best (if available)
    if prev_vec is not None:
        seeds.append(prev_vec)

    # ------------------------------------------------------------------
    # 2️⃣  Initial evaluation – keep the top‑k seeds
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    top_k = 5
    top_seeds = []

    print("\n--- Initial seed evaluation ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        vec = vec / vec.sum()
        val = _objective_surrogate(vec)
        print(f" seed #{idx + 1:03d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()
        # keep a sorted list of the best few seeds for later multi‑start
        if len(top_seeds) < top_k:
            top_seeds.append((val, vec.copy()))
            top_seeds.sort(key=lambda x: x[0])
        elif val < top_seeds[-1][0]:
            top_seeds[-1] = (val, vec.copy())
            top_seeds.sort(key=lambda x: x[0])

    print(f"[Init] global best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Adam smoothing on the current global best (and maybe on the other
    #      top‑k seeds if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Adam smoothing (smooth max) ---")
        # allocate roughly half of the remaining budget to Adam
        remain = deadline - time.time()
        adam_deadline = time.time() + 0.48 * remain
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

        # optionally try Adam on the other top seeds (quick, few iterations)
        for val, seed_vec in top_seeds[1:]:
            if time.time() > adam_deadline:
                break
            # a very short run – only the first β and 1500 inner steps
            short_dead = time.time() + 6.0  # seconds budget per extra seed
            seq_tmp, val_tmp = _adam_smoothing(seed_vec, n, short_dead)
            if val_tmp < best_val:
                best_val, best_seq = val_tmp, seq_tmp.copy()
                print(f"[Adam‑extra] improved to {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Exact exponentiated‑gradient polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        polishing_deadline = deadline - 5.0
        # proportion of remaining time dedicated to polishing
        remaining = polishing_deadline - time.time()
        max_iters = int(2_500_000 * remaining / (deadline - start_time))
        max_iters = max(300_000, min(max_iters, 4_000_000))
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=8.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 5️⃣  Convex mixing with diverse seeds (systematic line search)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Convex mixing with other seeds ---")
        mix_deadline = deadline - 4.0
        mix_seeds = [_uniform(n), _power_shape(n, 0.5, 0.5),
                     _triangular(n)]
        # add a few Dirichlet samples for diversity
        for conc in (0.1, 0.5, 1.0):
            mix_seeds.append(_random_dirichlet(n, conc, rng))
        for other in mix_seeds:
            if time.time() > mix_deadline:
                break
            cand_seq, cand_val = _mix_sequences(best_seq, other, n,
                                                mix_deadline,
                                                steps=31)
            if cand_val < best_val:
                best_val, best_seq = cand_val, cand_seq.copy()
                print(f"[Mix] improvement to {cand_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final intensive random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=300_000,
                                                sigma_start=0.08,
                                                sigma_end=0.01)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 7️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for potential warm‑starts on the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
