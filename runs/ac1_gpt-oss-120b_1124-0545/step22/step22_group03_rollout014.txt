"""
Improved search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

The algorithm performs several restarts:

1.  Generate an initial candidate (previous best, uniform, Dirichlet,
    or a “random‑phase” vector obtained from the Fourier domain).
2.  Refine the candidate with a *smoothed* exponentiated‑gradient
    optimiser.  The non‑smooth max‑convolution is approximated by a
    soft‑max (log‑sum‑exp) with an increasing temperature β; the exact
    gradient of this surrogate is computed by a correlation via FFT.
3.  Polish the refined vector with a simple simulated‑annealing
    routine that moves a tiny amount of probability mass between two
    random indices.

The whole process respects the 1000 s budget, stores the best answer
in ``height_sequence_1`` for reuse, and prints intermediate bounds.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic numerical utilities (FFT‑based linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact when the vector sums to 1)
# ----------------------------------------------------------------------
def _surrogate_obj(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _random_phase_candidate(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a non‑negative vector from a random‑phase Fourier spectrum.
    The DC term is set to 1 so that the mean of the time‑domain signal is 1/n.
    Afterwards the signal is shifted to be non‑negative and ℓ₁‑normalised.
    """
    N = _next_pow2(2 * n)                     # power‑of‑two for fast FFT
    A = np.zeros(N, dtype=complex)
    A[0] = 1.0                                 # DC component

    half = N // 2
    phases = rng.random(half - 1) * 2.0 * np.pi
    # magnitude = 1 for all non‑DC frequencies
    A[1:half] = np.exp(1j * phases)
    A[N - np.arange(1, half)] = np.conj(A[1:half])   # conjugate symmetry

    if N % 2 == 0:                              # Nyquist term must be real
        A[half] = 0.0

    a_complex = np.fft.ifft(A).real           # time‑domain signal (real)
    a = a_complex[:n]

    # shift to make every entry ≥ 0
    min_a = a.min()
    if min_a < 0.0:
        a = a - min_a                          # put the most negative value at 0

    # normalization to ℓ₁ = 1  (evaluate_sequence is scale‑invariant)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


def _dirichlet_candidate(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration parameter ``alpha``."""
    a = rng.dirichlet(np.full(n, alpha))
    # Numerical safety – Dirichlet never returns negatives, but clip just in case.
    a = np.maximum(0.0, a)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient optimiser (log‑space, simplex)
# ----------------------------------------------------------------------
def _exp_gradient_smooth(start_seq: np.ndarray,
                         deadline: float,
                         eta0: float = 10.0,
                         beta_start: float = 0.5,
                         beta_factor: float = 1.12,
                         beta_max: float = 500.0,
                         max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on a *softened* max‑convolution.
    The soft‑max temperature β is increased slowly, giving a smoother
    approximation early on and converging to the true max later.
    Returns the best sequence (ℓ₁‑normalised) seen and its exact surrogate value.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)                 # stay in simplex via soft‑max

    best_seq = a.copy()
    best_val = _surrogate_obj(a)

    beta = beta_start
    it = 0
    while it < max_iters and time.time() < deadline:
        it += 1

        # exact convolution (needed for the soft‑max weights)
        conv = _conv_fft(a, a)                # length 2n‑1

        # soft‑max weights w_i ≈ ∂max_β/∂conv_i
        scaled = conv * beta
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)   # subtraction for stability
        w = exp_scaled / exp_scaled.sum()          # shape (2n‑1,)

        # gradient of Σ_i w_i·conv_i w.r.t. a:
        # g_j = 2 * Σ_i w_i * a_{i-j}  (indices out of bounds are ignored)
        a_rev = a[::-1]                       # reversed a
        corr = _conv_fft(w, a_rev)            # length 3n‑2
        # indices (n‑1‑j) give exactly the needed sum
        idx = (n - 1) - np.arange(n)
        g = 2.0 * corr[idx]

        # mirror‑descent step in log‑space
        eta = eta0 / math.sqrt(it)            # decaying learning rate
        log_a -= eta * g
        a = _softmax(log_a)                    # stays on the probability simplex

        # periodic exact evaluation (every 1000 iterations) – cheap enough
        if it % 1000 == 0:
            cur_val = _surrogate_obj(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()

        # increase β occasionally – makes the approximation tighter
        if it % 5000 == 0:
            beta = min(beta * beta_factor, beta_max)

    # final exact evaluation (in case the loop ended before a periodic check)
    cur_val = _surrogate_obj(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (tiny mass transfers)
# ----------------------------------------------------------------------
def _simulated_annealing_refine(start_seq: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator,
                                max_attempts: int = 600_000,
                                T_start: float = 0.02,
                                T_end: float = 1e-9,
                                sigma_start: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Simple SA that moves a small amount of probability mass between two
    random indices.  Temperature and step‑size decay linearly.
    Returns the best sequence seen.
    """
    n = start_seq.shape[0]
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_obj(a)
    cur_val = best_val

    attempts = 0
    while time.time() < deadline and attempts < max_attempts:
        # linear cooling schedule
        frac = attempts / max_attempts
        T = T_start * (1.0 - frac) + T_end * frac
        sigma = sigma_start * (T / T_start)          # sigma shrinks with T

        i = rng.integers(n)
        if a[i] <= 1e-15:                     # nothing to give away
            attempts += 1
            continue

        # amount to move: a fraction of the donor mass
        amt = sigma * rng.random() * a[i]
        if amt <= 0.0:
            attempts += 1
            continue

        j = rng.integers(n)
        if i == j:
            attempts += 1
            continue

        ns = a.copy()
        ns[i] -= amt
        ns[j] += amt

        val = _surrogate_obj(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point – respects the overall 1000 s budget
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a height sequence that minimises ``evaluate_sequence``.
    The routine re‑uses any previous solution stored in the global variable
    ``height_sequence_1`` and then performs several restarts with
    smoothed exponentiated‑gradient descent followed by simulated‑annealing.
    The best sequence found (as a list) is stored back into
    ``height_sequence_1`` and returned.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0               # small safety margin

    # ------------------------------------------------------------------
    # Initialise – use a previously found vector if available, otherwise uniform
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim != 1 or arr.size == 0:
                raise ValueError
            s = arr.sum()
            if s <= 0.0:
                raise ValueError
            seq0 = arr / s                      # ℓ₁‑normalise
            n = seq0.shape[0]
        except Exception:
            seq0 = None
            n = None
    else:
        seq0 = None
        n = None

    if seq0 is None:
        n = 4000
        seq0 = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    print(f"\n=== Optimisation started – n = {n} ===")
    print(f"Initial surrogate value: {_surrogate_obj(seq0):.6f}")

    # ------------------------------------------------------------------
    # Helper that runs one optimisation round (EG + SA) inside a time slice
    # ------------------------------------------------------------------
    def _optimise_once(start_vec: np.ndarray, slice_deadline: float) -> tuple[np.ndarray, float]:
        """Run EG (smoothed) then SA on ``start_vec`` bounded by ``slice_deadline``."""
        # EG – allocate ~70 % of the slice to gradient descent
        now = time.time()
        eg_deadline = now + max(5.0, 0.70 * (slice_deadline - now))
        seq_eg, val_eg = _exp_gradient_smooth(start_vec, eg_deadline)
        # SA – use the rest of the slice
        now = time.time()
        if now < slice_deadline - 1.0:
            sa_deadline = slice_deadline
            seq_sa, val_sa = _simulated_annealing_refine(seq_eg, sa_deadline, rng)
        else:
            seq_sa, val_sa = seq_eg, val_eg
        # pick the better of the two
        if val_sa < val_eg:
            return seq_sa, val_sa
        else:
            return seq_eg, val_eg

    # ------------------------------------------------------------------
    # Phase 0 – polish the previously known solution (if any)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        seq_opt, val_opt = _optimise_once(seq0, deadline)
        if val_opt < _surrogate_obj(seq0):
            print(f"[Polish] improved surrogate to {val_opt:.6f}")
        seq0, best_val = seq_opt, val_opt
    else:
        best_val = _surrogate_obj(seq0)

    best_seq = seq0.copy()

    # ------------------------------------------------------------------
    # Further random restarts (up to a few, limited by the remaining budget)
    # ------------------------------------------------------------------
    restart_idx = 0
    while time.time() < deadline and restart_idx < 6:
        remaining = deadline - time.time()
        if remaining < 10.0:
            break

        # decide generation method
        if restart_idx % 3 == 0:
            cand = _dirichlet_candidate(n, rng, alpha=0.3)   # mildly peaked
        elif restart_idx % 3 == 1:
            cand = _random_phase_candidate(n, rng)          # Fourier‑based
        else:
            cand = _dirichlet_candidate(n, rng, alpha=0.01)  # very spiky

        # run a short optimisation inside the allocated slice
        slice_deadline = time.time() + remaining * 0.6
        seq_opt, val_opt = _optimise_once(cand, slice_deadline)

        print(f"[Restart {restart_idx + 1}] surrogate = {val_opt:.6f}")

        if val_opt < best_val:
            best_val = val_opt
            best_seq = seq_opt.copy()
            print(f"  *** New global best after restart {restart_idx + 1}: {best_val:.6f}")

        restart_idx += 1

    # ------------------------------------------------------------------
    # Final true evaluation (may differ slightly because of rounding)
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\nTrue evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:   # pragma: no cover
        true_val = None
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store globally for possible reuse
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
