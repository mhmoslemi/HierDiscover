def _project_onto_simplex(v):
    """Project v onto probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    rg = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0

    n = int(kwargs.get('n', 4000))

    # generate seed list
    seeds = []

    # deterministic seeds
    seeds.append(_uniform(n))
    seeds.append(_arcsine(n))
    seeds.append(_triangular(n))

    # beta shaped seeds
    for alpha,beta in [(0.2,0.2),(0.3,0.3),(0.5,0.5),(0.8,0.8),(1.2,1.2)]:
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        pdf = np.power(x, alpha-1) * np.power(1-x, beta-1) + 1e-12
        pdf /= pdf.sum()
        seeds.append(pdf)

    # power-law seeds
    for p in [0.3,0.4,0.5,0.6]:
        i = np.arange(n, dtype=np.float64)
        a = 1.0 / np.power((i+0.5)*(n-0.5-i)+1e-12, p)
        a = a / a.sum()
        seeds.append(a)

    # Dirichlet draws low conc
    for conc in (0.01,0.02,0.05):
        for _ in range(3):
            seeds.append(rg.dirichlet(np.full(n, conc)))

    # previous solution if present
    prev = globals().get('height_sequence_1')
    if prev is not None:
        arr = np.asarray(prev, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # evaluate seeds
    best_seq = None
    best_val = np.inf
    print("\n--- Seed evaluation ---")
    for idx,seq in enumerate(seeds):
        if time.time()>deadline:
            break
        seq = seq / seq.sum()
        val = _objective_surrogate(seq)
        print(f" seed {idx+1:02d}: val {val:.6f}")
        if val<best_val:
            best_val=val
            best_seq=seq.copy()
    print(f"[seed] best val {best_val:.6f}")

    # Stage 1: Adam smoothing
    if best_seq is not None and time.time()<deadline:
        remain = deadline - time.time()
        adam_deadline = time.time() + remain*0.45  # allocate ~45% time
        print("\n--- Adam smoothing ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n, adam_deadline)
        print(f"[Adam] best val {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   updated global best (Adam)")

    # Stage 2: projected subgradient descent (PGD)
    if best_seq is not None and time.time()<deadline:
        pgd_deadline = deadline - 5.0
        print("\n--- Projected subgradient descent ---")
        seq_pgd, val_pgd = _projected_subgradient_opt(best_seq, n, pgd_deadline,
                                                      max_iters=400_000,
                                                      step0=0.5,
                                                      step_decay=0.9)
        print(f"[PGD] best val {val_pgd:.6f}")
        if val_pgd < best_val:
            best_val, best_seq = val_pgd, seq_pgd.copy()
            print("   updated global best (PGD)")

    # Stage 3: optionally symmetrize
    if best_seq is not None and time.time()<deadline:
        sym_seq = (best_seq + best_seq[::-1]) / 2.0
        sym_seq = sym_seq / sym_seq.sum()
        sym_val = _objective_surrogate(sym_seq)
        print(f"[symmetrize] val {sym_val:.6f}")
        if sym_val < best_val:
            best_val, best_seq = sym_val, sym_seq.copy()
            print("   updated global best (sym)")

    # Stage 4: final random local refinement
    if best_seq is not None and time.time()<deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=200_000,
                                                sigma=0.04)
        print(f"[Local] best val {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   updated global best (local)")

    # Evaluate true metric
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate {best_val:.6f}, true {true_val:.6f}")
    except Exception as e:
        print(f"Evaluation exception: {e}")

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()

"""
candidate‑search for the step‑function evaluation problem.

The algorithm builds a large, diverse pool of deterministic and random
probability‑vectors (uniform, arcsine, triangular, Beta‑shaped,
power‑law‑shaped, low‑concentration Dirichlet draws and the previously
found solution).  All vectors are normalised to sum = 1.

1.  The best seed is selected using the exact surrogate
    ``2·n·max(conv(a,a))`` (the same value returned by
    ``evaluate_sequence`` for a normalised vector).

2.  A long Adam‑optimisation is run on the **smooth** log‑sum‑exp
    approximation of the max‑convolution.  The schedule contains many
    β‑values up to 1e8 and each β performs a large number of inner
    iterations (``max_iter_beta`` can be increased by the time budget).

3.  A second phase uses exponentiated‑gradient (mirror‑descent) on the
    true max‑convolution, with a large iteration budget and an adaptive
    step size.

4.  The current best sequence is symmetrised (a←(a+a[::-1])/2) – the
    optimal solution is known to be symmetric – and re‑evaluated.

5.  A cheap stochastic walk on the simplex (tiny random local search)
    is performed with a large number of attempts, moving a small random
    amount of mass from one coordinate to another and keeping improvements.

6.  Finally the sequence is evaluated with the provided ``evaluate_sequence``
    function and stored in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector (sum = 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


def _objective_surrogate(seq: np.ndarray) -> float:
    """Exact surrogate used throughout the optimisation:
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine‑like density."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular distribution."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    a = np.concatenate([left, right]).astype(np.float64)
    a = a / a.sum()
    return a


def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on the index set."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    pdf = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    pdf = np.where(pdf > 0, pdf, 0.0)
    s = pdf.sum()
    if s == 0.0:
        return _uniform(n)
    return pdf / s


def _powerlaw_distribution(n: int, p: float) -> np.ndarray:
    """U‑shaped power‑law: a_i ∝ ((i+½)(n‑½‑i))^{-p}."""
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.power((i + 0.5) * (n - 0.5 - i) + eps, p)
    a = a / a.sum()
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the smooth max‑convolution surrogate
# ----------------------------------------------------------------------


def _adam_smoothing(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iter_beta: int = 8000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the log‑sum‑exp (smooth‑max) surrogate.
    ``max_iter_beta`` controls the number of inner iterations per β.
    Returns the best probability‑vector found and its exact surrogate value.
    """
    logit = np.log(start_seq + 1e-30)               # initialise logits
    m = np.zeros_like(logit)
    v = np.zeros_like(logit)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0

    beta_schedule = [
        1.0, 5.0, 20.0, 80.0,
        200.0, 500.0, 1000.0,
        2000.0, 5000.0,
        20000.0, 50000.0,
        100000.0, 200000.0,
        500000.0, 1_000_000.0, 2_000_000.0,
        5_000_000.0, 10_000_000.0,
        20_000_000.0, 50_000_000.0,
        100_000_000.0,
    ]

    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    for beta in beta_schedule:
        if time.time() > deadline:
            break

        for _ in range(max_iter_beta):
            if time.time() > deadline:
                break

            a = _softmax(logit)                     # current probability vector
            conv = _conv_fft(a, a)
            max_c = np.max(conv)

            # smooth‑max (log‑sum‑exp) weights
            exp_beta = np.exp(beta * (conv - max_c))
            w = exp_beta / exp_beta.sum()

            # gradient of the weighted sum of convolutions w.r.t. a
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = n + (2n‑1) - 1
            grad_weighted = 2.0 * full[n - 1:n - 1 + n]   # d/d a of Σ w·conv
            grad_a = grad_weighted * (2.0 * n)            # outer factor 2·n

            # convert to gradient on the log‑softmax (mirror) space
            dot = np.dot(grad_a, a)
            grad_log = a * (grad_a - dot)

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            lr = 0.15
            logit = logit - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if it % 800 == 0:
                cur_val = _objective_surrogate(a)
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()

    # final check
    final_a = _softmax(logit)
    final_val = _objective_surrogate(final_a)
    if final_val < best_val:
        best_seq, best_val = final_a, final_val

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact exponentiated‑gradient polishing (sub‑gradient of the true max)
# ----------------------------------------------------------------------


def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 2_000_000,
                     eta0: float = 7.0) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution.
    Uses the sub‑gradient at the current active index.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # active sub‑gradient
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        eta = eta0 / math.sqrt(it)
        # exponentiated‑gradient update: a ∝ a * exp(-η g)
        a = a * np.exp(-eta * g)
        a = a / a.sum()

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny random local refinement
# ----------------------------------------------------------------------


def _local_random_search(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         attempts: int = 200_000,
                         sigma: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Stochastic walk on the simplex: repeatedly move a small random amount
    from one coordinate to another and keep improvements.
    """
    rng = np.random.default_rng()
    best_seq = start_seq.copy()
    best_val = _objective_surrogate(best_seq)

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = sigma * best_seq[i] * rng.random()
        if delta <= 0.0:
            continue
        new_seq = best_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        if new_seq[i] < 0.0:
            continue
        val = _objective_surrogate(new_seq)
        if val < best_val:
            best_seq = new_seq
            best_val = val

    return best_seq, best_val

        remain = deadline - time.time()
        # Spend roughly 45 % of the remaining budget on this stage
        adam_deadline = time.time() + remain * 0.45
        print("\n--- Adam smoothing (smooth max) ---")
        seq_adam, val_adam = _adam_smoothing(best_seq, n,
                                             adam_deadline,
                                             max_iter_beta=8000)
        print(f"[Adam] best exact surrogate = {val_adam:.6f}")
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print("   >>> Updated global best (Adam)")

    # ------------------------------------------------------------------
    # 2️⃣  Exact exponentiated‑gradient polishing (sub‑gradient of true max)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        polishing_deadline = deadline - 5.0
        remaining = polishing_deadline - time.time()
        # Allocate the remainder proportionally to the number of allowed steps
        max_iters = int(3_000_000 * remaining / (deadline - start_time))
        max_iters = max(300_000, min(max_iters, 5_000_000))
        print("\n--- Exponentiated‑gradient polishing (exact max) ---")
        seq_eg, val_eg = _exp_gradient_opt(best_seq, n,
                                           polishing_deadline,
                                           max_iters=max_iters,
                                           eta0=7.0)
        print(f"[EG] exact surrogate = {val_eg:.6f}")
        if val_eg < best_val:
            best_val, best_seq = val_eg, seq_eg.copy()
            print("   >>> Updated global best (EG)")

    # ------------------------------------------------------------------
    # 3️⃣  Symmetrisation (the optimum is symmetric)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        sym_seq = (best_seq + best_seq[::-1]) / 2.0
        sym_seq = sym_seq / sym_seq.sum()
        sym_val = _objective_surrogate(sym_seq)
        print(f"[Symmetrisation] surrogate = {sym_val:.6f}")
        if sym_val < best_val:
            best_val, best_seq = sym_val, sym_seq.copy()
            print("   >>> Updated global best (symmetrised)")

    # ------------------------------------------------------------------
    # 4️⃣  Final tiny random local refinement
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Final random local refinement ---")
        seq_loc, val_loc = _local_random_search(best_seq, n, deadline,
                                                attempts=300_000,
                                                sigma=0.04)
        print(f"[Local] exact surrogate = {val_loc:.6f}")
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print("   >>> Updated global best (local)")

    # ------------------------------------------------------------------
    # 5️⃣  Official evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, "
              f"evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
