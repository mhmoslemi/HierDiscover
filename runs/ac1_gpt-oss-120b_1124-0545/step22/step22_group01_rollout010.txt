def _smooth_exp_gradient_opt(start_seq, n, deadline, max_iters=300_000,
                            eta0=0.5, beta1=0.9, beta2=0.999, eps=1e-8,
                            alpha0=0.5, alpha_max=30.0):
    a = start_seq.copy()
    # Ensure sum=1
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / s
    # initialize log parameters
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        a = _softmax(log_a)   # recalc each iteration
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        # compute loss surrogate = max_c (or 2*n*max_c), we need gradient of max_c: will compute smoothed gradient using softmax weights.
        # compute smoothing parameter alpha: linear schedule

        # compute weight vector w = softmax(alpha * conv)

        w = exp_term / np.sum(exp_term)

"""
Improved step‑function optimiser (smooth‑gradient version).

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a))² ,   a_i ≥ 0 .

The function is scale‑invariant, therefore we can always normalise a to
∑a = 1 and minimise the surrogate 2·n·max(conv(a,a)).  The optimiser

1️⃣ builds a large pool of analytic seeds (uniform, exponential, power‑law,
   beta‑shapes, Dirichlet draws, …),
2️⃣ keeps the best few seeds,
3️⃣ refines each seed with a *smooth* mirror‑descent optimiser:
   the non‑smooth max is replaced by a soft‑max (temperature α) so that
   the whole gradient becomes a weighted sum of all convolution terms.
   The update is performed in log‑space (exponentiated gradient) using
   Adam‑style adaptive step sizes.
4️⃣ after the per‑seed phase a long polishing run is carried out on the
   current global best,
5️⃣ occasional random perturbations are tried to escape shallow minima,
6️⃣ finally the best sequence is evaluated with ``evaluate_sequence`` and
   stored in the global variable ``height_sequence_1`` for continuation.

The smooth optimiser is able to move through non‑smooth regions of the
landscape and, with a schedule that slowly increases the soft‑max
sharpness α, typically finds a lower maximal auto‑convolution than the
pure sub‑gradient method used previously.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (always sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    return e / s


# ----------------------------------------------------------------------
# Surrogate objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Return 2·n·max(conv(seq, seq)) for a vector whose sum is 1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian_shift(n: int, sigma_frac: float, mu_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = mu_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_exp_decay(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-rate * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(conc,…,conc). Guarantees strict positivity."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smooth mirror‑descent optimiser (Adam in log‑space)
# ----------------------------------------------------------------------
def _smooth_exp_gradient_opt(start_seq: np.ndarray,
                             n: int,
                             deadline: float,
                             max_iters: int = 300_000,
                             eta0: float = 0.5,
                             alpha0: float = 0.5,
                             alpha_max: float = 30.0) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the *smoothed* max‑convolution.
    The smoothing uses a soft‑max with temperature α, slowly increased from
    ``alpha0`` to ``alpha_max``.  All updates are performed in log‑space,
    guaranteeing non‑negativity and ℓ₁‑normalisation (via a softmax).
    Returns the best sequence seen and its surrogate objective value.
    """
    a = np.maximum(start_seq, 0.0)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s

    # log‑parameters (softmax enforces sum=1 and positivity)
    log_a = np.log(a + 1e-30)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # Current probability vector
        a = _softmax(log_a)

        # Convolution a*a
        conv = _conv_fft(a, a)

        # Current smoothing parameter (linear schedule)
        alpha = alpha0 + (alpha_max - alpha0) * min(1.0, it / max_iters)

        # Soft‑max weights over all convolution entries
        max_c = np.max(conv)
        exp_term = np.exp(alpha * (conv - max_c))   # stable
        w = exp_term / np.sum(exp_term)            # shape (2n‑1,)

        # Gradient of the weighted sum  Σ_k w_k conv_k
        a_rev = a[::-1]
        conv_w = _conv_fft(w, a_rev)               # length 3n‑2
        G = conv_w[n - 1:n - 1 + n]                 # vector of length n

        # Diagonal correction (terms where i = k‑i)
        diag = w[2 * np.arange(n)] * a

        grad = 2.0 * G - diag                      # ∇_a Σ w_k conv_k

        # Convert to gradient in log‑space (mirror descent)
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)

        # Adam update on log‑parameters
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        log_a = log_a - eta0 * m_hat / (np.sqrt(v_hat) + eps)

        # Surrogate objective for this iteration
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Original exponentiated‑gradient optimiser (kept for comparison)
# ----------------------------------------------------------------------
def _exp_gradient_opt(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 300_000,
                     eta0: float = 5.0):
    a = np.maximum(start_seq, 0.0)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    log_a = np.log(a + 1e-30)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # exact sub‑gradient of max(conv) w.r.t. a
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        diag_mask = idx_i == rev_idx
        if np.any(diag_mask):
            g[idx_i[diag_mask]] = a[idx_i[diag_mask]]

        eta = eta0 / math.sqrt(it)
        dot = np.dot(g, a)
        grad_log = a * (g - dot)
        log_a = log_a - eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Permutation‑based hill‑climbing (kept from the previous version)
# ----------------------------------------------------------------------
def _perm_hill_climb(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     rng: np.random.Generator,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    cur_seq = start_seq.copy()
    cur_val = _objective_surrogate(cur_seq)
    best_seq = cur_seq.copy()
    best_val = cur_val

    T_start = 1e-3 * cur_val
    T_end = 1e-6 * cur_val
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        T = T_start + (T_end - T_start) * (it / max_iters)

        if rng.random() < 0.55:  # swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            ns = cur_seq.copy()
            ns[i], ns[j] = ns[j], ns[i]
        else:  # Gaussian perturbation
            sigma = 0.001 / math.sqrt(it + 1)
            ns = cur_seq + sigma * rng.normal(size=n)
            ns = np.maximum(ns, 0.0)
            s = ns.sum()
            if s == 0.0:
                continue
            ns = ns / s

        val = _objective_surrogate(ns)

        if val < cur_val:
            cur_seq = ns
            cur_val = val
            if val < best_val:
                best_seq = ns.copy()
                best_val = val
        else:
            prob = math.exp(-(val - cur_val) / T) if T > 0 else 0.0
            if rng.random() < prob:
                cur_seq = ns
                cur_val = val

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds a large pool of analytical seeds,
    keeps the best few, refines each with permutation hill‑climbing,
    a classic exponentiated‑gradient, *and* a new smooth Adam‑based mirror‑descent.
    The current global best is polished until the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4000 provides a good trade‑off between expressivity
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous solution if it exists (useful for continuation)
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_arr = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
        except Exception:
            prev_arr = None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # 1. Assemble a large pool of seed vectors
    # ------------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))

    # sine windows – sweep gamma
    for gamma in np.linspace(0.2, 4.0, 60):
        seeds.append(_sine_window(n, gamma))

    # exponentials – wide log‑space sweep
    for rate in np.logspace(-7, 1.5, 80):
        seeds.append(_exp_decay(n, rate))
        seeds.append(_sym_exp_decay(n, rate))

    # power‑laws
    for p in np.linspace(0.0, 4.0, 80):
        seeds.append(_power(n, p))

    # beta‑shapes – go down to 0.01
    alphas = np.geomspace(0.01, 10.0, 24)
    betas = np.geomspace(0.01, 10.0, 24)
    for a in alphas:
        for b_ in betas:
            seeds.append(_beta_shape(n, a, b_))

    # random Dirichlet draws with many concentrations
    for conc in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(10):
            seeds.append(_random_dirichlet(n, conc, rng))

    # previous solution and noisy copies
    if prev_arr is not None:
        seeds.append(prev_arr)
        for _ in range(6):
            noisy = prev_arr + 0.005 * rng.normal(size=n)
            seeds.append(np.maximum(noisy, 0.0))

    # ------------------------------------------------------------------
    # 2. Quick surrogate evaluation – keep the best K seeds
    # ------------------------------------------------------------------
    evaluated = []
    for vec in seeds:
        if time.time() > deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n--- Top seed surrogates ---")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f" #{i+1}: {v:.6f}")

    # ------------------------------------------------------------------
    # 3. Initialise global best
    # ------------------------------------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] global best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 4. Refine each top seed
    # ------------------------------------------------------------------
    for si, seed_vec in enumerate(top_seeds):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        share = max(12.0, remaining * 0.10 / max(1, (top_k - si)))
        sub_deadline = time.time() + share
        print(f"\n[Seed {si+1}/{top_k}] allocated ≈ {share:.1f}s")

        # a) permutation hill‑climb (quick)
        seq_hc, _ = _perm_hill_climb(seed_vec, n,
                                     sub_deadline,
                                     rng,
                                     max_iters=150_000)

        # b) classic exponentiated‑gradient
        seq_exp, val_exp = _exp_gradient_opt(seed_vec, n,
                                             sub_deadline,
                                             max_iters=400_000,
                                             eta0=5.0)

        # c) smooth Adam‑based mirror descent (our new method)

                                                          sub_deadline,
                                                          max_iters=600_000,
                                                          eta0=0.5,
                                                          alpha0=0.5,
                                                          alpha_max=30.0)

                                                               sub_deadline,
                                                               max_iters=300_000,
                                                               eta0=0.5,
                                                               alpha0=0.5,
                                                               alpha_max=30.0)

        # pick the best of all locally generated candidates

            (seq_hc, _objective_surrogate(seq_hc)),
            (seq_exp, val_exp),
            (seq_smooth, val_smooth),
            (seq_smooth_hc, val_smooth_hc),
        ]
        local_best_seq, local_best_val = min(candidates, key=lambda x: x[1])

        print(f"[Seed {si+1}] local best surrogate = {local_best_val:.6f}")

        if local_best_val < global_best_val:

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    if time.time() < deadline:

                                                        alpha0=0.5,
                                                        alpha_max=40.0)
        if final_val < global_best_val:

    while time.time() < deadline - 5.0:

                                                    sub_deadline,
                                                    max_iters=400_000,
                                                    eta0=0.5,
                                                    alpha0=0.5,
                                                    alpha_max=30.0)
        if val_tmp < global_best_val:
            global_best_seq, global_best_val = seq_tmp, val_tmp
            print(f"[Escape] new best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 7. Official evaluation via the provided function
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # store for possible continuation runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
