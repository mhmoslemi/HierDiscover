def _two_gaussian(n, sigma_frac, offset_frac1, offset_frac2):
    sigma = sigma_frac * n
    i = np.arange(n, dtype=np.float64)
    mu1 = offset_frac1 * (n-1)
    mu2 = offset_frac2 * (n-1)
    a = np.exp(-((i - mu1)**2) / (2 * sigma**2)) + np.exp(-((i - mu2)**2) / (2 * sigma**2))
    return a / a.sum()

def _refine_sequence(seq, n, deadline, rng):
    # EG
    seq_eg, val_eg = _exp_gradient_opt_full(seq.copy(), n, deadline, max_iters=400_000, eta0=7.0)
    # LBFGS
    if time.time() < deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(seq_eg, n, deadline,
                                           beta_schedule=[...],
                                           max_iter_per_beta=4000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg
    # SA
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(seq_lbfgs, n, deadline, rng,
                                              max_attempts=250_000,
                                              start_sigma=0.001,
                                              T_start=0.01,
                                              T_end=1e-9)
    else:
        seq_sa, val_sa = seq_lbfgs, val_lbfgs
    # pick best
    candidates = [(seq, _objective_surrogate(seq)), (seq_eg, val_eg), (seq_lbfgs, val_lbfgs), (seq_sa, val_sa)]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val

"""
Propose a step‑function (non‑negative sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where conv is the full linear convolution (np.convolve).  

The optimiser works in two stages:

1. **Broad seeding** – generate many analytic families (uniform,
   sine‑window, Beta, exponential, Gaussian, two‑Gaussian mixtures,
   power‑law) and random Dirichlet draws.  All candidates are
   ℓ₁‑normalised and evaluated with the cheap surrogate
   `2·n·max(conv(a,a))` (identical to the true score when the sum is 1).

2. **Local refinement** – for the best seeds a three‑step
   optimiser is applied:
        * exponentiated‑gradient (mirror‑descent on the exact max‑convolution
          using the full active set);
        * a smooth‑max L‑BFGS pass (log‑sum‑exp surrogate) with an
          extended β‑schedule;
        * a high‑budget simulated‑annealing sweep.
   The current global best is updated whenever an improvement is found.
   Finally a long EG polishing and a final SA sweep are performed.

The routine respects a hard 1000 s budget (leaving a small safety margin).
If a previous solution exists it is up‑scaled and re‑optimised, otherwise
a new search starts at length 4096.  Larger length 8192 is attempted if time
remains.  The best sequence is stored in the global variable
`height_sequence_1` and returned as a plain list of floats.
"""

import time
import math
import numpy as np

# optional SciPy for L‑BFGS
try:
    import scipy.optimize as _opt
except Exception:  # pragma: no cover
    _opt = None


# ----------------------------------------------------------------------
# Numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable soft‑max (returns a probability vector that sums to 1)."""
    m = np.max(v)
    e = np.exp(v - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return e / s


# ----------------------------------------------------------------------
# Objective (exact for ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _objective_surrogate(a: np.ndarray) -> float:
    """2·n·max(conv(a, a)) assuming sum(a)=1."""
    n = len(a)
    conv = _conv_fft(a, a)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families – all return ℓ₁‑normalised vectors
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_left(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_right(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * (n - 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float, centre_frac: float = 0.5) -> np.ndarray:
    sigma = sigma_frac * n
    mu = centre_frac * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _two_gaussian(n: int, sigma_frac: float,
                  offset1: float, offset2: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu1 = offset1 * (n - 1)
    mu2 = offset2 * (n - 1)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu1) ** 2) / (2.0 * sigma ** 2))
    a += np.exp(-((i - mu2) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_law(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else a.copy()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv(a,a)) – full active set
# ----------------------------------------------------------------------
def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """Averaged sub‑gradient of max(conv(a,a)) w.r.t. a."""
    n = len(a)
    conv = _conv_fft(a, a)
    max_val = conv.max()
    eps = max_val * 1e-12
    active = (conv >= max_val - eps).astype(np.float64)          # length 2n‑1
    cnt = active.sum()
    if cnt == 0:
        return np.zeros_like(a)

    rev_a = a[::-1]
    # Convolution of active mask with rev_a gives gradient contributions
    full = _conv_fft(active, rev_a)          # length 3n‑2
    grad = 2.0 * full[n - 1:n - 1 + n] / cnt
    return grad


# ----------------------------------------------------------------------
# Exponentiated‑gradient (full‑active) optimiser
# ----------------------------------------------------------------------
def _exp_gradient_opt_full(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 6.0):
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution,
    using the full active set.
    Returns best sequence and its surrogate value.
    """
    a = start_seq.copy()
    log_a = np.log(a + 1e-30)
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        grad = _subgradient_max_conv(a)
        dot = np.dot(grad, a)
        grad_log = a * (grad - dot)          # chain rule for log‑softmax
        eta = eta0 / math.sqrt(it)
        log_a -= eta * grad_log
        a = _softmax(log_a)

        cur_val = _objective_surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS optimisation (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _lbfgs_smooth_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           beta_schedule=None,
                           max_iter_per_beta: int = 4000):
    """
    Optimise the smoothed max‑convolution using L‑BFGS‑B.
    Works in log‑softmax space, guaranteeing non‑negativity and ℓ₁‑norm.
    """
    if _opt is None:
        # SciPy not present – return the start vector unchanged
        return start_vec

    if beta_schedule is None:
        beta_schedule = [
            0.5, 1.0, 5.0, 20.0, 80.0,
            200.0, 500.0, 1_000.0, 2_000.0, 5_000.0,
            20_000.0, 50_000.0, 100_000.0, 200_000.0,
            500_000.0, 1_000_000.0, 2_000_000.0, 5_000_000.0,
            1e7, 2e7
        ]

    v = np.log(start_vec + 1e-30)               # log‑softmax variables

    for beta in beta_schedule:
        if time.time() > deadline:
            break
        beta_val = float(beta)

        def fun(v_local):
            a = _softmax(v_local)
            conv = _conv_fft(a, a)
            shift = conv.max()
            exp_beta = np.exp(beta_val * (conv - shift))
            sum_exp = exp_beta.sum()
            obj = 2.0 * n * (shift + np.log(sum_exp) / beta_val)

            # gradient w.r.t a (smooth max)
            w = exp_beta / sum_exp
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            grad_a = 2.0 * full[n - 1:n - 1 + n]

            # chain‑rule for log‑softmax
            dot = np.dot(grad_a, a)
            grad_v = a * (grad_a - dot)
            return obj, grad_v

        def callback(_):
            if time.time() > deadline:
                raise KeyboardInterrupt()

        try:
            res = _opt.minimize(fun=fun,
                                x0=v,
                                method='L-BFGS-B',
                                jac=True,
                                callback=callback,
                                options={'maxiter': max_iter_per_beta,
                                         'disp': False,
                                         'ftol': 1e-12})
            v = res.x
        except KeyboardInterrupt:
            break
        except Exception as e:   # pragma: no cover
            print(f"[L‑BFGS] exception {e} at β={beta_val}")
            break

    return _softmax(v)


# ----------------------------------------------------------------------
# Simulated‑annealing refinement
# ----------------------------------------------------------------------
def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_attempts: int = 250_000,
                         start_sigma: float = 0.001,
                         T_start: float = 0.01,
                         T_end: float = 1e-9):
    """
    Simple SA: Gaussian perturbations with a linear temperature schedule.
    Returns the best sequence seen before the deadline.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)
    cur_val = best_val
    attempts = 0

    while time.time() < deadline and attempts < max_attempts:
        T = T_start * (1.0 - attempts / max_attempts) + T_end * (attempts / max_attempts)
        sigma = start_sigma * (T / T_start)

        ns = a + sigma * rng.normal(size=n)
        ns = np.maximum(ns, 0.0)
        s = ns.sum()
        if s == 0.0:
            attempts += 1
            continue
        ns /= s

        val = _objective_surrogate(ns)
        delta = val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / (T + 1e-12)):
            a = ns
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = ns.copy()
        attempts += 1

    return best_seq, best_val

# Seed generation
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator, prior_seq: np.ndarray | None):
    """Create a broad list of candidate sequences (ℓ₁‑normalised)."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    for gamma in (0.2, 0.5, 1.0, 2.0):
        seeds.append(_sine_window(n, gamma=gamma))

    # beta shapes – a grid including very small alphas/betas
    alphas = np.concatenate([np.logspace(-2, -0.3, 6), np.linspace(0.2, 4.0, 7)])
    betas = alphas.copy()
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # exponential decays (both sides)
    rates = np.logspace(-4, -0.2, 20)
    for r in rates:
        seeds.append(_exp_decay_left(n, r))
        seeds.append(_exp_decay_right(n, r))

    # Gaussian family (different sigmas, centre positions)
    for sigma_frac in (0.08, 0.15, 0.25):
        for centre_frac in (0.3, 0.5, 0.7):
            seeds.append(_gaussian(n, sigma_frac=sigma_frac, centre_frac=centre_frac))

    # Two‑Gaussian mixtures (symmetric, off‑centre)
    for sigma_frac in (0.07, 0.12, 0.2):
        seeds.append(_two_gaussian(n, sigma_frac, 0.25, 0.75))
        seeds.append(_two_gaussian(n, sigma_frac, 0.15, 0.85))

    # Power‑law tails
    for p in (0.5, 1.0, 2.0, 3.5):
        seeds.append(_power_law(n, p))

    # Mixtures of exponential left/right with uniform
    for r in rates[::5]:
        exp_l = _exp_decay_left(n, r)
        exp_r = _exp_decay_right(n, r)
        for alpha in (0.2, 0.5, 0.8):
            seeds.append(_mix(_uniform(n), exp_l, alpha))
            seeds.append(_mix(_uniform(n), exp_r, alpha))

    # Random Dirichlet draws – explore a wide concentration range
    for conc in (0.001, 0.005, 0.01, 0.05, 0.2, 0.6):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Perturb prior solution (if supplied)
    if prior_seq is not None and prior_seq.sum() > 0.0:
        base = prior_seq.copy()
        seeds.append(base)
        for _ in range(6):
            pert = base + 0.003 * rng.normal(size=n)
            pert = np.maximum(pert, 0.0)
            if pert.sum() > 0.0:
                seeds.append(pert / pert.sum())

    return seeds

# Refinement of a single seed
# ----------------------------------------------------------------------
def _refine_sequence(seq: np.ndarray,
                     n: int,
                     sub_deadline: float,
                     rng: np.random.Generator):
    """Run EG → L‑BFGS → SA on one seed and return the best result."""
    if time.time() > sub_deadline:
        return seq, _objective_surrogate(seq)

    # 1) Exponentiated‑gradient (full active set)
    seq_eg, val_eg = _exp_gradient_opt_full(seq.copy(),
                                            n,
                                            sub_deadline,
                                            max_iters=400_000,
                                            eta0=7.0)

    # 2) L‑BFGS smoothing (if time permits)
    if time.time() < sub_deadline:
        seq_lbfgs = _lbfgs_smooth_optimize(seq_eg,
                                           n,
                                           sub_deadline,
                                           beta_schedule=None,
                                           max_iter_per_beta=4000)
        val_lbfgs = _objective_surrogate(seq_lbfgs)
    else:
        seq_lbfgs, val_lbfgs = seq_eg, val_eg

    # 3) Simulated annealing (final sweep)
    if time.time() < sub_deadline:
        seq_sa, val_sa = _simulated_annealing(seq_lbfgs,
                                              n,
                                              sub_deadline,
                                              rng,
                                              max_attempts=250_000,
                                              start_sigma=0.001,
                                              T_start=0.012,
                                              T_end=5e-10)
    else:
        seq_sa, val_sa = seq_lbfgs, val_lbfgs

    candidates = [
        (seq,      _objective_surrogate(seq)),
        (seq_eg,   val_eg),
        (seq_lbfgs, val_lbfgs),
        (seq_sa,   val_sa)
    ]
    best_seq, best_val = min(candidates, key=lambda x: x[1])
    return best_seq, best_val

# Main search routine for a given length
# ----------------------------------------------------------------------
def _search_length(n: int,
                  rng: np.random.Generator,
                  prior_seq: np.ndarray | None,
                  overall_deadline: float):
    """
    Perform the 3‑stage optimisation for a fixed length n.
    Returns the best sequence (ℓ₁‑normalised) and its surrogate value.
    """
    t_start = time.time()
    print(f"\n[Info] Starting search for length n={n}")

    # ---- 1. Broad seeding ------------------------------------------------
    seeds = _generate_seeds(n, rng, prior_seq)

    # ---- 2. Fast surrogate evaluation of all seeds -----------------------
    evaluated = []
    for vec in seeds:
        if time.time() > overall_deadline:
            break
        vec = np.maximum(vec, 0.0)
        s = vec.sum()
        if s == 0.0:
            continue
        vec = vec / s
        val = _objective_surrogate(vec)
        evaluated.append((vec, val))

    if not evaluated:
        raise RuntimeError("No valid seeds generated.")

    evaluated.sort(key=lambda x: x[1])
    top_k = min(15, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print("\n[Seed scan] top surrogate values (first few):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1:2d}: {v:.6f}")

    # ---- 3. Initialise global best ---------------------------------------
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"[Init] best surrogate = {global_best_val:.6f}")

    # ---- 4. Local refinement of the top seeds ----------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() >= overall_deadline:
            break
        # allocate a modest slice of the remaining budget per seed
        remaining = overall_deadline - time.time()
        per_seed = max(5.0,
                       remaining * 0.12 / max(1, (top_k - idx)))
        sub_deadline = time.time() + per_seed

        seq_refined, val_refined = _refine_sequence(seed_vec, n, sub_deadline, rng)
        print(f"[Seed {idx+1}/{top_k}] refined surrogate = {val_refined:.6f}")

        if val_refined < global_best_val:
            global_best_seq = seq_refined
            global_best_val = val_refined
            print(f"    *** New global best = {global_best_val:.6f}")

    # ---- 5. Final polishing (high‑budget EG + SA) -----------------------
    if time.time() < overall_deadline:
        print("\n--- Final intensive EG polishing (full‑active) ---")
        final_seq, final_val = _exp_gradient_opt_full(global_best_seq,
                                                      n,
                                                      overall_deadline,
                                                      max_iters=2_500_000,
                                                      eta0=6.0)
        if final_val < global_best_val:
            global_best_seq = final_seq
            global_best_val = final_val
            print(f"[Final EG] improved to {global_best_val:.6f}")

    if time.time() < overall_deadline:
        print("\n--- Final SA polishing (high‑budget) ---")
        final_sa_seq, final_sa_val = _simulated_annealing(global_best_seq,
                                                          n,
                                                          overall_deadline,
                                                          rng,
                                                          max_attempts=350_000,
                                                          start_sigma=9e-4,
                                                          T_start=4e-3,
                                                          T_end=1e-9)
        if final_sa_val < global_best_val:
            global_best_seq = final_sa_seq
            global_best_val = final_sa_val
            print(f"[Final SA] improved to {global_best_val:.6f}")

    print(f"\n[Done length {n}] best surrogate = {global_best_val:.6f}")
    return global_best_seq, global_best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects the time budget and stores the best found sequence in the
    global variable ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 2.0                 # small safety margin

    # --------------------------------------------------------------
    # Retrieve previous solution (if any) and decide on lengths to try
    # --------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # start with a power‑of‑two length – fast FFTs
    lengths = [4096]               # default
    # If we have enough time we will also try a larger problem size
    if time.time() + 250 < deadline:    # enough slack for a second stage
        lengths.append(8192)

    best_overall_seq = None
    best_overall_val = float("inf")

    for n in lengths:
        # adapt previous solution to the new length if possible
        prior_rescaled = None
        if prev_arr is not None:
            old_n = prev_arr.shape[0]
            if old_n != n:
                # linear interpolation to the new grid
                x_old = np.linspace(0.0, 1.0, old_n)
                x_new = np.linspace(0.0, 1.0, n)
                prior_rescaled = np.interp(x_new, x_old, prev_arr)
                prior_rescaled = np.maximum(prior_rescaled, 0.0)
                s = prior_rescaled.sum()
                if s > 0.0:
                    prior_rescaled = prior_rescaled / s
            else:
                prior_rescaled = prev_arr.copy()

        seq, val = _search_length(n, rng, prior_rescaled, deadline)

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            print(f"\n>>> New overall best surrogate = {best_overall_val:.6f} (n={n})")

        # if time is exhausted, break early
        if time.time() >= deadline:
            break

    # --------------------------------------------------------------
    # Official evaluation via the provided function
    # --------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_overall_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate = {best_overall_val:.6f}, "
              f"true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for later runs
    globals()["height_sequence_1"] = best_overall_seq.tolist()
    return best_overall_seq.tolist()
