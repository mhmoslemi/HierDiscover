### Projection onto simplex

Algorithm (Duchi et al.):

Given v ∈ ℝ^n, we want x = argmin ||x - v||_2 s.t. x ≥ 0, sum x = 1.

Procedure:

- Sort v in descending order: u = sort(v)[::-1] (descending).

- Compute cumulative sum: cssv = np.cumsum(u) - 1

- Find rho = max{i | u_i - cssv_i / (i+1) > 0}

- Theta = cssv[rho] / (rho+1)

- x = np.maximum(v - theta, 0)

Implementation stable.

Algorithm
---------
1. Initialise a pool of deterministic seeds (uniform, arcsine, triangular,

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, soft‑max etc.)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search:
    2·n·max(conv(seq, seq)), assuming sum(seq)=1.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise to sum 1 exactly
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(w, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (exact for the active lag)
# ----------------------------------------------------------------------
def _max_conv_and_subgrad(a: np.ndarray):
    """Return (max_conv, active_lag, subgradient) for vector a (sum=1)."""
    n = a.shape[0]
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))          # active lag (0 … 2n‑2)
    max_val = conv[k]

    # build sub‑gradient g_i = 2·a_{k‑i} on the overlapping region
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = k - idx_i
    g[idx_i] = 2.0 * a[rev_idx]
    return max_val, k, g


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (Euclidean mirror)
# ----------------------------------------------------------------------
def _subgradient_descent(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         eta0: float = 0.8):
    """Refine ``start_seq`` using projected sub‑gradient steps."""
    a = start_seq.copy()
    best_seq = a.copy()
    max_val, *_ = _max_conv_and_subgrad(a)
    best_val = 2.0 * n * max_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        cur_max, k, g = _max_conv_and_subgrad(a)
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:
            best_val, best_seq = cur_val, a.copy()

        # step‑size schedule
        eta = eta0 / math.sqrt(it)

        # gradient step and projection onto simplex
        a = _project_simplex(a - eta * g)

    # final check
    final_max, *_ = _max_conv_and_subgrad(a)
    final_val = 2.0 * n * final_max
    if final_val < best_val:
        best_val, best_seq = final_val, a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Sparse candidate generation and fast max‑convolution evaluation
# ----------------------------------------------------------------------
def _max_convolution_sparse(pos: np.ndarray, w: np.ndarray, n: int) -> float:
    """
    Compute max_{k} Σ_{i+j=k} a_i a_j for a sparse vector:
    positions ``pos`` (int) and corresponding weights ``w`` (float, sum=1).
    """
    L = 2 * n - 1
    conv = np.zeros(L, dtype=np.float64)
    m = len(pos)

    # self‑terms (i=j)
    for i in range(m):
        idx = 2 * pos[i]
        conv[idx] += w[i] * w[i]

    # cross‑terms (i<j) – each contributes twice
    for i in range(m):
        pi = pos[i]
        wi = w[i]
        for j in range(i + 1, m):
            pj = pos[j]
            wj = w[j]
            idx = pi + pj
            conv[idx] += 2.0 * wi * wj

    return float(conv.max())


def _random_sparse_candidate(n: int, rng: np.random.Generator,
                            m_min: int = 30, m_max: int = 250,
                            conc_options: tuple = (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0)):
    """
    Produce a full‑length candidate (len=n) that has exactly ``m`` non‑zero entries.
    The non‑zeros are placed on randomly chosen distinct positions.
    """
    m = rng.integers(m_min, m_max + 1)
    conc = rng.choice(conc_options)
    pos = rng.choice(n, size=m, replace=False)
    w = rng.dirichlet(np.full(m, conc)).astype(np.float64)

    seq = np.zeros(n, dtype=np.float64)
    seq[pos] = w
    max_c = _max_convolution_sparse(pos, w, n)
    surrogate = 2.0 * n * max_c
    return seq, surrogate


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     n: int = 4000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises the supplied
    ``evaluate_sequence`` objective.  The routine respects the 1000 s
    time budget and stores the best sequence in the global variable
    ``height_sequence_1`` for warm‑starts.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # 0️⃣ Load any existing solution that matches the required length
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    prev_arr = None
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.size == n and arr.sum() > 0.0:
                prev_arr = arr / arr.sum()
        except Exception:
            prev_arr = None

    # ------------------------------------------------------------------
    # 1️⃣ Deterministic seed pool (all normalised to sum = 1)
    # ------------------------------------------------------------------
    def _as_prob(v):
        return v / v.sum()

    seeds = [_as_prob(np.full(n, 1.0 / n))]                 # uniform
    if prev_arr is not None:
        seeds.append(prev_arr)                              # carry‑over
    # arcsine
    i = np.arange(n, dtype=np.float64)
    eps = 1e-12
    a = 1.0 / np.sqrt((i + 0.5) * (n - 0.5 - i) + eps)
    seeds.append(_as_prob(a))
    # triangular
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1)
        right = left[::-1]
    else:
        left = np.arange(1, half + 2)
        right = left[:-1][::-1]
    tri = np.concatenate([left, right]).astype(np.float64)
    seeds.append(_as_prob(tri))

    # ------------------------------------------------------------------
    # 1️⃣ Evaluate deterministic seeds, keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf
    print("\n--- Evaluating deterministic seeds ---")
    for idx, vec in enumerate(seeds):
        if time.time() > deadline:
            break
        val = _objective_surrogate(vec)
        print(f" seed #{idx+1:02d}: surrogate = {val:.6f}")
        if val < best_val:
            best_val, best_seq = val, vec.copy()
    print(f"[Init] best surrogate = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Random Dirichlet draws (dense vectors) – a quick broad search
    # ------------------------------------------------------------------
    draw_limit = 8000
    draws = 0
    conc_options = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 4.0, 8.0]
    while draws < draw_limit and time.time() < deadline:
        conc = rng.choice(conc_options)
        vec = rng.dirichlet(np.full(n, conc)).astype(np.float64)
        val = _objective_surrogate(vec)
        if val < best_val:
            best_val, best_seq = val, vec.copy()
            print(f"[Dense] new best = {best_val:.6f} (draw {draws})")
        draws += 1

    # ------------------------------------------------------------------
    # 3️⃣ Sparse random candidates – the core innovation
    # ------------------------------------------------------------------
    print("\n--- Sparse random search ---")
    sparse_attempts = 0
    while time.time() < deadline:
        seq, surrogate = _random_sparse_candidate(n, rng)
        if surrogate < best_val:
            best_val, best_seq = surrogate, seq.copy()
            print(f"[Sparse] new best = {best_val:.6f} (attempt {sparse_attempts})")
        sparse_attempts += 1
        # keep the loop lightweight – the cost per candidate is O(m²) with m≈100

    # ------------------------------------------------------------------
    # 4️⃣ Refine the current best with projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        print("\n--- Projected sub‑gradient refinement ---")
        # allocate remaining time roughly equally to refinement
        seq_opt, val_opt = _subgradient_descent(best_seq, n,
                                                deadline,
                                                max_iters=500_000,
                                                eta0=0.8)
        print(f"[Refine] refined surrogate = {val_opt:.6f}")
        if val_opt < best_val:
            best_val, best_seq = val_opt, seq_opt.copy()
            print("   >>> Updated global best after refinement")

    # ------------------------------------------------------------------
    # 5️⃣ Final evaluation with the provided ``evaluate_sequence``
    # ------------------------------------------------------------------
    true_val = None
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {best_val:.6f}, evaluate_sequence = {true_val:.6f}")
    except Exception as exc:                                          # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    # Store for possible warm‑start
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
