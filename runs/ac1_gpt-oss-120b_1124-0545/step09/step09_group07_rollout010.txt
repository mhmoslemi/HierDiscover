def project_onto_simplex(v):
    """Projects a vector v onto the probability simplex (nonnegative, sum=1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate – gradient‑based smoothing + multi‑restart + annealing

The objective to minimise is

    F(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

with a_i ≥ 0, a not identically zero.
Because the ratio is scale‑invariant we fix the simplex
    Σ a_i = 1 .
The hard max is replaced by a smooth “log‑sum‑exp” surrogate

    G(a) = (1/α)·log Σ_k exp(α·conv_k)

which has the same minimisers as the original problem when α is large.
Its gradient is a weighted average of the gradients of the individual
convolution entries, i.e.

    ∇G = Σ_k  w_k · ∇conv_k ,   w_k = exp(α·conv_k) / Σ_j exp(α·conv_j).

For a fixed a the gradient of conv_k w.r.t. a is the linear
convolution of the reversed vector with the weight vector w.
Thus each optimisation step only needs two FFT‑based convolutions.

The algorithm
---------------
1. Choose a problem size n (4000, a power‑of‑two‑friendly length).
2. Build a small pool of inexpensive seeds:
   * previous best (if any),
   * a uniform vector,
   * a Dirichlet‑type sparse vector,
   * a deterministic Sidon‑like support.
3. Repeatedly (until time runs out) pick a seed and run a
   *smooth gradient descent* on G(a) for a short fixed budget.
   The current best sequence across all runs is stored.
4. With the remaining time run a cheap simulated‑annealing finetuner
   that swaps small masses between two coordinates.
5. Project the final vector onto the probability simplex, store it in the
   global variable ``height_sequence_1`` and return it as a plain list.

All heavy work respects the supplied time budget (default 1000 s) and
all vectors remain non‑negative.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, power‑of‑two padding)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using FFT (full length).  Both a and b are real.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (mirrors evaluate_sequence without the safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seeding families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _dirichlet_sparse(n: int, rng: np.random.Generator,
                      concentration: float = 0.3) -> np.ndarray:
    """Gamma‑generated sparse Dirichlet sample."""
    samples = rng.gamma(concentration, 1.0, size=n)
    if samples.sum() == 0:
        samples = np.ones(n)
    return samples / samples.sum()


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic “triangular‑numbers” support (mod n)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


# ----------------------------------------------------------------------
# Projection onto probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, n + 1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed gradient descent on the log‑sum‑exp surrogate
# ----------------------------------------------------------------------
def _smooth_optimize(start_seq: np.ndarray,
                     deadline: float,
                     alpha: float = 200.0,
                     lr0: float = 0.3) -> tuple[np.ndarray, float]:
    """
    Minimises the smooth surrogate of the max‑convolution using
    (projected) gradient descent.  Returns the best sequence observed
    and its exact objective.
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1

        # 1) convolution a*a
        conv = _conv_fft(a, a)                     # length 2n‑1
        maxc = conv.max()

        # 2) soft‑max weights (log‑sum‑exp gradient)
        #    w_k = exp(alpha * (conv_k - maxc))  (max weight = 1)
        w_raw = np.exp(alpha * (conv - maxc))
        w = w_raw / w_raw.sum()

        # 3) gradient of the surrogate
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        g = 2.0 * full[n - 1:n - 1 + n]           # slice to length n

        # 4) step size (inverse‑sqrt schedule) and projected update
        lr = lr0 / math.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        # 5) keep incumbent
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Smooth] it={it:>7d} best={cur_val:.6f}")

        # early exit if we are very close to the deadline
        if it % 5000 == 0 and time.time() > deadline - 5.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very lightweight simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis‑Hastings walk that swaps a fraction of mass between
    two random positions and occasionally performs a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.05)
    decay = 0.9994
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        # move a small fraction of a[i] to a[j]
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:>7d} best={new_val:.6f}")

        # global perturbation every 2000 steps
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            candidate = cur + noise
            candidate = np.clip(candidate, 0.0, None)
            candidate /= candidate.sum()
            cand_val = _objective(candidate)
            accept = cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / (T + 1e-12))
            if accept:
                cur = candidate
                cur_val = cand_val
                if cand_val < best_val:
                    best_val = cand_val
                    best = candidate.copy()
                    print(f"[Anneal] it={it:>7d} (jump) best={cand_val:.6f}")

    return best, best_val
