"""
Non‑negative step‑function optimiser (multi‑restart, smoothed‑gradient).

The optimiser works in three phases:

1️⃣  **Seed creation** – deterministic families (uniform, triangular,
    sine‑window, symmetric / unsymmetric geometric, power‑law,
    beta‑distribution) plus a few random Dirichlet draws.  All candidates
    are normalised to sum = 1 and evaluated with the cheap surrogate
    ``2·n·max(conv(a,a))`` (computed by an FFT).  The best *k* seeds are
    kept.

2️⃣  **Refinement** – each of the selected seeds is refined with a
    smoothed‑gradient descent on the true objective.  The non‑smooth
    ``max`` is approximated by a log‑sum‑exp with a temperature that is
    slowly increased (α grows).  The gradient of the smooth proxy is
    obtained via two FFT‑based convolutions.  After each gradient step we
    project back onto the probability simplex (ensuring non‑negativity
    and ∑a = 1).  The best vector seen across all restarts is retained.

3️⃣  **Polish** – a short simulated‑annealing pass that reshuffles the
    mass of a random small subset of indices.  This helps escape any
    remaining local minima.

The whole routine respects a global time budget (default 1000 s) and
stores its result in the global name ``height_sequence_1`` so that a
future run can start from it.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (fast O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """
    Cheap surrogate of the true objective.
    ``seq`` must be non‑negative and sum to 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic families (all normalised to sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure (unsymmetric) geometric decay a_i = r**i."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)^‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Return a k‑sparse Dirichlet vector (k ≤ n)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a

# Projection onto the probability simplex

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum x = 1}."""
    # implementation of the algorithm by Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerically degenerate – fallback to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _smooth_grad_step(

    f_α(a) = (1/α) * log Σ_k exp(α·conv_k(a,a)).
    The gradient is Σ_k w_k ∇conv_k with
    w_k = softmax_α(conv_k).
    Returns the updated vector (already projected onto the simplex).
    """
    # full convolution a*a
    conv = _conv_fft(a, a)                     # length 2n-1
    maxc = np.max(conv)
    # stable softmax
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / exp_term.sum()              # shape (2n-1,)

    # gradient = Σ_k w_k ∇conv_k
    # ∇conv_k[j] = a_{k-j} + a_{j-k} (where indices in range)
    # The two terms can be obtained via two convolutions:
    #  g1 = a * w  (correlates a with w, producing Σ_i a_i w_{j+i})
    #  g2 = a * w_rev   (with w reversed, producing Σ_i a_i w_{i-j})
    w_rev = w[::-1]

    # perform the two convolutions; both inputs are real, use the same FFT helper
    g1_full = _conv_fft(a, w)          # length n + (2n-1) -1 = 3n-2
    g2_full = _conv_fft(a, w_rev)

    # Extract the portions that correspond to j = 0 … n‑1
    # For g1 we need indices (n‑1 .. n‑1 + n‑1) → slice [n-1 : 2n-1)
    # For g2 we need indices (n‑1 .. n‑1 + n‑1) as well.
    start = n - 1
    g1 = g1_full[start : start + n]
    g2 = g2_full[start : start + n]

    grad = g1 + g2                         # shape (n,)

    # gradient ascent on the *negative* objective → descent
    a_next = a - lr * grad
    a_proj = _project_simplex(a_next)
    return a_proj


def _refine_with_smoothing(
    start: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    alpha0: float = 1.0,
    alpha_factor: float = 1.02,
    lr0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Smoothed‑gradient descent with a slowly increasing temperature.
    Returns the best vector (according to the true surrogate) seen.
    """
    a = start.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    it = 0
    alpha = alpha0
    lr = lr0

    while it < max_iters and time.time() < deadline:
        it += 1
        # learning‑rate schedule (inverse‑sqrt) + slight decay
        lr = lr0 / np.sqrt(it)
        a = _smooth_grad_step(a, n, alpha, lr)

        # slowly raise α (sharpen the approximation of the max)
        if it % 500 == 0:
            alpha *= alpha_factor

        if it % 2000 == 0:
            cur_val = _surrogate(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SmoothGrad] it={it:6d}  α={alpha:.2f}  best≈{best_val:.6f}")

    # final clean‑up with the raw sub‑gradient (α → ∞) for a few hundred steps
    final_iters = min(5_000, max_iters // 20)
    for _ in range(final_iters):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = lr0 / np.sqrt(it + 1)
        a = _project_simplex(np.log(a + 1e-30) - lr * g)  # exponentiated‑gradient style
        it += 1
        cur_val = _surrogate(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (lightweight)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """Simple SA that redistributes mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9997, min_temp)

        k = rng.integers(2, 8)           # 2 … 7 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _surrogate(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:6d}  best≈{best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a list.
    The routine is fully deterministic w.r.t. the given ``seed``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin (5 s) before the hard timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Previous solution (if any) – we reuse it as an extra seed
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the step‑function – 4000 is a good compromise
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – create a large pool of seeds and keep the best few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    seed_pool = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – refine each seed with smoothed‑gradient descent
    # ------------------------------------------------------------------
    global_best_seq = None
    global_best_val = np.inf

    for idx, s in enumerate(seed_pool):
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        # allocate a modest share of the remaining budget
        sub_deadline = time.time() + remaining * 0.6 / (len(seed_pool) - idx)
        print(f"\n--- Refining seed #{idx+1} (deadline in {remaining:.1f}s) ---")
        refined, val = _refine_with_smoothing(
            start_seq=s,
            n=n,
            deadline=sub_deadline,
            max_iters=120_000,
            alpha0=1.0,
            alpha_factor=1.015,
            lr0=0.7,
        )
        print(f"[Refine] seed #{idx+1} best≈{val:.6f}")
        if val < global_best_val:
            global_best_val = val
            global_best_seq = refined.copy()
            print(f"[Overall] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional simulated‑annealing polishing (if time left)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        polished, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=120_000
        )
        if val_sa < global_best_val:
            global_best_seq = polished.copy()
            global_best_val = val_sa
            print(f"[Polish] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        # Evaluate once just to ensure there is no hidden exception.
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose for reuse in future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
