def smooth_mirror_descent(start_seq, n, deadline, lr0=1.0, tau0=0.05, tau_min=1e-8, max_iters=500000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a) # surrogate.
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        # compute conv and w
        conv = _conv_fft(a, a)
        max_conv = np.max(conv)
        # compute temperature
        tau = max(tau_min, tau0 * np.exp(-it / 50000.0))
        # compute softmax weights
        # shift to avoid overflow
        shifted = conv - max_conv  # shift for stability
        exps = np.exp(shifted / tau)
        w = exps / np.sum(exps)
        # gradient via convolution: use reversed a
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w)
        grad = 2.0 * grad_full[n-1:n-1+n]
        # exponentiated gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * grad
        log_a -= np.max(log_a)  # for stability
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum
        # evaluate surrogate
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            # optionally print progress
            if it % 10000 == 0:
                print(f"[Smooth] it={it} best={best_val:.6f}")
    return best_seq, best_val

"""
Non‑negative step‑function optimiser (new version).

This implementation builds on the previous mirror‑descent approach but
adds a *soft‑max* surrogate for the max‑convolution.  The idea is to
replace the hard sub‑gradient (which only looks at the single index
where the convolution attains its maximum) by a gradient of the
log‑sum‑exp of the convolution:

    f(a) = τ·log Σ_k exp( (a*a)_k / τ )

When τ→0 this converges to the true max, while for larger τ the
gradient is a smooth weighted average of many convolution entries.
Using this smoother gradient together with exponentiated‑gradient
updates (mirror descent on the probability simplex) yields a more
exploratory descent and usually reaches a lower objective.

Algorithm outline
-----------------
1. **Seed generation** – deterministic families (uniform, triangular,
   sine‑window, symmetric/unsymmetric geometric, power‑law, beta) and a
   large set of random Dirichlet draws.  All candidates are normalised
   to sum = 1 and evaluated with the cheap surrogate `2·n·max(conv)`.
   The best `k` seeds are kept.

2. **Smoothed mirror‑descent** – each top seed is refined with the
   soft‑max gradient.  Temperature `τ` is exponentially decayed,
   learning‑rate follows `η = η₀ / sqrt(t)`.  The best vector over all
   runs is remembered.

3. **Exact sub‑gradient refinement** – the current global best is
   further polished with the original max‑only sub‑gradient (the
   “exponentiated‑gradient” step from the reference implementation).

4. **Upsampling (optional)** – if time permits the best sequence is
   interpolated to a finer grid (twice as many points) and refined
   again with the smoothed method.

The routine respects the supplied time budget (default 1000 s) and
stores the result in the global name `height_sequence_1` for possible
reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (fast, O(n log n)).
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    ``seq`` is assumed to be non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction (diverse, cheap)
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric – fine grid close to 1
    for r in np.linspace(0.985, 0.9999, 12):
        seeds.append(_geom_decay(n, r))

    # power‑law decay
    for p in np.arange(0.1, 5.01, 0.3):
        seeds.append(_power_decay(n, p))

    # beta family – moderate grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # mixtures uniform ↔ symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.90, 7):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 4):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws – richer set
    for conc in (0.02, 0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(10):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns – very small support
    for k in (3, 5, 10, 20):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Smoothed mirror‑descent (exponentiated‑gradient with soft‑max)
# ----------------------------------------------------------------------


def _smooth_mirror_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 1.5,
    tau0: float = 0.05,
    tau_min: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the *soft‑max* surrogate of max‑convolution.
    - ``lr0`` : initial learning‑rate factor (decays as 1/√t).
    - ``tau0``: initial temperature for the soft‑max.
    - ``tau_min``: final temperature (approaches the true max).
    Returns the best sequence seen and its surrogate objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- convolution and temperature -------------------------------------------------
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_conv = conv.max()
        tau = max(tau_min, tau0 * np.exp(-it / 50000.0))

        # soft‑max weights (log‑sum‑exp trick)
        shifted = conv - max_conv                  # now max = 0
        w = np.exp(shifted / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the log‑sum‑exp -------------------------------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, w)            # length 3n‑2
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]   # slice of length n

        # --- exponentiated‑gradient update -----------------------------------------------
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        # stabilise before exponentiation
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        # --- bookkeeping -----------------------------------------------------------------
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[Smooth] it={it:9d} best≈{best_val:.6f}")

    return best_seq, best_val

# Exact max‑convolution sub‑gradient descent (as in the reference)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10_000_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) using the true max
    sub‑gradient (i.e. only the index where the convolution attains its
    maximum).  Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient for the max index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:9d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (lightweight)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)               # size of the random subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA]  it={it:8d} best≈{best_val:.6f}")

    return best_seq, best_val
