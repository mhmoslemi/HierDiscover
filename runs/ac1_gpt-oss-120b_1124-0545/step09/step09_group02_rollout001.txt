def _adam_optimize(start_seq, n, deadline):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    # Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # schedule temps
    temp = 0.5
    while time.time() < deadline and temp > 1e-6:
        for _ in range(2000):
            if time.time() > deadline:
                break
            # conv
            conv = _conv_fft(a, a)
            # compute softmax weights
            # Use scaling: w_i = exp(conv_i / temp) to focus on larger conv.
            # Actually when temp small, weight heavily on max, as desired.
            # Use: exponent = conv / temp
            # subtract max exponent for numerical stability.
            exp_arg = conv / temp
            max_exp = np.max(exp_arg)
            w_exp = np.exp(exp_arg - max_exp)
            w = w_exp / w_exp.sum()
            # gradient
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            grad = 2.0 * full[n-1:n-1+n]
            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.1 * m_hat / (np.sqrt(v_hat) + eps)   # lr=0.1
            a = a - step
            # projection onto simplex
            a = _project_simplex(a)
            # evaluate
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] temp={temp:.6f} iter={t} best={cur_val:.6f}")
        temp *= 0.7   # decrease temperature
    return best_seq, best_val

"""
Adam‑smoothed optimisation for the step‑function attack.

The objective (with sum = 1) is 2·n·max(conv(a,a)).
We minimise a smoothed version
    F_T(a) = log‑sum‑exp(conv(a,a)/T)               (T > 0)
whose gradient is
    ∇F_T = 2 * conv( rev(a), softmax(conv/T) ).
Using Adam on this smooth surrogate and a geometric decay of T
gives a very effective descent, after which a short sub‑gradient
refinement (as in the reference implementation) cleans up the
remaining non‑smoothness.

The routine respects the 1000 s budget, keeps the best vector found,
stores it globally as ``height_sequence_1`` and returns it as a plain
list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution (full mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by the grader (2·n·max(conv(seq,seq)) / (sum(seq)**2)).
    The sequence is assumed non‑negative.  Returns np.inf for degenerate inputs.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector: sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse set of start vectors (all normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    for c in np.linspace(0.5, 12.0, 48):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    for p in np.linspace(0.0, 4.0, 41):
        seeds.append(_power(n, p))

    for c in np.linspace(0.5, 12.0, 24):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 9):
            seeds.append(_geom_power(n, r, p))

    for c in np.linspace(0.5, 12.0, 24):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 8):
        seeds.append(_tukey_window(n, alpha))

    for sigma_factor in np.linspace(0.06, 0.30, 13):
        seeds.append(_gaussian(n, sigma_factor))

    for alpha in np.linspace(0.3, 2.5, 23):
        seeds.append(_beta_window(n, alpha))

    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(4):
            seeds.append(_random_dirichlet(n, alpha, rng))

    for _ in range(10):
        seeds.append(_random_softmax(n, rng))

    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean‑up (clip, normalise)
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α = 1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient refinement – same as the reference implementation
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 600000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth objective.
    Returns best vector found and its value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max(rel_tol, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # early exit if we already beat the target
        if best_val < 1.503:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smoothed (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on F_T(a) = log‑sum‑exp(conv(a,a)/T).
    The temperature T is decreased geometrically.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # start temperature – large enough to give an almost flat weight vector
    T = 0.5
    # we will run several temperature stages until T becomes tiny
    while time.time() < deadline and T > 1e-6:
        # a few thousand inner iterations per temperature
        inner_iters = 2500
        for _ in range(inner_iters):
            if time.time() > deadline:
                break

            conv = _conv_fft(a, a)          # shape (2n‑1,)
            # softmax weights w_i ∝ exp(conv_i / T)
            # use shift for numerical stability
            exp_arg = conv / T
            max_exp = np.max(exp_arg)
            w = np.exp(exp_arg - max_exp)
            w /= w.sum()

            # gradient of the smoothed objective
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

            # Adam update
            t += 1
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = 0.12 * m_hat / (np.sqrt(v_hat) + eps)   # learning‑rate 0.12
            a = a - step
            a = _project_simplex(a)

            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Adam] T={T:.6f} it={t:7d} best={cur_val:.6f}")

            # early stopping if we already beat the target
            if best_val < 1.503:
                return best_seq, best_val

        # geometric cooling
        T *= 0.7

    return best_seq, best_val
