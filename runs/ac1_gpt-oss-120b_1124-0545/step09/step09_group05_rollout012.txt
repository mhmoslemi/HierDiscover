def _permute_anneal(start_seq, n, deadline, max_iters=400000):
    rng = np.random.default_rng()
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val
    # initial temperature as fraction of best_val
    T = best_val * 0.5  # higher initial temperature
    min_T = 1e-10
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        # decay
        T = max(T * 0.9999, min_T)
        # choose move type
        if rng.random() < 0.5:
            # simple swap of two indices
            i, j = rng.integers(n, size=2)
            if i==j: continue
            cand = a.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # block relocate: choose block length L (2-20)
            L = rng.integers(2, 21)
            i = rng.integers(0, n - L + 1)
            j = rng.integers(0, n - L + 1)
            if abs(i-j) < L: continue  # avoid overlap
            cand = a.copy()
            block = cand[i:i+L].copy()
            cand[i:i+L] = cand[j:j+L]
            cand[j:j+L] = block
        # compute objective
        cand_val = _objective(cand)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            a = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                print(f"[PermSA] it={it} best={cand_val:.6f}")
    return best_seq, best_val

"""
Non‑negative step‑function optimiser (v3).

This version builds on the previous pipeline but adds a *permutation annealing*
stage that explicitly reorders the coordinates of the vector.  Reordering can
dramatically change the self‑convolution because the index sums i+j are
different; a simple hill‑climbing / simulated‑annealing over swaps (and small
block moves) is therefore able to lower the peak of the autocorrelation beyond
what value‑only updates achieve.

Main stages
-----------
1.  Generate a rich seed pool (deterministic families, random Dirichlet draws,
    block patterns and a few random permutations of the deterministic seeds).
2.  Keep the best few seeds (by the fast surrogate 2·n·max(conv)).
3.  Refine the best seed with three deterministic gradient methods:
    – Adam on a smoothed log‑sum‑exp surrogate,
    – Exponentiated‑gradient sub‑gradient descent on the true max,
    – Euclidean‑projection gradient descent.
4.  *Permutation annealing*: repeatedly swap two positions or exchange two short
    blocks, accepting moves that improve the surrogate or with a Metropolis
    probability.  This re‑orders the mass while keeping the sum unchanged.
5.  Final sanity check with the real ``evaluate_sequence`` function.

The algorithm respects a user‑supplied time budget (default 1000 s) and prints
progress information.  The best sequence found is stored in the global name
``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Fast surrogate of ``evaluate_sequence`` used during the search.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (algorithm of Duchi et al. 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw sorted in descending order."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_blocks(n: int, max_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """
    Very simple block‑wise pattern: choose k blocks of equal length,
    spread them evenly with gaps, uniform mass inside each block.
    """
    k = rng.integers(2, max_blocks + 1)
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos:pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction (large but cheap)
# ----------------------------------------------------------------------


def _generate_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    families = [
        _uniform,
        _triangular,
        _sine_window,
    ]
    for fn in families:
        seeds.append(fn(n))
        # also a random permutation of each deterministic seed
        perm = fn(n).copy()
        rng.shuffle(perm)
        seeds.append(perm)

    # geometric decay – fine grid
    for r in np.linspace(0.70, 0.99, 13):
        seeds.append(_geom_decrease(n, r))

    # power‑law – fine grid
    for p in np.arange(0.3, 3.01, 0.3):
        seeds.append(_power_decrease(n, p))

    # sorted Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # block‑wise patterns
    for max_blocks in (3, 5, 7, 10):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # random Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (5, 10, 20, 40):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

# Optimisation primitives

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 120000) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a smooth log‑sum‑exp surrogate (keeps the simplex)."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5
    beta_mul = 1.001
    beta_max = 500.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                  # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] iter={it:6d} best={cur_val:.6f}")

        # temperature annealing for the log‑sum‑exp smoothing
        beta = min(beta * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # map to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.9 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 250000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the true
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] iter={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)            # numerical stabilisation
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _euc_proj_subgradient(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the true max‑convolution with Euclidean projection
    onto the probability simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[EucProj] iter={it:7d} best={cur_val:.6f}")

        # gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)

    return best_seq, best_val


def _permutation_annealing(start_seq: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 400000,
                           init_temp_factor: float = 0.5,
                           min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing over permutations of the coordinates.
    Moves are either a simple swap of two positions or an exchange of two short blocks.
    The temperature schedule is geometric; moves that improve the surrogate are always kept,
    otherwise they are accepted with probability exp(-Δ / T).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9999, min_temp)

        # decide move type
        if rng.random() < 0.55:
            # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # block exchange
            L = rng.integers(2, 21)                # block length 2‑20
            i = rng.integers(0, n - L + 1)
            j = rng.integers(0, n - L + 1)
            if abs(i - j) < L:                     # avoid overlap
                continue
            cand = cur.copy()
            block = cand[i:i + L].copy()
            cand[i:i + L] = cand[j:j + L]
            cand[j:j + L] = block

        cand_val = _objective(cand)
        delta = cand_val - cur_val
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                print(f"[PermSA] iter={it:7d} best={cand_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (as a list of floats).
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the search vector – a balance between expressiveness and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # Normalise all seeds to sum = 1
    seeds = [s / s.sum() if s.sum() > 0 else _uniform(n) for s in seeds]

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deterministic gradient refinements (single promising seed)
    # ------------------------------------------------------------------
    print("\n--- Gradient refinement of the best seed ---")
    # 1) Adam
    seq_adam, val_adam = _adam_optimize(global_best_seq, n, deadline)
    if seq_adam is not None and val_adam < global_best_val:
        global_best_seq, global_best_val = seq_adam, val_adam
        print(f"[global‑Adam] new best = {val_adam:.6f}")

    # 2) Exponentiated‑gradient sub‑gradient
    seq_sub, val_sub = _subgradient_refine(global_best_seq, n, deadline)
    if val_sub < global_best_val:
        global_best_seq, global_best_val = seq_sub, val_sub
        print(f"[global‑SubGrad] new best = {val_sub:.6f}")

    # 3) Euclidean‑projection gradient descent
    seq_euc, val_euc = _euc_proj_subgradient(global_best_seq, n, deadline)
    if val_euc < global_best_val:
        global_best_seq, global_best_val = seq_euc, val_euc
        print(f"[global‑EucProj] new best = {val_euc:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – permutation annealing (reordering of mass)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Permutation annealing ---")
        seq_perm, val_perm = _permutation_annealing(global_best_seq, n, deadline)
        if val_perm < global_best_val:
            global_best_seq, global_best_val = seq_perm, val_perm
            print(f"[global‑PermSA] new best = {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – final sanity check and bookkeeping
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
