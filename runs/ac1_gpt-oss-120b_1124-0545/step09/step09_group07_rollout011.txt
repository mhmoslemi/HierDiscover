def propose_candidate(seed=42, budget_s=1000, **kwargs):
    # initialization of rng
    start_time = time.time()
    deadline = start_time + budget_s - 1
    rng = np.random.default_rng(seed)

    # Choose n (maybe 8000 for better potential)
    n = 8000
    max_collisions = 2   # target threshold
    attempts = 30

    best_seq = None
    best_val = float('inf')
    best_c = None
    best_m = None

    # Greedy building of support for several attempts
    for att in range(attempts):
        # call greedy
        support_mask = _greedy_build_support(n, max_collisions, rng)
        m = support_mask.sum()
        if m == 0:
            continue
        # compute max collision
        conv_counts = _conv_fft(support_mask.astype(np.float64), support_mask.astype(np.float64))
        c = int(np.rint(conv_counts.max()))  # should be int
        # assign equal weights
        seq = support_mask.astype(np.float64) / m
        val = _objective(seq)
        # note use _objective (our internal objective) by default sum=1 => matches evaluate_sequence
        if val < best_val:
            best_val = val
            best_seq = seq
            best_c = c
            best_m = m
            print(f"[Init] attempt {att}, m={m}, c={c}, obj={val:.6f}")

        # break early if time near deadline
        if time.time() > deadline - 300:  # reserve time
            break

    # If still None fallback uniform
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)

    # Phase 2: subgradient refinement on support (exponentiated gradient)
    sub_deadline = deadline - 30
    if time.time() < sub_deadline:
        print("[Phase 1] Subgradient refinement on support")
        # Use _subgradient_refine but ensure nonzero entries only for support? Actually it works on full n.
        seq_sub, val_sub = _subgradient_refine(best_seq, sub_deadline)
        if val_sub < best_val:
            best_seq = seq_sub
            best_val = val_sub
            print(f"[Phase 1] improved obj {val_sub:.6f}")

    # Phase 3: LP refinement if cvxpy available (same as before)
    lp_deadline = deadline - 10
    if CVXPY_AVAILABLE and time.time() < lp_deadline:
        print("[Phase 2] LP refinement")
        seq_lp, val_lp = _lp_linearised_refine(best_seq, lp_deadline)
        if val_lp < best_val:
            best_seq = seq_lp
            best_val = val_lp
            print(f"[Phase 2] LP improved obj {val_lp:.6f}")

    # Phase 4: anneal fine-tune
    if time.time() < deadline - 0.5:
        print("[Phase 3] Annealing")
        seq_final, val_final = _anneal_optimize(best_seq, deadline)
        if val_final < best_val:
            best_seq = seq_final
            best_val = val_final
            print(f"[Phase 3] final obj {val_final:.6f}")

    # final normalization
    best_seq = np.maximum(best_seq, 0)
    total = best_seq.sum()
    if total == 0:
        best_seq = _uniform(n)
        total = best_seq.sum()
    best_seq = best_seq / total
    final_obj = _objective(best_seq)
    print(f"\n=== Done: best obj {final_obj:.6f}, n={n}, m={best_m}, c={best_c}")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()

def _greedy_support(n: int, max_c: int, rng: np.random.Generator) -> np.ndarray:
    mask = np.zeros(n, dtype=np.bool_)
    conv_counts = np.zeros(2*n-1, dtype=np.int32)
    current = []
    # random order of indices
    order = np.arange(n)
    rng.shuffle(order)
    for idx in order:
        # quick check: potential self count
        if conv_counts[2*idx] + 1 > max_c:
            continue
        # compute sums with current indices
        exceed = False
        for j in current:
            s = idx + j
            if conv_counts[s] + 2 > max_c:
                exceed = True
                break
        if exceed:
            continue
        # accept
        mask[idx] = True
        # update conv_counts
        for j in current:
            s = idx + j
            conv_counts[s] += 2
        conv_counts[2*idx] += 1
        current.append(idx)
    return mask

"""
Propose candidate step function.

Algorithm:
1. Greedy construction of a support set with a bound on the maximal number
   of ordered pairs (the convolution of the binary support) per sum.
   This yields a sparse vector with a provable upper bound on the desired
   objective (2·n·max(conv)/sum^2) using equal weights on the support.
2. Keep the best support found among several random attempts and several
   upper‑collision limits.
3. Refine the weights of the selected support with exponentiated‑gradient
   sub‑gradient descent on the exact objective (full convolution).
4. Optionally perform a linear‑program refinement (if cvxpy is available)
   and a short simulated‑annealing fine‑tuning.
The whole routine respects the given time budget.
"""

import time, math, numpy as np

# optional cvxpy import
try:
    import cvxpy as cp
    CVXPY_AVAILABLE = True
except Exception:
    CVXPY_AVAILABLE = False

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    s = seq.sum()
    if s < 1e-12:
        return float('inf')
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n

def _subgradient_refine(start_seq: np.ndarray, deadline: float):
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12
    it = 0
    while True:
        it += 1
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)  # length = 3n‑2, but we only need the middle n terms
            g = 2.0 * full[n - 1:n - 1 + n]
        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:8d} best={cur_val:.6f}")
    return best_seq, best_val

def _lp_linearised_refine(a: np.ndarray, deadline: float,
                         max_refine: int = 8,
                         eps_active: float = 1e-9):
    if not CVXPY_AVAILABLE:
        return a, _objective(a)
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)
    x_var = cp.Variable(n)
    M_var = cp.Variable()
    constraints = [x_var >= 0,
                   cp.sum(x_var) == 1]
    for it in range(max_refine):
        if time.time() > deadline - 0.3:
            break
        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]
        if active.size == 0:
            break
        A = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            A[r, i_range] = a[k - i_range]
        lin_constraints = [2 * (A @ x_var) <= M_var + conv[active]]
        prob = cp.Problem(cp.Minimize(M_var), constraints + lin_constraints)
        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=2000,
                       abstol=1e-9,
                       reltol=1e-9)
        except Exception:
            break
        if x_var.value is None or M_var.value is None:
            break
        a_new = np.asarray(x_var.value).flatten()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s
        val_new = _objective(a_new)
        print(f"[LP] iter={it+1:2d} M={M:.6f} new_obj={val_new:.6f}")
        if val_new < best_val - 1e-9:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
        else:
            break
    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray, deadline: float, max_iters: int | None = None):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val
    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)
        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")
    return best, best_val

def _greedy_support(n: int, max_c: int, rng: np.random.Generator) -> np.ndarray:
    """Return a boolean mask of length n.
    The mask represents a set of indices such that the binary auto‑convolution
    never exceeds max_c (ordered‑pair count). Greedy random insertion.
    """
    mask = np.zeros(n, dtype=np.bool_)
    conv_counts = np.zeros(2 * n - 1, dtype=np.int32)
    current = []
    order = np.arange(n)
    rng.shuffle(order)
    for idx in order:
        # quick check for self contribution
        if conv_counts[2 * idx] + 1 > max_c:
            continue
        # check with existing indices
        ok = True
        for j in current:
            s = idx + j
            if conv_counts[s] + 2 > max_c:
                ok = False
                break
        if not ok:
            continue
        # accept this index
        mask[idx] = True
        # update conv_counts
        for j in current:
            s = idx + j
            conv_counts[s] += 2
        conv_counts[2 * idx] += 1
        current.append(idx)
    return mask

def _search_best_support(n: int, rng: np.random.Generator, deadline: float):
    """Explore a few max_c values and random greedy attempts.
    Returns a dense probability vector (sum = 1) and its exact objective.
    """
    best_seq = None
    best_val = float('inf')
    best_m = None
    best_c = None
    for max_c in (1, 2, 3, 4):
        for attempt in range(30):
            if time.time() > deadline - 30:  # keep some time for refinements
                break
            mask = _greedy_support(n, max_c, rng)
            m = int(mask.sum())
            if m < 2:
                continue
            # evaluate max count directly from conv_counts (already limited to max_c)
            # For safety recompute using FFT (fast) to get true max (may be slightly < max_c)
            conv = _conv_fft(mask.astype(np.float64), mask.astype(np.float64))
            c = int(np.rint(conv.max()))
            # compute objective with equal weights
            seq = mask.astype(np.float64) / m
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_m = m
                best_c = c
                print(f"[Init] max_c={max_c} att={attempt} m={m} c={c} obj={val:.6f}")
    return best_seq, best_val, best_m, best_c


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The method builds a sparse Sidon‑like support with a bounded convolution
    count, then refines the weights using sub‑gradient descent, an optional
    LP refinement and a final simulated‑annealing fine‑tuning.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # safety margin

    # problem size - we keep it moderate to allow many iterations
    n = 4000
    print(f"\n=== Optimising step function of length {n} ===")

    # try to reuse previous solution if it matches the current size
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                seed_vec = prev_arr / prev_arr.sum()
                print("[Init] Re‑using saved sequence")
            else:
                seed_vec = None
        except Exception:
            seed_vec = None
    else:
        seed_vec = None

    # if no usable previous sequence, run the greedy support search
    if seed_vec is None:
        support_deadline = deadline - 200.0  # reserve time for later phases
        seq, val, m, c = _search_best_support(n, rng, support_deadline)
        if seq is None:
            # fallback to simple uniform
            seq = _uniform(n)
            val = _objective(seq)
        seed_vec = seq
        print(f"[Init] Greedy support objective = {val:.6f} (m={m}, c={c})")

    # Phase 1 – sub‑gradient refinement (exponentiated gradient)
    sub_deadline = deadline - 30.0
    if time.time() < sub_deadline:
        print("[Phase 1] Sub‑gradient refinement")
        seq_sub, val_sub = _subgradient_refine(seed_vec, sub_deadline)
        if val_sub < _objective(seed_vec):
            seed_vec = seq_sub
            print(f"[Phase 1] improved objective = {val_sub:.6f}")
        else:
            print("[Phase 1] no improvement")
    else:
        print("[Phase 1] skipped (no time)")

    # Phase 2 – LP linearised refinement (optional)
    lp_deadline = deadline - 10.0
    if CVXPY_AVAILABLE and time.time() < lp_deadline:
        print("[Phase 2] LP refinement")
        seq_lp, val_lp = _lp_linearised_refine(seed_vec, lp_deadline)
        if val_lp < _objective(seed_vec):
            seed_vec = seq_lp
            print(f"[Phase 2] improved objective = {val_lp:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped")

    # Phase 3 – simulated annealing fine‑tuning
    if time.time() < deadline - 0.5:
        print("[Phase 3] Simulated annealing")
        seq_final, val_final = _anneal_optimize(seed_vec, deadline)
        if val_final < _objective(seed_vec):
            seed_vec = seq_final
            print(f"[Phase 3] final objective = {val_final:.6f}")
        else:
            print("[Phase 3] annealing did not improve")
    else:
        print("[Phase 3] no time left")

    # final normalisation
    final_seq = np.maximum(seed_vec, 0.0)
    tot = final_seq.sum()
    if tot < 1e-12:
        final_seq = _uniform(n)
        tot = final_seq.sum()
    final_seq = final_seq / tot
    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()

"""
Propose a step function that minimises `evaluate_sequence`.

Algorithm
---------
1. **Sparse Sidon‑like construction** – Randomly insert indices into a
   binary mask while keeping the ordered‑pair convolution count
   (`conv(mask,mask)`) bounded by a small constant `max_c`.  This yields a
   sparse vector with a provable upper bound on the objective
   `2·n·max(conv)/sum²` (equal weights on the support).

2. **Search over several `max_c` values** – For each bound we run a few
   independent greedy attempts, keep the best support and evaluate the
   exact objective with the equal‑weight vector.

3. **Sub‑gradient (exponentiated‑gradient) refinement** – Starting from
   the best sparse vector we perform mirror‑descent on the true
   non‑smooth objective.  The update stays on the probability simplex
   and quickly improves the bound.

4. **Optional LP refinement** – If `cvxpy` is available we linearise the
   quadratic constraints around the current point and solve a small LP.

5. **Simulated annealing fine‑tuning** – A short Metropolis‑Hastings walk
   cleans up any remaining irregularities.

All phases respect the supplied time budget (default 1000 s).  The final
vector is normalised to sum = 1, stored globally as ``height_sequence_1``
and returned as a plain Python list.
"""

import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full convolution of two real vectors using FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective: 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.ones(n, dtype=np.float64) / n

# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray, deadline: float):
    """Mirror‑descent on the max‑convolution objective."""
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12
    it = 0
    while True:
        it += 1
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over all active constraints
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                         # numerical stability
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:8d} best={cur_val:.6f}")

    return best_seq, best_val

# Linearised LP refinement (convex–concave procedure)
# ----------------------------------------------------------------------
def _lp_linearised_refine(a: np.ndarray, deadline: float,
                         max_refine: int = 8,
                         eps_active: float = 1e-9):
    """Perform a few LP refinements where quadratic constraints are linearised."""
    if not CVXPY_AVAILABLE:
        return a, _objective(a)

    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    x_var = cp.Variable(n)
    M_var = cp.Variable()
    constraints = [x_var >= 0,
                   cp.sum(x_var) == 1]

    for it in range(max_refine):
        if time.time() > deadline - 0.3:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]
        if active.size == 0:
            break

        # Build matrix A where A[row,i] = a[k-i] for k = active[row]
        A = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            A[r, i_range] = a[k - i_range]

        lin_constraints = [2 * (A @ x_var) <= M_var + conv[active]]
        prob = cp.Problem(cp.Minimize(M_var), constraints + lin_constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=2000,
                       abstol=1e-9,
                       reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if x_var.value is None or M_var.value is None:
            break

        a_new = np.asarray(x_var.value).flatten()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s

        val_new = _objective(a_new)
        print(f"[LP] iter={it+1:2d} M={M:.6f} new_obj={val_new:.6f}")

        if val_new < best_val - 1e-9:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
        else:
            break

    return best_seq, best_val

def _anneal_optimize(start_seq: np.ndarray, deadline: float,
                     max_iters: int | None = None):
    """Tiny Metropolis‑Hastings walk that swaps a little mass between two positions."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

            else:

            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:

                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

# Greedy construction of a Sidon‑like support with bounded convolution count

def _greedy_support(n: int, max_c: int, rng: np.random.Generator) -> np.ndarray:
    """
    Return a boolean mask of length `n`.
    The mask represents a set of indices such that the binary auto‑convolution
    never exceeds `max_c` (ordered‑pair count).  Insertion order is random.
    """
    mask = np.zeros(n, dtype=np.bool_)
    conv_counts = np.zeros(2 * n - 1, dtype=np.int32)
    current = []
    order = np.arange(n)
    rng.shuffle(order)

    for idx in order:
        # self‑pair check
        if conv_counts[2 * idx] + 1 > max_c:
            continue

        # check all sums with already selected indices
        ok = True
        for j in current:
            s = idx + j
            if conv_counts[s] + 2 > max_c:
                ok = False
                break
        if not ok:
            continue

        # accept this index
        mask[idx] = True
        for j in current:
            conv_counts[idx + j] += 2
        conv_counts[2 * idx] += 1
        current.append(idx)

    return mask
