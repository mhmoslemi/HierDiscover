Implementation details: Use efficient simplex projection algorithm:

def _project_simplex(v):
    """Project vector v onto simplex {x >=0, sum x = 1}."""
    # Standard algorithm: sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a **smoothed exponential‑gradient descent**:

1.  Start from several cheap seeds (uniform, random soft‑max, Sidon‑like,
    Dirichlet) and keep the best.
2.  For each seed run a mirror‑descent on the *soft‑max* approximation of
    the max‑convolution:
        fτ(a) = τ·log∑_k exp(conv_k(a)/τ) .
    The gradient of fτ is a weighted sum of the individual quadratic
    constraints with weights given by a soft‑max over the convolution.
    The update is performed in the exponentiated‑gradient (multiplicative)
    form and the vector is renormalised onto the simplex after each step.
    The temperature τ is slowly decreased (starting at the current
    max‑convolution and decaying to ≈2 % of it), which gradually focuses
    the optimisation on the true max.
3.  The best sequence of the seeds is fine‑tuned with a short simulated‑
    annealing walk (pairwise mass swaps and occasional global Gaussian
    jumps).
4.  The final vector is normalised, stored in the global ``height_sequence_1``
    and returned as a plain list of floats.

The whole routine respects a 1000 s budget, runs on a single core
(and can use a second core for the optional LP refinement if ``cvxpy``
is available, though it is not required for the core search).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full mode) using real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used in the challenge: 2·n·max(conv)/sum²."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (return a probability vector on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    # stable soft‑max
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    a /= a.sum()
    return a


def _dirichlet_seed(n: int, rng: np.random.Generator, concentration: float = 0.5) -> np.ndarray:
    """A Dirichlet draw – useful for mildly sparse starting points."""
    a = rng.dirichlet(np.full(n, concentration))
    return a


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho index
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent on the soft‑max of the convolution
# ----------------------------------------------------------------------
def _smooth_expgd(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the smoothed objective
        fτ(a) = τ·log ∑ exp(conv_k(a)/τ)
    The update is multiplicative (log‑space) and the vector is
    re‑projected onto the simplex after each step.
    """
    a = start_seq.copy()
    a = a / a.sum()
    n = a.size
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1

        # --- convolution and current max ---
        conv = _conv_fft(a, a)                # length 2n‑1
        max_conv = conv.max()
        if max_conv <= 0.0:            # numerical safeguard
            max_conv = 1e-12

        # --- temperature schedule (slowly focus onto the true max) ---
        # τ starts at the current max and decays to ≈2 % of it.
        tau = max_conv * max(0.02, math.exp(-it / 2000.0))

        # --- soft‑max weights (stable) ---
        scaled = conv / tau
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        w /= w.sum()                    # w_i = exp(conv_i/τ) / Σexp(...)

        # --- gradient of the smoothed objective ---
        # Gradient = 2 * H(a)·w, where H is the Hankel matrix with entries w_{i+j}.
        # Efficiently computed by a convolution with the reversed weight vector.
        w_rev = w[::-1]
        conv_rev = _conv_fft(a, w_rev)          # length 3n‑2
        # extract the n‑dimensional slice and reverse it
        grad = 2.0 * conv_rev[n - 1 : 2 * n - 1][::-1]

        # --- step‑size (decreasing) ---
        lr = 0.5 / math.sqrt(it)   # simple schedule; can be tuned

        # --- exponentiated‑gradient update (log‑space) ---
        log_a = np.log(a + eps) - lr * grad
        # keep numbers numerically stable
        log_a -= np.max(log_a)
        a_new = np.exp(log_a)
        a_new /= a_new.sum()

        cur_val = _objective(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a_new.copy()
            print(f"[SmoothEGD] it={it:7d} val={cur_val:.6f}")

        a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuner (same idea as the previous code)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray, deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Tiny Metropolis–Hastings walk that swaps a little mass between two indices."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # pick two distinct indices
        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        # move a small fraction of mass from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val
