def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # safety margin

    rng = np.random.default_rng(seed+12345)

    n = 8000  # length of sequence

    # Initialise seed vectors list
    seeds = []

    # previous sequence if matches length
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n,):
                seeds.append(arr / arr.sum())
        except Exception:
            pass

    # Add some generic seeds
    seeds.append(_uniform(n))
    seeds.append(_random_softmax(n, rng))
    # maybe add sawtooth pattern
    idx = np.arange(0, n, 2)
    saw = np.zeros(n)
    saw[idx] = 1.0
    saw = saw / saw.sum()
    seeds.append(saw)

    # evaluate seeds and choose best
    best_val = float('inf')
    best_seq = None
    for s in seeds:
        val = _objective(s)
        if val < best_val:
            best_val = val
            best_seq = s.copy()
    print(f"[Init] best seed objective = {best_val:.6f}")

    # Phase 1: softmax Adam
    phase1_deadline = deadline - 200.0  # reserve 200 sec for later phases
    if time.time() < phase1_deadline:
        print("[Phase 1] Softmax‑Adam gradient descent")
        seq1, val1 = _softmax_adam_optimize(best_seq, phase1_deadline, rng=rng)
        if val1 < best_val:
            best_val = val1
            best_seq = seq1.copy()
            print(f"[Phase 1] achieved {val1:.6f}")
    else:
        seq1 = best_seq

    # Phase 2: Exact sub‑gradient refinement
    phase2_deadline = deadline - 30.0
    if time.time() < phase2_deadline:
        print("[Phase 2] Exact sub‑gradient refinement")
        seq2, val2 = _subgradient_refine(best_seq, phase2_deadline)
        if val2 < best_val:
            best_val = val2
            best_seq = seq2.copy()
            print(f"[Phase 2] improved to {val2:.6f}")
    else:
        print("[Phase 2] skipped (time)")

    # Phase 3: Simulated annealing fine‑tuning
    if time.time() < deadline - 0.5:
        print("[Phase 3] Simulated annealing")
        seq3, val3 = _anneal_optimize(best_seq, deadline)
        if val3 < best_val:
            best_val = val3
            best_seq = seq3.copy()
            print(f"[Phase 3] final best {val3:.6f}")

    final_seq = np.maximum(best_seq, 0.0)
    tot = final_seq.sum()
    if tot < 1e-12:
        final_seq = _uniform(n)
        tot = final_seq.sum()
    final_seq /= tot

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()

def _softmax_adam_optimize(start_seq, deadline, rng):
    a = start_seq.copy()
    n = a.size
    loga = np.log(a + 1e-30)   # small epsilon
    m = np.zeros_like(loga)
    v = np.zeros_like(loga)
    factor = 0.4
    best_seq = a.copy()
    best_val = _objective(a)
    last_improve_iter = 0
    it = 0
    while time.time() < deadline:
        it += 1
        a = np.exp(loga)
        a_sum = a.sum()
        if a_sum == 0:
            a = _uniform(n)
            a_sum = a.sum()
        a /= a_sum
        # convolution
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # compute softmax weights
        shift = max_c
        exp_w = np.exp((conv - shift) / (factor * max_c + 1e-20))
        w = exp_w / exp_w.sum()
        # gradient
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n-1:n-1+n]   # gradient wrt a
        # Adam update on loga
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        # bias correction
        m_hat = m / (1 - 0.9**it)
        v_hat = v / (1 - 0.999**it)
        step = 0.05 * m_hat / (np.sqrt(v_hat) + 1e-12)
        loga -= step
        # keep numerical stability
        loga -= np.max(loga)
        # evaluate objective
        a_candidate = np.exp(loga)
        a_candidate /= a_candidate.sum()
        cur_val = _objective(a_candidate)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a_candidate.copy()
            last_improve_iter = it
            print(f"[SoftmaxAdam] it={it:8d} best={cur_val:.6f}")

        # schedule temperature factor
        factor *= 0.9995
        if factor < 1e-4:
            factor = 1e-4

        # occasional jitter to escape stagnation
        if it % 50000 == 0:
            # add Gaussian noise to logits
            loga += rng.normal(scale=jitter_scale, size=n)
            loga -= np.max(loga)  # re‑center
            # reset Adam moments to avoid polluted state
            m.fill(0.0)
            v.fill(0.0)
            print(f"[SoftmaxAdam] it={it} jitter applied")

        # early exit if no improvement for long time
        if it - last_improve_iter > 200000 and factor <= 1e-4:
            # stop early
            break
    return best_seq, best_val

"""
Search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Soft‑max Adam** – optimise a smooth approximation of the
   non‑smooth max using an Adam update in the log‑domain (mirror
   descent).  The temperature of the soft‑max is slowly reduced,
   giving increasingly sharp focus on the worst convolution entries.

2. **Exact sub‑gradient refinement** – a few hundred thousand
   exponentiated‑gradient steps using the true sub‑gradient of the
   max‑convolution (active constraints only) to polish the solution.

3. **Simulated annealing** – a final Metropolis‑Hastings walk that
   swaps small amounts of mass between coordinates to escape any
   remaining local minima.

The routine respects the supplied time budget (default 1000 s) and
stores the best sequence in the global variable ``height_sequence_1``
for warm‑starting future runs.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (full linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple initial seeds (producing probability vectors on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    # stability
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sawtooth(n: int) -> np.ndarray:
    """Alternating 1/0 pattern – often a good starting point."""
    a = np.zeros(n, dtype=np.float64)
    a[::2] = 1.0
    return a / a.sum()


# ----------------------------------------------------------------------
# Soft‑max Adam optimisation (smooth surrogate of the max‑convolution)
# ----------------------------------------------------------------------
def _softmax_adam_optimize(start_seq: np.ndarray,
                           deadline: float,
                           rng: np.random.Generator,
                           lr: float = 0.05,
                           beta1: float = 0.9,
                           beta2: float = 0.999,
                           eps: float = 1e-12,
                           factor_init: float = 0.5,
                           factor_decay: float = 0.9995,
                           eval_interval: int = 50,
                           jitter_interval: int = 50000,
                           jitter_scale: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Optimise the max‑convolution using a soft‑max approximation.
    Returns the best sequence seen and its true objective value.
    """
    n = start_seq.size
    # logits for the mirror‑descent (log‑probabilities)
    loga = np.log(start_seq + eps)
    m = np.zeros_like(loga)
    v = np.zeros_like(loga)

    best_seq = None
    best_val = float("inf")
    last_improve = 0

    factor = factor_init
    it = 0

    while time.time() < deadline:
        it += 1

        # ----- current probabilities -----
        a = np.exp(loga)
        s = a.sum()
        if s == 0:
            a = _uniform(n)
            s = a.sum()
        a /= s

        # ----- convolution and soft‑max weights -----
        conv = _conv_fft(a, a)                     # length 2n‑1
        max_c = conv.max()
        # temperature τ = factor * max_c   (never too small)
        tau = max(factor * max_c, 1e-8)

        # soft‑max over convolution values, numerically stable
        shift = max_c
        exp_w = np.exp((conv - shift) / tau)
        w = exp_w / exp_w.sum()                    # length 2n‑1, sums to 1

        # ----- gradient of the smooth surrogate (w.r.t. a) -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        g = 2.0 * full[n - 1:n - 1 + n]            # shape (n,)

        # ----- Adam update on the logits (mirror‑descent) -----
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        loga -= step

        # keep numbers well‑scaled
        loga -= np.max(loga)

        # ----- occasional jitter to escape plateaus -----
        if it % jitter_interval == 0:
            loga += rng.normal(scale=jitter_scale, size=n)
            loga -= np.max(loga)      # re‑centre
            m.fill(0.0)
            v.fill(0.0)
            print(f"[SoftmaxAdam] it={it:8d} jitter applied")

        # ----- schedule the temperature factor -----
        factor = max(factor * factor_decay, 1e-4)

        # ----- evaluate the true objective from time to time -----
        if it % eval_interval == 0:
            cur_val = _objective(a)
            if cur_val < best_val - 1e-12:
                best_val = cur_val
                best_seq = a.copy()
                last_improve = it
                print(f"[SoftmaxAdam] it={it:8d} best={cur_val:.6f}")

            # early stop if stagnation for a long while
            if it - last_improve > 300_000 and factor <= 1e-4:
                break

    # if we never recorded a best (should not happen) fall back to the last a
    if best_seq is None:
        best_seq = a.copy()
        best_val = _objective(best_seq)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient (exponentiated‑gradient) refinement – same as before
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution objective."""
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over all active constraints
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                         # keep numbers stable
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:8d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (tiny random walk)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis‑Hastings walk that swaps tiny amounts of mass."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global Gaussian perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin – we want to finish cleanly before the wall‑clock limit
    deadline = start_time + budget_s - 2.0

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------
    # Problem size – a larger n gives more freedom to flatten the convolution.
    # ------------------------------------------------------------------
    n = 8000
    print(f"\n=== Optimising step function of length {n} ===")

    # ------------------------------------------------------------------
    # Initialise a decent starting point (reuse previous if possible)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    seed_vec = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.shape == (n,):
                seed_vec = prev_arr / prev_arr.sum()
        except Exception:
            seed_vec = None

    # If no suitable previous vector, build a handful of inexpensive seeds
    if seed_vec is None:
        seeds = [
            _uniform(n),
            _random_softmax(n, rng),
            _sawtooth(n),
        ]
        best_val = float("inf")
        best_seq = None
        for s in seeds:
            val = _objective(s)
            if val < best_val:
                best_val = val
                best_seq = s
        seed_vec = best_seq
        print(f"[Init] best seed objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Soft‑max Adam optimisation
    # ------------------------------------------------------------------
    phase1_deadline = deadline - 200.0   # keep ~200 s for later polishing
    if time.time() < phase1_deadline:
        print("[Phase 1] Soft‑max Adam optimisation start")
        seq1, val1 = _softmax_adam_optimize(seed_vec, phase1_deadline, rng)
        if val1 < _objective(seed_vec):
            seed_vec = seq1
            print(f"[Phase 1] best objective = {val1:.6f}")
    else:
        print("[Phase 1] skipped (time)")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient refinement (mirrored‑descent)
    # ------------------------------------------------------------------
    phase2_deadline = deadline - 30.0
    if time.time() < phase2_deadline:
        print("[Phase 2] Sub‑gradient refinement start")
        seq2, val2 = _subgradient_refine(seed_vec, phase2_deadline)
        if val2 < _objective(seed_vec):
            seed_vec = seq2
            print(f"[Phase 2] improved objective = {val2:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped (insufficient time)")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 3] Simulated annealing fine‑tuning")
        seq3, val3 = _anneal_optimize(seed_vec, deadline)
        if val3 < _objective(seed_vec):
            seed_vec = seq3
            print(f"[Phase 3] final objective = {val3:.6f}")
        else:
            print("[Phase 3] annealing did not improve")
    else:
        print("[Phase 3] no time left for annealing")

    # ------------------------------------------------------------------
    # Final normalisation and export
    # ------------------------------------------------------------------
    final_seq = np.maximum(seed_vec, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq /= total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
