    n = len(seq)
    conv = _conv_fft(seq, seq)
    maxc = conv.max()
    if lam <= 0:

def _adam_optimize(start, deadline, max_iters=200000, lam_start=5.0, lam_increase=1.001):
    a = start.copy()
    l = np.log(a + 1e-30)  # log-probabilities
    m_vec = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    best_seq = a.copy()
    best_val = _objective(a)
    t = 0
    lam = lam_start
    while time.time() < deadline and t < max_iters:
        t += 1
        # compute a from l
        log_a = l - np.max(l)
        a = np.exp(log_a)
        a = a / a.sum()
        # gradient
        g_a = _grad(a, lam=lam)
        # log gradient
        dot = np.dot(a, g_a)
        g_l = a * (g_a - dot)
        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        m_vec = beta1 * m_vec + (1 - beta1) * g_l
        v_vec = beta2 * v_vec + (1 - beta2) * (g_l * g_l)
        m_hat = m_vec / (1 - beta1**t)
        v_hat = v_vec / (1 - beta2**t)
        lr = 0.2 / (np.sqrt(t) + 1)  # adapt
        l = l - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # evaluate
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] iter={t:7d} best={cur_val:.6f} lam={lam:.2f}")
        # increase smoothing parameter gradually
        lam *= lam_increase
        # if no improvement for many iterations, break maybe, but we can just continue.
    return best_seq, best_val

"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a **smooth‑gradient Adam optimiser on the simplex**
combined with a few cheap random restarts and a short final simulated‑annealing
finetune.  The gradient of the (softened) max‑convolution is obtained
exactly by FFT, and the probabilities are kept on the simplex via a
soft‑max (exponentiated) parametrisation.  A temperature‑like
parameter λ controls how sharply the max is approximated; λ is increased
gradually so the optimisation first sees a smooth landscape and later
focuses on the true non‑smooth objective.

The routine respects a 1000 s budget, stores the best sequence in the
global ``height_sequence_1`` and returns it as a plain list of floats.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact evaluation used for reporting (same formula as evaluate_sequence)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Seed families (return a probability vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    # use stable soft‑max
    m = np.max(logits)
    exps = np.exp(logits - m)
    return exps / exps.sum()


def _exponential_decay(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate·i)  (then normalised)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    a = a / a.sum()
    return a


def _triangular(n: int) -> np.ndarray:
    """Increasing then decreasing linear shape."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    dec = inc[::-1] if n % 2 == 0 else np.arange(half, 0, -1, dtype=np.float64)
    a = np.concatenate([inc, dec])
    a = a / a.sum()
    return a


def _sidon_like(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support (binary, not enough points for optimum)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


# ----------------------------------------------------------------------
# Gradient of the (soft) max‑convolution
# ----------------------------------------------------------------------
def _soft_max_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Return a probability vector w (sum to 1, length = len(conv))
    that is a soft‑max over the entries of `conv`.
    The argument `lam` plays the role of an inverse temperature.
    """
    # subtract the maximum for numerical stability
    max_c = conv.max()
    if lam <= 0.0:
        # hard max – uniform weight on all ties
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.where(active, 1.0 / np.count_nonzero(active), 0.0)
        return w
    rel = (conv - max_c) / max_c  # ≤ 0
    exps = np.exp(lam * rel)
    w = exps / exps.sum()
    return w


def _gradient_wrt_a(a: np.ndarray, lam: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution with respect to the
    probability vector `a`.  Implements   g = 2 * rev(a) ⊛ w,
    where w are the soft‑max weights over the convolution entries.
    """
    n = a.size
    conv = _conv_fft(a, a)
    w = _soft_max_weights(conv, lam)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                # length = n + (2n-1) -1 = 3n-2
    grad = 2.0 * full[n - 1:n - 1 + n]        # extract the slice that matches a
    return grad


# ----------------------------------------------------------------------
# Adam optimisation on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _adam_optimize(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 300_000,
                   lam_start: float = 5.0,
                   lam_factor: float = 1.001) -> tuple[np.ndarray, float]:
    """
    Runs Adam on the log‑probabilities (soft‑max) of `start`.
    The smoothing parameter λ is increased slowly (lam_factor>1) so the
    optimiser starts on a smooth approximation and finishes on the true
    max‑convolution objective.
    Returns the best probability vector found and its objective value.
    """
    # initialise logarithmic variables
    eps = 1e-30
    logp = np.log(start + eps)          # shape (n,)
    n = start.size
    m = np.zeros_like(logp)             # first moment
    v = np.zeros_like(logp)             # second moment

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    best_seq = start.copy()
    best_val = _objective(start)

    t = 0
    lam = lam_start

    while t < max_iters and time.time() < deadline:
        t += 1

        # compute current probability vector (soft‑max, numerically stable)
        logp_shift = logp - np.max(logp)
        a = np.exp(logp_shift)
        a /= a.sum()

        # gradient w.r.t. a (smooth max)
        g_a = _gradient_wrt_a(a, lam)

        # chain rule to get gradient w.r.t. log‑probabilities
        dot = np.dot(a, g_a)
        g_log = a * (g_a - dot)               # shape (n,)

        # Adam updates (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log * g_log)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # step size – a small constant divided by sqrt(t) (works well in practice)
        lr = 0.15 / (math.sqrt(t) + 1.0)

        # parameter update
        logp -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # evaluate the true (non‑smoothed) objective
        cur_val = _objective(a)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d}  best={cur_val:.6f}  λ={lam:.2f}")

        # increase the smoothing parameter slowly (approaches the hard max)
        lam *= lam_factor

        # occasional random kick‑restart if we have made no progress for a while
        if t % 25000 == 0 and best_val - cur_val > 1e-6:
            # add a tiny Gaussian perturbation to log‑probabilities to escape shallow plateaus
            noise = np.random.default_rng().normal(scale=0.02, size=n)
            logp += noise

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing neighbour search (final polishing)
# ----------------------------------------------------------------------
def _anneal_refine(start: np.ndarray,
                   deadline: float,
                   max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis–Hastings walk that swaps a small fraction of mass
    between random coordinates and occasionally performs a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    n = cur.size
    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a small fraction from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d}  best={new_val:.6f}")

        # every ~2000 steps try a global perturbation
        if it % 2000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0.0:
                new_seq /= total
            else:
                new_seq = _uniform(n)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val
