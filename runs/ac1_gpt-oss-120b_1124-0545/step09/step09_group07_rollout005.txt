Pseudo:

a = a0.copy()
best_seq = a.copy()
best_val = _objective(a)

log_a = np.log(a + 1e-30)

alpha = initial_alpha
it = 0
while time.time() < deadline:
    it += 1
    # compute conv and gradient with current a (not log)
    a = np.exp(log_a)
    a = a / a.sum()
    conv = _conv_fft(a, a)
    # gradient:
    # compute softmax weights for conv
    # w = np.exp(alpha * (conv - conv.max())) ; w = w / w.sum()
    # compute gradient g = 2 * conv_fft(rev_a, w)[n-1:n-1+n]
    rev_a = a[::-1]
    w = np.exp(alpha * (conv - conv.max()))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.ones_like(conv) / conv.size
    else:
        w /= w_sum
    g = 2.0 * _conv_fft(rev_a, w)[len(a)-1:len(a)-1+len(a)]

    # learning rate schedule: e.g., lr = lr0 / sqrt(it)
    lr = lr0 / math.sqrt(it)
    # update log_a:
    log_a -= lr * g

    # normalization (to avoid overflow)
    log_a -= np.max(log_a)  # keep stable
    # compute new a
    a = np.exp(log_a)
    a = a / a.sum()
    cur_val = _objective(a)
    if cur_val < best_val:
        best_val = cur_val
        best_seq = a.copy()
        print(f"[Smooth] it={it} best={cur_val:.6f} alpha={alpha:.2f}")

    # increase alpha gradually after certain iteration count:
    if it % 2000 == 0 and alpha < max_alpha:
        alpha = min(alpha * alpha_factor, max_alpha)

    # optionally break if improvement tiny and time short.

Return best_seq, best_val

"""
Robust optimisation of a non‑negative step function for
`evaluate_sequence`.

The algorithm consists of three stages:

1️⃣ **Initialisation** – generate a few cheap seeds (uniform,
   random soft‑max, Sidon‑like, random sparse) and keep the
   best on the true objective.

2️⃣ **Smoothed max–convolution descent** – minimise a smooth
   approximation of the max‑convolution
   `M_α(a)= (1/α)·log(∑_k exp(α·(a*a)_k))` using
   exponentiated‑gradient / Adam updates on the probability simplex.
   The smoothing parameter `α` is increased geometrically (continuation)
   so that the optimisation first sees a gentle landscape and later
   focuses on the true max.

3️⃣ **Fine‑tuning** – a short simulated‑annealing walk (Metropolis‑Hastings)
   is applied to the best vector found to escape possible tiny
   local minima.

All operations respect the supplied time budget (default 1000 s) and
the returned sequence is normalised to sum = 1, guaranteeing that the
value of `evaluate_sequence` equals `2·n·max(conv(a,a))`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT, next‑power‑of‑2)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using an FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective   2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


def _random_sparse_seed(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k random positions, equal mass, normalised."""
    if k >= n:
        return _uniform(n)
    pos = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[pos] = 1.0 / k
    return a / a.sum()


# ----------------------------------------------------------------------
# Smoothed max–convolution optimisation (Adam on the simplex)
# ----------------------------------------------------------------------
def _smooth_max_opt(start_seq: np.ndarray,
                    deadline: float,
                    init_alpha: float = 1.0,
                    max_alpha: float = 500.0,
                    alpha_mult: float = 2.0,
                    lr: float = 0.5,
                    beta1: float = 0.9,
                    beta2: float = 0.999,
                    eps: float = 1e-8,
                    alpha_step: int = 2000) -> tuple[np.ndarray, float]:
    """
    Minimises a smooth approximation of `max(a*a)` using Adam
    on the probability simplex. Returns the best vector encountered.
    """
    a = start_seq.copy()
    n = a.size
    # work in log‑space to keep positivity automatically
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_seq = a.copy()
    best_val = _objective(a)

    alpha = init_alpha
    it = 0
    rel_tol = 1e-12

    while True:
        if time.time() > deadline:
            break
        it += 1

        # ----- obtain current a from log‑parameters -----
        a = np.exp(log_a)
        a /= a.sum()

        # ----- convolution and smoothed weights -----
        conv = _conv_fft(a, a)                # length 2n‑1
        max_c = conv.max()
        # softmax with current smoothing parameter
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (exact) -----
        rev_a = a[::-1]
        # gradient = 2 * (rev_a * w) (conv of rev_a with w)
        grad = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

        # ----- Adam update on log‑a -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep log‑values numerically stable
        log_a -= np.max(log_a)

        # ----- occasional exact evaluation -----
        if it % 500 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Smooth] it={it:7d} α={alpha:.1f} best={cur_val:.6f}")

        # ----- continuation on α -----
        if it % alpha_step == 0 and alpha < max_alpha:
            alpha = min(alpha * alpha_mult, max_alpha)

    # final evaluation (in case we never hit the 500‑step check)
    final_val = _objective(np.exp(log_a) / np.exp(log_a).sum())
    if final_val < best_val:
        best_val = final_val
        best_seq = np.exp(log_a) / np.exp(log_a).sum()
        print(f"[Smooth] final check best={final_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (same as the previous script)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")
