def propose_candidate(seed=123, budget_s=1000, **kwargs):
    import time, math, numpy as np
    rng = np.random.default_rng(seed)
    # Determine n
    n = 4000
    prior = globals().get('height_sequence_1')
    if prior is not None:
        arr = np.asarray(prior, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0:
            n = arr.size
            a = np.maximum(arr, 0.0)
            a_sum = a.sum()
            if a_sum < 1e-12:
                a = np.ones(n, dtype=np.float64) / n
            else:
                a /= a_sum
            print(f"[Init] Using prior sequence of length {n}")
        else:
            a = None
    else:
        a = None

    if a is None:
        n = 4000
        a = np.ones(n, dtype=np.float64) / n
        print("[Init] No prior sequence; starting from uniform")
    
    # initialise conv
    def conv_full(x):
        L = x.size + x.size - 1
        N = 1 << ((L - 1).bit_length())
        fx = np.fft.rfft(x, n=N)
        conv = np.fft.irfft(fx * fx, n=N)[:L]
        return conv

    conv = conv_full(a)
    max_val = conv.max()
    best_a = a.copy()
    best_val = 2.0 * n * max_val
    print(f"[Init] initial objective = {best_val:.6f}")

    deadline = time.time() + budget_s - 1.0
    moves = 0
    # temperature schedule
    T0 = best_val * 0.02  # initial temperature proportional to objective
    decay = 0.9995

    # loop
    while time.time() < deadline:
        moves += 1
        # current max index
        k = int(np.argmax(conv))

        # range of indices that contribute to this sum
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx_i = np.arange(low, high + 1)
        idx_j = k - idx_i
        # contributions to conv[k]
        contrib = a[idx_i] * a[idx_j]
        # choose source i with probability ~ contribution (to target most damaging)
        sum_contrib = contrib.sum()
        if sum_contrib > 0:
            probs = contrib / sum_contrib
            i = rng.choice(idx_i, p=probs)
        else:
            i = rng.integers(low, high + 1)

        # choose a destination j (different from i)
        # prefer locations with small a[k - j] (i.e., small contribution to current max)
        # compute candidate contributions for all j where k - j in range
        contrib_j = np.full(n, np.inf)
        valid_j = np.arange(n)
        # compute k - j in [0, n-1]
        mask = (k - valid_j >= 0) & (k - valid_j < n)
        contrib_j[mask] = a[k - valid_j[mask]]
        # set very small for out-of-range (they give 0 contribution)
        contrib_j[~mask] = 0.0
        # choose among low contribution indices
        low_contrib = np.percentile(contrib_j, 25)  # choose quarter lowest
        candidates = np.where(contrib_j <= low_contrib)[0]
        if candidates.size == 0:
            candidates = np.arange(n)
        # ensure i not in candidates
        if candidates.size > 1:
            # remove i if present
            candidates = candidates[candidates != i]
        if candidates.size == 0:
            # fallback to any index not i
            j = (i + 1) % n
        else:
            j = rng.choice(candidates)

        # mass to move
        max_move = a[i] * 0.5  # at most half of source
        if max_move <= 0:
            continue
        delta = rng.random() * max_move  # uniform in [0, max_move)

        # prepare new convolution
        conv_new = conv.copy()
        di = -delta
        dj = delta
        # update slices
        conv_new[i:i+n] += 2.0 * di * a
        conv_new[j:j+n] += 2.0 * dj * a
        # quadratic adjustments
        conv_new[2*i] += di * di
        conv_new[2*j] += dj * dj
        conv_new[i + j] += 2.0 * di * dj

        new_max = conv_new.max()

        # acceptance
        if new_max < max_val - 1e-15:
            accept = True
        else:
            T = T0 * (decay ** moves)
            prob = math.exp(-(new_max - max_val) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:
            # apply move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            max_val = new_max

            cur_obj = 2.0 * n * max_val
            if cur_obj < best_val - 1e-12:
                best_val = cur_obj
                best_a = a.copy()
                print(f"[HillClimb] move={moves:d}, best={best_val:.6f}")

        # optional renormalisation every 10000 moves
        if moves % 10000 == 0:
            tot = a.sum()
            if abs(tot - 1.0) > 1e-12:
                a /= tot
                conv = conv_full(a)   # recompute from scratch to avoid drift

    # final normalisation
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total < 1e-12:
        best_a = np.ones(n, dtype=np.float64) / n
        total = best_a.sum()
    best_a /= total

    final_obj = 2.0 * n * conv_full(best_a).max()
    print(f"\n=== Done ===\nBest objective ≈ {best_val:.6f} (length {n})")
    globals()["height_sequence_1"] = best_a.copy()
    return best_a.tolist()

"""
Hill‑climbing optimizer for the evaluation function

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works directly on the convolution array with O(n) incremental updates.
It repeatedly moves a small amount of mass from a point that contributes most to the
current maximal convolution value to a random destination.  The move is accepted
whenever it reduces the maximal convolution, otherwise it may be accepted with a
decreasing temperature (simulated‑annealing) to escape local minima.

A previous best sequence can be supplied via the global variable
``height_sequence_1``; otherwise the search starts from a uniform distribution.
All arithmetic is performed with NumPy for speed.  The routine respects the given
time budget (default 1000 s) and returns a list of non‑negative floats that can be
passed to ``evaluate_sequence``.
"""

import time
import math
import numpy as np


def _conv_full(x: np.ndarray) -> np.ndarray:
    """Linear full convolution of a real vector with itself using FFT."""
    L = x.size + x.size - 1
    N = 1 << ((L - 1).bit_length())            # next power of two
    fx = np.fft.rfft(x, n=N)
    conv = np.fft.irfft(fx * fx, n=N)[:L]
    return conv


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The best sequence found (as a plain list of floats) is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # ------------------------------------------------------------------
    # Determine problem size and initialise a starting sequence
    # ------------------------------------------------------------------
    prior = globals().get("height_sequence_1")
    if prior is not None:
        try:
            a0 = np.asarray(prior, dtype=np.float64)
            if a0.ndim == 1 and a0.size > 0:
                n = a0.size
                a0 = np.maximum(a0, 0.0)
                s = a0.sum()
                if s < 1e-12:
                    a0 = np.ones(n, dtype=np.float64) / n
                else:
                    a0 /= s
                print(f"[Init] Using prior sequence of length {n}")
            else:
                raise ValueError
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        n = 4000
        a0 = np.ones(n, dtype=np.float64) / n
        print("[Init] No usable prior – starting from uniform distribution")

    # ------------------------------------------------------------------
    # Initial convolution and bookkeeping
    # ------------------------------------------------------------------
    a = a0.copy()
    conv = _conv_full(a)               # length 2*n-1
    max_val = conv.max()
    best_a = a.copy()
    best_obj = 2.0 * n * max_val
    print(f"[Init] initial objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Hill‑climbing parameters
    # ------------------------------------------------------------------
    moves = 0
    T0 = best_obj * 0.02                # initial temperature (relative scale)
    decay = 0.9995

    # ------------------------------------------------------------------
    # Main optimisation loop (incremental convolution updates)
    # ------------------------------------------------------------------
    while time.time() < deadline:
        moves += 1

        # 1) locate current maximal convolution index
        k = int(np.argmax(conv))

        # 2) pick a source index i that contributes most to this max
        low = max(0, k - (n - 1))
        high = min(k, n - 1)
        idx = np.arange(low, high + 1, dtype=int)
        opp = k - idx                               # opposite indices
        prod = a[idx] * a[opp]                      # contribution a_i * a_{k-i}
        if prod.size == 0:
            continue
        i = idx[np.argmax(prod)]                    # most damaging source

        # 3) choose a destination j (different from i)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # 4) amount of mass to transfer (up to 30 % of source)
        if a[i] <= 0.0:
            continue
        delta = rng.random() * a[i] * 0.3
        if delta <= 0.0:
            continue

        # 5) compute the updated convolution (O(n) work)
        conv_new = conv.copy()
        di = -delta
        dj =  delta

        # linear term 2 * a * Δ
        conv_new[i:i + n] += 2.0 * di * a
        conv_new[j:j + n] += 2.0 * dj * a

        # quadratic term Δ * Δ
        conv_new[2 * i] += di * di
        conv_new[2 * j] += dj * dj
        conv_new[i + j] += 2.0 * di * dj   # i+j ≤ 2n‑2 guaranteed

        new_max = conv_new.max()

        # 6) acceptance test (always accept improvement, otherwise anneal)
        if new_max < max_val - 1e-15:
            accept = True
        else:
            T = T0 * (decay ** moves)
            prob = math.exp(-(new_max - max_val) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:
            # apply the move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            max_val = new_max

            cur_obj = 2.0 * n * max_val
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[HillClimb] move={moves:6d}  best={best_obj:.6f}")

        # ------------------------------------------------------------------
        # Periodic re‑normalisation to avoid drift (full recompute)
        # ------------------------------------------------------------------
        if moves % 10000 == 0:
            total = a.sum()
            if abs(total - 1.0) > 1e-12:
                a /= total
            conv = _conv_full(a)
            max_val = conv.max()
            cur_obj = 2.0 * n * max_val
            if cur_obj < best_obj - 1e-12:
                best_obj = cur_obj
                best_a = a.copy()
                print(f"[ReNorm] move={moves:6d}  best={best_obj:.6f}")

    # ------------------------------------------------------------------
    # Final safety normalisation and export
    # ------------------------------------------------------------------
    best_a = np.maximum(best_a, 0.0)
    total = best_a.sum()
    if total < 1e-12:
        best_a = np.ones(n, dtype=np.float64) / n
    else:
        best_a /= total

    final_obj = 2.0 * n * _conv_full(best_a).max()
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    # store for the next run
    globals()["height_sequence_1"] = best_a.copy()
    return best_a.tolist()
