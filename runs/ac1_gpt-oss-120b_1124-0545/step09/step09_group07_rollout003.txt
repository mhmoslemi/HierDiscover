            best_seq = a.copy()

        # compute softmax weights

"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

using a smoothed exponential‑gradient descent on the simplex,
followed optionally by a linearised LP refinement (if cvxpy is
available) and a short simulated‑annealing fine‑tune.  The
gradient of a smooth approximation of the max‑convolution is obtained
by a soft‑max weighting of all convolution entries; the smoothing
parameter is increased during the run, allowing the optimizer to
escape shallow local minima that trap a pure sub‑gradient method.
The algorithm respects a 1000 s time budget and stores the best
sequence found in the global ``height_sequence_1`` for warm‑starting
future calls.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional cvxpy – used only for the LP refinement
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Low‑level convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (return a probability vector on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    a /= a.sum()
    return a


def _exponential_seed(n: int, rate: float = 0.97) -> np.ndarray:
    """Geometric decay a_i ∝ rate**i."""
    a = rate ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _cosine_seed(n: int) -> np.ndarray:
    """Half‑cosine shape (non‑negative, symmetric)."""
    x = np.arange(n, dtype=np.float64)
    a = (np.cos(np.pi * x / (n - 1)) + 1.0) / 2.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent on the simplex
# ----------------------------------------------------------------------
def _smooth_exponential_gradient(start_seq: np.ndarray,
                                 deadline: float,
                                 start_time: float,
                                 rng: np.random.Generator,
                                 alpha_start: float = 5.0,
                                 alpha_end: float = 200.0,
                                 lr0: float = 0.7,
                                 stagnation_limit: int = 20000) -> tuple[np.ndarray, float]:
    """
    Minimise the objective using a smooth approximation of the max‑convolution.
    The smoothing parameter `alpha` is increased linearly from `alpha_start`
    to `alpha_end`.  The update is performed in the log‑domain (mirror descent
    on the simplex).  If progress stalls for `stagnation_limit` iterations,
    a small random perturbation is added to escape the local basin.
    Returns the best sequence encountered and its objective value.
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    # momentum for a little extra stability
    momentum = np.zeros_like(a)
    beta = 0.9

    last_improve_it = 0

    it = 0
    while True:
        it += 1
        now = time.time()
        if now > deadline:
            break

        # 1) full auto‑convolution and objective
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            last_improve_it = it
            print(f"[Smooth] it={it:7d} best={cur_val:.6f}")

        # 2) smoothing schedule (linear in elapsed fraction)
        elapsed = now - start_time
        total = deadline - start_time
        frac = min(1.0, elapsed / total)
        alpha = alpha_start + frac * (alpha_end - alpha_start)

        # 3) soft‑max weights over convolution entries
        max_conv = conv.max()
        # stable soft‑max: subtract max before exponentiation
        w = np.exp(alpha * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0:   # fallback – should not happen
            w = np.zeros_like(conv)
            w[np.isclose(conv, max_conv)] = 1.0
            w_sum = w.sum()
        w /= w_sum

        # 4) gradient of the smooth objective: 2 * rev_a ⋆ w
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]

        # 5) momentum smoothing
        momentum = beta * momentum + (1.0 - beta) * g

        # 6) step‑size (decaying like 1/√t)
        lr = lr0 / math.sqrt(it)

        # 7) mirror‑descent in log‑space
        log_a = np.log(a + 1e-30) - lr * momentum
        log_a -= np.max(log_a)                     # numerical stability
        a = np.exp(log_a)
        a /= a.sum()

        # 8) occasional random jitter when stagnating
        if it - last_improve_it > stagnation_limit:
            # Add a tiny, zero‑mean multiplicative noise, then re‑normalise.
            noise = rng.normal(scale=5e-4, size=n)
            a = a * (1.0 + noise)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 1e-12:
                a /= s
            else:
                a = _uniform(n)
            last_improve_it = it  # reset counter after perturbation

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linearised LP refinement (same as the previous code)
# ----------------------------------------------------------------------
def _lp_linearised_refine(a: np.ndarray,
                         deadline: float,
                         max_refine: int = 8,
                         eps_active: float = 1e-9) -> tuple[np.ndarray, float]:
    """Linearise the quadratic constraints around `a` and solve a small LP."""
    if not CVXPY_AVAILABLE:
        return a, _objective(a)

    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    x_var = cp.Variable(n)
    M_var = cp.Variable()
    base_constraints = [x_var >= 0,
                        cp.sum(x_var) == 1]

    for it in range(max_refine):
        if time.time() > deadline - 0.3:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]
        if active.size == 0:
            break

        # Build matrix A where A[row,i] = a[active[row] - i] (if the index is valid)
        A = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            A[r, i_range] = a[k - i_range]

        lin_constraints = [2 * (A @ x_var) <= M_var + conv[active]]
        prob = cp.Problem(cp.Minimize(M_var), base_constraints + lin_constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=2000,
                       abstol=1e-9,
                       reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if x_var.value is None or M_var.value is None:
            break

        a_new = np.asarray(x_var.value).flatten()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s

        val_new = _objective(a_new)
        print(f"[LP] iter={it + 1:2d} M={M:.6f} new_obj={val_new:.6f}")

        if val_new < best_val - 1e-9:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new  # continue from improved point
        else:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuner (unchanged)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Tiny Metropolis‑Hastings walk that swaps a small amount of mass."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of floats) that minimises
    `evaluate_sequence`.  The routine respects the supplied time budget,
    keeps the best sequence in the global `height_sequence_1` and returns it.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    rng = np.random.default_rng(seed + 12345)

    # --------------------------------------------------------------
    # Problem size – a trade‑off between quality and speed
    # --------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function of length {n} ===")

    # --------------------------------------------------------------
    # Initialise a decent starting point (reuse previous if possible)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    seed_vec = None
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                seed_vec = arr / arr.sum()
        except Exception:
            pass

    if seed_vec is None:
        # generate a few cheap seeds and keep the best
        candidate_seeds = [
            _uniform(n),
            _random_softmax(n, rng),
            _sidon_like_seed(n, min(120, n)),
            _exponential_seed(n, rate=0.97),
            _cosine_seed(n),
        ]
        best_val = float("inf")
        best_seq = None
        for cand in candidate_seeds:
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_seq = cand
        seed_vec = best_seq
        print(f"[Init] best initial objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Smoothed exponential‑gradient descent
    # --------------------------------------------------------------
    sub_deadline = deadline - 30.0   # reserve ~30 s for LP + annealing
    if sub_deadline > time.time():
        print("[Phase 1] Smoothed exponential‑gradient start")
        seq_sub, val_sub = _smooth_exponential_gradient(
            seed_vec, sub_deadline, start_time, rng,
            alpha_start=5.0, alpha_end=180.0, lr0=0.7, stagnation_limit=20000)
        print(f"[Phase 1] best objective after smoothing = {val_sub:.6f}")
    else:
        seq_sub, val_sub = seed_vec, _objective(seed_vec)

    # --------------------------------------------------------------
    # Phase 2 – Linearised LP refinement (optional)
    # --------------------------------------------------------------
    lp_deadline = deadline - 10.0   # keep a few seconds for annealing
    if CVXPY_AVAILABLE and lp_deadline > time.time():
        print("[Phase 2] LP linearised refinement start")
        seq_lp, val_lp = _lp_linearised_refine(seq_sub, lp_deadline)
        if val_lp < val_sub:
            seq_sub, val_sub = seq_lp, val_lp
            print(f"[Phase 2] improved objective = {val_lp:.6f}")
        else:
            print("[Phase 2] no improvement")
    else:
        print("[Phase 2] skipped (no cvxpy or insufficient time)")

    # --------------------------------------------------------------
    # Phase 3 – Simulated annealing fine‑tuning (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 3] Simulated annealing fine‑tuning")
        seq_final, val_final = _anneal_optimize(seq_sub, deadline)
        if val_final < val_sub:
            seq_sub, val_sub = seq_final, val_final
            print(f"[Phase 3] final objective = {val_final:.6f}")
        else:
            print("[Phase 3] annealing did not improve")
    else:
        print("[Phase 3] no time left for annealing")

    # --------------------------------------------------------------
    # Final safety normalisation and export
    # --------------------------------------------------------------
    final_seq = np.maximum(seq_sub, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    # store for warm start in future calls
    globals()["height_sequence_1"] = final_seq.copy()
    # return as plain Python list (the grader will convert to float)
    return final_seq.tolist()
