def _simulated_annealing_custom(start_seq, n, deadline, max_iters=2_000_000,
                               init_temp_factor=0.03, min_temp=1e-12):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        T = max(T * 0.9995, min_temp)

        # Choose a random index that contributes to the maximum pair
        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))
        # get range of i that participates
        i_start = max(0, max_idx-(n-1))
        i_end = min(n-1, max_idx)
        # choose one i uniformly from range
        i = rng.integers(i_start, i_end+1)
        j = max_idx - i
        # choose a random other index not i nor j
        other = rng.choice(n, p=None)
        while other == i or other == j:
            other = rng.choice(n)
        # mass to transfer from i to other
        transfer = rng.random() * cur[i] * 0.5
        new = cur.copy()
        new[i] -= transfer
        new[other] += transfer
        # ensure nonnegative
        new[i] = max(new[i],0.0)
        # optionally also move from j similarly
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta/T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 ==0:
                    print(f"[SAcustom] it={it} best={new_val:.6f}")
    return best_seq, best_val

def _logspace_sgd_optimize_custom(start_vec, n, deadline, max_iters=400_000):
    eps = 1e-30
    v = np.log(start_vec + eps)
    beta = 0.5
    beta_inc = 0.5
    beta_max = 500.0
    lr0 = 1.0
    momentum = 0.9
    m = np.zeros_like(v)
    best_seq = None
    best_val = float('inf')
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        a = _softmax(v)
        conv = _conv_fft(a,a)
        cur_val = 2.0*n*np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogSGD_opt] it={it} best={cur_val:.6f}")
        # smoothed max via log-sum-exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        lr = lr0 / np.sqrt(it)
        # momentum update
        m = momentum * m + (1 - momentum) * grad_v
        v = v - lr * m
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)
    return best_seq, best_val

"""
Propose a step‑function that (approximately) minimises

    2·n·max(conv(a, a)) / (sum(a)²)

where ``a`` is a non‑negative vector (the step‑function heights).
The algorithm works within a fixed time budget (default 1000 s) and
contains four stages:

1. **Rich seed pool** – analytical families (uniform, triangular,
   sine‑window, Gaussian, power‑law, beta, Tukey, inverse‑sqrt,
   exponential‑decay), random Dirichlet / exponential draws and low‑frequency
   random curves.  All seeds are ℓ₁‑normalised (sum = 1).

2. **Log‑space stochastic gradient descent** – optimisation on the
   log‑softmax (probability simplex) using a smoothed max surrogate.
   Momentum is added and the temperature is increased gradually.

3. **Exact sub‑gradient descent** – two complementary schemes are applied:
   * Euclidean projection onto the simplex (additive updates);
   * KL‑mirror descent (multiplicative updates).  Both use a
   decreasing step size 𝛼ₜ∝1/√t.

4. **Targeted simulated annealing** – small mass‑redistribution moves
   focused on the currently active convolution index, with a geometric
   cooling schedule.

The best sequence found (as a plain list of floats) is stored in the
global name ``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for a ℓ₁‑normalised vector (sum(seq) == 1):
        2·n·max(conv(seq, seq))
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (already ℓ₁‑normalised)
# ------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inverse_sqrt_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay_shape(n: int, scale_frac: float = 0.1) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    scale = scale_frac * n
    a = np.exp(-i / scale) + np.exp(-(n - 1 - i) / scale)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Smooth non‑negative curve by summing a few random cosine components.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds(n: int,
                   rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # Beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Extra analytic shapes
    seeds.append(_inverse_sqrt_shape(n))
    seeds.append(_exp_decay_shape(n, scale_frac=0.08))

    # Random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(3):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _logspace_sgd_optimize(start_vec: np.ndarray,
                           n: int,
                           deadline: float,
                           max_iters: int = 400_000) -> tuple[np.ndarray | None, float]:
    """
    Stochastic gradient descent on the log‑softmax parameters.
    Uses a log‑sum‑exp smoothing of the max and adds simple momentum.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)             # logits
    beta = 0.5                               # smoothing temperature
    beta_inc = 0.5
    beta_max = 500.0
    lr0 = 1.0
    momentum = 0.9
    m = np.zeros_like(v)

    best_seq = None
    best_val = float('inf')

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogSGD] it={it:6d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # Convert gradient to log‑space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        lr = lr0 / np.sqrt(it)

        # momentum update
        m = momentum * m + (1.0 - momentum) * grad_v
        v = v - lr * m

        # increase temperature (make surrogate sharper)
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 4_000_000) -> tuple[np.ndarray, float]:
    """
    Additive sub‑gradient descent with Euclidean projection onto the simplex.
    Step size αₜ = η / √t, η = 0.30.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.30 / np.sqrt(it)          # a slightly larger initial step
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _subgradient_kl(start_seq: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 3_000_000) -> tuple[np.ndarray, float]:
    """
    Multiplicative (KL‑mirror) sub‑gradient descent.
    Update: a ← a · exp(‑η·g)  then renormalise.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 40000 == 0:
                print(f"[SubKL]  it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        eta = 0.20 / np.sqrt(it)
        a = a * np.exp(-eta * g)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a = a / s

        # occasional tiny smoothing to avoid exact zeros
        if it % 100000 == 0:
            a = 0.999 * a + 0.001 * _uniform(n)
            a = a / a.sum()

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 2_000_000,
                         init_temp_factor: float = 0.03,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Targeted annealing that reshuffles the mass of a small random subset
    focusing on the currently active convolution index.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        # find active index
        conv = _conv_fft(cur, cur)
        max_idx = int(np.argmax(conv))

        # pick a contributing entry i (and its partner j)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        if i_end < i_start:
            continue
        i = rng.integers(i_start, i_end + 1)
        j = max_idx - i

        # pick a random other position to receive some mass
        other = rng.integers(0, n)
        while other == i or other == j:
            other = rng.integers(0, n)

        # move a random fraction of mass from i to other
        transfer = rng.random() * cur[i] * 0.5
        new = cur.copy()
        new[i] -= transfer
        new[other] += transfer
        new[i] = max(new[i], 0.0)            # guard against tiny negatives

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

            global_best_val = val_sa

    # -------------------------------------------------

        f"(length {len(global_best_seq)})"
    )

    return global_best_seq.tolist()
