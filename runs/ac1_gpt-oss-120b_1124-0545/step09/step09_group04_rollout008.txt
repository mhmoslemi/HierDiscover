"""
Non‑negative step‑function optimiser (new version).

Key ideas
---------

* Diverse families of analytic shapes are generated **and their entries are
  randomly permuted** – this breaks the symmetry that many earlier methods
  keep, creating a richer set of starting points.

* A *smoothed* version of the max‑convolution objective is minimised:
  ```
      Fβ(a) = (1/β) * log Σj exp(β·c_j)   ,  c = a ∗ a
  ```
  For small β the landscape is smooth; β is increased slowly so that the
  optimiser finally concentrates on the true peak.  The gradient of Fβ can
  be written in closed form and evaluated with a single FFT per iteration.

* The optimisation is performed on the probability simplex using an
  exponentiated‑gradient (mirror‑descent) update with an Adam‑style
  adaptive learning‑rate.  After every update the vector is normalised,
  guaranteeing non‑negativity and Σa = 1.

* A short simulated‑annealing polishing phase is applied to the best
  candidate discovered by the gradient loop.

The routine respects the 1000 s time‑budget, runs entirely on the CPU and
returns a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, N: int) -> np.ndarray:
    """Linear convolution of two real vectors using FFT of length N."""
    A = np.fft.rfft(a, n=N)
    B = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(A * B, n=N)
    L = len(a) + len(b) - 1
    return conv[:L]


def _full_conv(a: np.ndarray, N: int) -> np.ndarray:
    """Convolution a ∗ a (full length) using a pre‑chosen FFT size N."""
    A = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(A * A, n=N)
    L = 2 * len(a) - 1
    return conv[:L]


def _corr_fft(a: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Linear correlation Σ_i a_i w_{i+k} for k = 0 … len(a)-1.
    Implemented as ifft(fft(a) * conj(fft(w))) with zero‑padding.
    """
    A = np.fft.fft(a, n=N)
    W = np.fft.fft(w, n=N)
    corr = np.fft.ifft(A * np.conj(W)).real
    return corr[: len(a)]


def _objective(a: np.ndarray, N: int) -> float:
    """Exact evaluation (2·n·max(conv)) for a vector with Σa = 1."""
    conv = _full_conv(a, N)
    return float(2.0 * len(a) * np.max(conv))


# ----------------------------------------------------------------------
# Analytic shape families (all returned normalised to Σ = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction (large but cheap)
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Create a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric (grid)
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))
        seeds.append(rng.permutation(_sym_geom(n, r)))   # permuted variant

    # unsymmetric geometric
    for r in np.linspace(0.985, 0.9999, 12):
        base = _geom_decay(n, r)
        seeds.append(base)
        seeds.append(rng.permutation(base))

    # power‑law decay
    for p in np.arange(0.1, 5.01, 0.3):
        base = _power_decay(n, p)
        seeds.append(base)
        seeds.append(rng.permutation(base))

    # beta family – finer grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            base = _beta_shape(n, a, b)
            seeds.append(base)
            seeds.append(rng.permutation(base))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitive – smoothed‑max exponentiated‑gradient (Adam)
# ----------------------------------------------------------------------


def _smooth_expgrad_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 0.8,
    beta_start: float = 1.0,
    beta_end: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Adam‑style exponentiated‑gradient descent on the smooth surrogate
    (log‑sum‑exp) of the max‑convolution.
    Returns the best vector seen (in terms of the exact objective) and its
    exact objective value.
    """

    a = start_seq.copy()
    eps = 1e-30

    # Adam parameters
    b1, b2 = 0.9, 0.999
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # FFT size – must accommodate correlation of a (len n) with w (len 2n-1)
    N = _next_pow2(3 * n)

    best_seq = a.copy()
    best_val = _objective(a, N)

    beta = beta_start
    beta_factor = (beta_end / beta_start) ** (1.0 / max_iters)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- convolution a∗a (full length) ---------------------------------
        conv_full = _full_conv(a, N)                     # length 2n-1
        max_conv = conv_full.max()

        # --- smooth max (log‑sum‑exp) ------------------------------------
        beta = min(beta_end, beta * beta_factor)        # geometric increase
        z = beta * conv_full
        z_max = z.max()
        w = np.exp(z - z_max)
        w_sum = w.sum()
        w /= w_sum                                      # soft‑max weights

        # --- gradient of smooth max (correlation) -------------------------
        # grad_k = 2 * Σ_i a_i * w_{i+k}
        grad = 2.0 * _corr_fft(a, w, N)

        # --- Adam update on the *log* of the simplex variable -------------
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad ** 2)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)

        lr = lr0 / np.sqrt(it)                         # decaying base LR
        # Exponentiated‑gradient step on log‑space
        log_a = np.log(a + eps) - lr * (m_hat / (np.sqrt(v_hat) + eps))
        log_a -= np.max(log_a)                         # stabilise
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

        # --- keep exact best ------------------------------------------------
        cur_val = _objective(a, N)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report (quiet for speed)
            if it % 200_000 == 0:
                print(f"[ExpGrad] it={it:,d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp_factor: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles mass of a small random subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur, _next_pow2(3 * n))
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, 1e-14)

        k = rng.integers(2, 7)                # subset size 2…6
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new, _next_pow2(3 * n))

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA]    it={it:,d} best≈{best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal ``evaluate_sequence``
    value.  The function respects the global runtime budget, keeps the best
    sequence across several optimisation runs and finally stores it in the
    global name ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0    # safety margin

    # ------------------------------------------------------------------
    # Re‑use a previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose length – 4000 is a good compromise between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate a large pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate all seeds with the exact objective (they are already normalised)
    N_fft = _next_pow2(3 * n)
    evaluated = [(vec, _objective(vec, N_fft)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – aggressive smoothed‑max exponentiated‑gradient optimisation
    # ------------------------------------------------------------------
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq, N_fft)
    print(f"\n[Init] best seed exact = {global_best_val:.6f}")

    # allocate roughly equal time slices to the top seeds
    remaining = deadline - time.time()
    per_seed_deadline = time.time() + remaining / max(1, top_k)

    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimising seed #{idx+1} ---")
        # give each seed a fraction of the remaining budget
        seed_deadline = min(deadline, time.time() + (deadline - time.time()) / (top_k - idx))
        seq_opt, val_opt = _smooth_expgrad_optimize(
            seed_vec,
            n,
            seed_deadline,
            max_iters=3_000_000,
            lr0=0.9,
            beta_start=1.0,
            beta_end=250.0,
        )
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[Overall] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – optional simulated‑annealing polish
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=150_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[Overall SA] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the external evaluator will later be called)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # Store for reuse and return
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
