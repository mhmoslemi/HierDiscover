Pseudo:

def build_low_collision_mask(n, c, deadline):
    best_mask = None
    best_size = -1
    rng = np.random.default_rng()
    while time.time() < deadline:
        # random greedy
        perm = rng.permutation(n)
        mask = np.zeros(n, dtype=bool)
        counts = np.zeros(2*n-1, dtype=np.uint8)
        for idx in perm:
            # compute contributions if added:
            # compute self sum = 2*idx
            if counts[2*idx] + 1 > c:
                continue
            # compute sums with existing
            existing = np.where(mask)[0]
            if existing.size>0:
                sums = idx + existing
                # contributions +2 each
                if np.any(counts[sums] + 2 > c):
                    continue
            # accept:
            mask[idx]=True
            counts[2*idx] += 1
            if existing.size>0:
                counts[sums] += 2
        size = mask.sum()
        if size > best_size:
            best_size = size
            best_mask = mask.copy()
    return best_mask
```

                continue
            # check feasibility
            if cur_counts[2*cand] + 1 > max_c:
                continue
            if cur_elements.size:
                sums = cand + cur_elements
                if np.any(cur_counts[sums] + 2 > max_c):
                    continue

            cur_mask[cand] = True
            cur_counts[2*cand] += 1
            if cur_elements.size:
                cur_counts[sums] += 2
                cur_elements = np.append(cur_elements, cand)
            else:
                cur_elements = np.array([cand], dtype=np.int32)

"""
propose_candidate
-----------------
A 1000‑second time‑bounded optimiser that focuses on the combinatorial
core of the problem: building a large “low‑collision’’ support set
S⊂{0,…,n‑1}.  For a uniform distribution on S the convolution value at
position k equals (number of ordered pairs (i,j)∈S×S with i+j=k) /
|S|².  Hence the objective for a uniform distribution is

    2·n·max_k count_k / |S|²

and we simply have to maximise |S| while keeping the maximal ordered
count ≤ c (c=4,5,…).  The code runs a greedy construction many times,
followed by a fast local‑search (additions and swaps) that respects the
collision bound.  The best mask found is turned into a uniform step
function and returned.  If a previous solution exists it is kept as a
fallback.

The algorithm is deliberately simple – it does not perform the expensive
gradient‑based phases of the previous solution, because a good low‑collision
support already yields an objective below the target 1.503.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# Greedy construction of a low‑collision set
# ------------------------------------------------------------
def _greedy_build(n: int, max_c: int, rng: np.random.Generator):
    """Return mask, ordered‑pair count array and element list for a random
    greedy construction that never exceeds ``max_c`` ordered collisions."""
    mask = np.zeros(n, dtype=bool)
    counts = np.zeros(2 * n - 1, dtype=np.int32)          # ordered pair counts
    order = rng.permutation(n)                           # random insertion order
    elements = np.empty(0, dtype=np.int32)                # current support

    for idx in order:
        # self‑pair
        if counts[2 * idx] + 1 > max_c:
            continue
        # collisions with already chosen elements
        if elements.size:
            sums = idx + elements
            if np.any(counts[sums] + 2 > max_c):
                continue

        # accept the candidate
        mask[idx] = True
        counts[2 * idx] += 1
        if elements.size:
            counts[sums] += 2
            elements = np.append(elements, idx)
        else:
            elements = np.array([idx], dtype=np.int32)

    return mask, counts, elements


# ------------------------------------------------------------
# Local improvement (additions and swaps) keeping feasibility
# ------------------------------------------------------------
def _local_improve(mask, counts, elements, n, max_c, rng, deadline):
    """Try to enlarge / reshuffle the set while never violating ``max_c``."""
    cur_mask = mask
    cur_counts = counts
    cur_elements = elements
    cur_size = cur_elements.shape[0]

    while time.time() < deadline:
        # ----------------------------------------------------------------
        # 1) try a plain addition (increase the support size)
        # ----------------------------------------------------------------
        if cur_size == 0 or (cur_size < n and rng.random() < 0.5):
            cand = rng.integers(n)
            if cur_mask[cand]:
                continue
            # feasibility test for the addition
            if cur_counts[2 * cand] + 1 > max_c:
                continue
            if cur_size:
                sums = cand + cur_elements
                if np.any(cur_counts[sums] + 2 > max_c):
                    continue
            # perform the addition
            cur_mask[cand] = True
            cur_counts[2 * cand] += 1
            if cur_size:
                cur_counts[sums] += 2
                cur_elements = np.append(cur_elements, cand)
            else:
                cur_elements = np.array([cand], dtype=np.int32)
            cur_size += 1
            continue

        # ----------------------------------------------------------------
        # 2) try a swap (remove one element, insert another)
        # ----------------------------------------------------------------
        if cur_size == 0:
            continue
        i_pos = rng.integers(cur_size)      # element that will be removed
        i = cur_elements[i_pos]
        cand = rng.integers(n)               # new element to insert
        if cur_mask[cand]:
            continue

        # copy counts and simulate the removal of i
        temp_counts = cur_counts.copy()
        temp_counts[2 * i] -= 1
        rest = np.delete(cur_elements, i_pos)      # set after removal
        if rest.size:
            temp_counts[i + rest] -= 2

        # feasibility test for the insertion of cand
        if temp_counts[2 * cand] + 1 > max_c:
            continue
        if rest.size:
            sums2 = cand + rest
            if np.any(temp_counts[sums2] + 2 > max_c):
                continue
            temp_counts[sums2] += 2

        temp_counts[2 * cand] += 1

        if temp_counts.max() > max_c:           # safety check
            continue

        # accept the swap
        cur_mask[i] = False
        cur_mask[cand] = True
        cur_counts = temp_counts
        cur_elements = np.append(rest, cand)
        # size unchanged – only the composition altered

    return cur_mask, cur_counts, cur_elements


# ------------------------------------------------------------
# Search for the best feasible mask within a given time budget
# ------------------------------------------------------------
def _search_best_mask(n: int, max_c: int, rng: np.random.Generator,
                     deadline: float):
    """Repeated greedy + local‑search attempts; return best mask and its
    uniform‑distribution objective."""
    best_mask = None
    best_obj = float("inf")
    best_m = 0

    while time.time() < deadline:
        mask, counts, elements = _greedy_build(n, max_c, rng)

        # give a tiny slice of time to polish the current mask
        improve_deadline = min(deadline, time.time() + 0.06)
        mask, counts, elements = _local_improve(
            mask, counts, elements, n, max_c, rng, improve_deadline
        )

        m = elements.shape[0]
        if m == 0:
            continue
        max_cnt = counts.max()
        obj = 2.0 * n * max_cnt / (m * m)   # objective for uniform weights

        if obj < best_obj:
            best_obj = obj
            best_mask = mask.copy()
            best_m = m
            print(
                f"[c={max_c}] new best size={m} max_cnt={max_cnt} "
                f"obj={obj:.6f}"
            )
        # optional early exit if we already reached the target bound
        if best_obj < 1.503:
            break

    return best_mask, best_obj

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    Search for a non‑negative step function (list of floats) that minimises

    start_time = time.time()

    target_obj = 1.503                       # desired bound
    best_seq = None
    best_val = float("inf")

    # ----------------------------------------------------------------

    # we try several maximal collision thresholds; start with the tightest
    max_c_list = [4, 5, 6]
