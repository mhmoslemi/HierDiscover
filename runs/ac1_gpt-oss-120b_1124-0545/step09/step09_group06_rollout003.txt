def generate_set(n, max_ordered, rng):
    mask = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2*n - 1, dtype=np.int32)
    order = rng.permutation(n)
    for i in order:
        # compute new sums with existing elements.
        if sum_counts[2*i] + 1 > max_ordered: continue
        existing = np.nonzero(mask)[0]
        if existing.size > 0:
            sums = i + existing
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue
        # add
        mask[i] = True
        sum_counts[2*i] += 1
        if existing.size > 0:
            sum_counts[sums] += 2
    return mask

"""
propose_candidate
-----------------

A time‑bounded optimiser that focuses on constructing a large low‑collision
support set S⊂{0,…,n‑1} such that every ordered sum i+j (i,j∈S) occurs only a few
times.  For a uniform distribution on S we have

    evaluate_sequence(a) = 2·n·max_k N_k / |S|²,

where N_k is the number of ordered pairs with sum k.
Hence minimising max N_k while maximising |S| drives the objective down.
The algorithm works in two stages:

1. **Aggressive low‑collision search** (≈60 % of the time budget).  
   Many greedy constructions are tried with random orderings and with
   maximal ordered‑collision limits of 4 or 5.  The best mask found is
   turned into a uniform probability vector.

2. **Lightweight Adam refinement** (remaining time).  
   Starting from the uniform mask we run a few Adam steps on a smooth
   approximation of the max‑convolution, which usually yields a small
   further reduction.

The final sequence is normalised to sum to 1, clipped to non‑negative,
and stored in the global ``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (exact, real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective_fast(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq))/(sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Greedy low‑collision set construction
# ----------------------------------------------------------------------
def _greedy_low_coll(n: int, max_ordered: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a mask of indices S⊂{0,…,n‑1} such that each ordered sum i+j
    occurs at most ``max_ordered`` times.
    """
    mask = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    support = []                     # list of chosen indices
    order = rng.permutation(n)
    max_possible = int(math.isqrt(8 * n))   # theoretical upper bound for max_ordered=4

    for cand in order:
        # self‑pair
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue

        if support:
            # sums with already chosen elements
            sums = cand + np.array(support, dtype=int)
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue

        # accept candidate
        mask[cand] = True
        sum_counts[2 * cand] += 1
        if support:
            sum_counts[sums] += 2
        support.append(int(cand))

        if len(support) >= max_possible:
            break

    return mask


def _evaluate_mask(mask: np.ndarray, n: int) -> tuple[int, int, float]:
    """
    Given a boolean mask, compute:
        m       = |S|
        max_cnt = max ordered‑pair count for any sum
        obj     = 2·n·max_cnt / m²   (same as evaluate_sequence for uniform weights)
    """
    idx = np.where(mask)[0]
    m = idx.size
    if m == 0:
        return 0, 0, np.inf

    # compute ordered‑pair counts (O(m²) – cheap for m≈200)
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

    for i, a in enumerate(idx):
        sum_counts[2 * a] += 1                     # (a,a)
        for b in idx[i + 1:]:
            sum_counts[a + b] += 2                 # (a,b) and (b,a)

    max_cnt = int(sum_counts.max())
    obj = 2.0 * n * max_cnt / (m * m)
    return m, max_cnt, obj


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the exact objective, using a temperature‑annealed
    soft‑max as a smooth approximation of the max‑convolution.
    Returns the best vector seen and its true objective.
    """
    # initialise logits (log‑softmax) to stay on the probability simplex
    v = np.log(seed + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # temperature for the soft‑max over the convolution
    beta = 1.5
    beta_mul = 1.006      # increase slowly → tighter approximation

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                       # current probability vector
        cur_val = _objective_fast(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ----- smooth‑max weights -----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of Σ w_k·(a*a)_k -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a ~1000 s time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed + 12345)
    start_time = time.time()
    # keep a small safety margin for interpreter overhead
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 6000 offers a good balance between quality and speed.
    # ------------------------------------------------------------------
    n = 6000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive search for a large low‑collision support set
    # ------------------------------------------------------------------
    # Allocate roughly 60 % of the budget for this stage.
    lowcoll_deadline = start_time + 0.60 * (deadline - start_time)

    best_mask = None
    best_obj = float("inf")
    best_m = 0
    best_maxcnt = 0
    attempts = 0

    while time.time() < lowcoll_deadline:
        attempts += 1
        # randomly pick a maximum ordered collision budget (4 is ideal,
        # 5 is allowed – the objective will capture the true penalty)
        max_ordered = rng.choice([4, 5])

        seq = np.zeros(n, dtype=np.float64)
        seq[best_mask] = 1.0 / best_m
        print(f"[Info] selected support size {best_m} (max ordered count {best_maxcnt}) "

    # ------------------------------------------------------------------
    # Phase 2 – brief Adam refinement (the remaining budget)
    # ------------------------------------------------------------------

        max_iters = int(min(20000, (adam_deadline - time.time()) / 0.0015))
        seq_opt, val_opt = _adam_optimize(seq, n, adam_deadline, max_iters=max_iters)
        if seq_opt is not None and val_opt < best_obj:

    else:

    print(f"\n=== Finished ===\nBest objective ≈ {best_obj:.6f} (length {len(seq)})")

    # expose for the next iteration (if any)
    globals()["height_sequence_1"] = seq.copy()
    return seq.tolist()
