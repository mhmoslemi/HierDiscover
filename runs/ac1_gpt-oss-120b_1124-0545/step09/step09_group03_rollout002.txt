Algorithm (≈ 1000 s budget)

1. **Analytic seed phase** – evaluate a collection of classic windows
   (uniform, Hann, Hamming, Blackman, Catalan, Gaussian, arcsine,
   DPSS).  Keep the best shape.

2. **Global parametric optimisation** – model the distribution as a
   mixture of three Beta‑pdfs.  Optimise the 9 free parameters with
   SciPy's ``differential_evolution`` (a population‑based global search)
   while respecting the time budget.

3. **Local smooth optimisation** – starting from the best mixture,
   run a high‑temperature Adam optimiser on the log‑softmax
   (log‑its) representation, smoothly approximating the max‑convolution
   via a log‑sum‑exp.  Temperature is annealed from 5 to a large value
   (≈ 500) to obtain a very flat autocorrelation.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:

def _beta_mixture_sequence(params: np.ndarray, n: int, x_grid: np.ndarray) -> np.ndarray:
    """
    Construct a probability vector as a mixture of three Beta pdfs.
    params[0:3] – raw (log) weights,
    params[3:6] – alphas,
    params[6:9] – betas.
    The result is already normalised to sum = 1.
    """
    raw_w = params[0:3]
    alphas = params[3:6]
    betas = params[6:9]

    # softmax weights to ensure they sum to one and are positive
    w_exp = np.exp(raw_w - np.max(raw_w))
    weights = w_exp / w_exp.sum()

    seq = np.zeros(n, dtype=np.float64)
    for i in range(3):
        a = alphas[i]
        b = betas[i]
        # compute beta pdf on the fixed grid
        comp = (x_grid ** (a - 1.0)) * ((1.0 - x_grid) ** (b - 1.0))
        comp_sum = comp.sum()
        if comp_sum == 0.0:
            comp = np.full_like(comp, 1.0 / n)
        else:
            comp = comp / comp_sum
        seq += weights[i] * comp
    # numeric safety – renormalise
    total = seq.sum()
    if total == 0.0:
        return _uniform(n)
    return seq / total

# Smoothed Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    The sequence is represented via log‑softmax (log‑its).  A
    temperature schedule anneals from beta_start to beta_end.
    Returns the best sequence found and its exact objective.
    """
    n = len(start_seq)
    eps = 1e-30

    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max to obtain probabilities
        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum

        # convolution a * a
        conv = _conv_fft(a, a)

        # temperature schedule (linear up‑to 60 % of iterations)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # grad wrt a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            # early exit if already excellent
            if best_val < 1.45:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Aggressive sub‑gradient mirror descent (aggregated active set)
# ----------------------------------------------------------------------


def _refine_aggregated_subgradient(start_seq: np.ndarray, deadline: float, eps_rel: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* near‑maximal
    convolution entries (within a relative tolerance eps_rel).  Returns the
    best sequence found and its exact objective.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # stabilise exponentiation
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")

            if time.time() > deadline - 1.0:
                break

    return best_seq, best_val

        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (FFT‑friendly, moderately large)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000]

    best_seq = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds (including DPSS)
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
        _dpss_seed,
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                best_n = n
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # try to reuse previous solution when length matches
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                best_n = n
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – global optimisation of a three‑Beta mixture (DE)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and differential_evolution is not None and best_n is not None:
        # allocate about 30 % of remaining time to DE
        de_deadline = time.time() + remaining * 0.30
        global _DE_DEADLINE
        _DE_DEADLINE = de_deadline

        n = best_n
        x_grid = (np.arange(n, dtype=np.float64) + 0.5) / n

        def de_obj(params):
            seq = _beta_mixture_sequence(params, n, x_grid)
            return _objective(seq)

        bounds = [(-5, 5)] * 3 + [(0.1, 10.0)] * 6

        # Keep the number of evaluations modest to stay within the budget
        result = differential_evolution(
            de_obj,
            bounds,
            maxiter=25,         # ~25 generations
            popsize=15,
            tol=1e-6,
            callback=_de_time_callback,
            polish=False,
            updating='deferred',
            disp=False,
        )
        seq_de = _beta_mixture_sequence(result.x, n, x_grid)
        val_de = _objective(seq_de)
        if val_de < best_val:
            best_val = val_de
            best_seq = seq_de.copy()
            print(f"[DE] length={n} val={val_de:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – local Adam refinement (smooth max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 3.0 and best_seq is not None:
        adam_deadline = time.time() + remaining * 0.20
        seq_adam, val_adam = _smooth_adam_optimize(
            start_seq=best_seq,
            deadline=adam_deadline,
            beta_start=5.0,
            beta_end=500.0,
            max_iter=5_000_000,
            lr0=0.03,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = seq_adam.copy()
            print(f"[Adam‑refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – aggressive mirror‑descent polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0 and best_seq is not None:
        seq_final, val_final = _refine_aggregated_subgradient(best_seq, deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[Agg‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will also call evaluate_sequence)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )

    # store for possible reuse by future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
