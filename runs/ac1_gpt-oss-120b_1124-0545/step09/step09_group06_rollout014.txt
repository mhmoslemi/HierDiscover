def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).  ``seq`` ≥ 0."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:

    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.

def _greedy_low_collision_mask(n: int, C: int, rng: np.random.Generator) -> tuple[np.ndarray, list[int]]:
    """
    Greedy construction of a 0/1 mask with at most C ordered pairs for any sum.
    Returns mask and list of active indices.
    """
    mask = np.zeros(n, dtype=bool)
    support = []
    # counts of ordered pairs for each possible sum (0 … 2n‑2)
    sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)

    order = rng.permutation(n)
    for cand in order:
        # check self‑pair
        if sum_counts[2 * cand] + 1 > C:
            continue

        if support:
            idxs = cand + np.array(support, dtype=int)
            if np.any(sum_counts[idxs] + 2 > C):
                continue

        # accept the candidate
        if support:
            sum_counts[cand + np.array(support, dtype=int)] += 2
        sum_counts[2 * cand] += 1
        mask[cand] = True
        support.append(cand)

    return mask, support


def _augment_mask(mask: np.ndarray, support: list[int],
                  sum_counts: np.ndarray, C: int,
                  rng: np.random.Generator,
                  deadline: float) -> tuple[np.ndarray, list[int], np.ndarray]:
    """
    Try to insert additional indices while respecting the C‑collision bound.
    Returns the (potentially) enlarged mask, its support list and updated sum_counts.
    """
    n = mask.shape[0]
    # a simple loop – abort when time is out or no progress after many trials
    trials_without_success = 0
    max_trials_without_success = 2000

    while time.time() < deadline:
        if trials_without_success > max_trials_without_success:
            break
        # pick a random index not currently used
        cand = rng.integers(0, n)
        if mask[cand]:
            continue

        # fast feasibility test
        ok = True
        if sum_counts[2 * cand] + 1 > C:
            ok = False
        else:
            if support:
                idxs = cand + np.array(support, dtype=int)
                if np.any(sum_counts[idxs] + 2 > C):
                    ok = False

        if ok:
            # perform the insertion
            if support:
                sum_counts[cand + np.array(support, dtype=int)] += 2
            sum_counts[2 * cand] += 1
            mask[cand] = True
            support.append(cand)
            trials_without_success = 0
        else:
            trials_without_success += 1

    return mask, support, sum_counts

                              C: int,
                              rng: np.random.Generator,
                              deadline: float) -> np.ndarray:
    """
    Repeated greedy attempts (with cheap augmentation) to find a large
    low‑collision mask (ordered collisions ≤ C).  Returns the index array of
    the best mask found.
    """
    best_mask = None
    best_size = -1
    attempts = 0
    theoretical_limit = int(math.sqrt(2.0 * C * n))

    while time.time() < deadline:
        attempts += 1
        mask, support = _greedy_low_collision_mask(n, C, rng)

        # augment greedily
        sum_counts = np.convolve(mask.astype(np.uint8), mask.astype(np.uint8))

    if best_mask is None:

    return np.where(best_mask)[0]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

def _adam_optimize(seed: np.ndarray, n: int, deadline: float,
                   max_iters: int = 40000):
    """
    Adam on a smoothed‑max surrogate of max(conv(a,a)).
    Returns best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)                # logits on simplex
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5               # temperature for soft‑max over convolution
    beta_mul = 1.007         # slowly increase → tighter approximation

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ---- smoothed‑max over convolution ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·(a*a)_k ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 1_000_000):
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray, n: int, deadline: float,
               t_mix: float = 0.3, max_lp_iters: int = 5) -> np.ndarray:
    """
    Cheap LP refinements: for the current max M, maximise Σ g subject to
    f * g ≤ M (with g ≥ 0).  Mix the result back into f.
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff = np.zeros(n, dtype=np.float64)
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        if g_opt.sum() == 0:
            break
        g_opt = g_opt / g_opt.sum()

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass between
    two positions and occasionally performs a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation (including the improved low‑collision mask)
# ----------------------------------------------------------------------
def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator,
                lowcoll_idx: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of starting vectors (all summed to 1).
    """
    seeds = []

    # classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # sparse random seeds – support sizes 30 … 200
    for m in range(30, 210, 20):
        for _ in range(5):
            seeds.append(_sparse_random_seed(n, m, rng))

    # deterministic Sidon‑like structures
    for m in (50, 70, 90, 110):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # low‑collision uniform support (if we built one)
    if lowcoll_idx is not None and lowcoll_idx.size:
        a = np.zeros(n, dtype=np.float64)
        a[lowcoll_idx] = 1.0 / lowcoll_idx.size
        a /= a.sum()
        seeds.append(a)
        print(f"[seed] added low‑collision uniform seed (size {lowcoll_idx.size})")
        # a few random Dirichlet perturbations of that support
        for _ in range(4):
            w = rng.dirichlet(np.ones(lowcoll_idx.size))
            a2 = np.zeros(n, dtype=np.float64)
            a2[lowcoll_idx] = w
            a2 /= a2.sum()
            seeds.append(a2)

    # reuse previous solution if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean‑up: ensure non‑negativity and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a ~1000 s time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # reuse a previous construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 4000 gives a good balance; can be changed if desired
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive low‑collision set search (C = 5, then maybe 6)
    # ------------------------------------------------------------------
    lowcoll_idx = None
    # allocate ~30 % of the budget for the low‑collision construction
    phase1_deadline = start_time + 0.30 * (deadline - start_time)

    for C in (5, 6):
        if time.time() >= phase1_deadline:
            break
        print(f"[Phase1] searching low‑collision mask with C={C}")
        idx = _search_low_collision_set(
            n=n,
            C=C,
            rng=rng,
            deadline=phase1_deadline,
        )
        if idx.size == 0:
            continue
        # keep the best (largest) set across the tried C values
        if lowcoll_idx is None or idx.size > lowcoll_idx.size:
            lowcoll_idx = idx
            best_C = C
        # early exit if we already hit the theoretical limit for C
        if idx.size >= int(math.sqrt(2.0 * C * n)):
            break

    if lowcoll_idx is not None:
        print(f"[Info] best low‑collision set size {lowcoll_idx.size} (C={best_C})")
    else:
        print("[Info] low‑collision search failed – proceeding without it")

    # ------------------------------------------------------------------
    # Phase 2 – build a rich pool of seeds
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng, lowcoll_idx)

    # ------------------------------------------------------------------
    # Quick evaluation of seeds – keep the best few for the expensive phases
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(12, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] initial best obj = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on top seeds
    # ------------------------------------------------------------------
    phase3_deadline = start_time + 0.55 * (deadline - start_time)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > phase3_deadline:
            break
        remaining = phase3_deadline - time.time()
        max_iters = int(min(50000, remaining / 0.0012))
        seq, val = _adam_optimize(seed_vec, n, phase3_deadline, max_iters=max_iters)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact sub‑gradient refinement (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        sub_deadline = deadline - 7.0   # reserve a few seconds for later phases
        remaining = sub_deadline - time.time()
        max_sub_iters = int(min(900000, remaining / 0.0014))
        seq, val = _subgradient_refine(global_best_seq, n, sub_deadline,
                                       max_iters=max_sub_iters)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – LP refinements (if cvxpy available)
    # ------------------------------------------------------------------
    if CVXPY_AVAILABLE and time.time() < deadline - 6.0:
        lp_seq = global_best_seq.copy()
        while time.time() + 5.0 < deadline:
            lp_seq = _lp_refine(lp_seq, n, deadline, t_mix=0.3, max_lp_iters=8)
            # brief sub‑gradient clean‑up after each LP pass
            sub_seq, sub_val = _subgradient_refine(lp_seq, n, deadline,
                                                   max_iters=25000)
            if sub_seq is not None and sub_val < global_best_val:
                global_best_seq = sub_seq.copy()
                global_best_val = sub_val
                print(f"[LP+Sub] new best {sub_val:.6f}")
                lp_seq = sub_seq.copy()
            else:
                break

    # ------------------------------------------------------------------
    # Phase 6 – Simulated annealing fine‑tuning (use all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and safety checks
    # ------------------------------------------------------------------
    final_seq = np.maximum(global_best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(final_seq)})")

    # expose for the next iteration (if any)
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
