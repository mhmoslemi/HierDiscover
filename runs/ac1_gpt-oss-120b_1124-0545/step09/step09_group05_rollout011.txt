"""
Improved non‑negative step‑function optimiser.

The optimiser works in three stages:

1.  Seed generation – a rich collection of analytic families (uniform,
    geometric, power‑law, Hann/Hamming/Blackman windows, Dirichlet draws,
    block‑wise patterns) and – if available – the previous best solution.
    All candidates are normalised to a unit sum.

2.  Global optimisation – a smoothed‑max Adam optimiser (log‑sum‑exp
    smoothing) followed by a few iterations of exponentiated‑gradient
    sub‑gradient descent.  This phase uses fast FFT‑based convolutions.

3.  Fine‑grained simulated annealing – mass is transferred between the
    two coordinates that contribute most/least to the currently maximal
    convolution entry.  The convolution array is updated **incrementally**
    (O(n) per move) so that thousands of cheap moves can be performed
    before the time budget expires.

The whole routine respects a user‑supplied time budget (default 1000 s)
and finally stores the best sequence found in the global name
``height_sequence_1``.  The returned list can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective used during the search (sum(seq)=1).

    Returns the exact quantity ``2 * n * max(conv(seq,seq))`` which is
    the value that ``evaluate_sequence`` would produce for a unit‑sum
    vector.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Duchi et al.)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw sorted in descending order."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_blocks(n: int, max_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """Simple block‑wise pattern – a few equally sized blocks spread out."""
    k = rng.integers(2, max_blocks + 1)
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos : pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _interpolate_seq(old: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalised sequence to a new length."""
    old_n = len(old)
    if old_n == new_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    interp = np.interp(x_new, x_old, old)
    s = interp.sum()
    return interp / s if s > 0 else _uniform(new_n)


# ----------------------------------------------------------------------
# Seed construction
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Return a diverse collection of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # geometric – grid
    for r in np.linspace(0.70, 0.99, 13):
        seeds.append(_geom_decrease(n, r))

    # power‑law – grid
    for p in np.arange(0.3, 3.01, 0.3):
        seeds.append(_power_decrease(n, p))

    # sorted Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # block‑wise patterns
    for max_blocks in (3, 5, 7, 10):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse draws
    for k in (5, 10, 20, 40):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous best (if compatible)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------


def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 60000,
) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a smoothed max (log‑sum‑exp) surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5                # smoothing parameter – will be increased
    beta_mul = 1.001
    beta_max = 300.0          # enough to approximate the true max

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                   # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam]   it={it:6d}  best={cur_val:.6f}")

        # anneal smoothing
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1 : n - 1 + n]

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.9 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d}  best={cur_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float) -> None:
    """
    Apply a change ``a[idx] ← a[idx] + delta`` and update the full
    convolution array ``conv = a * a`` *incrementally*.
    The routine assumes ``conv`` already contains the convolution of the
    *current* vector ``a`` (i.e. before the change).
    """
    n = a.shape[0]

    # contribution of the cross terms
    conv[idx : idx + n] += 2.0 * delta * a

    # self‑convolution term at position 2*idx
    conv[2 * idx] += delta * delta

    # finally update the coefficient
    a[idx] += delta


def _local_sa(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    init_temp_factor: float = 0.06,
    decay: float = 0.9999,
    min_eps: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Targeted simulated‑annealing that moves mass between the two
    coordinates that contribute most / least to the current maximal
    convolution entry.  The convolution array is kept up‑to‑date by
    ``_apply_delta`` which runs in O(n) time.
    """
    a = start_seq.copy()
    conv = _conv_fft(a, a)                     # full convolution, length 2n‑1
    cur_max = conv.max()
    best_seq = a.copy()
    best_val = 2.0 * n * cur_max

    T = max(init_temp_factor * cur_max, 1e-12)
    it = 0

    while time.time() < deadline:
        it += 1
        T = max(T * decay, 1e-12)

        max_idx = int(np.argmax(conv))

        # build sub‑gradient for the active index (only the overlapping part)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_range = np.arange(i_start, i_end + 1, dtype=int)
        grad = np.zeros(n, dtype=np.float64)
        grad[idx_range] = 2.0 * a[max_idx - idx_range]

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # ensure we actually move some mass
        if a[i_high] <= min_eps:
            # nothing to take from – pick a random non‑zero entry instead
            nonzero = np.where(a > min_eps)[0]
            if nonzero.size == 0:
                break
            i_high = int(np.random.choice(nonzero))

        eps = min(a[i_high] * 0.2, 0.001)   # small but not too tiny
        if eps < min_eps:
            break

        # apply the move
        _apply_delta(a, conv, i_high, -eps)
        _apply_delta(a, conv, i_low, +eps)

        new_max = conv.max()
        delta = new_max - cur_max

        if delta < 0.0 or np.random.rand() < np.exp(-delta / T):
            cur_max = new_max
            # keep global best if improved
            cur_val = 2.0 * n * cur_max
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SA]    it={it:7d}  best={best_val:.6f}")
        else:
            # revert the move
            _apply_delta(a, conv, i_low, -eps)
            _apply_delta(a, conv, i_high, +eps)

        # occasional re‑normalisation to fight rounding errors
        if it % 2000 == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-12:
                a /= s
                conv = _conv_fft(a, a)   # cheap re‑sync

    return best_seq, best_val

# Main entry point

def propose_candidate(
    seed: int = 123,
    budget_s: int = 1000,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  The routine respects the supplied time budget
    and stores the best sequence in the global name ``height_sequence_1``.
    """
    # --------------------------------------------------------------
    # reproducibility
    # --------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Length of the search vector – a larger n often yields a smaller bound
    # --------------------------------------------------------------
    n = 8000

    # --------------------------------------------------------------
    # Load previous solution if available (allows warm‑start across runs)
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Stage 1 – generate a diverse set of seeds and keep the few best
    # --------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – Adam + sub‑gradient refinement on the top seeds
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=50000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # sub‑gradient (exponentiated‑gradient) refinement
        seq_sub, val_sub = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=120000
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – long targeted simulated annealing
    # --------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Long simulated‑annealing polishing ---")
        seq_sa, val_sa = _local_sa(
            global_best_seq, n, deadline, init_temp_factor=0.06, decay=0.9999
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # --------------------------------------------------------------
    # Optional re‑ordering test (descending vs ascending)
    # --------------------------------------------------------------
    if time.time() < deadline:
        for ordering in ("desc", "asc"):
            cand = (
                np.sort(global_best_seq)[::-1]
                if ordering == "desc"
                else np.sort(global_best_seq)
            )
            cand_val = _objective(cand)
            if cand_val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = cand_val
                print(
                    f"[Reorder] new global best with {ordering} order = {cand_val:.6f}"
                )

    # --------------------------------------------------------------
    # Final sanity check
    # --------------------------------------------------------------
    try:
        # evaluate_sequence is provided by the environment
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
