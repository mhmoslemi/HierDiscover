def _smoothed_gradient(a, alpha):
    conv = _conv_fft(a, a)
    maxc = conv.max()
    # w = exp(alpha*(conv - maxc))
    exp_term = np.exp(alpha * (conv - maxc))
    w = exp_term / exp_term.sum()
    # Compute gradient
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w) # full length 3n-2
    grad = 2.0 * full[len(a)-1:len(a)-1+len(a)]
    return grad, conv.max()

"""
Propose a non‑negative step function (list of floats) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm combines several ideas that differ from the previous submission:

1. **Smoothed exponential‑gradient descent**  
   Instead of the raw sub‑gradient of the max‑convolution we optimise a
   smooth approximation  

        f_α(a) = (1/α)·log Σ_k exp(α·conv_k(a)) ,

   whose gradient is a weighted sum of the convolution gradients.
   For a given smoothing parameter α we perform an exponentiated‑gradient
   (mirror‑descent on the simplex) update

        log a ← log a – η·∇f_α(a) ,

   followed by normalisation.  The smoothing parameter is increased
   geometrically (α = 0.5, 2, 5, 20, 100, 500, 2000) so that the
   optimisation first sees a soft landscape and then focusses on the
   true max.  Small random perturbations are injected after each α‑stage
   to escape shallow local minima.

2. **Multiple restarts** – a handful of diverse initialisations
   (uniform, random soft‑max, Dirichlet, a sparse spike pattern and, if
   available, the previous best sequence) are each refined by the
   smoothed descent; the overall best is kept.

3. **Optional LP refinement** – if `cvxpy` is installed we perform a
   single linearised‑constraint refinement (as in the reference code).

4. **Final simulated‑annealing fine‑tuning** – a cheap Metropolis walk
   uses the remaining budget to polish the solution.

All steps respect the time budget (default 1000 s).  The returned
sequence is normalised to sum = 1, non‑negative, and stored in the global
``height_sequence_1`` for reuse.

The implementation uses only NumPy, SciPy (if available) and cvxpy
(where present).  No file or network I/O is performed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (real FFT, full mode)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution (full) of two real vectors using FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (the quantity we try to minimise)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (probability vectors on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _dirichlet_seed(n: int, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.ones(n))


def _sparse_spike_seed(n: int, rng: np.random.Generator, sparsity: float = 0.01) -> np.ndarray:
    """Sparse seed: a few random positions carry most of the mass."""
    m = max(1, int(n * sparsity))
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.random(m) + 1e-3
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smoothed gradient (soft‑max) utilities
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Return soft‑max weights Σ w_k = 1 for a given α."""
    # stable soft‑max
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    w = exp_term / exp_term.sum()
    return w


def _smoothed_gradient(a: np.ndarray, alpha: float) -> tuple[np.ndarray, float]:
    """
    Compute gradient of f_α(a) = (1/α)·log Σ_k exp(α·conv_k(a))
    and return also the current max convolution value.
    """
    conv = _conv_fft(a, a)
    max_conv = conv.max()
    w = _softmax_weights(conv, alpha)          # shape (2n-1,)
    rev_a = a[::-1]
    full = _conv_fft(rev_a, w)                 # length = 3n-2
    grad = 2.0 * full[a.size - 1:a.size - 1 + a.size]
    return grad, max_conv


# ----------------------------------------------------------------------
# Smoothed (exponentiated) gradient descent
# ----------------------------------------------------------------------
def _smooth_descent(start_seq: np.ndarray,
                    deadline: float,
                    alphas: list[float],
                    iters_per_alpha: int,
                    eta0: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a multi‑stage smoothed exponentiated‑gradient descent.
    Returns the best sequence encountered and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it_total = 0
    for alpha in alphas:
        eta = eta0
        for it in range(iters_per_alpha):
            if time.time() > deadline:
                return best_seq, best_val

            grad, _ = _smoothed_gradient(a, alpha)

            # exponentiated‑gradient update (mirror descent on the simplex)
            log_a = np.log(a + 1e-30) - eta * grad
            log_a -= np.max(log_a)                      # avoid overflow
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum <= 0:
                a = _uniform(a.size)
            else:
                a /= a_sum

            # simple learning‑rate schedule
            eta *= 0.9995

            it_total += 1
            if it_total % 20 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Smooth] α={alpha:.0f} it={it_total} best={val:.6f}")

        # after each α‑stage inject a tiny random perturbation
        eps = 0.01
        noise = _dirichlet_seed(a.size, rng)
        a = (1.0 - eps) * a + eps * noise
        a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linearised LP refinement (convex‑concave procedure, optional)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    CVXPY_AVAILABLE = False


def _lp_linearised_refine(a: np.ndarray,
                         deadline: float,
                         max_refine: int = 6,
                         eps_active: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements where the quadratic constraints are
    linearised around the current point `a`. Returns the best vector
    discovered and its objective.
    """
    if not CVXPY_AVAILABLE:
        return a, _objective(a)

    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    x = cp.Variable(n)
    M = cp.Variable()
    constraints = [x >= 0, cp.sum(x) == 1]

    for it in range(max_refine):
        if time.time() > deadline - 0.3:
            break

        conv = _conv_fft(a, a)
        M_cur = conv.max()
        active = np.where(conv >= M_cur - eps_active)[0]
        if active.size == 0:
            break

        # Build matrix A where A[row, i] = a[k-i] for row offset k = active[row]
        A = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            A[r, i_range] = a[k - i_range]

        lin_cons = [2 * (A @ x) <= M + conv[active]]
        prob = cp.Problem(cp.Minimize(M), constraints + lin_cons)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True,
                       max_iters=2000, abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if x.value is None or M.value is None:
            break

        a_new = np.maximum(np.asarray(x.value).flatten(), 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s

        val_new = _objective(a_new)
        print(f"[LP] iter={it+1} M={M_cur:.6f} new_obj={val_new:.6f}")

        if val_new < best_val - 1e-9:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
        else:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuner (as in the reference)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis‑Hastings walk that swaps a small amount of mass
    between two random positions and occasionally makes a global
    Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns the best sequence found (list of floats).
    The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # small safety margin

    rng = np.random.default_rng(seed + 12345)

    # --------------------------------------------------------------
    # Problem size – a slightly larger n gives more freedom while still
    # staying comfortably fast (FFT of length ~16384).
    # --------------------------------------------------------------
    n = 5000
    print(f"\n=== Optimising step function of length {n} ===")

    # --------------------------------------------------------------
    # Initialise a collection of diverse seeds
    # --------------------------------------------------------------
    seeds = []

    # previous run (if compatible)
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            arr = np.asarray(previous, dtype=np.float64)
            if arr.shape == (n,):
                arr = np.maximum(arr, 0.0)
                arr = arr / arr.sum()
                seeds.append(arr)
                print("[Init] Loaded previous sequence")
        except Exception:
            pass

    # cheap deterministic seeds
    seeds.append(_uniform(n))
    seeds.append(_random_softmax(n, rng))
    seeds.append(_dirichlet_seed(n, rng))
    seeds.append(_sparse_spike_seed(n, rng, sparsity=0.01))

    # pick the best of the cheap seeds as a fallback
    best_initial = None
    best_initial_val = float("inf")
    for s in seeds:
        val = _objective(s)
        if val < best_initial_val:
            best_initial_val = val
            best_initial = s.copy()
    print(f"[Init] best initial objective = {best_initial_val:.6f}")

    # --------------------------------------------------------------
    # Smoothed exponentiated‑gradient descent (multiple restarts)
    # --------------------------------------------------------------
    alphas = [0.5, 2.0, 5.0, 20.0, 100.0, 500.0, 2000.0]
    iters_per_alpha = 200          # modest number – many restarts fit in budget
    eta0 = 0.5

    # reserve part of the budget for later phases
    smoothing_deadline = deadline - 30.0
    if smoothing_deadline < time.time():
        smoothing_deadline = time.time() + 0.1

    overall_best_seq = best_initial.copy()
    overall_best_val = best_initial_val

    # run a few restarts (including the previous best if present)
    for restart_id, init_seq in enumerate(seeds[:4]):   # at most 4 restarts
        if time.time() > smoothing_deadline:
            break
        print(f"[Restart {restart_id+1}] Smoothed descent start")
        seq, val = _smooth_descent(init_seq.copy(),
                                   smoothing_deadline,
                                   alphas,
                                   iters_per_alpha,
                                   eta0,
                                   rng)
        if val < overall_best_val:
            overall_best_val = val
            overall_best_seq = seq.copy()
            print(f"[Restart {restart_id+1}] New best objective = {val:.6f}")

    # --------------------------------------------------------------
    # Optional LP refinement (convex‑concave step)
    # --------------------------------------------------------------
    lp_deadline = deadline - 10.0
    if CVXPY_AVAILABLE and lp_deadline > time.time():
        print("[LP] Linearised refinement start")
        seq_lp, val_lp = _lp_linearised_refine(overall_best_seq, lp_deadline)
        if val_lp < overall_best_val:
            overall_best_val = val_lp
            overall_best_seq = seq_lp.copy()
            print(f"[LP] Improvement, new objective = {val_lp:.6f}")
        else:
            print("[LP] No improvement")
    else:
        print("[LP] Skipped (cvxpy unavailable or not enough time)")

    # --------------------------------------------------------------
    # Final simulated‑annealing fine‑tuning with whatever time remains
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Anneal] Final fine‑tuning")
        seq_final, val_final = _anneal_optimize(overall_best_seq, deadline)
        if val_final < overall_best_val:
            overall_best_val = val_final
            overall_best_seq = seq_final.copy()
            print(f"[Anneal] New best objective = {val_final:.6f}")
        else:
            print("[Anneal] No improvement")
    else:
        print("[Anneal] No time left for fine‑tuning")

    # --------------------------------------------------------------
    # Normalise and store the final answer
    # --------------------------------------------------------------
    final_seq = np.maximum(overall_best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq /= total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
