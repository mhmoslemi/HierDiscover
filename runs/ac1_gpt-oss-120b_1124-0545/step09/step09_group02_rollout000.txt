"""
Hybrid optimiser – smoothed‐gradient (Adam) + final annealing

Goal
----
Minimise 2·n·max(conv(a,a)) / (sum(a)**2) for a non‑negative vector a.
Because the objective is homogeneous we work on the probability simplex
∑a = 1 (the scaling does not affect the score).

Method
------
1. Build a modest but diverse pool of analytic seeds (uniform, geometric,
   power‑law, beta‑pdf, random Dirichlet, soft‑max).  Keep the few best.
2. Refine every kept seed with a *smoothed* version of the max‑convolution.
   We replace the non‑smooth max by a temperature‑controlled soft‑max
   (log‑sum‑exp).  The gradient of the smoothed objective is a simple
   convolution, so each iteration costs two FFTs (O(n log n)).
   An Adam optimiser on the simplex (projected after each step) runs for
   as many iterations as time permits.
3. The best vector found so far is further polished by a short simulated‑
   annealing walk (mass‑transfer, swaps, jitter).  This helps escape the
   remaining tiny local minima.
4. The final sequence is stored in the global ``height_sequence_1`` variable.

The whole search respects the 1000 s time budget and prints progress
messages.  All helper functions are top‑level, with no closures or lambdas.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑convolution via FFT (real input, full mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (the score that the grader will compute)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise (should already sum to 1)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ----------------------------------------------------------------------
# Analytic seed families (all returned as probability vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1), discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


def _make_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a modestly sized list of candidate probability vectors."""
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    for r in np.linspace(0.2, 0.99, 9):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 4.0, 9):
        seeds.append(_power(n, p))
    for alpha in np.linspace(0.3, 2.0, 9):
        seeds.append(_beta_window(n, alpha))

    # stochastic families
    for _ in range(8):
        seeds.append(_random_softmax(n, rng))
    for a in [0.05, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, a, rng))

    # reuse previous construction if given
    if previous is not None and previous.shape == (n,):
        prev = np.clip(previous, 0.0, None)
        ss = prev.sum()
        if ss > 0:
            seeds.append(prev / ss)

    # final clean‑up – ensure proper normalisation
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Smoothed‑gradient (Adam) optimiser on the simplex
# ----------------------------------------------------------------------


def _adam_refine(start_vec: np.ndarray,
                 n: int,
                 deadline: float,
                 init_lr: float = 0.5,
                 beta1: float = 0.9,
                 beta2: float = 0.999,
                 eps: float = 1e-8,
                 eval_every: int = 100) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smoothed objective:
        f_T(a) = 2·n·T·log( Σ_i exp( conv_i(a)/T ) )
    The temperature T is decreased geometrically.  After each step the
    iterate is projected onto the probability simplex.
    Returns the best vector seen (according to the *exact* objective).
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule – start relatively warm
    conv0 = _conv_fft(a, a)
    T = max(1e-4, conv0.max() * 0.2)   # initial temperature ≈ 20 % of the current peak
    decay = 0.9999                     # decay per iteration

    it = 0
    while time.time() < deadline:
        it += 1

        # convolution of current iterate
        conv = _conv_fft(a, a)

        # soft‑max weights (numerically stable)
        maxc = conv.max()
        # keep temperature positive
        T = max(1e-7, T * decay)
        shifted = (conv - maxc) / T
        w = np.exp(shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient of the smoothed max (ignoring the constant factor 2·n)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]   # shape (n,)

        # Adam update (step size absorbs the constant 2·n)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = init_lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a = _project_simplex(a_new)

        # occasional exact evaluation
        if it % eval_every == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] iter={it:7d} best={val:.6f}")

        # safety break if we are out of time
        if time.time() >= deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine tuning (same as in the reference implementation)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Simple Metropolis–Hastings walk on the simplex."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – fraction of current objective
    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.68:                               # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.18 * cur[i]   # up to 18 % of source mass
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif rnd < 0.86:                             # swap
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                                        # Gaussian jitter
            noise = rng.normal(scale=0.0013, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        if it % 2500 == 0:
            # occasional jump from the current best
            cand = best + rng.normal(scale=0.0035, size=n)
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a list of floats) found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------------
    # Vector length – 4000 is a good trade‑off between speed and quality
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous construction (if any) – it will be stored globally
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 0) Initialise seed pool
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous)

    # ------------------------------------------------------------------
    # 1) Evaluate all seeds and keep the top few
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    best_seq = top_seeds[0].copy()
    best_val = _objective(best_seq)
    print(f"[init] best seed value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed‑gradient (Adam) refinement of each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 5.0:
            break
        print(f"[refine] start Adam on seed {idx}")

        seq, val = _adam_refine(seed_vec, n, deadline,
                                init_lr=0.4,   # a bit more conservative
                                eval_every=200)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] new best after Adam = {val:.6f}")

    # ------------------------------------------------------------------
    # 3) Final simulated‑annealing fine‑tuning (uses remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] start annealing")
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[final] best after anneal = {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    # store for future runs
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
