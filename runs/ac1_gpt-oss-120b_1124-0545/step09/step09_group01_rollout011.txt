def _polyak_subgradient(start_seq, n, deadline, max_iters=1_000_000):
    a = start_seq.copy()
    best = a.copy()
    best_val = _objective(a)
    best_iter = 0

    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max
        # update best
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            best_iter = it
        # active indices
        mask = np.isclose(conv, cur_max, atol=1e-15)
        active_idx = np.where(mask)[0]
        # subgradient
        g = np.zeros_like(a)
        for k in active_idx:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end+1)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]
        # compute step size Polyak
        grad_norm_sq = np.dot(g, g)
        if grad_norm_sq < 1e-12:
            # add small noise
            a = _proj_simplex(a + 1e-12 * (np.random.rand(n)-0.5))
            continue
        # step size
        eta = (cur_val - best_val) / grad_norm_sq
        # ensure positive step
        if eta <= 0:
            eta = 1e-4  # fallback
        # optionally limit step size
        eta = min(eta, 1.0)  # avoid too large jumps
        a = a - eta * g
        a = _proj_simplex(a)
    return best, best_val

"""
Propose a non‑negative step‑function with a low ``evaluate_sequence`` value.

Algorithm
---------
1. **Seed generation** – generate a huge, diverse pool of ℓ₁‑normalised
   candidates (analytic families, many Dirichlet draws with a wide range of
   concentration parameters, random exponentials and the previous best).

2. **Polyak sub‑gradient descent** – for the top seeds run a dedicated
   sub‑gradient method with a *Polyak step size* (ηₜ = (f(xₜ)‑f_best)/‖gₜ‖²)
   on the exact objective 2·n·max(conv(seq,seq)).  The update is followed by
   Euclidean projection onto the simplex.  This converges far faster than a
   fixed 1/√t schedule.

3. **Adam on a smooth max** – polish the best result with the Adam optimiser
   that minimises a log‑sum‑exp surrogate of the max‑convolution.
   Temperature is raised slowly up to a large value (β≈250) to approach the
   true max.

4. **Simulated annealing** – final stochastic polishing that swaps the mass
   of a few coordinates via Dirichlet re‑allocations.

All work respects a 1000 s time budget; the best sequence found is stored in
the global variable ``height_sequence_1`` and returned as a plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate (sum = 1) used throughout the optimisation.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Σ=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the simplex (concentration ``conc``)."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 1_000_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Parameters are the logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits on the simplex

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule for the soft‑max surrogate
    beta_temp = 5.0
    beta_mul = 1.004
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # raise temperature slowly so we approach the true max
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                # exact convolution
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical safeguard
            continue
        w /= w_sum                               # softmax over the convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _polyak_subgradient(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent with Polyak step size.
    The objective is 2·n·max(conv(seq,seq)).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_max = np.max(conv)
        cur_val = 2.0 * n * cur_max

        # Track the best solution seen so far
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[Polyak] it={it:7d} best={best_val:.6f}")

        # active set (indices where the max is attained)
        mask = np.isclose(conv, cur_max, atol=1e-15, rtol=0.0)
        active = np.where(mask)[0]

        # sub‑gradient of the max‑convolution
        g = np.zeros_like(a)
        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        grad_norm_sq = np.dot(g, g)
        if grad_norm_sq < 1e-14:
            # degenerate gradient – add a tiny random perturbation
            a = _proj_simplex(a + 1e-12 * (np.random.rand(n) - 0.5))
            continue

        # Polyak step size (uses the best value seen so far as a lower bound)
        eta = (cur_val - best_val) / grad_norm_sq
        # Ensure a positive step; fallback to a small fixed step if needed
        if eta <= 0 or np.isnan(eta) or np.isinf(eta):
            eta = 0.1 / np.sqrt(it)

        # limit the step size to avoid runaway updates
        eta = min(eta, 1.0)

        # gradient step + projection onto the simplex
        a = a - eta * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.025,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that randomly reshuffles the mass of a few
    coordinates (Dirichlet re‑allocation).  Works directly on the exact
    objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 20000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a heterogeneous pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    # Geometric families
    for r in np.linspace(0.2, 0.98, 12):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 18):
        seeds.append(_power(n, p))

    # Beta family – broaden the search ranges
    for _ in range(40):
        a = rng.uniform(0.05, 3.0)
        b = rng.uniform(0.05, 5.0)
        seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.12, 0.18, 0.24, 0.30, 0.36):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.05, 0.15, 0.25, 0.35, 0.45, 0.55, 0.75, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – a wide range of concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Extra random draws with many concentration levels
    for conc in np.logspace(-2, 1, 10):
        for _ in range(20):
            seeds.append(_random_dirichlet(n, conc, rng))

    for _ in range(20):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # tiny safety margin

    # retrieve possible previous solution
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list):
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose the length – a few thousand offers a good trade‑off.
    # ------------------------------------------------------------------
    n = 2000

    # ------------------------------------------------------------------
    # Stage 0 – massively diverse seed pool.
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # cheap surrogate evaluation (exact because sum=1)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = 8
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – Polyak sub‑gradient descent on the exact objective.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Polyak sub‑gradient run {idx+1}/{len(best_seeds)} ---")
        seq_sub, val_sub = _polyak_subgradient(seed_vec, n, deadline,
                                               max_iters=500_000)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[Polyak] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimiser on the smooth max surrogate.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate([global_best_seq] + best_seeds[:3]):
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation run {idx+1} ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline,
                                            max_iters=2_000_000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (Simulated Annealing).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=1_500_000,
            init_temp_factor=0.03
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – feed the candidate to the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
