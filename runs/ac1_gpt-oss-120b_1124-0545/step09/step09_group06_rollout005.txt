"""
Propose a step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

by constructing a large *low‑collision* support set S ⊂ {0,…,n‑1}
such that every ordered pair‑sum i+j (i,j∈S) occurs at most
`max_ordered` times.  For a uniform distribution on S the objective
is exactly

    2·n·max_ordered / |S|² .

Choosing `max_ordered = 4` and a set of size ≈√(8·n) gives a
theoretical bound ≈1.0.  This implementation spends the majority of
the time budget searching greedily (with many random restarts) for a
set S with as many elements as possible under the collision bound.
The best set found is returned as a uniform probability vector.
If the search fails to obtain a set large enough to beat the target
(≈1.503) a simple fallback (uniform over all indices) is used.

The algorithm runs completely within the 1000 s time limit, uses only
NumPy and standard library, and stores the best sequence in the global
variable `height_sequence_1` for possible reuse by a subsequent run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (exact, real‑valued via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).  seq ≥ 0."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Greedy low‑collision set construction
# ----------------------------------------------------------------------
def _search_low_collision_set(
    n: int,
    max_ordered: int,
    target_size: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """
    Greedy random construction of a set S ⊂ {0,…,n‑1}
    such that each ordered sum i+j (i,j∈S) occurs at most `max_ordered`
    times.  Returns the index array of the best set found before `deadline`.
    """
    best_mask = None
    best_size = 0
    attempts = 0

    while time.time() < deadline:
        attempts += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)
        # keep a python list of chosen indices for quick access
        chosen = []

        order = rng.permutation(n)
        for cand in order:
            # self‑pair (i,i) adds exactly 1 ordered collision
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue

            if chosen:
                cand_arr = np.full(len(chosen), cand, dtype=np.int32)
                sums = cand_arr + np.array(chosen, dtype=np.int32)
                # each cross‑pair contributes 2 ordered collisions
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue
                # safe to add – update all cross‑pair counts
                sum_counts[sums] += 2

            # update self‑pair count and record the element
            sum_counts[2 * cand] += 1
            mask[cand] = True
            chosen.append(int(cand))

            # stop early if we reached the target size
            if len(chosen) >= target_size:
                break

        cur_size = len(chosen)
        if cur_size > best_size:
            best_size = cur_size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:5d} size {cur_size}")

        # if we already hit the desired size we can stop early
        if best_size >= target_size:
            break

    if best_mask is None:
        return np.array([], dtype=int)

    return np.where(best_mask)[0]

    # --------------------------------------------------------------
    n = 4000

    # theoretical optimum size ≈ sqrt(8·n); we aim for ≳85 % of it.

    print(f"\n=== Finished ===\nBest objective ≈ {obj:.6f} (length {len(seq)})")

    # expose for possible reuse in a later run
    globals()["height_sequence_1"] = seq.copy()
    return seq.tolist()
