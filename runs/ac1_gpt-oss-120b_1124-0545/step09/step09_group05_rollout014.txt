"""
Non‑negative step‑function optimiser – version 3.

Key ideas
---------

*   **Two‑stage resolution** – first optimise a relatively small vector
    (n≈1024) where many iterations are cheap, then up‑sample the best
    candidate to the target length (≈4000) and polish it again.
*   **Richer seed families** – beyond the previous uniform / triangular /
    Dirichlet / block families we also add raised‑cosine, alternating and
    fine‑grained geometric / power‑law families.
*   **Longer gradient phases** – we allocate more Adam / sub‑gradient / Euclidean‑projection
    steps (still respecting the global time budget).
*   **Uniform background search** – after all optimisation stages we
    systematically add a tiny constant to every entry (grid‐search over
    epsilons).  This raises the total mass without increasing the peak
    convolution proportionally and often yields a lower final
    “evaluate_sequence” value.
*   **Final re‑ordering** – we try both descending and ascending order of
    the final vector; sometimes this reduces the maximum of the
    convolution.

The function ``propose_candidate`` is the required entry point.  It
returns a list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _raised_cosine(n: int) -> np.ndarray:
    """a_i = (1 - cos(2π i/(n‑1))) / 2, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) * 0.5
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _alternating(n: int, high: float = 0.9, low: float = 0.1) -> np.ndarray:
    """Two‑value alternating pattern, then normalised."""
    a = np.empty(n, dtype=np.float64)
    a[::2] = high
    a[1::2] = low
    s = a.sum()
    return a / s


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric a_i = r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]                # descending
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_blocks(n: int, max_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """
    Simple block pattern: k blocks of equal length spread evenly.
    """
    k = rng.integers(2, max_blocks + 1)
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos:pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction (large but cheap)
# ----------------------------------------------------------------------
def _generate_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))
    seeds.append(_raised_cosine(n))
    seeds.append(_alternating(n))

    # geometric – dense grid up to very close to 1
    for r in np.linspace(0.70, 0.9999, 15):
        seeds.append(_geom_decrease(n, r))

    # power‑law – dense grid
    for p in np.arange(0.2, 4.01, 0.2):
        seeds.append(_power_decrease(n, p))

    # sorted Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0):
        for _ in range(3):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # block‑wise patterns
    for max_blocks in (3, 5, 7, 10, 15):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # random Dirichlet draws
    for conc in (0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (5, 10, 20, 40, 80):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if present and same length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 100000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp (smoothed max) surrogate.
    The vector stays on the simplex via a log‑softmax parametrisation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5
    beta_mul = 1.001
    beta_max = 800.0          # larger smoothing => tighter approximation

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                    # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        beta = min(beta * beta_mul, beta_max)

        # --- smoothed max via log‑sum‑exp ---
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.9
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on
    the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _euc_proj_subgradient(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the true max‑convolution with Euclidean projection
    onto the probability simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[EucProj] it={it:7d} best={cur_val:.6f}")

        # gradient of the active convolution entry (same as in sub‑grad)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)               # a bit more aggressive
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simulated annealing that reshuffles mass among a few coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # pick a small subset (3‑9 indices) and redistribute their total mass
        k = rng.integers(3, 10)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Helper for up‑sampling
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to target_n, then re‑normalise."""
    src_n = len(seq)
    if src_n == target_n:
        return seq.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, target_n)
    up = np.interp(x_tgt, x_src, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up = _uniform(target_n)
    else:
        up = up / s
    return up


# ----------------------------------------------------------------------
# Uniform‑background search
# ----------------------------------------------------------------------
def _add_uniform_background(seq: np.ndarray,
                            deadline: float,
                            max_evals: int = 30) -> np.ndarray:
    """
    Grid‑search over a tiny constant added to every entry.
    The objective is evaluated with the *real* evaluate_sequence
    function (which is available in the global namespace).
    """
    best_seq = seq.copy()
    best_val = evaluate_sequence(best_seq.tolist())

    # scale factors relative to the mean mass
    mean = best_seq.mean()
    factors = np.logspace(-8, -1, num=max_evals)

    for f in factors:
        if time.time() > deadline:
            break
        eps = f * mean
        cand = best_seq + eps
        try:
            val = evaluate_sequence(cand.tolist())
        except Exception:
            # safety – should never happen
            continue
        if val < best_val:
            best_val = val
            best_seq = cand
            print(f"[Uniform‑bg] eps={eps:.2e} -> val={val:.6f}")

    return best_seq

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` value.  Returns a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 0 – small “pilot” optimisation (fast, many seeds)
    # ------------------------------------------------------------------
    base_n = 1024
    print(f"\n=== Stage 0 – pilot optimisation (n={base_n}) ===")
    base_seeds = _generate_seeds(base_n, rng, previous)
    print(f"Generated {len(base_seeds)} pilot seeds.")

    evaluated = [(vec, _objective(vec)) for vec in base_seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    pilot_best = evaluated[0][0].copy()
    pilot_best_val = evaluated[0][1]
    print(f"Best pilot seed objective = {pilot_best_val:.6f}")

    # optimise the top pilots
    for idx, seed_vec in enumerate([v for v, _ in evaluated[:top_k]]):
        if time.time() > deadline:
            break
        print(f"\n--- Pilot optimisation run {idx + 1}/{top_k} ---")
        seq_adam, val_adam = _adam_optimize(seed_vec, base_n, deadline,
                                            max_iters=50000)
        if seq_adam is not None and val_adam < pilot_best_val:
            pilot_best, pilot_best_val = seq_adam, val_adam
            print(f"[pilot‑Adam] new best = {val_adam:.6f}")

        seq_sub, val_sub = _subgradient_refine(pilot_best, base_n, deadline,
                                               max_iters=120000)
        if val_sub < pilot_best_val:
            pilot_best, pilot_best_val = seq_sub, val_sub
            print(f"[pilot‑SubGrad] new best = {val_sub:.6f}")

        seq_euc, val_euc = _euc_proj_subgradient(pilot_best, base_n, deadline,
                                                 max_iters=80000)
        if val_euc < pilot_best_val:
            pilot_best, pilot_best_val = seq_euc, val_euc
            print(f"[pilot‑EucProj] new best = {val_euc:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – upscale to target length and coarse refinement
    # ------------------------------------------------------------------
    target_n = 4000
    print(f"\n=== Stage 1 – up‑sample to n={target_n} ===")
    cur_seq = _upsample(pilot_best, target_n)
    cur_val = _objective(cur_seq)
    print(f"Upsampled surrogate objective = {cur_val:.6f}")

    # a few more optimisation passes at full size
    seq_adam, val_adam = _adam_optimize(cur_seq, target_n, deadline,
                                        max_iters=60000)
    if seq_adam is not None and val_adam < cur_val:
        cur_seq, cur_val = seq_adam, val_adam
        print(f"[full‑Adam] new best = {val_adam:.6f}")

    seq_sub, val_sub = _subgradient_refine(cur_seq, target_n, deadline,
                                           max_iters=200000)
    if val_sub < cur_val:
        cur_seq, cur_val = seq_sub, val_sub
        print(f"[full‑SubGrad] new best = {val_sub:.6f}")

    seq_euc, val_euc = _euc_proj_subgradient(cur_seq, target_n, deadline,
                                             max_iters=150000)
    if val_euc < cur_val:
        cur_seq, cur_val = seq_euc, val_euc
        print(f"[full‑EucProj] new best = {val_euc:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(cur_seq, target_n, deadline,
                                              max_iters=300000,
                                              init_temp_factor=0.07)
        if val_sa < cur_val:
            cur_seq, cur_val = seq_sa, val_sa
            print(f"[full‑SA] new best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – uniform background addition
    # ------------------------------------------------------------------
    if time.time() < deadline:
        cur_seq = _add_uniform_background(cur_seq, deadline, max_evals=30)
        # recompute the surrogate after the background addition
        cur_val = _objective(cur_seq)
        print(f"[post‑bg] surrogate objective = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional simple re‑ordering (desc / asc)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for ordering in ("desc", "asc"):
            cand = np.sort(cur_seq)[::-1] if ordering == "desc" else np.sort(cur_seq)
            cand_val = _objective(cand)
            if cand_val < cur_val:
                cur_seq, cur_val = cand, cand_val
                print(f"[reorder] new best with {ordering} order = {cand_val:.6f}")

    # ------------------------------------------------------------------
    # Final validation with the true evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(cur_seq.tolist())
        print(f"\n=== Finished ===\nFinal evaluate_sequence score = {final_score:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse
    globals()["height_sequence_1"] = cur_seq.tolist()
    return cur_seq.tolist()
