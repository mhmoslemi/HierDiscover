def _smooth_adam(start, n, deadline, max_iters=50000,
                 alpha_start=0.5, alpha_end=12.0, alpha_ramp=20000):
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        # compute convolution a*a
        conv = _conv_fft(a, a)
        maxc = conv.max()
        # schedule alpha
        if it <= alpha_ramp:
            alpha = alpha_start + (alpha_end-alpha_start) * it / alpha_ramp
        else:
            alpha = alpha_end
        # weight vector (softmax of conv)
        w = np.exp(alpha * (conv - maxc))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.shape[0]
        else:
            w /= w_sum

        a_rev = a[::-1]
        full = _conv_fft(w, a_rev)
        grad = 2.0 * full[n-1:n-1+n]

        # Adam update
        m = beta1*m + (1-beta1)*grad
        v = beta2*v + (1-beta2)*(grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        lr = 0.1 / math.sqrt(it)

def _anneal(start, n, deadline, max_iters=200000):
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val
    T0 = max(1e-7, cur_val*0.07)
    decay = 0.9994
    it=0
    while it < max_iters and time.time()<deadline:
        it+=1
        T = T0 * (decay**it)
        # random move: pick two indices and move fraction
        i, j = rng.integers(0,n, size=2)

            delta = rng.random()*0.15*cur[i]

            cand[i] -= delta
            cand[j] += delta

"""
Smooth‑Adam + sub‑gradient + annealing optimizer.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)**2)

which is homogeneous, therefore we optimise over the probability simplex
(∑a = 1).  The core idea is to replace the non‑smooth “max’’ by a
soft‑max (log‑sum‑exp) with a temperature that is slowly increased.
The gradient of the smoothed functional is a weighted sum of the
convolutions and can be computed with a single FFT‑based Hankel
multiplication.  An Adam‑style adaptive step is taken and the iterate is
projected onto the simplex.  After a few hundred ‑ thousand iterations
the temperature is high enough that the smoothed functional approximates
the true max‑convolution, yielding a high‑quality local minimum.

The final solution is further polished by a short sub‑gradient mirror‑
descent phase and a lightweight simulated‑annealing walk.

All helper functions are top‑level; no closures or lambdas are used.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (full mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (all return a probability vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet distribution with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    total = x.sum()
    return x / total if total > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Adam optimiser with soft‑max smoothing of the max‑convolution
# ----------------------------------------------------------------------


def _smooth_adam(start: np.ndarray,
                 n: int,
                 deadline: float,
                 max_iters: int = 50000,
                 alpha_start: float = 0.5,
                 alpha_end: float = 12.0,
                 alpha_ramp: int = 20000) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimisation of the soft‑max approximated objective.
    The temperature (alpha) grows linearly from `alpha_start` to
    `alpha_end` over `alpha_ramp` iterations and then stays at `alpha_end`.
    Returns the best vector seen and its (exact) objective value.
    """
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- convolution of a with itself (needed for the soft‑max) ---
        conv = _conv_fft(a, a)
        max_c = conv.max()

        # linear schedule for the smoothing parameter
        if it <= alpha_ramp:
            alpha = alpha_start + (alpha_end - alpha_start) * it / alpha_ramp
        else:
            alpha = alpha_end

        # soft‑max weights (max gets weight 1, others ≤ 1)
        w = np.exp(alpha * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient = 2·(Hankel‑matrix(w) · a)
        a_rev = a[::-1]
        full = _conv_fft(w, a_rev)                     # length 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]             # central n entries

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.10 / math.sqrt(it)                     # decreasing base step
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement (mirror descent) – identical to the reference
# ----------------------------------------------------------------------


def _subgrad_refine(start: np.ndarray,
                    n: int,
                    deadline: float,
                    max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth objective.
    Returns the best vector seen and its objective value.
    """
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max(rel_tol, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        w = np.zeros(2 * n - 1, dtype=np.float64)
        if len(active) > 0:
            w[active] = 1.0 / len(active)

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.12 / math.sqrt(it)
        a = a - lr * grad
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing polishing
# ----------------------------------------------------------------------


def _anneal_optimize(start: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """A small Metropolis‑Hastings walk to escape shallow minima."""
    rng = np.random.default_rng()
    cur = start.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # ---- move: transfer a fraction of mass from i to j ----
        i, j = rng.integers(0, n, size=2)
        if i != j:
            delta = rng.random() * 0.16 * cur[i]      # ≤ 16 % of source
            cand = cur.copy()
            cand[i] -= delta
            cand[j] += delta
        else:
            # global jitter
            cand = cur + rng.normal(scale=0.0012, size=n)
            cand = np.clip(cand, 0.0, None)
            total = cand.sum()
            if total > 0:
                cand /= total
            else:
                cand = _uniform(n)

        cand = _project_simplex(cand)
        cand_val = _objective(cand)

        if cand_val < cur_val or rng.random() < math.exp(-(cand_val - cur_val) / (T + 1e-12)):
            cur = cand
            cur_val = cand_val
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] it={it:6d} best={cand_val:.6f}")

        # occasional stronger perturbation
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.0035, size=n)
            jump = best + jitter
            jump = np.clip(jump, 0.0, None)
            jump = _project_simplex(jump)
            jump_val = _objective(jump)
            if jump_val < best_val:
                best_val = jump_val
                best = jump.copy()
                print(f"[Anneal] (jump) it={it:6d} best={jump_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm consists of:

      1.  Assemble a few diverse seeds (uniform, previous best,
          Dirichlet draws, power‑law shapes).
      2.  Run a soft‑max Adam optimiser from each seed.
      3.  Refine the current best vector with a sub‑gradient mirror descent.
      4.  Finish with a short simulated‑annealing walk.

    The best sequence found before the deadline is returned.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # 2 s safety margin

    # --------------------------------------------------------------
    # Length of the vector – reuse previous construction if available
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            prev_arr = np.clip(prev_arr, 0.0, None)
            if prev_arr.sum() > 0:
                prev_arr = prev_arr / prev_arr.sum()
            n = int(prev_arr.shape[0])
        except Exception:
            prev_arr = None
            n = 4000
    else:
        prev_arr = None
        n = 4000

    # --------------------------------------------------------------
    # 0) – initialise a few seeds
    # --------------------------------------------------------------
    rng = np.random.default_rng(123456)
    seeds = [_uniform(n)]
    if prev_arr is not None and prev_arr.shape[0] == n:
        seeds.append(prev_arr)
    for a in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        seeds.append(_random_dirichlet(n, a, rng))
    for p in [0.3, 0.7, 1.5, 2.5]:
        seeds.append(_power(n, p))

    # keep the best few seeds
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(5, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    best_seq = None
    best_val = float("inf")
    for s, v in seed_vals:
        if v < best_val:
            best_val = v
            best_seq = s.copy()
    print(f"[Init] best objective so far = {best_val:.6f}")

    # --------------------------------------------------------------
    # 1) Smooth‑Adam optimisation from each promising seed
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 12.0:
            break
        print(f"[Phase1] Adam optimiser on seed {idx}")
        seq, val = _smooth_adam(seed_vec, n, deadline,
                                max_iters=30000,
                                alpha_start=0.5,
                                alpha_end=12.0,
                                alpha_ramp=15000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Phase1] new global best = {val:.6f}")

    # --------------------------------------------------------------
    # 2) Deterministic sub‑gradient refinement (mirror descent)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8.0:
        print("[Phase2] Sub‑gradient refinement")
        seq, val = _subgrad_refine(best_seq, n, deadline,
                                   max_iters=100000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Phase2] new global best = {val:.6f}")

    # --------------------------------------------------------------
    # 3) Light annealing fine‑tuning
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 4.0:
        print("[Phase3] Simulated‑annealing polishing")
        seq, val = _anneal_optimize(best_seq, n, deadline,
                                    max_iters=200000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Phase3] new global best = {val:.6f}")

    # --------------------------------------------------------------
    # 4) Fallback – should never be needed
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[Fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
