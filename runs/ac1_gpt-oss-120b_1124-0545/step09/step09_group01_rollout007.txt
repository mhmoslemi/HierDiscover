def _lp_refine(a, n, deadline, max_iter=5, t_init=0.5):
    best_a = a.copy()
    best_val = _objective(best_a)
    t = t_init
    for it in range(max_iter):
        if time.time() > deadline:
            break

        # compute M
        conv = _conv_fft(best_a, best_a)
        max_conv = conv.max()

        # Build constraint matrix
        try:
            A = _build_conv_matrix(best_a)   # sparse matrix
        except Exception as e:
            print("Failed to build conv matrix:", e)
            break

        c = -np.ones(n)  # maximize sum(b)
        b_ub = np.full(2*n-1, max_conv)

        bounds = [(0, None)] * n

        try:
            res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                                         method='highs', options={'presolve': True})
        except Exception as e:
            print(f"LP solve error: {e}")
            break

        if not res.success:
            # fallback: break
            print("LP not successful:", res.message)
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            break

        # perform convex combination
        a_new = (1 - t) * best_a + t * g

"""
Improved step‑function search for the `evaluate_sequence` challenge.

The algorithm consists of three stages.

1️⃣  **Seed generation** – a heterogeneous pool (analytic families, random
    Dirichlet draws, monotone‑sorted Dirichlet draws, previous solution)
    of ℓ₁‑normalised vectors.

2️⃣  **Linear‑programming refinement** – for a seed `a` we compute
    `M = max(conv(a,a))`.  The LP
        maximise   Σ b_j
        subject to conv(a,b) ≤ M   (2·n‑1 linear inequalities)
                   b ≥ 0
    yields a vector `b` that can be mixed with `a` while keeping the
    convolution bound.  Repeated mixing (convex combination)
    with a decreasing step size quickly lowers the objective.
    The LP matrix is built as a sparse Toeplitz operator, so the procedure
    works up to a few thousand coefficients inside the time budget.

3️⃣  **Fine tuning** – a short Adam optimisation on a smooth
    log‑sum‑exp surrogate, an exact projected sub‑gradient descent,
    and finally a lightweight simulated‑annealing polishing step.

All stages respect the supplied time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1`` and
returned as a plain list of Python floats.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) (sum(seq)=1 assumed)."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax – returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σv=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (each returns an ℓ₁‑normalised vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw then sorted descending – gives monotone shapes."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]                     # descending
    return a / a.sum()

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """Adam on the smooth log‑sum‑exp surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.004
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                         # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a of Σ w_k·conv_k  (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient descent + Euclidean projection onto the simplex."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.15 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 200_000,
                         init_temp_factor: float = 0.025,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles a small random subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _build_conv_matrix(a: np.ndarray) -> sparse.csr_matrix:
    """
    Build the sparse matrix `A` such that A @ b = conv(a, b)
    (full linear convolution, length 2·n‑1).
    """
    n = a.shape[0]
    rows, cols, data = [], [], []
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(k - i)          # column index of b
            data.append(a[i])
    return sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_refine(a: np.ndarray, n: int, deadline: float,
               max_iter: int = 6, t_init: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Perform a few LP‑guided mixing steps.
    Returns a refined vector (still sum‑to‑1) together with its objective.
    """
    best_a = a.copy()
    best_val = _objective(best_a)
    t = t_init

    for it in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        # current convolution bound
        conv = _conv_fft(best_a, best_a)
        M = conv.max()

        # LP: maximise 1ᵀb  s.t. conv(best_a, b) ≤ M , b ≥ 0
        A = _build_conv_matrix(best_a)
        c = -np.ones(n)                    # maximising Σb_j
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0, None)] * n

        # solve
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            print(f"[LP] iteration {it} – linprog failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0:
            print("[LP] degenerate solution (sum=0).")
            break

        g = b / sum_b                      # normalise to sum=1

        # convex combination
        new_a = (1 - t) * best_a + t * g
        new_a = _proj_simplex(new_a)        # enforce simplex exactly

        new_val = _objective(new_a)

        if new_val < best_val:
            best_val = new_val
            best_a = new_a
            t = max(t * 0.9, 0.05)          # slowly decrease step size
            print(f"[LP] iter {it} improved → {best_val:.6f}, t={t:.3f}")
        else:
            t = max(t * 0.5, 0.01)          # shrink step if no progress
            print(f"[LP] iter {it} no improvement, t reduced to {t:.3f}")

    return best_a, best_val

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a heterogeneous pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_geom(n, 0.6))
    seeds.append(_sym_geom(n, 0.6))
    seeds.append(_power(n, 1.0))
    seeds.append(_beta_shape(n, 0.7, 1.2))
    seeds.append(_gaussian(n, 0.20))
    seeds.append(_tukey_window(n, 0.3))

    # random Dirichlet draws – several concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sorted (monotone) Dirichlet draws – often give good anti‑symmetric shapes
    for conc in (0.05, 0.2, 0.5):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    return seeds


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # tiny safety margin

    # ------------------------------------------------------------------
    # Length choice – 3000 gives a good trade‑off between expressivity
    # and LP memory consumption.
    # ------------------------------------------------------------------
    n = 3000

    # ------------------------------------------------------------------
    # Previous solution (if it exists) – bring it to the required length.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        prev_arr = np.asarray(previous, dtype=np.float64)
        if prev_arr.size == n:
            prev_arr = np.maximum(prev_arr, 0.0)
            if prev_arr.sum() > 0:
                prev_arr = prev_arr / prev_arr.sum()
            else:
                prev_arr = None
        else:
            # reshape/truncate or pad with zeros to match the new length
            if prev_arr.size > n:
                prev_arr = prev_arr[:n]
            else:
                extra = np.zeros(n - prev_arr.size, dtype=np.float64)
                prev_arr = np.concatenate([prev_arr, extra])
            prev_arr = prev_arr / prev_arr.sum() if prev_arr.sum() > 0 else None
    else:
        prev_arr = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a pool of seeds and keep the most promising ones.
    # ------------------------------------------------------------------
    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, prev_arr)
    print(f"Generated {len(seeds)} seeds.")

    # cheap exact surrogate (since all seeds are ℓ₁‑normalised)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = 3
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – LP‐guided refinement (the new ingredient).
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- LP refinement run {idx+1}/{len(best_seeds)} ---")
        refined_seq, refined_val = _lp_refine(seed_vec, n, deadline,
                                              max_iter=6, t_init=0.5)
        if refined_val < global_best_val:
            global_best_seq = refined_seq.copy()
            global_best_val = refined_val
            print(f"[LP] new global best = {refined_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – gradient‑based fine tuning (Adam → sub‑gradient → SA)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_adam, val_adam = _adam_optimize(
            global_best_seq, n, deadline, max_iters=300_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Adam] new global best = {val_adam:.6f}")

    if time.time() < deadline:
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=300_000
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[SubProj] new global best = {val_proj:.6f}")

    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=200_000,
            init_temp_factor=0.03
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for reuse in a subsequent call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
