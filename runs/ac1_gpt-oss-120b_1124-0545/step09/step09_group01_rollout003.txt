"""
Propose a non‑negative step‑function (a probability vector) that
minimises  ``evaluate_sequence``  = 2·n·max(conv(p,p))  (the ℓ₁‑norm is
irrelevant).  The algorithm works in three phases inside the allotted
time budget:

1. **Seed generation** – a broad pool of analytic families
   (beta, power‑law, geometric, trapezoidal, etc.) together with
   random Dirichlet draws (a wide range of concentrations).  The best
   few (default 10) are kept for optimisation.

2. **Smooth‑max Adam optimisation** – the surrogate
   `f_β(p)= (1/β)·log Σ exp(β·conv(p,p))` is minimised with Adam.
   The temperature β is increased continuously up to a very large
   value (≈5 000) so that the surrogate approaches the true max.
   This phase exploits cheap gradients obtained via FFT.

3. **Exact sub‑gradient descent with line‑search** – after the smooth
   phase a true sub‑gradient of the max‑convolution is applied.
   A simple back‑tracking line‑search selects the largest step size
   that yields a strict improvement, which gives a more aggressive
   descent than the fixed‑step version used previously.

4. **Final polishing** – a brief simulated‑annealing stage that
   re‑allocates a random subset of coordinates (Dirichlet reshuffle)
   to escape possible shallow local minima.

All operations respect the time budget (default 1000 s).  The best
sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a probability vector (sum == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (ℓ₁‑normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)
    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]
    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

# Optimisers

def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 800_000,
                   beta_max: float = 5_000.0) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the smooth‐max surrogate.
    Parameters are the logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 5.0               # start temperature
    beta_mul = 1.003
    best_seq = None
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        it += 1
        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # increase temperature slowly, but never exceed beta_max
        beta = min(beta * beta_mul, beta_max)

        # ----- gradient of the smooth‑max surrogate -----
        conv = _conv_fft(a, a)
        shift = np.max(conv)                # for numerical stability
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:       # all underflow – shouldn't happen with shift
            continue
        w /= w_sum                                 # soft‑max over conv entries

        # gradient of Σ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_line_search(start_seq: np.ndarray, n: int, deadline: float,
                            max_iters: int = 400_000,
                            base_lr: float = 0.6,
                            min_lr: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the *exact* max‑convolution objective.
    Uses a simple back‑tracking line‑search to pick the largest step
    size that yields a strict improvement.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 10_000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index (average over ties)
        max_val = conv[max_idx]
        active = np.where(conv == max_val)[0]
        g = np.zeros_like(a)

        for k in active:
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = k - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        g /= float(len(active))                     # average of the active set

        # back‑tracking line‑search
        lr = base_lr / np.sqrt(it)
        while lr > min_lr:
            cand = _proj_simplex(a - lr * g)
            cand_val = _objective(cand)
            if cand_val < cur_val - 1e-12:          # strict decrease
                a = cand
                break
            lr *= 0.5
        # if no improvement we still keep the current a (no update)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 200_000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that randomly reshuffles a small subset of
    coordinates (Dirichlet draw).  Very lightweight but often escapes
    shallow minima.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 14)                # move 2 … 13 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 5_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation utilities
# ----------------------------------------------------------------------


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a heterogeneous pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_trapezoidal(n, width_fraction=0.30))
    seeds.append(_beta_shape(n, 0.5, 0.5))

    # Geometric families (dense sweep)
    for r in np.linspace(0.05, 0.95, 18):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law families
    for p in np.linspace(0.0, 6.0, 20):
        seeds.append(_power(n, p))

    # Beta families – wide sweep
    for a in np.linspace(0.05, 4.0, 12):
        for b in np.linspace(0.05, 6.0, 12):
            seeds.append(_beta_shape(n, a, b))

    # Gaussian windows
    for sigma_frac in (0.10, 0.15, 0.20, 0.25, 0.30):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows
    for alpha in (0.05, 0.15, 0.25, 0.35, 0.45, 0.55, 0.70, 0.85):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – a dense spread of concentrations
    for conc in (0.001, 0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(7):
        seeds.append(_random_exponential(n, rng))

    # Include the previously discovered solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    # ------------------------------------------------------------------
    # Initialise RNG and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0          # 1 s safety margin

    # ------------------------------------------------------------------
    # Length selection – 6000 offers a good trade‑off between granularity
    # and speed (FFT of length 2²⁴ ≈ 16 384).
    # ------------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------------
    # Retrieve previous solution, if any, and build the seed pool
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    print(f"\n--- Collecting seeds (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # quick surrogate evaluation (exact because sum=1)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = 10
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Smooth‑max Adam optimisation on each elite seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation run {idx+1}/{len(best_seeds)} ---")
        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline,
            max_iters=800_000,      # a generous budget per seed
            beta_max=5_000.0
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent with line‑search
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient refinement ---")
        seq_sub, val_sub = _subgradient_line_search(
            global_best_seq, n, deadline,
            max_iters=400_000,
            base_lr=0.8
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=200_000,
            init_temp_factor=0.03
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse in a subsequent call
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation expands the search space, raises the smooth‑max temperature to a much larger value, introduces an aggressive line‑search sub‑gradient phase and a brief annealing polish.  All work is performed inside the 1000 s budget, and the best sequence is returned as a plain list of floats.
