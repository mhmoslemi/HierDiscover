"""
Improved step‑function search.

The algorithm proceeds in three phases:

1. **Initialisation** – start from the best known sequence (or a
   triangular seed) and normalise it to ℓ₁‑norm = 1.

2. **Linear‑programming refinement** – for the current shape `f` we
   compute `M = max(conv(f,f))`.  Holding `f` fixed we solve
   a linear programme that maximises the ℓ₁‑norm of a non‑negative
   vector `b` under the linear constraints `conv(f,b) ≤ M`.  If the
   optimum has sum > 1 we normalise `b` and replace `f`.  Each LP
   strictly reduces the objective `2·n·max(conv(f,f))`.

3. **Local polishing** – after the LP phase we run the same
   Adam + projected sub‑gradient routine from the previous state‑of‑the‑art
   implementation, followed by a short simulated‑annealing pass.

All work respects the supplied time budget (default ≈ 1000 s).  The
best sequence found is stored in the global variable
``height_sequence_1`` and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation for sequences that are ℓ₁‑normalised to sum = 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (Σ=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 1_200_000) -> tuple[np.ndarray | None, float]:
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0
    beta_mul = 1.004
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v

        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val

        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 800_000,
                         init_temp_factor: float = 0.025,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve:
        maximize   sum(b)
        subject to conv(f, b) ≤ M  (component‑wise)
                   b ≥ 0
    Returns the optimal `b` (not normalised).  Returns None on failure.
    """
    n = f.shape[0]

    # Build sparse matrix A where A @ b = conv(f, b)
    # A has shape (2*n-1, n) and each diagonal offset d (0≤d<n) contains f[d]
    data = []
    offsets = []
    for d in range(n):
        # length of this diagonal = n - d
        data.append(np.full(n - d, f[d], dtype=np.float64))
        offsets.append(d)

    A = diags(data, offsets, shape=(2 * n - 1, n), format='csr')

    # Upper‑bound vector
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme: minimise -1·b  ⇔ maximise sum(b)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})

    if not res.success:
        print(f"[LP] linprog failed: {res.message}")
        return None
    return res.x


def _lp_improve(f: np.ndarray, deadline: float,
                max_outer_iters: int = 8,
                min_sum_improve: float = 1e-8) -> tuple[np.ndarray, float]:
    """Iteratively apply the LP refinement described in the docstring."""
    best_f = f.copy()
    best_val = _objective(best_f)

    for it in range(1, max_outer_iters + 1):
        if time.time() > deadline:
            break

        M = np.max(_conv_fft(best_f, best_f))
        b = _solve_lp_max_sum(best_f, M)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b <= 1.0 + min_sum_improve:
            # No useful increase → stop
            break

        f_new = b / sum_b                     # renormalise to ℓ₁ = 1
        val_new = _objective(f_new)

        if val_new < best_val - 1e-12:
            print(f"[LP] iter={it:2d} sum(b)={sum_b:.6f}  val={val_new:.6f}")
            best_f = f_new
            best_val = val_new
        else:
            break

    return best_f, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.

    The pipeline is
      * initialise from the best known sequence (or a triangular seed)
      * run a few iterations of a linear‑programming refinement
      * polish with Adam → projected sub‑gradient → simulated annealing
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0  # tiny safety margin

    # ------------------------------------------------------------------
    # Load the best previously found candidate if it exists.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.ndim != 1:
                prev = None
        except Exception:
            prev = None
    else:
        prev = None

    # ------------------------------------------------------------------
    # Determine the working length.
    # Larger n gives a finer shape but also a larger LP.
    # We stick to n = 4000 (the size used previously) unless a
    # previous solution with a different length is present.
    # ------------------------------------------------------------------
    if prev is not None:
        n = prev.shape[0]
        init_seq = prev.copy()
    else:
        n = 4000
        init_seq = _triangular(n)

    # Normalise to ℓ₁ = 1 (objective is scale‑invariant)
    s = init_seq.sum()
    if s <= 0.0:
        init_seq = _uniform(n)
        s = init_seq.sum()
    seq = init_seq / s

    print(f"\n--- Starting optimisation (n={n}) ---")
    print(f"Initial surrogate value: {_objective(seq):.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – linear‑programming improvement
    # ------------------------------------------------------------------
    seq, cur_val = _lp_improve(seq, deadline, max_outer_iters=6)
    print(f"[LP‑phase] best surrogate after LP = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deterministic local optimisation (Adam + sub‑gradient)
    # ------------------------------------------------------------------
    # Adam optimisation (smooth max surrogate)
    seq_adam, val_adam = _adam_optimize(seq, n, deadline,
                                        max_iters=1_500_000)
    if seq_adam is not None and val_adam < cur_val:
        seq = seq_adam
        cur_val = val_adam
        print(f"[Adam] new best = {cur_val:.6f}")

    # Exact projected sub‑gradient descent
    seq_proj, val_proj = _subgradient_proj(seq, n, deadline,
                                           max_iters=2_500_000)
    if val_proj < cur_val:
        seq = seq_proj
        cur_val = val_proj
        print(f"[SubProj] new best = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(seq, n, deadline,
                                              max_iters=1_200_000,
                                              init_temp_factor=0.03)
        if val_sa < cur_val:
            seq = seq_sa
            cur_val = val_sa
            print(f"[SA] final best = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {cur_val:.6f}, "
              f"official = {final_score:.6f}, length = {len(seq)}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for possible reuse
    globals()["height_sequence_1"] = seq.tolist()
    return seq.tolist()
