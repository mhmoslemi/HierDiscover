"""
Propose a step‑function (non‑negative sequence) with a low value of

    2·n·max(conv(a,a)) / (sum(a)^2)

for the given evaluate_sequence() scorer.

The main idea is to search for a large *low‑collision* support set S⊂{0,…,n‑1}
such that each ordered pair‑sum i+j (i,j∈S) appears at most
``max_ordered`` times (here 4).  For a uniform distribution on S we have

    max(conv) ≤ max_ordered / |S|²   →   objective ≤ 2·n·max_ordered / |S|².

Since the theoretical optimum satisfies |S| ≈ √(max_ordered·2n),
the objective can be driven close to 1 as |S| approaches that bound.
The optimizer therefore spends most of the 1000 s budget constructing
a large low‑collision set via many cheap greedy attempts, then refines
the probabilities on that support with a short sub‑gradient descent.

If the greedy phase does not reach the target size, the code falls back
to a generic Adam optimisation (the same routine used in the previous
solution).  The final sequence is normalised to sum = 1 and stored in the
global variable ``height_sequence_1`` for possible reuse by a later run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (exact, real‐valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution via a real FFT (full mode).
    Returns length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    Assumes seq non‑negative and already normalised to sum=1.
    """
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple Adam optimiser (soft‑max surrogate for the max)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray, n: int, deadline: float,
                   max_iters: int = 30_000):
    """
    Adam on a smooth approximation of max(conv(a,a)).
    Returns the best sequence seen (as a numpy array) and its true objective.
    """
    # work in log‑space (softmax → simplex)
    v = np.log(seed + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # temperature for the soft‑max over the convolution
    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ---- smoothed‑max weights ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·(a*a)_k ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam update ----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)
        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true (non‑smooth) objective
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 200_000):
    """
    Exponentiated‑gradient (mirror descent) on the exact objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]

        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:6d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Low‑collision set generator (ordered sums ≤ max_ordered)
# ----------------------------------------------------------------------
def _search_low_collision_set(n: int,
                              max_ordered: int,
                              deadline: float) -> np.ndarray:
    """
    Repeated cheap greedy constructions.  Returns the index array of the
    best set (as a NumPy ``int`` array).  The greedy algorithm respects
    the bound on ordered pair‑sums.
    """
    best_mask = None
    best_size = 0
    attempts = 0

    # theoretical upper bound on the size (m² ≤ max_ordered·(2n−1))
    theoretical_m = int(math.sqrt(max_ordered * (2 * n - 1)))

    while time.time() < deadline:
        attempts += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)

        # keep a Python list of the current indices for fast updates
        cur_idx = []

        # random order each attempt
        order = np.random.permutation(n)
        for cand in order:
            # self‑pair test
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue

            if cur_idx:
                cand_sums = cand + np.array(cur_idx, dtype=np.int32)
                if np.any(sum_counts[cand_sums] + 2 > max_ordered):
                    continue

            # accept candidate
            mask[cand] = True
            cur_idx.append(cand)
            sum_counts[2 * cand] += 1
            if len(cur_idx) > 1:
                sum_counts[cand_sums] += 2

        cur = mask.sum()
        if cur > best_size:
            best_size = cur
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:4d} size {cur} "
                  f"(theoretical {theoretical_m})")

        # early exit if we already hit the theoretical optimum
        if best_size >= theoretical_m:
            break

    if best_mask is None:
        return np.array([], dtype=int)
    return np.where(best_mask)[0]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` score.  The routine respects a ~1000 s budget
    and returns the best sequence it found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    max_ordered = 4

    print(f"\n=== Optimising length {n} (max_ordered={max_ordered}) ===")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive search for a large low‑collision support set
    # ------------------------------------------------------------------
    set_deadline = start_time + 0.70 * (deadline - start_time)   # ~70 % of budget
    lowcoll_idx = _search_low_collision_set(
        n=n,
        max_ordered=max_ordered,
        deadline=set_deadline,
    )
    if lowcoll_idx.size == 0:
        print("[Info] No low‑collision set found – falling back.")
    else:
        print(f"[Info] best low‑collision set size {lowcoll_idx.size}")

    # ------------------------------------------------------------------
    # Phase 2 – build a uniform distribution on that support
    # ------------------------------------------------------------------
    if lowcoll_idx.size:
        m = lowcoll_idx.size
        seq = np.zeros(n, dtype=np.float64)
        seq[lowcoll_idx] = 1.0 / m
        seq /= seq.sum()          # normalise (should already be 1)
    else:
        # fallback – uniform over all indices
        seq = np.ones(n, dtype=np.float64) / n

    cur_val = _objective(seq)
    print(f"[Init] objective on uniform low‑collision support = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – if the objective is still above the target, run a short
    #           refinement (Adam → Sub‑gradient) using the remaining time
    # ------------------------------------------------------------------
    target = 1.503
    time_left = deadline - time.time()
    if cur_val > target and time_left > 2.0:
        # Adam refinement (smooth max surrogate)
        adam_deadline = time.time() + min(30.0, time_left * 0.4)
        seq_adam, val_adam = _adam_optimize(seq, n, adam_deadline,
                                            max_iters=40_000)
        if seq_adam is not None and val_adam < cur_val:
            seq = seq_adam
            cur_val = val_adam
            print(f"[Refine] after Adam best = {cur_val:.6f}")

        # Sub‑gradient refinement (exact max)
        time_left = deadline - time.time()
        if cur_val > target and time_left > 2.0:
            sub_deadline = time.time() + min(45.0, time_left * 0.5)
            seq_sub, val_sub = _subgradient_refine(seq, n, sub_deadline,
                                                   max_iters=300_000)
            if seq_sub is not None and val_sub < cur_val:
                seq = seq_sub
                cur_val = val_sub
                print(f"[Refine] after sub‑gradient best = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks, normalisation and export
    # ------------------------------------------------------------------
    final_seq = np.maximum(seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.ones(n, dtype=np.float64) / n
    else:
        final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} "
          f"(support size {int(final_seq.sum()>0)})")

    # expose for possible reuse
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
