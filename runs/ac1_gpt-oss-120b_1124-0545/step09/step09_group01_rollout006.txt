def _exp_gradient_descent(start_vec, n, deadline, max_iters=..., eta0=0.2):
   a = start_vec.copy()
   best_seq = a.copy()
   best_val = _objective(a)
   for it in range(1, max_iters+1):
      if time.time() > deadline: break
      # compute conv
      conv = _conv_fft(a, a)
      k = int(np.argmax(conv))
      # compute gradient g_i = 2 a_{k-i}
      g = np.zeros_like(a)
      # i from max(0, k-(n-1)) ... min(n-1, k)
      i0 = max(0, k-(n-1))
      i1 = min(n-1, k)
      idx = np.arange(i0, i1+1)
      rev_idx = k - idx
      g[idx] = 2 * a[rev_idx]
      # update with exponentiated gradient
      eta = eta0 / np.sqrt(it)
      w = a * np.exp(-eta * g)
      # avoid zeros: if any w is zero due to underflow, set small positive.
      # Compute sum and renormalize:
      s = w.sum()
      if s == 0:
          a = _uniform(n)
      else:
          a = w / s
      # evaluate objective every certain steps
      if it % 1000 == 0:
          cur_val = _objective(a)
          if cur_val < best_val:
              best_val = cur_val
              best_seq = a.copy()
              print("[ExpGD] it={} best={}".format(it, best_val))
   return best_seq, best_val

"""
Propose a non‑negative step‑function (probability vector) that minimises
the black‑box ``evaluate_sequence`` objective.

Approach
--------
1. Build a large, diverse pool of ℓ₁‑normalised candidates.
   Families include uniform, triangular, trapezoidal, geometric,
   symmetric geometric, power‑law, sine‑window, beta‑shaped,
   Gaussian, Tukey and random Dirichlet / exponential draws.
   All candidates are evaluated with the exact objective
   (since the sum is forced to 1, the surrogate coincides with the
   true value).

2. Keep the best few seeds and refine them with three complementary
   optimisation phases, each respecting the time budget.

   • **Exponentiated‑gradient descent** – a simple EG update on the
     true max‑convolution sub‑gradient.  It works directly on the
     simplex and converges quickly on the non‑smooth landscape.

   • **Adam on a smooth log‑sum‑exp surrogate** – the surrogate turns
     the max into a soft‑max controlled by a temperature β that is
     increased slowly, allowing large steps early on and focusing on
     the true max later.

   • **Exact projected sub‑gradient descent** – the classic descent
     used in the reference solution (g_i = 2·a_{k−i} where k is the
     index of the current maximal convolution entry).

3. **Simulated annealing polishing** – random Dirichlet reshuffles of
   a small subset of coordinates are performed with an exponentially
   decreasing temperature to escape shallow local minima.

All stages stop as soon as the allotted time expires.  The best sequence
found is stored in the global variable ``height_sequence_1`` and returned
as a list of floats.

The implementation respects the required ``propose_candidate`` signature
and uses only the standard library together with NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation for a sequence that is already ℓ₁‑normalised to sum = 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σ=1, v≥0."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Parameters are the logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule (start low, slowly increase)
    beta_temp = 5.0
    beta_mul = 1.004
    beta_max = 500.0  # allow a larger final temperature

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # raise temperature slowly so we approach the true max
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)                # exact convolution
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over the convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact projected sub‑gradient descent on the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # Polyak‑style step size (decrease slowly)
        lr = 0.18 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _exp_gradient_descent(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the true max‑convolution.
    The update is a = a * exp(-η·g) followed by simplex projection.
    """
    a = start_vec.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # sub‑gradient of (a*a)_k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx
        g[idx] = 2.0 * a[rev_idx]

        # EG step (η decreases like 1/√t)
        eta = eta0 / np.sqrt(it)
        w = a * np.exp(-eta * g)
        s = w.sum()
        if s == 0.0 or np.isnan(s):
            a = _uniform(n)
        else:
            a = w / s

        if it % 3000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[ExpGD] it={it:7d} best={best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.025,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that randomly redistributes the mass
    of a small subset of coordinates (Dirichlet reshuffle).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Assemble a heterogeneous pool of ℓ₁‑normalised candidates.
    The pool is deliberately larger than in the reference code
    to improve the chance of finding a superior start point.
    """
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.10))
    seeds.append(_trapezoidal(n, width_fraction=0.30))
    seeds.append(_sine_window(n, gamma=0.5))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    # Geometric families
    for r in np.linspace(0.15, 0.99, 14):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family (wide sweep)
    for p in np.linspace(0.0, 8.0, 25):
        seeds.append(_power(n, p))

    # Beta family – exhaustive random sampling
    for _ in range(120):
        a = rng.uniform(0.05, 4.5)
        b = rng.uniform(0.05, 4.5)
        seeds.append(_beta_shape(n, a, b))

    # Gaussian windows – varied widths
    for sigma_frac in (0.08, 0.12, 0.16, 0.20, 0.24, 0.30):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows – many alphas
    for alpha in np.linspace(0.05, 0.90, 18):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – diverse concentrations
    for conc in (0.001, 0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws – heavy tailed
    for _ in range(8):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any) – interpolated to current length
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.ndim == 1 and prev.shape[0] > 0:
                # Simple linear interpolation to the new length
                x_old = np.linspace(0, 1, prev.shape[0])
                x_new = np.linspace(0, 1, n)
                interp = np.interp(x_new, x_old, prev)
                interp = np.maximum(interp, 0.0)
                s = interp.sum()
                if s > 0.0:
                    seeds.append(interp / s)
        except Exception:
            pass

    return seeds

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    print(f"Top {top_k} surrogate values:")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – exponentiated‑gradient descent (very cheap & robust)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_exp, val_exp = _exp_gradient_descent(
            global_best_seq, n, deadline, max_iters=600_000, eta0=0.35
        )
        if val_exp < global_best_val:
            global_best_seq, global_best_val = seq_exp, val_exp
            print(f"[overall‑ExpGD] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – Adam optimiser on the smooth log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_adam, val_adam = _adam_optimize(
            global_best_seq, n, deadline, max_iters=800_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq, global_best_val = seq_adam, val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – exact projected sub‑gradient descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=1_200_000
        )
        if val_proj < global_best_val:
            global_best_seq, global_best_val = seq_proj, val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=900_000, init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check – feed the candidate to the official evaluator.
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest approx ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
