"""
propose_candidate – multi‑phase search for a low‑value step‑function.

The algorithm explores a rich collection of analytic families (uniform,
Beta‑pdf, Gaussian, power‑law, Hann, Blackman, …) on several lengths,
augments this with massive random Dirichlet sampling and random mixtures,
then refines the best candidates with a smooth‑max Adam optimiser,
optional L‑BFGS (if SciPy is present), a short simulated‑annealing
exploration, and a final aggressive KL‑mirror‑descent polish.
All sequences are kept on the probability simplex (sum = 1) so that the
internal objective equals the value returned by ``evaluate_sequence``.
The routine respects a user‑supplied time budget (default 1000 s) and
prints progress messages.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# optional SciPy import for L‑BFGS‑B
try:
    from scipy.optimize import minimize
except Exception:  # pragma: no cover
    minimize = None

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Internal objective for a *normalized* sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all returned already normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total > 0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * math.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * math.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * math.pi * i / (n - 1)) + a2 * np.cos(4 * math.pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _powerlaw(n: int, p: float) -> np.ndarray:
    """w_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-p)
    total = w.sum()
    return w / total


def _tent(n: int) -> np.ndarray:
    """Symmetric triangular (tent) shape."""
    half = n // 2
    if n % 2 == 0:
        left = np.arange(1, half + 1, dtype=np.float64)
        right = left[::-1]
        w = np.concatenate([left, right])
    else:
        left = np.arange(1, half + 2, dtype=np.float64)
        right = left[:-1][::-1]
        w = np.concatenate([left, right])
    return w / w.sum()

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


def _random_mixture(n: int, rng: np.random.Generator) -> np.ndarray:
    """Mixture of a few Beta PDFs with random mixing weights."""
    k = rng.integers(2, 5)             # number of components 2–4
    alphas = rng.uniform(0.2, 5.0, size=k)
    betas = rng.uniform(0.2, 5.0, size=k)
    weights = rng.dirichlet(np.ones(k))
    mix = np.zeros(n, dtype=np.float64)
    for w, a, b in zip(weights, alphas, betas):
        mix += w * _beta_shape(n, a, b)
    return mix / mix.sum()

# Smooth‑max Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(start_seq: np.ndarray,
                          rng: np.random.Generator,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 200.0,
                          max_iter: int = 5_000_000,
                          lr0: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam on
    the log‑softmax (logits) parametrisation.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    # Adam moments
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # soft‑max → probability vector a
        a = np.exp(logits)
        a /= a.sum()

        # convolution a*a
        conv = _conv_fft(a, a)

        # annealed temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()          # soft max‑weights, sum=1

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain‑rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / math.sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:        # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.45:   # early stop if we already beat a very good target
                break

    return best_seq, best_val

def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smooth‑max objective.
    Returns the best exact objective and the associated sequence.
    """
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits):
        a = np.exp(logits)
        a /= a.sum()
        conv = _conv_fft(a, a)

        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOutException(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOutException()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 300, 'disp': False},
            callback=callback,
        )
    except TimeOutException:
        best_logits = x0
    else:
        best_logits = res.x

    a = np.exp(best_logits)
    a /= a.sum()
    return a, _objective(a)

# Aggressive aggregated KL‑mirror descent (polishing)
# ----------------------------------------------------------------------
def _refine_aggregated_subgradient(start_seq: np.ndarray,
                                   deadline: float,
                                   eps_rel: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* indices whose
    convolution entry is within a relative tolerance of the current maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = 0.7 / math.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")
            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val

def _simulated_annealing_optimize(start_seq: np.ndarray,

                                  deadline: float,

    Very cheap SA that moves a tiny amount of mass between two random
    coordinates.  Accepts uphill moves with an exponentially decreasing
    temperature schedule.

    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    for it in range(1, max_steps + 1):
        if time.time() >= deadline:
            break

        i, j = rng.choice(n, size=2, replace=False)
        # step size decays like 1/√it; keep it small to maintain feasibility
        step = 0.02 / math.sqrt(it)
        delta = step * a[i] * rng.random()
        if delta <= 0.0:
            continue
        if delta > a[i]:
            delta = a[i]

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective(a_new)

        if val < cur_val:
            a = a_new
            cur_val = val
            if val < best_val:
                best_val = val
                best_seq = a_new.copy()
        else:
            # temperature drops linearly to ~1e-6
            T = max(1e-6, 0.02 * (1.0 - it / max_steps))
            if rng.random() < math.exp(-(val - cur_val) / T):
                a = a_new
                cur_val = val

        if (it & 2047) == 0:
            print(f"[SA] it={it:5d} cur={cur_val:.6f} best={best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the supplied time budget.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Possibly reuse a previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – we pick a mixture of modest and larger sizes.
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 12000, 16000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families (including a grid of Beta / power‑law)
    # ------------------------------------------------------------------
    analytic_fns = [
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, 0.12),
        lambda n: _gaussian(n, 0.15),
        lambda n: _gaussian(n, 0.20),
        lambda n: _gaussian(n, 0.25),
        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 0.3, 0.3),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
        lambda n: _beta_shape(n, 1.5, 1.5),
        lambda n: _powerlaw(n, 0.5),
        lambda n: _powerlaw(n, 1.0),
        lambda n: _powerlaw(n, 1.5),
        lambda n: _powerlaw(n, 2.0),
        _tent,
    ]

    # additional fine‑grained Beta sweep (small grid)
    beta_grid = [0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0]

    for n in lengths:
        for fn in analytic_fns:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        # systematic beta grid
        for a in beta_grid:
            for b in beta_grid:
                seq = _beta_shape(n, a, b)
                val = _objective(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(f"[BetaGrid] n={n:5d} α={a:.1f} β={b:.1f} val={val:.6f}")

        # try to reuse a previously stored sequence when the length matches
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Prev] n={n:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy random sampling (Dirichlet + random mixtures)
    # ------------------------------------------------------------------
    now = time.time()
    # about 30 % of the remaining time for sampling
    time_for_sampling = (deadline - now) * 0.30
    sample_deadline = now + time_for_sampling

    alphas = [0.2, 0.5, 1.0, 2.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet draw
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta shape
        a_par = float(rng.choice(beta_grid))
        b_par = float(rng.choice(beta_grid))
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Beta] n={n:5d} α={a_par:.2f} β={b_par:.2f} val={val:.6f}")

        # Random mixture of Betas
        seq = _random_mixture(n, rng)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rnd-Mix] n={n:5d} val={val:.6f}")

        # early break if we discover an exceptionally low value
        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # keep a few diverse candidates for Adam restarts
        candidates = [best_seq]
        # add a couple of fresh random starts of the same length as the best
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=float(rng.choice(alphas)))
            candidates.append(cand)

        # allocate roughly 25 % of the remaining budget to Adam runs
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25
        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                rng=rng,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=200.0,
                max_iter=4_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional L‑BFGS refinement (if SciPy is available)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 4.0:
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=200.0)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – cheap simulated‑annealing exploration from the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            sa_deadline = time.time() + remaining * 0.15
            seq_sa, val_sa = _simulated_annealing_optimize(best_seq, rng, sa_deadline)
            if val_sa < best_val:
                best_val = val_sa
                best_seq = seq_sa
                print(f"[SA‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – aggressive mirror‑descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_aggregated_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the scorer will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    # make the sequence a plain Python list of floats
    return best_seq.tolist()
