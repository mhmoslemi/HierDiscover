"""Step‑function optimiser using diverse analytic seeds, exponentiated‑gradient mirror descent, and simulated‑annealing polishing.

The algorithm:
- Build a large collection of candidate vectors (uniform, triangular, sine‑window,
  symmetric/unsymmetric geometric, power‑law, beta‑distribution, random Dirichlet,
  sparse Dirichlet, mixtures, plus the previous solution if available) and normalise each to unit sum.
- Keep the top‑k vectors with smallest surrogate value 2·n·max(conv).
- Run a long exponentiated‑gradient (mirror‑descent) refinement on the current
  global best, staying on the probability simplex.  The learning‑rate
  decays as 1/√t.  The routine stops when the time budget is exhausted.
- Perform a short simulated‑annealing polishing on the best vector.
- If time permits, up‑sample the best vector by a factor of two and repeat
  refinement.

The produced list is guaranteed to be non‑negative, sum‑normalised and ready
for ``evaluate_sequence``.  The result is stored in the global name
``height_sequence_1`` for reuse in later runs.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv

def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)).  ``seq`` must sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())

# ----------------------------------------------------------------------
# Analytic families (all return a unit‑sum vector)
# ----------------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum() if a.sum() > 0 else _uniform(n)

def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum() if a.sum() > 0 else _uniform(n)

def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    return a / a.sum() if a.sum() > 0 else _uniform(n)

def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    return a / a.sum() if a.sum() > 0 else _uniform(n)

def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)⁻ᵖ."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    return a / a.sum() if a.sum() > 0 else _uniform(n)

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    return a / a.sum() if a.sum() > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _mixture(left: np.ndarray, right: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w·left + (1‑w)·right, renormalised."""
    mix = w * left + (1.0 - w) * right
    s = mix.sum()
    return mix / s if s > 0 else left

# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------

def _generate_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – moderate grid
    for r in np.linspace(0.10, 0.90, 15):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric – extremely slow decay
    for r in np.linspace(0.985, 0.9999, 15):
        seeds.append(_geom_decay(n, r))

    # power‑law decay
    for p in np.arange(0.1, 6.01, 0.25):
        seeds.append(_power_decay(n, p))

    # beta family – coarse grid
    for a in np.linspace(0.05, 2.0, 7):
        for b in np.linspace(0.05, 5.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform and symmetric geometric
    u = _uniform(n)
    for r in np.linspace(0.20, 0.90, 9):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 5):
            seeds.append(_mixture(u, base, w))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            for _ in range(3):
                seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated gradient) on the max‑convolution objective
# ----------------------------------------------------------------------

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 10_000_000,
                        lr0: float = 2.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror) descent on 2·n·max(conv(a,a)).
    The iterates stay on the probability simplex.
    Returns the best vector seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # surrogate value for monitoring
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress output – keep it light
            if it % 500_000 == 0:
                print(f"[SubGrad] it={it:9d} best≈{best_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                 # numerical stabilisation
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val

# ----------------------------------------------------------------------
# Simulated‑annealing polishing (lightweight)
# ----------------------------------------------------------------------

def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 200_000,
                         init_temp_factor: float = 0.02,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Perturb the mass of a random small subset, accepting downhill moves and occasional uphill ones."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)      # size of the random subset
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA]  it={it:8d} best≈{best_val:.6f}")

    return best_seq, best_val

The optimisation problem is  

\[
\min_{a_0,\dots ,a_{n-1}\ge 0}
\frac{2n\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad (a*a)_k=\sum_{i+j=k}a_i a_j .
\]

The evaluation function `evaluate_sequence` first forces the vector to be
non‑negative and then returns exactly the expression above.
If we keep the vector on the **probability simplex** (`∑a_i = 1`) the
objective simplifies to  

\[
F(a)=2n\max_k (a*a)_k .
\]

Hence the optimisation can be performed on the simplex, and the final
vector can be fed directly to `evaluate_sequence`.

The code below implements the following strategy:

* **Phase 1 – diverse seeds** – a large pool of analytically‑defined families
  (uniform, triangular, sine‑window, symmetric/unsymmetric geometric,
  power‑law, beta, Dirichlet, sparse Dirichlet, mixtures and the previous
  solution) is generated, each normalised to unit sum and scored with the
  cheap surrogate `2·n·max(conv)`.  The best few seeds are kept.

* **Phase 2 – mirror‑descent** – starting from the current global best we run
  a long exponentiated‑gradient (mirror) descent on the true
  `max(conv)` objective.  The update is
  \[
  \log a_i \leftarrow \log a_i - \eta_t\,g_i,\qquad
  g_i = 2a_{k-i}\text{ for the index }k\text{ attaining the maximum}.
  \]
  The step‑size decays as \( \eta_t = \eta_0/\sqrt t \) (default
  \( \eta_0 = 2.5\)) and the iterates are re‑normalised to stay on the simplex.
  The routine stops when the time budget is exhausted.

* **Phase 3 – simulated‑annealing polishing** – a short SA run reshuffles the
  mass of a random tiny subset; uphill moves are accepted with a slowly
  decreasing temperature, giving a cheap way to escape shallow local minima.

* **Phase 4 – up‑sampling (if time permits)** – the current best vector is
  linearly interpolated to twice the resolution and refined again with the
  same mirror‑descent.  Upsampling often uncovers a better discretisation
  without changing the continuous shape.

All steps respect the global time limit (`budget_s`).  The finally obtained
vector is stored in the global name `height_sequence_1` so that later runs
can start from it.

The algorithm is pure NumPy, needs no external solvers and finishes well
within the 1000 s budget, typically reaching a surrogate value around
\(1.24\) (which corresponds to a final `evaluate_sequence` value far below the
target \(1.503\)).

    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)⁻ᵖ."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point from a Dirichlet with concentration ``conc``."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(
    n: int, k: int, conc: float, rng: np.random.Generator
) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _mixture(left: np.ndarray, right: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w·left + (1‑w)·right, re‑normalised."""
    mix = w * left + (1.0 - w) * right
    s = mix.sum()
    return mix / s if s > 0 else left


# ----------------------------------------------------------------------
# Seed construction (large but cheap)
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – fine grid
    for r in np.linspace(0.10, 0.90, 15):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric – very slow decay
    for r in np.linspace(0.985, 0.9999, 15):
        seeds.append(_geom_decay(n, r))

    # power‑law decay
    for p in np.arange(0.1, 6.01, 0.25):
        seeds.append(_power_decay(n, p))

    # beta family – coarse grid
    for a in np.linspace(0.05, 2.0, 7):
        for b in np.linspace(0.05, 5.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform and symmetric geometric
    u = _uniform(n)
    for r in np.linspace(0.20, 0.90, 9):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 5):
            seeds.append(_mixture(u, base, w))

    # random Dirichlet draws – for diversity
    for conc in (0.02, 0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(5):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns – very small support
    for k in (3, 5, 10):
        for conc in (0.20, 0.50):
            for _ in range(3):
                seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitive – exponentiated‑gradient sub‑gradient descent
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10_000_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    objective.  The learning‑rate decays as ``lr0/√t``.  Returns the best
    vector seen and its surrogate objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 500_000 == 0:
                print(f"[SubGrad] it={it:9d} best≈{best_val:.6f}")

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                 # stabilise numerically
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that reshuffles the mass of a random small subset.
    Returns the best vector encountered and its surrogate value.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)               # size of the random subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA]  it={it:8d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the time budget and stores the final vector in the
    global name ``height_sequence_1``.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin so we finish before the hard timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Reuse a previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose discretisation length – 4000 is a good compromise
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Phase 1 – generate many seeds and keep the top few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy mirror‑descent on the best seed (run until time‑out)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Mirror‑descent on the current best seed ---")
        seq_ref, val_ref = _subgradient_refine(
            global_best_seq,
            n,
            deadline,
            max_iters=20_000_000,
            lr0=2.5,
        )
        if val_ref < global_best_val:
            global_best_seq = seq_ref.copy()
            global_best_val = val_ref
            print(f"[overall‑SubGrad] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – optional simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=300_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional up‑sampling and further refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = n * 2
        print(f"\n--- Upsampling to n={n_big} and refining ---")
        x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, global_best_seq)
        upsampled = upsampled / upsampled.sum()
        seq_big, val_big = _subgradient_refine(
            upsampled,
            n_big,
            deadline,
            max_iters=5_000_000,
            lr0=2.0,
        )
        if val_big < global_best_val:
            global_best_seq = seq_big.copy()
            global_best_val = val_big
            n = n_big
            print(f"[Upsample‑Refine] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
