"""
Non‑negative step‑function optimiser – version 3.

This implementation builds on the previous version and adds several
independent ideas that have proven useful for shaving a few
hundredths off the bound:

*   A richer set of deterministic “window” families (Hann, Hamming,
    Blackman) and a finer grid of geometric / power‑law decays.
*   An extra global refinement based on a tiny linear programme:
    given a candidate f we solve
        minimise   M
        subject to  (f * g)_k ≤ M   (for the most critical indices)
                    Σ g_i = 1,  g_i ≥ 0 .
    The LP is very cheap because we only keep the ~200 most binding
    convolution constraints.  The resulting g is accepted when it
    improves the true objective.
*   Longer Adam, sub‑gradient and Euclidean‑projection phases
    (more iterations, slightly larger step‑size schedules).
*   A short “mix‑and‑match” step that averages the current best vector
    with the LP solution – this mimics the original convex‑combination
    update of the classic Frank‑Wolfe style scheme.
*   All steps respect the user supplied time‑budget (default 1000 s).

The function ``propose_candidate`` returns the best sequence found as
a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    The input sequence is assumed to sum to 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (all normalised to sum = 1)
# --------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw sorted in descending order (strongly skewed)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_blocks(n: int, max_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """
    Very simple block‑wise pattern:
    choose k (2 ≤ k ≤ max_blocks) blocks of equal length,
    place them evenly with gaps, mass is spread uniformly inside each block.
    """
    k = rng.integers(2, max_blocks + 1)
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos:pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# --------------------------------------------------------------
# Seed construction (diverse, cheap)
# --------------------------------------------------------------

def _generate_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))

    # geometric – finer grid
    for r in np.linspace(0.65, 0.99, 17):
        seeds.append(_geom_decrease(n, r))

    # power‑law – finer grid
    for p in np.arange(0.2, 3.01, 0.2):
        seeds.append(_power_decrease(n, p))

    # sorted Dirichlet draws
    for conc in (0.02, 0.05, 0.10, 0.20, 0.50, 1.0):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # block‑wise patterns
    for max_blocks in (3, 5, 7, 10, 15):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (5, 10, 20, 40, 80):
        for conc in (0.10, 0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

# Optimisation primitives
# --------------------------------------------------------------

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 120000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp (smoothed max) surrogate.
    The vector stays on the simplex via a log‑softmax parametrisation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule for the log‑sum‑exp smoothing
    beta = 0.5
    beta_mul = 1.001
    beta_max = 2000.0            # more smoothing → closer to the true max

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                    # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # anneal temperature
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.9
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the true
    max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _euc_proj_subgradient(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 250000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the true max‑convolution with Euclidean projection
    onto the probability simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[EucProj] it={it:7d} best={cur_val:.6f}")

        # gradient of the active convolution entry (same as sub‑grad)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)               # a bit more aggressive than exp‑grad
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simulated annealing that randomly reshuffles the mass of a few coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # pick a small subset (3‑7 indices) and redistribute their total mass
        k = rng.integers(3, 8)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray,
               n: int,
               deadline: float,
               max_constraints: int = 200,
               max_trials: int = 3) -> tuple[np.ndarray, float]:
    """
    Tiny linear‑programming refinement.
    For the current candidate ``f`` we solve
        minimise   M
        subject to  (f * g)_k ≤ M   for the most critical indices k,
                    Σ g_i = 1,  g_i ≥ 0.
    Only the top ``max_constraints`` indices (by (f*f) value) are used.
    The LP is solved with SciPy's high‑performance ``highs`` solver.
    The returned vector ``g`` is accepted only if it improves the true
    objective; otherwise the original ``f`` is kept.
    """
    if time.time() > deadline:
        return f, _objective(f)

    conv_ff = _conv_fft(f, f)
    M_current = np.max(conv_ff)

    # Indices where the convolution is close to the peak
    idxs = np.where(conv_ff >= 0.9 * M_current)[0]
    if len(idxs) > max_constraints:
        rng = np.random.default_rng()
        idxs = rng.choice(idxs, size=max_constraints, replace=False)

    # Build sparse constraint matrix for the selected indices.
    rows = []
    cols = []
    data = []
    b_ub = []

    for row, k in enumerate(idxs):
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        # loop over admissible j (the variable index)
        for j in range(start, end + 1):
            i = k - j
            rows.append(row)
            cols.append(j)
            data.append(f[i])
        # coefficient for the auxiliary variable M (last column)
        rows.append(row)
        cols.append(n)          # column n holds M
        data.append(-1.0)
        b_ub.append(0.0)

    A_ub = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(len(b_ub), n + 1))
    b_ub = np.array(b_ub, dtype=np.float64)

    # Equality: sum g_i = 1
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    # Variable bounds (g_i ≥ 0, M ≥ 0)
    bounds = [(0.0, None) for _ in range(n)] + [(0.0, None)]

    # Objective: minimise M
    c = np.concatenate([np.zeros(n), np.ones(1)])

    # Solve
    res = scipy.optimize.linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        A_eq=A_eq,
        b_eq=b_eq,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # If LP failed, simply return the original vector.
        return f, _objective(f)

    g = res.x[:n]
    M_new = res.x[n]

    # Evaluate the true objective of g (full convolution)
    obj_g = _objective(g)

    # Accept if we really improved the bound.
    if obj_g < _objective(f):
        print(f"[LP] improved objective from {_objective(f):.6f} → {obj_g:.6f}")
        return g, obj_g
    else:
        return f, _objective(f)

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # small safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length – 5000 offers a fine discretisation while staying fast
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Stage 1 – generate a broad pool of seeds and keep the best few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation of the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        # 1) Adam (smooth max)
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=120000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # 2) Sub‑gradient refinement
        seq_sub, val_sub = _subgradient_refine(global_best_seq, n, deadline, max_iters=300000)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

        # 3) Euclidean‑projection gradient descent
        seq_euc, val_euc = _euc_proj_subgradient(global_best_seq, n, deadline, max_iters=250000)
        if val_euc < global_best_val:
            global_best_seq = seq_euc.copy()
            global_best_val = val_euc
            print(f"[overall‑EucProj] new global best = {val_euc:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=500000,
            init_temp_factor=0.07
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – tiny LP refinement (applied a few times)
    # ------------------------------------------------------------------
    for trial in range(3):
        if time.time() > deadline:
            break
        print(f"\n--- LP refinement pass {trial+1} ---")
        new_seq, new_val = _lp_refine(global_best_seq, n, deadline,
                                      max_constraints=250, max_trials=3)
        if new_val < global_best_val:
            global_best_seq = new_seq.copy()
            global_best_val = new_val
            print(f"[overall‑LP] new global best = {new_val:.6f}")

    # ------------------------------------------------------------------
    # Optional re‑ordering test (descending vs ascending)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for ordering in ("desc", "asc"):
            cand = np.sort(global_best_seq)[::-1] if ordering == "desc" else np.sort(global_best_seq)
            cand_val = _objective(cand)
            if cand_val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = cand_val
                print(f"[Reorder] new global best with {ordering} = {cand_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the public evaluate_sequence
    # ------------------------------------------------------------------
    try:
        # we keep the list form for the external evaluator
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose for possible reuse in later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
