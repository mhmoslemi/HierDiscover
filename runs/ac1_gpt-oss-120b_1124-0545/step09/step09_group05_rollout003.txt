"""
Non‑negative step‑function optimiser – version 3.

Goal
----
Find a non‑negative sequence a_0,…,a_{n‑1} (n≈4000) minimising

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum a)^2 .

Since the objective is homogeneous we may restrict ourselves to
probability vectors (sum = 1); the objective then simplifies to
`2·n·max(conv(a,a))`.  The algorithm proceeds in three stages:

1. **Rich parametric seeding** – a dense collection of analytically
   defined families (uniform, triangular, geometric, power‑law,
   double‑exponential, symmetric Beta) is generated and evaluated.
   Only the best few candidates are kept.

2. **Projected sub‑gradient descent** – starting from each top seed we
   perform a mirror‑descent on the true max‑convolution (the
   sub‑gradient is non‑zero only on the active convolution index).
   After each step we project back onto the probability simplex.
   The best vector seen across all runs is retained.

3. **Simulated‑annealing polishing** – a Metropolis‑type random walk
   that repeatedly redistributes the mass of a small random subset of
   coordinates.  The temperature is slowly cooled; any improvement
   updates the current best.

All stages respect the global time budget (default 1000 s).  The final
vector is returned as a plain Python list and stored in the global
name ``height_sequence_1`` for reuse by later runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray | None = None) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    if y is None:
        y = x
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective for a probability vector (sum == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq)
    return 2.0 * n * np.max(conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero after projection – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i,   0<r<1."""
    if not (0.0 < r < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-p)
    s = a.sum()
    return a / s


def _double_geometric(n: int, r: float) -> np.ndarray:
    """Two‑sided exponential: a_i ∝ r**i + r**(n‑1‑i)."""
    if not (0.0 < r < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s


def _beta_symmetric(n: int, a_param: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) shape: a_i ∝ (i+½)^{α‑1}·(n‑½‑i)^{α‑1}.
    α>0; α<1 gives U‑shaped, α>1 gives mound‑shaped.
    """
    if a_param <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    x = i + 0.5
    y = n - 0.5 - i
    # raise to potentially negative powers safely
    a = np.power(x, a_param - 1.0) * np.power(y, a_param - 1.0)
    # replace non‑finite values (possible when α≈0) by a large number,
    # then normalise – this yields a very U‑shaped distribution.
    a[~np.isfinite(a)] = np.max(a[np.isfinite(a)]) if np.any(np.isfinite(a)) else 0.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_asymmetric(n: int, a_param: float, b_param: float) -> np.ndarray:
    """General Beta(α,β) shape, α,β>0."""
    if a_param <= 0.0 or b_param <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    x = i + 0.5
    y = n - 0.5 - i
    a = np.power(x, a_param - 1.0) * np.power(y, b_param - 1.0)
    a[~np.isfinite(a)] = np.max(a[np.isfinite(a)]) if np.any(np.isfinite(a)) else 0.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient (mirror‑descent) on the true max‑convolution.
    Keeps the best vector ever seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size – 0.6/√t works well in practice
        lr = 0.6 / np.sqrt(it)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.08,
                         min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Random redistribution of a small subset of coordinates with a
    slow exponential cooling schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature based on current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling (very gentle)
        T = max(T * 0.9999, min_temp)

        # pick a random subset of 5‑10 indices and reshuffle their total mass
        k = rng.integers(5, 11)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Seed generation (analytical families + previous solution)
# ----------------------------------------------------------------------


def _generate_seed_pool(n: int,
                        rng: np.random.Generator,
                        previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse collection of normalised candidate vectors.
    """
    seeds: list[np.ndarray] = []

    # classic deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # geometric and power‑law sweeps
    for r in np.linspace(0.70, 0.99, 13):
        seeds.append(_geom_decrease(n, r))
    for p in np.arange(0.3, 3.01, 0.3):
        seeds.append(_power_decrease(n, p))

    # double‑sided exponential (two‑ended) family
    for r in np.linspace(0.55, 0.95, 13):
        seeds.append(_double_geometric(n, r))

    # symmetric Beta family (α = β)
    for a in np.linspace(0.15, 2.0, 20):
        seeds.append(_beta_symmetric(n, a))

    # a few asymmetric Beta draws (randomly sampled)
    for _ in range(12):
        a = rng.uniform(0.15, 2.0)
        b = rng.uniform(0.15, 2.0)
        seeds.append(_beta_asymmetric(n, a, b))

    # include the previous best if compatible
    if previous is not None and previous.shape == (n,):
        seq = np.asarray(previous, dtype=np.float64)
        if seq.sum() > 0.0:
            seeds.append(seq / seq.sum())

    # random Dirichlet draws to add stochastic variety
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(2):
            a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
            seeds.append(a / a.sum())

    return seeds

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` value.
    Returns the best sequence found as a plain list of floats.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # ------------------------------------------------------------------
    # Length of the search vector – 4000 works well for the bound we target
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool of analytical seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seed_pool(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    # ------------------------------------------------------------------
    # Evaluate all seeds with the cheap surrogate objective
    # ------------------------------------------------------------------
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    top_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Top {top_k} seed objectives (surrogate):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    # keep the best ever‑seen vector
    global_best_seq = top_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy local optimisation from each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Sub‑gradient optimisation run {idx + 1}/{top_k} ---")
        seq_opt, val_opt = _subgradient_optimize(
            seed_vec, n, deadline, max_iters=150_000
        )
        if val_opt < global_best_val:
            global_best_seq = seq_opt.copy()
            global_best_val = val_opt
            print(f"[overall‑SubGrad] new global best = {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=600_000,
            init_temp_factor=0.08
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the real evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n>>> True evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
