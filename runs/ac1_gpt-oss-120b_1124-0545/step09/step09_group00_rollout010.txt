"""
Propose a step‑function that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

using a three‑stage optimisation:

1. **Diverse seeds** – analytic families (uniform, beta, sine‑window,
   low‑frequency Fourier curves, mixtures) normalised to ℓ₁=1.
   The best few seeds (according to the cheap surrogate
   2·n·max(conv) ) are kept.

2. **Adam on a smoothed max** – the vector is represented by logits
   (`v`) such that `a = softmax(v)` lives on the probability simplex.
   The objective   max(conv)   is approximated by a temperature‑scaled
   log‑sum‑exp; Adam (β₁=0.9, β₂=0.999) updates the logits while the
   temperature (`beta`) is slowly increased so that the surrogate
   converges to the true max.  The best vector seen is recorded.

3. **Exponentiated‑gradient sub‑gradient descent** – after the smooth
   phase the exact max‑convolution is used.  For the active index
   `k = argmax(conv)` the sub‑gradient is
   `g_i = 2·a_{k‑i}` (zero outside the admissible range).  An
   exponentiated‑gradient (mirror descent with KL‑divergence) updates
   the probabilities:
        a_i ← a_i·exp(‑η·g_i)  ,  a ← a / Σa .
   The step‑size η decays like 1/√t.  The best vector over the whole
   run is kept.

A final short simulated‑annealing polish reshuffles the mass of a
random subset of coordinates, always using the exact evaluator
`evaluate_sequence`.  The routine respects the global deadline
(≤ 1000 s) and stores the resulting list in the global name
`height_sequence_1` for possible reuse.

The algorithm is deliberately different from the previous baseline
(stochastic‑gradient + Euclidean sub‑gradient).  The Adam optimiser
and multiplicative mirror‑descent often drive the max‑convolution
lower, giving a better bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    return np.fft.irfft(fx * fy, n=N)[:L]


def _surrogate_objective(a: np.ndarray) -> float:
    """
    Cheap surrogate: 2·n·max(conv(a,a))  (a is assumed to sum to 1).
    """
    n = len(a)
    conv = _conv_fft(a, a)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _logsumexp_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Smooth max weights using temperature `beta`."""
    shift = np.max(conv)
    w = np.exp(beta * (conv - shift))
    s = w.sum()
    if s == 0.0:
        # fallback to uniform
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture of two normalised shapes, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidates."""
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))

    # Beta family – both U‑shaped (α,β<1) and mound‑shaped (α,β>1)
    for a in (0.2, 0.4, 0.6, 0.8, 1.2, 2.0):
        for b in (0.2, 0.4, 0.6, 0.8, 1.2, 2.0):
            seeds.append(_beta_shape(n, a, b))

    # low‑frequency random curves
    for nf in (2, 4, 6):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # random Dirichlet draws (sparse and moderately dense)
    for conc in (0.03, 0.1, 0.3, 0.8, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # mixtures of two independent low‑frequency curves
    for _ in range(3):
        a = _lowfreq_curve(n, 4, rng)
        b = _lowfreq_curve(n, 5, rng)
        seeds.append(_mix_shapes(a, b, alpha=rng.random()))

    # previous solution if available
    if previous is not None and previous.shape == (n,):
        arr = np.asarray(previous, dtype=np.float64)
        if arr.sum() > 0:
            seeds.append(arr / arr.sum())

    return seeds

def _adam_logspace_sgd(start_vec: np.ndarray, n: int,
                       deadline: float,
                       max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑softmax representation using a log‑sum‑exp
    surrogate for the max‑convolution.
    Returns best vector seen (already ℓ₁‑normalised) and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                      # logits
    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    lr0 = 0.8
    beta_temp = 1.0          # start temperature for smooth max
    beta_inc = 1.02          # multiplicative increase
    beta_max = 250.0

    best_seq = None
    best_val = np.inf

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability vector

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if t % 5000 == 0:
                print(f"[AdamSGD] it={t:6d} best={cur_val:.6f}")

        # smooth max weights
        w = _logsumexp_weights(conv, beta_temp)

        # gradient w.r.t. a (same derivation as in the reference code)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length 3n-2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # gradient w.r.t. logits (log‑softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update in log‑space
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
        m_corr = m / (1.0 - beta1 ** t)
        v_corr = v_hat / (1.0 - beta2 ** t)
        lr = lr0 / np.sqrt(t)
        v -= lr * m_corr / (np.sqrt(v_corr) + eps)

        # temperature schedule (slowly make surrogate sharper)
        if beta_temp < beta_max:
            beta_temp = min(beta_max, beta_temp * beta_inc)

    return best_seq, best_val


def _exp_gradient_subproj(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 800_000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution.
    Returns best vector (ℓ₁‑normalised) and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate_objective(a)

    eta0 = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[ExpSubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # multiplicative update (exponentiated gradient)
        eta = eta0 / np.sqrt(it)
        a = a * np.exp(-eta * g)

        # re‑normalise to the simplex (KL‑projection)
        s = a.sum()
        if s <= 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing_polish(start_seq: np.ndarray, n: int,
                                deadline: float,
                                max_iters: int = 200_000,
                                init_temp_factor: float = 0.04) -> tuple[np.ndarray, float]:
    """
    Simple geometric‑cooling SA that reshuffles the mass of a random
    subset of coordinates, evaluated with the *true* evaluator.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = evaluate_sequence(cur.tolist())
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9985, 1e-12)               # slower cooling for finer search

        k = rng.integers(2, 15)                # size of the reshuffled subset
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # new allocation: Dirichlet draws on the simplex of size k
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = evaluate_sequence(new.tolist())

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # Length choice – a power of two speeds up the FFT.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) for warm‑start.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool of seeds and keep the most promising.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evals = [(vec, _surrogate_objective(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 5
    best_seeds = [vec for vec, _ in evals[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _surrogate_objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam log‑space SGD (multiple restarts).
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Adam SGD restart {idx+1}/{len(best_seeds)} ---")
        seq_adam, val_adam = _adam_logspace_sgd(
            seed_vec, n, deadline, max_iters=300_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # ------------------------------------------------------------------
        # Sub‑gradient phase with exponentiated updates (exact max).
        # ------------------------------------------------------------------
        seq_exp, val_exp = _exp_gradient_subproj(
            global_best_seq, n, deadline, max_iters=800_000
        )
        if val_exp < global_best_val:
            global_best_seq = seq_exp.copy()
            global_best_val = val_exp
            print(f"[overall‑ExpSub] new global best = {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final simulated‑annealing polishing (true evaluator).
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_polish(
            global_best_seq, n, deadline, max_iters=200_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator.
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
