    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _surrogate(seq: np.ndarray) -> float:
    """
    Cheap surrogate used during the search:
    ``2·n·max(conv(seq,seq))`` (seq is already normalised to sum = 1).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum=1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    return a / a.sum()


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    return a / a.sum()


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    return a / a.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    return a / a.sum()


def _spaced_uniform(n: int, k: int) -> np.ndarray:
    """k points equally spaced across the array (mass 1/k each)."""
    if k <= 0:
        return _uniform(n)
    indices = np.round(np.linspace(0, n - 1, k)).astype(int)
    a = np.zeros(n, dtype=np.float64)
    a[indices] = 1.0 / k
    return a


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – modest grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric – dense log‑grid near 1
    for r in np.geomspace(0.90, 0.999999, 20):
        seeds.append(_geom_decay(n, r))

    # power‑law decay
    for p in np.arange(0.1, 5.01, 0.3):
        seeds.append(_power_decay(n, p))

    # beta family – fine grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # spaced‑uniform patterns (varying sparsity)
    for k in (8, 16, 32, 64, 128, 256):
        seeds.append(_spaced_uniform(n, k))

    # random Dirichlet draws – diverse concentrations
    for conc in (0.01, 0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns – very small support
    for k in (3, 5, 10, 20, 40):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the true max‑conv objective
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 3.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent.
    Returns the best sequence observed during the run and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _surrogate(a)

    eps = 1e-30
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # current exact objective (used only for logging)
        cur_val = 2.0 * n * conv[max_idx]

        # if surrogate improves, remember it
        surrogate = _surrogate(a)
        if surrogate < best_val:
            best_val = surrogate
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:,d} best≈{best_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numerical stabilisation
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)  # 2 … 6 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _surrogate(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA]  it={it:,d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # Reproducibility
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a safety margin for the judge
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Warm‑start from previous run (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length that is a power of two (FFT friendly)
    # ------------------------------------------------------------------
    n = 4096  # 2**12 – fast FFT and enough resolution

    # ------------------------------------------------------------------
    # Stage 1 – generate many diverse seeds and keep the elite few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _surrogate(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    elite = [vec for vec, _ in evaluated[:top_k]]

    print(f"Top {top_k} seeds (surrogate values):")
    for i, (_, v) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {v:.6f}")

    global_best_seq = elite[0].copy()
    global_best_val = _surrogate(global_best_seq)
    print(f"\n[Init] best elite surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – mirror‑descent on each elite seed, time‑slice equally
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    per_seed_budget = remaining / (top_k + 0.5)  # a bit extra for polishing

    for idx, seed_vec in enumerate(elite):
        sub_deadline = time.time() + per_seed_budget
        print(f"\n--- Mirror‑descent on elite #{idx+1} (budget≈{per_seed_budget:.1f}s) ---")
        refined_seq, refined_val = _subgradient_refine(
            seed_vec, n, sub_deadline, max_iters=5_000_000, lr0=3.0
        )
        if refined_val < global_best_val:
            global_best_seq = refined_seq
            global_best_val = refined_val
            print(f"[Overall‑SubGrad] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        polished_seq, polished_val = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=300_000
        )
        if polished_val < global_best_val:
            global_best_seq = polished_seq
            global_best_val = polished_val
            print(f"[Overall‑SA] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional up‑sampling to double resolution
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n_big = n * 2
        print(f"\n--- Upsampling to n={n_big} and refining ---")
        x_small = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_small, global_best_seq)
        upsampled = upsampled / upsampled.sum()
        # give the up‑sampled version a fresh time slice
        up_deadline = deadline
        refined_big, val_big = _subgradient_refine(
            upsampled, n_big, up_deadline, max_iters=1_000_000, lr0=2.5
        )
        if val_big < global_best_val:
            global_best_seq = refined_big
            global_best_val = val_big
            n = n_big
            print(f"[Upsample‑Refine] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose for possible warm‑start in a later invocation
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
