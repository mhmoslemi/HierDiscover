def _project_simplex(v):
    """Return projection of v onto simplex {x>=0, sum=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Non‑negative step‑function optimiser – smooth‑max gradient descent.

The optimiser works in three stages.

1️⃣  **Seed creation** – a diverse collection of analytic families
    (uniform, triangular, sine‑window, symmetric/unsymmetric geometric,
    power‑law, beta) together with a few random Dirichlet draws.  All
    candidates are normalised to sum = 1 and evaluated with the exact
    objective  ``2·n·max(conv(a,a))`` (the same value returned by
    ``evaluate_sequence`` for a normalised vector).  The top‑k seeds
    are retained.

2️⃣  **Smooth‑max projected gradient descent** – starting from each kept
    seed we minimise the true objective using a differentiable
    approximation of the ``max``:
        M_α(a) = (1/α)·log(∑_k exp(α·c_k)) ,   c = conv(a,a) .
    For a given α the gradient is computed analytically
    (via a second convolution) and a projected gradient step onto the
    probability simplex is performed.  We run three continuation phases
    with increasing α (5 → 20 → 80) and a learning‑rate that decays as
    1/√t.  The best vector seen in any phase is kept.

3️⃣  **Optional polishing** – a short exponentiated‑gradient (mirror‑descent)
    sub‑gradient pass and a tiny simulated‑annealing run are applied
    if time permits.

The whole routine respects the 1000 s budget, periodically checks the
deadline and finally stores the best vector in the global name
``height_sequence_1`` so that a later run can reuse it.

The returned list can be fed directly to ``evaluate_sequence``.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact surrogate used during the search.
    ``seq`` is assumed to be non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))`` – the same value that
    ``evaluate_sequence`` will return for a normalised vector.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D vector")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – coarse grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric decay (very shallow)
    for r in np.linspace(0.985, 0.9999, 12):
        seeds.append(_geom_decay(n, r))

    # power‑law decay
    for p in np.arange(0.1, 5.01, 0.3):
        seeds.append(_power_decay(n, p))

    # beta family – coarse grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform and symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.90, 7):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 4):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws – a handful for diversity
    for conc in (0.02, 0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns – very small support
    for k in (3, 5, 10):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent
# ----------------------------------------------------------------------


def _smooth_max_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_total_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise 2·n·max(conv) using a smooth approximation of the max.
    The algorithm runs three continuation phases with increasing
    smoothing parameter ``alpha`` and a decaying learning rate.
    Returns the best sequence seen and its exact objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # continuation schedule: (alpha, initial_lr, max_iters_in_phase)
    schedule = [
        (5.0, 0.8, int(0.4 * max_total_iters)),
        (20.0, 0.4, int(0.4 * max_total_iters)),
        (80.0, 0.2, int(0.2 * max_total_iters)),
    ]

    it = 0
    for alpha, lr0, phase_lim in schedule:
        for _ in range(phase_lim):
            if time.time() > deadline:
                break
            it += 1

            # ----- convolution and soft‑max weights -----
            conv = _conv_fft(a, a)                 # length 2·n‑1
            mx = conv.max()
            if mx < 1e-30:      # degenerate – should not happen
                break
            # scale to avoid numerical issues (max term gets weight 1)
            w = np.exp(alpha * (conv / mx - 1.0))
            w_sum = w.sum()
            if w_sum == 0.0:
                break
            w /= w_sum                               # now Σ w = 1

            # ----- gradient of the smooth max -----
            a_rev = a[::-1]
            tmp = _conv_fft(a_rev, w)                # length n + (2n‑1) – 1
            grad = tmp[n - 1 : n - 1 + n] + a         # (a * w) + a

            # ----- projected gradient step -----
            lr = lr0 / np.sqrt(it)
            a_new = a - lr * grad
            a = _project_simplex(a_new)

            # ----- keep best -----
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                # occasional progress report
                if it % 10000 == 0:
                    print(f"[Smooth] it={it:8d} α={alpha:.0f} best≈{best_val:.6f}")

            # time guard
            if time.time() > deadline:
                break

        if time.time() > deadline:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient refinement (same as earlier code)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    objective.  Stops when the deadline is reached.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # exact objective (for bookkeeping, not used for the step)
        cur_val = 2.0 * n * conv[max_idx]

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:9d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a small random subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # size of the random subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA]      it={it:8d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` value.
    Returns the best sequence found as a plain ``list[float]``.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Possibly reuse a previous solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a sensible length (4000 is a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse pool of seeds and keep the best few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    scored = [(vec, _objective(vec)) for vec in seeds]
    scored.sort(key=lambda x: x[1])
    top_k = min(4, len(scored))
    best_seeds = [vec for vec, _ in scored[:top_k]]

    print(f"Top {top_k} seed objectives (exact):")
    for i, (_, val) in enumerate(scored[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max projected gradient descent from each top seed
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Smooth‑max descent from seed #{idx+1} ---")
        seq, val = _smooth_max_descent(seed_vec, n, deadline, max_total_iters=400_000)
        if val < global_best_val:
            global_best_seq, global_best_val = seq.copy(), val
            print(f"[global] new best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – (optional) fine‑grained exponentiated‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient (exponentiated‑gradient) refinement ---")
        seq_refine, val_refine = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=2_000_000, lr0=2.0
        )
        if val_refine < global_best_val:
            global_best_seq, global_best_val = seq_refine.copy(), val_refine
            print(f"[global] after sub‑grad best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – brief simulated‑annealing polish (if we still have time)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=150_000
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa.copy(), val_sa
            print(f"[global] after SA best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a larger length (if time permits)
    # ------------------------------------------------------------------
    if time.time() + 30 < deadline:
        n_big = n * 2
        x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
        x_big = np.linspace(0.0, 1.0, n_big, dtype=np.float64)
        upsampled = np.interp(x_big, x_old, global_best_seq)
        upsampled = upsampled / upsampled.sum()
        print(f"\n--- Upsampling to n={n_big} and smooth‑max descent ---")
        seq_big, val_big = _smooth_max_descent(
            upsampled, n_big, deadline, max_total_iters=250_000
        )
        if val_big < global_best_val:
            global_best_seq, global_best_val = seq_big.copy(), val_big
            n = n_big
            print(f"[Upsample] new best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (prevent accidental NaNs / infinities)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
