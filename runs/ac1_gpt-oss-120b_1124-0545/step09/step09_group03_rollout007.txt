"""
Propose a step‑function with a (hopefully) lower ``evaluate_sequence`` value.

The search proceeds in several stages, each respecting the global time budget
(~1000 s):

1. **Deterministic analytic seeds** – several families (uniform, Beta,
   Gaussian, Hann, etc.) for a range of lengths.
2. **Heavy random sampling** – Dirichlet draws and random Beta shapes.
3. **Continuous Beta‑parameter optimisation** – a cheap two‑parameter search
   (SciPy / Nelder‑Mead) on a medium length.
4. **Smoothed Adam refinement** – start from the best candidates (including
   up‑sampled versions) and run Adam on a log‑sum‑exp smooth approximation of
   ``max(conv(a,a))`` with a slowly increasing temperature.
5. **Aggressive KL‑mirror‑descent polishing** – final sub‑gradient descent
   that aggregates all (almost) active convolution entries.

The algorithm also tries to *up‑sample* the current best sequence to larger
sizes (up to 12 000 entries); a finer discretisation reduces the
discretisation error of the underlying continuous problem and typically
lowers the constant factor in the objective.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, pi, log, exp

# ---------------------------------------------------------------
# Helper utilities (FFT‑based full convolution)
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    ``evaluate_sequence`` for a *normalised* sequence (sum == 1).

    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())


# ---------------------------------------------------------------
# Analytic families (always returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = w.sum()
    return w / s if s else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    s = w.sum()
    return w / s if s else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta density on the discrete grid (shifted by 0.5 to avoid the exact ends)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    s = w.sum()
    return w / s if s else _uniform(n)


# ---------------------------------------------------------------
# Random Dirichlet sampling
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Return a normalised Dirichlet sample (α parameter = alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s else _uniform(n)


# ---------------------------------------------------------------
# Interpolation (up‑sampling / down‑sampling)
# ---------------------------------------------------------------

def _interpolate_sequence(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linearly interpolate a probability vector to a new length."""
    if new_len == len(seq):
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, len(seq), endpoint=False) + 0.5 / len(seq)
    new_x = np.linspace(0.0, 1.0, new_len, endpoint=False) + 0.5 / new_len
    new_y = np.interp(new_x, old_x, seq)
    new_y = np.maximum(new_y, 0.0)
    new_y /= new_y.sum() if new_y.sum() else 1.0
    return new_y


# ---------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parametrisation)
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 300.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑softmax parametrisation, minimising a smooth
    approximation of ``max(conv(a,a))`` via log‑sum‑exp.
    """
    n = len(start_seq)
    eps = 1e-30

    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)

        # temperature schedule (linear)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()          # soft‑max weights over convolution entries

        # gradient w.r.t. a (see derivation in the original code)
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:       # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            # early stop if we are already very good
            if best_val < 1.45:
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Aggressive KL‑mirror‑descent polishing (aggregated active set)
# ---------------------------------------------------------------

def _refine_aggregated_subgradient(start_seq: np.ndarray, deadline: float,
                                   eps_rel: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* convolution
    entries whose value is within a relative tolerance ``eps_rel`` of the
    current maximum.  Returns the best sequence found before ``deadline``.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        # indices that are “almost maximal”
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])          # length = 3n‑2
        g = 2.0 * conv_ag[n - 1:n - 1 + n]            # slice = n

        # KL‑mirror step (exponentiated gradient)
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)        # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:         # periodic exact check
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")

            # very early exit if almost out of time
            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val


# ---------------------------------------------------------------
# Optional L‑BFGS refinement (smooth max)
# ---------------------------------------------------------------

try:
    from scipy.optimize import minimize
except Exception:   # pragma: no cover
    minimize = None


def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 200.0) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B on the same log‑sum‑exp smooth objective.
    Returns the best exact (unsmoothed) objective value.
    """
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a /= a.sum()
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        exps = np.exp(beta * (conv - max_c))
        lse = max_c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOut(RuntimeError):
        pass

    def callback(_):
        if time.time() >= deadline:
            raise TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'disp': False},
            callback=callback,
        )
    except TimeOut:
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    return best_a, best_val

# Main entry point
# ---------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Possibly reuse a previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths we will explore (all FFT‑friendly, up to ~12k)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 6000, 8000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic families
    # ------------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – heavy random sampling (Dirichlet + Beta)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.35
    sample_deadline = now + time_for_sampling
    alphas = [0.5, 1.0, 2.0]
    beta_params = [(0.3, 0.3), (0.5, 0.5), (1.0, 1.0), (2.0, 2.0),
                   (0.5, 2.0), (2.0, 0.5)]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rand‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta shape
        a_par, b_par = rng.choice(beta_params)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rand‑Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – continuous Beta‑parameter optimisation (if SciPy available)
    # ------------------------------------------------------------------
    if minimize is not None:
        # Choose a medium length that is still cheap to evaluate
        opt_len = 6000
        def beta_obj(params):
            log_alpha, log_beta = params
            alpha = np.exp(log_alpha)
            beta_par = np.exp(log_beta)
            seq = _beta_shape(opt_len, alpha, beta_par)
            return _objective(seq)

        best_beta_val = float('inf')
        best_beta_params = None
        # a few simple initialisations
        inits = [(0.4, 0.4), (0.8, 0.8), (1.2, 1.2), (2.0, 2.0), (0.5, 2.0), (2.0, 0.5)]
        for a0, b0 in inits:
            x0 = np.log([a0, b0])
            try:
                res = minimize(beta_obj, x0, method='Nelder-Mead',
                               options={'maxiter': 250, 'disp': False})
            except Exception:
                continue
            if res.fun < best_beta_val:
                best_beta_val = res.fun
                best_beta_params = np.exp(res.x)

        if best_beta_params is not None:
            seq = _beta_shape(opt_len,
                              float(best_beta_params[0]),
                              float(best_beta_params[1]))
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Beta‑Opt] n={opt_len} α={best_beta_params[0]:.4f} β={best_beta_params[1]:.4f} val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Smoothed Adam refinement (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Build a small pool of candidates: current best and several up‑sampled versions
        candidate_pool = [best_seq]
        for target_len in [8000, 12000]:
            if target_len > len(best_seq):
                up = _interpolate_sequence(best_seq, target_len)
                candidate_pool.append(up)

        # Allocate about 25 % of the remaining budget to Adam runs
        remaining = deadline - time.time()
        if remaining > 5.0:
            adam_deadline = time.time() + remaining * 0.25
            for idx, cand in enumerate(candidate_pool):
                seq_adam, val_adam = _smooth_adam_optimize(
                    start_seq=cand,
                    deadline=adam_deadline,
                    beta_start=5.0,
                    beta_end=400.0,
                    max_iter=4_500_000,
                    lr0=0.04,
                )
                if val_adam < best_val:
                    best_val = val_adam
                    best_seq = seq_adam
                    print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – optional L‑BFGS refinement (smooth objective, if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0 and minimize is not None:
            lbfgs_deadline = time.time() + remaining * 0.15
            seq_lb, val_lb = _lbfgs_refine(best_seq, lbfgs_deadline, beta=300.0)
            if val_lb < best_val:
                best_val = val_lb
                best_seq = seq_lb
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – aggressive mirror‑descent polishing (final)
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_aggregated_subgradient(best_seq, deadline,
                                                             eps_rel=5e-7)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    # return as plain Python list
    return best_seq.tolist()
