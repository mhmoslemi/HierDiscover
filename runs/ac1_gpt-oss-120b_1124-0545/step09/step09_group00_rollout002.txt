        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

"""
Propose a non‑negative step function that (approximately) minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm works in three stages:

1. **Diverse seed pool** – analytic families (uniform, triangular,
   cosine, Gaussian, power‑law, beta, Tukey), random Dirichlet/exponential
   draws and low‑frequency random curves.  The previous best solution is
   interpolated to the current length and added to the pool.

2. **Log‑space Adam optimisation** – the vector is represented by logits
   (log‑softmax).  A smoothed max (log‑sum‑exp) surrogate of the convolution
   peak is used; the gradient is computed via one FFT per iteration.
   Adam’s adaptive steps together with a slowly increasing temperature give
   a fast descent on the surrogate.

3. **Exact mirror‑descent (KL‑projection) on the true max‑convolution**
   – after the surrogate phase we switch to the true objective.
   The sub‑gradient of the max‑convolution is obtained analytically,
   and an exponentiated‑gradient (mirror descent on the simplex) updates
   the probabilities.

4. **Stochastic annealing polish** – random redistributions of the mass
   on a small random subset of coordinates are accepted according to a
   geometric cooling schedule.  The best sequence found throughout all
   stages is returned.

All convolutions are performed with a single FFT (self‑convolution),
making even tens of thousands of iterations cheap.  The routine respects
the 1000 s deadline, and stores the best vector in the global name
``height_sequence_1`` for possible reuse.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _self_conv_fft(a: np.ndarray) -> np.ndarray:
    """Self‑convolution a*a using a single FFT."""
    L = len(a) * 2 - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact objective (sum is forced to 1)."""
    n = len(seq)
    conv = _self_conv_fft(seq)
    return float(2.0 * n * np.max(conv))


def _resample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = len(seq)
    if old_n == new_n:
        return np.asarray(seq, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s == 0.0:
        return np.full(new_n, 1.0 / new_n, dtype=np.float64)
    return new / s

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve built from a few cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the logits (log‑softmax) using a log‑sum‑exp
    surrogate for the max‑convolution.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    beta = 1.0              # temperature for soft‑max smoothing
    beta_inc = 0.6
    beta_max = 250.0

    lr0 = 0.6               # base learning rate

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current probability vector
        conv = _self_conv_fft(a)
        cur_val = 2.0 * n * np.max(conv)    # exact objective (since sum=1)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogAdam] it={it:6d} best={cur_val:.6f}")

        # ---------- smoothed max (log‑sum‑exp) ----------
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k  w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # convert to logits gradient (softmax Jacobian)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v2_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _mirror_descent_kl(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective,
    using KL‑mirror updates (exponentiated gradient on the simplex).
    """
    # initialise log‑probabilities
    loga = np.log(start_seq + 1e-30)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(loga)          # probabilities, sum to 1
        conv = _self_conv_fft(a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20_000 == 0:
                print(f"[KL‑Descent] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max‑convolution at the active index
        g = np.zeros_like(a)
        i_start = max_idx - (n - 1)
        if i_start < 0:
            i_start = 0
        i_end = max_idx if max_idx < n else n - 1
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]      # each pair counted twice

        # step size (decreasing)
        eta = 0.35 / np.sqrt(it)

        # KL‑mirror step: loga ← loga – η·g   (softmax later enforces simplex)
        loga = loga - eta * g

    return best_seq, best_val


def _simulated_annealing_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.025,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that redistributes the mass of a small random subset.
    Acceptance follows the Metropolis rule with a geometric cooling schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, 1e-12)

        # pick a small random subset
        k = rng.integers(2, 15)               # subset size 2 … 14
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # propose a new allocation on this subset (Dirichlet with low concentration)
        new_alloc = rng.dirichlet(np.full(k, 0.5)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function within the given wall‑clock budget.  Returns the best
    sequence as a plain list of floats and stores it in the global name
    ``height_sequence_1`` for possible reuse.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0      # safety margin

    # ------------------------------------------------------------------
    # Choose a length – more points give a better asymptotic bound.
    # ------------------------------------------------------------------
    n = 8000                      # feel free to adjust (kept ≤ 1 e4 for speed)

    # ------------------------------------------------------------------
    # Load previous solution (if present) and up‑sample it.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich pool of seeds.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n = {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # Evaluate surrogate (exact objective – cheap because all are ℓ₁‑normalised)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 7
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – log‑space Adam optimisation (smooth surrogate).
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Log‑Adam run {idx+1}/{len(best_seeds)} ---")
        seq_sgd, val_sgd = _logspace_adam_optimize(
            seed_vec, n, deadline, max_iters=200_000
        )
        if seq_sgd is not None and val_sgd < global_best_val:
            global_best_seq = seq_sgd.copy()
            global_best_val = val_sgd
            print(f"[overall‑Adam] new global best = {val_sgd:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact KL‑mirror descent on the true max‑convolution.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Exact KL‑mirror descent ---")
        seq_kl, val_kl = _mirror_descent_kl(
            global_best_seq, n, deadline, max_iters=300_000
        )
        if val_kl < global_best_val:
            global_best_seq = seq_kl.copy()
            global_best_val = val_kl
            print(f"[overall‑KL] new global best = {val_kl:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – stochastic annealing polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing_refine(
            global_best_seq, n, deadline, max_iters=300_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
