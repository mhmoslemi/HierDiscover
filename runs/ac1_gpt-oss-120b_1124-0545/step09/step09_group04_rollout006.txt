def smooth_objective_and_grad(a, beta):
    conv = _conv_fft(a, a)  # length 2n-1
    scaled = beta * conv
    # compute max via logsumexp
    max_weight = np.exp(scaled - np.max(scaled)) # to avoid overflow
    Z = max_weight.sum()
    softmax = max_weight / Z  # weights w_k

    # gradient: g = 2 * correlate(a, softmax) (with appropriate offset)
    # softmax length = 2n-1, compute reversed:
    w_rev = softmax[::-1]
    # convolve a and w_rev via FFT
    conv2 = _conv_fft(a, w_rev)  # length 3n-2
    # get g values for j=0..n-1: conv2[(len(w)-1) - j] where len(w)=2n-1
    L = len(softmax)
    # compute indices
    idx = (L - 1) - np.arange(len(a))
    g = 2.0 * conv2[idx]
    # final objective = (1/beta) * logsumexp(scaled) = (1/beta)*(np.log(Z)+max_scaled)
    # Actually logsumexp = max_scaled + log(sum(exp(beta*conv-max_scaled)))
    m = np.max(scaled)
    logz = m + np.log(np.sum(np.exp(scaled - m)))
    obj = logz / beta
    return obj, g

def _project_simplex(v):
    # ensure non-negative sum to 1
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback uniform
        return np.full(n, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # Efficient projection onto simplex (nonnegative, sum=1)
    n = len(v)
    # shift v to have sum>0? Not needed; algorithm handles.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho
    rho_arr = u - cssv / ind
    rho = np.where(rho_arr > 0)[0]
    if len(rho) == 0:
        # all <=0, return uniform
        return np.full(n, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _smooth_gradient_descent(start_seq, n, deadline, lr0=0.5, beta0=0.5, beta_max=2000.0, beta_inc=1.001, max_iters=5_000_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_obj, _ = _smooth_obj_and_grad(a, beta_max)  # we could compute with large beta to approximate max
    # Actually compute exact max via _objective maybe easier.
    best_val = _objective(a)  # approximate.

    it = 0
    beta = beta0
    while time.time() < deadline and it < max_iters:
        it += 1
        obj, grad = _smooth_obj_and_grad(a, beta)
        # Use gradient descent step
        lr = lr0 / np.sqrt(it)
        a_new = a - lr * grad
        # project onto simplex
        a = _project_simplex(a_new)
        # Evaluate exact objective occasionally
        if it % 1000 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Grad] it={it} best≈{best_val:.6f}")
        # increase beta
        beta = min(beta_max, beta * beta_inc)
    # after loop, compute best final value
    return best_seq, best_val

"""
Non‑negative step‑function optimiser – hybrid of analytic seeds,
deterministic smoothed‑gradient descent (soft‑max annealing) and a
final mirror‑descent polish.

The search proceeds in three phases:

1️⃣  **Seed generation** – a large, cheap collection of normalised
    families (uniform, triangular, sine, symmetric/unsymmetric
    geometric, power‑law, beta, random Dirichlet/sparse).  The best
    few seeds (by the exact max‑convolution objective) are retained.

2️⃣  **Smooth‑gradient annealing** – starting from the current global
    best we minimise the *soft‑max* surrogate
        φβ(a)=log∑exp(β·(a*a))/β
    using projected gradient descent on the probability simplex.
    β is increased slowly (β←min(βmax,β·βinc)) so the optimisation
    gradually focusses on the true maximum.
    The exact objective 2·n·max(a*a) is monitored and the best
    vector is saved.

3️⃣  **Mirror‑descent polish** – a short exponentiated‑gradient
    sub‑gradient loop (as in the reference implementation) is applied
    to the best candidate found in phase 2, yielding a final fine‑tuned
    result.

If time permits the final sequence is up‑sampled (linear interpolation)
to double the length and refined once more.

The routine stores the result in the global name ``height_sequence_1``
so that a later run can start from it.
"""

import time
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)), seq assumed to sum to 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta_: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta_ <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta_ - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> List[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: List[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric decay – grid (very slow decay)
    for r in np.linspace(0.985, 0.9999, 12):
        seeds.append(_geom_decay(n, r))

    # pure power‑law decay
    for p in np.arange(0.1, 5.01, 0.3):
        seeds.append(_power_decay(n, p))

    # beta family – grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform and symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.90, 7):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 4):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (3, 5, 10):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible)
    if previous is not None and previous.shape == (n,):
        prev = np.asarray(previous, dtype=np.float64)
        if prev.sum() > 0:
            seeds.append(prev / prev.sum())

    return seeds


# ----------------------------------------------------------------------
# Projection onto probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the simplex {x >= 0, sum x = 1}.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpinán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = ind[cond][-1]               # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Smoothed objective (soft‑max) and its gradient
# ----------------------------------------------------------------------


def _smooth_obj_and_grad(a: np.ndarray, beta: float) -> Tuple[float, np.ndarray]:
    """
    Returns (φβ(a), ∇φβ(a)) where
        φβ(a) = (1/β)·log∑_k exp(β·conv_k(a,a))
    and the gradient is the soft‑max weighted correlation.
    """
    n = a.shape[0]

    # 1. convolution a*a
    conv = _conv_fft(a, a)                # length 2n‑1

    # 2. soft‑max weights (stable)
    scaled = beta * conv
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    Z = exps.sum()
    w = exps / Z                         # shape (2n‑1,)

    # objective value
    obj = (m + np.log(Z)) / beta

    # 3. gradient = 2 * correlation(a, w)
    w_rev = w[::-1]                      # reverse for correlation via convolution
    conv2 = _conv_fft(a, w_rev)          # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)     # pick the required slice
    grad = 2.0 * conv2[idx]

    return obj, grad


# ----------------------------------------------------------------------
# Smoothed gradient descent (annealing)
# ----------------------------------------------------------------------


def _smooth_gradient_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    lr0: float = 0.5,
    beta0: float = 0.5,
    beta_max: float = 2000.0,
    beta_inc: float = 1.001,
    max_iters: int = 5_000_000,
) -> Tuple[np.ndarray, float]:
    """
    Projected gradient descent on the soft‑max surrogate.
    The temperature β grows multiplicatively (β←β·β_inc) up to βmax.
    Returns the best sequence discovered (exact objective evaluated).
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)   # exact max‑conv objective

    beta = beta0
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # evaluate smooth objective & gradient
        smooth_obj, grad = _smooth_obj_and_grad(a, beta)

        # step size schedule
        lr = lr0 / np.sqrt(it)

        # gradient step + projection onto simplex
        a = _project_simplex(a - lr * grad)

        # increase β
        beta = min(beta_max, beta * beta_inc)

        # occasional exact evaluation
        if it % 2000 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[SmoothGD] it={it:7d} β={beta:.1f} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) – same as reference, used for polish
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 10_000_000,
    lr0: float = 2.0,
) -> Tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent directly on max(conv).
    Keeps the best vector seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the maximal entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponential update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        # keep best (checking only periodically to save work)
        if it % 200_000 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[SubGrad] it={it:9d} best≈{best_val:.6f}")

    # final check
    final_val = _objective(a)
    if final_val < best_val:
        best_val = final_val
        best_seq = a.copy()
    return best_seq, best_val
