def project_simplex(v):
    """Projects onto simplex: sum=1, v>=0."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Smooth‐gradient optimisation for the step‑function bound.

The objective 2·n·max(conv(a,a)) / (∑a)² is equivalent (for ∑a=1) to
minimising the maximum auto‑convolution value.
We replace the non‑smooth max by a temperature‑controlled softmax:
    F_T(a) = T·log(∑_k exp(conv_k / T))

∇F_T(a) can be written in closed form:
    ∂F_T/∂a_i = 2·(rev_a * w)_i,
where w_k = exp(conv_k/T) / ∑_j exp(conv_j/T) and rev_a is a reversed copy
of a.  Using this smooth surrogate we run an Adam optimiser,
project the iterate onto the probability simplex after each step and
gradually decrease the temperature to recover the original max‑objective.
Occasional random perturbations help escape plateaus.
The best vector found (according to the exact objective) is returned.

The routine respects a 1000 s budget and stores the result in the global
variable ``height_sequence_1`` for warm‑starts in future calls.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i ≥ 0, ∑ x_i = 1}.
    """
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, n + 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

        for s in seeds:
            s = _project_simplex(s)
            val = _objective(s)
            if val < best_val:
                best_val = val
                best_seq = s
        a = best_seq
        print(f"[Init] best seed objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters for the smooth optimisation
    # ------------------------------------------------------------------
    # Adam parameters
    lr = 0.08                     # base learning rate
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Temperature schedule (start warm, end very cold)
    T_start = 1.0
    T_end = 1e-6
    # Linear decay in log‑space: log T = log T_start + (log T_end - log T_start) * t / Tmax
    # We will approximate by exponential decay per iteration.
    decay_T = 0.99995

    T = T_start

    best_seq = a.copy()
    best_obj = _objective(best_seq)
    last_improve = 0
    improve_patience = 3000   # iterations without improvement before a perturbation

    it = 0
    max_iters = 10 ** 7

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    print("[Main] Starting smooth Adam optimisation")
    while time.time() < deadline and it < max_iters:
        it += 1

        # temperature update (exponential decay, never below T_end)
        T = max(T_end, T * decay_T)

        # 1) convolution a*a
        conv = _conv_fft(a, a)               # length 2n‑1

        # 2) soft‑max weights for the smooth max
        max_c = conv.max()
        # (conv - max_c) ≤ 0, so exponentials stay ≤ 1 → no overflow
        w = np.exp((conv - max_c) / T)
        w_sum = w.sum()
        if w_sum == 0.0:                     # numerical safety (should not happen)
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # 3) gradient of the smooth surrogate
        rev_a = a[::-1]
        grad_full = _conv_fft(rev_a, w)      # length 3n‑2
        grad = 2.0 * grad_full[n - 1:n - 1 + n]

        # 4) Adam update (unconstrained)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # 5) projection onto the simplex (ensures non‑negativity & sum=1)
        a = _project_simplex(a)

        # ------------------------------------------------------------------
        # Periodic evaluation on the true (non‑smooth) objective
        # ------------------------------------------------------------------
        if it % 500 == 0:
            cur_obj = _objective(a)
            if cur_obj < best_obj:
                best_obj = cur_obj
                best_seq = a.copy()
                last_improve = it
                print(f"[Iter {it:6d}] new best objective = {cur_obj:.6f}")

            # If we have not improved for a while, perform a random perturbation
            if it - last_improve > improve_patience:
                # Small Gaussian jitter + re‑projection
                jitter = rng.normal(scale=0.015, size=n)
                a = a + jitter
                a = _project_simplex(a)
                m.fill(0.0)
                v.fill(0.0)
                last_improve = it
                print(f"[Iter {it:6d}] perturbation to escape plateau")

        # Keep a light check on time
        if it % 2000 == 0 and time.time() > deadline - 5.0:
            # final few seconds – break early to avoid cutting off prints
            break

    # ------------------------------------------------------------------
    # Final evaluation & safety normalisation
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.ones(n) / n
    else:
        final_seq = final_seq / total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (iterations={it})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
