Propose a low‑value step‑function for ``evaluate_sequence``.

Algorithm
---------

* **Seed generation** – a rich pool of ℓ₁‑normalised candidates:
  * classic families (uniform, triangular, trapezoidal, sine, Tukey, Gaussian)
  * geometric / symmetric‑geometric families with a dense grid of ratios
  * power‑law families with exponents up to 10
  * asymmetric beta families (α∈[0.01,2], β∈[1,20]) and their mirror images
  * random Dirichlet draws with many concentration values
  * the previous best solution (if any)

* **Coarse filtering** – evaluate the cheap exact surrogate  
  `obj(seq) = 2·n·max(conv(seq,seq))` (here `∑ seq = 1`) and keep the
  `top_k` best seeds.

* **Iterative optimisation** (performed while time permits):
  1. **Adam on a log‑sum‑exp surrogate** – we optimise the smooth

     `β` starts at 5 and is multiplied by `1.005` each iteration up to
     `β_max = 1500`.  Adam’s adaptive step‑size provides fast early

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation used during the search.
    Works for any non‑negative sequence (no automatic normalisation).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def _objective_normalised(seq: np.ndarray) -> float:
    """
    Fast surrogate for sequences already normalised to sum = 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * conv.max())


def _softmax(v: np.ndarray) -> np.ndarray:
    """Stable softmax, returns a probability vector."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex Σ=1, v≥0.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _trapezoidal(n: int, width_fraction: float = 0.2) -> np.ndarray:
    half_flat = max(1, int(np.round(width_fraction * n / 2.0)))
    a = np.zeros(n, dtype=np.float64)

    ramp_len = (n // 2) - half_flat
    if ramp_len > 0:
        ramp = np.linspace(0.0, 1.0, ramp_len, endpoint=False)
        a[:ramp_len] = ramp
        a[-ramp_len:] = ramp[::-1]

    a[half_flat:n - half_flat] = 1.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Seed generation utilities (more extensive than the previous version)
# ----------------------------------------------------------------------


def _collect_seeds(n: int, rng: np.random.Generator,
                   previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a heterogeneous pool of ℓ₁‑normalised candidates.
    The parameter ranges have been broadened to explore more asymmetric shapes.
    """
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_trapezoidal(n, width_fraction=0.15))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_sine_window(n, gamma=4.0))      # extra gamma

    # Geometric families (dense grid)
    for r in np.linspace(0.2, 0.999, 30):
        seeds.append(_geom(n, r))
        seeds.append(_sym_geom(n, r))

    # Power‑law family – extended exponent range
    for p in np.linspace(0.0, 10.0, 30):
        seeds.append(_power(n, p))

    # Beta family – much larger asymmetric range
    for _ in range(80):
        a = rng.uniform(0.01, 2.5)
        b = rng.uniform(0.5, 20.0)
        seeds.append(_beta_shape(n, a, b))
        # mirror image (swap α,β) gives a right‑skewed counterpart
        seeds.append(_beta_shape(n, b, a))

    # Gaussian windows – several σ
    for sigma_frac in (0.08, 0.12, 0.18, 0.24, 0.30, 0.36):
        seeds.append(_gaussian(n, sigma_frac))

    # Tukey windows – fine‑grained α
    for alpha in (0.05, 0.10, 0.15, 0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85, 0.95):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – wide concentration range
    for conc in (0.005, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(12):
        seeds.append(_random_exponential(n, rng))

    # Previous solution (if any) after re‑normalisation
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _adam_optimize_logits(start_vec: np.ndarray, n: int, deadline: float,
                          max_iters: int = 2_000_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the *log‑softmax* representation.
    Uses a log‑sum‑exp (smooth‑max) surrogate with a temperature that
    grows up to β_max = 1500.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)           # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule
    beta = 5.0
    beta_mul = 1.005
    beta_max = 1500.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                        # stays on simplex, sum = 1
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)       # exact objective for sum=1

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 4000 == 0:
                print(f"[Adam] it={it:7d} best={best_val:.6f}")

        # increase temperature slowly
        beta = min(beta * beta_mul, beta_max)

        # smooth max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # transform to log‑softmax coordinates (as in the reference code)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # gradient clipping for stability
        max_norm = 8.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_opt(start_seq: np.ndarray, n: int, deadline: float,
                    max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Projection onto the simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_normalised(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak step size – a little larger than the reference
        lr = 0.3 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _mirror_descent_opt(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent (multiplicative updates) using the exact max‑gradient.
    This often yields small additional improvements after the sub‑gradient phase.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_normalised(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))

        # gradient of the active convolution term
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev = k - idx
        g[idx] = 2.0 * a[rev]

        # multiplicative update
        eta = 0.6 / np.sqrt(it)
        mult = np.exp(-eta * g)
        a = a * mult
        a = a / a.sum()          # projection onto simplex (KL‑mirror)

        cur_val = 2.0 * n * conv[k]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 30000 == 0:
                print(f"[Mirror] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 1_500_000,
                         init_temp_factor: float = 0.03,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Stochastic polishing – randomly reshuffle the mass of a small subset
    of coordinates.  The temperature decays geometrically.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_normalised(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9994, min_temp)

        # pick a small random subset to re‑allocate
        k = rng.integers(2, 14)          # 2 … 13 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective_normalised(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine obeys the supplied
    time‑budget and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any) to use as a seed.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length choice – 6000 gives a finer discretisation while staying
    # comfortably fast for FFTs.
    # ------------------------------------------------------------------
    n = 6000
    print(f"\n=== Starting optimisation, length = {n} ===")

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse seed pool and keep the most promising.
    # ------------------------------------------------------------------
    print("\n--- Collecting seeds ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # cheap exact surrogate (since all seeds are ℓ₁‑normalised)
    evaluated = [(vec, _objective_normalised(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])

    top_k = 7            # a slightly larger set than before
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"\nTop {top_k} surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_normalised(global_best_seq)
    print(f"\n[Init] best surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy deterministic optimisation.
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        # 2.1 Adam on smooth surrogate
        seq_adam, val_adam = _adam_optimize_logits(
            seed_vec, n, deadline, max_iters=800_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # 2.2 Exact sub‑gradient descent
        seq_sub, val_sub = _subgradient_opt(
            global_best_seq, n, deadline, max_iters=1_500_000
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

        # 2.3 Mirror‑descent refinement
        seq_mir, val_mir = _mirror_descent_opt(
            global_best_seq, n, deadline, max_iters=1_000_000
        )
        if val_mir < global_best_val:
            global_best_seq = seq_mir.copy()
            global_best_val = val_mir
            print(f"[overall‑Mirror] new global best = {val_mir:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing.
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=1_200_000,
            init_temp_factor=0.04
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sanity check using the official evaluator.
    # ------------------------------------------------------------------
    try:
        eval_res = evaluate_sequence(global_best_seq.tolist())
        print(f"Official evaluator returned {eval_res:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse in subsequent calls
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
