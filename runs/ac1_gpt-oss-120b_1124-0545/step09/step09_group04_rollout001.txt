"""
Optimiser for the step‑function attack (minimise evaluate_sequence).

Algorithm
---------
1️⃣  **Seed generation** – a large, diverse pool of candidate vectors.
    Families include uniform, triangular, sine‑window, symmetric / unsymmetric
    geometric, power‑law, beta‐distribution, Dirichlet draws and sparse
    patterns.  All candidates are normalised to sum = 1 (the objective is
    homogeneous, so the scale does not matter).

2️⃣  **Screening** – each seed is evaluated with the cheap surrogate
    ``2·n·max(conv(a,a))`` (computed via an FFT) and the best few (default
    12) are retained.

3️⃣  **Refinement** – each elite seed is improved by a long mirror‑descent
    (exponentiated‑gradient) loop on the *true* objective
    ``2·n·max(conv(a,a))``.  The update is
        aᵗ⁺¹ ∝ aᵗ ⊙ exp(−ηₜ·g) ,
    where ``g`` is the sub‑gradient of the maximum convolution entry.
    The learning‑rate decays as ``η₀/√t``; we start with η₀=4.0.
    The best vector seen during the run is kept.

4️⃣  **Optional Euclidean step** – after EG we perform a short Euclidean
    projected‑gradient descent (same sub‑gradient, projection onto the
    probability simplex) to fine‑tune the solution.

5️⃣  **Polishing** – a lightweight simulated‑annealing phase that swaps
    mass between a random small subset of coordinates.

6️⃣  **Result** – the final vector (as a plain ``list[float]``) is returned
    and stored in the global name ``height_sequence_1`` for reuse in a
    later invocation.

The whole routine respects the 1000 s budget (a 5 s safety margin) and
uses only NumPy, ``time`` and the provided ``evaluate_sequence`` function.
"""

from __future__ import annotations

import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two real vectors using an FFT.
    Returns ``len(x)+len(y)-1`` real values.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate (and also exact) objective for a vector with sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
# ----------------------------------------------------------------------


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric series: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure (unsymmetric) geometric decay a_i = r**i."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta pdf on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(
    n: int, k: int, conc: float, rng: np.random.Generator
) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> List[np.ndarray]:
    """
    Produce a fairly large and diverse list of normalised candidate vectors.
    The list is deliberately over‑complete; later we keep only the best
    few according to the surrogate objective.
    """
    seeds: List[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – coarse grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric – very close to 1 (long plateau)
    for r in np.linspace(0.985, 0.9999, 12):
        seeds.append(_geom_decay(n, r))

    # pure power‑law decay
    for p in np.arange(0.1, 5.01, 0.3):
        seeds.append(_power_decay(n, p))

    # beta family – moderate grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform & symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.90, 7):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 4):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # Dirichlet draws with a wide range of concentration parameters
    for conc in (0.01, 0.02, 0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Sparse patterns – extremely small support
    for k in (3, 5, 8, 12):
        for conc in (0.1, 0.3, 0.7):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Sub‑gradient refinement – exponentiated‑gradient (mirror descent)
# ----------------------------------------------------------------------


def _subgradient_refine_exp(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 4.0,
) -> Tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑convolution objective.
    Uses exponentiated‑gradient updates (multiplicative).
    Returns the best vector encountered and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]  # exact objective (identical to surrogate)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient at the maximal entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # numerical stabilisation
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement – Euclidean projected gradient (optional)
# ----------------------------------------------------------------------


def _subgradient_refine_euclid(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    lr0: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the max‑convolution objective.
    After each Euclidean step we project back onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient at the maximal entry (same as in EG)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx
        g[idx] = 2.0 * a[rev_idx]

        # Euclidean step with decaying learning rate
        lr = lr0 / np.sqrt(it)
        a = a - lr * g
        # keep feasibility
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (lightweight)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> Tuple[np.ndarray, float]:
    """Simple SA that redistributes mass among a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)  # subset size 2–6
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:8d} best≈{best_val:.6f}")

    return best_seq, best_val
