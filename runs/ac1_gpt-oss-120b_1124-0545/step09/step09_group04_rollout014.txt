```
def _smooth_expgrad_refine(start_seq, n, deadline, max_iters=200_000, lr0=2.0, gamma0=5.0, gamma_inc=1.001):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    gamma = gamma0
    eps = 1e-30
    N = _next_pow2(3*n)
    L = 2*n - 1
    for it in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # compute B and weights
        fft_a = np.fft.rfft(a, N)
        B_full = np.fft.irfft(fft_a * fft_a, N)[:L]
        maxB = B_full.max()
        # smooth weights
        # reduce B to avoid overflow: shift by maxB
        exp_term = np.exp(gamma * (B_full - maxB))
        w = exp_term / exp_term.sum()
        # compute gradient via two convolutions
        fft_w = np.fft.rfft(w, N)
        conv1_full = np.fft.irfft(fft_w * fft_a, N)[:(L + n - 1)]
        conv1 = conv1_full[:n]
        w_rev = w[::-1]
        fft_w_rev = np.fft.rfft(w_rev, N)
        conv2_full = np.fft.irfft(fft_w_rev * fft_a, N)[:(L + n - 1)]
        conv2 = conv2_full[(L - n):L][::-1]
        grad = 2.0 * (conv1 + conv2)
        # step size
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum
        # evaluate surrogate for best tracking
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:,d} best≈{best_val:.6f}")
        # increase gamma gradually
        gamma *= gamma_inc
    return best_seq, best_val
```

"""
Optimiser for the step‑function bound.
Uses a richer family of analytical seeds, a smoothed
exponential‑gradient descent (mirror descent on a log‑sum‑exp
surrogate of the max‑convolution) and a final simulated‑annealing
polish.  The routine respects the 1000 s time budget and stores its
result in the global name ``height_sequence_1``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    ``seq`` is assumed to be non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:

def _smooth_expgrad_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    lr0: float = 2.0,
    gamma0: float = 5.0,
    gamma_inc: float = 1.001,
) -> tuple[np.ndarray, float]:

    Mirror‑descent on the smooth log‑sum‑exp surrogate of the max‑convolution.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    L = 2 * n - 1
    N = _next_pow2(3 * n)            # long enough for all convolutions

    gamma = gamma0

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---------- conv(a,a) ----------
        fft_a = np.fft.rfft(a, N)
        B = np.fft.irfft(fft_a * fft_a, N)[:L]

        # ---------- log‑sum‑exp weights ----------
        maxB = B.max()
        # shift to avoid overflow
        exp_term = np.exp(gamma * (B - maxB))
        w = exp_term / exp_term.sum()

        # ---------- gradient of the smooth surrogate ----------
        # first convolution: w * a
        fft_w = np.fft.rfft(w, N)
        conv1_full = np.fft.irfft(fft_w * fft_a, N)[: (L + n - 1)]
        conv1 = conv1_full[:n]                     # indices 0 … n‑1

        # second convolution: reversed w * a
        w_rev = w[::-1]
        fft_w_rev = np.fft.rfft(w_rev, N)
        conv2_full = np.fft.irfft(fft_w_rev * fft_a, N)[: (L + n - 1)]
        # take the last n entries and reverse them
        conv2 = conv2_full[(L - n) : L][::-1]

        grad = 2.0 * (conv1 + conv2)               # ∂f/∂a

        # ---------- mirror‑descent update ----------
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)                     # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        # keep the best ever seen
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:,d} best≈{best_val:.6f}")

        # slowly increase the smoothing parameter
        gamma *= gamma_inc

    return best_seq, best_val

def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # size of the random subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA]  it={it:,d} best≈{best_val:.6f}")

    return best_seq, best_val

Optimiser for the step‑function bound.

The algorithm works in three stages:

1️⃣  **Seed generation** – a wide variety of analytical families

        w = exp_term / exp_term.sum()

        # --------------------------------------------------------------
        # 3) Gradient of the smooth surrogate
        #    ∂f/∂a_j = 2·[ (w∗a)_j + (rev(w)∗a)_{n‑1‑j} ]
        # --------------------------------------------------------------
        fft_w = np.fft.rfft(w, N)
        conv1_full = np.fft.irfft(fft_w * fft_a, N)[: (L + n - 1)]
        conv1 = conv1_full[:n]                         # indices 0 … n‑1

        w_rev = w[::-1]
        fft_w_rev = np.fft.rfft(w_rev, N)
        conv2_full = np.fft.irfft(fft_w_rev * fft_a, N)[: (L + n - 1)]
        conv2 = conv2_full[(L - n) : L][::-1]           # last n entries, reversed

        grad = 2.0 * (conv1 + conv2)                   # ∂f/∂a

        # --------------------------------------------------------------
        # 4) Exponentiated‑gradient update (mirror descent on simplex)
        # --------------------------------------------------------------
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * grad
        log_a -= np.max(log_a)                         # improve numerical stability
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        # --------------------------------------------------------------
        # 5) Keep the best ever seen
        # --------------------------------------------------------------
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[ExpGrad] it={it:,d} best≈{best_val:.6f}")

        # slowly increase the smoothing parameter
        gamma *= gamma_inc

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)           # size of the random subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA]  it={it:,d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal ``evaluate_sequence`` value.
    Returns the best sequence found as a list of floats.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin before the hard timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Re‑use a previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the candidate (4000 offers a good trade‑off)
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse pool of seeds
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k], start=1):
        print(f"  #{i}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – smooth mirror‑descent from the strongest seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() >= deadline:
            break
        print(f"\n--- Smooth exp‑gradient from seed #{idx+1} ---")
        seq_ref, val_ref = _smooth_expgrad_refine(
            seed_vec,
            n,
            deadline,
            max_iters=250_000,
            lr0=2.0,
            gamma0=5.0,
            gamma_inc=1.001,
        )
        if val_ref < global_best_val:
            global_best_seq = seq_ref.copy()
            global_best_val = val_ref
            print(f"[Update] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=300_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[SA] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
