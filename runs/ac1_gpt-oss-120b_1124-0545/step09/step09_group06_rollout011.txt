def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    seq is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _greedy_mask(n: int, max_ordered: int, rng: np.random.Generator):
    """
    Return a mask, the corresponding sum‑counts array and a Python list of
    selected indices obtained by greedy insertion in a random order.
    """
    mask = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2 * n - 1, dtype=np.uint16)  # small counts, uint16 is safe
    support = []  # list of indices
    order = rng.permutation(n)
    for cand in order:
        # self‑pair constraint
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue

        feasible = True
        for e in support:
            if sum_counts[cand + e] + 2 > max_ordered:
                feasible = False
                break
        if not feasible:
            continue
        # add candidate
        for e in support:
            sum_counts[cand + e] += 2
        support.append(int(cand))
        mask[cand] = True
        sum_counts[2 * cand] += 1
    return mask, sum_counts, support


def _add_candidate(mask, sum_counts, support, cand, max_ordered):
    """
    Try to add cand to the current feasible set.
    Return True if successful, otherwise False (the data structures stay unchanged).
    """
    if mask[cand]:
        return False
    if sum_counts[2 * cand] + 1 > max_ordered:
        return False
    for e in support:
        if sum_counts[cand + e] + 2 > max_ordered:
            return False
    # perform addition
    for e in support:
        sum_counts[cand + e] += 2
    support.append(int(cand))
    mask[cand] = True
    sum_counts[2 * cand] += 1
    return True


def _remove_candidate(mask, sum_counts, support, idx):
    """
    Remove the element at position idx (value) from the set.
    The caller must guarantee that idx is present in support.
    """
    # idx is the element value (not position in list)
    # remove contributions
    mask[idx] = False
    sum_counts[2 * idx] -= 1
    # subtract cross contributions
    for e in support:
        if e == idx:

    # remove from list
    support.remove(idx)


def _local_improve(mask, sum_counts, support, n, max_ordered, rng,
                   target_m, deadline):
    """
    Simple local search that repeatedly tries random additions.
    When stuck, it removes a randomly chosen element to free capacity.
    The process stops when the time budget expires or the desired size is reached.

    best_support = list(support)

                cur_size = len(support)

        rem = rng.choice(support)
        _remove_candidate(mask, sum_counts, support, rem)
        # note: size decreased by one, continue loop which will try to add again
    return best_mask, best_support

    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)

def _adam_optimize(seed_vec, n, deadline, max_iters=20000):
    """
    Adam on a smoothed‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its exact objective.
    """
    v = np.log(seed_vec + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta_temp = 1.2
    beta_mul = 1.006

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smoothed‑max weights
        beta_temp *= beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·(a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)  # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam updates
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    if best_seq is None:
        best_seq = seed_vec.copy()
        best_val = _objective(best_seq)
    return best_seq, best_val

        best_seq = seed_vec.copy()
        best_val = _objective(best_seq)
    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):

    # ------------------------------------------------------------------

    )

        # Fallback: uniform over the whole domain
        seq = np.full(n, 1.0 / n, dtype=np.float64)
