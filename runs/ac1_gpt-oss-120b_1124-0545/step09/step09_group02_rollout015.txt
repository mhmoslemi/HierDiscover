def _smooth_objective(a, alpha):
    conv = _conv_fft(a, a)
    # compute weighted sum via softmax
    max_val = conv.max()  # for numerical stability to shift
    # shift by subtracting max, but we can compute directly using np.exp(alpha*(conv - max_val))
    exp_vals = np.exp(alpha * (conv - max_val))
    Z = exp_vals.sum()
    M = max_val + np.log(Z) / alpha
    # gradient w.r.t a: 2 * conv_rev where w = exp(alpha*conv) / Z

    grad = 2.0 * grad_full[n-1: n-1 + n]  # note shape
    return M, grad

"""
Optimise a non‑negative step function for the objective

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm proceeds in four stages

1️⃣  Build a rich pool of analytic seeds (uniform, windows, power‑law,
    geometric, random Dirichlet, sparse Dirichlet, mixtures of Gaussians,
    etc.) and keep the best few.

2️⃣  Refine every kept seed with a *smoothed* mirror‑descent optimiser.
    Instead of the hard max we minimise a log‑sum‑exp surrogate
    (a smooth approximation of the max).  The surrogate is differentiable,
    the gradient is obtained by a single FFT‑based convolution and the
    parameters are updated with an Adam‑style exponentiated‑gradient step.
    The “temperature’’ α of the soft‑max is increased gradually so that the
    algorithm first explores a smooth landscape and then focuses on the
    true max.

3️⃣  The best sequence obtained so far is further polished with the
    deterministic additive descent from the original code and finally with
    a short simulated‑annealing walk to escape any remaining plateaux.

4️⃣  Return the best sequence found (as a plain list).  The function respects
    the 1000 s time‑budget and stores the best construction in the global
    variable ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# FFT based linear convolution (real inputs, full mode)
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# --------------------------------------------------------------
# Analytic families (all returned vectors are normalised to sum = 1)
# --------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _random_sparse(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries with random positive masses."""
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = rng.random(k) + 1e-12
    a /= a.sum()
    return a


def _mix_gaussians(n: int, rng: np.random.Generator, ncomp: int = 3) -> np.ndarray:
    """Mixture of a few Gaussian‑shaped bumps."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.zeros(n, dtype=np.float64)
    for _ in range(ncomp):
        mu = rng.integers(0, n)
        sigma = rng.uniform(0.02, 0.12) * n
        comp = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        weight = rng.random()
        a += weight * comp
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Construct a diverse start‑vector pool (all normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # elementary families -------------------------------------------------
    seeds.append(_uniform(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_sine_window(n))

    # geometric and power‑law sweeps --------------------------------------
    for r in np.linspace(0.95, 0.9999, 30):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 4.0, 41):
        seeds.append(_power(n, p))
    for r in np.linspace(0.9, 0.999, 15):
        for p in np.linspace(0.0, 2.0, 7):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric --------------------------------------------------
    for r in np.linspace(0.9, 0.999, 15):
        seeds.append(_sym_geom(n, r))

    # Tukey and beta windows (finer grid) --------------------------------
    for alpha in np.linspace(0.2, 0.8, 13):
        seeds.append(_tukey_window(n, alpha))
    for a in np.linspace(0.1, 2.5, 25):
        seeds.append(_beta_window(n, a))

    # Gaussian family (wide sweep) ---------------------------------------
    for sigma_factor in np.linspace(0.05, 0.35, 13):
        seeds.append(_gaussian(n, sigma_factor))

    # Random Dirichlet draws (very low concentration) --------------------
    for alpha in [0.01, 0.02, 0.05, 0.1, 0.2]:
        for _ in range(6):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # Random soft‑max vectors ---------------------------------------------
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # Random sparse vectors ------------------------------------------------
    for k in [int(np.sqrt(n)), int(np.sqrt(n) * 2), int(np.sqrt(n) * 4)]:
        for _ in range(4):
            seeds.append(_random_sparse(n, k, rng))

    # Mixture of Gaussians -------------------------------------------------
    for _ in range(12):
        seeds.append(_mix_gaussians(n, rng, ncomp=rng.integers(2, 5)))

    # Re‑use previous construction -----------------------------------------
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up -------------------------------------------------------
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned

# Euclidean projection onto the probability simplex
# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# --------------------------------------------------------------
# Smoothed mirror‑descent optimiser (log‑sum‑exp surrogate)
# --------------------------------------------------------------
def _smooth_adam_optimize(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Adam on a soft‑max surrogate of the max‑convolution.
    Returns the best sequence seen and its exact objective value.
    """
    a = start_seq.copy()
    # start from a log‑parameterisation to keep positivity
    log_a = np.log(a + 1e-30)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = np.exp(log_a)
        a /= a.sum()

        # soft‑max temperature – grows with the iteration count
        alpha = 0.5 + 49.5 * (it - 1) / max_iters          # from 0.5 to 50

        conv = _conv_fft(a, a)                 # exact convolution
        max_val = conv.max()
        # soft‑max weighting (numerically stable)
        shift = conv - max_val
        exp_w = np.exp(alpha * shift)
        w = exp_w / exp_w.sum()                # ∑ w = 1, w concentrates on active indices

        # gradient of the surrogate (see derivation in the doc‑string)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # same size as conv
        grad = 2.0 * full[n - 1: n - 1 + n]    # ∂/∂a of the smooth max

        # Adam on log‑variables  (∂U/∂log a = a·∂U/∂a)
        g_log = grad * a
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.12 / np.sqrt(it)               # base learning‑rate schedule
        log_a += -lr * m_hat / (np.sqrt(v_hat) + eps)

        # renormalise (log‑space translation is harmless)
        log_a -= np.max(log_a)

        # evaluate true objective only every few iterations to save time
        if it % 400 == 0 or it == max_iters:
            cur_val = _objective(np.exp(log_a) / np.exp(log_a).sum())
            if cur_val < best_val:
                best_val = cur_val
                best_seq = np.exp(log_a).copy()
                best_seq /= best_seq.sum()
                print(f"[SmoothAdam] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val

def _additive_descent(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 200000) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max(rel_tol, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.12 / np.sqrt(it)
        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        a = a_next
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AddGD] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val

# Simulated‑annealing fine‑tuning (same as the original code)
# --------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.68:                               # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.18 * cur[i]
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif rnd < 0.86:                             # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                                        # global Gaussian jitter
            noise = rng.normal(scale=0.0013, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        if it % 2500 == 0:
            jitter = rng.normal(scale=0.0035, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val


# --------------------------------------------------------------
# Main entry point required by the evaluator
# --------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative vector that (approximately) minimises
    ``evaluate_sequence``.
    Returns the best vector found as a list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # small safety margin

    # ------------------------------------------------------------
    # Problem size – 4000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------
    # Retrieve a previously discovered construction, if any
    # ------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------
    # 0)  evaluate the stored construction (if compatible)
    # ------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] val={val:.6f}")

    # ------------------------------------------------------------
    # 1)  Build a very rich pool of seeds and keep the top few
    # ------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 10
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # ------------------------------------------------------------
    # 2)  Deterministic refinements
    #      – smooth Adam optimiser (log‑sum‑exp surrogate)
    #      – additive gradient descent (original method)
    # ------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 12.0:
            break
        print(f"[refine] start seed {idx}")

        seq, val = _smooth_adam_optimize(seed_vec, n, deadline,
                                         max_iters=180000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] smooth‑adam best={val:.6f}")

        seq, val = _additive_descent(seq, n, deadline,
                                     max_iters=120000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] additive best={val:.6f}")

    # ------------------------------------------------------------
    # 3)  Simulated‑annealing fine‑tuning (uses remaining time)
    # ------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall] anneal best={val:.6f}")

    # ------------------------------------------------------------
    # 4)  Fallback (unlikely)
    # ------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
