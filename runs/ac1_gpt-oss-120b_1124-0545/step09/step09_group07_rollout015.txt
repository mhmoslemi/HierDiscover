def simplex_projection(v):
    # Project onto the simplex {x >=0, sum(x)=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Robust initialisation** – generate several cheap seeds (uniform,
   random soft‑max, power‑law, exponential) and keep the best one.

2. **Adam‑based sub‑gradient optimisation** – we minimise the
   non‑smooth objective by repeatedly taking a sub‑gradient of the
   currently active convolution entry (the single maximal entry) and
   performing an Adam update.  After each step the vector is projected
   onto the probability simplex.

3. **Dirichlet local search** – a fast stochastic refinement.  Starting
   from the best vector found above we repeatedly draw a Dirichlet
   sample with concentration proportional to the current point (the
   concentration is slowly reduced).  Any improvement is kept.

The whole routine respects the supplied time‑budget (default 1000 s) and
stores the best vector in the global ``height_sequence_1`` for the next
invocation.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT based convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("simplex_projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _power_law(n: int, p: float = 0.5) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (heavy tail on the left side)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    return a / a.sum()


def _exponential(n: int, rate: float = 0.001) -> np.ndarray:
    """a_i ∝ exp(-rate·i)  (decreasing geometric shape)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-rate * i)
    return a / a.sum()


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangle: rise then fall."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)  # peak at centre
    return a / a.sum()


# ----------------------------------------------------------------------
# Adam based sub‑gradient optimisation (single active constraint)
# ----------------------------------------------------------------------
def _adam_subgradient(start_seq: np.ndarray,
                      deadline: float,
                      eta: float = 0.15,
                      beta1: float = 0.9,
                      beta2: float = 0.999,
                      eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the max‑convolution objective using Adam.
    Returns the best vector encountered and its objective value.
    """
    a = start_seq.copy()
    n = a.size
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    best_seq = a.copy()
    best_val = _objective(a)

    t = 0
    while True:
        t += 1
        if time.time() > deadline:
            break

        # convolution and index of the current maximum
        conv = _conv_fft(a, a)
        k = int(conv.argmax())             # single active index
        # gradient of the active quadratic term
        w = np.zeros(2 * n - 1, dtype=np.float64)
        w[k] = 1.0
        rev_a = a[::-1]
        grad = 2.0 * _conv_fft(rev_a, w)[n - 1:n - 1 + n]

        # Adam update (minimise!)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step                     # move opposite to sub‑gradient
        a = _simplex_projection(a)       # keep on the simplex

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Dirichlet stochastic refinement
# ----------------------------------------------------------------------
def _dirichlet_refine(start_seq: np.ndarray,
                      deadline: float,
                      start_conc: float = 2000.0,
                      end_conc: float = 50.0,
                      decay: float = 0.9995) -> tuple[np.ndarray, float]:
    """
    Repeatedly draw a Dirichlet sample with concentration proportional
    to the current point.  The concentration slowly shrinks, allowing
    a coarse‑to‑fine stochastic search.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    best = cur.copy()
    best_val = _objective(cur)
    conc = start_conc
    it = 0
    while time.time() < deadline and conc > end_conc:
        it += 1
        # Dirichlet draw centred at cur
        candidate = rng.dirichlet(conc * cur)
        val = _objective(candidate)
        if val < best_val:
            best_val = val
            best = candidate.copy()
            cur = candidate.copy()
            print(f"[Dir] it={it:7d} best={val:.6f}")
        # geometric decay of concentration
        conc = max(end_conc, conc * decay)
    return best, best_val
