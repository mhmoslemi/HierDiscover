"""
Non‑negative step‑function optimiser – version 3.

The strategy is a focussed, long‑run refinement of a single high‑quality
seed (the previous best sequence if available).  It combines several
gradient‑based phases and a final simulated‑annealing polishing stage.
All phases respect a global time budget (default 1000 s) and constantly
track the best sequence seen with respect to the true evaluation metric
``evaluate_sequence`` (which equals ``2·n·max(conv(a,a))`` for a
normalised probability vector).
"""

import time
import numpy as np

# --------------------------------------------------------------
# Low‑level utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    The vector is assumed to be non‑negative and normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))`` – exactly the value returned by
    ``evaluate_sequence`` for a normalised vector.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# --------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


# --------------------------------------------------------------
# Optimisation primitives
# --------------------------------------------------------------

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 300000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp (smoothed max) surrogate.
    The vector stays on the simplex via a log‑softmax parametrisation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    # temperature schedule for the log‑sum‑exp smoothing
    beta = 0.5
    beta_mul = 1.002
    beta_max = 2000.0          # larger → closer to the true max

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                    # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # anneal temperature
        beta = min(beta * beta_mul, beta_max)

        # smoothed max via log‑sum‑exp on the convolution entries
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr = 0.9 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 800000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on
    the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _euc_proj_subgradient(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 400000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the true max‑convolution with Euclidean projection
    onto the simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100000 == 0:
                print(f"[EucProj] it={it:7d} best={cur_val:.6f}")

        # gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 900000,
                         init_temp_factor: float = 0.1,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simulated annealing that reshuffles the mass of a few coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # choose a random subset (3‑12 indices) and redistribute its mass
        k = rng.integers(3, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` value.

    The routine
      * starts from the previous best sequence if present (otherwise a
        uniform vector of length 4000),
      * refines it with a long Adam optimiser on a smoothed max‑convolution,
      * follows with an exponentiated‑gradient sub‑gradient phase,
      * applies an Euclidean‑projection gradient phase,
      * finishes with a simulated‑annealing polishing stage,
    always respecting the supplied time budget.
    The best vector found is stored in the global name ``height_sequence_1``
    and returned as a plain list.
    """
    # ----------------------------------------------------------
    # initialise
    # ----------------------------------------------------------
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + budget_s - 5.0

    # try to reuse a previous solution
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            init_seq = np.asarray(previous, dtype=np.float64)
        except Exception:
            init_seq = None
    else:
        init_seq = None

    # ----------------------------------------------------------
    # length of the search vector – 4000 is a well‑balanced choice
    # ----------------------------------------------------------
    n = 4000
    if init_seq is None or init_seq.shape != (n,):
        # No usable previous vector – fall back to a simple seed
        init_seq = _uniform(n)
        # throw in a couple of alternative deterministic seeds and keep the best
        seed_candidates = [
            _triangular(n),
            _sine_window(n),
            _geom_decrease(n, 0.85),
            _power_decrease(n, 1.2)
        ]
        best_val = float("inf")
        for cand in seed_candidates:
            v = _objective(cand)
            if v < best_val:
                best_val = v
                init_seq = cand

    # normalise to a probability vector (scale does not matter for the metric)
    init_seq = np.maximum(init_seq, 0.0)
    s = init_seq.sum()
    if s == 0.0:
        init_seq = _uniform(n)
    else:
        init_seq = init_seq / s

    print(f"\n--- Starting optimisation (n={n}) ---")
    print(f"Initial surrogate objective = {_objective(init_seq):.6f}")

    # ----------------------------------------------------------
    # Stage 1 – Adam on smoothed max
    # ----------------------------------------------------------
    seq_adam, val_adam = _adam_optimize(
        init_seq, n, deadline, max_iters=300000
    )
    if seq_adam is not None:
        best_seq = seq_adam
        best_val = val_adam
        print(f"[Stage‑Adam] best objective = {val_adam:.6f}")
    else:
        best_seq = init_seq
        best_val = _objective(best_seq)

    # ----------------------------------------------------------
    # Stage 2 – sub‑gradient (exponentiated‑gradient) refinement
    # ----------------------------------------------------------
    seq_sub, val_sub = _subgradient_refine(
        best_seq, n, deadline, max_iters=800000
    )
    if val_sub < best_val:
        best_seq, best_val = seq_sub, val_sub
        print(f"[Stage‑SubGrad] improved objective = {val_sub:.6f}")

    # ----------------------------------------------------------
    # Stage 3 – Euclidean‑projection gradient descent
    # ----------------------------------------------------------
    seq_euc, val_euc = _euc_proj_subgradient(
        best_seq, n, deadline, max_iters=400000
    )
    if val_euc < best_val:
        best_seq, best_val = seq_euc, val_euc
        print(f"[Stage‑EucProj] improved objective = {val_euc:.6f}")

    # ----------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing
    # ----------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            best_seq, n, deadline, max_iters=900000,
            init_temp_factor=0.1
        )
        if val_sa < best_val:
            best_seq, best_val = seq_sa, val_sa
            print(f"[Stage‑SA] improved objective = {val_sa:.6f}")

    # ----------------------------------------------------------
    # Optional: try sorted orders (sometimes helps numerically)
    # ----------------------------------------------------------
    if time.time() < deadline:
        for ordering in ("desc", "asc"):
            cand = np.sort(best_seq)[::-1] if ordering == "desc" else np.sort(best_seq)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_seq, best_val = cand, cand_val
                print(f"[Reorder] new best with {ordering} order = {cand_val:.6f}")

    # ----------------------------------------------------------
    # Final sanity check with the real evaluate_sequence
    # ----------------------------------------------------------
    try:
        true_val = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluate_sequence value = {true_val:.6f}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose globally for possible reuse in future runs
    globals()["height_sequence_1"] = best_seq.tolist()
    print(f"\n=== Finished ===\nBest surrogate objective ≈ {best_val:.6f} (length {len(best_seq)})")
    if true_val is not None:
        print(f"True objective (after evaluation) ≈ {true_val:.6f}")
    return best_seq.tolist()
