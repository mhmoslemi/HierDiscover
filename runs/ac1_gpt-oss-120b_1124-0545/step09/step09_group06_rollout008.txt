def repair_set(S, n, max_allowed):
    S = set(S)
    sum_counts = np.convolve(np.array([1 if i in S else 0 for i in range(n)], dtype=int), 
                             np.array([1 if i in S else 0 for i in range(n)], dtype=int))
    while True:
        if sum_counts.max() <= max_allowed:
            break
        k = int(sum_counts.argmax())
        # compute contributions for each element in S to sum k
        contrib = {}
        for i in S:
            j = k - i
            if j in S:
                if i == j:
                    contrib[i] = 1
                else:
                    contrib[i] = 2
            else:
                contrib[i] = 0
        # find element with max contrib
        i_remove = max(contrib, key=contrib.get)
        # update sum_counts
        for j in S:
            if j == i_remove:
                sum_counts[2*i_remove] -= 1
            else:
                sum_counts[i_remove + j] -= 2
        S.remove(i_remove)

"""
Low‑collision step‑function optimiser

The algorithm concentrates on building a *large* support S ⊂ {0,…,n‑1}
such that every ordered pair‑sum i+j (i,j∈S) occurs at most
``max_allowed = 4`` times.  For a uniform distribution on such a set the
objective becomes

    2·n·max(conv)/sum² = 2·n·c_max / |S|²  ≤  8·n / |S|² .

Hence, maximising |S| under the collision bound directly yields a very
small evaluation value.  The search consists of many random initial
subsets, a greedy *repair* that removes offending elements, followed by a
maximal greedy *expansion*.  The best support discovered is turned into a
uniform probability vector; if the obtained bound is still above the
target (1.503) a short sub‑gradient refinement is performed.
The final sequence is stored in the global ``height_sequence_1`` for reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for low‑collision support construction
# ----------------------------------------------------------------------


def _compute_sum_counts(S: set[int], n: int) -> np.ndarray:
    """Ordered‑pair‑sum multiplicities for a given support S."""
    cnt = np.zeros(2 * n - 1, dtype=np.int32)
    S_list = sorted(S)
    L = len(S_list)
    for idx in range(L):
        i = S_list[idx]
        cnt[2 * i] += 1                     # (i,i)
        for j in S_list[idx + 1 :]:
            s = i + j
            cnt[s] += 2                     # (i,j) and (j,i)
    return cnt


def _can_add(cand: int, S: set[int], cnt: np.ndarray, max_allowed: int) -> bool:
    """Check whether adding ``cand`` would keep every sum ≤ max_allowed."""
    if cnt[2 * cand] + 1 > max_allowed:
        return False
    for j in S:
        if cnt[cand + j] + 2 > max_allowed:
            return False
    return True


def _add_element(cand: int, S: set[int], cnt: np.ndarray) -> None:
    """Insert ``cand`` into the support and update the sum‑count array."""
    for j in S:
        cnt[cand + j] += 2
    cnt[2 * cand] += 1
    S.add(cand)


def _remove_element(rem: int, S: set[int], cnt: np.ndarray) -> None:
    """Delete ``rem`` from the support and update the sum‑count array."""
    cnt[2 * rem] -= 1
    for j in S:
        if j == rem:
            continue
        cnt[rem + j] -= 2
    S.remove(rem)


def _repair_set(
    S: set[int],
    cnt: np.ndarray,
    n: int,
    max_allowed: int,
    rng: np.random.Generator,
) -> tuple[set[int], np.ndarray]:
    """
    Greedy removal: while any sum exceeds ``max_allowed`` remove the element
    that contributes the most to the most overloaded sum.
    """
    while cnt.max() > max_allowed:
        k = int(cnt.argmax())
        # Find element with the largest contribution to sum k
        best_elem = None
        best_contrib = -1
        for i in list(S):
            j = k - i
            if j in S:
                contrib = 2 if i != j else 1
                if contrib > best_contrib:
                    best_contrib = contrib
                    best_elem = i
                if best_contrib == 2:   # cannot beat this
                    break
        if best_elem is None:           # safety – should not happen
            break
        _remove_element(best_elem, S, cnt)
    return S, cnt


def _expand_set(
    S: set[int],
    cnt: np.ndarray,
    n: int,
    max_allowed: int,
    rng: np.random.Generator,
) -> tuple[set[int], np.ndarray]:
    """
    Greedy insertion of any element that does not violate the collision bound.
    The process repeats until a full pass adds nothing.
    """
    changed = True
    while changed:
        changed = False
        candidates = list(set(range(n)) - S)
        rng.shuffle(candidates)
        for cand in candidates:
            if _can_add(cand, S, cnt, max_allowed):
                _add_element(cand, S, cnt)
                changed = True
    return S, cnt


# ----------------------------------------------------------------------
# Objective (fast, exact) – identical to evaluate_sequence for positive data
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact value of 2·n·max(conv(seq,seq))/sum(seq)²."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = np.convolve(seq, seq)          # exact linear convolution (real)
    return 2.0 * len(seq) * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple sub‑gradient refinement (fallback if the low‑collision set is not enough)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true non‑smooth objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = np.convolve(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = np.convolve(rev_a, w)          # length 3n‑2
            g = 2.0 * full[n - 1 : n - 1 + n]

        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                     # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur = _objective(a)
        if cur < best_val:
            best_val = cur
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a comfortable theoretical bound (~1.02)
    # ------------------------------------------------------------------
    n = 8000
    max_allowed = 4
    target_m = int(0.95 * math.sqrt(8.0 * n))      # ≈ 0.95·√(8n)
    print(f"\n=== Optimising length {n} (target support {target_m}) ===")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive random search for a large low‑collision set
    # ------------------------------------------------------------------
    lowcoll_deadline = time.time() + 0.35 * (deadline - time.time())
    best_set = None
    best_size = 0
    attempt = 0

    while time.time() < lowcoll_deadline:
        attempt += 1
        # start with a random over‑dense subset
        init_frac = 1.2 * target_m / n
        mask = rng.random(n) < init_frac
        S = set(np.where(mask)[0].tolist())
        if not S:
            continue

        cnt = _compute_sum_counts(S, n)

        # repair (remove offending elements) and then expand greedily
        S, cnt = _repair_set(S, cnt, n, max_allowed, rng)
        S, cnt = _expand_set(S, cnt, n, max_allowed, rng)

        cur_size = len(S)
        if cur_size > best_size:
            best_size = cur_size
            best_set = np.array(sorted(S), dtype=int)
            print(f"[LowColl] attempt {attempt:4d} size {cur_size}")

        # early exit if we have already reached the theoretical target
        if best_size >= target_m:
            break

    if best_set is None:
        # fall back to a trivial uniform vector
        best_set = np.arange(n, dtype=int)
        best_size = n

    # ------------------------------------------------------------------
    # Build the uniform probability vector on the discovered support
    # ------------------------------------------------------------------
    seq = np.zeros(n, dtype=np.float64)
    seq[best_set] = 1.0 / best_size

    best_val = _objective(seq)
    print(f"[Initial] support {best_size}, objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – optional sub‑gradient refinement if we are still above target
    # ------------------------------------------------------------------
    target = 1.503
    if best_val > target and time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        remaining = sub_deadline - time.time()
        max_iters = int(min(200_000, remaining / 0.0015))
        seq, best_val = _subgradient_refine(seq, n, sub_deadline, max_iters=max_iters)
        print(f"[Refine] post‑subgrad objective ≈ {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and safety checks
    # ------------------------------------------------------------------
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total < 1e-12:
        seq = np.ones(n, dtype=np.float64) / n
    else:
        seq /= total

    # ------------------------------------------------------------------
    # Reuse a previously stored solution if it is better
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            prev_arr = np.asarray(prev, dtype=np.float64)
            if prev_arr.shape == (n,):
                prev_val = _objective(prev_arr)
                if prev_val < best_val:
                    seq = prev_arr.copy()
                    best_val = prev_val
                    print(f"[Reuse] kept previous best {prev_val:.6f}")
        except Exception:
            pass

    # expose for possible future runs
    globals()["height_sequence_1"] = seq.copy()

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (support {best_size})")
    return seq.tolist()
