"""
Propose a step‑function (a non‑negative vector summing to 1) that minimizes
the black‑box ``evaluate_sequence`` objective.

Algorithm (≈ 1000 s budget)
-------------------------------------------------
1. Analytic seeds – generate many classic windows (uniform, Hann,
   Blackman, Gaussian, Tukey, sin‑power, Beta‑pdf) over several lengths.
   Keep the best seed.

2. Random Dirichlet / Beta sampling – keep drawing random vectors
   (Dirichlet with a few α’s and Beta shapes with random (α,β))
   while ~30 % of the time budget remains.

3. Smoothed Adam – run Adam on a log‑softmax parametrisation with a
   log‑sum‑exp smooth approximation of the max‑convolution.
   Use a longer temperature schedule (β∈[5, 500]) and a higher iteration
   budget.

4. L‑BFGS refinement – minimise the same smooth objective with SciPy's
   L‑BFGS‑B (more iterations than the reference code).

5. Aggressive KL‑mirror descent – sub‑gradient descent on the *exact*
   max‑convolution, aggregating *all* near‑maximal indices (relative
   tolerance 1e‑8).  A larger step‑size constant (1.0 / √t) is used.

6. Fine‑grained random local search – tiny mass transfers between two
   random coordinates, accepted only if they improve the true objective.
   This polish is run on the remaining budget.

All phases respect the global deadline; the best sequence found is
returned as a plain Python list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import pi, sqrt, log, exp

# optional SciPy import – L‑BFGS will be skipped if unavailable

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a *normalised* sequence (sum == 1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _tukey(n: int, alpha: float = 0.5) -> np.ndarray:
    """Standard Tukey (tapered cosine) window. 0≤alpha≤1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann(n)
    i = np.arange(n, dtype=np.float64) / (n - 1)
    w = np.ones_like(i)
    left = i < alpha / 2
    right = i > 1 - alpha / 2
    w[left] = 0.5 * (1 + np.cos(pi * (2 * i[left] / alpha - 1)))
    w[right] = 0.5 * (1 + np.cos(pi * (2 * i[right] / alpha - 2 / alpha + 1)))
    return w / w.sum()


def _sin_power(n: int, power: float = 2.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = np.sin(pi * i / (n - 1)) ** power
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf sampled on the centre of each bin."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()

# Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector; α=1 gives uniform on simplex."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(start_seq: np.ndarray,
                          deadline: float,
                          beta_start: float = 5.0,
                          beta_end: float = 500.0,
                          max_iter: int = 8_000_000,
                          lr0: float = 0.04) -> tuple[np.ndarray, float]:
    """Adam on a log‑softmax parameterisation with a log‑sum‑exp smooth max."""
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max (guarantees non‑negativity & sum‑to‑1)
        a = np.exp(logits)
        a = a / a.sum()

        # self‑convolution
        conv = _conv_fft(a, a)                 # length = 2n‑1

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # log‑sum‑exp smooth max
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()          # soft weights, sum=1

        # gradient w.r.t. a (see paper)
        conv_pa = _conv_fft(p, a[::-1])        # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain‑rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:      # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            # early exit if approaching target
            if best_val < 1.45:
                break

    return best_seq, best_val

# Optional L‑BFGS refinement (smooth max with large β)
# ----------------------------------------------------------------------
def _lbfgs_refine(start_seq: np.ndarray,
                  deadline: float,
                  beta: float = 500.0,
                  maxiter: int = 400) -> tuple[np.ndarray, float]:
    """L‑BFGS‑B on the smooth max objective (if SciPy is available)."""
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a = a / a.sum()
        conv = _conv_fft(a, a)          # length = 2n‑1

        # smooth max via log‑sum‑exp
        c = np.max(conv)
        exps = np.exp(beta * (conv - c))
        lse = c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        # gradient w.r.t a
        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain‑rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class _TimeOut(RuntimeError):
        pass

    def callback(_):
        if time.time() >= deadline:
            raise _TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': maxiter, 'disp': False},
            callback=callback,
        )
        best_logits = res.x
    except _TimeOut:
        best_logits = x0

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive KL‑mirror descent (aggregated active set)
# ----------------------------------------------------------------------
def _aggressive_mirror_descent(start_seq: np.ndarray,
                               deadline: float,
                               eps_rel: float = 1e-8) -> tuple[np.ndarray, float]:
    """KL‑mirror descent using *all* near‑maximal indices (relative tolerance)."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        # indices whose value is within relative tolerance of the max
        thr = max_val - eps_rel * max_val
        active_idx = np.where(conv >= thr)[0]

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # aggregated sub‑gradient = 2 * (active * a_rev) convolved
        conv_ag = _conv_fft(active, a[::-1])          # length = 3n‑2
        g = 2.0 * conv_ag[n - 1:n - 1 + n]            # slice = n

        # KL‑mirror step
        lr = 1.0 / sqrt(it)          # a slightly larger prefactor than the reference
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)       # for numerical stability
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:        # every 1024 iterations
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggMirror] it={it:7d} n={n:5d} val={val:.6f}")

            # early exit when very close to deadline
            if time.time() > deadline - 0.8:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Fine‑grained random local search
# ----------------------------------------------------------------------
def _random_local_search(start_seq: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator,
                         init_step: float = 0.02) -> tuple[np.ndarray, float]:
    """Swap tiny amounts of mass between random coordinates; keep improvements."""
    a = start_seq.copy()
    n = len(a)
    best_val = _objective(a)
    best_seq = a.copy()
    step = init_step

    while time.time() < deadline:
        # choose two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # amount we are willing to move (bounded by current mass at i)
        max_move = min(a[i], step)
        if max_move <= 0:
            continue
        delta = rng.random() * max_move

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            # slowly tighten the step as we improve
            step = max(step * 0.995, 1e-5)
            print(f"[Local] improve to {val:.6f} step={step:.6e}")
        else:
            # occasionally shrink step even on non‑improvement to allow finer moves later
            step = max(step * 0.9995, 1e-5)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the supplied time budget.
    Returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Possible reuse of a previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – a mix of modest and larger values
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 4500, 5000, 6000, 7000, 8000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds
    # ------------------------------------------------------------------
    analytic_families = (
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        lambda n: _tukey(n, alpha=0.4),
        lambda n: _tukey(n, alpha=0.6),
        lambda n: _sin_power(n, power=1.5),
        lambda n: _sin_power(n, power=3.0),
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
    )
    for n in lengths:
        for fn in analytic_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – random sampling (Dirichlet + Beta)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.30
    sample_deadline = now + time_for_sampling

    dirichlet_alphas = [0.2, 0.5, 1.0, 2.0, 5.0]
    beta_param_grid = [(rng.uniform(0.1, 5.0), rng.uniform(0.1, 5.0)) for _ in range(30)]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet draw
        alpha = float(rng.choice(dirichlet_alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rand-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta pdf shape
        a_par, b_par = rng.choice(beta_param_grid)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Rand-Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        # early break if already very good (safety)
        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # keep the three most promising candidates (current best + two random restarts)
        candidates = [best_seq]
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(dirichlet_alphas))
            candidates.append(cand)

        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.30   # approx 30 % of what is left

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=500.0,
                max_iter=8_000_000,
                lr0=0.04,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional L‑BFGS refinement (more iterations)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 4.0 and minimize is not None:
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline,
                                                 beta=500.0, maxiter=600)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – aggressive KL‑mirror descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _aggressive_mirror_descent(best_seq, deadline,
                                                          eps_rel=1e-8)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑Mirror] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – fine‑grained random local search (very small moves)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            seq_local, val_local = _random_local_search(best_seq, time.time() + remaining,
                                                        rng, init_step=0.02)
            if val_local < best_val:
                best_val = val_local
                best_seq = seq_local
                print(f"[Local‑Search] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
