    v0 = np.log(start_vec + 1e-30)

        conv = _conv_fft(a, a)

"""
Improved step‑function optimiser.

The optimiser builds a large, diverse pool of normalised seed vectors,
keeps the best few, and refines them with a high‑precision L‑BFGS‑B
optimiser (working on the log‑softmax parametrisation, i.e. on the
probability simplex).  After the quasi‑Newton phase a cheap
sub‑gradient mirror‑descent (exponentiated‑gradient) and a
simulated‑annealing polishing stage are applied.  The whole search is
time‑boxed (default 1000 s) and returns the best sequence seen.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps ℝⁿ to the probability simplex."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (all are returned normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """a_i = r**i , 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^(-p) , p>0."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine density  f(x)=1/(π√{x(1‑x)})  on (0,1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-16)
    s = a.sum()
    return a / s


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta(α,β) density discretised on the grid i+½ / n."""
    if a <= 0.0 or b <= 0.0:
        return _uniform(n)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a_vals = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    a_vals[np.isnan(a_vals)] = 0.0
    s = a_vals.sum()
    return a_vals / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float,
                   rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet draw."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _random_blocks(n: int, max_blocks: int,
                   rng: np.random.Generator) -> np.ndarray:
    """Simple block‑wise pattern."""
    k = rng.integers(2, max_blocks + 1)
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos:pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()

def _generate_seeds(n: int, rng: np.random.Generator,
                    previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))
    seeds.append(_arcsine(n))

    # geometric – fine grid
    for r in np.linspace(0.80, 0.998, 15):
        seeds.append(_geom_decrease(n, r))

    # power‑law – fine grid
    for p in np.arange(0.2, 4.01, 0.2):
        seeds.append(_power_decrease(n, p))

    # Beta‑shapes (different edge‑behaviour)
    beta_params = [
        (0.3, 0.3), (0.5, 0.5), (0.8, 0.8),
        (0.5, 2.0), (2.0, 0.5), (1.0, 1.0)
    ]
    for a, b in beta_params:
        seeds.append(_beta_shape(n, a, b))

    # random Dirichlet draws (different concentrations)
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (5, 10, 20, 40):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # block‑wise patterns
    for max_blocks in (3, 5, 7, 10):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # previous best (if supplied and compatible)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Surrogate “objective” – 2·n·max(conv)  (a is assumed to sum to 1)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) for a probability vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _lbfgs_func(v: np.ndarray, n: int) -> float:
    """Function value for L‑BFGS (2·n·max(conv))."""
    a = _softmax(v)
    conv = _conv_fft(a, a)
    return 2.0 * n * float(np.max(conv))


def _lbfgs_grad(v: np.ndarray, n: int) -> np.ndarray:
    """
    Gradient of the objective w.r.t. the softmax logits.
    Uses the (sub‑)gradient of the true max‑convolution entry.
    """
    a = _softmax(v)
    conv = _conv_fft(a, a)
    k = int(np.argmax(conv))

    # sub‑gradient of the k‑th convolution entry w.r.t. a
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    idx_i = np.arange(i_start, i_end + 1, dtype=int)
    rev_idx = k - idx_i
    g[idx_i] = 2.0 * a[rev_idx]

    # convert to gradient in log‑softmax space
    dot = np.dot(g, a)
    return a * (g - dot)


def _lbfgs_optimize(start_vec: np.ndarray, n: int,
                    maxfun: int = 1500) -> tuple[np.ndarray, float]:
    """
    L‑BFGS‑B optimiser on the true max‑convolution objective.
    Uses the log‑softmax parametrisation to stay on the simplex.
    """
    v0 = np.log(start_vec + 1e-30)
    # fmin_l_bfgs_b expects f(x) and fprime(x) separately and can pass extra args
    x_opt, f_opt, _ = scipy.optimize.fmin_l_bfgs_b(
        func=_lbfgs_func,
        x0=v0,
        fprime=_lbfgs_grad,
        args=(n,),
        maxfun=maxfun,
        disp=False
    )
    a_opt = _softmax(x_opt)
    return a_opt, _objective(a_opt)


def _subgradient_refine(start_seq: np.ndarray, n: int,
                        deadline: float,
                        max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient for the active index k
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 500000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Small‑neighbourhood SA that reshuffles the mass of a few coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # pick 3‑7 indices and randomly re‑allocate their total mass
        k = rng.integers(3, 8)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (as a plain list of floats) discovered within
    the allotted time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # 5 s safety margin

    # ------------------------------------------------------------------
    # Length of the vector – 4000 gives a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Retrieve any previous solution (if present)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # Stage 1 – generate a rich seed pool and keep the best few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(8, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – L‑BFGS‑B refinement of the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- L‑BFGS pass {idx+1}/{len(best_seeds)} ---")
        seq_lbfgs, val_lbfgs = _lbfgs_optimize(seed_vec, n, maxfun=1500)
        print(f"[L‑BFGS] seed #{idx+1} result = {val_lbfgs:.6f}")
        if val_lbfgs < global_best_val:
            global_best_seq = seq_lbfgs.copy()
            global_best_val = val_lbfgs
            print(f"[overall] new global best = {val_lbfgs:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exponentiated‑gradient sub‑gradient polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient polishing ---")
        seq_sub, val_sub = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=200000
        )
        print(f"[SubGrad] best = {val_sub:.6f}")
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing final polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=500000,
            init_temp_factor=0.07
        )
        print(f"[SA] best = {val_sa:.6f}")
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Optional ordering test (descending / ascending)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for ordering in ("desc", "asc"):
            cand = np.sort(global_best_seq)[::-1] if ordering == "desc" else np.sort(global_best_seq)
            cand_val = _objective(cand)
            if cand_val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = cand_val
                print(f"[Reorder] new global best with {ordering} order = {cand_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (call evaluate_sequence once)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```

This implementation follows the required constraints, introduces a deterministic L‑BFGS‑B phase (which typically yields a lower max‑convolution than pure Adam), enriches the seed pool with several analytic families (including the arcsine and Beta shapes), and retains the proven sub‑gradient and simulated‑annealing polishing steps. The whole search respects the 1000 s time budget and returns the best step‑function found.
