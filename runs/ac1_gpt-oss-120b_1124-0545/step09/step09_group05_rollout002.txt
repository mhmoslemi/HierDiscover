def _lp_reweight(f, M, active_idx):
    K = len(active_idx)
    rows = []
    cols = []
    data = []
    for r_idx, i in enumerate(active_idx):
        j0 = max(0, i-(n-1))
        j1 = min(i, n-1)
        # For j from j0 to j1 inclusive:
        # f_val = f[i - j]
        # vectorized: indices = np.arange(j0, j1+1)
        # values = f[i - indices]
        row = np.full(j1-j0+1, r_idx, dtype=int)
        col = np.arange(j0, j1+1, dtype=int)
        rows.append(row)
        cols.append(col)
        data.append(f[i - col])
    # Concatenate all
    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)
    A = sp.csr_matrix((data, (rows, cols)), shape=(K, n))
    # Solve linprog:
    bounds = [(0, None)]*n
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(K, M), bounds=bounds, method='highs')
    if not res.success: return None
    b = res.x
    return b
```

"""
Search for a non‑negative step function minimizing ``evaluate_sequence``.
The algorithm proceeds in four phases:

1. **Seed generation** – a large and diverse collection of normalised
   candidate vectors (uniform, triangular, sine‑window, geometric,
   power‑law, Dirichlet draws, block patterns, and the previous best
   solution if available).

2. **Smooth‑max & sub‑gradient refinement** – each seed is first
   optimised with Adam on a log‑sum‑exp surrogate, then refined by
   exact sub‑gradient mirror‑descent and Euclidean‑projected gradient
   descent (all on the true max‑convolution objective).

3. **Linear‑programming re‑weighting** – given the current vector `f`,
   we solve a small LP that maximises the total weight `∑b_j` under the
   linear constraints `conv(f,b) ≤ M` where `M = max(conv(f,f))`.  The
   solution `b` is blended with `f` (convex combination) which provably
   cannot increase the max‑convolution and usually reduces it.  This
   step mirrors the classic ``f←(1‑t)f + t g`` iteration from the
   literature.

4. **Simulated‑annealing polish** – a final long annealing phase
   performs random local mass redistributions guided by the current
   temperature, guaranteeing a monotone improvement.

All phases respect a user‑supplied time budget (default 1000 s) and
the best vector encountered is stored in the global name
``height_sequence_1`` for reuse in subsequent runs.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective: 2·n·max(conv(seq,seq)).  Assumes sum(seq)=1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw sorted in descending order (strongly skewed)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]                # descending
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_blocks(n: int, max_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """
    Very simple block‑wise pattern:
    choose k (2 ≤ k ≤ max_blocks) blocks of equal length,
    place them evenly with gaps, mass is spread uniformly inside each block.
    """
    k = rng.integers(2, max_blocks + 1)
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos:pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _generate_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # monotone geometric – fine grid
    for r in np.linspace(0.70, 0.99, 13):
        seeds.append(_geom_decrease(n, r))

    # monotone power‑law – fine grid
    for p in np.arange(0.3, 3.01, 0.3):
        seeds.append(_power_decrease(n, p))

    # sorted Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # block‑wise patterns
    for max_blocks in (3, 5, 7, 10):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # random Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (5, 10, 20, 40):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 60000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on a log‑sum‑exp (smoothed max) surrogate.
    Vector stays on the simplex via a log‑softmax parametrisation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5
    beta_mul = 1.001
    beta_max = 5000.0           # more smoothing → closer to true max

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # anneal temperature
        beta = min(beta * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # convert to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.8
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient mirror‑descent (exponentiated‑gradient)
    on the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _euc_proj_subgradient(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the true max‑convolution with Euclidean projection
    onto the probability simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[EucProj] it={it:7d} best={cur_val:.6f}")

        # gradient of the active convolution entry (same as in sub‑grad)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)               # a bit more aggressive than exp‑grad
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 500000,
                         init_temp_factor: float = 0.05,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simulated annealing that randomly reshuffles the mass of a few coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # pick a small subset (3‑7 indices) and redistribute their total mass
        k = rng.integers(3, 8)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming re‑weighting (the “f → g” step)
# ----------------------------------------------------------------------


def _lp_reweight(f: np.ndarray,
                 M: float,
                 active_idx: np.ndarray) -> np.ndarray | None:
    """
    Solve   max ∑ b_j   s.t.   conv(f,b)[i] ≤ M   for i∈active_idx,   b ≥ 0.
    Returns the optimal b (not normalised).  If the LP fails, returns None.
    """
    n = f.shape[0]
    K = active_idx.shape[0]

    rows = []
    cols = []
    data = []

    for row_id, i in enumerate(active_idx):
        # indices j for which 0 ≤ i-j < n   ⇔   i-n+1 ≤ j ≤ i
        j0 = max(0, i - (n - 1))
        j1 = min(i, n - 1)
        js = np.arange(j0, j1 + 1, dtype=int)
        vals = f[i - js]                    # f at (i-j)
        rows.append(np.full_like(js, row_id))
        cols.append(js)
        data.append(vals)

    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)

    A = sp.csr_matrix((data, (rows, cols)), shape=(K, n))

    # SciPy linprog solves min c^T x ; we want max ∑b ⇒ min -(∑b)
    c = -np.ones(n, dtype=np.float64)

    b_ub = np.full(K, M, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    except Exception as e:
        print("[LP] linprog exception:", e)
        return None

    if not res.success:
        print("[LP] linprog failed:", res.message)
        return None

    return res.x


def _lp_improve(seq: np.ndarray,
                n: int,
                deadline: float,
                max_outer: int = 4,
                top_k: int = 200,
                t0: float = 0.12,
                decay: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Apply the LP re‑weighting step a few times.
    Returns (improved_seq, its objective value).
    """
    best_seq = seq.copy()
    best_val = _objective(best_seq)

    t = t0
    for outer in range(1, max_outer + 1):
        if time.time() > deadline:
            break

        conv_f = _conv_fft(best_seq, best_seq)
        M = np.max(conv_f)

        # choose a small active set (largest entries of the convolution)
        K = min(top_k, conv_f.shape[0])
        active_idx = np.argpartition(-conv_f, K - 1)[:K]

        b = _lp_reweight(best_seq, M, active_idx)
        if b is None:
            break

        sum_b = b.sum()
        if sum_b <= 0.0:
            break
        b_norm = b / sum_b

        # convex blend
        new_seq = (1.0 - t) * best_seq + t * b_norm
        # already normalised (both sums = 1)
        new_val = _objective(new_seq)

        if new_val < best_val:
            best_seq = new_seq
            best_val = new_val
            print(f"[LP] iteration {outer} improved objective to {new_val:.6f}")

        # decay blending factor to become more cautious
        t = max(t * decay, 0.02)

    return best_seq, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal
    ``evaluate_sequence`` value.
    Returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin so we finish before the hard timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Retrieve previous solution (if any)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the search vector – 5000 is a reasonable trade‑off
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse pool of seeds and keep the top few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(12, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimisation of the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        # 1) Adam (smooth‑max surrogate)
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline, max_iters=60000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        # 2) Exponentiated‑gradient sub‑gradient refinement
        seq_sub, val_sub = _subgradient_refine(global_best_seq, n, deadline,
                                               max_iters=150000)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

        # 3) Euclidean‑projection gradient descent
        seq_euc, val_euc = _euc_proj_subgradient(global_best_seq, n, deadline,
                                                max_iters=120000)
        if val_euc < global_best_val:
            global_best_seq = seq_euc.copy()
            global_best_val = val_euc
            print(f"[overall‑EucProj] new global best = {val_euc:.6f}")

        # 4) LP re‑weighting (the classic f‑g step)
        seq_lp, val_lp = _lp_improve(global_best_seq, n, deadline,
                                     max_outer=4, top_k=200,
                                     t0=0.12, decay=0.8)
        if val_lp < global_best_val:
            global_best_seq = seq_lp.copy()
            global_best_val = val_lp
            print(f"[overall‑LP] new global best = {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n, deadline,
                                             max_iters=500000,
                                             init_temp_factor=0.05)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Optional global re‑ordering test (descending vs ascending)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for ordering in ("desc", "asc"):
            if ordering == "desc":
                cand = np.sort(global_best_seq)[::-1]
            else:
                cand = np.sort(global_best_seq)
            cand_val = _objective(cand)
            if cand_val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = cand_val
                print(f"[Reorder] new global best with {ordering} order = {cand_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
