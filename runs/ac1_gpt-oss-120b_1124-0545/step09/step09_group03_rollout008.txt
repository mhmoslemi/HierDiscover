def _hill_climb(start_seq, deadline, init_sigma=0.5, min_sigma=1e-6):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    sigma = init_sigma
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline:
        it += 1
        # propose new sequence by adding gaussian noise to logits
        logits = np.log(a + 1e-30)
        perturb = rng.normal(scale=sigma, size=logits.shape)
        new_logits = logits + perturb
        # compute new sequence
        new_a = np.exp(new_logits)
        new_a = new_a / new_a.sum()
        val = _objective(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
            # optionally reduce sigma for fine search
            sigma = max(sigma * 0.9, min_sigma)
            print(f"[HillClimb] it={it} val={val:.6f} sigma={sigma:.6e}")
        else:
            # maybe occasionally accept worse to escape local minima? 
            # For simplicity, reject.
            pass
        # Decrease sigma gradually if not improved
        if it % 1000 == 0:
            sigma = max(sigma * 0.99, min_sigma)
        # early exit if sigma very small and no improvement for many steps
        if sigma < min_sigma + 1e-12:
            break
    return best_seq, best_val

"""
Search for a non‑negative step‑function that minimises `evaluate_sequence`.

Algorithm:
1. Load any previously found sequence and normalise.
2. Explore analytic families (uniform, arcsine, beta, sin^p) and
   smoothed/arcsine‑ε variants for several lengths.
3. Heavy random sampling: Dirichlet draws and random beta‑mixtures.
4. Refine the best candidates with a smooth‑max Adam optimiser
   (log‑softmax parametrisation, annealed temperature up to β=500).
5. Local stochastic hill‑climbing on the log‑it space.
6. Aggressive KL‑mirror‑descent polishing.
All phases respect a global time budget (default 1000 s)."""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, pi, log, exp

# Optional SciPy for L‑BFGS (used if available)
try:
    from scipy.optimize import minimize
except Exception:  # pragma: no cover
    minimize = None

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a normalised sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# ---------------------------------------------------------------
# Analytic and parametric families
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    w /= w.sum()
    return w

def _arcsine_eps_shape(n: int, eps: float) -> np.ndarray:
    """Arcsine with a small epsilon to smooth the ends."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt((i + eps) * (n + 0.5 - i + eps))
    w /= w.sum()
    return w

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid (shifted by 0.5 to avoid end‑points)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _sin_power_shape(n: int, p: float) -> np.ndarray:
    """Shape ∝ sin(π·(i+0.5)/n)^p, symmetric and zero at the ends."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n) ** p
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _smooth_shape(base: np.ndarray, sigma: float, rng: np.random.Generator) -> np.ndarray:
    """Gaussian smoothing of a base shape (same length, mode='same')."""
    # make a reasonably sized kernel
    L = max(3, int(8 * sigma) | 1)  # odd length
    x = np.arange(L, dtype=np.float64) - (L // 2)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(base, kernel, mode='same')
    smoothed = np.clip(smoothed, 0.0, None)
    s = smoothed.sum()
    if s == 0.0:
        return base
    return smoothed / s

def _beta_mix_uniform(n: int, alpha: float, beta: float, lam: float) -> np.ndarray:
    """Mixture λ·Uniform + (1‑λ)·Beta(α,β)."""
    beta_seq = _beta_shape(n, alpha, beta)
    uni_seq = _uniform(n)
    w = lam * uni_seq + (1.0 - lam) * beta_seq
    return w / w.sum()

# ---------------------------------------------------------------
# Random sampling utilities
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# Optimisation primitives
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    max_iter: int = 8_000_000,
    lr0: float = 0.025,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam on
    the log‑softmax (logits) parametrisation.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max
        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum

        # convolution
        conv = _conv_fft(a, a)  # length = 2n‑1

        # temperature schedule (linear annealing)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.7))

        # smooth‑max via log‑sum‑exp
        cmax = np.max(conv)
        exp_beta = np.exp(beta * (conv - cmax))
        p = exp_beta / exp_beta.sum()   # soft weights, sum=1

        # gradient w.r.t a
        conv_pa = _conv_fft(p, a[::-1])   # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule ∝ 1/√it
        lr = lr0 / sqrt(it)

        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:  # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            # early break if we achieve a very low bound
            if best_val < 1.45:
                break

    return best_seq, best_val

def _hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    init_sigma: float = 0.5,
    min_sigma: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing in log‑it space.
    Random Gaussian perturbations are applied to the logits; improvements
    are kept and the step‑size is annealed.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    sigma = init_sigma
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        logits = np.log(a + 1e-30)
        perturb = rng.normal(scale=sigma, size=logits.shape)
        new_logits = logits + perturb
        new_a = np.exp(new_logits)
        new_a /= new_a.sum()
        val = _objective(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
            sigma = max(sigma * 0.9, min_sigma)
            print(f"[HillClimb] it={it:7d} val={val:.6f} sigma={sigma:.2e}")
        # occasional decay of sigma even without improvement
        if (it & 1023) == 0:
            sigma = max(sigma * 0.99, min_sigma)
            if sigma <= min_sigma:
                break
    return best_seq, best_val

def _refine_aggregated_subgradient(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* indices whose
    convolution entry is within a relative tolerance of the current maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")
            if time.time() > deadline - 1.0:
                break
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm combines analytic seeds, massive random sampling,
    smooth Adam optimisation, stochastic hill‑climbing and a final aggregated
    mirror‑descent polishing.  It respects the total time budget.
    Returns the best sequence as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Load a previously stored solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            if prev_arr.size > 0 and np.all(prev_arr >= 0):
                previous = prev_arr / prev_arr.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore (FFT‑friendly)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 4000, 5000, 6000, 8000, 10000, 12000, 16000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic families and cheap parametric sweeps
    # ------------------------------------------------------------------
    # For speed we cap the number of parametrised samples per length
    param_samples_per_len = 80
    for n in lengths:
        # uniform
        seq = _uniform(n)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Seed] n={n:5d} type=uniform   val={val:.6f}")

        # pure arcsine
        seq = _arcsine_shape(n)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Seed] n={n:5d} type=arcsine   val={val:.6f}")

        # arcsine with epsilon smoothing
        for _ in range(4):
            eps = rng.uniform(0.0, 5.0)  # epsilon ∈ [0,5]
            seq = _arcsine_eps_shape(n, eps)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] n={n:5d} type=arc_eps   eps={eps:.2f} val={val:.6f}")

        # random beta shapes
        for _ in range(param_samples_per_len // 4):
            a_par = 10 ** rng.uniform(-1.0, 1.0)   # α ∈ [0.1,10]
            b_par = 10 ** rng.uniform(-1.0, 1.0)   # β ∈ [0.1,10]
            seq = _beta_shape(n, a_par, b_par)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] n={n:5d} type=beta   α={a_par:.3f} β={b_par:.3f} val={val:.6f}")

        # beta‑mix‑uniform
        for _ in range(param_samples_per_len // 8):
            a_par = 10 ** rng.uniform(-1.0, 1.0)
            b_par = 10 ** rng.uniform(-1.0, 1.0)
            lam = rng.random()
            seq = _beta_mix_uniform(n, a_par, b_par, lam)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] n={n:5d} type=beta_mix λ={lam:.2f} α={a_par:.3f} β={b_par:.3f} val={val:.6f}")

        # sin‑power shapes
        for _ in range(param_samples_per_len // 8):
            p = rng.uniform(0.5, 5.0)
            seq = _sin_power_shape(n, p)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] n={n:5d} type=sin_pow p={p:.2f} val={val:.6f}")

        # optionally reuse previous solution if it matches this length
        if previous is not None and previous.shape[0] == n:
            val = _objective(previous)
            if val < best_val:
                best_val, best_seq = val, previous.copy()
                print(f"[Seed] n={n:5d} type=previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive random sampling (Dirichlet & beta mixtures)
    # ------------------------------------------------------------------
    now = time.time()
    remaining = deadline - now
    sampling_time = remaining * 0.35
    sample_deadline = now + sampling_time

    alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet draw with random concentration
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Random‑Dir] n={n:5d} α={alpha:.3f} val={val:.6f}")

        # Random beta mixture (beta + uniform)
        a_par = 10 ** rng.uniform(-1.0, 1.0)
        b_par = 10 ** rng.uniform(-1.0, 1.0)
        lam = rng.random()
        seq = _beta_mix_uniform(n, a_par, b_par, lam)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Random‑BetaMix] n={n:5d} λ={lam:.2f} α={a_par:.3f} β={b_par:.3f} val={val:.6f}")

        # Early stop if we've already crossed a good threshold
        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – smooth Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # Keep a few top candidates for Adam restarts
        candidates = [best_seq]
        # Generate a couple of fresh random starts with same length
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        # Allocate ~25 % of the remaining budget to Adam
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=500.0,
                max_iter=6_000_000,
                lr0=0.025,
            )
            if val_adam < best_val:
                best_val, best_seq = val_adam, seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – stochastic hill‑climbing local search
    # ------------------------------------------------------------------
    if best_seq is not None:

"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Load any previously stored solution and normalise it.
2. Generate a broad set of analytic / parametric families (uniform,
   arcsine, arcsine‑ε, beta, beta‑mixed‑uniform, sin‑power, Gaussian‑smoothed
   arcsine) for several lengths (up to 16000).  Keep the best.
3. Heavy random sampling: Dirichlet draws with many concentration values
   and random beta‑mix‑uniform shapes.
4. Refine a few top candidates with a smooth‑max Adam optimiser
   (log‑softmax parametrisation, temperature annealed up to β=500).
5. Stochastic hill‑climbing on the log‑it space (Gaussian perturbations,
   annealed step‑size) starting from the best Adam result.
6. Aggressive KL‑mirror‑descent polishing (aggregated active set).
All phases respect a global time budget (default 1000 s).  The function
returns the best sequence found as a plain list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, pi

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a normalised sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# ---------------------------------------------------------------
# Analytic / parametric families
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return w / w.sum()

def _arcsine_eps_shape(n: int, eps: float) -> np.ndarray:
    """Arcsine with a small epsilon to smooth the ends."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt((i + eps) * (n + 0.5 - i + eps))
    return w / w.sum()

def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _beta_mix_uniform(n: int, alpha: float, beta: float, lam: float) -> np.ndarray:
    """Mixture λ·Uniform + (1‑λ)·Beta(α,β)."""
    return (lam * _uniform(n) + (1.0 - lam) * _beta_shape(n, alpha, beta))

def _sin_power_shape(n: int, p: float) -> np.ndarray:
    """Shape ∝ sin(π·(i+0.5)/n)^p, zero at the ends."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = np.sin(pi * i / n) ** p
    total = w.sum()
    if total == 0.0:
        return _uniform(n)
    return w / total

def _smooth_shape(base: np.ndarray, sigma: float) -> np.ndarray:
    """Gaussian smoothing of a base shape (mode='same')."""
    L = max(3, int(8 * sigma) | 1)  # odd kernel length
    x = np.arange(L, dtype=np.float64) - (L // 2)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(base, kernel, mode='same')
    smoothed = np.clip(smoothed, 0.0, None)
    s = smoothed.sum()
    if s == 0.0:
        return base
    return smoothed / s

# ---------------------------------------------------------------
# Random utilities
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s

# ---------------------------------------------------------------
# Optimisation primitives
# ---------------------------------------------------------------

def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    max_iter: int = 6_000_000,
    lr0: float = 0.025,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑softmax parametrisation with annealed temperature."""
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)                 # length 2n‑1
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.7))

        cmax = np.max(conv)
        exp_beta = np.exp(beta * (conv - cmax))
        p = exp_beta / exp_beta.sum()          # smooth‑max weights

        conv_pa = _conv_fft(p, a[::-1])        # length 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:      # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.45:
                break
    return best_seq, best_val

def _hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    init_sigma: float = 0.5,
    min_sigma: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """Simple stochastic hill‑climbing in log‑it space."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    sigma = init_sigma
    rng = np.random.default_rng()
    it = 0
    while time.time() < deadline:
        it += 1
        logits = np.log(a + 1e-30)
        perturb = rng.normal(scale=sigma, size=logits.shape)
        new_logits = logits + perturb
        new_a = np.exp(new_logits)
        new_a /= new_a.sum()
        val = _objective(new_a)
        if val < best_val:
            best_val, best_seq, a = val, new_a.copy(), new_a
            sigma = max(sigma * 0.9, min_sigma)
            print(f"[HillClimb] it={it:7d} val={val:.6f} sigma={sigma:.2e}")
        if (it & 1023) == 0:
            sigma = max(sigma * 0.99, min_sigma)
            if sigma <= min_sigma:
                break
    return best_seq, best_val

def _refine_aggregated_subgradient(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """KL‑mirror descent using all indices within a relative tolerance of the max."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")
            if time.time() > deadline - 1.0:
                break
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
