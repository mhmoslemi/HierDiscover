"""
optimiser_step_function_v5.py

A time‑bounded optimiser that aims at the target bound 1.503​ by

1. Using a larger problem size (n≈8000).  With the same collision bound C=4
   the theoretical optimum 2·n·C/m²≈1 is reachable when the support size
   approaches √(C·(2n‑1))≈250.

2. Aggressively searching for a large low‑collision support set S⊂{0,…,n‑1}.
   A fast greedy construction is run many times; after each run a simple
   local‑search pass tries to insert any still‑missing index without
   violating the “ordered pair‑sum appears at most C times’’ constraint.
   The best mask found is kept.

3. Building a pool of seeds:
      * uniform distribution on the low‑collision support,
      * random Dirichlet weights on that support,
      * a variety of classic window families (Gaussian, Hann, etc.)
      * the previous global solution if present.

4. Optimising the weights on the chosen support with three phases:
      – Adam on a smoothed‑max surrogate,
      – Exact sub‑gradient (mirror‑descent) on the true objective,
      – Optional simulated‑annealing fine‑tuning.

All phases respect the global 1000 s budget.  The final vector is
normalised to have total mass 1 (the evaluate_sequence routine does *not*

# Basic utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ------------------------------------------------------------
# Window families (used as auxiliary seeds)
# ------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a /= a.sum()
    return a


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()

def _greedy_low_collision(n: int, max_ordered: int, rng: np.random.Generator):
    """Return a mask and sum‑count array produced by a single greedy pass."""
    mask = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)

    order = rng.permutation(n)
    for cand in order:
        # self‑pair
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue
        existing = np.nonzero(mask)[0]
        if existing.size:
            sums = cand + existing
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue
        # accept
        mask[cand] = True
        sum_counts[2 * cand] += 1
        if existing.size:
            sum_counts[sums] += 2
        # early stop when we cannot increase further (optional)
    return mask, sum_counts


def _improve_mask(mask: np.ndarray, sum_counts: np.ndarray,
                  max_ordered: int, rng: np.random.Generator):
    """Local search: repeatedly try to insert any still‑available index."""
    n = mask.size
    changed = True
    while changed:
        changed = False
        candidates = np.where(~mask)[0]
        rng.shuffle(candidates)
        for cand in candidates:
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue
            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue
            # insertion is feasible
            mask[cand] = True
            sum_counts[2 * cand] += 1
            if existing.size:
                sum_counts[sums] += 2
            changed = True
    return mask, sum_counts


def _search_low_collision_set(
    n: int,
    target_m: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """
    Greedy + local‑search construction of a set S⊂{0,…,n‑1} such that every
    ordered pair‑sum i+j (i,j∈S) occurs at most ``max_ordered`` times.
    Returns the index array of the best set found before the deadline.
    """
    best_mask = None
    best_size = 0
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        mask, sum_counts = _greedy_low_collision(n, max_ordered, rng)
        mask, sum_counts = _improve_mask(mask, sum_counts, max_ordered, rng)
        cur = mask.sum()
        if cur > best_size:
            best_size = cur
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:4d} size {cur}")
            if cur >= target_m:
                break
    if best_mask is None:
        return np.array([], dtype=int)
    return np.where(best_mask)[0]


# ------------------------------------------------------------
# Optimisation primitives
# ------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _adam_optimize(seed: np.ndarray, n: int, deadline: float,
                   max_iters: int = 40000):
    """Adam on a smoothed‑max surrogate of max(conv(a,a))."""
    v = np.log(seed + 1e-30)          # logits
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    beta = 1.5
    beta_mul = 1.007
    best_seq = None
    best_val = float("inf")
    t = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # smoothed max over convolution
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·(a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)         # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # chain rule through softmax
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam updates
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)
        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)
        lr = 0.7 / np.sqrt(t)
        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)
    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 800_000):
    """Exponentiated‑gradient on the exact nonsmooth objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")
    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Simple Metropolis‑Hastings annealing on the true objective."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        if it % 1000 == 0:
            # occasional global Gaussian jump
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")
    return best, best_val


# ------------------------------------------------------------
# Seed construction
# ------------------------------------------------------------
def _make_seeds(n: int, previous: np.ndarray | None,
                lowcoll_idx: np.ndarray | None,
                rng: np.random.Generator) -> list[np.ndarray]:
    """Collect a diverse set of initial probability vectors."""
    seeds = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_gaussian(n, sigma_factor=0.12))
    seeds.append(_gaussian(n, sigma_factor=0.22))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for _ in range(5):
        seeds.append(_random_softmax(n, rng))

    # low‑collision uniform support
    if lowcoll_idx is not None and lowcoll_idx.size:
        a = np.zeros(n, dtype=np.float64)
        a[lowcoll_idx] = 1.0 / lowcoll_idx.size
        seeds.append(a)
        # random Dirichlet perturbations on that support
        for _ in range(4):
            w = rng.dirichlet(np.ones(lowcoll_idx.size))
            a2 = np.zeros_like(a)
            a2[lowcoll_idx] = w
            a2 /= a2.sum()
            seeds.append(a2)

    # previous solution (if same length)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # ensure proper normalisation and non‑negativity
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned

# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  Returns the best sequence found as a list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # --------------------------------------------------------
    # Problem size – we use a larger n to obtain a tighter theoretical bound
    # --------------------------------------------------------
    n = 8000
    print(f"\n=== Optimising length {n} ===")

    rng = np.random.default_rng(seed + 1)

    # --------------------------------------------------------
    # Phase 1 – search for a large low‑collision support set
    # --------------------------------------------------------
    # theoretical max m = sqrt(C * (2n‑1)) with C=4
    theoretical_m = int(math.sqrt(4 * (2 * n - 1)))
    target_m = int(0.97 * theoretical_m)            # aim for ~97 % of bound
    max_ordered = 4
    t_set_deadline = start_time + 0.28 * (deadline - start_time)   # ~28 % budget
    lowcoll_idx = _search_low_collision_set(
        n=n,
        target_m=target_m,
        max_ordered=max_ordered,
        rng=rng,
        deadline=t_set_deadline,
    )
    if lowcoll_idx.size:
        print(f"[Info] low‑collision set size {lowcoll_idx.size} (target {target_m})")
    else:
        print("[Info] low‑collision search failed – will fall back to generic seeds")
        lowcoll_idx = None

    # --------------------------------------------------------
    # Phase 2 – build a pool of seeds
    # --------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    seeds = _make_seeds(n, previous, lowcoll_idx, rng)

    # quick evaluation and keep top seeds
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(15, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] initial best objective = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Phase 3 – Adam optimisation on the top seeds
    # --------------------------------------------------------
    t_adam_deadline = start_time + 0.55 * (deadline - start_time)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > t_adam_deadline:
            break
        remaining = t_adam_deadline - time.time()
        # estimated time per Adam iteration ≈ 0.0012 s
        max_iters = max(1, int(min(60000, remaining / 0.0012)))
        seq, val = _adam_optimize(seed_vec, n, t_adam_deadline, max_iters=max_iters)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # --------------------------------------------------------
    # Phase 4 – Exact sub‑gradient refinement (majority of remaining time)
    # --------------------------------------------------------
    if time.time() < deadline - 12.0:
        sub_deadline = deadline - 7.0
        remaining = sub_deadline - time.time()
        max_sub_iters = max(1, int(min(1_200_000, remaining / 0.0014)))
        seq, val = _subgradient_refine(global_best_seq, n, sub_deadline,
                                       max_iters=max_sub_iters)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall SubGrad] best={val:.6f}")

    # --------------------------------------------------------
    # Phase 5 – Simulated‑annealing fine‑tuning (all leftover time)
    # --------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Anneal] best={val:.6f}")

    # --------------------------------------------------------
    # Final normalisation and safety checks
    # --------------------------------------------------------
    final_seq = np.maximum(global_best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} (length {len(final_seq)})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
