"""
Binary‑mask Simulated Annealing for the max‑convolution objective.

The evaluation function is scale‑invariant, so we restrict ourselves to
probability vectors (sum = 1).  By fixing the weight of every non‑zero
entry to the same value w = 1/m (m = number of ones) the objective
becomes

    2·n·max_i ( conv(mask,mask)[i] ) / m²

where ``mask`` is a {0,1} vector.  The convolution entries are exactly the
counts of ordered pairs (j,k)∈S×S with j+k=i (``S`` = support of the
mask).  A random dense mask already spreads these counts fairly evenly,
giving values around 1.0.  We improve further by a simple simulated‑
annealing that repeatedly swaps a 1 ↔ 0 while accepting uphill moves
with a decreasing temperature.

The algorithm:
    1. Choose a length n (4000 – a good trade‑off between speed and
       quality).  Try several densities ρ = m/n.
    2. Initialise a random mask with exactly m = round(ρ·n) ones.
    3. Perform a long SA run (swap‑moves only) while the time budget
       lasts, keeping the best mask seen.
    4. Convert the best mask to a float vector (weight = 1/m) and return
       it.  The global variable ``height_sequence_1`` is updated for
       possible reuse.

All heavy work is done with FFT‑based convolution (O(n log n)) and the
code respects the 1000 s time limit.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective_mask(mask: np.ndarray, m: int, n: int) -> float:
    """
    Exact objective for a binary mask with exactly m ones.
    The mask is 0/1, sum(mask) == m.
    Returns 2·n·max(conv(mask,mask)) / m².
    """
    # convolution of the integer mask (ordered pair counts)
    conv = _conv_fft(mask.astype(np.float64), mask.astype(np.float64))
    # rounding eliminates tiny FFT noise
    max_int = int(round(np.max(conv)))
    return 2.0 * n * max_int / (m * m)


# ----------------------------------------------------------------------
# Simulated‑annealing on a binary mask
# ----------------------------------------------------------------------


def _sa_binary(mask: np.ndarray,
               n: int,
               deadline: float,
               rng: np.random.Generator,
               init_temp_factor: float = 0.04,
               min_temp: float = 1e-12,
               cool_interval: int = 2000,
               cool_factor: float = 0.95) -> tuple[np.ndarray, float]:
    """
    Perform SA on a given binary mask (fixed number of ones).
    Returns the best mask seen and its objective value.
    """
    m = int(mask.sum())
    cur_mask = mask.copy()
    cur_val = _objective_mask(cur_mask, m, n)

    best_mask = cur_mask.copy()
    best_val = cur_val

    # temperature proportional to the current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    # maintain explicit lists of positions for O(1) swaps
    ones = list(np.flatnonzero(cur_mask))
    zeros = list(np.where(cur_mask == 0)[0])

    it = 0
    while time.time() < deadline and T > min_temp:
        it += 1

        # pick a random 1 and a random 0 to swap
        idx_one = rng.integers(len(ones))
        idx_zero = rng.integers(len(zeros))
        pos_one = ones[idx_one]
        pos_zero = zeros[idx_zero]

        # perform the tentative swap
        cur_mask[pos_one] = 0
        cur_mask[pos_zero] = 1

        new_val = _objective_mask(cur_mask, m, n)
        delta = new_val - cur_val

        # Metropolis acceptance
        accept = False
        if delta < 0.0:
            accept = True
        else:
            # probability of accepting uphill move
            prob = math.exp(-delta / T) if T > 0 else 0.0
            if rng.random() < prob:
                accept = True

        if accept:
            cur_val = new_val
            # update the index lists to reflect the swap
            ones[idx_one] = pos_zero
            zeros[idx_zero] = pos_one

            if cur_val < best_val:
                best_val = cur_val
                best_mask = cur_mask.copy()
                # occasional progress output
                if it % 20000 == 0:
                    print(f"[SA] it={it:7d} best={best_val:.6f}")
        else:
            # revert swap
            cur_mask[pos_one] = 1
            cur_mask[pos_zero] = 0

        # cooling schedule
        if it % cool_interval == 0:
            T *= cool_factor

    return best_mask, best_val

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function with a low value of
    ``evaluate_sequence``.  The method builds binary masks of different
    densities and refines them with simulated annealing.  The best mask
    is returned as a list of floats (uniform weight on the support).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 1.0    # safety margin

    # Length of the sequence – 4000 balances speed and solution quality.
    n = 4000

    # ------------------------------------------------------------------
    # Densities to explore (fraction of non‑zero entries)
    # ------------------------------------------------------------------
    densities = [0.25, 0.33, 0.40, 0.50, 0.60, 0.70]
    # Ensure we respect the time budget across densities
    remaining = len(densities)

    global_best_seq = None
    global_best_val = float("inf")

    for rho in densities:
        if time.time() >= deadline:
            break
        remaining -= 1

        m = max(1, int(round(rho * n)))          # number of ones
        # initialise a random mask with exactly m ones
        mask = np.zeros(n, dtype=np.int8)
        mask[rng.choice(n, size=m, replace=False)] = 1

        # allocate a slice of the remaining time for this density
        # (give a generous share, but leave a safety margin for the last run)
        slice_deadline = time.time() + (deadline - time.time()) * (1.0 / max(remaining, 1))

        best_mask, best_val = _sa_binary(
            mask,
            n=n,
            deadline=slice_deadline,
            rng=rng,
            init_temp_factor=0.04,
            min_temp=1e-12,
            cool_interval=3000,
            cool_factor=0.945,
        )

        # Convert the mask to a probability vector (uniform weight on support)
        w = 1.0 / m
        seq = best_mask.astype(np.float64) * w

        # Evaluate with the official evaluator (scale‑invariant, but we do it
        # for a sanity check and to keep the same metric as the competition).
        val = evaluate_sequence(seq.tolist())
        print(f"[Result] density={rho:.2f}, m={m}, obj={val:.6f}")

        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"*** New global best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and expose the result for future runs
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback – uniform distribution (should never happen)
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest value ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
