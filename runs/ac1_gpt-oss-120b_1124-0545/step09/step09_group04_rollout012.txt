The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 ,
\]

where `(a*a)` is the linear convolution of the vector with itself.  
The value is *scale‑invariant*, therefore we can always keep the
vector normalised to `sum(a)=1` and simply minimise  

\[
\Phi(a)=2\,n\;\max_k (a*a)_k .
\]

`Φ` reaches its theoretical lower bound `≈1` when the convolution is
perfectly flat (its total mass is always `1`).  A uniform vector gives
`Φ=2`.  By shaping the distribution we can spread the pairwise
products over many different indices and obtain a much smaller
maximum.  

The optimiser works in three phases

1. **Diverse seed generation** – many analytical families (uniform,
   triangular, sine‑window, symmetric/unsymmetric geometric,
   power‑law, beta, U‑shaped *1/√(i·(n−i))* …) plus random Dirichlet /
   sparse vectors are created and normalised to sum = 1.

2. **Multi‑start exponentiated‑gradient descent** – the best few seeds
   are refined with a mirror‑descent (exponentiated‑gradient) that
   follows a sub‑gradient of the current maximal convolution entry.
   Several starts are tried with different learning‑rate schedules;
   the best result of all runs is kept.

3. **Simulated‑annealing polishing** – a short stochastic local‑search
   reshuffles the mass of a small random subset; it can escape shallow
   local minima found by the gradient phase.

All heavy work (convolution, sub‑gradient) is performed with the FFT
so that a single iteration costs `O(n log n)`.  
The whole procedure respects the 1000 s time budget, stops a few
seconds early, stores the final vector in the global name
`height_sequence_1` and returns it as a plain list ready for
`evaluate_sequence`.

```python
"""
Non‑negative step‑function optimiser (new version).

The optimiser consists of three stages:

1️⃣  Seed generation – a large, diverse collection of analytic families
    (uniform, triangular, sine‑window, geometric decay, power‑law,
    beta, U‑shape 1/√(i·(n‑i)) etc.) and random Dirichlet / sparse vectors.
    All candidates are normalised to sum = 1 and evaluated with the fast
    surrogate  `Φ(a) = 2·n·max(conv(a,a))` computed via FFT.

2️⃣  Multi‑start exponentiated‑gradient (mirror‑descent) – the best
    several seeds are refined by iteratively moving in the direction
    of a *sub‑gradient* of the current maximal convolution entry.
    The update is multiplicative (log‑gradient) which guarantees
    non‑negativity and a projection onto the probability simplex.
    Different learning‑rate schedules are tried; the best sequence from
    all runs is kept.

3️⃣  Simulated‑annealing polishing – a short stochastic local search
    that repeatedly redistributes the mass of a tiny random subset.
    Accept‑reject follows the Metropolis rule with an exponentially
    decreasing temperature.

The final vector is stored in the global variable ``height_sequence_1``
and returned as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    ``seq`` is assumed to be non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _u_shape(n: int) -> np.ndarray:
    """U‑shaped 1/√(i·(n‑i)) – higher at the ends, lower in the centre."""
    i = np.arange(n, dtype=np.float64) + 1.0         # 1 … n
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a dense Dirichlet vector."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector (k non‑zero entries)."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _sparse_uniform(n: int, k: int) -> np.ndarray:
    """Uniform mass on k equally spaced positions."""
    if k <= 0:
        return _uniform(n)
    positions = np.linspace(0, n - 1, k, dtype=int)
    a = np.zeros(n, dtype=np.float64)
    a[positions] = 1.0 / k
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction (large but cheap)
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))
    seeds.append(_u_shape(n))

    # symmetric geometric – fine grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric decay
    for r in np.linspace(0.95, 0.9995, 15):
        seeds.append(_geom_decay(n, r))

    # pure power‑law decay
    for p in np.arange(0.1, 3.01, 0.3):
        seeds.append(_power_decay(n, p))

    # beta family – fine grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # sparse uniform patterns
    for k in (8, 16, 32, 64):
        seeds.append(_sparse_uniform(n, k))

    # random Dirichlet draws – a handful for diversity
    for conc in (0.02, 0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns – very small support
    for k in (3, 5, 10, 20):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient descent (mirror‑descent)
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    objective.  ``lr0`` is the initial learning‑rate factor;
    it decays as 1/√t.  The routine stops when the deadline is reached.
    Returns the best sequence seen and its surrogate objective value.
    """
    a = start_seq.copy()               # already normalised to sum=1
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30                         # for log safety

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full convolution (FFT)
        conv = _conv_fft(a, a)

        # index where the max is attained
        max_idx = int(np.argmax(conv))

        # current surrogate value (exact because sum=1)
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:9d} best≈{best_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)               # size of the random subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA]  it={it:8d} best≈{best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal ``evaluate_sequence``
    value.  The routine is deterministic for a given ``seed`` and respects a
    hard time limit (``budget_s`` seconds).  The final sequence is stored in
    the global name ``height_sequence_1`` so that subsequent runs can reuse it.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # small safety margin so we finish before the hard timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Possible reuse of a previous solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length choice – a compromise between quality and speed
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate many seeds and keep the top few
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(5, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – multi‑start mirror‑descent
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Multi‑start mirror‑descent ---")
        # allocate roughly equal time to each start
        remaining = deadline - time.time()
        per_start = remaining / top_k
        for idx, seed_vec in enumerate(best_seeds):
            this_deadline = time.time() + per_start
            lr0 = 2.0 * (1.0 + 0.5 * (idx % 3))   # a few different starts
            seq_refine, val_refine = _subgradient_refine(
                seed_vec, n, this_deadline, max_iters=3_000_000, lr0=lr0
            )
            if val_refine < global_best_val:
                global_best_seq = seq_refine.copy()
                global_best_val = val_refine
                print(f"[overall‑SubGrad] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=250_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best ≈ {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (optional)
    # ------------------------------------------------------------------
    try:
        # The external evaluate_sequence will be called later; we just
        # make sure the vector is well‑formed.
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose the result for possible reuse in a later run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
```
