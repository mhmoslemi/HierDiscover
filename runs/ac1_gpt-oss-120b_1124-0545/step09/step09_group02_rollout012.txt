"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a list of heights) that
minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

where ``conv`` is the linear convolution (full mode) of the vector
with itself.  The algorithm is deliberately different from the
previous “Hybrid optimiser” in three respects:

1. **Richer seed set** – more parameter sweeps (very fine geometric /
   power‑law grids, extra Beta‑distribution shapes and many random
   Dirichlet draws).

2. **Smoothed exponentiated‑gradient descent** – instead of using only
   the active constraints we weight *all* convolution lags with a
   soft‑max (temperature β).  This yields a smooth surrogate of the
   max‑convolution, whose gradient can be computed by a single FFT‑based
   convolution.  β is slowly increased so that the surrogate converges
   to the true max.

3. **Final annealing** – a short Metropolis–Hastings walk with the
   same move set as the reference code, but started from the best
   vector found by the smoothed EG.

The whole search respects the 1000 s budget, returns the best
sequence found (as a plain list of ``float``) and stores it globally
as ``height_sequence_1`` for potential reuse.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution (real FFT, full mode)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, same as evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple probability‑simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic families (all return a *probability* vector, i.e. sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(α,…,α)."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()

def _make_extended_seeds(
    n: int, previous: np.ndarray | None, rng: np.random.Generator
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # uniform baseline
    seeds.append(_uniform(n))

    # geometric – dense sweep
    for c in np.linspace(0.5, 12.0, 60):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power‑law – dense sweep
    for p in np.linspace(0.0, 4.0, 61):
        seeds.append(_power(n, p))

    # combined geometric‑power (moderate grid)
    for c in np.linspace(0.5, 12.0, 30):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 13):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 12.0, 30):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.9, 15):
        seeds.append(_tukey_window(n, alpha))

    # Gaussian family – finer sweep
    for sigma_factor in np.linspace(0.04, 0.35, 25):
        seeds.append(_gaussian(n, sigma_factor))

    # Beta‑pdf family – extra heavy‑edge range
    for alpha in np.linspace(0.05, 3.0, 35):
        seeds.append(_beta_window(n, alpha))

    # Dirichlet draws – many concentrations
    for alpha in np.logspace(-2, 1, 30):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # random soft‑max draws
    for _ in range(20):
        seeds.append(_random_softmax(n, rng))

    # reuse previous if appropriate
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean up (clip, normalise)
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Smoothed exponentiated‑gradient descent (soft‑max weighting)
# ----------------------------------------------------------------------


def _smooth_exponential_gradient(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on a soft‑max surrogate of the max‑convolution.
    The temperature β starts small (smooth) and is slowly increased,
    turning the surrogate into the true max.  The update is
    multiplicative (exponentiated‑gradient) followed by a projection
    onto the probability simplex.
    Returns the best vector seen and its exact objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    beta = 0.3           # start temperature (smoothness)
    beta_factor = 1.001  # geometric increase each iteration
    base_lr = 0.55

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full convolution
        conv = _conv_fft(a, a)        # length 2n‑1
        max_c = conv.max()

        # soft‑max weights over all lags
        expo = np.exp(beta * (conv - max_c))   # shift for numerical stability
        w = expo / expo.sum()

        # gradient of the smooth surrogate
        rev_a = a[::-1]
        g_full = _conv_fft(rev_a, w)          # length 3n‑2
        g = 2.0 * g_full[n - 1 : n - 1 + n]    # keep central n entries

        # step‑size (decreasing)
        lr = base_lr / math.sqrt(it)

        # exponentiated‑gradient update (mirror descent on the simplex)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # prevent overflow
        a = np.exp(log_a)

        # renormalise – stays on the simplex
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = _uniform(n)

        # increase temperature – focus more on the true maximum
        beta = min(beta * beta_factor, 800.0)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SmoothEG] it={it:7d} val={cur_val:.6f} beta={beta:.2f}")

        # occasional random jitter to escape plateaus
        if it % 4200 == 0:
            noise = np.random.default_rng().normal(scale=0.0012, size=n)
            a = np.clip(a + noise, 0.0, None)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = _uniform(n)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple additive projected‑gradient descent (sub‑gradient of max‑conv)
# ----------------------------------------------------------------------


def _additive_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact (non‑smooth) max‑convolution.
    Uses the same sub‑gradient construction as the reference code.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max(1e-12, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1 : n - 1 + n]

        lr = 0.11 / math.sqrt(it)
        a_next = a - lr * g
        a = _project_simplex(a_next)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AddGD]   it={it:7d} val={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (same move set as reference)
# ----------------------------------------------------------------------


def _anneal_optimize(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int | None = None,
) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with several move types and a slow
    temperature decay.  Returns the best vector encountered.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # three basic move types
        rnd = rng.random()
        if rnd < 0.66:  # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.18 * cur[i]
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif rnd < 0.86:  # swap
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:  # global jitter
            new_seq = cur + rng.normal(scale=0.0015, size=n)
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s > 0:
                new_seq /= s
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # every few thousand steps a stronger jump
        if it % 2600 == 0:
            cand = best + rng.normal(scale=0.0032, size=n)
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Length – 4000 gives a good trade‑off between speed and quality
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load previous construction if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 0) evaluate previous construction (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev]   val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) Build a very rich seed pool and keep the top few
    # ------------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous, rng)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed]   new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Refine each kept seed with smoothed EG + additive GD
    # ------------------------------------------------------------------
    remaining_time = deadline - time.time()
    per_seed_deadline = deadline  # we will reuse the global deadline but
    # keep an eye on the remaining budget inside the loop.

    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 6.0:
            break
        print(f"[refine] start seed {idx}")

        # smoothed exponentiated‑gradient
        seq, val = _smooth_exponential_gradient(seed_vec, n, deadline)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] smoothEG best={val:.6f}")

        # short additive gradient descent to clean up residuals
        seq, val = _additive_descent(seq, n, deadline, max_iters=80000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] additive  best={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing fine‑tuning (uses whatever time remains)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall] anneal best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
