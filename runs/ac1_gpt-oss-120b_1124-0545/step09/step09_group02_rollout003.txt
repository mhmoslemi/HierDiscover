"""
Propose a step‑function (non‑negative vector) that minimises

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The objective is homogeneous, so we work on the probability simplex
(∑a = 1).  The optimiser consists of three phases:

0.  Load the previous best construction (if any) and use it as a start.
1.  Run a smooth‑max gradient descent (Adam) on the full objective.
    The max is approximated by a soft‑max (log‑sum‑exp) whose
    temperature is increased during the run.  After each step we
    project onto the simplex.
2.  With the remaining time perform a cheap random‑local‑search
    (mass‑transfer moves) to squeeze out a few extra improvements.

All heavy lifting (convolutions) is performed with an FFT based
implementation for speed.  The routine respects the 1000 s budget
by checking the deadline in the main loop.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT convolution, simplex projection, objective)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution (full mode) of two real vectors using a real FFT.
    Length of the result is a.size + b.size - 1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    The sequence is assumed to be non‑negative; no clipping is performed.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Optimisation core – smooth‑max (log‑sum‑exp) Adam descent
# ----------------------------------------------------------------------


def _smooth_max_adam(start: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """
    Minimise max(conv(a,a)) on the simplex using an Adam optimiser
    with a gradually sharpening soft‑max.

    Returns the best vector seen and its objective value.
    """
    a = start.copy()
    best_a = a.copy()
    best_val = _objective(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.2                     # base learning rate

    # soft‑max schedule
    lambda_val = 5.0             # start temperature (small → smooth)
    lambda_factor = 1.05         # multiplicative increase
    lambda_step = 1500           # iterations between temperature updates

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1

        # ---------- evaluate ----------
        conv = _conv_fft(a, a)               # full convolution
        max_conv = conv.max()
        cur_val = 2.0 * n * max_conv          # because sum(a)=1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam] it={it:7d} best={cur_val:.6f}")

        # ---------- temperature schedule ----------
        if it % lambda_step == 0:
            lambda_val *= lambda_factor

        # ---------- soft‑max weights ----------
        diff = conv - max_conv                 # ≤ 0
        range_ = max_conv - conv.min()
        if range_ == 0.0:
            w = np.ones_like(conv)
        else:
            w = np.exp(lambda_val * diff / (range_ + 1e-15))
        w_sum = w.sum()
        if w_sum == 0.0:
            w_norm = np.full_like(w, 1.0 / w.size)
        else:
            w_norm = w / w_sum

        # ---------- gradient of smoothed max ----------
        rev_a = a[::-1]
        conv_rev = _conv_fft(rev_a, w_norm)          # length 2n‑1
        grad = 2.0 * conv_rev[n - 1:n - 1 + n]       # same shape as a

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_simplex(a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Tiny random local‑search (mass transfer) used at the end
# ----------------------------------------------------------------------


def _local_search(start: np.ndarray,
                  n: int,
                  deadline: float,
                  max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly transfer a small random fraction
    of mass from one index to another and keep the move if the objective
    improves.  Operates on the simplex via projection.
    """
    rng = np.random.default_rng()
    a = start.copy()
    best_val = _objective(a)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer up to 15 % of source mass
        delta = rng.random() * 0.15 * a[i]
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_simplex(cand)
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            a = cand
            print(f"[Local] it={it:7d} best={val:.6f}")
        else:
            # occasional random restart to escape tiny plateaus
            if rng.random() < 0.0005:
                a = best_a.copy()
    return best_a, best_val

            power = 1.0 / (idx ** p)
            power /= power.sum()
            seeds.append(power)

        # pick the best of them
        best_val = float("inf")
        best_seq = None
        for s in seeds:
            val = _objective(s)
            if val < best_val:
                best_val = val
                best_seq = s.copy()
                print(f"[seed] val={val:.6f}")
        a0 = best_seq

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    print("[phase 1] smooth‑max Adam descent")
    a_opt, val_opt = _smooth_max_adam(a0, n, deadline)

    # ------------------------------------------------------------------
    # Phase 2 – cheap local search if we still have time left
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[phase 2] random local improvement")
        a_opt, val_opt = _local_search(a_opt, n, deadline)

    # ------------------------------------------------------------------
    # Finalise – make sure the vector is non‑negative and clip to the
    # allowed range (the grader will clip anyway, we just stay safe).
    # ------------------------------------------------------------------
    a_opt = np.clip(a_opt, 0.0, None)

    # Store for possible reuse in a future run
    globals()["height_sequence_1"] = a_opt.copy()

    final_val = _objective(a_opt)
    print(f"\n=== Finished ===\nBest objective ≈ {final_val:.6f} (length {len(a_opt)})")
    return a_opt.tolist()
