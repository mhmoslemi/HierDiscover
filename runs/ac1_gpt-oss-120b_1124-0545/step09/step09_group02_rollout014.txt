def _sidon_set_greedy(n, rng):
    # Generate random ordering of indices
    order = rng.permutation(n)
    S = []
    sums = set()
    for i in order:
        # Check if adding i would cause duplicate sums
        conflict = False
        # self sum
        if (i + i) in sums:
            continue
        for j in S:
            s = i + j
            if s in sums:
                conflict = True
                break
        if conflict:
            continue
        # no conflict, add i
        for j in S:
            sums.add(i + j)
        sums.add(i + i)
        S.append(i)
    return np.array(S, dtype=np.int32)

Goal
----
Minimise  2·n·max(conv(a,a)) / (sum(a)**2)  for a non‑negative vector `a`.

Strategy
--------
1.  Build a large pool of start vectors.
    – Classic analytic windows (uniform, Hann, Blackman, …) – already good.
    – Many random sparse vectors whose support is a *Sidon* set
       (all pairwise sums are distinct).  For such a support the
       convolution at any index is the sum of at most one ordered pair,
       hence the maximum value is roughly 2/|S|², often well below 2.
    – Random Dirichlet draws with tiny concentration for extra sparsity.

2.  Keep the best few seeds and refine each of them with two deterministic
    first‑order methods:
        • exponentiated‑gradient (mirror descent on the exact non‑smooth objective)
        • additive projected gradient descent (sub‑gradient of the max‑convolution)

3.  Finish with a simulated‑annealing walk that uses mass‑transfer,
    swaps and Gaussian jitter.  It runs on the remaining time budget.

All heavy work is bound by the given time limit (default 1000 s);
the code stops early if the deadline is reached.

The final sequence is stored in the global ``height_sequence_2`` variable
so that later runs can reuse it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Simple utilities ------------------------------------------------------
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic families (probability vectors, sum = 1) --------------------
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from a Dirichlet distribution."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Sidon set construction (no pairwise sum collisions) ---------------
# ----------------------------------------------------------------------


def _sidon_set_greedy(length: int, rng: np.random.Generator) -> np.ndarray:
    """
    Greedy random construction of a Sidon set S ⊂ {0,…,length‑1}
    (i.e. i+j is unique for unordered pairs).  Returns the index array.
    """
    order = rng.permutation(length)
    S = []
    used_sums = set()
    for idx in order:
        # test whether adding idx would create a duplicate sum
        conflict = (idx + idx) in used_sums
        if not conflict:
            for j in S:
                if (idx + j) in used_sums:
                    conflict = True
                    break
        if conflict:
            continue
        # accept idx
        for j in S:
            used_sums.add(idx + j)
        used_sums.add(idx + idx)
        S.append(idx)
    return np.array(S, dtype=np.int32)


def _uniform_on_support(n: int, support: np.ndarray) -> np.ndarray:
    """Create a probability vector that is uniform on the given support."""
    a = np.zeros(n, dtype=np.float64)
    m = len(support)
    if m:
        a[support] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex --------------------
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, ∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Sub‑gradient (exponentiated‑gradient) refinement --------------------
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 600000) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the exact non‑smooth objective.
    Returns the best vector seen and its objective value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max(rel_tol, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # numeric stability
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Additive projected gradient descent ----------------------------------
# ----------------------------------------------------------------------


def _additive_descent(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 300000) -> tuple[np.ndarray, float]:
    """
    Simple projected gradient descent using a sub‑gradient of the max‑convolution.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max(rel_tol, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.12 / np.sqrt(it)
        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        a = a_next
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AddGD] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing final polishing ----------------------------------
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Metropolis–Hastings walk with several move types.
    Returns the best vector found and its objective.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a small fraction of current objective
    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.68:               # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.18 * cur[i]
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif rnd < 0.86:             # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                        # global Gaussian jitter
            noise = rng.normal(scale=0.0013, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        if it % 2500 == 0:
            jitter = rng.normal(scale=0.0035, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Construction of a rich seed pool ------------------------------------
# ----------------------------------------------------------------------


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse set of start vectors (all normalised to sum = 1).
    """
    rng = np.random.default_rng(12345)
    seeds: list[np.ndarray] = []

    # classic analytic families
    seeds.append(_uniform(n))
    seeds.append(_hann_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_gaussian(n, sigma_factor=0.12))
    seeds.append(_geom(n, r=0.9995))

    # random Dirichlet draws (very sparse)
    for alpha in [0.02, 0.05, 0.1, 0.2]:
        for _ in range(6):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # Sidon‑set based supports
    for _ in range(30):
        supp = _sidon_set_greedy(n, rng)
        if len(supp) >= 30:                     # keep only reasonably large supports
            seeds.append(_uniform_on_support(n, supp))

    # reuse previous construction if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up – guarantee non‑negativity and normalisation
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)

    return cleaned


# ----------------------------------------------------------------------
# Main entry point required by the grader ------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # safety margin

    # --------------------------------------------------------------
    # Length – 4000 offers a good speed / quality trade‑off
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # Try to reuse a construction from a previous run
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_2")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # --------------------------------------------------------------
    # 0) evaluate previous construction (if any)
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] objective={val:.6f}")

    # --------------------------------------------------------------
    # 1) Build a very rich pool of seeds and keep the top few
    # --------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Deterministic refinements (mirror‑descent + additive GD)
    # --------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 8.0:
            break
        print(f"[refine] start seed {idx}")

        # mirror‑descent (exponentiated gradient)
        seq, val = _subgradient_refine(seed_vec, n, deadline,
                                       max_iters=250000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] mirror best={val:.6f}")

        # additive projected gradient descent
        seq, val = _additive_descent(seq, n, deadline,
                                     max_iters=150000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] additive best={val:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated‑annealing fine‑tuning (uses remaining time)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall] anneal best={val:.6f}")

    # --------------------------------------------------------------
    # 4) Fallback (unlikely)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_2"] = best_seq.copy()
    return best_seq.tolist()
