def _generate_random_seeds(n, rng, N=2000):
    seeds = []
    for _ in range(N):
        typ = rng.integers(0,5) # define options
        if typ == 0:
            conc = rng.uniform(0.01, 5.0)
            seeds.append(_random_dirichlet(n, conc, rng))
        elif typ == 1:
            conc = rng.uniform(0.01, 1.0)
            seeds.append(_sorted_dirichlet(n, conc, rng))
        elif typ == 2:
            # block pattern
            max_blocks = rng.choice([3,5,7,10])
            seeds.append(_random_blocks(n, max_blocks, rng))
        elif typ==3:
            # piecewise linear decreasing
            s = rng.integers(1, n-1)
            h0 = rng.uniform(0.001, 0.01) # will normalize anyway
            # generate decreasing linear to zero at s
            # compute line: a_i = max(0, h0*(1 - i/s))
            i = np.arange(n)
            a = np.maximum(0, h0 * (1 - i/s))
            a_sum = a.sum()
            if a_sum == 0:
                continue
            seeds.append(a / a_sum)
        elif typ==4:
            # mixture of two Dirichlets
            conc1 = rng.uniform(0.01, 5.0)
            conc2 = rng.uniform(0.01, 5.0)
            d1 = rng.dirichlet(np.full(n, conc1)).astype(np.float64)
            d2 = rng.dirichlet(np.full(n, conc2)).astype(np.float64)
            lam = rng.random()
            a = lam * d1 + (1-lam) * d2
            a /= a.sum()
            seeds.append(a)
    return seeds

"""
Step‑function optimiser (v3)

This implementation expands on the previous version by:
* Adding a rich family of *beta‑skewed* seeds (continuous analogue of
  monotone power‑law / geometric families) and a massive random Dirichlet
  search.
* Using a longer, higher‑temperature simulated‑annealing phase.
* Running the Adam optimiser for more iterations and with a larger
  smoothing‐parameter ceiling.
* Dynamically allocating the time budget among the stages.

The search works on normalised vectors (sum = 1); the evaluation
objective is therefore   2·n·max(conv(a,a))   (identical to
``evaluate_sequence`` up to the scale‑invariance).  The best sequence
found is stored in the global name ``height_sequence_1`` for reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    Assumes ``seq`` is non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (Duchi et al. 2008)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """
    Skewed beta distribution on {0,…,n‑1}.
    Parameters a,b > 0.  For a<1,b>1 we obtain a decreasing shape.
    """
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    logp = (a - 1.0) * np.log(np.clip(x, eps, 1.0 - eps)) + \
           (b - 1.0) * np.log(np.clip(1.0 - x, eps, 1.0 - eps))
    p = np.exp(logp - np.max(logp))
    p_sum = p.sum()
    return p / p_sum if p_sum > 0 else _uniform(n)


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw sorted in descending order."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Simple Dirichlet draw."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


def _random_blocks(n: int, max_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """Very simple block‑wise pattern."""
    k = rng.integers(2, max_blocks + 1)
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos:pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()

            seeds.append(_beta_shape(n, a, b))

    # sorted Dirichlet draws
    for conc in (0.02, 0.10, 0.30, 0.80, 1.5):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # block‑wise patterns
    for max_blocks in (3, 5, 7, 10, 15):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # random Dirichlet draws (large batch)
    for _ in range(n_random):
        conc = 10 ** rng.uniform(-3, 1)          # 0.001 … 10
        seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns
    for k in (5, 10, 20, 40, 80):
        for conc in (0.2, 0.6):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds

def _adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150000,
) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a log‑sum‑exp (smoothed max) surrogate."""
    eps = 1e-30
    v = np.log(start_vec + eps)               # logits on the simplex

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5
    beta_mul = 1.001
    beta_max = 2000.0          # stronger smoothing → tighter to true max

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                               # stay on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # anneal temperature
        beta = min(beta * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of weighted sum of convolution entries
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        # transform to log‑softmax space
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clip for stability
        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.9
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 250000,
) -> tuple[np.ndarray, float]:
    """Exact sub‑gradient mirror‑descent (exponentiated‑gradient) on the true max."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient of the active convolution entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update on simplex
        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _euc_proj_subgradient(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200000,
) -> tuple[np.ndarray, float]:
    """Gradient descent on the true max‑convolution with Euclidean projection."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[EucProj] it={it:7d} best={cur_val:.6f}")

        # gradient of the active entry (same pattern as sub‑gradient)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800000,
    init_temp_factor: float = 0.07,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simulated annealing with random mass reshuffling."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9998, min_temp)

        # pick a small random set of indices and re‑distribute their mass
        k = rng.integers(3, 10)                     # 3‑9 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a list of floats.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Acquire previous solution if present
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length of the optimisation vector – 4000 is a good trade‑off
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – massive random seed generation
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    seeds = _generate_seeds(n, rng, previous, n_random=2500)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(10, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed objectives:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] best seed objective = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – heavy optimiser on the elite seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx+1}/{len(best_seeds)} ---")

        seq_adam, val_adam = _adam_optimize(
            seed_vec, n, deadline, max_iters=150000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[overall‑Adam] new global best = {val_adam:.6f}")

        seq_sub, val_sub = _subgradient_refine(
            global_best_seq, n, deadline, max_iters=250000
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[overall‑SubGrad] new global best = {val_sub:.6f}")

        seq_euc, val_euc = _euc_proj_subgradient(
            global_best_seq, n, deadline, max_iters=200000
        )
        if val_euc < global_best_val:
            global_best_seq = seq_euc.copy()
            global_best_val = val_euc
            print(f"[overall‑EucProj] new global best = {val_euc:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – long simulated‑annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline,
            max_iters=800000,
            init_temp_factor=0.07
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] new global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – optional ordering test (descending vs ascending)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        for ordering in ("desc", "asc"):
            cand = np.sort(global_best_seq)[::-1] if ordering == "desc" else np.sort(global_best_seq)
            cand_val = _objective(cand)
            if cand_val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = cand_val
                print(f"[Reorder] new global best with {ordering} order = {cand_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:               # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
