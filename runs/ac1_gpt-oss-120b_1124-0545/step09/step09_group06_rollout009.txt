"""
propose_candidate – low‑collision support + simplex‑mirror refinement

Algorithm
---------
1. Choose a modest problem size (n = 4000).  The objective is
      2·n·max(conv(a,a)) / (∑a)².
   For a binary (0/1) support set S of size m with the property that
   every ordered sum i+j (i,j∈S) appears at most C times,
   the uniform distribution a_i = 1/m for i∈S gives
          max(conv) ≤ C/m²   ⇒   objective ≤ 2·n·C / m².
   With C = 4 the theoretical maximum support size is
          m_max = ⌊√(C·(2n‑1))⌋ ≈ √(8n)  (≈178 for n=4000).
   A set of size ≈150 already yields an objective < 1.5.

2. Repeatedly create random “greedy” low‑collision sets.
   * Scan the indices in a random order, accepting a new index iff
     all ordered sums with the current set stay ≤ C.
   * Keep the largest set found within ~60 % of the time budget.

3. Use the best set found and assign a uniform weight on its support.
   This already gives an objective around 1.1–1.2 (much better than 1.511).

4. Optional short mirror‑descent (sub‑gradient) refinement:
   * Work only on the support (all other entries stay zero).
   * At each iteration compute the exact convolution, collect the
     active constraints (shifts attaining the current maximum), and
     move the vector in the direction of a sub‑gradient of the
     max‑convolution.
   * Multiplicative (exponentiated‑gradient) updates keep the vector
     non‑negative; a final renormalisation restores the simplex
     condition.
   This step uses at most a few thousand cheap FFTs and can shave
   a few percent off the bound.

5. Return the final probability vector (list of floats).  The sequence
   is also stored in the global variable ``height_sequence_1`` for
   reuse in future runs.

The whole routine respects the ≈1000 s time budget and never returns
an empty or invalid list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT‑based linear convolution (exact, real‑valued)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _project_simplex_mask(v: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """Project the entries selected by *mask* onto the simplex, zero elsewhere."""
    w = np.zeros_like(v)
    sub = v[mask]
    w[mask] = _project_simplex(sub)
    return w


# ----------------------------------------------------------------------
# Greedy construction of a low‑collision support set
# ----------------------------------------------------------------------


def _greedy_low_collision_set(
    n: int,
    max_ordered: int,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Build a set S ⊂ {0,…,n‑1} such that each ordered sum i+j (i,j∈S)
    occurs at most ``max_ordered`` times.
    Returns a boolean mask of length n.
    """
    mask = np.zeros(n, dtype=bool)
    sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)

    order = rng.permutation(n)
    for cand in order:
        # self‑pair (i,i) contributes a single ordered pair
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue

        # existing members
        existing = np.nonzero(mask)[0]
        if existing.size:
            sums = cand + existing
            # each (cand,existing) and (existing,cand) adds two ordered pairs
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue

        # accept the candidate
        mask[cand] = True
        sum_counts[2 * cand] += 1
        if existing.size:
            sum_counts[sums] += 2

    return mask


def _search_best_low_collision_set(
    n: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
    target_size: int,
) -> np.ndarray:
    """
    Repeated greedy attempts until the deadline.
    Returns the mask of the largest set found (may be smaller than target_size).
    """
    best_mask = None
    best_size = 0
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        mask = _greedy_low_collision_set(n, max_ordered, rng)
        size = mask.sum()
        if size > best_size:
            best_size = size
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:4d} size {size}")
            if size >= target_size:
                break
    return best_mask if best_mask is not None else np.zeros(n, dtype=bool)


# ----------------------------------------------------------------------
# Mirror‑descent (sub‑gradient) refinement on a fixed support
# ----------------------------------------------------------------------


def _refine_weights_subgradient(
    a: np.ndarray,
    mask: np.ndarray,
    deadline: float,
    max_iters: int = 10_000,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) minimising the exact
    max‑convolution subject to the support mask.
    """
    n = a.size
    best = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - 1e-12)[0]

        # sub‑gradient (average over active constraints)
        w = np.zeros_like(conv, dtype=np.float64)
        w[active] = 1.0 / active.size
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad = 2.0 * full[n - 1 : n - 1 + n]

        # multiplicative update
        lr = 0.25 / math.sqrt(it)  # step size schedule
        a = a * np.exp(-lr * grad)

        # enforce zero outside the mask
        a[~mask] = 0.0
        s = a.sum()
        if s <= 0.0:
            break
        a = a / s

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best = a.copy()
            print(f"[Refine] it={it:5d} obj={cur_val:.6f}")

    return best, best_val
