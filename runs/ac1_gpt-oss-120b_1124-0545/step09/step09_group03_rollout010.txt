def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    """Return 2 * n * max(conv(seq, seq)) assuming sum(seq) == 1."""
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))

def _beta_mixture_seq(n: int, weights: np.ndarray, alphas: np.ndarray, betas: np.ndarray) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    # compute each component and sum weighted
    pdf = np.zeros(n, dtype=np.float64)
    for w, a, b in zip(weights, alphas, betas):
        pdf += w * (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    total = pdf.sum()
    if total <= 0.0:
        # fallback to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return pdf / total
```

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0):
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return x / s

def _smooth_adam_optimize(start_seq: np.ndarray, deadline: float,
                          beta_start: float = 5.0, beta_end: float = 200.0,
                          max_iter: int = 5_000_000, lr0: float = 0.03):
    n = start_seq.shape[0]
    eps = 1e-30
    logits = np.log(start_seq + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)
    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum
        conv = _conv_fft(a, a)
        # annealing beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)
        if (it & 1023) == 0:  # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.45:
                break
    return best_seq, best_val

def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 200.0):
    if minimize is None:
        return start_seq, _objective(start_seq)
    n = start_seq.shape[0]
    eps = 1e-30
    def fun_and_grad(logits):
        a = np.exp(logits)
        a = a / a.sum()
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        exps = np.exp(beta * (conv - max_c))
        lse = max_c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse
        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)
    x0 = np.log(start_seq + eps)
    class TimeOutException(RuntimeError):
        pass
    def callback(xk):
        if time.time() >= deadline:
            raise TimeOutException()
    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'disp': False},
            callback=callback,
        )
    except TimeOutException:
        best_logits = x0
    else:
        best_logits = res.x
    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    return best_a, best_val

def _refine_aggregated_subgradient(start_seq: np.ndarray, deadline: float,
                                  eps_rel: float = 1e-6):
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]
        active = np.zeros_like(conv)
        active[active_idx] = 1.0
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")
            if time.time() > deadline - 1.0:
                break
    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """Search for a non‑negative step‑function that minimises evaluate_sequence.
    
    Combines analytic seeds, Dirichlet/Beta‑mixture random draws, 
    smoothed Adam optimisation, optional L‑BFGS refinement and aggressive 
    KL‑mirror descent polishing. Also explores several lengths and 
    up‑scales the best solution to larger sizes.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin
    
    # Prefer reusing previously stored solution if available.
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.size > 0 and np.all(prev >= 0):
                prev = prev / prev.sum()
                previous = prev
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None
    
    # lengths to test
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000]
    
    best_seq = None
    best_val = float("inf")
    
    # -------------------------------------------------------------
    # Phase 1 – analytic seeds (including beta‑mixture)
    # -------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
    )
    for n in lengths:
        for fn in seed_families:
            try:
                seq = fn(n)
                val = _objective(seq)
                if val < best_val:
                    best_val = val
                    best_seq = seq.copy()
                    print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")
            except Exception as e:
                # protect against any numerical mishap
                print(f"[Seed] n={n} type={fn.__name__} raised {e}")
        # also try previous solution if same length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")
    
    # -------------------------------------------------------------
    # Phase 2 – random Dirichlet sampling
    # -------------------------------------------------------------
    now = time.time()
    # allocate ~30% of remaining time for Dirichlet sampling
    time_for_dirichlet = (deadline - now) * 0.30
    dirichlet_deadline = now + time_for_dirichlet
    alphas = [0.5, 1.0, 2.0, 5.0]  # added 5.0 for more extreme draws
    while time.time() < dirichlet_deadline:
        n = int(rng.choice(lengths))
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")
        # early stop if we reach a very low bound
        if best_val < 1.45:
            break
    
    # -------------------------------------------------------------
    # Phase 3 – random Beta‑mixture sampling
    # -------------------------------------------------------------
    now = time.time()
    time_for_mix = (deadline - now) * 0.15
    mix_deadline = now + time_for_mix
    K_mix = 3
    while time.time() < mix_deadline:
        n = int(rng.choice(lengths))
        seq = _sample_beta_mixture(n, rng, K=K_mix)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random-Mix] n={n:5d} K={K_mix} val={val:.6f}")
        # also try K=4 occasionally
        if rng.random() < 0.1:
            seq = _sample_beta_mixture(n, rng, K=4)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Random-Mix] n={n:5d} K=4 val={val:.6f}")
        if best_val < 1.45:
            break
    
    # -------------------------------------------------------------
    # Phase 4 – smoothed Adam optimisation (multiple restarts)
    # -------------------------------------------------------------
    if best_seq is not None:
        # create a few restart candidates: best plus some random Dirichlet of same length
        candidates = [best_seq]
        top_len = len(best_seq)
        for _ in range(3):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)
        # also include a mixture candidate of same length
        mix_cand = _sample_beta_mixture(top_len, rng, K=K_mix)
        candidates.append(mix_cand)
        # allocate ~25% of remaining time
        remaining = max(0.0, deadline - time.time())
        adam_deadline = time.time() + remaining * 0.25
        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=500.0,  # larger final beta for tighter approximation
                max_iter=4_000_000,
                lr0=0.04,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")
            # shrink deadline after each run to avoid overshoot
            remaining = max(0.0, deadline - time.time())
            if remaining <= 0.0:
                break
    
    # -------------------------------------------------------------
    # Phase 5 – optional L‑BFGS refinement
    # -------------------------------------------------------------
    if best_seq is not None:
        remaining = max(0.0, deadline - time.time())
        if remaining > 5.0:
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=500.0)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")
    
    # -------------------------------------------------------------
    # Phase 6 – aggressive mirror‑descent polishing (more tolerant)
    # -------------------------------------------------------------
    if best_seq is not None:
        remaining = max(0.0, deadline - time.time())
        # use a slightly larger tolerance to incorporate more active indices
        final_seq, final_val = _refine_aggregated_subgradient(best_seq, deadline, eps_rel=1e-4)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑Refine] improved best to {best_val:.6f}")
    
    # -------------------------------------------------------------
    # Phase 7 – upsample to a larger length and polish again
    # -------------------------------------------------------------
    if best_seq is not None:
        # try to expand to the next longer length if time permits
        current_n = len(best_seq)
        larger_lengths = [L for L in lengths if L > current_n]
        for L in larger_lengths:
            if time.time() >= deadline - 5.0:
                break
            # linear interpolation to new size L (preserve total mass)
            x_old = np.linspace(0, 1, current_n, endpoint=False) + 0.5/current_n
            x_new = np.linspace(0, 1, L, endpoint=False) + 0.5/L
            up_seq = np.interp(x_new, x_old, best_seq)
            up_seq = np.maximum(up_seq, 0.0)
            up_seq /= up_seq.sum()
            # do a short Adam refinement on the upsampled vector
            sub_deadline = time.time() + min(30.0, deadline - time.time() - 2.0)
            up_refined, up_val = _smooth_adam_optimize(
                start_seq=up_seq,
                deadline=sub_deadline,
                beta_start=5.0,
                beta_end=300.0,
                max_iter=1_000_000,
                lr0=0.03,
            )
            if up_val < best_val:
                best_val = up_val
                best_seq = up_refined
                print(f"[Upsample] length {L} improved best to {best_val:.6f}")
            # optional extra polishing
            if time.time() < deadline - 2.0:
                final_seq2, final_val2 = _refine_aggregated_subgradient(best_seq, deadline, eps_rel=1e-4)
                if final_val2 < best_val:
                    best_val = final_val2
                    best_seq = final_seq2
                    print(f"[Upsample‑Agg] length {L} further improved to {best_val:.6f}")
            current_n = len(best_seq)
    
    # Final sanity check
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)
    
    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
search for a step‑function that minimises ``evaluate_sequence``.

The algorithm proceeds in several independent phases that together
explore a wide variety of shapes while respecting the 1000 s time‑budget:

1. **Deterministic analytic seeds** – several classic windows (uniform,
   Catalan, Hann, Hamming, Blackman, Gaussian, arcsine, single‑beta PDFs).

2. **Random Dirichlet draws** – Dirichlet vectors with diverse concentration
   parameters to obtain both spiky and flat profiles.

3. **Random Beta‑mixture draws** – mixtures of up to four Beta PDFs give a
   flexible low‑dimensional family; each candidate is evaluated directly.

4. **Smoothed Adam optimisation** – a log‑sum‑exp smooth approximation of
   ``max(conv(a,a))`` is minimised on the probability simplex using Adam.
   Several restart points (best so far, random Dirichlet, mixture) are used
   and a larger final temperature (β≈500) provides a tighter approximation.

5. **Optional L‑BFGS‑B refinement** (if SciPy is available) on the same
   smooth objective.

6. **Aggressive KL‑mirror descent** – a sub‑gradient method that aggregates
   *all* near‑maximal convolution entries (relative tolerance 1e‑4) to push
   the exact max lower.

7. **Upsampling & re‑polishing** – the current best vector is linearly
   interpolated to larger admissible lengths and a short Adam + mirror‑
   descent polishing is performed; this exploits the observed ``O(1/n)`` 
   scaling of the convolution peak.

All phases respect the overall deadline and keep the globally best
sequence.  The returned list is already normalised (sum = 1) and
non‑negative – ``evaluate_sequence`` performs the final clipping.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, pi

# optional SciPy import – L‑BFGS‑B will be skipped if unavailable

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence (sum == 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic seed families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2 * pi * i / (n - 1))
         + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    """Discrete approximation of the arcsine density on {0,…,n‑1}."""
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf shape on a discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑distributed vector (concentration = alpha)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Beta‑mixture families (low‑dimensional parameterisation)
# ----------------------------------------------------------------------
def _beta_mixture_seq(n: int, weights: np.ndarray,
                      alphas: np.ndarray, betas: np.ndarray) -> np.ndarray:
    """
    Build a mixture of Beta PDFs on a uniform grid of size ``n``.
    The raw mixture is normalised to sum = 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    pdf = np.zeros(n, dtype=np.float64)
    for w, a, b in zip(weights, alphas, betas):
        pdf += w * (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    total = pdf.sum()
    if total <= 0.0:
        return _uniform(n)
    return pdf / total


def _sample_beta_mixture(n: int, rng: np.random.Generator, K: int = 3) -> np.ndarray:
    """Randomly sample a K‑component Beta mixture."""
    weights = rng.dirichlet(np.ones(K))
    alphas = rng.uniform(0.1, 10.0, size=K)
    betas = rng.uniform(0.1, 10.0, size=K)
    return _beta_mixture_seq(n, weights, alphas, betas)


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑sum‑exp smoothing)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(start_seq: np.ndarray, deadline: float,
                          beta_start: float = 5.0, beta_end: float = 200.0,
                          max_iter: int = 5_000_000, lr0: float = 0.03) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    The parametrisation is a soft‑max of logits to keep the vector on the simplex.
    """
    n = start_seq.shape[0]
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max
        a = np.exp(logits)
        a = a / a.sum()

        # self‑convolution
        conv = _conv_fft(a, a)

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp
        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()      # soft weights, sum=1

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])    # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:               # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.45:            # early exit if already very good
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS‑B refinement (smooth max with large β)
# ----------------------------------------------------------------------
def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 200.0) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smooth max objective.
    Returns the best sequence and its exact objective value.
    """
    if minimize is None:                 # SciPy not available
        return start_seq, _objective(start_seq)

    n = start_seq.shape[0]
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a = a / a.sum()
        conv = _conv_fft(a, a)

        max_c = np.max(conv)
        exps = np.exp(beta * (conv - max_c))
        lse = max_c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        # gradient
        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOutException(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOutException()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'disp': False},
            callback=callback,
        )
    except TimeOutException:
        best_logits = x0
    else:
        best_logits = res.x

    best_a = np.exp(best_logits)
    best_a /= best_a.sum()
    best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Aggressive KL‑mirror descent (aggregated active set)
# ----------------------------------------------------------------------
def _refine_aggregated_subgradient(start_seq: np.ndarray, deadline: float,
                                  eps_rel: float = 1e-6) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates all convolution
    entries within a *relative* tolerance of the current maximum.
    """
    a = start_seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")

            if time.time() > deadline - 1.0:
                break

    return best_seq, best_val
