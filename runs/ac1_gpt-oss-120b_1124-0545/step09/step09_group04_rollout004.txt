"""
Step‑function optimiser – multi‑scale mirror descent + optional SA polish.

The algorithm explores several problem sizes (n = 256 … 4000).  For each
size it builds a diverse collection of “seed” vectors (uniform,
triangular, sinusoidal, geometric, power‑law, beta, random Dirichlet,
sparse Dirichlet, …).  After normalising every seed to Σa = 1 the cheap
surrogate  2·n·max(conv(a,a))  is evaluated (this equals the true
evaluate_sequence value as long as the sum stays equal to 1).

The best few seeds for the current size are refined with a long
exponentiated‑gradient (mirror‑descent) loop that directly minimises the
exact max‑convolution.  The update is

    log a ← log a – η·g      (g is a sub‑gradient of the current max)
    a ← softmax(log a)      (i.e. a_i = exp(log a_i) / Σexp)

Learning‑rate η decays as η₀/√t, which gives rapid early progress and
stable later behaviour.  The loop stops as soon as the global time budget
(≈1000 s) is exhausted.

If time permits a short simulated‑annealing (SA) polishing phase is
performed: a random small subset of coordinates reshuffles its mass;
changes are accepted with the usual Metropolis rule.

The final best vector (any length) is stored in the global variable
``height_sequence_1`` for possible reuse and returned as a plain ``list``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective (identical to evaluate_sequence when Σseq = 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure (unsymmetric) geometric decay a_i = r**i."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction (diverse cheap candidates)
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Create a list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – fine grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric decay (near‑constant)
    for r in np.linspace(0.985, 0.9999, 12):
        seeds.append(_geom_decay(n, r))

    # pure power‑law decay
    for p in np.arange(0.1, 5.01, 0.3):
        seeds.append(_power_decay(n, p))

    # beta family – modest grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform & symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.90, 7):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 4):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws – several concentrations
    for conc in (0.02, 0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns – very small support
    for k in (3, 5, 10):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent that minimises 2·n·max(conv(a,a)).
    The current sequence is kept on the probability simplex.
    Returns the best sequence seen and its surrogate objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30  # for numerical stability in log

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # full convolution, locate the maximal entry
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # current surrogate objective
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[SubGrad] it={it:,d} best≈{best_val:.6f}")

        # sub‑gradient of max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)        # keep numbers bounded
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing polish (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Tiny SA that repeatedly redistributes the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)               # subset size 2 … 6
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 25_000 == 0:
                    print(f"[SA]  it={it:,d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine tries several problem sizes,
    refines a handful of promising seeds with mirror‑descent and,
    if time permits, finishes with a brief simulated‑annealing polish.
    The final vector is stored in the global name ``height_sequence_1``.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously found solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Problem sizes to explore (smaller sizes give smaller 2·n factor)
    # ------------------------------------------------------------------
    n_candidates = [256, 512, 1024, 2048, 4000]

    global_best_seq = None
    global_best_val = float("inf")

    for n in n_candidates:
        if time.time() > deadline:
            break

        print(f"\n=== Exploring n = {n} ===")
        # build seeds (reuse previous only if its length matches)
        prev_for_n = previous if previous is not None and previous.shape == (n,) else None
        seeds = _generate_seeds(n, rng, prev_for_n)

        # cheap evaluation of all seeds
        evaluated = [(vec, _objective(vec)) for vec in seeds]
        evaluated.sort(key=lambda p: p[1])

        top_k = min(5, len(evaluated))
        top_seeds = [vec for vec, _ in evaluated[:top_k]]

        print(f"  Generated {len(seeds)} seeds – best surrogate = {evaluated[0][1]:.6f}")

        # mirror‑descent refinement for each top seed
        for i, seed_vec in enumerate(top_seeds, 1):
            if time.time() > deadline:
                break
            print(f"  [Refine] seed #{i}/{top_k} (n={n}) …")
            seq_ref, val_ref = _subgradient_refine(
                seed_vec, n, deadline, max_iters=800_000, lr0=2.0
            )
            if val_ref < global_best_val:
                global_best_seq = seq_ref.copy()
                global_best_val = val_ref
                print(f"    *** New global best (n={n}) = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional simulated‑annealing polishing (if time remains)
    # ------------------------------------------------------------------
    if global_best_seq is not None and time.time() < deadline:
        print("\n=== Simulated‑annealing polishing ===")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, len(global_best_seq), deadline, max_iters=250_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"  *** After SA new best = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call the official evaluator once
    # ------------------------------------------------------------------
    try:
        final_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f}")
        print(f"Corresponding evaluate_sequence value = {final_val:.6f}")
        print(f"Length of final sequence = {len(global_best_seq)}")
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # expose for future runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
