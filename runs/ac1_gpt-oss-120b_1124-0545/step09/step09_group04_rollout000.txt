"""
non‑negative step‑function optimiser.

The optimiser works in three stages:

1️⃣  **Seed generation** – a large, diverse set of candidate vectors is
    built from simple analytic families (uniform, triangular, sine‑window,
    symmetric/unsymmetric geometric, power‑law, beta) together with a few
    random Dirichlet draws and very sparse patterns.  All vectors are
    normalised to have L1‑norm = 1, because the evaluation function
    ``evaluate_sequence`` is homogeneous (scaling does not change the
    result).

2️⃣  **Mirror‑descent (exponentiated‑gradient) refinement** – each of the
    best few seeds is improved by a sub‑gradient method that directly
    minimises the true objective
    ``2·n·max(conv(a,a))`` while staying on the probability simplex.
    The update uses a large initial learning‑rate (≈2) which decays as
    1/√t.  The routine stops when the global time budget is exhausted.

3️⃣  **Optional simulated‑annealing polish** – a cheap SA that reshuffles
    the mass of a random small subset can occasionally escape shallow
    basins and yields a modest extra gain.

The whole search respects the supplied time budget (default 1000 s) and
stores the best vector in the global name ``height_sequence_1`` for
reuse in a later run.  The returned list can be passed directly to the
provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via a real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate (and exact) objective used during the search.
    ``seq`` must be non‑negative and already normalised to L1‑norm = 1.
    Returns ``2·n·max(conv(seq, seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all returned normalised to L1 = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Full‑support Dirichlet draw."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Seed construction (cheap, many candidates)
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – medium grid
    for r in np.linspace(0.1, 0.9, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric – very high decay (near‑constant)
    for r in np.linspace(0.985, 0.9999, 12):
        seeds.append(_geom_decay(n, r))

    # pure power‑law decay
    for p in np.arange(0.1, 5.01, 0.3):
        seeds.append(_power_decay(n, p))

    # beta family – broad grid
    for a in np.linspace(0.05, 2.0, 6):
        for b in np.linspace(0.05, 5.0, 6):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform and symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.2, 0.9, 7):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 4):
            mixed = w * uniform + (1.0 - w) * base
            mixed /= mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.2, 0.5, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns (tiny support)
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution, if compatible
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Mirror‑descent / exponentiated‑gradient sub‑gradient optimiser
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    objective.  The learning‑rate decays as 1/√t.
    Returns the best sequence seen and its surrogate objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # compute full convolution and locate the maximal entry
        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # current surrogate (exact) objective
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[SubGrad] it={it:9d} best≈{best_val:.6f}")

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # stabilise numerically
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polish (lightweight)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 200_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that redistributes the mass of a small random subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)            # size of the subset (2…6)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:8d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Returns the best sequence as a plain ``list[float]``.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin before the hard cut‑off
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Re‑use a previous solution if it exists
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a few candidate lengths – 4000 works well, but we also try a
    # smaller size because the objective scales with n.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4000]
    global_best_seq = None
    global_best_val = np.inf

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ----- Stage 1 : generate seeds ---------------------------------
        seeds = _generate_seeds(n, rng, previous)
        print(f"generated {len(seeds)} seeds")

        evaluated = [(vec, _objective(vec)) for vec in seeds]
        evaluated.sort(key=lambda x: x[1])
        top_k = min(6, len(evaluated))
        best_seeds = [vec for vec, _ in evaluated[:top_k]]
        print(f"top {top_k} seed objectives:")
        for i, (_, val) in enumerate(evaluated[:top_k]):
            print(f"  #{i+1}: {val:.6f}")

        # ----- Stage 2 : mirror‑descent on each top seed ---------------
        for idx, seed_vec in enumerate(best_seeds, start=1):
            if time.time() > deadline:
                break
            print(f"\n[Refine] seed #{idx}/{top_k} (n={n})")
            refined_seq, refined_val = _subgradient_refine(
                seed_vec, n, deadline, max_iters=5_000_000, lr0=2.0
            )
            print(f"   refined surrogate = {refined_val:.6f}")
            if refined_val < global_best_val:
                global_best_val = refined_val
                global_best_seq = refined_seq.copy()
                print(f"   >>> new global best (n={n}) ≈ {global_best_val:.6f}")

        # ----- Stage 3 : optional SA polish ----------------------------
        if global_best_seq is not None and time.time() < deadline:
            print("\n[SA polish] on current global best")
            seq_sa, val_sa = _simulated_annealing(
                global_best_seq, n, deadline, max_iters=150_000
            )
            if val_sa < global_best_val:
                global_best_val = val_sa
                global_best_seq = seq_sa.copy()
                print(f"   >>> improved by SA → {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check – call evaluate_sequence once to ensure validity
    # ------------------------------------------------------------------
    if global_best_seq is None:
        # fallback: uniform vector of length 1024
        global_best_seq = _uniform(1024)
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {global_best_val:.6f}"
        f" (length {len(global_best_seq)})"
    )

    # expose for a possible subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
