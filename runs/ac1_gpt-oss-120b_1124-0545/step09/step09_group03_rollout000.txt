"""
Propose a step‑function with a lower bound on ``evaluate_sequence``.

Algorithm (≈ 1000 s)

1. **Deterministic parametric families** – generate a rich set of
   analytic sequences (uniform, geometric decay, power‑law, gamma,
   Catalan, Beta, Hann, …) for a few carefully chosen lengths.
2. **Random Dirichlet / Beta sampling** – draw many random candidates.
3. **Smoothed Adam** – run Adam on a log‑softmax parametrisation,
   using a log‑sum‑exp approximation of the max with a temperature
   schedule 5 → 500.  Multiple restarts are performed.
4. **Projected sub‑gradient descent** on the *exact* max‑objective.
5. **Simple hill‑climbing** – random pairwise mass transfers improve
   the solution locally.

All phases respect the global time budget.  The returned list is a
non‑negative vector normalised to sum = 1; ``evaluate_sequence`` will
clip it if necessary.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, log, pi, exp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Equivalent to ``evaluate_sequence`` for a *normalised* sequence
    (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, sum w = 1}.
    """
    # Implementation of Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all values are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w


# ----------------------------------------------------------------------
# Analytic families (always normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    """Coefficients ∝ binom(2i,i)/4**i – the (1‑z)^‑½ series."""
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta‑pdf on a grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _geometric_decay(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i, 0<r<1 (r≈1 gives uniform)."""
    if r <= 0.0:
        return _uniform(n)
    if r >= 1.0:
        return _uniform(n)
    geom = r ** np.arange(n, dtype=np.float64)
    total = geom.sum()
    return geom / total


def _power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = idx ** (-p)
    return w / w.sum()


def _gamma_shape(n: int, k: float, theta: float) -> np.ndarray:
    """a_i ∝ (i+1)^{k-1} * exp(-(i+1)/θ)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    w = (idx ** (k - 1.0)) * np.exp(-idx / theta)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Random generation
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑like vector (α governs concentration)."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed Adam on log‑softmax parametrisation
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    max_iter: int = 8_000_000,
    lr0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)).
    Optimises the logits of a soft‑max representation.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max
        a = np.exp(logits)
        a /= a.sum()

        # convolution
        conv = _conv_fft(a, a)

        # temperature schedule (linear)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()           # smooth‑max weights

        # gradient w.r.t a
        conv_pa = _conv_fft(p, a[::-1])         # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:  # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            if best_val < 1.45:      # early exit if already very good
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the exact max‑objective
# ----------------------------------------------------------------------
def _subgradient_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel: float = 1e-5,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the true
    ``max(conv)`` objective.  The active set contains *all* indices
    within eps_rel·max.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        # active set (relative tolerance)
        active_idx = np.where(conv >= max_val - eps_rel * max_val)[0]

        # indicator on full conv length
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # sub‑gradient = 2 * (active * a_rev) convolved
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # step size schedule
        eta = 1.5 / sqrt(it)   # a slightly larger constant than plain mirror‑descent
        a_new = a - eta * g
        a_new = _project_simplex(a_new)

        # evaluate
        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            print(f"[Subgrad] it={it:7d} n={n:5d} best={val:.6f}")

        a = a_new

        # early stop if we are close to the target
        if best_val < 1.45:
            break

        # prevent endless loops when time is almost over
        if it % 1024 == 0 and time.time() > deadline - 2.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Random hill‑climbing refinement (pairwise mass transfers)
# ----------------------------------------------------------------------
def _hillclimb_refine(
    start_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 4000,
    step_fraction: float = 0.08,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search: repeatedly move a small fraction
    of mass from a random index to another.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    for _ in range(attempts):
        if time.time() >= deadline:
            break
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] < 1e-12:
            continue
        delta = rng.uniform(0.0, min(step_fraction * a[i], a[i]))
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        # sum unchanged, non‑negativity guaranteed by construction
        val = _objective(a_new)
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            print(f"[HillClimb] val={val:.6f}")
        # else keep the old a (simple greedy acceptance)

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):

    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Optional reuse of a previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths that we will explore.  We keep them FFT‑friendly.
    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    seed_families = [
        _uniform,
        _catalan_shape,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.12),

        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),

        # power‑law
        for p in power_ps:

            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()

        # gamma‑like shapes
        for k, th in gamma_params:
            seq = _gamma_shape(n, k, th * n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Gamma] n={n:5d} k={k:.2f} θ={th:.2f} val={val:.6f}")

        # previous solution reuse (if length matches)

        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
