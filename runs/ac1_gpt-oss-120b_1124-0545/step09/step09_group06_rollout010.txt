"""
Optimiser for the non‑negative step‑function objective

Goal
----
Find a list of non‑negative heights (a “step function”) that minimises

    2·n·max(conv(f,f)) / (sum(f)^2)

where `conv` is the linear (full) convolution.  The objective is
scale‑invariant, so we may assume `sum(f)=1`.

Strategy
--------
1. **Random low‑collision support search** –  
   Pick random supports of size `m` and keep the one that gives the
   smallest bound `2·n·max_order/m²`.  The allowed ordered‑pair count
   (`max_order`) is not fixed; we simply minimise the bound directly.

2. **Local greedy refinement** –  
   Starting from the best support, repeatedly try to add a new index.
   The addition is kept if the bound improves (even if the raw
   `max_order` grows a little).  This hill‑climbing quickly pushes the
   support size beyond the naïve random limit.

3. **Seed pool** –  
   Assemble many initial sequences (window functions, random draws,
   the refined support with uniform and Dirichlet weights, the previous
   best solution, …).  Keep the top few with the smallest true objective
   as starting points for the expensive phases.

4. **Adam optimisation** –  
   Optimise the weights on the full vector using a smoothed‑max surrogate
   (soft‑max over the convolution entries).  The surrogate temperature is
   increased slowly so that the algorithm converges to a true minimiser.

5. **Exact sub‑gradient descent** –  
   Refine the solution on the original non‑smooth objective using an
   exponentiated‑gradient (mirror‑descent) scheme.

6. **Optional LP‑based mixing** –  If `cvxpy` is available, a cheap LP
   maximises the sum of a partner vector under the current worst‑case
   convolution bound; the partner is mixed back into the current iterate.

7. **Simulated annealing** –  Final fine‑tuning via a Metropolis walk.

All phases respect the overall 1000 s budget.  The best sequence found
is stored in the global variable `height_sequence_1` for reuse across
runs.

The implementation is self‑contained and only requires NumPy (and
optionally `cvxpy`).
"""

# ----------------------------------------------------------------------
# FFT‑based convolution (exact, real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple window families (return probability vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Sparse seeds
# ----------------------------------------------------------------------
def _sparse_random_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform mass on a random support of size ``m`` (1 ≤ m ≤ n)."""
    if m >= n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic “Sidon‑like” support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _quadratic_residue_seed(n: int, m: int) -> np.ndarray:
    """Positions are i² (mod n)."""
    idx = np.mod(np.arange(m) ** 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Low‑collision support generator (randomised + greedy refinement)
# ----------------------------------------------------------------------
def _random_lowcollision_set(
    n: int,
    m_min: int,
    m_max: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """
    Randomly draw support sets and keep the one that minimises
    2·n·max_order / m² (the uniform‑weight bound).
    Returns the index array of the best set found.
    """
    best_idx = None
    best_obj = float("inf")
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        m = rng.integers(m_min, m_max + 1)
        idx = rng.choice(n, size=m, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0
        conv = _conv_fft(a, a)
        max_order = int(conv.max())
        obj = 2.0 * n * max_order / (m * m)
        if obj < best_obj:
            best_obj = obj
            best_idx = idx.copy()
            print(f"[LowCollRand] attempt {attempts:4d} m={m:3d} max_order={max_order:3d} obj={obj:.6f}")
            # early exit if we already beat the target bound
            if best_obj < 1.5:
                break
    if best_idx is None:
        return np.array([], dtype=int)
    return best_idx


def _refine_support_additive(
    idx: np.ndarray,
    n: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """
    Greedy hill‑climbing: repeatedly try to add a new index.
    Keep the addition if the uniform‑weight bound improves.
    Returns the refined index list.
    """
    if idx.size == 0:
        return idx

    a = np.zeros(n, dtype=np.uint8)
    a[idx] = 1
    cur_idx = idx.copy()
    cur_m = cur_idx.size
    conv = _conv_fft(a.astype(np.float64), a.astype(np.float64)).astype(np.int32)
    cur_max = int(conv.max())
    cur_obj = 2.0 * n * cur_max / (cur_m * cur_m)

    attempts = 0
    while time.time() < deadline:
        attempts += 1
        cand = rng.integers(0, n)
        if a[cand]:
            continue

        # compute incremental convolution
        conv_new = conv.copy()
        conv_new[2 * cand] += 1
        sums = cur_idx + cand
        conv_new[sums] += 2
        new_m = cur_m + 1
        new_max = int(conv_new.max())
        new_obj = 2.0 * n * new_max / (new_m * new_m)

        if new_obj < cur_obj:
            # accept
            a[cand] = 1
            cur_idx = np.append(cur_idx, cand)
            cur_m = new_m
            conv = conv_new
            cur_max = new_max
            cur_obj = new_obj
            print(f"[Refine] added {cand:4d} -> m={cur_m:3d} max={cur_max:3d} obj={cur_obj:.6f}")
            # stop early if we beat the target bound
            if cur_obj < 1.5:
                break
        # safety break if we have exhausted attempts
        if attempts % 5000 == 0 and time.time() > deadline:
            break
    return cur_idx


# ----------------------------------------------------------------------
# Simple utilities
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(seed: np.ndarray, n: int, deadline: float, max_iters: int = 40000):
    """
    Adam on a smoothed‑max surrogate of max(conv(a,a)).
    Returns best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)          # logits
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    lr0 = 0.7

    beta_temp = 1.5          # temperature for the soft‑max over the convolution
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                # current distribution (sum=1)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # --- smoothed max weights ---
        beta_temp *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # --- gradient of Σ w_k·(a*a)_k ---
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # --- chain rule through soft‑max ---
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # --- Adam moment update ---
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray, n: int, deadline: float,
                        max_iters: int = 1_000_000):
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                 # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _lp_refine(f: np.ndarray, n: int, deadline: float,
               t_mix: float = 0.3, max_lp_iters: int = 5) -> np.ndarray:
    """
    Cheap LP refinement: for the current max M, maximise Σ g subject to
    f * g ≤ M (with g ≥ 0).  Mix the result back into f.
    """
    if not CVXPY_AVAILABLE:
        return f

    a = f.copy()
    for lp_it in range(max_lp_iters):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - 1e-9)[0]

        g = cp.Variable(n)
        constraints = [g >= 0]

        for k in active:
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i = np.arange(i_low, i_high + 1)
            j = k - i
            coeff = np.zeros(n, dtype=np.float64)
            coeff[j] = a[i]
            constraints.append(coeff @ g <= M + 1e-12)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000,
                       abstol=1e-9, reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if g.value is None:
            break

        g_opt = np.asarray(g.value).flatten()
        if g_opt.sum() == 0:
            break
        g_opt = g_opt / g_opt.sum()

        a = (1.0 - t_mix) * a + t_mix * g_opt
        a = _project_simplex(a)

    return a


def _anneal_optimize(start_seq: np.ndarray, n: int, deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass
    between two positions and occasionally performs a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = _uniform(n)
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Seed generation (including many sparse candidates)
# ----------------------------------------------------------------------
def _make_seeds(n: int, previous: np.ndarray | None, rng: np.random.Generator,
                lowcoll_idx: np.ndarray | None) -> list[np.ndarray]:
    """
    Assemble a diverse set of starting vectors (all summed to 1).
    """
    seeds = []

    # classic families
    seeds.append(_uniform(n))
    for c in np.linspace(0.5, 4.5, 9):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 2.0, 9):
        seeds.append(_power(n, p))
    for c in np.linspace(0.5, 3.5, 5):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 1.8, 5):
            seeds.append(_geom_power(n, r, p))
    for c in np.linspace(0.5, 4.0, 5):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 4):
        seeds.append(_tukey_window(n, alpha))
    for sigma_factor in np.linspace(0.08, 0.30, 7):
        seeds.append(_gaussian(n, sigma_factor))
    for alpha in np.linspace(0.55, 0.95, 9):
        seeds.append(_beta_window(n, alpha))
    for _ in range(7):
        seeds.append(_random_softmax(n, rng))

    # sparse random seeds – support sizes 30 … 200
    for m in range(30, 210, 20):
        for _ in range(5):
            seeds.append(_sparse_random_seed(n, m, rng))

    # deterministic Sidon‑like structures
    for m in (50, 70, 90, 110):
        seeds.append(_sidon_like_seed(n, min(m, n)))
        seeds.append(_quadratic_residue_seed(n, min(m, n)))

    # low‑collision uniform support (if we managed to build one)
    if lowcoll_idx is not None and lowcoll_idx.size:
        a = np.zeros(n, dtype=np.float64)
        a[lowcoll_idx] = 1.0 / lowcoll_idx.size
        a /= a.sum()
        seeds.append(a)
        print(f"[seed] added low‑collision uniform seed (size {lowcoll_idx.size})")

        # random Dirichlet perturbations of that support
        for _ in range(4):
            w = rng.dirichlet(np.ones(lowcoll_idx.size))
            a2 = np.zeros(n, dtype=np.float64)
            a2[lowcoll_idx] = w
            a2 /= a2.sum()
            seeds.append(a2)

    # reuse previous solution if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # clean‑up: ensure non‑negativity and normalise
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point expected by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence``.  The routine respects a ~1000 s time budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for interpreter overhead
    deadline = start_time + budget_s - 5.0

    # try to reuse a previous construction if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive random low‑collision support search
    # ------------------------------------------------------------------
    # target size range around sqrt(8·n) ≈ 179
    approx = int(math.sqrt(8.0 * n))
    m_min = max(80, int(0.6 * approx))
    m_max = min(n, int(1.2 * approx))
    t_set_deadline = start_time + 0.25 * (deadline - start_time)   # ~25 % of budget

    raw_idx = _random_lowcollision_set(
        n=n,
        m_min=m_min,
        m_max=m_max,
        rng=rng,
        deadline=t_set_deadline,
    )
    # Local greedy refinement (additive hill‑climb)
    if raw_idx.size:
        refine_deadline = t_set_deadline + 0.07 * (deadline - start_time)  # extra 7 %
        lowcoll_idx = _refine_support_additive(
            idx=raw_idx,
            n=n,
            rng=rng,
            deadline=refine_deadline,
        )
        print(f"[Info] final low‑collision support size {lowcoll_idx.size}")
    else:
        lowcoll_idx = None

    # ------------------------------------------------------------------
    # Phase 2 – build a diverse pool of seeds
    # ------------------------------------------------------------------
    seeds = _make_seeds(n, previous, rng, lowcoll_idx)

    # ------------------------------------------------------------------
    # Quick evaluation of seeds – keep the best few for the expensive phases
    # ------------------------------------------------------------------
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = min(12, len(seed_vals))
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    global_best_seq = seed_vals[0][0].copy()
    global_best_val = seed_vals[0][1]
    print(f"[seed] initial best obj = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam optimisation on top seeds
    # ------------------------------------------------------------------
    t_adam_deadline = start_time + 0.55 * (deadline - start_time)
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > t_adam_deadline:
            break
        remaining = t_adam_deadline - time.time()
        max_iters = int(min(50000, remaining / 0.0012))
        seq, val = _adam_optimize(seed_vec, n, t_adam_deadline, max_iters=max_iters)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Adam] seed={idx} best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact sub‑gradient refinement (majority of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12.0:
        sub_deadline = deadline - 7.0   # leave a cushion for later phases
        remaining = sub_deadline - time.time()
        max_sub_iters = int(min(800000, remaining / 0.0014))
        seq, val = _subgradient_refine(global_best_seq, n, sub_deadline,
                                       max_iters=max_sub_iters)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall SubGrad] best={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Optional LP refinement (if cvxpy is available)
    # ------------------------------------------------------------------
    if CVXPY_AVAILABLE and time.time() < deadline - 6.0:
        lp_seq = global_best_seq.copy()
        while time.time() + 5.0 < deadline:
            lp_seq = _lp_refine(lp_seq, n, deadline, t_mix=0.3, max_lp_iters=8)
            # short sub‑gradient clean‑up after each LP pass
            sub_seq, sub_val = _subgradient_refine(lp_seq, n, deadline,
                                                   max_iters=25000)
            if sub_seq is not None and sub_val < global_best_val:
                global_best_seq = sub_seq.copy()
                global_best_val = sub_val
                print(f"[LP+Sub] new best {sub_val:.6f}")
                lp_seq = sub_seq.copy()
            else:
                break

    # ------------------------------------------------------------------
    # Phase 6 – Simulated annealing fine‑tuning (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq, val = _anneal_optimize(global_best_seq, n, deadline, max_iters=None)
        if seq is not None and val < global_best_val:
            global_best_seq = seq.copy()
            global_best_val = val
            print(f"[overall Anneal] best={val:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and safety checks
    # ------------------------------------------------------------------
    final_seq = np.maximum(global_best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {global_best_val:.6f} "
          f"(length {len(final_seq)})")

    # expose for the next iteration (if any)
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
