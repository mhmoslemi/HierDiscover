def _smooth_exponential_gradient(start_seq, deadline, beta_start=0.5, beta_max=20.0, beta_inc=0.005, lr0=1.0):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    beta = beta_start
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_conv = conv.max()
        # objective approx: 2 * n * max_conv
        # compute softmax weights
        # exponent = beta * (conv - max_conv)  # shift for numerical stability
        exp_vals = np.exp(beta * (conv - max_conv))
        w = exp_vals / exp_vals.sum()
        # gradient

def _upsample(seq, new_n):
    old_n = len(seq)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = new_seq / new_seq.sum()
    return new_seq

"""
Propose a non‑negative step function that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm is a three‑phase procedure:

1. **Coarse optimisation on a small problem** (n≈2000) using a
   smoothed exponential‑gradient descent (soft‑max approximation of the
   max‑convolution).  This stage runs quickly and discovers a good shape.

2. **Upsample the coarse solution** to a larger dimension (n≈8000) and
   repeat the smoothed exponential‑gradient optimisation, now with a
   gradually increasing “softness’’ parameter β to focus on the true
   maximum.  The updates are performed on the simplex by an
   exponentiated‑gradient step.

3. **Fine‑tune** the best sequence with a tiny simulated‑annealing walk
   that moves a small amount of mass between two positions.

The routine respects the user‑supplied time budget (default 1000 s) and
stores the final sequence in the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective for a (sum‑normalised) sequence."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (probability vectors on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape, peak in the middle."""
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(x - centre) / (centre + 1.0)
    a = np.maximum(a, 0.0)
    return a / a.sum()


# ----------------------------------------------------------------------
# Upsample a probability vector by linear interpolation
# ----------------------------------------------------------------------
def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    tot = new_seq.sum()
    if tot > 0.0:
        new_seq /= tot
    else:
        new_seq = _uniform(new_n)
    return new_seq


# ----------------------------------------------------------------------
# Smoothed exponential‑gradient descent (log‑sum‑exp approximation)
# ----------------------------------------------------------------------
def _smooth_exponential_gradient(start_seq: np.ndarray,
                                 deadline: float,
                                 n: int,
                                 lr0: float = 1.0,
                                 beta_start: float = 0.5,
                                 beta_max: float = 30.0,
                                 beta_inc: float = 0.005) -> tuple[np.ndarray, float]:
    """
    Optimises the max‑convolution using a soft‑max (log‑sum‑exp)
    surrogate. The update is performed on the simplex via an
    exponentiated‑gradient step.
    """
    a = start_seq.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()                     # ensure normalisation
    best_seq = a.copy()
    # initial objective (sum is 1 ⇒ objective = 2·n·max_conv)
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * conv.max()
    it = 0
    beta = beta_start
    eps = 1e-30

    while True:
        it += 1
        if time.time() >= deadline:
            break

        # full convolution of a with itself
        conv = _conv_fft(a, a)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SmoothEG] it={it:8d} beta={beta:.2f} best={best_val:.6f}")

        # soft‑max weights over the convolution entries
        # shift by max_c for numerical stability
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to uniform weights (should not happen)
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed max (2·conv(rev_a, w))
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)            # length = 3·n‑2
        g = 2.0 * full[n - 1:n - 1 + n]

        # diminishing learning rate
        lr = lr0 / math.sqrt(it)

        # exponentiated‑gradient update on the simplex
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise before exponentiation
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum > 0.0:
            a /= a_sum
        else:
            a = _uniform(n)

        # increase β (slowly sharpen the approximation)
        beta = min(beta_max, beta + beta_inc)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Very light simulated‑annealing finetuner
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis–Hastings style walk that swaps a small fraction of
    mass between two random positions and accepts downhill moves or
    probabilistic uphill moves.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    n = cur.size
    cur_val = 2.0 * n * _conv_fft(cur, cur).max()
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # move up to 2 % of the mass at i to position j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = 2.0 * n * _conv_fft(new_seq, new_seq).max()

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={best_val:.6f}")

        # occasional global Gaussian perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0.0:
                new_seq /= tot
            else:
                new_seq = _uniform(n)
            new_val = 2.0 * n * _conv_fft(new_seq, new_seq).max()
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={best_val:.6f}")

    return best, best_val

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The best sequence found (as a plain list of
    floats) is returned.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)

    start_time = time.time()
    # small safety margin to avoid being terminated exactly at the deadline
    final_deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Phase split (roughly 40 % / 50 % / 10 % of the total time)
    # ------------------------------------------------------------------
    small_end = start_time + 0.40 * budget_s
    big_end = start_time + 0.90 * budget_s

    # ------------------------------------------------------------------
    # ---- Phase 1 : coarse optimisation on a small problem ------------
    # ------------------------------------------------------------------
    n_small = 2000
    seeds_small = [
        _uniform(n_small),
        _random_softmax(n_small, rng),
        _triangular(n_small),
    ]

    best_small_seq = None
    best_small_val = float("inf")
    for s in seeds_small:
        val = _objective(s)
        if val < best_small_val:
            best_small_val = val
            best_small_seq = s
    print(f"[Phase 1] best initial small objective = {best_small_val:.6f}")

    # smoothed exponential‑gradient on the small problem
    if time.time() < small_end:
        print("[Phase 1] smoothed exponential‑gradient (small)")
        seq_small, val_small = _smooth_exponential_gradient(
            best_small_seq,
            deadline=min(small_end, final_deadline),
            n=n_small,
            lr0=1.2,
            beta_start=0.3,
            beta_max=25.0,
            beta_inc=0.008
        )
        if val_small < best_small_val:
            best_small_seq, best_small_val = seq_small, val_small
            print(f"[Phase 1] improved small objective = {best_small_val:.6f}")
    else:
        print("[Phase 1] no time for small optimisation")

    # ------------------------------------------------------------------
    # ---- Phase 2 : upsample and heavy optimisation on a larger grid --
    # ------------------------------------------------------------------
    n_big = 8000

    # candidate seeds for the big problem
    seeds_big = []

    # 1) upsampled version of the best small solution
    up_big = _upsample(best_small_seq, n_big)
    seeds_big.append(up_big)

    # 2) global previous solution (if it matches n_big)
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, np.ndarray)):
        try:
            arr = np.asarray(prev, dtype=np.float64)
            if arr.shape == (n_big,):
                seeds_big.append(arr / arr.sum())
        except Exception:
            pass

    # 3) fresh generic seeds
    seeds_big.append(_uniform(n_big))
    seeds_big.append(_random_softmax(n_big, rng))
    seeds_big.append(_triangular(n_big))

    # pick the best among the above to start the large optimisation
    best_big_seq = None
    best_big_val = float("inf")
    for s in seeds_big:
        val = _objective(s)
        if val < best_big_val:
            best_big_val = val
            best_big_seq = s
    print(f"[Phase 2] best initial big objective = {best_big_val:.6f}")

    # smoothed exponential‑gradient on the big problem
    if time.time() < big_end:
        print("[Phase 2] smoothed exponential‑gradient (big)")
        seq_big, val_big = _smooth_exponential_gradient(
            best_big_seq,
            deadline=min(big_end, final_deadline),
            n=n_big,
            lr0=1.0,
            beta_start=0.4,
            beta_max=30.0,
            beta_inc=0.005
        )
        if val_big < best_big_val:
            best_big_seq, best_big_val = seq_big, val_big
            print(f"[Phase 2] improved big objective = {best_big_val:.6f}")
    else:
        print("[Phase 2] no time for big optimisation")

    # ------------------------------------------------------------------
    # ---- Phase 3 : final fine‑tuning with simulated annealing ----------
    # ------------------------------------------------------------------
    if time.time() < final_deadline - 0.5:
        print("[Phase 3] simulated‑annealing fine‑tuning")
        seq_final, val_final = _anneal_optimize(best_big_seq, final_deadline)
        if val_final < best_big_val:
            best_big_seq, best_big_val = seq_final, val_final
            print(f"[Phase 3] final objective after anneal = {val_final:.6f}")
        else:
            print("[Phase 3] annealing did not improve")
    else:
        print("[Phase 3] insufficient time for annealing")

    # ------------------------------------------------------------------
    # Normalise and return the best sequence found
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_big_seq, 0.0)
    tot = final_seq.sum()
    if tot < 1e-12:

        tot = final_seq.sum()
    final_seq = final_seq / tot

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n_big})")

    # store for the next invocation
    globals()["height_sequence_1"] = final_seq.copy()

    return final_seq.tolist()
