"""
Propose a step‑function that (hopefully) beats the 1.503 bound.

Algorithm
---------

1. **Rich seed pool** – deterministic analytic windows (uniform,
   triangular, sine, Gaussian, Tukey, beta, power‑law) and many
   random Dirichlet draws.  All seeds are ℓ₁‑normalised (∑a_i = 1).

2. **Log‑space Adam optimisation** – we optimise the logits
   `v = log(a)` on the probability simplex using the smooth
   log‑sum‑exp surrogate for the max‑convolution.  An Adam optimiser
   (first‑ and second‑moment adaptation) replaces the plain SGD of
   the previous code, which gives a more stable and faster convergence.
   The temperature `β` of the soft‑max is increased slowly from an
   initial value (≈0.1) to a large one (≈200) so that the surrogate
   gradually approaches the true max.

3. **Exact sub‑gradient descent** – after the smooth phase we switch
   to the true objective.  At each iteration we move in the direction
   of the sub‑gradient at the currently maximal convolution index and
   project back onto the simplex (Euclidean projection).

4. **Aggressive simulated annealing** – a final polishing stage that
   repeatedly reshuffles the mass of a random subset of coordinates.
   The temperature drops geometrically and uphill moves are accepted
   with probability `exp(-Δ / T)`.

All stages respect a global deadline (default 1000 s).  The best
sequence found is stored as ``height_sequence_1`` for possible reuse
and returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT convolution, simplex projection, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """Exact surrogate: 2·n·max(conv(seq,seq)) assuming ∑seq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Smooth non‑negative curve generated by a few random cosine waves.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised to sum = 1."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _collect_seeds(n: int, rng: np.random.Generator, previous: np.ndarray | None) -> list[np.ndarray]:
    """Build a diversified pool of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.18))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_gaussian(n, sigma_frac=0.35))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # random Dirichlet draws
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7, 1.5):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # low‑frequency curves
    for nf in (2, 3, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # mixtures with uniform
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # previous best (if available)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

def _logspace_adam_optimize(start_vec: np.ndarray,

    v_hat = np.zeros_like(v)             # second moment

    eps_adam = 1e-8

    beta_inc = 0.4
    beta_max = 250.0

    lr0 = 0.6           # initial learning‑rate (log‑space)
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        # ----- smooth max via log‑sum‑exp -----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]   # central n entries

        # log‑space gradient
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_v ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat_corr = v_hat / (1.0 - beta2 ** it)
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps_adam)

        # increase smoothing temperature (slowly)
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    The step size decays like 1/√t and each iterate is projected
    back onto the probability simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size
        lr = 0.13 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 1_500_000,
                         init_temp_factor: float = 0.04,
                         min_temp: float = 1e-14) -> tuple[np.ndarray, float]:
    """
    Simple stochastic annealing that randomly reshuffles the mass of
    a small subset of coordinates.  The temperature cools geometrically.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9993, min_temp)

        # pick a random subset (size 2‑12)
        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # redistribute uniformly on the simplex of size k
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 60_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

# Main entry point required by the judge
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 314159, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` evaluation function within the given time budget.
    The returned object is a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Length selection – a bit larger than the previous 4000.
    # ------------------------------------------------------------------
    n = 6000          # trade‑off: more degrees of freedom, still fast FFT

    # ------------------------------------------------------------------
    # Retrieve previous best if present (so we can keep improving).
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and quick screening.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seed_pool = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seed_pool)} raw seeds.")

    # Evaluate surrogate for each seed (fast, no need to call evaluate_sequence)
    evaluated = [(vec, _objective(vec)) for vec in seed_pool]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in evaluated[:top_k]]

    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – Adam on log‑softmax (multiple restarts)
    # ------------------------------------------------------------------
    for run_idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Adam optimisation run {run_idx+1}/{len(best_seeds)} ---")
        seq_adam, val_adam = _logspace_adam_optimize(
            seed_vec, n, deadline, max_iters=400_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Overall‑Adam] new global best = {val_adam:.6f}")

        # Exact sub‑gradient projection using the current best
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=1_800_000
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[Overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – aggressive simulated annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=1_500_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[Overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (optional).
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
