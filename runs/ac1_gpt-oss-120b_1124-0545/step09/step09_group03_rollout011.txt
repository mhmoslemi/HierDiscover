assistantfinal**Solution Overview**

The evaluation function  

```
E(a) = 2·n·max( a*a ) / (∑a)²
```  

is scale‑invariant – we may normalise the step heights so that `∑a = 1`.  
After normalisation  

```
E(a) = 2·n·max( a*a )
```  

where `a*a` is the *full* linear convolution (length `2·n‑1`).  
The problem is therefore to minimise the peak of the a‑periodic
autocorrelation of a non‑negative probability vector.

The code below follows a **multi‑phase search** that proved very
effective for this task and adds a few new tricks that improve the
final bound:

1. **Deterministic seeds** – several analytic families (β‑shapes,
   arcsine, Hann, etc.) for several lengths.  
2. **Heavy random sampling** – Dirichlet draws and random β‑pdf shapes.  
3. **Smoothed Adam optimisation** – we minimise a smooth
   approximation of the `max` using the log‑sum‑exp with a slowly
   increasing temperature (β from `5` to `500`).  The optimisation is
   performed in the log‑softmax (logits) space and uses the Adam
   optimizer.  Several restarts are performed.
4. **Aggressive KL‑mirror descent** – a true sub‑gradient method for
   the non‑smooth objective.  The active set contains **all**
   convolution indices whose value is within a *relative* tolerance of
   the current maximum; the coefficient of the KL step is larger
   (`γ = 1.2`) than in the reference implementation.  The tolerance is
   decreased from `1e‑3` to `1e‑6` during the run.
5. **Local random perturbations** – occasional Gaussian noise added to
   the current best vector, followed by re‑normalisation and acceptance
   only when the objective improves.  This helps to escape tiny local
   minima.

All phases respect a global time budget (default 1000 s).  The best
vector found is returned as a plain Python list; the grader will call
`evaluate_sequence` on it.

```python
"""
minimise the peak of a self‑convolution (non‑negative step heights)

The algorithm consists of several stages:
  1) analytic seed families (uniform, Hann, Blackman, Gaussian, arcsine,
     β‑pdfs, Catalan)
  2) massive Dirichlet / β‑pdf random sampling
  3) smoothed Adam optimisation on a log‑sum‑exp approximation of max
  4) aggressive KL‑mirror descent with a decreasing active‑set tolerance
  5) short random‑perturbation polishing

All phases run inside the supplied time budget and keep the best
candidate found.  The returned list is non‑negative and will be
automatically normalised by ``evaluate_sequence``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# FFT based convolution (full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full convolution via real FFT, O(N log N)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (sum is assumed to be 1)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) – the objective for a normalised vector."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic seed families (each returns a normalised probability vector)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total > 0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0
         - a1 * np.cos(2.0 * np.pi * i / (n - 1))
         + a2 * np.cos(4.0 * np.pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.20) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return w / w.sum() if w.sum() > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return w / w.sum() if w.sum() > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    return w / w.sum() if w.sum() > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (log‑sum‑exp approximation of max)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 500.0,
    max_iter: int = 10_000_000,
    lr0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑softmax representation.

    The temperature β is linearly increased from `beta_start` to
    `beta_end`.  The loss that is minimised is
        L = log‑sum‑exp(β·conv(a,a))   (up to an additive constant)
    and its gradient is computed analytically.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max → probability vector a
        a = np.exp(logits)
        a /= a.sum()

        # self‑convolution
        conv = _conv_fft(a, a)

        # annealed temperature
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max (log‑sum‑exp) weights
        c = np.max(conv)
        w = np.exp(beta * (conv - c))
        p = w / w.sum()                     # ∑p = 1, emphasises large entries

        # gradient of smooth‑max w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])     # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1 : n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)              # simple 1/√t decay
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 2047) == 0:                # every 2048 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f}")

            # early stop if already quite good
            if best_val < 1.45:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Aggressive KL‑mirror descent (aggregated active set)
# ----------------------------------------------------------------------
def _mirror_descent(
    start_seq: np.ndarray,
    deadline: float,
    eps_start: float = 1e-3,
    eps_end: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """KL‑mirror descent using *all* indices whose convolution value is
       within a relative tolerance of the current maximum.
       The tolerance is linearly decreased from `eps_start` to `eps_end`.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    eps = 1e-30
    gamma = 1.2                     # step‑size multiplier (larger than default)

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # linear schedule for the relative tolerance
        rel_eps = eps_start + (eps_end - eps_start) * min(1.0, it / 200_000)

        active_idx = np.where(conv >= max_val - rel_eps * max_val)[0]
        if active_idx.size == 0:
            active_idx = np.array([np.argmax(conv)], dtype=int)

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # aggregated sub‑gradient
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1 : n - 1 + n]

        # KL‑mirror (exponentiated gradient) step
        lr = gamma / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)               # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        # periodic exact evaluation
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Mirror] it={it:7d} n={n:5d} best={val:.6f}")

            # exit early if we are about to run out of time
            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Tiny random‑perturbation polishing
# ----------------------------------------------------------------------
def _local_noise_search(
    seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    attempts: int = 200,
    init_scale: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Add small Gaussian noise, keep improvements."""
    a = seq.copy()
    best_val = _objective(a)
    scale = init_scale

    while attempts > 0 and time.time() < deadline:
        attempts -= 1
        noise = rng.normal(loc=0.0, scale=scale, size=len(a))
        cand = a * (1.0 + noise)
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        val = _objective(cand)
        if val < best_val:
            best_val = val
            a = cand
            print(f"[Noise] new best {val:.6f}")
        # slowly shrink the perturbation size
        scale = max(scale * 0.995, 5e-5)

    return a, best_val

    """

    # Possibly reuse a previously stored solution
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            previous = np.maximum(previous, 0.0)
            s = previous.sum()
            if s > 0:
                previous = previous / s
            else:
                previous = None
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – a few FFT‑friendly sizes
    # ------------------------------------------------------------------
    lengths = [4000, 6000, 8000, 10000, 12000]

    best_val = float("inf")

    # ------------------------------------------------------------------

        _catalan_shape,
        _uniform,
        _hann,
        _hamming,

        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),

            if val < best_val:

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:

                print(f"[Seed] n={n:5d} type=Previous  val={val:.6f}")

    # ------------------------------------------------------------------
    now = time.time()

        # Dirichlet draw

        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random β‑pdf
        a_par, b_par = rng.choice(beta_params)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        if best_val < 1.45:          # good enough – early break
            break

    # ------------------------------------------------------------------
    # Phase 3 – smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # keep the three best candidates (current + two fresh Dirichlet draws)
        candidates = [best_seq]
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        # allocate ~35 % of remaining time to Adam runs
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.35

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=500.0,
                max_iter=8_000_000,
                lr0=0.05,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – aggressive KL‑mirror‑descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            # use a two‑stage tolerance schedule
            mirror_deadline = time.time() + remaining
            seq_mirror, val_mirror = _mirror_descent(
                start_seq=best_seq,
                deadline=mirror_deadline,
                eps_start=1e-3,
                eps_end=1e-6,
            )
            if val_mirror < best_val:
                best_val = val_mirror
                best_seq = seq_mirror
                print(f"[Mirror‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – tiny random‑noise polishing (helps to leave tiny basins)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 0.5:
            seq_noise, val_noise = _local_noise_search(
                seq=best_seq,
                deadline=time.time() + remaining,
                rng=rng,
                attempts=500,
                init_scale=0.015,
            )
            if val_noise < best_val:
                best_val = val_noise
                best_seq = seq_noise
                print(f"[Noise‑Polish] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will also check)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())          # type: ignore[name-defined]
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
```
