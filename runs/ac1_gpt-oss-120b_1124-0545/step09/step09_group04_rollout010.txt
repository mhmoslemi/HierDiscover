"""
Minimal‑max‑convolution optimiser.

Search strategy:
  1. Build a diverse catalogue of analytical families (uniform, triangular,
     geometric decay, power‑law, beta‑shaped, plus random Dirichlet draws).
  2. Keep the best few seeds according to the fast surrogate
     `2·n·max(conv(a,a))`.
  3. Run an exponentiated‑gradient (mirror‑descent) sub‑gradient optimiser
     on the true max‑convolution objective while staying on the probability
     simplex.  Learning‑rate decays as 1/√t.
  4. Finish with a relatively long simulated‑annealing pass that reshuffles
     the mass of a random small subset.
The routine respects the 1000 s wall‑time, stores the result in the global
`height_sequence_1` variable for reuse, and finally returns the best vector
as a plain list of floats (sum ≈ 1, all entries non‑negative).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution via real FFT (O(n log n)).
    ``x`` and ``y`` are 1‑D real arrays.
    """
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    ``seq`` is assumed non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))`` – exactly the value that
    ``evaluate_sequence`` would produce for the same normalised vector.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Analytic families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the probability simplex."""
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet pattern."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the *exact* max‑convolution.
    The iteration stops either after ``max_iters`` steps or when ``deadline``
    is reached.  Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30  # for numerical stability

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the max at index ``max_idx``
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        # derivative of a_i * a_{max_idx-i}
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        # re‑normalise in the log‑domain to avoid overflow
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

        # keep the best ever seen
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[MD] iter={it:,} best≈{best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Very simple simulated‑annealing polish.  At each step a random small
    subset of coordinates is re‑drawn from a Dirichlet distribution that
    preserves the total mass.  Acceptance follows the Metropolis rule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)                 # size of the perturbed subset
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] iter={it:,} best≈{best_val:.6f}")

    return best_seq, best_val

        # sub‑gradient of the max entry
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient update
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)           # keep numbers in a stable range
        a = np.exp(log_a)
        s = a.sum()
        a = a / s if s > 0 else _uniform(n)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[MD] iter={it:,} best≈{best_val:.6f}")

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that repeatedly reshuffles the mass of a random small subset.
    Acceptance follows the Metropolis rule with an exponential cooling schedule.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 7)                  # subset size
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] iter={it:,} best≈{best_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal ``evaluate_sequence``
    value.  Returns the best sequence (list of floats, sum ≈ 1).  The result is
    stored in the global ``height_sequence_1`` for possible reuse.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # safety margin for the outer harness
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------
    # Length of the step function – 4000 is a good trade‑off.
    # --------------------------------------------------------------
    n = 4000

    # --------------------------------------------------------------
    # 1️⃣  Seed generation
    # --------------------------------------------------------------
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric – denser grid focusing on the promising region
    for r in np.linspace(0.90, 0.9995, 20):
        seeds.append(_geom_decay(n, r))

    # power‑law decay – broad sweep
    for p in np.arange(0.1, 5.01, 0.25):
        seeds.append(_power_decay(n, p))

    # beta shapes – coarse 7×7 grid
    for a in np.linspace(0.05, 2.0, 7):
        for b in np.linspace(0.05, 5.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # sparse random patterns – very few non‑zero entries
    for k in (3, 5, 10):
        for conc in (0.2, 0.5):
            seeds.append(_random_sparse(n, k, conc, rng))

    # a handful of random Dirichlet draws for extra diversity
    for conc in (0.02, 0.05, 0.15, 0.3, 0.8, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # ------------------------------------------------------------------
    # Previous solution (if any) – either reuse directly or interpolate.
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            prev = np.asarray(previous, dtype=np.float64)
            if prev.shape == (n,):
                seeds.append(prev / prev.sum())
            else:
                # interpolate to current length
                x_old = np.linspace(0.0, 1.0, prev.shape[0])
                x_new = np.linspace(0.0, 1.0, n)
                interp = np.interp(x_new, x_old, prev)
                seeds.append(interp / interp.sum())
        except Exception:
            pass

    # --------------------------------------------------------------
    # Evaluate seeds with the fast surrogate and keep the best few.
    # --------------------------------------------------------------
    print(f"\n--- seed generation (n={n}) ---")
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda t: t[1])
    top_k = min(12, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"kept top-{top_k} seeds; best surrogate = {evaluated[0][1]:.6f}")

    # --------------------------------------------------------------
    # 2️⃣  Mirror‑descent refinement (multiple rounds if time permits)
    # --------------------------------------------------------------
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)

    while time.time() < deadline:
        slice_deadline = min(deadline, time.time() + 120.0)  # ≤2 min per round
        seq_ref, val_ref = _subgradient_refine(
            global_best_seq, n, slice_deadline, max_iters=300_000, lr0=2.0
        )
        if val_ref < global_best_val:
            global_best_seq, global_best_val = seq_ref, val_ref
            print(f"[MD] new global best ≈ {global_best_val:.6f}")

        # stop early if this round gave almost no improvement
        if val_ref >= global_best_val * (1 - 1e-5):
            break

    # --------------------------------------------------------------
    # 3️⃣  Simulated‑annealing polishing (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=250_000
        )
        if val_sa < global_best_val:
            global_best_seq, global_best_val = seq_sa, val_sa
            print(f"[SA] polished global best ≈ {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final sanity check – let the official evaluator verify the vector.
    # --------------------------------------------------------------
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== finished ===\nbest surrogate ≈ {global_best_val:.6f} (length {len(global_best_seq)})"
    )
    # expose for reuse in later runs
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
