"""
Propose a non‑negative step‑function with a low evaluate_sequence() value.

Approach
--------
We treat the objective (for a sequence with ℓ₁‑norm 1)

    obj(a) = 2·n·max(a * a)

as a convex minimisation problem in the sense of a Frank‑Wolfe scheme.
For a current iterate `f` we compute its worst convolution value

    M = max(conv(f, f))

and then solve a linear programme

    maximise   1ᵀ b
    subject to  conv(f, b) ≤ M        (element‑wise)
                b ≥ 0

The feasible set is a polytope; the optimiser `b` points towards a direction
that can lower the worst autocorrelation.  After solving the LP we
renormalise `b` to ℓ₁‑norm 1 and perform a short line‑search on the segment
`(1‑t)·f + t·b`.  Repeating this step a few times rapidly drives the
maximum convolution down.

When the LP‐based phase stalls we polish the current best sequence with a
very cheap simulated‑annealing routine that randomly reshuffles a few
coordinates (Dirichlet re‑allocation).  All work respects the 1000 s
budget and the final sequence is stored in the global variable
``height_sequence_1`` for possible reuse.

The code is fully self‑contained; only NumPy, SciPy and the supplied
``evaluate_sequence`` are used.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Low‑level helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a sequence that is (or will be) ℓ₁‑normalised.
    Returns 2·n·max(conv(seq, seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Linear‑programming utilities
# ----------------------------------------------------------------------


def _make_constraint_structure(n: int):
    """
    For a fixed length `n` pre‑compute the row / column index arrays that
    describe the Toeplitz structure of the convolution constraints
    conv(f, b) ≤ M.
    """
    rows = []
    cols = []
    # each row k corresponds to shift k of the convolution (0 … 2n‑2)
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # the column index j = k - i runs over a contiguous interval
        cols.extend(range(k - i_end, k - i_start + 1))
        rows.extend([k] * (i_end - i_start + 1))
    rows = np.asarray(rows, dtype=np.int32)
    cols = np.asarray(cols, dtype=np.int32)
    # offset such that f_index = rows - cols
    offset = rows - cols
    return rows, cols, offset


def _solve_lp(f: np.ndarray, M: float, rows, cols, offset, time_left):
    """
    Solve the LP:
        maximise   1ᵀ b
        subject to  conv(f, b) ≤ M,   b ≥ 0
    using the pre‑computed constraint sparsity pattern.
    """
    n = len(f)
    # data for the sparse matrix A_ub : each entry is f[rows - cols]
    data = f[offset]
    A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                              # maximise sum(b)  →  minimise -sum(b)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)  # right‑hand side
    bounds = [(0.0, None)] * n

    # SciPy's HiGHS backend respects a time limit (seconds)
    res = linprog(
        c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
        method='highs',
        options={'time_limit': max(0.0, time_left)}
    )
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


# ----------------------------------------------------------------------
# Random helpers
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Draw a random point on the probability simplex."""
    return rng.dirichlet(np.ones(n))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective.  The routine respects the supplied
    time budget and returns the best sequence found as a list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Choose a modest problem size – small enough for repeated LP solves,
    # large enough that the factor 2·n does not dominate the bound.
    # ------------------------------------------------------------------
    n = 1000

    # ------------------------------------------------------------------
    # Initial seed – uniform distribution (known to be feasible).
    # ------------------------------------------------------------------
    f = np.full(n, 1.0 / n, dtype=np.float64)       # ℓ₁‑norm = 1
    best_f = f.copy()
    best_val = _objective(best_f)
    print(f"[Init] n={n}, objective={best_val:.6f}")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    iteration = 0

        except Exception as exc:

        # Normalise the LP solution to the probability simplex.
        b_sum = b.sum()
        if b_sum <= 0:

        g = b / b_sum                               # ℓ₁‑norm = 1

        # ------------------------------------------------------------------
        # Simple line‑search on the segment (1‑t)·f + t·g.
        # ------------------------------------------------------------------
        best_local_f = f

        idx = rng.choice(n, size=k, replace=False)

        if total <= 0.0:

        new_alloc = rng.dirichlet(np.ones(k)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)
