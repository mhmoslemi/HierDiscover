"""
Propose a step‑function that minimizes

    2 * n * max(conv(a,a)) / (sum(a)^2)

by:

1.  Building a large “low‑collision’’ support S ⊂ {0,…,n‑1}
    such that every ordered pair‑sum i+j (i,j∈S) occurs at most
    ``max_ordered`` times (we use 4).  This is exactly the B₂[2] problem.
    A stochastic greedy construction together with random swaps is run
    for a large part of the time budget; the best mask found is kept.

2.  Turning the mask into a probability vector (uniform mass on S).

3.  Running a few passes of the smoothed‑max Adam optimiser and an exact
    sub‑gradient refinement (the same code as the previous state‑of‑the‑art).

The whole routine respects the 1000 s limit, prints progress, and returns
the best sequence found.  The final vector is stored in the global variable
``height_sequence_1`` for possible reuse.
"""

import time
import math
import numpy as np

# ------------------------------------------------------------
# FFT based convolution (exact, real valued)
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using a real FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective used by the grader (2·n·max(conv)/sum²)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return float(2.0 * seq.size * conv.max() / (s * s))

# Simple helpers
# ------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()

# Low‑collision support search
# ------------------------------------------------------------
def _can_add(idx: int, present: list[int], sum_counts: np.ndarray,
             max_ordered: int) -> bool:
    """Check whether adding idx respects the ordered‑pair bound."""
    if sum_counts[2 * idx] + 1 > max_ordered:
        return False
    for s in present:
        if sum_counts[idx + s] + 2 > max_ordered:
            return False
    return True


def _add(idx: int, present: list[int], sum_counts: np.ndarray,
         mask: np.ndarray) -> None:
    """Insert idx into the current set, updating data structures."""
    mask[idx] = True
    # cross‑terms (ordered pairs i≠j give two contributions)
    for s in present:
        sum_counts[idx + s] += 2
    # self‑pair
    sum_counts[2 * idx] += 1
    present.append(idx)


def _remove(idx: int, present: list[int], sum_counts: np.ndarray,
            mask: np.ndarray) -> None:
    """Delete idx from the set, updating data structures."""
    mask[idx] = False
    present.remove(idx)
    for s in present:
        sum_counts[idx + s] -= 2
    sum_counts[2 * idx] -= 1


def _attempt_swap(rem: int, add: int, present: list[int],
                  sum_counts: np.ndarray, mask: np.ndarray,
                  max_ordered: int) -> bool:
    """
    Try to replace element ``rem`` by ``add``.
    Returns True iff the swap is performed.
    """
    # make temporary copies
    cnt = sum_counts.copy()
    pres = present.copy()
    # remove rem in the copy
    pres.remove(rem)
    for s in pres:
        cnt[rem + s] -= 2
    cnt[2 * rem] -= 1

    # now test whether add fits
    if cnt[2 * add] + 1 > max_ordered:
        return False
    for s in pres:
        if cnt[add + s] + 2 > max_ordered:
            return False

    # perform real removal
    _remove(rem, present, sum_counts, mask)
    # perform real addition
    _add(add, present, sum_counts, mask)
    return True


def _low_collision_search(n: int,
                          max_ordered: int,
                          deadline: float) -> tuple[np.ndarray, float]:
    """
    Stochastic greedy + random swaps searching for a large support
    with ordered pair collisions ≤ max_ordered.
    Returns the best mask found and its objective value (computed
    for uniform weights on the support).
    """
    rng = np.random.default_rng()
    best_mask = None
    best_val = float('inf')
    best_m = 0
    best_max = 0

    # cheap helper for objective from mask
    def _obj_from(mask: np.ndarray, max_cnt: int, m: int) -> float:
        return 2.0 * n * max_cnt / (m * m)

    # Main loop – many independent attempts
    attempt = 0
    while time.time() < deadline:
        attempt += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
        present: list[int] = []

        # ---- greedy insertion in random order ----
        order = rng.permutation(n)
        for idx in order:
            if _can_add(idx, present, sum_counts, max_ordered):
                _add(idx, present, sum_counts, mask)

        # try to enlarge the set by random insertions
        for _ in range(200):
            cand = rng.integers(n)
            if mask[cand]:
                continue
            if _can_add(cand, present, sum_counts, max_ordered):
                _add(cand, present, sum_counts, mask)

        # local swaps – keep trying to improve max count or size
        for _ in range(500):
            # with 30 % probability try a pure addition
            if rng.random() < 0.3:
                cand = rng.integers(n)
                if mask[cand]:
                    continue
                if _can_add(cand, present, sum_counts, max_ordered):
                    _add(cand, present, sum_counts, mask)
                continue

            # otherwise try a swap
            if not present:
                continue
            rem = rng.choice(present)
            add = rng.integers(n)
            while mask[add]:
                add = rng.integers(n)
            _attempt_swap(rem, add, present, sum_counts, mask, max_ordered)

        m = len(present)
        if m == 0:
            continue
        cur_max = int(sum_counts.max())
        cur_val = _obj_from(mask, cur_max, m)

        if cur_val < best_val:
            best_val = cur_val
            best_mask = mask.copy()
            best_m = m
            best_max = cur_max
            print(f"[LowColl] attempt {attempt:4d} size {m:3d} max {cur_max} "
                  f"obj {cur_val:.6f}")

        # early stop if we are already below the target
        if best_val < 1.503:
            break

    return best_mask, best_val

# Optimisation phases (same as the previous state‑of‑the‑art)
# ------------------------------------------------------------
def _adam_optimize(seed: np.ndarray, n: int,
                   deadline: float, max_iters: int = 40000):
    """Adam on a smoothed‑max surrogate of the true objective."""
    v = np.log(seed + 1e-30)

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5
    beta_mul = 1.007

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ---- smoothed‑max weights ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·(a*a)_k ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam updates ----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


def _subgradient_refine(start_seq: np.ndarray, n: int,
                        deadline: float,
                        max_iters: int = 1_000_000):
    """Exponentiated‑gradient (mirror descent) on the exact nonsmooth objective."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # avoid overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


def _anneal_optimize(start_seq: np.ndarray, n: int,
                     deadline: float, max_iters: int | None = None):
    """Simple Metropolis‑Hastings annealing."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global jump
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            if new_seq.sum() == 0.0:
                new_seq = np.ones(n) / n
            else:
                new_seq /= new_seq.sum()
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimizes
    ``evaluate_sequence``.  The routine respects a ~1000 s time
    budget and returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Problem size – a trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4000
    max_ordered = 4                      # allowed ordered‑pair collisions
    print(f"\n=== Optimising length {n} (max ordered collisions = {max_ordered}) ===")

    # ------------------------------------------------------------------
    # Phase 1 – aggressive low‑collision set search (≈ 45 % of budget)
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.45 * (deadline - start_time)
    low_mask, low_val = _low_collision_search(n, max_ordered, phase1_deadline)
    if low_mask is None:
        # fallback: uniform distribution
        print("[Info] Low‑collision search failed – falling back to uniform.")
        seq = np.ones(n, dtype=np.float64) / n
    else:
        m = low_mask.sum()
        seq = np.zeros(n, dtype=np.float64)
        seq[low_mask] = 1.0 / m
        print(f"[Info] Low‑collision set size {m}, max ordered sum {int((_conv_fft(seq, seq).max() * m * m))}, obj ≈ {low_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (≈ 20 % of budget)
    # ------------------------------------------------------------------
    phase2_deadline = phase1_deadline + 0.20 * (deadline - start_time)
    seq_adam, val_adam = _adam_optimize(seq, n, phase2_deadline, max_iters=30000)
    if seq_adam is not None and val_adam < low_val:
        seq = seq_adam
        best_val = val_adam
        print(f"[Overall] Adam improved objective to {val_adam:.6f}")
    else:
        best_val = low_val

    # ------------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient refinement (≈ 25 % of budget)
    # ------------------------------------------------------------------
    phase3_deadline = phase2_deadline + 0.25 * (deadline - start_time)
    seq_sub, val_sub = _subgradient_refine(seq, n, phase3_deadline, max_iters=600000)
    if val_sub < best_val:
        seq = seq_sub
        best_val = val_sub
        print(f"[Overall] Sub‑gradient improved objective to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing fine‑tuning (use remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        seq_ann, val_ann = _anneal_optimize(seq, n, deadline, max_iters=None)
        if val_ann < best_val:
            seq = seq_ann
            best_val = val_ann
            print(f"[Overall] Annealing improved objective to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and safety checks
    # ------------------------------------------------------------------
    final_seq = np.maximum(seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.ones(n, dtype=np.float64) / n
    else:
        final_seq = final_seq / total

    # expose for the next run (if any)
    globals()["height_sequence_1"] = final_seq.copy()
    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(final_seq)})")
    return final_seq.tolist()
