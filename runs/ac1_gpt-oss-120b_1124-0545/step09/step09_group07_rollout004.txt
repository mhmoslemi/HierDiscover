"""
Improved optimisation for the step‑function bound.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)**2)

with a ≥ 0 and sum(a) > 0.  The previous implementation used a hard
sub‑gradient on the exact max.  Here we employ a *smooth* exponential‑gradient
procedure: the max is approximated by a soft‑max (log‑sum‑exp) whose
gradient can be written as a convolution with a soft‑weight vector.
During optimisation the temperature ε is gradually annealed, which
focuses the gradient first on the whole autocorrelation and later on the
few dominant peaks.  A simple learning‑rate schedule together with
exponentiated‑gradient updates keeps the iterate on the simplex
automatically.  After the smooth phase we optionally run a short LP
refinement (if cvxpy is present) and finish with a lightweight simulated
annealing fine‑tuner.

The method is fast (≈1 ms per iteration for n≈4000) and fits comfortably
into the 1000 s budget, typically producing a bound well below the previous
1.511197.
"""

import time
import math
import numpy as np

# Optional cvxpy for the LP refinement step
try:
    import cvxpy as cp
    CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    CVXPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Core convolution utilities (FFT based, full linear convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    Assumes seq is non‑negative.
    """
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple initial seeds (probability vectors on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Phase 1 – Smooth exponential‑gradient descent on a soft‑max of the
#           auto‑convolution.
# ----------------------------------------------------------------------
def _smooth_expgrad(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise the true max‑convolution using a log‑sum‑exp smoothing.
    Returns the best sequence seen and its exact objective value.
    """
    a = start_seq.copy()
    n = a.size

    best_seq = a.copy()
    best_val = _objective(a)

    it = 0
    # Rough estimate of total iterations we can afford (used for epsilon schedule)
    # 0.8 s per 1000 iterations ≈ 1 ms per iteration for n≈4000.
    max_est_iters = int((deadline - time.time()) / 0.001)
    max_est_iters = max(max_est_iters, 1_000)   # protect against very short runs

    # ε schedule: start fairly large (focus on many peaks) and decay exponentially
    eps_start = 5.0
    eps_end = 0.001

    while time.time() < deadline:
        it += 1
        # ---------- compute current convolution ----------
        conv = _conv_fft(a, a)
        max_conv = conv.max()

        # ---------- annealing schedule for the smoothing temperature ----------
        frac = min(it / max_est_iters, 1.0)
        eps = eps_start * (eps_end / eps_start) ** frac   # geometric decay

        # ---------- soft‑max weights (higher weight on larger conv entries) ----------
        # shift by max to keep values ≤ 1 and avoid overflow
        w_raw = np.exp(eps * (conv - max_conv))
        w = w_raw / w_raw.sum()          # length 2n‑1, sums to 1

        # ---------- gradient of the smoothed max ----------
        # g = 2 * (reverse(a) convolved with w)  (slice to length n)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length = 3n‑2, but only first 2n‑1 needed
        g = 2.0 * full[n - 1:n - 1 + n]           # extract the n‑component part

        # ---------- step size ----------
        lr = 0.5 / math.sqrt(it)                  # diminishing steps

        # ---------- exponentiated‑gradient update (keeps positivity) ----------
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                    # numerical stabilisation
        a = np.exp(log_a)
        a /= a.sum()

        # ---------- bookkeeping ----------
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SmoothEG] it={it:8d} eps={eps:.5f} best={cur_val:.6f}")

        # occasional random jitter to escape plateaus
        if it % 5000 == 0:
            rng = np.random.default_rng(it)   # deterministic per iteration
            noise = rng.normal(scale=0.001, size=n)
            a = np.clip(a + noise, 0.0, None)
            a /= a.sum()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 2 – Linearised LP refinement (convex‑concave procedure)
# ----------------------------------------------------------------------
def _lp_linearised_refine(a: np.ndarray, deadline: float,
                         max_refine: int = 8,
                         eps_active: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Perform a handful of LP refinements where the quadratic constraints are
    linearised around the current point `a`.  Returns the best sequence seen.
    """
    if not CVXPY_AVAILABLE:
        return a, _objective(a)

    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    # cvxpy variables (created once, re‑used)
    x_var = cp.Variable(n)
    M_var = cp.Variable()

    base_constraints = [x_var >= 0,
                        cp.sum(x_var) == 1]

    for it in range(max_refine):
        if time.time() > deadline - 0.3:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]
        if active.size == 0:
            break

        # build the linearised constraints:   2·(A @ x) ≤ M + conv[active]
        A = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            A[r, i_range] = a[k - i_range]

        lin_constraints = [2 * (A @ x_var) <= M_var + conv[active]]

        prob = cp.Problem(cp.Minimize(M_var), base_constraints + lin_constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=2000,
                       abstol=1e-9,
                       reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if x_var.value is None or M_var.value is None:
            break

        a_new = np.asarray(x_var.value, dtype=np.float64)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s

        val_new = _objective(a_new)
        print(f"[LP] iter={it+1:2d} M={M:.6f} new_obj={val_new:.6f}")

        if val_new < best_val - 1e-9:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new  # continue from improved point
        else:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Phase 3 – Very light simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Simple Metropolis‑Hastings walk that moves a tiny amount of mass
    between two positions and occasionally adds a global Gaussian jitter.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        # move a small fraction of mass from i to j
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # global jitter every 1000 steps
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val
