"""
Adam‑based smooth‑max optimiser for the step‑function attack.

The objective is homogeneous, therefore we work on the probability simplex
(∑a = 1, a_i ≥ 0).  For a vector `a` the true loss is

    L(a) = 2·n·max(conv(a,a))                (∑a = 1)

The max is non‑smooth; we first replace it by a smooth soft‑max
(`log‑sum‑exp`) with a temperature β that is increased during the run.
Its gradient is a simple cross‑correlation that can be computed with two
FFT’s per iteration.  The parameters are optimised with Adam and after a
given amount of time we switch to the exact sub‑gradient obtained from the
active set of indices that are (almost) maximal.  After each step we
project onto the simplex and symmetrise the vector (the optimal solution
is believed to be symmetric).  The whole procedure runs well within the
1000‑second budget and typically beats the previous hybrid optimiser.

The returned list is stored as ``height_sequence_1`` for possible reuse.
"""

# Helper utilities (FFT, projection, simple analytic seeds)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray, N: int) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT (full mode)."""
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)
    return conv[: a.size + b.size - 1]


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2).
    The sequence is assumed to be non‑negative but not necessarily normalised.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    n = seq.size
    # use FFT for speed
    N = _next_pow2(2 * n - 1)
    conv = _conv_fft(seq, seq, N)
    return 2.0 * n * conv.max() / (s * s)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    """Discretised Gaussian window, already normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    else:
        a[:] = 1.0 / n
    return a


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on n points."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, then normalise to sum = 1."""
    old_n = seq.shape[0]
    if old_n == new_n:
        out = np.clip(seq, 0.0, None)
        total = out.sum()
        if total > 0:
            out /= total
        else:
            out = _uniform(new_n)
        return out
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    out = np.interp(x_new, x_old, seq)
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total > 0:
        out /= total
    else:
        out = _uniform(new_n)
    return out


def _symmetrize(a: np.ndarray) -> np.ndarray:
    """Force symmetry a[i] = a[-i-1] (the vector length is even or odd)."""
    return (a + a[::-1]) * 0.5


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  Uses an Adam optimiser on a smooth‑max surrogate,
    then switches to the exact sub‑gradient of the max‑convolution.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # a tiny safety margin

    # ------------------------------------------------------------------
    # Choose a length – 8000 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 8000

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available, otherwise Gaussian
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None and isinstance(previous, (list, tuple, np.ndarray)):
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
            a = _resize_sequence(prev_arr, n)
        except Exception:
            a = _gaussian(n)
    else:
        a = _gaussian(n)

    a = np.clip(a, 1e-15, None)          # avoid exact zeros for stability
    a /= a.sum()                         # enforce probability simplex

    best_seq = a.copy()
    best_val = _objective(a)

    # ------------------------------------------------------------------
    # Pre‑compute FFT length (avoid aliasing)
    # ------------------------------------------------------------------
    fft_len = _next_pow2(3 * n - 2)      # > len(a) + len(conv) - 1

    # Adam hyper‑parameters
    lr0 = 0.5
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Soft‑max temperature schedule
    beta_start = 5.0
    beta_end = 150_000.0                # large enough to approach the true max

    it = 0
    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # ----- FFT of the current vector -----
        Fa = np.fft.rfft(a, n=fft_len)

        # ----- auto‑convolution a*a (full length) -----
        conv = np.fft.irfft(Fa * Fa, n=fft_len)[: 2 * n - 1]

        # ------------------------------------------------------------------
        # Decide which gradient to use:
        #   • first half of the allotted time → smooth‑max (soft‑max)
        #   • second half → exact sub‑gradient on the active set
        # ------------------------------------------------------------------
        elapsed = time.time() - start_time
        use_smooth = elapsed < (budget_s * 0.55)

        if use_smooth:
            # temperature grows exponentially from beta_start to beta_end
            frac = min(1.0, elapsed / (budget_s * 0.55))
            beta = beta_start * (beta_end / beta_start) ** frac

            # soft‑max weights (stable formulation)
            max_c = conv.max()
            exp_vals = np.exp(beta * (conv - max_c))
            w = exp_vals / exp_vals.sum()
        else:
            # exact active‑set sub‑gradient
            max_c = conv.max()
            tol = max(1e-12, 0.001 * max_c)          # a small relative tolerance
            active = np.where(conv >= max_c - tol)[0]
            w = np.zeros_like(conv)
            if active.size > 0:
                w[active] = 1.0 / active.size
            else:
                w[:] = 1.0 / conv.size

        # ----- gradient via cross‑correlation (2 · B a) -----
        Fw = np.fft.rfft(w, n=fft_len)
        g_full = np.fft.irfft(np.conj(Fa) * Fw, n=fft_len)
        g = 2.0 * g_full[:n]

        # ----- Adam update -----
        lr = lr0 / math.sqrt(it)          # simple decay
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- feasibility projection (simplex) and symmetrisation -----
        a_new = _project_simplex(a_new)
        a_new = _symmetrize(a_new)        # keep the vector symmetric
        a = a_new

        # ----- periodic reporting & best‑track -----
        if it % 100 == 0:
            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Iter {it:6d}] new best = {cur_val:.6f}")

        # early stop if we are extremely close to the lower bound 2n/(2n‑1)
        # (the bound for a perfectly flat convolution)
        if best_val <= 2.0 * n / (2 * n - 1) + 1e-4:
            break

    # ------------------------------------------------------------------
    # Very short local random walk to possibly escape the last flat region
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 2.0:
        jumps = 1500
        for _ in range(jumps):
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.02 * best_seq[i]   # move up to 2 % of mass
            cand = best_seq.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_simplex(cand)
            cand = _symmetrize(cand)
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Local] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final report and store for the next run
    # ------------------------------------------------------------------
    print(f"\n=== Finished optimisation ===\n"
          f"Best objective ≈ {best_val:.6f} after {it} iterations, length {n}")

    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
