    while time.time() < deadline and no_improve_iters < 20000:
        # try random addition

        if non_support.size == 0:
            break
        cand = rng.choice(non_support)
        # check if add allowed
        if sum_counts[2*cand] + 1 <= L:
            if support.size == 0:
                can_add = True
            else:
                sums = cand + support
                if not np.any(sum_counts[sums] + 2 > L):
                    can_add = True
                else:
                    can_add = False
            if can_add:
                # perform addition
                mask[cand] = True
                sum_counts[2*cand] += 1
                if support.size:
                    sum_counts[sums] += 2
                support = np.append(support, cand)
                no_improve_iters = 0
                continue
        # try swap
        if support.size > 0:
            # pick removal
            rem = rng.choice(support)
            other = support[support != rem]
            # remove rem
            sum_counts[2*rem] -= 1
            if other.size:
                sum_counts[rem + other] -= 2
            mask[rem] = False
            # try to add cand (the same random candidate was previously chosen and didn't work)
            # after removal, recompute non_support (includes rem)
            # choose a new candidate because we might want to add rem itself back, but that would revert.
            cand2 = rng.choice(np.where(~mask)[0])
            # check add after removal
            ok = False
            if sum_counts[2*cand2] + 1 <= L:

                else:

                    if not np.any(sum_counts[sums2] + 2 > L):

"""
propose_candidate
-----------------

A 1000 s time‑bounded search for a non‑negative step function that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works in three stages:

1. **Large low‑collision support** –  
   We greedily construct a binary mask whose ordered pair‑sums never
   exceed ``max_ordered`` (set to 5).  After a quick greedy phase we
   improve the mask by random insertions and swaps while respecting the
   collision bound.  The final mask used for the step function is a
   uniform distribution on this support.

2. **Adam optimisation** –  
   The uniform mask is refined with Adam on a smooth approximation of the
   max‑convolution, allowing the weights to become non‑uniform while
   keeping the objective low.

3. **Exact sub‑gradient refinement** –  
   Finally an exponentiated‑gradient (mirror‑descent) step uses the true
   non‑smooth objective to polish the solution.

The whole pipeline respects the overall time budget and stores the best
sequence in the global variable ``height_sequence_1`` for possible reuse
in a later run.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution (exact, real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)²).
    ``seq`` is assumed non‑negative and not all zero.
    """
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    """Uniform probability vector of length n."""
    return np.ones(n, dtype=np.float64) / n


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimisation using a smoothed max over the convolution
# ----------------------------------------------------------------------
def _adam_optimize(seed: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 40000):
    """
    Adam on a smoothed‑max surrogate of max(conv(a,a)).
    Returns the best vector seen and its true objective.
    """
    v = np.log(seed + 1e-30)          # logits on the simplex

    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5               # temperature for the soft‑max over the convolution
    beta_mul = 1.007         # increase slowly → tighter approx. of max

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)                      # current point on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ---- smoothed‑max weights ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·(a*a)_k ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ---- Adam moment update ----
        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1.0 - beta1 ** t)
        m2_hat = m2 / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient refinement (mirror descent on the true objective)
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 1_000_000):
    """
    Exponentiated‑gradient (mirror descent) on the exact non‑smooth objective.
    Returns the best sequence encountered.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)          # prevent overflow
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val

        cand = rng.choice(non_support)

        # ---- try simple insertion ----
        if sum_counts[2 * cand] + 1 <= max_ordered:
            if support.size == 0:
                can_add = True
            else:
                sums = cand + support
                can_add = not np.any(sum_counts[sums] + 2 > max_ordered)

            if can_add:
                mask[cand] = True
                sum_counts[2 * cand] += 1
                if support.size:
                    sum_counts[sums] += 2
                support = np.append(support, cand)
                no_improve_iters = 0
                continue

        # ---- try a swap (remove one, insert another) ----
        if support.size:
            rem = rng.choice(support)
            other = support[support != rem]

            # remove `rem`
            sum_counts[2 * rem] -= 1
            if other.size:
                sum_counts[rem + other] -= 2
            mask[rem] = False

            # after removal pick a (maybe different) candidate
            cand2 = rng.choice(np.where(~mask)[0])

            ok = False
            if sum_counts[2 * cand2] + 1 <= max_ordered:
                if other.size == 0:
                    ok = True
                else:
                    sums2 = cand2 + other
                    ok = not np.any(sum_counts[sums2] + 2 > max_ordered)

            if ok:
                mask[cand2] = True
                sum_counts[2 * cand2] += 1
                if other.size:
                    sum_counts[cand2 + other] += 2
                support = np.append(other, cand2)
                no_improve_iters = 0
                continue
            else:
                # revert the removal
                mask[rem] = True
                sum_counts[2 * rem] += 1
                if other.size:
                    sum_counts[rem + other] += 2
                # support unchanged
        no_improve_iters += 1

    return mask


def _search_low_collision_set_improved(n: int,
                                       target_m: int,
                                       max_ordered: int,
                                       rng: np.random.Generator,
                                       deadline: float) -> np.ndarray:
    """
    Greedy construction of a set S⊂{0,…,n‑1} with at most ``max_ordered``
    ordered collisions per sum.  The function first runs many greedy
    attempts, keeps the largest mask, and then applies a local random
    improvement (insert+swap) within the remaining time.
    Returns the index array of the best set found.
    """
    best_mask = np.zeros(n, dtype=bool)
    best_size = 0
    attempts = 0

    # ---- greedy phase -------------------------------------------------
    greedy_deadline = deadline - 5.0  # reserve a few seconds for improvement
    while time.time() < greedy_deadline:
        attempts += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.int16)

        order = rng.permutation(n)
        for cand in order:
            # self‑pair
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue
            existing = np.where(mask)[0]
            if existing.size:
                sums = cand + existing
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue
            # accept candidate
            mask[cand] = True
            sum_counts[2 * cand] += 1
            if existing.size:
                sum_counts[sums] += 2
            if mask.sum() >= target_m:
                break

        cur_sz = mask.sum()
        if cur_sz > best_size:
            best_size = cur_sz
            best_mask = mask.copy()
            print(f"[Greedy] attempt {attempts:4d} size {cur_sz}")

        # early exit if we already reached the target
        if best_size >= target_m:
            break

    # ---- local improvement -------------------------------------------
    if time.time() < deadline and best_mask.sum() > 0:
        # recompute exact ordered‑pair counts for the best mask
        indicator = np.zeros(n, dtype=np.float64)
        indicator[best_mask] = 1.0
        sum_counts = np.rint(_conv_fft(indicator, indicator)).astype(np.int16)

        improved_mask = _improve_mask(best_mask.copy(),
                                      sum_counts,
                                      max_ordered,
                                      rng,
                                      deadline)
        improved_sz = improved_mask.sum()
        if improved_sz > best_size:
            best_size = improved_sz
            best_mask = improved_mask
            print(f"[Improve] increased size to {improved_sz}")

    return np.where(best_mask)[0]

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    # safety margin for interpreter overhead
    deadline = start_time + budget_s - 5.0

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – 8000 gives a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 8000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Phase 1 – find a large low‑collision support set
    # ------------------------------------------------------------------
    max_ordered = 5                                      # allow ≤5 ordered collisions
    # theoretical optimum: |S| ≈ sqrt(2·n·max_ordered)
    target_m = int(0.94 * math.sqrt(2.0 * n * max_ordered))
    set_deadline = start_time + 0.35 * (deadline - start_time)   # ~35 % of total time
    lowcoll_idx = _search_low_collision_set_improved(
        n=n,
        target_m=target_m,
        max_ordered=max_ordered,
        rng=rng,
        deadline=set_deadline,
    )
    if lowcoll_idx.size == 0:
        print("[Info] Low‑collision set not found – falling back to previous solution.")
        previous = globals().get("height_sequence_1")
        if previous is not None:
            seq = np.asarray(previous, dtype=np.float64)
            seq = np.clip(seq, 0.0, None)
            if seq.sum() == 0.0:
                seq = _uniform(n)
        else:
            seq = _uniform(n)
    else:
        # uniform mass on the discovered support
        seq = np.zeros(n, dtype=np.float64)
        seq[lowcoll_idx] = 1.0 / lowcoll_idx.size

    # ------------------------------------------------------------------
    # Normalise and evaluate the initial candidate
    # ------------------------------------------------------------------
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = _uniform(n)
    else:
        seq /= seq.sum()
    best_seq = seq.copy()
    best_val = _objective(best_seq)
    print(f"[Init] support size {len(lowcoll_idx)} → objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (smooth max approximation)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        adam_deadline = min(deadline - 10.0, time.time() + 200.0)  # up to ~200 s
        seq_adam, val_adam = _adam_optimize(best_seq, n, adam_deadline,
                                            max_iters=50000)
        if seq_adam is not None and val_adam < best_val:
            best_seq, best_val = seq_adam, val_adam
            print(f"[Adam] improved objective → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 2.0
        seq_sub, val_sub = _subgradient_refine(best_seq, n, sub_deadline,
                                               max_iters=300_000)
        if seq_sub is not None and val_sub < best_val:
            best_seq, best_val = seq_sub, val_sub
            print(f"[SubGrad] improved objective → {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation and safety checks
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(final_seq)})")

    # expose for the next iteration (if any)
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
