"""
Algorithm:
1. Generate a diverse pool of ℓ₁‑normalised candidate sequences of length n.
   Includes uniform, triangular, several analytic families, a
   1/√((i+1)(n-i)) shape, power‑law shapes and random Dirichlet draws.
2. Keep the few best seeds w.r.t. the cheap surrogate 2·n·max(conv).
3. Starting from the best seed run a *flattening* optimisation that
   minimises the L2 deviation of the convolution from the constant value
   1/(2n‑1).  The gradient of this surrogate can be computed with two FFT‑based
   convolutions per iteration; after each step we project back onto the
   probability simplex.
4. Refine the obtained vector with an exact sub‑gradient descent on the true
   max‑convolution objective.
5. Finish with a simulated‑annealing polishing phase that randomly reshuffles
   the mass of a small subset of coordinates.
All stages respect the global deadline (default 1000 s).  The best sequence
found is stored in the global name ``height_sequence_1``.
"""

import time
import numpy as np

# Helper FFT convolution
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv

# Surrogate objective (assumes sum(x)=1)
def _objective(seq: np.ndarray) -> float:
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Projection onto the probability simplex (Euclidean)
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (already ℓ₁‑normalised)
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _inv_sqrt_shape(n: int) -> np.ndarray:
    # a_i ∝ 1 / sqrt((i+1)*(n-i))
    i = np.arange(1, n + 1, dtype=np.float64)  # 1 … n
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _power_shape(n: int, p: float) -> np.ndarray:
    # a_i ∝ ((i+1)*(n-i))^{-p/2}
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i * (n - i + 1.0)) ** (-p / 2.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)

def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)

def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed generation
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    seeds = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))
    seeds.append(_inv_sqrt_shape(n))

    # Power‑law family (different exponents)
    for p in np.linspace(0.1, 2.0, 12):
        seeds.append(_power_shape(n, p))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mix of uniform with inverse‑sqrt shape
    for mix_alpha in (0.2, 0.5, 0.8):
        seeds.append(_mix_shapes(_uniform(n), _inv_sqrt_shape(n), mix_alpha))

    # Previous solution if available
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

# ---------- L2 flattening optimizer ----------
def _flatten_l2_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise sum_k (conv(a,a)[k] - C)^2 where C = 1/(2n-1) using projected
    gradient descent.  Returns best sequence (w.r.t. the true max‑convolution
    surrogate) seen during the run.
    """
    a = start_vec.copy()
    # make sure we are on the simplex
    a = _proj_simplex(a)

    C = 1.0 / (2 * n - 1)           # desired constant value of the convolution
    best_seq = a.copy()
    best_val = _objective(a)       # surrogate max‑conv value

    # momentum variables
    velocity = np.zeros_like(a)
    beta = 0.9
    lr0 = 0.3

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)                     # full linear convolution
        # surrogate objective for early stopping
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[L2] it={it:6d} best={cur_val:.6f}")

        # error vector
        e = conv - C

        # gradient = 2 * (e * a_rev)_{central n}
        rev_a = a[::-1]
        full = _conv_fft(e, rev_a)                 # length 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]

        # learning rate schedule (1/√t)
        lr = lr0 / np.sqrt(it)

        # momentum update
        velocity = beta * velocity + (1.0 - beta) * grad
        a = a - lr * velocity

        # projection onto the simplex (ensures non‑negativity and sum=1)
        a = _proj_simplex(a)

    # final evaluation
    final_val = _objective(best_seq)
    return best_seq, final_val

# ---------- Exact sub‑gradient descent ----------
def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size decays as 1/√t
        lr = 0.25 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val

# ---------- Simulated annealing polishing ----------
def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling
        T = max(T * 0.9995, min_temp)

        # choose a random subset of coordinates (size 2…12)
        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # redistribute the total mass using a Dirichlet draw
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

"""
Algorithm Overview
------------------

1. **Seed generation** – Build a diverse pool of ℓ₁‑normalised candidates of length
   *n* (4000).  The pool contains uniform/triangular windows, a few sinusoidal
   and Gaussian shapes, a dedicated “inverse‑sqrt” shape
   a_i ∝ 1/√((i+1)(n‑i)), several power‑law families
   a_i ∝ ((i+1)(n‑i))^{-p/2}, random Dirichlet draws and mixtures with the
   uniform vector.  If a previous solution exists it is also added.

2. **Cheap screening** – Evaluate the cheap surrogate ``2·n·max(conv)`` (the
   denominator is 1 because every seed is ℓ₁‑normalised).  Keep the best *k*
   seeds (k=5).

3. **L2 flattening** – Starting from each of the retained seeds we minimise the
   L₂ deviation of the convolution from the constant value
   C = 1/(2n‑1):

       Φ(a) = Σ_k ( (a*a)_k – C )² .

   The gradient of Φ can be written in closed form and computed with two
   FFT‑based convolutions per iteration.  After each gradient step we project
   back onto the probability simplex (non‑negative, sum = 1).  A simple
   momentum scheme with a 1/√t learning‑rate schedule is used.  The best
   vector (according to the true max‑convolution objective) is kept.

4. **Exact sub‑gradient descent** – Using the vector obtained from step 3 we
   perform a classic sub‑gradient descent on the genuine objective
   ``2·n·max(conv(a,a))``.  At each iteration the active convolution index is
   identified, the corresponding sub‑gradient is applied and the result is
   projected onto the simplex.

5. **Simulated‑annealing polishing** – Finally a stochastic annealing loop
   reshuffles the mass of a random subset of coordinates.  The temperature
   decays geometrically; uphill moves are accepted with the usual Metropolis
   probability.  The best sequence of the whole run is stored in the global
   variable ``height_sequence_1`` for possible reuse.

All stages respect the global deadline (default 1000 s).  The function
``propose_candidate`` returns the best sequence found as a plain ``list`` of
floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)).
    Assumes Σseq = 1, which holds for all vectors we keep on the simplex.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _inv_sqrt_shape(n: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt((i+1)*(n-i)) – symmetric, heavily weighted at the edges."""
    i = np.arange(1, n + 1, dtype=np.float64)          # 1 … n
    a = 1.0 / np.sqrt(i * (n - i + 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^{-p/2}, symmetric power‑law family."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i * (n - i + 1.0)) ** (-p / 2.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ------------------------------------------------------------
# Seed collection
# ------------------------------------------------------------

def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))
    seeds.append(_inv_sqrt_shape(n))

    # Power‑law families (different exponents)
    for p in np.linspace(0.1, 2.0, 12):
        seeds.append(_power_shape(n, p))

    # Random Dirichlet draws (various concentrations)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Mixtures of uniform with the inverse‑sqrt shape
    for mix_alpha in (0.2, 0.5, 0.8):
        seeds.append(_mix_shapes(_uniform(n), _inv_sqrt_shape(n), mix_alpha))

    # Previous solution, if any
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


# ------------------------------------------------------------
# L2 flattening optimiser (smooth surrogate)
# ------------------------------------------------------------

def _flatten_l2_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Minimise Σ_k (conv(a,a)[k] – C)² with C = 1/(2n‑1) using projected
    gradient descent (momentum).  Returns the best vector seen w.r.t. the true
    max‑convolution surrogate.
    """
    a = _proj_simplex(start_vec.copy())
    C = 1.0 / (2 * n - 1)                     # target constant value of the convolution

    best_seq = a.copy()
    best_val = _objective(a)

    velocity = np.zeros_like(a)
    beta = 0.9
    lr0 = 0.3

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)                # full linear convolution
        cur_val = 2.0 * n * np.max(conv)

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[L2] it={it:6d} best={cur_val:.6f}")

        # error vector (how far we are from the flat target)
        e = conv - C

        # gradient = 2 * (e * a_rev)_{central n}
        rev_a = a[::-1]
        full = _conv_fft(e, rev_a)            # length 3n‑2
        grad = 2.0 * full[n - 1:n - 1 + n]

        # learning‑rate schedule (1/√t) with momentum
        lr = lr0 / np.sqrt(it)
        velocity = beta * velocity + (1.0 - beta) * grad
        a = a - lr * velocity

        # projection back onto the simplex (keeps non‑negativity & sum=1)
        a = _proj_simplex(a)

    # Final evaluation of the best among all iterates
    final_val = _objective(best_seq)
    return best_seq, final_val


# ------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ------------------------------------------------------------

def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on 2·n·max(conv(a,a))."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size decays as 1/√t
        lr = 0.25 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ------------------------------------------------------------
# Simulated annealing polishing
# ------------------------------------------------------------

def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 800_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles a random subset of coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        # pick a random subset (size 2 … 12) and redistribute its total mass
        k = rng.integers(2, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val
