def _lp_refine(f, deadline, max_iters=10, t=0.5):
    n = len(f)
    best_f = f.copy()
    best_val = _objective(f)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        M = np.max(_conv_fft(f,f))
        # Build A_ub
        L = 2*n - 1
        # using broadcasting
        idx = np.arange(L)[:, None] - np.arange(n)[None, :]
        mask = (idx >= 0) & (idx < n)
        A = np.zeros((L, n), dtype=np.float64)
        A[mask] = f[idx[mask]]
        # Solve LP
        c = -np.ones(n, dtype=np.float64)

        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                      method='highs', options={'presolve': True})
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g
        # Mix
        f = (1 - t) * f + t * g
        # Ensure simplex (numerical)
        f = _proj_simplex(f)
        # Symmetrize
        f = (f + f[::-1]) / 2
        val = _objective(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP] iter {it+1} best {val:.6f}")
        # Optionally reduce t
        t = max(0.1, t * 0.9)

def _subgradient_descent(f, deadline, max_iters=100000):
    n= len(f)
    best_f = f.copy()
    best_val = _objective(f)
    for it in range(1, max_iters+1):
        if time.time()>deadline:
            break
        conv = _conv_fft(f,f)
        max_k = np.argmax(conv)
        # subgradient vector:
        g = np.zeros(n)
        i_start = max(0, max_k - (n-1))
        i_end = min(n-1, max_k)
        # for i in [i_start, i_end]
        i_range = np.arange(i_start, i_end+1)
        rev_idx = max_k - i_range
        g[i_range] = 2.0 * f[rev_idx]
        # step size:

        # project onto simplex

"""
Non‑negative step‑function optimizer.

The algorithm proceeds in four stages within the given time budget:

1. **Initialisation** – start from the previous solution (if any) or a uniform
   distribution, normalised to sum = 1.

2. **Linear‑programming refinement** – given the current sequence `f`,
   compute `M = max(conv(f,f))`.  Solve the LP
       maximise   Σ g_j
       subject to conv(f,g) ≤ M  (linear in g) ,  g ≥ 0 .
   The solution `g` is re‑normalised and mixed with `f`.  Repeating this a
   handful of times quickly lowers the max‑convolution.

3. **Smooth‑surrogate Adam optimisation** – optimise the log‑sum‑exp
   (smooth‑max) surrogate of the max‑convolution while staying on the
   probability simplex (log‑softmax parametrisation).

4. **Exact projected sub‑gradient descent** – finish with a few hundred
   thousand iterations of true sub‑gradient descent on the max‑convolution.
   A final short simulated‑annealing polish is performed if time permits.

All intermediate sequences are kept on the simplex, are symmetrised
(`a[i]=a[n-1-i]`) and projected onto the simplex after each update.
The best sequence found is stored in the global variable ``height_sequence_1``
and returned as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact evaluation for L1‑normalised sequences (Σ a_i = 1).
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return float(2.0 * n * np.max(conv))


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Analytic families (producing ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """U‑shaped geometric: a_i = r^i + r^{n-1-i}."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = r ** i + r ** (n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑shaped density on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1.0) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray, n: int, deadline: float,
                   max_iters: int = 1_000_000) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Parameters are the logits of the soft‑max representation.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)          # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta_temp = 5.0          # start temperature
    beta_mul = 1.004
    beta_max = 250.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # stays on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[Adam] it={it:6d} best={best_val:.6f}")

        # raise temperature slowly so we approach the true max
        beta_temp = min(beta_temp * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta_temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum                               # softmax over the convolution entries

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)              # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]    # central n entries

        # transform to log‑softmax coordinates
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # clipping for stability
        max_norm = 6.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        # Adam update
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.6
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int, deadline: float,
                      max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # Polyak‑style step size (decrease slowly)
        lr = 0.15 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

        # keep symmetry (helps flattening)
        a = (a + a[::-1]) / 2.0

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int, deadline: float,
                         max_iters: int = 300_000,
                         init_temp_factor: float = 0.025,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that redistributes the mass of a small
    subset of indices (Dirichlet reshuffle).
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        k = rng.integers(2, 13)          # move 2 … 12 indices
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 10000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, deadline: float,
               max_iters: int = 12, mix_factor: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Linear‑programming stage described in the problem statement.
    Starting from `f` (already on the simplex), each iteration solves

        maximise   Σ g_j
        subject to conv(f,g) ≤ M   (M = max conv(f,f))
                   g ≥ 0.

    The solution `g` is re‑normalised, mixed with `f` (mix_factor),
    projected back onto the simplex and symmetrised.
    """
    n = len(f)
    best_f = f.copy()
    best_val = _objective(best_f)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current maximum of the self‑convolution
        M = np.max(_conv_fft(f, f))

        # build constraint matrix A (shape (2n‑1, n)):
        # conv(f,g)[k] = Σ_i f_i * g_{k-i}
        L = 2 * n - 1
        # broadcasting approach – modest memory for n ≤ 3000
        idx = np.arange(L)[:, None] - np.arange(n)[None, :]
        mask = (idx >= 0) & (idx < n)
        A = np.where(mask, f[idx], 0.0)          # (L, n)

        # Linear program: maximise Σ g_j  → minimise -Σ g_j
        c = -np.ones(n, dtype=np.float64)
        b = M * np.ones(L, dtype=np.float64)

        # Solve with HiGHS (fast interior‑point/simplex)
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True})

        if not res.success:
            print(f"[LP] iteration {it}: linprog failed – {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # normalise and mix
        g = g / sum_g
        f = (1.0 - mix_factor) * f + mix_factor * g
        f = _proj_simplex(f)
        f = (f + f[::-1]) / 2.0                     # enforce symmetry

        cur_val = _objective(f)
        if cur_val < best_val:
            best_val = cur_val
            best_f = f.copy()
            print(f"[LP] iter {it} best={cur_val:.6f}")

        # slowly reduce the mixing factor (helps convergence)
        mix_factor = max(0.05, mix_factor * 0.9)

    return best_f, best_val
