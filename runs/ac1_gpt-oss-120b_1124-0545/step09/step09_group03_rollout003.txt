"""
Propose a step‑function (non‑negative sequence) that minimises the
``evaluate_sequence`` objective.

The algorithm consists of several stages that respect a 1000 s budget:

1. **Analytic seed sweep** – a rich collection of families (uniform,
   arcsine, Beta‑pdf, edge‑power, Gaussian) is evaluated for a set of
   lengths.  The best seed is kept.
2. **Random Dirichlet / Beta sampling** – many random draws explore the
   simplex.  Occasionally mixture of two seeds is also sampled.
3. **Smoothed Adam optimisation** – the log‑sum‑exp smooth approximation
   of the max‑convolution is minimised using Adam.  Several restarts with
   different hyper‑parameters are performed.
4. **Multi‑tolerance mirror‑descent** – a KL‑mirror‑descent with an
   adaptive active‑set (all convolution entries within a relative
   tolerance of the current maximum) is run.  The tolerance decays
   from 0.1 to 1e‑6.
5. **Local stochastic hill‑climbing** – small random mass transfers
   between coordinates are accepted only if they improve the exact
   objective.  The step size is adapted on the fly.

All phases continuously check the remaining time and stop early if the
budget is exhausted.  The best sequence found (normalised to sum = 1)
is returned as a plain Python list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, log, pi, exp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used for comparison.
    For a normalised sequence (sum == 1) this equals ``2·n·max(conv)``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total > 0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = (a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1)))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _edge_power_shape(n: int, p: float) -> np.ndarray:
    """Edge‑heavy shape: a_i ∝ (i+1)^-p + (n-i)^-p."""
    i = np.arange(n, dtype=np.float64)
    w = (i + 1) ** (-p) + (n - i) ** (-p)
    total = w.sum()
    return w / total if total > 0 else _uniform(n)

# Random generators
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smoothed Adam optimisation (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    max_iter: int = 4_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam.
    ``start_seq`` must be normalised (sum == 1).
    Returns the best sequence found and its exact objective value.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max
        a = np.exp(logits)
        a_sum = a.sum()
        a = a / a_sum

        # convolution
        conv = _conv_fft(a, a)

        # temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp
        c = conv.max()
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # occasional exact evaluation
        if (it & 1023) == 0:  # every 1024 iterations
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f}")

            if best_val < 1.45:  # early exit if we are already excellent
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Aggressive KL‑mirror descent with decreasing tolerance
# ----------------------------------------------------------------------
def _mirror_descent_refine(
    start_seq: np.ndarray,
    deadline: float,
    start_tol: float = 0.1,
    end_tol: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates all convolution
    entries within a relative tolerance of the current maximum.
    The tolerance decays exponentially from ``start_tol`` to ``end_tol``.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())

        # exponential tolerance decay
        # decay factor chosen so that after ~1e6 iterations we are near end_tol
        decay = 0.999995
        current_tol = max(end_tol, start_tol * (decay ** it))

        delta = current_tol * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        # build indicator on full convolution length
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # aggregated sub‑gradient
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # exponentiated‑gradient step (KL‑mirror)
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)  # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        # periodic evaluation
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Mirror] it={it:7d} n={n:5d} best={val:.6f}")

            if time.time() > deadline - 1.0:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Stochastic hill‑climbing (local perturbations)
# ----------------------------------------------------------------------
def _hill_climb(
    start_seq: np.ndarray,
    deadline: float,
    init_step: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing: repeatedly move a small amount of
    mass from one coordinate to another and keep the change if the exact
    objective improves.  The step size adapts during the run.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    rng = np.random.default_rng()
    step = init_step

    while time.time() < deadline:
        # pick source with enough mass
        src = rng.integers(0, n)
        if a[src] < step * 0.1:
            continue
        dst = rng.integers(0, n)
        if src == dst:
            continue
        delta = step * rng.random()
        delta = min(delta, a[src])
        new_a = a.copy()
        new_a[src] -= delta
        new_a[dst] += delta
        val = _objective(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
            step = min(step * 1.05, 0.1)  # slowly enlarge step on success
            if best_val < 1.45:
                break
        else:
            step = max(step * 0.99, 1e-4)  # shrink step on failure

    return best_seq, best_val

        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
        lambda n: _edge_power_shape(n, 0.3),
        lambda n: _edge_power_shape(n, 0.5),
        lambda n: _edge_power_shape(n, 0.8),
        lambda n: _edge_power_shape(n, 1.2),
    )

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:15s} val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive random sampling (Dirichlet + Beta + mixture)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.30
    sample_deadline = now + time_for_sampling
    alphas = [0.5, 1.0, 2.0]
    beta_params = [(0.3, 0.3), (0.5, 0.5), (1.0, 1.0),
                   (2.0, 2.0), (0.5, 2.0), (2.0, 0.5)]
    mix_families = list(seed_families)

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet draw
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Random‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Random Beta‑pdf shape
        a_par, b_par = rng.choice(beta_params)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Random‑Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        # Random mixture of two analytic families
        f1, f2 = rng.choice(mix_families, size=2, replace=False)
        w = rng.random()
        seq = w * f1(n) + (1.0 - w) * f2(n)
        seq = seq / seq.sum()
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Mix] n={n:5d} w={w:.2f} ({f1.__name__},{f2.__name__}) val={val:.6f}")

        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # create a small pool of promising starts (best + a few random)
        candidates = [best_seq]
        top_n = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_n, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=300.0,
                max_iter=5_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val, best_seq = val_adam, seq_adam.copy()
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

            # stop early if no time left for next run
            if time.time() >= adam_deadline:
                break

    # ------------------------------------------------------------------
    # Phase 4 – aggressive mirror‑descent polishing with decaying tolerance
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 2.0:
            md_deadline = time.time() + remaining * 0.30
            seq_md, val_md = _mirror_descent_refine(
                start_seq=best_seq,
                deadline=md_deadline,
                start_tol=0.1,
                end_tol=1e-6,
            )
            if val_md < best_val:
                best_val, best_seq = val_md, seq_md.copy()
                print(f"[Mirror‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – stochastic hill‑climbing fine‑tuning
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 1.0:
            hc_deadline = time.time() + remaining * 0.20
            seq_hc, val_hc = _hill_climb(best_seq, hc_deadline, init_step=0.02)
            if val_hc < best_val:
                best_val, best_seq = val_hc, seq_hc.copy()
                print(f"[Hill‑Climb] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will run evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
