"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm is a multi‑stage search that respects a 1000 s budget.

1. **Analytic seeds** – a variety of classic windows (uniform, Hann,
   Gaussian, Beta‑pdf, Catalan, …) are generated for several lengths.
   The best seed (lowest exact objective) is kept.

2. **Heavy random sampling** – Dirichlet draws with a wide range of
   concentration parameters (α∈[0.1,5]) together with random Beta‑pdf
   shapes and simple mixtures of two Beta shapes are sampled while
   ≈30 % of the total budget remains.

3. **Smoothed Adam optimisation** – starting from the few best
   candidates we run an Adam optimiser on a smooth approximation of
   ``max(conv(a,a))`` obtained via a log‑sum‑exp with an annealed
   temperature β∈[5,β_end].  Several independent restarts with
   different β_end values (200 … 600) are performed; the best exact
   objective is retained.

4. **Aggressive KL‑mirror‑descent polishing** – a sub‑gradient method
   where *all* near‑maxima of the convolution (indices whose value is
   within a relative tolerance of the current maximum) are aggregated.
   The update is an exponentiated‑gradient step (mirror descent on the
   simplex).  This phase usually squeezes the last few digits.

5. **Optional L‑BFGS‑B refinement** – if SciPy is available a few
   L‑BFGS‑B steps on the same smooth objective are done as a final
   polish.

All phases monitor the remaining time and stop early if the budget is
exhausted.  The returned list is non‑negative, sums to 1 (the
evaluation function is scale‑invariant) and is already a plain Python
list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, log, pi, exp, ceil

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` when the sequence is
    already normalised (sum == 1).  Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total > 0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1 - np.cos(2 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2 * pi * i / (n - 1)) + a2 * np.cos(4 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return w / w.sum() if w.sum() > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    return w / w.sum() if w.sum() > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑pdf on a regular grid (shifted by 0.5)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    return w / w.sum() if w.sum() > 0 else _uniform(n)


def _beta_mixture(n: int, a1: float, b1: float, a2: float, b2: float, w: float) -> np.ndarray:
    """Convex combination of two Beta shapes."""
    s1 = _beta_shape(n, a1, b1)
    s2 = _beta_shape(n, a2, b2)
    mixed = w * s1 + (1.0 - w) * s2
    return mixed / mixed.sum()


# ----------------------------------------------------------------------
# Random Dirichlet sampling
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector – exponential for α=1, gamma otherwise."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Smooth Adam optimiser (log‑softmax parametrisation)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 400.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth approximation of max(conv(a,a)) using Adam on
    the log‑softmax (logits) representation.
    Returns the best exact sequence found and its exact objective.
    """
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1

        # soft‑max to get a probability vector
        a = np.exp(logits)
        a /= a.sum()

        # convolution
        conv = _conv_fft(a, a)

        # temperature schedule (linear annealing)
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp
        c = conv.max()
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()          # soft weights, sum=1

        # gradient w.r.t a (see derivation in the original code)
        conv_pa = _conv_fft(p, a[::-1])       # length = 3n‑2
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)                   # simple 1/√t schedule
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        # periodic exact evaluation
        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")

            # early exit if we are already very good
            if best_val < 1.45:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Aggressive mirror‑descent polishing (aggregated active set)
# ----------------------------------------------------------------------
def _agg_subgrad_refine(
    start_seq: np.ndarray,
    deadline: float,
    eps_rel: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    KL‑mirror descent where the sub‑gradient aggregates *all* indices whose
    convolution entry is within a relative tolerance of the current maximum.
    """
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        # indices that are “almost maximal”
        active_idx = np.where(conv >= max_val - eps_rel * max_val)[0]

        # construct indicator of active offsets
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # aggregated sub‑gradient = 2 * (active * a_rev) convolution
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # exponentiated‑gradient step (KL‑mirror)
        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Agg] it={it:7d} n={n:5d} best={val:.6f}")

            if time.time() > deadline - 0.5:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS‑B refinement (smooth‑max objective)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import minimize
except Exception:  # pragma: no cover
    minimize = None


def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 400.0) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS‑B on the smooth‑max objective (log‑sum‑exp with temperature β).
    Returns the best exact objective and the corresponding sequence.
    """
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a /= a.sum()
        conv = _conv_fft(a, a)
        max_c = conv.max()
        exps = np.exp(beta * (conv - max_c))
        lse = max_c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        # gradient w.r.t a
        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # chain‑rule through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class _TimeOut(RuntimeError):
        pass

    def _callback(xk):
        if time.time() >= deadline:
            raise _TimeOut()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method='L-BFGS-B',
            options={'maxiter': 200, 'disp': False},
            callback=_callback,
        )
    except _TimeOut:
        best_logit = x0
    else:
        best_logit = res.x

    a = np.exp(best_logit)
    a /= a.sum()
    return a, _objective(a)

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **_):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective under the supplied time budget.
    Returns the best sequence as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # --------------------------------------------------------------
    # Possibly reuse a previously stored solution
    # --------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------------
    # Lengths to explore (FFT‑friendly, not too large)
    # --------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000, 10000]

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – deterministic analytic seeds
    # --------------------------------------------------------------
    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
    )
    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val, best_seq = val, seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val, best_seq = val, previous.copy()
                print(f"[Seed] n={n:5d} type=Previous val={val:.6f}")

        # early exit if we already beat the target by a comfortable margin
        if best_val < 1.45:
            break

    # --------------------------------------------------------------
    # Phase 2 – massive random sampling (Dirichlet + random Beta)
    # --------------------------------------------------------------
    now = time.time()
    time_left = deadline - now
    # reserve roughly 30 % for the later phases
    sample_deadline = now + time_left * 0.30
    alphas = np.logspace(-1, log(5), 7, base=np.e)  # 0.1 … 5 on log‑scale
    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet draw
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Dir] n={n:5d} α={alpha:.3f} val={val:.6f}")

        # Random Beta‑pdf shape
        a_par = float(rng.uniform(0.2, 5.0))
        b_par = float(rng.uniform(0.2, 5.0))
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        # Random mixture of two Beta pdfs
        a1 = float(rng.uniform(0.2, 5.0))
        b1 = float(rng.uniform(0.2, 5.0))
        a2 = float(rng.uniform(0.2, 5.0))
        b2 = float(rng.uniform(0.2, 5.0))
        w = float(rng.uniform(0.1, 0.9))
        seq = _beta_mixture(n, a1, b1, a2, b2, w)
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[MixBeta] n={n:5d} w={w:.2f} val={val:.6f}")

        if best_val < 1.45:
            break

    # --------------------------------------------------------------
    # Phase 3 – smoothed Adam optimisation (multiple restarts)
    # --------------------------------------------------------------
    # keep a few promising seeds for Adam‐restarts
    candidates = []
    if best_seq is not None:
        candidates.append(best_seq)
        # two extra random starts with the same length as the current best
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=float(rng.choice(alphas)))
            candidates.append(cand)

    # allocate about 40 % of the remaining budget to Adam
    remaining = deadline - time.time()
    adam_deadline = time.time() + remaining * 0.40

    beta_ends = [200.0, 400.0, 600.0]  # a few increasing temperature caps
    for idx, cand in enumerate(candidates):
        # choose a β_end that roughly matches the remaining budget for this run
        bd = beta_ends[idx % len(beta_ends)]
        seq_adam, val_adam = _smooth_adam_optimize(
            start_seq=cand,
            deadline=adam_deadline,
            beta_start=5.0,
            beta_end=bd,
            max_iter=4_000_000,
            lr0=0.03,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam
            print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

        if time.time() >= adam_deadline:
            break

    # --------------------------------------------------------------
    # Phase 4 – optional L‑BFGS refinement (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0 and minimize is not None:
        lbfgs_deadline = time.time() + remaining * 0.15
        seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=400.0)
        if val_lbfgs < best_val:
            best_val, best_seq = val_lbfgs, seq_lbfgs
            print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – aggressive mirror‑descent polishing
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 1.0:
        final_seq, final_val = _agg_subgrad_refine(best_seq, deadline, eps_rel=1e-6)
        if final_val < best_val:
            best_val, best_seq = final_val, final_seq
            print(f"[Agg‑Refine] improved best to {best_val:.6f}")

    # --------------------------------------------------------------
    # Final sanity‑check (the grader will evaluate again anyway)
    # --------------------------------------------------------------
    try:
        # The evaluator clips and rescales internally; we just invoke it
        # to make sure the sequence is accepted.
        import builtins  # noqa: F401 (ensure evaluate_sequence is in globals)
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
