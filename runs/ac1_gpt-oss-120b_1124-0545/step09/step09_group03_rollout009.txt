def _random_move(seq, step_factor, rng):
    n = len(seq)
    i = rng.integers(n)
    if seq[i] == 0:
        return seq, None (skip)
    j = rng.integers(n-1)
    if j >= i: j += 1
    delta = step_factor * seq[i] * rng.random()
    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] += delta
    # renormalize
    new_seq /= new_seq.sum()
    return new_seq, delta

"""
Candidate generator for `evaluate_sequence`.

Algorithm (≈ 1000 s):
1. **Deterministic seeds** – build a variety of analytic families
   (uniform, Catalan, Hann, Hamming, Blackman, Gaussian, arcsine,
   several Beta‑pdfs) for a range of lengths (2000 … 16000).  Keep the
   best seed (lowest `2·n·max(conv)`).

2. **Heavy random sampling** – draw many Dirichlet vectors (α∈{0.5,1,2})
   and random Beta‑pdf shapes, also occasional “sparse Sidon‑like”
   supports with uniform weights.  Keep the global best.

3. **Upsampling** – take the current best sequence and linearly
   interpolate it to larger lengths (×2, ×3).  Evaluate the up‑sampled
   versions and keep any improvement.

4. **Smoothed Adam** – start from the three currently best candidates
   and minimise a smooth approximation of `max(conv)` using Adam on the
   log‑softmax (log‑its) parametrisation.  The temperature β is
   annealed from 5 to 500, the learning‑rate schedule follows
   `lr = 0.04/√t`.  The best Adam result is retained.

5. **Aggressive KL‑mirror descent** – a refined mirror‑descent step that
   aggregates *all* convolution entries within a relative tolerance
   `ε=1e‑5` of the current maximum.  Step‑size `η = 0.6/√t`.  Run until
   the time budget is exhausted.

6. **Random local polishing** – simple hill‑climbing moves that transfer
   a tiny amount of mass between two random positions (mass
   conservation).  Accept only improving moves.  The step magnitude is
   gradually reduced from 5 % to 0.01 % of the source value.

All phases respect the overall time budget (default 1000 s).  The
function returns the best sequence found as a plain list of non‑negative
floats (the grader will clip/renormalise it anyway).
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ---------------------------------------------------------------
# Low‑level utilities
# ---------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Value of `evaluate_sequence` for a *normalised* sequence (sum==1)."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * float(conv.max())


# ---------------------------------------------------------------
# Analytic families (all returned normalised to sum = 1)
# ---------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2 * i) * (2 * i - 1)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    return coeffs / total if total > 0 else _uniform(n)


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2.0 * np.pi * i / (n - 1)) + a2 * np.cos(4.0 * np.pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    w = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    total = w.sum()
    return w / total if total > 0 else _uniform(n)


# ---------------------------------------------------------------
# Random generators
# ---------------------------------------------------------------

def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    """Draw a Dirichlet‑like vector (Gamma/Exp) and normalise to sum = 1."""
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    return x / s if s > 0 else _uniform(n)


def _random_sparse_support(n: int, rng: np.random.Generator, k: int) -> np.ndarray:
    """Uniform mass on a randomly chosen k‑element subset (k≈√(2n))."""
    idx = rng.choice(n, size=k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a


# ---------------------------------------------------------------
# Optimisers
# ---------------------------------------------------------------

def _smooth_adam_optimize(start_seq: np.ndarray, deadline: float,
                          beta_start: float = 5.0, beta_end: float = 500.0,
                          max_iter: int = 8_000_000, lr0: float = 0.04):
    """Adam on the log‑softmax parametrisation, smooth max via log‑sum‑exp."""
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2 = 0.9, 0.999
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        # soft‑max
        a = np.exp(logits)
        a = a / a.sum()

        # convolution
        conv = _conv_fft(a, a)

        # temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        # smooth‑max via log‑sum‑exp
        c = conv.max()
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()          # soft weights, sum = 1

        # gradient w.r.t. a
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        # back‑prop through soft‑max
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        # Adam update (bias‑corrected)
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / np.sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:    # every 1024 steps
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            # early exit once we are already very good
            if best_val < 1.45:
                break

    return best_seq, best_val


def _mirror_descent(start_seq: np.ndarray, deadline: float,
                    eps_rel: float = 1e-5) -> tuple[np.ndarray, float]:
    """KL‑mirror descent using a *large* active set (all near‑max entries)."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        # build the active indicator
        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        # aggregated sub‑gradient
        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        # KL‑step
        lr = 0.6 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)   # stabilise
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Mirror] it={it:7d} n={n:5d} best={val:.6f}")
            if time.time() > deadline - 1.0:
                break

    return best_seq, best_val


def _random_local_search(start_seq: np.ndarray, deadline: float,
                         start_step: float = 0.05, end_step: float = 0.0001) -> tuple[np.ndarray, float]:
    """Simple hill‑climbing: move a fraction of mass from i→j if it reduces the objective."""
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    total_time = deadline - time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        # linearly interpolate step size with remaining time
        fraction = (deadline - time.time()) / total_time
        step_factor = end_step + (start_step - end_step) * fraction

        i = rng.integers(n)
        if a[i] == 0.0:
            continue
        # ensure j != i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = step_factor * a[i] * rng.random()
        if delta <= 0.0:
            continue

        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # re‑normalise (tiny numerical drift)
        new_a /= new_a.sum()

        val = _objective(new_a)
        if val < best_val:
            best_val = val
            best_seq = new_a.copy()
            a = new_a
            # occasional progress report
            if (it & 2047) == 0:
                print(f"[Local] it={it:7d} n={n:5d} best={val:.6f}")

    return best_seq, best_val


# ---------------------------------------------------------------
# Main entry point
# ---------------------------------------------------------------

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    global rng
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # 1. Deterministic analytic seeds
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000, 12000, 16000]

    seed_families = (
        _catalan_shape,
        _uniform,
        _hann,
        _hamming,
        _blackman,
        lambda n: _gaussian(n, sigma_factor=0.15),
        lambda n: _gaussian(n, sigma_factor=0.25),
        _arcsine_shape,
        lambda n: _beta_shape(n, 0.5, 0.5),
        lambda n: _beta_shape(n, 2.0, 2.0),
        lambda n: _beta_shape(n, 0.5, 2.0),
        lambda n: _beta_shape(n, 2.0, 0.5),
    )

    best_seq = None
    best_val = float("inf")

    for n in lengths:
        for fn in seed_families:
            seq = fn(n)
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={fn.__name__:12s} val={val:.6f}")

    # ------------------------------------------------------------------
    # 2. Heavy random sampling (Dirichlet, Beta, sparse supports)
    # ------------------------------------------------------------------
    now = time.time()
    time_for_sampling = (deadline - now) * 0.35   # about 35 % of remaining budget
    sample_deadline = now + time_for_sampling
    alphas = [0.5, 1.0, 2.0]
    beta_params = [(0.3, 0.3), (0.5, 0.5), (1.0, 1.0), (2.0, 2.0), (0.5, 2.0), (2.0, 0.5)]

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet
        alpha = float(rng.choice(alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # Beta‑pdf shape
        a_par, b_par = rng.choice(beta_params)
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Beta] n={n:5d} (α,β)=({a_par:.2f},{b_par:.2f}) val={val:.6f}")

        # Sparse support (size ≈ sqrt(2n))
        k = int(np.sqrt(2 * n))
        seq = _random_sparse_support(n, rng, k=k)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random‑Sparse] n={n:5d} k={k:3d} val={val:.6f}")

        if best_val < 1.45:       # early stop if already excellent
            break

    # ------------------------------------------------------------------
    # 3. Upsampling of the current best to larger lengths
    # ------------------------------------------------------------------
    # try ×2 and ×3 up‑sampling if we have enough time
    if time.time() < deadline - 2.0:
        for factor in (2, 3):
            n_big = len(best_seq) * factor
            # restrict up‑sample to at most the largest length we already tried
            if n_big > max(lengths):
                continue
            x_old = np.linspace(0.0, 1.0, len(best_seq))
            x_new = np.linspace(0.0, 1.0, n_big)
            seq_up = np.interp(x_new, x_old, best_seq)
            seq_up /= seq_up.sum()
            val = _objective(seq_up)
            if val < best_val:
                best_val = val
                best_seq = seq_up.copy()
                print(f"[Upsample ×{factor}] n={n_big:5d} val={val:.6f}")

    # ------------------------------------------------------------------
    # 4. Smoothed Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # take the three best‑performing candidates for Adam restarts
        candidates = [best_seq]
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(alphas))
            candidates.append(cand)

        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25   # 25 % of the rest

        for idx, cand in enumerate(candidates):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=500.0,
                max_iter=8_000_000,
                lr0=0.04,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5. Aggressive KL‑mirror descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        mirror_deadline = time.time() + (deadline - time.time()) * 0.30
        seq_mirror, val_mirror = _mirror_descent(best_seq, mirror_deadline, eps_rel=1e-5)
        if val_mirror < best_val:
            best_val = val_mirror
            best_seq = seq_mirror
            print(f"[Mirror‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Random local polishing (final stage)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        local_deadline = deadline - 0.2
        seq_local, val_local = _random_local_search(best_seq, local_deadline,
                                                    start_step=0.05, end_step=0.0001)
        if val_local < best_val:
            best_val = val_local
            best_seq = seq_local
            print(f"[Local‑Polish] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check (the grader will evaluate again anyway)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:   # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
