"""
Non‑negative step‑function optimiser – version 3.

Main ideas
-----------
*   Run two independent optimisation processes in parallel (2 CPUs).
    One works on length 4000 (the “classic” size) and the other on
    length 5000 – the extra degrees of freedom sometimes give a
    slightly smaller max‑convolution.
*   Enrich the local‑search phase:
    –   Adam on a log‑sum‑exp smoothed max (as before) but with a
        larger iteration budget.
    –   A *multi‑constraint* sub‑gradient / mirror‑descent that
        accounts for all convolution entries that are within a tiny
        tolerance of the current maximum (instead of only the single
        worst index).  This gives a smoother descent and helps to
        push the true maximum down further.
    –   Euclidean‑projection gradient descent (unchanged) for a
        complementary search direction.
*   Longer simulated‑annealing polishing with a slower cooling schedule.
*   Simple re‑ordering test (ascending/descending) as a final
    cheap tweak.

The programme respects the global time budget (default 1000 s) and
stores the best vector in the global name ``height_sequence_1`` for

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    The sequence must already be normalised to sum = 1.
    Returns 2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (all normalised to sum = 1)
# --------------------------------------------------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i, 0<r<1."""
    if r <= 0.0 or r >= 1.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw sorted in descending order."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_blocks(n: int, max_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """Simple block‑wise pattern with equal‑size blocks."""
    k = rng.integers(2, max_blocks + 1)
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos:pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _generate_seeds(n: int,
                    rng: np.random.Generator,
                    previous: np.ndarray | None) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # Geometric decay grid
    for r in np.linspace(0.70, 0.99, 13):
        seeds.append(_geom_decrease(n, r))

    # Power‑law grid
    for p in np.arange(0.3, 3.01, 0.3):
        seeds.append(_power_decrease(n, p))

    # Sorted Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # Block‑wise patterns
    for max_blocks in (3, 5, 7, 10):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # Random Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse patterns
    for k in (5, 10, 20, 40):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # Previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# --------------------------------------------------------------
# Optimisation primitives
# --------------------------------------------------------------

def _adam_optimize(start_vec: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 120000) -> tuple[np.ndarray | None, float]:
    """Adam optimiser on a log‑sum‑exp smoothed max (same as before)."""
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits

    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    beta = 0.5
    beta_mul = 1.001
    beta_max = 500.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                    # stays on the simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:6d} best={cur_val:.6f}")

        beta = min(beta * beta_mul, beta_max)

        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]      # central n entries

        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        max_norm = 12.0
        norm = np.linalg.norm(grad_v)
        if norm > max_norm:
            grad_v = grad_v * (max_norm / norm)

        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-12
        lr0 = 0.9
        lr = lr0 / np.sqrt(it)

        m = beta1 * m + (1.0 - beta1) * grad_v
        v2 = beta2 * v2 + (1.0 - beta2) * (grad_v ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v2_hat = v2 / (1.0 - beta2 ** it)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + eps_adam)

    return best_seq, best_val


def _subgradient_multi_refine(start_seq: np.ndarray,
                              n: int,
                              deadline: float,
                              max_iters: int = 300000,
                              eps_active: float = 1e-5,
                              lr0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true max‑convolution
    but using *all* indices whose value is within eps_active of the
    current maximum.  This yields a smoother descent direction.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = np.max(conv)
        cur_val = 2.0 * n * max_val

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad‑Multi] it={it:7d} best={cur_val:.6f}")

        # active set: all indices within eps_active of the peak
        thresh = max_val - eps_active * max_val
        active_idxs = np.where(conv >= thresh)[0]
        # limit to a handful for speed
        if active_idxs.size > 7:
            # keep the 7 largest entries only
            top = np.argpartition(conv, -7)[-7:]
            active_idxs = top

        # accumulate gradient over the active set
        g = np.zeros_like(a)
        for idx in active_idxs:
            i_start = max(0, idx - (n - 1))
            i_end = min(n - 1, idx)
            idx_i = np.arange(i_start, i_end + 1, dtype=int)
            rev_idx = idx - idx_i
            g[idx_i] += 2.0 * a[rev_idx]

        # average if several active constraints
        g = g / max(1, active_idxs.size)

        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

    return best_seq, best_val


def _euc_proj_subgradient(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """Gradient descent on the true max‑convolution with Euclidean projection."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 50000 == 0:
                print(f"[EucProj] it={it:7d} best={cur_val:.6f}")

        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.6 / np.sqrt(it)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 600000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simulated annealing with a slower cooling schedule."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling, a bit slower than before
        T = max(T * 0.9998, min_temp)

        # pick a small subset (3‑9 indices) and redistribute their total mass
        k = rng.integers(3, 10)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# --------------------------------------------------------------
# Core optimisation runner (executed in a separate process)
# --------------------------------------------------------------

def _optimisation_worker(seed: int,
                         n: int,
                         deadline: float,
                         out_queue: mp.Queue):
    """
    Run the full optimisation pipeline for a given length `n` and
    random seed.  The best sequence (as a plain list) and its surrogate
    objective are put on `out_queue`.
    """
    print(f"\n--- Worker {seed} (n={n}) starting ---")
    rng = np.random.default_rng(seed)

    # use any previously stored solution if it matches the length
    previous = None
    global_prev = globals().get("height_sequence_1")
    if isinstance(global_prev, list):
        try:
            arr = np.asarray(global_prev, dtype=np.float64)
            if arr.shape == (n,):
                previous = arr
        except Exception:
            pass

    # --------------------------------------------------------
    # Stage 1 – seed generation
    # --------------------------------------------------------
    seeds = _generate_seeds(n, rng, previous)
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = min(6, len(evaluated))
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Worker {seed}: generated {len(seeds)} seeds, top {top_k} selected.")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"Worker {seed}: initial best objective = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Stage 2 – heavy optimisation of the top seeds
    # --------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\nWorker {seed}: optimisation run {idx+1}/{len(best_seeds)}")

        # Adam on smoothed max
        seq_adam, val_adam = _adam_optimize(seed_vec, n, deadline,
                                            max_iters=120000)
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Worker {seed}] new global best after Adam = {val_adam:.6f}")

        # Multi‑constraint sub‑gradient refinement
        seq_sub, val_sub = _subgradient_multi_refine(global_best_seq, n,
                                                    deadline,
                                                    max_iters=300000)
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[Worker {seed}] new global best after SubGrad‑Multi = {val_sub:.6f}")

        # Euclidean‑projection gradient descent (as a complementary step)
        seq_euc, val_euc = _euc_proj_subgradient(global_best_seq, n,
                                                  deadline,
                                                  max_iters=200000)
        if val_euc < global_best_val:
            global_best_seq = seq_euc.copy()
            global_best_val = val_euc
            print(f"[Worker {seed}] new global best after EucProj = {val_euc:.6f}")

    # --------------------------------------------------------
    # Stage 3 – simulated‑annealing polishing
    # --------------------------------------------------------
    if time.time() < deadline:
        seq_sa, val_sa = _simulated_annealing(global_best_seq, n,
                                              deadline,
                                              max_iters=600000,
                                              init_temp_factor=0.07)
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[Worker {seed}] new global best after SA = {val_sa:.6f}")

    # --------------------------------------------------------
    # Stage 4 – optional re‑ordering test
    # --------------------------------------------------------
    if time.time() < deadline:
        for ordering in ("desc", "asc"):
            cand = np.sort(global_best_seq)[::-1] if ordering == "desc" else np.sort(global_best_seq)
            cand_val = _objective(cand)
            if cand_val < global_best_val:
                global_best_seq = cand.copy()
                global_best_val = cand_val
                print(f"[Worker {seed}] new global best after {ordering} reorder = {cand_val:.6f}")

    # final sanity check (calls the provided evaluate_sequence if available)
    try:
        evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print(f"[Worker {seed}] warning – evaluate_sequence raised: {exc}")

    print(f"\n[Worker {seed}] finished. Best surrogate ≈ {global_best_val:.6f}")
    out_queue.put((global_best_seq.tolist(), float(global_best_val)))


# --------------------------------------------------------------
# Public entry point
# --------------------------------------------------------------

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal `evaluate_sequence`
    value.  Two parallel optimisation workers (length 4000 and 5000)
    are launched; the best result among them is returned.
    """
    # reproducibility – the two workers use slightly different seeds
    np.random.seed(seed)

    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------
    # Set up parallel workers (max 2 CPUs)
    # --------------------------------------------------------
    manager = mp.Manager()
    result_queue = manager.Queue()

    # Worker 0: classic length 4000
    p0 = mp.Process(target=_optimisation_worker,
                    args=(seed, 4000, deadline, result_queue))
    # Worker 1: a bit larger length 5000
    p1 = mp.Process(target=_optimisation_worker,
                    args=(seed + 1, 5000, deadline, result_queue))

    p0.start()
    p1.start()

    # Wait until the global deadline approaches (a tiny buffer)
    while time.time() < deadline:
        time.sleep(0.5)
        # if both processes have finished we can break early
        if not p0.is_alive() and not p1.is_alive():
            break

    # Ensure termination
    for p in (p0, p1):
        if p.is_alive():
            p.terminate()
        p.join()

    # --------------------------------------------------------
    # Collect results
    # --------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    while not result_queue.empty():
        seq, val = result_queue.get()
        if val < best_val:
            best_val = val
            best_seq = seq

    if best_seq is None:
        # Fallback: return a simple uniform vector (should never happen)
        print("Warning: no result returned from workers – falling back to uniform.")
        best_seq = _uniform(4000).tolist()
        best_val = _objective(np.asarray(best_seq, dtype=np.float64))

    print(f"\n=== Overall finished ===\nBest surrogate objective ≈ {best_val:.6f} "
          f"(length {len(best_seq)})")

    # store for possible reuse
    globals()["height_sequence_1"] = best_seq
    return best_seq
