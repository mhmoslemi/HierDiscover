"""
Propose a step‑function that reduces 2·n·max(conv(seq,seq))/(sum(seq)²).

Algorithm
---------

1. **Rich seed pool** – deterministic families (uniform, triangular,
   sine‑window, Gaussian, power‑law, beta, exponential‑decay, Tukey)
   together with random Dirichlet / low‑frequency curves.
   All candidates are ℓ₁‑normalised (sum = 1).

2. **Log‑space Adam** – optimisation on the logits `v = log(a)` of the
   probability vector `a`.  The objective is approximated by a
   temperature‑smoothed max (log‑sum‑exp) with a temperature that is
   increased gradually.  Adam’s adaptive steps together with a simple
   symmetrisation (`a ← (a+rev(a))/2`) accelerate convergence.

3. **Exact sub‑gradient descent** – after the smooth phase we switch to
   the true max‑convolution objective.  The sub‑gradient is non‑zero
   only on the active lag.  We use an additive update with a
   decreasing step size and a cheap simplex projection
   (`max(v,0); v/=v.sum()`).  After each projection the vector is kept
   symmetric.

4. **Stochastic polishing** – a final simulated‑annealing loop that
   redistributes the mass of a random small subset of coordinates.
   Accepted moves are governed by a geometric temperature schedule.

All stages respect a global deadline (default 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` and
also returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1."""
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _simplex_proj(v: np.ndarray) -> np.ndarray:
    """
    Fast projection onto the probability simplex:
    clip to non‑negative then renormalise.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        return v / s
    # fallback – uniform
    return np.full_like(v, 1.0 / v.shape[0])


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Make the vector symmetric w.r.t. reversal (preserves sum)."""
    v = (v + v[::-1]) * 0.5
    return v


# ----------------------------------------------------------------------
# Analytic shape families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    if lam <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-lam * np.abs(i - centre))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


# ----------------------------------------------------------------------
# Random families
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Smooth non‑negative curve built from a few random cosine components."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection (enhanced)
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised candidates."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.15))
    seeds.append(_gaussian(n, sigma_frac=0.25))
    seeds.append(_gaussian(n, sigma_frac=0.35))

    # Power‑law family (extended range)
    for p in np.linspace(0.0, 8.0, 17):
        seeds.append(_power(n, p))

    # Beta family (dense grid, covering edge‑heavy shapes)
    alphas = np.geomspace(0.05, 4.0, 12)
    betas = np.geomspace(0.05, 4.0, 12)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.05, 0.15, 0.25, 0.4, 0.6, 0.8, 0.95):
        seeds.append(_tukey_window(n, alpha))

    # Exponential decays (different lengths)
    for lam in (0.001, 0.002, 0.005, 0.01, 0.02):
        seeds.append(_exponential_decay(n, lam))

    # Random Dirichlet draws (varied concentrations)
    for conc in (0.02, 0.05, 0.1, 0.3, 0.7):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(4):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 5):
        for _ in range(2):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

# Optimisation primitives

def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on logits with a temperature‑smoothed max (log‑sum‑exp)
    surrogate.  Returns the best probability vector seen and its surrogate
    value (2·n·max(conv)).
    """
    eps = 1e-30
    v = np.log(start_vec + eps)  # logits
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)        # second‑moment accumulator

    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # temperature schedule for the smooth max
    temp = 5.0
    temp_inc = 0.8
    temp_max = 500.0

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # probability vector (sum = 1)
        # optional symmetrisation to keep the search space small
        a = _symmetrize(a)

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogAdam] it={it:6d} best={cur_val:.6f}")

        # ----- smooth max via log‑sum‑exp -----
        shift = np.max(conv)
        w = np.exp(temp * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)          # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # convert to gradient w.r.t. logits v
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v2 = beta2 * v2 + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1 ** it)
        v2_hat = v2 / (1 - beta2 ** it)

        lr = 0.6 / np.sqrt(it)   # mild decay
        v -= lr * m_hat / (np.sqrt(v2_hat) + adam_eps)

        # temperature schedule (slowly sharpen)
        if temp < temp_max:
            temp = min(temp + temp_inc, temp_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a simple Euclidean‑like simplex projection (clip‑and‑renorm) and
    optional symmetrisation.
    """
    a = start_seq.copy()
    a = _symmetrize(a)
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size decays with √t (tuned a bit larger than before)
        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _simplex_proj(a)          # fast clip‑and‑renorm
        a = _symmetrize(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-14,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random small subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9993, min_temp)

        # pick a random subset of coordinates (size 3 … 12)
        k = rng.integers(3, 13)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        # redistribute uniformly on the simplex of size k
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        # keep symmetry – copy the changes to the mirrored positions
        if n % 2 == 0:
            # even length: mirror by reversal
            mirror = n - 1 - idx
            new[mirror] = new_alloc
        else:
            # odd length: central element maps to itself
            mirror = n - 1 - idx
            new[mirror] = new_alloc

        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 40000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    # ------------------------------------------------------------------
    # reproducibility
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # ------------------------------------------------------------------
    # retrieve previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Length – 4000 offers a good trade‑off between quality and speed.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and quick pre‑selection.
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    # evaluate surrogate for all seeds
    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seeds (surrogate values):")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    # initialise global best
    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – optimisation loops (log‑space Adam + exact sub‑gradient)
    # ------------------------------------------------------------------
    for run_idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {run_idx+1}/{len(best_seeds)} ---")

        # 2.1 Log‑space Adam
        seq_adam, val_adam = _logspace_adam_optimize(
            seed_vec, n, deadline, max_iters=300_000
        )
        if seq_adam is not None and val_adam < global_best_val:
            global_best_seq = seq_adam.copy()
            global_best_val = val_adam
            print(f"[Overall‑Adam] new global best = {val_adam:.6f}")

        # 2.2 Exact sub‑gradient descent (starting from the latest best)
        seq_sub, val_sub = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=2_500_000
        )
        if val_sub < global_best_val:
            global_best_seq = seq_sub.copy()
            global_best_val = val_sub
            print(f"[Overall‑Sub] new global best = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=1_200_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[Overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator (optional)
    # ------------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
