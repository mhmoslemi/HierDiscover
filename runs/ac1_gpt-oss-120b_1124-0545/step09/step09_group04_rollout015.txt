"""
search_step_function.py

Search for a non‑negative step‑function (a list of floats) that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq)) / (sum(seq)**2)

The code follows three stages

1️⃣  Generate a very diverse set of “seed’’ vectors from analytic families
    (uniform, triangular, sine‑window, geometric, power‑law, beta, …)
    and from a dense grid of a few key parameters (p, r, α, β).

2️⃣  Keep the K best seeds and run a long exponentiated‑gradient (mirror‑descent)
    sub‑gradient optimisation on each of them.  The sub‑gradient of the
    max‑convolution is cheap to compute (it is just 2·a shifted by the
    index attaining the maximum).  An AdaGrad‑style learning‑rate
    (η₀/√t) is used and the best vector ever seen is retained.

3️⃣  Finish with a relatively cheap simulated‑annealing phase that randomly
    reshuffles a tiny subset of masses; this often yields a modest final
    polish.

All operations stay on the probability simplex (the vector is normalised
to sum = 1 after every update) so the scale‑invariance of the objective is
exploited automatically.  The routine respects the 1000 s time‑budget,
stores the best vector in the global name ``height_sequence_1`` for
possible reuse in a later run, and finally returns the vector as a plain
list of floats.
"""

import numpy as np

# Low‑level utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using real FFT (fast)."""
    L = x.size + y.size - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    ``seq`` is assumed to be non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = seq.size
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)

# Analytic shape families (all are normalised to sum = 1)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _subgradient_refine(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 5_000_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent) on the exact max‑convolution.
    Returns the best sequence ever seen and its surrogate objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30
    it = 0

    while it < max_iters and time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        # index of the (first) maximal convolution entry
        max_idx = int(np.argmax(conv))

        # sub‑gradient of the max at index max_idx
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)                # normalise in log‑space
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s

        # keep the best ever seen
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            if it % 200_000 == 0:
                print(f"[SubGrad] iter={it:9d}  best≈{best_val:.6f}")

    return best_seq, best_val
