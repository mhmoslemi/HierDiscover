def _interleave_desc(vec):
    # descending sort
    s = np.sort(vec)[::-1]
    out = np.empty_like(s)
    l, r = 0, len(s)-1
    i = 0
    while l <= r:
        if i % 2 == 0:
            out[i] = s[l]; l+=1
        else:
            out[i] = s[r]; r-=1
        i+=1

"""
Improved step‑function search for the evaluation metric

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The optimiser works in three stages:

1. **Rich seed pool** – deterministic families (uniform, triangular,
   Gaussian, power‑law, beta, Tukey), many random Dirichlet draws with a
   broad range of concentrations, low‑frequency cosine curves and a
   *phase‑random Fourier* construction.  For each seed we also add a
   “interleaved” ordering (large numbers far apart) which is known to
   reduce the self‑convolution peak.

2. **Log‑space SGD with momentum** – optimisation is performed on the
   logits of a soft‑max (so the simplex constraint is satisfied automatically).
   The true max‑convolution is approximated by a temperature‑controlled
   log‑sum‑exp, the temperature is increased slowly, and a momentum term
   (β=0.9) accelerates convergence.  The best vector seen is recorded.

3. **Exact sub‑gradient descent + long‑run simulated annealing** – after the
   smooth phase we switch to the exact non‑smoothed objective.
   A classic sub‑gradient step with Euclidean projection onto the simplex
   is run for as long as the remaining budget permits.
   Finally a very long annealing loop reshuffles the mass of a random
   subset of coordinates (size up to 30) with a geometric cooling schedule.
   The best sequence found over all stages is returned and stored in the
   global name ``height_sequence_1`` for reuse.

All stages respect the global deadline (default 1000 s).  Progress and
intermediate objective values are printed to the console.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    (If Σseq ≠ 1 the returned value is scaled accordingly, which does not
    affect ordering of candidates.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero -> uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (already ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi *
                                    (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _random_fourier(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random phase on a flat spectrum (inverse FFT).  The resulting real
    signal is shifted to be non‑negative and ℓ₁‑normalised.
    """
    # phases for the real FFT (n//2+1 coefficients)
    phases = rng.uniform(0.0, 2.0 * np.pi, size=n // 2 + 1)
    spectrum = np.exp(1j * phases)
    sig = np.fft.irfft(spectrum, n=n)
    sig -= sig.min()          # make non‑negative
    s = sig.sum()
    if s <= 0.0:
        return _uniform(n)
    return sig / s


def _interleave_desc(v: np.ndarray) -> np.ndarray:
    """Place the largest values far apart: descending → interleaved."""
    s = np.sort(v)[::-1]                     # descending
    out = np.empty_like(s)
    l, r = 0, len(s) - 1
    i = 0
    while l <= r:
        if i % 2 == 0:
            out[i] = s[l]
            l += 1
        else:
            out[i] = s[r]
            r -= 1
        i += 1
    out_sum = out.sum()
    if out_sum == 0.0:
        return _uniform(len(v))
    return out / out_sum


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection
# ----------------------------------------------------------------------
def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # Power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # Beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – many concentrations and many repeats
    concs = (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0)
    for conc in concs:
        for _ in range(6):          # six draws per concentration
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential draws (heavy‑tailed)
    for _ in range(6):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Random Fourier seeds (flat spectrum with random phase)
    for _ in range(6):
        seeds.append(_random_fourier(n, rng))

    # Interleaved versions of a subset of the above (large values far apart)
    for base in seeds[:30]:          # only the first 30 to keep size reasonable
        seeds.append(_interleave_desc(base))

    # Mixtures of uniform with low‑frequency curves
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds

# Optimisation primitives

def _logspace_sgd_momentum_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
) -> tuple[np.ndarray | None, float]:
    """
    SGD on logits (soft‑max) using a log‑sum‑exp smooth approximation of the max.
    Momentum (β=0.9) and a slow temperature ramp improve convergence.
    Returns the best vector seen and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                     # logits
    m = np.zeros_like(v)                           # momentum term

    beta = 0.5               # initial smoothing temperature
    beta_inc = 0.7
    beta_max = 300.0

    lr0 = 0.9                # initial learning‑rate (log‑space)

    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                     # current simplex point

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 5000 == 0:
                print(f"[LogSGD‑M] it={it:7d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of Σ w_k·conv_k w.r.t. a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]       # central n entries

        # convert to log‑space gradient
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # momentum update
        m = 0.9 * m + 0.1 * grad_v

        # learning‑rate schedule (1/√t)
        lr = lr0 / np.sqrt(it)

        v = v - lr * m

        # increase smoothing temperature slowly
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses additive updates with Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 20000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient at the active index (linear in a)
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size – slowly decreasing, with a slight boost when far from optimum
        lr = 0.2 / np.sqrt(it) * (cur_val / best_val)  # adaptive factor
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing_improved(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
    init_temp_factor: float = 0.04,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Extended SA that uses two move types:
      * redistributing the mass of a random subset (size 2‑30)
      * swapping two coordinates.
    Cooling is slower (factor 0.9998) to allow thorough exploration.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9998, min_temp)

        if rng.random() < 0.7:        # majority of moves = redistribution
            k = rng.integers(2, 31)   # subset size 2 … 30
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            # Dirichlet draw (concentration 1) – uniform over the simplex
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                         # occasional swap
            i, j = rng.choice(n, size=2, replace=False)
            new = cur.copy()
            new[i], new[j] = new[j], new[i]

        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA‑imp] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluator.
    Returns the best sequence found as a plain list of floats.
    """
    # reproducibility
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # retrieve previous best if present
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # --------------------------------------------------------
    # Choose a length – 4000 is a good balance.
    # --------------------------------------------------------
    n = 4000

    # --------------------------------------------------------
    # Stage 1 – generate a rich pool of seeds and keep the most promising.
    # --------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 7                     # keep a few elite seeds
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # --------------------------------------------------------
    # Stage 2 – log‑space SGD with momentum (multiple restarts).
    # --------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Log‑space SGD‑M run {idx+1}/{len(best_seeds)} ---")
        seq_sgd, val_sgd = _logspace_sgd_momentum_optimize(
            seed_vec, n, deadline, max_iters=1_200_000
        )
        if seq_sgd is not None and val_sgd < global_best_val:
            global_best_seq = seq_sgd.copy()
            global_best_val = val_sgd
            print(f"[overall‑SGD‑M] new global best = {val_sgd:.6f}")

        # --------------------------------------------------------
        # Sub‑gradient projection phase (exact max).
        # --------------------------------------------------------
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # --------------------------------------------------------
    # Stage 3 – final stochastic polishing (improved simulated annealing).
    # --------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing (improved) ---")
        seq_sa, val_sa = _simulated_annealing_improved(
            global_best_seq, n, deadline, max_iters=2_500_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA‑imp] final global best = {val_sa:.6f}")

    # --------------------------------------------------------
    # Final sanity check with the official evaluator (optional).
    # --------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
