"""
Propose a non‑negative step‑function that (approximately) minimises
``evaluate_sequence``.  The algorithm is a multi‑phase search:

1. **Rich analytic seeds** – a battery of parametrised families
   (uniform, window functions, Gaussian, Beta, exponential‑/polynomial‑
   decay, inverse‑power, Catalan, arcsine).  All are normalised to sum = 1
   and evaluated; the best is kept.

2. **Heavy random sampling** – Dirichlet draws with a wide range of
   α‑values and random Beta‑pdf shapes (α,β drawn from a log‑uniform
   range).  The best sample is recorded.

3. **Smoothed Adam optimisation** – a few restarts from the current
   elite candidates, minimising a log‑sum‑exp smooth approximation of the
   max‑convolution (the same machinery used in the reference solution).

4. **Optional L‑BFGS‑B refinement** – if SciPy is available, a final
   smooth‑max optimisation is performed.

5. **Aggressive KL‑mirror descent** – a sub‑gradient method that
   aggregates *all* near‑maximal convolution entries.

6. **Permutation hill‑climbing** – random swaps of two positions
   (preserving the sum) are accepted only when they lower the exact
   objective (`2·n·max(conv)`).  This cheap local search often pushes the
   solution a few thousandths further.

The total running time never exceeds ``budget_s`` seconds (default 1000 s).
The returned list is a plain Python list of floats; all numbers are
non‑negative.  The global variable ``height_sequence_1`` (if present) is
used as a warm start.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft
from math import sqrt, log, pi, exp

# optional SciPy import -------------------------------------------------
try:
    from scipy.optimize import minimize
except Exception:  # pragma: no cover
    minimize = None

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast for large vectors)."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = rfft(x, n=N)
    fy = rfft(y, n=N)
    conv = irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Exact objective for a *normalised* sequence (sum == 1):
    2·n·max(conv(seq,seq)).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic families (all return a normalised numpy array)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _catalan_shape(n: int) -> np.ndarray:
    coeffs = np.empty(n, dtype=np.float64)
    c = 1.0
    coeffs[0] = c
    for i in range(1, n):
        c = c * ((2.0 * i) * (2.0 * i - 1.0)) / (4.0 * i * i)
        coeffs[i] = c
    total = coeffs.sum()
    if total == 0.0:
        return _uniform(n)
    return coeffs / total


def _hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 * (1.0 - np.cos(2.0 * pi * i / (n - 1)))
    return w / w.sum()


def _hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * pi * i / (n - 1))
    return w / w.sum()


def _blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    w = a0 - a1 * np.cos(2.0 * pi * i / (n - 1)) + a2 * np.cos(4.0 * pi * i / (n - 1))
    return w / w.sum()


def _gaussian(n: int, sigma_factor: float = 0.2) -> np.ndarray:
    sigma = sigma_factor * n
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    w = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _arcsine_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 0.5
    w = 1.0 / np.sqrt(i * (n + 0.5 - i))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    w = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _exp_decay(n: int, rate: float = 0.9) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = rate ** i
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _poly_decay(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (n - i) ** power
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


def _inv_power(n: int, power: float = 1.0) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    w = i ** (-power)
    if w.sum() == 0.0:
        return _uniform(n)
    return w / w.sum()


# ----------------------------------------------------------------------
# Random drawing utilities
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, rng: np.random.Generator, alpha: float = 1.0) -> np.ndarray:
    if alpha == 1.0:
        x = rng.exponential(1.0, size=n)
    else:
        x = rng.gamma(alpha, 1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return _uniform(n)
    return x / s


# ----------------------------------------------------------------------
# Smoothed Adam optimiser (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_adam_optimize(
    start_seq: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 200.0,
    max_iter: int = 5_000_000,
    lr0: float = 0.03,
) -> tuple[np.ndarray, float]:
    n = len(start_seq)
    eps = 1e-30
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        a = np.exp(logits)
        a /= a.sum()

        conv = _conv_fft(a, a)

        # linear temperature schedule
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iter * 0.6))

        c = np.max(conv)
        exp_beta = np.exp(beta * (conv - c))
        p = exp_beta / exp_beta.sum()

        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]

        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)

        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / sqrt(it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Adam] it={it:7d} n={n:5d} val={val:.6f} β={beta:.1f}")
            if best_val < 1.45:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Optional L‑BFGS‑B refinement (smooth max)
# ----------------------------------------------------------------------
def _lbfgs_refine(start_seq: np.ndarray, deadline: float, beta: float = 200.0) -> tuple[np.ndarray, float]:
    if minimize is None:
        return start_seq, _objective(start_seq)

    n = len(start_seq)
    eps = 1e-30

    def fun_and_grad(logits: np.ndarray):
        a = np.exp(logits)
        a /= a.sum()
        conv = _conv_fft(a, a)
        max_c = np.max(conv)
        exps = np.exp(beta * (conv - max_c))
        lse = max_c + (1.0 / beta) * np.log(exps.sum())
        value = 2.0 * n * lse

        p = exps / exps.sum()
        conv_pa = _conv_fft(p, a[::-1])
        g_a = 2.0 * conv_pa[n - 1:n - 1 + n]
        dot = np.dot(g_a, a)
        grad_logits = a * (g_a - dot)
        return value, grad_logits.astype(np.float64)

    x0 = np.log(start_seq + eps)

    class TimeOutException(RuntimeError):
        pass

    def callback(xk):
        if time.time() >= deadline:
            raise TimeOutException()

    try:
        res = minimize(
            fun=lambda x: fun_and_grad(x)[0],
            x0=x0,
            jac=lambda x: fun_and_grad(x)[1],
            method="L-BFGS-B",
            options={"maxiter": 200, "disp": False},
            callback=callback,
        )
    except TimeOutException:
        best_logits = x0
    else:
        best_logits = res.x

    a = np.exp(best_logits)
    a /= a.sum()
    return a, _objective(a)


# ----------------------------------------------------------------------
# Aggressive KL‑mirror descent (aggregated sub‑gradient)
# ----------------------------------------------------------------------
def _refine_aggregated_subgradient(start_seq: np.ndarray, deadline: float, eps_rel: float = 1e-6) -> tuple[np.ndarray, float]:
    a = start_seq.copy()
    n = len(a)
    best_seq = a.copy()
    best_val = _objective(a)
    it = 0
    eps = 1e-30

    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = float(conv.max())
        delta = eps_rel * max_val
        active_idx = np.where(conv >= max_val - delta)[0]

        active = np.zeros_like(conv)
        active[active_idx] = 1.0

        conv_ag = _conv_fft(active, a[::-1])
        g = 2.0 * conv_ag[n - 1:n - 1 + n]

        lr = 0.7 / sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        if (it & 1023) == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[AggRefine] it={it:7d} n={n:5d} best={val:.6f}")
            if time.time() > deadline - 1.0:
                break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Local permutation hill‑climbing
# ----------------------------------------------------------------------
def _permute_hill_climb(start_seq: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Random swaps that are kept only when they strictly improve the exact objective."""
    a = start_seq.copy()
    n = len(a)
    best_val = _objective(a)
    it = 0
    while time.time() < deadline:
        it += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        a_swapped = a.copy()
        a_swapped[i], a_swapped[j] = a_swapped[j], a_swapped[i]
        val = _objective(a_swapped)
        if val < best_val:
            best_val = val
            a = a_swapped
            print(f"[Swap] it={it:7d} n={n:5d} val={val:.6f}")
    return a, best_val


# ----------------------------------------------------------------------
# Generate a rich collection of seed candidates for a given length
# ----------------------------------------------------------------------
def _seed_candidates_for_n(n: int):
    """Yield (seq, description) pairs for analytic families."""
    # uniform and simple windows
    yield _uniform(n), "uniform"
    yield _catalan_shape(n), "catalan"
    yield _hann(n), "hann"
    yield _hamming(n), "hamming"
    yield _blackman(n), "blackman"
    yield _arcsine_shape(n), "arcsine"

    # Gaussians with a few sigma factors
    for sf in (0.12, 0.15, 0.20, 0.25):
        yield _gaussian(n, sigma_factor=sf), f"gaussian_sf={sf}"

    # Beta distributions – a modest grid
    beta_params = [
        (0.3, 0.3),
        (0.5, 0.5),
        (1.0, 1.0),
        (2.0, 2.0),
        (0.5, 2.0),
        (2.0, 0.5),
        (0.2, 0.2),
        (3.0, 3.0),
    ]
    for a_par, b_par in beta_params:
        yield _beta_shape(n, a_par, b_par), f"beta_{a_par}_{b_par}"

    # Exponential decays
    for rate in (0.8, 0.85, 0.9, 0.92, 0.95, 0.97):
        yield _exp_decay(n, rate), f"exp_{rate}"

    # Polynomial decays
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        yield _poly_decay(n, p), f"poly_{p}"

    # Inverse‑power
    for p in (0.5, 1.0, 1.5, 2.0):
        yield _inv_power(n, p), f"invpow_{p}"

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored solution, if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, (list, tuple, np.ndarray)):
        try:
            previous = np.asarray(previous, dtype=np.float64)
            if previous.size > 0 and np.all(previous >= 0):
                previous = previous / previous.sum()
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Lengths to explore – we allow a few larger values
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000, 5000, 6000, 8000, 12000]

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – analytic seeds (rich family)
    # ------------------------------------------------------------------
    for n in lengths:
        for seq, label in _seed_candidates_for_n(n):
            val = _objective(seq)
            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                print(f"[Seed] n={n:5d} type={label:12s} val={val:.6f}")

        # incorporate previous solution if same length
        if previous is not None and previous.shape == (n,):
            val = _objective(previous)
            if val < best_val:
                best_val = val
                best_seq = previous.copy()
                print(f"[Seed] n={n:5d} type=previous val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – massive random sampling (Dirichlet + random Beta)
    # ------------------------------------------------------------------
    now = time.time()
    # allocate ~35 % of remaining budget for sampling
    time_for_sampling = (deadline - now) * 0.35
    sample_deadline = now + time_for_sampling

    dirichlet_alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    # log‑uniform Beta parameters (range 0.1 … 5)
    beta_log_min, beta_log_max = -1.0, log(5.0)

    while time.time() < sample_deadline:
        n = int(rng.choice(lengths))
        # Dirichlet draw
        alpha = float(rng.choice(dirichlet_alphas))
        seq = _random_dirichlet(n, rng, alpha=alpha)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random-Dir] n={n:5d} α={alpha:.2f} val={val:.6f}")

        # random Beta shape with log‑uniform parameters
        a_par = exp(rng.uniform(beta_log_min, beta_log_max))
        b_par = exp(rng.uniform(beta_log_min, beta_log_max))
        seq = _beta_shape(n, a_par, b_par)
        val = _objective(seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Random-Beta] n={n:5d} (α,β)=({a_par:.3f},{b_par:.3f}) val={val:.6f}")

        if best_val < 1.45:
            break

    # ------------------------------------------------------------------
    # Phase 3 – Smoothed Adam optimisation (few restarts)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # keep a few elite candidates for Adam restarts
        elite = [best_seq]
        # generate a couple more random Dirichlet starts of the same length
        top_len = len(best_seq)
        for _ in range(2):
            cand = _random_dirichlet(top_len, rng, alpha=rng.choice(dirichlet_alphas))
            elite.append(cand)

        # allocate ~25 % of remaining budget to Adam runs
        remaining = deadline - time.time()
        adam_deadline = time.time() + remaining * 0.25

        for idx, cand in enumerate(elite):
            seq_adam, val_adam = _smooth_adam_optimize(
                start_seq=cand,
                deadline=adam_deadline,
                beta_start=5.0,
                beta_end=200.0,
                max_iter=4_000_000,
                lr0=0.03,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = seq_adam
                print(f"[Adam‑Refine {idx}] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – optional L‑BFGS refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0 and minimize is not None:
            lbfgs_deadline = time.time() + remaining * 0.20
            seq_lbfgs, val_lbfgs = _lbfgs_refine(best_seq, lbfgs_deadline, beta=200.0)
            if val_lbfgs < best_val:
                best_val = val_lbfgs
                best_seq = seq_lbfgs
                print(f"[L‑BFGS] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – aggressive KL‑mirror descent polishing
    # ------------------------------------------------------------------
    if best_seq is not None:
        final_seq, final_val = _refine_aggregated_subgradient(best_seq, deadline)
        if final_val < best_val:
            best_val = final_val
            best_seq = final_seq
            print(f"[Agg‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 6 – permutation hill‑climbing (local swap search)
    # ------------------------------------------------------------------
    if best_seq is not None:
        remaining = deadline - time.time()
        if remaining > 5.0:
            swap_deadline = time.time() + remaining
            seq_swapped, val_swapped = _permute_hill_climb(best_seq, swap_deadline, rng)
            if val_swapped < best_val:
                best_val = val_swapped
                best_seq = seq_swapped
                print(f"[Swap‑Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (the grader will call evaluate_sequence again)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_seq.tolist())
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})"
    )
    return best_seq.tolist()
