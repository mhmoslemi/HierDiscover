def _project_simplex(v):
    # v is 1D np.array
    n = v.shape[0]
    u = np.sort(v)[::-1]   # descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _smooth_gradient_descent(start_seq, deadline):
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * conv.max()
    best_iter=0

    it = 0
    alpha = 20.0
    lr0 = 0.5
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a) # length 2n-1
        max_c = conv.max()
        # compute softmax
        shift = max_c
        exp_arg = alpha * (conv - shift)
        # clip for numerical stability
        exp_arg = np.clip(exp_arg, -700, 700)
        exp_vals = np.exp(exp_arg)
        Z = exp_vals.sum()
        if Z == 0:
            w = np.full_like(exp_vals, 1.0/exp_vals.size)
        else:
            w = exp_vals / Z

        # gradient
        rev_a = a[::-1]
        conv_rev = _conv_fft(rev_a, w)
        g = 2.0 * conv_rev[n-1:n-1+n]

        # step size
        lr = lr0 / np.sqrt(it)
        a_new = a - lr * g
        # project onto simplex
        a_new = _project_simplex(a_new)

        a = a_new

        # evaluate true objective
        conv_new = _conv_fft(a, a)
        cur_max = conv_new.max()
        cur_val = 2.0 * n * cur_max

        if cur_val < best_val:

            best_seq = a.copy()
            best_iter = it
            print(f"[SmoothGrad] it={it:8d} best={cur_val:.6f}")

        # increase alpha slowly
        if it % 500 == 0:
            alpha = min(alpha * 1.05, 500.0)
    return best_seq, best_val

def _project_simplex(v):
    n = v.shape[0]
    if np.all(v <= 0):

    u = np.sort(v)[::-1]

    if rho.size == 0:

"""
Propose a low‑objective step‑function.

The algorithm consists of four phases that respect a 1000 s budget:

1. **Initialisation** – load a previous best or generate a few cheap seeds
   (uniform, random soft‑max, Sidon‑like support) and keep the best.

2. **Smooth‑gradient descent** – minimise a smooth approximation of the
   max‑convolution using a soft‑max (temperature α) and Euclidean projection
   onto the probability simplex.  α is increased gradually so that the
   optimisation starts global and finishes focussed on the true max.

3. **Exact sub‑gradient (exponentiated‑gradient) refinement** – the classic
   non‑smooth mirror‑descent from the previous state‑of‑the‑art code, used for
   a short remaining interval.

4. **Optional LP linearisation** (if cvxpy is present) and a final
   simulated‑annealing fine‑tuner.

All sequences are kept non‑negative and normalised to sum = 1, so the returned
objective equals ``2·n·max(conv(a,a))``.  The best sequence found is stored in
the global ``height_sequence_1`` for the next call.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional CVXPY import (used only for the LP refinement)
# ----------------------------------------------------------------------
try:
    import cvxpy as cp
    CVXPY_AVAILABLE = True
except Exception:  # pragma: no cover
    CVXPY_AVAILABLE = False


# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution (full mode) of two real vectors using FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) (sum(seq) is assumed to be 1)."""
    n = seq.size
    conv = _conv_fft(seq, seq)
    return 2.0 * n * conv.max()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    { x | x_i ≥ 0, Σ_i x_i = 1 }.
    """
    n = v.shape[0]
    if np.all(v <= 0):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed families (return a probability vector on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support (triangular numbers modulo n)."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


# ----------------------------------------------------------------------
# Smooth‑gradient descent on a soft‑max approximation of the max‑convolution
# ----------------------------------------------------------------------
def _smooth_gradient_descent(start_seq: np.ndarray,
                             deadline: float) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of  max(conv(a,a)) using gradient descent.
    The soft‑max temperature α grows slowly, turning the objective sharper.
    After each gradient step the iterate is projected onto the simplex.
    Returns the best sequence found and its true (non‑smoothed) objective.
    """
    a = start_seq.copy()
    n = a.size
    # best so far
    conv = _conv_fft(a, a)
    best_val = 2.0 * n * conv.max()
    best_seq = a.copy()

    # hyper‑parameters
    alpha = 20.0          # start temperature for the soft‑max
    lr0 = 0.5             # base learning rate
    it = 0
    rel_tol = 1e-12

    while time.time() < deadline:
        it += 1

        # ---------- compute current convolution & soft‑max ----------
        conv = _conv_fft(a, a)               # length 2n‑1
        max_c = conv.max()

        # soft‑max weights (numerically stable)
        shift = max_c
        exp_arg = alpha * (conv - shift)
        # clip avoid overflow/underflow
        exp_arg = np.clip(exp_arg, -700, 700)
        exp_vals = np.exp(exp_arg)
        Z = exp_vals.sum()
        if Z == 0.0:                # extremely unlikely, but keep safety
            w = np.full_like(exp_vals, 1.0 / exp_vals.size)
        else:
            w = exp_vals / Z        # sum(w) == 1

        # ---------- gradient of the smoothed max ----------
        rev_a = a[::-1]                     # reversed a
        conv_rev = _conv_fft(rev_a, w)      # length 3n‑2
        g = 2.0 * conv_rev[n - 1:n - 1 + n]  # gradient of smooth max

        # ---------- step and projection ----------
        lr = lr0 / math.sqrt(it)
        a_new = a - lr * g
        a = _project_simplex(a_new)

        # ---------- evaluate true objective ----------
        conv_new = _conv_fft(a, a)
        cur_val = 2.0 * n * conv_new.max()
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SmoothGrad] it={it:8d} best={cur_val:.6f}")

        # gradually sharpen the max
        if it % 500 == 0:
            alpha = min(alpha * 1.05, 500.0)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient (exponentiated‑gradient) phase (same as before)
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (exponentiated‑gradient) on the true non‑smooth max‑convolution.
    Returns the best sequence encountered and its true objective.
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    it = 0
    while time.time() < deadline:
        it += 1

        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over all active constraints
        g = np.zeros_like(a)
        if active.size:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)               # length = 3n‑2
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                         # numerical stability
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:8d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linearised LP refinement (convex‑concave procedure)
# ----------------------------------------------------------------------
def _lp_linearised_refine(a: np.ndarray,
                         deadline: float,
                         max_refine: int = 8,
                         eps_active: float = 1e-9) -> tuple[np.ndarray, float]:
    """
    Perform a few LP refinements where the quadratic constraints are linearised
    around the current point `a`. Returns the best vector discovered and its
    objective.
    """
    if not CVXPY_AVAILABLE:
        return a, _objective(a)

    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    # warm‑start variables for cvxpy
    x_var = cp.Variable(n)
    M_var = cp.Variable()
    constraints = [x_var >= 0,
                   cp.sum(x_var) == 1]

    for it in range(max_refine):
        if time.time() > deadline - 0.3:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]
        if active.size == 0:
            break

        # Build matrix A where A[row, i] = a[k-i] for k = active[row]
        A = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            A[r, i_range] = a[k - i_range]

        # Linearised constraints 2·A·x ≤ M + conv[active]
        lin_constraints = [2 * (A @ x_var) <= M_var + conv[active]]
        prob = cp.Problem(cp.Minimize(M_var), constraints + lin_constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=2000,
                       abstol=1e-9,
                       reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if x_var.value is None or M_var.value is None:
            break

        a_new = np.asarray(x_var.value).flatten()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s

        val_new = _objective(a_new)
        print(f"[LP] iter={it+1:2d} M={M:.6f} new_obj={val_new:.6f}")

        if val_new < best_val - 1e-9:
            best_val = val_new
            best_seq = a_new.copy()
            a = a_new
        else:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuner
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Tiny Metropolis‑Hastings walk that swaps a small amount of mass between
    two random positions and occasionally makes a global Gaussian jump.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    # leave a small safety margin

    rng = np.random.default_rng(seed + 12345)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising step function of length {n} ===")

    # ------------------------------------------------------------------
    # Initialise a decent starting point
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    seed_vec = None
    if previous is not None:
        try:

                seed_vec = prev_arr / prev_arr.sum()
            else:
                seed_vec = None
        except Exception:
            seed_vec = None

    if seed_vec is None:

    # ------------------------------------------------------------------

    grad_deadline = deadline - 150.0      # reserve ~150 s for later phases

        print("[Phase 1] skipped (insufficient time)")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    sub_deadline = deadline - 30.0

        seq_lp, val_lp = _lp_linearised_refine(seq_best, lp_deadline)
        if val_lp < best_val:

            print(f"[Phase 3] improvement: {val_lp:.6f}")
        else:
            print("[Phase 3] no improvement")
    else:

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing fine‑tuning (remaining time)

    if time.time() < deadline - 0.5:

    final_seq = np.maximum(seq_best, 0.0)
    total = final_seq.sum()

    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")

    return final_seq.tolist()
