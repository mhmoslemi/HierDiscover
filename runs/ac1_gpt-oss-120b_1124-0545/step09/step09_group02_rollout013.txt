def _best_shift(seq, n, deadline, max_trials=30):
    best_seq = seq
    best_val = _objective(seq)
    rng = np.random.default_rng(1234)
    for _ in range(max_trials):
        if time.time()>deadline: break
        shift = rng.integers(0, n)
        shifted = np.roll(seq, shift)
        val = _objective(shifted)
        if val < best_val:
            best_val = val
            best_seq = shifted.copy()
    return best_seq, best_val

"""
Propose a candidate step‑function (non‑negative vector) that minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

The evaluator is homogeneous, therefore we optimise on the probability
simplex (∑a = 1).  The search proceeds in four phases:

0.  If a construction from a previous run exists, evaluate it.
1.  Build a very large pool of analytic “seed” vectors (geometric,
    power‑law, classic windows, random Dirichlet / soft‑max draws, and a
    fine sweep of the Kaiser‑beta parameter).  Keep the best few.
2.  Refine each top seed with three deterministic first‑order methods:
        – sub‑gradient (mirror) descent on the exact max‑convolution,
        – Euclidean projected (additive) gradient descent,
        – Adam‑style optimisation of a *smoothed* max (log‑sum‑exp).
   The best vector from the three runs is kept.
3.  Post‑process the best vector:
        – try a few simple power‑transformations (a←a^γ, renormalise),
        – try a handful of random cyclic shifts,
        – finally run a simulated‑annealing Metropolis–Hastings walk to
          escape any remaining local minima.

All optimisation respects the simplex constraint by a Euclidean projection.
The routine is deliberately time‑aware: each phase checks the remaining
budget and aborts early if needed, guaranteeing a return before the
1000 s deadline.
"""

import numpy as np

# Optional SciPy imports (only for a few classic windows)

# ----------------------------------------------------------------------
# FFT utilities – linear convolution (full mode) using a real FFT
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (homogeneous, independent of overall scale)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic families – all return a *probability* vector (sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    if a.sum() == 0:
        return _uniform(n)
    return a / a.sum()


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    if a.sum() == 0:
        return _uniform(n)
    return a / a.sum()


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    if a.sum() == 0:
        return _uniform(n)
    return a / a.sum()


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    if a.sum() == 0:
        return _uniform(n)
    return a / a.sum()


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window, β ≥ 0."""
    if not SCIPY_AVAILABLE or beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    nminus1 = n - 1
    x = (2.0 * i / nminus1) - 1.0
    a = i0(beta * np.sqrt(1.0 - x * x)) / i0(beta)
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on (0,1), discretised on mid‑points.
    α ∈ (0,1] gives edge‑heavy shapes; α=1 is uniform.
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0:
        return _uniform(n)
    return a / a.sum()


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return _uniform(n)
    return x / x.sum()


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Rich set of starting vectors on the probability simplex."""
    rng = np.random.default_rng(20231123)  # reproducible but not tied to the seed argument
    seeds: list[np.ndarray] = []

    # classic families
    seeds.append(_uniform(n))
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for a in np.linspace(0.2, 0.8, 9):
        seeds.append(_tukey_window(n, a))
    if SCIPY_AVAILABLE:
        for beta in np.linspace(2.0, 12.0, 11):
            seeds.append(_kaiser_window(n, beta))

    # geometric / power sweep – much finer than the original code
    for r in np.linspace(0.80, 0.9999, 80):
        seeds.append(_geom(n, r))
    for p in np.linspace(0.0, 5.0, 81):
        seeds.append(_power(n, p))
    for r in np.linspace(0.80, 0.9999, 40):
        for p in np.linspace(0.0, 3.0, 31):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for r in np.linspace(0.8, 0.999, 30):
        seeds.append(_sym_geom(n, r))

    # beta‑pdf family
    for a in np.linspace(0.3, 3.0, 31):
        seeds.append(_beta_window(n, a))

    # random draws
    for alpha in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(6):
            seeds.append(_random_dirichlet(n, alpha, rng))
    for _ in range(20):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction if available
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up (clip, normalise)
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        total = s.sum()
        if total > 0:
            cleaned.append(s / total)
    return cleaned


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex (used by optimisers)
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Mirror‑descent (exponentiated‑gradient) on the *exact* max‑convolution
# ----------------------------------------------------------------------


def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 400000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror descent) on the true max‑convolution."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        # a slightly looser active set helps to move off plateaus
        tol = max(rel_tol, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        # sub‑gradient (average over active constraints)
        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        # exponentiated‑gradient step
        lr = 0.9 / np.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)                # numerical safety
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected additive gradient descent on a sub‑gradient of max‑conv
# ----------------------------------------------------------------------


def _additive_descent(start_seq: np.ndarray,
                      n: int,
                      deadline: float,
                      max_iters: int = 250000) -> tuple[np.ndarray, float]:
    """Simple projected gradient descent using a (averaged) sub‑gradient."""
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_val = conv.max()
        tol = max(rel_tol, 0.001 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        g = np.zeros_like(a)
        if len(active) > 0:
            w = np.zeros(2 * n - 1, dtype=np.float64)
            w[active] = 1.0 / len(active)
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]

        lr = 0.15 / np.sqrt(it)   # a tad larger than the original
        a_next = a - lr * g
        a_next = _project_simplex(a_next)

        a = a_next
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[AddGD]  it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam‑style optimiser on a *smoothed* max (log‑sum‑exp)
# ----------------------------------------------------------------------


def _adam_optimize(start_seq: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth approximation of ``max(conv)``.
    The smoothing parameter β grows with iterations so that the
    algorithm first sees a gentle landscape and then refines the true
    max.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2 = 0.90, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # modest initial learning rate – the gradient magnitude is O(1)
    lr0 = 0.12

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current full convolution
        conv = _conv_fft(a, a)

        # ---- smoothed max (log‑sum‑exp) ----
        beta = min(800.0, 5.0 + 0.01 * it)                # increase slowly
        scaled = conv * beta
        max_scaled = scaled.max()
        # numerical safe soft‑max weights
        w = np.exp(scaled - max_scaled)
        w /= w.sum()

        # gradient of the smoothed max w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        g = 2.0 * full[n - 1:n - 1 + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr0 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # enforce non‑negativity + simplex
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam]   it={it:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (identical to the previous version,
# but with a few extra move probabilities)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk with several move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-7, cur_val * 0.07)   # initial temperature
    decay = 0.99945

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.65:                 # mass transfer (most common)
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.random() * 0.22 * cur[i]   # up to 22 % of source
                new_seq = cur.copy()
                new_seq[i] -= delta
                new_seq[j] += delta
            else:
                continue
        elif rnd < 0.86:               # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                          # global Gaussian jitter
            noise = rng.normal(scale=0.0015, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        # Metropolis acceptance
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-13)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # occasional larger jitter jump
        if it % 2600 == 0:
            jitter = rng.normal(scale=0.0038, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val

def _power_transform(seq: np.ndarray, gamma: float) -> np.ndarray:
    """Apply a ↦ a^γ (renormalised).  γ≈1 leaves the vector unchanged."""
    if gamma <= 0:
        return seq.copy()
    a = np.power(seq, gamma)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = _uniform(seq.size)
    return a


def _try_power_transforms(seq: np.ndarray,
                          n: int,
                          deadline: float,
                          gammas: list[float]) -> tuple[np.ndarray, float]:
    """Test a few power‑law distortions of ``seq``."""
    best_seq = seq.copy()
    best_val = _objective(seq)
    for g in gammas:
        if time.time() > deadline:
            break
        cand = _power_transform(seq, g)
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Power]  γ={g:.3f} best={val:.6f}")
    return best_seq, best_val


def _try_random_shifts(seq: np.ndarray,
                       n: int,
                       deadline: float,
                       trials: int = 30) -> tuple[np.ndarray, float]:
    """Cyclically shift a few times and keep the best."""
    best_seq = seq.copy()
    best_val = _objective(seq)
    rng = np.random.default_rng()
    for _ in range(trials):
        if time.time() > deadline:
            break
        shift = rng.integers(0, n)
        cand = np.roll(seq, int(shift))
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Shift]  s={shift} best={val:.6f}")
    return best_seq, best_val

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # give ourselves a small safety margin
    deadline = start_time + budget_s - 3.0

    # ------------------------------------------------------------------
    # Problem size – 4000 is a good trade‑off for speed / quality
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Load a previous construction if it exists (global persistence)
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0) Evaluate previous construction (if any)
    # ------------------------------------------------------------------
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev]  val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) Build a very rich seed pool and keep the top few
    # ------------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    # evaluate every seed (fast – uses FFT internally)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 10
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed]  new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Deterministic refinements (sub‑gradient, additive GD, Adam)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 30.0:       # keep some room for later stages
            break
        print(f"[refine] start seed {idx}")

        # a) sub‑gradient (mirror‑descent)
        seq, val = _subgradient_refine(seed_vec, n, deadline,
                                       max_iters=300000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] subgrad best={val:.6f}")

        # b) additive projected GD
        seq, val = _additive_descent(seq, n, deadline,
                                     max_iters=180000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] additive best={val:.6f}")

        # c) Adam on smoothed max
        seq, val = _adam_optimize(seq, n, deadline,
                                  max_iters=150000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] Adam best={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Light post‑processing (power transforms + random shifts)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        # power exponents between 0.7 and 1.3
        gammas = np.linspace(0.70, 1.30, 13).tolist()
        seq, val = _try_power_transforms(best_seq, n, deadline, gammas)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

        # a few random cyclic shifts
        seq, val = _try_random_shifts(best_seq, n, deadline, trials=25)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning (uses remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline,
                                    max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[final] anneal best={val:.6f}")

    # ------------------------------------------------------------------
    # 5) Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
