def _logspace_sgd_optimize(start_vec, n, deadline, max_iters=300_000):
    eps = 1e-30
    v = np.log(start_vec + eps)
    beta = 0.5
    beta_inc = 0.5
    beta_max = 300.0
    lr0 = 0.8
    momentum = np.zeros_like(v)
    best_seq = None
    best_val = float('inf')
    for it in range(1, max_iters+1):
        if time.time() > deadline: break
        a = _softmax(v)
        # compute surrogate objective
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute gradient
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w = w / w_sum
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)
        # momentum update
        momentum = 0.9 * momentum + grad_v
        lr = lr0 / np.sqrt(it)
        v = v - lr * momentum
        beta = min(beta + beta_inc, beta_max)
    return best_seq, best_val

def _subgradient_proj(start_seq,n,deadline,max_iters=2_000_000):
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    # use decreasing step size with factor = c / sqrt(t)
    c = 0.4
    for it in range(1,max_iters+1):
        if time.time()>deadline: break
        conv = _conv_fft(a,a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # compute subgradient
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = c / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)
    return best_seq, best_val

    for it in range(1,max_iters+1):
        if time.time()>deadline: break
        T = max(T*0.9995, min_temp)
        k = rng.integers(2, 20)   # random subset size
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0: continue
        new_alloc = rng.dirichlet(np.full(k,1.0))*total
        new = cur.copy()
        new[idx]=new_alloc
        new_val = _objective(new)
        delta = new_val - cur_val

            cur_val = new_val

                best_val = new_val
                best_seq = new.copy()

"""
Propose a step‑function minimizing

    2·n·max(conv(seq, seq)) / (sum(seq)²)

by a multi‑stage search:

1.  Rich seed collection (analytic families, random Dirichlet,
    low‑frequency curves, mixtures) – extended parameter ranges
    (including very flat beta / arcsine shapes).

2.  Surrogate ranking (objective assumes Σseq = 1).

3.  Log‑space stochastic gradient descent on a smooth
    log‑sum‑exp approximation of the max (with momentum).

4.  Exact sub‑gradient descent on the true max‑convolution objective
    (projected onto the probability simplex).

5.  A short simulated‑annealing polishing that reshuffles the mass of
    a random subset of coordinates.

All stages respect a global time budget (default 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    (If Σseq ≠ 1 the returned value is scaled accordingly, which does not affect
    the ordering of candidates.)
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1, v ≥ 0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero → uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic shape families (all ℓ₁‑normalised)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a smooth non‑negative curve by summing a few random cosine
    components.  The result is ℓ₁‑normalised.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

def _logspace_sgd_optimize(start_vec: np.ndarray, n: int,
                           deadline: float,
                           max_iters: int = 300_000) -> tuple[np.ndarray | None, float]:
    """
    Stochastic gradient descent on log‑softmax parameters using a
    log‑sum‑exp surrogate for the max.  Includes a simple momentum term.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                       # logits
    beta = 0.5
    beta_inc = 0.5
    beta_max = 300.0
    lr0 = 0.8
    momentum = np.zeros_like(v)

    best_seq = None
    best_val = float("inf")

    it = 0
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break

        a = _softmax(v)                                # point on simplex
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # smoothed max via log‑sum‑exp
        conv = _conv_fft(a, a)
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient wrt a (linear convolution)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                     # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]           # central n entries

        # convert to log‑space gradient
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # momentum + learning‑rate schedule
        momentum = 0.9 * momentum + grad_v
        lr = lr0 / np.sqrt(it)
        v = v - lr * momentum

        # raise temperature slowly
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(start_seq: np.ndarray, n: int,
                      deadline: float,
                      max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing step size c/√t and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    c = 0.40          # initial step‑size factor

    it = 0
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient at the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = c / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)           # keep on simplex

    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray, n: int,
                         deadline: float,
                         max_iters: int = 500_000,
                         init_temp_factor: float = 0.03,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Simple SA that reshuffles the mass of a random subset of coordinates.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    it = 0
    while it < max_iters:
        it += 1
        if time.time() > deadline:
            break

        # geometric cooling
        T = max(T * 0.9995, min_temp)

        # pick a random subset (size 2 … 19)
        k = rng.integers(2, 20)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # redistribute total mass uniformly on the simplex of size k
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """Make a sequence symmetric (a[i] = a[n‑1‑i]) and project onto the simplex."""
    arr = np.asarray(seq, dtype=np.float64)
    rev = arr[::-1]
    sym = (arr + rev) * 0.5
    sym = _proj_simplex(sym)            # keep sum = 1, non‑negative
    return sym


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function minimising the evaluation
    function within the given time budget.  Returns the best sequence
    as a plain list of floats and stores it in the global variable
    ``height_sequence_1``.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # safety margin for the environment
    deadline = start_time + budget_s - 1.0

    # ------------------------------------------------------------------
    # Choose a length – 4000 is a good trade‑off between expressive power
    # and computational cost.
    # ------------------------------------------------------------------
    n = 4000
    print(f"\n=== Optimising length {n} ===")

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and surrogate ranking
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    seeds = _collect_seeds_extended(n, rng, previous)
    print(f"Generated {len(seeds)} candidate seeds.")

    surrogate_vals = [(vec, _objective(vec)) for vec in seeds]
    surrogate_vals.sort(key=lambda x: x[1])
    top_k = 5
    top_seeds = [vec for vec, _ in surrogate_vals[:top_k]]
    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(surrogate_vals[:top_k]):
        print(f"  #{i + 1}: {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – optimisation loops (log‑SGD → subgradient → symmetrisation)
    # ------------------------------------------------------------------
    best_seq = None
    best_real_val = float("inf")

    for idx, seed_vec in enumerate(top_seeds, 1):
        if time.time() > deadline:
            break
        print(f"\n--- Optimisation run {idx}/{top_k} ---")

        # 2a. Log‑space SGD (smooth surrogate)
        seq_sgd, val_sgd = _logspace_sgd_optimize(seed_vec, n, deadline,
                                                  max_iters=300_000)
        if seq_sgd is None:
            seq_sgd = seed_vec

        # 2b. Exact sub‑gradient descent
        seq_sub, val_sub = _subgradient_proj(seq_sgd, n, deadline,
                                             max_iters=1_200_000)

        # 2c. Symmetrise the result – most optimal solutions are symmetric
        seq_sym = _symmetrize(seq_sub)

        # 2d. Brief simulated‑annealing polishing
        seq_polished, val_polished = _simulated_annealing(seq_sym, n,
                                                          deadline,
                                                          max_iters=200_000)

        # ------------------------------------------------------------------
        # Evaluate the *real* objective (the one used for scoring)
        # ------------------------------------------------------------------
        for candidate in (seq_sgd, seq_sub, seq_sym, seq_polished):
            if candidate is None:
                continue
            try:
                real_val = evaluate_sequence(candidate.tolist())
            except Exception as exc:
                print("evaluate_sequence raised:", exc)
                real_val = float("inf")
            if real_val < best_real_val:
                best_real_val = real_val
                best_seq = candidate.copy()
                print(f"[global] New best real value = {real_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (in case no candidate improved the bound)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to a simple uniform distribution (guaranteed to be valid)
        best_seq = _uniform(n)

    # Expose for possible reuse by future calls
    globals()["height_sequence_1"] = best_seq.tolist()

    print(
        f"\n=== Finished ===\nBest real objective ≈ {best_real_val:.6f} "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
