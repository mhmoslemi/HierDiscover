"""
Non‑negative step‑function optimiser (robust multi‑restart version).

The algorithm works in three stages:

1️⃣ **Seed construction** – a large, cheap collection of candidate shapes
   (uniform, triangular, sine‑window, symmetric/unsymmetric geometric,
   power‑law, beta‑distribution, exponential decay, log‑decay and a few
   random Dirichlet vectors).  All candidates are normalised to sum = 1
   and evaluated with the fast surrogate ``2·n·max(conv(a,a))`` (FFT).

2️⃣ **Mirror‑descent (exponentiated‑gradient) refinement** – the best few
   seeds are refined by a sub‑gradient loop that directly minimises the
   maximum convolution while staying on the probability simplex.
   The learning‑rate starts at ~2.5 and decays as 1/√t.
   Multiple restarts are performed until the time budget is exhausted.

3️⃣ **Optional simulated‑annealing polish** – a cheap SA that reshuffles a
   tiny random subset of the mass and accepts downhill moves or
   Metropolis‑type uphill moves.  This often squeezes a little more out of
   the current best.

The final vector is returned as a plain ``list[float]`` ready for
``evaluate_sequence``.  The result is also stored in the global variable
``height_sequence_1`` for possible reuse in a later run.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (fast, O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.ifft(fx * fy, n=N)[:L].real
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    ``seq`` must be non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))``.
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    """Symmetric sum of two geometric progressions: r**i + r**(n‑1‑i)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i) + np.power(r, n - 1 - i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decay(n: int, r: float) -> np.ndarray:
    """Pure geometric decay a_i = r**i (unsymmetric)."""
    if r <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exp_decay(n: int, alpha: float) -> np.ndarray:
    """Exponential decay a_i = exp(-alpha*i)."""
    if alpha <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _log_decay(n: int, beta: float) -> np.ndarray:
    """Log‑decay a_i = 1 / (log(i+2)**beta)."""
    if beta <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.power(np.log(i + 2.0), beta)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decay(n: int, p: float) -> np.ndarray:
    """Power‑law decay a_i = (i+1)**‑p."""
    if p < 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta distribution on {1,…,n}."""
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, alpha - 1.0) * np.power(n - i + 1.0, beta - 1.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()


# ----------------------------------------------------------------------
# Seed construction (large but cheap)
# ----------------------------------------------------------------------


def _generate_seeds(
    n: int, rng: np.random.Generator, previous: np.ndarray | None
) -> list[np.ndarray]:
    """Return a diverse list of normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # symmetric geometric – medium grid
    for r in np.linspace(0.10, 0.90, 13):
        seeds.append(_sym_geom(n, r))

    # unsymmetric geometric decay – fine grid near 1
    for r in np.linspace(0.985, 0.9999, 12):
        seeds.append(_geom_decay(n, r))

    # exponential decay – also fine grid
    for a in np.linspace(0.001, 0.02, 12):
        seeds.append(_exp_decay(n, a))

    # log‑decay – grid
    for b in np.linspace(0.2, 2.0, 9):
        seeds.append(_log_decay(n, b))

    # pure power‑law decay – dense grid
    for p in np.arange(0.05, 3.01, 0.15):
        seeds.append(_power_decay(n, p))

    # beta family – finer grid (including very small exponents)
    for a in np.linspace(0.05, 2.0, 7):
        for b in np.linspace(0.05, 5.0, 7):
            seeds.append(_beta_shape(n, a, b))

    # mixtures of uniform with a symmetric geometric
    uniform = _uniform(n)
    for r in np.linspace(0.20, 0.90, 7):
        base = _sym_geom(n, r)
        for w in np.linspace(0.0, 1.0, 4):
            mixed = w * uniform + (1.0 - w) * base
            mixed = mixed / mixed.sum()
            seeds.append(mixed)

    # random Dirichlet draws – a handful for diversity
    for conc in (0.02, 0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(4):
            seeds.append(_random_dirichlet(n, conc, rng))

    # random sparse patterns – tiny support
    for k in (3, 5, 10):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous solution (if compatible length)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    return seeds


# ----------------------------------------------------------------------
# Optimisation primitive – exponentiated‑gradient (mirror descent)
# ----------------------------------------------------------------------


def _mirror_descent(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
    lr0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the max‑convolution objective.
    The update uses a learning‑rate that decays as 1/√t.
    Returns the best sequence seen and its surrogate value.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    eps = 1e-30

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        # construct sub‑gradient of the active convolution term
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # exponentiated‑gradient step
        lr = lr0 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise numerics
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = _uniform(n)
        else:
            a /= a_sum

        # track best
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 250_000 == 0:
                print(f"[Mirror] it={it:9d} best≈{best_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (optional)
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 150_000,
    init_temp_factor: float = 0.02,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Simple SA that reshuffles the mass of a random tiny subset."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        T = max(T * 0.9995, min_temp)

        # pick a random subset of positions (size 2…6)
        k = rng.integers(2, 7)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA]  it={it:8d} best≈{best_val:.6f}")

    return best_seq, best_val

    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

        seq_ref, val_ref = _mirror_descent(
            seed_vec, n, deadline, max_iters=max_iters, lr0=2.0

    # ------------------------------------------------------------------

    print(

    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
