def build_set_greedy(n,m_target, max_ordered, rng, time_deadline):
    best_mask = np.zeros(n, dtype=bool)
    best_sz = 0
    attempts = 0
    while time.time()<time_deadline:
        attempts+=1
        # start with random set of size target_m maybe all indices random?
        # or start with empty and add greedily random order.
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2*n-1, dtype=np.int16)

        # Add random permutation elements until cannot add anymore
        order = rng.permutation(n)
        for cand in order:
            # compute prospective increase
            # quickly compute if sum_counts[2*cand] + 1 <= max_ordered
            if sum_counts[2*cand] + 1 > max_ordered:
                continue
            # existing support
            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing
                # for each sum, adding candidate adds 2 to sum_counts[sum]; need to test if any exceed max
                # vectorized check:
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue
            # accept:
            mask[cand] = True
            sum_counts[2*cand] += 1
            if existing.size:
                sum_counts[sums] += 2
            if mask.sum() >= m_target:
                break

        # try to improve mask by adding more from leftover elements when no collision? Actually after loop we have size maybe less than target.

        # perform local additions: shuffle leftover and attempt to add while possible
        leftover = np.where(~mask)[0]
        rng.shuffle(leftover)
        for cand in leftover:
            if sum_counts[2*cand] + 1 > max_ordered:
                continue
            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue
            mask[cand] = True
            sum_counts[2*cand] += 1
            if existing.size:
                sum_counts[sums] += 2
            if mask.sum() >= m_target:
                break

        cur = mask.sum()
        if cur > best_sz:
            best_sz = cur
            best_mask = mask.copy()
            print(f"[LowCollGreedy] attempt {attempts} size {cur}")
            # possible early exit when size close to theoretical upper bound sqrt(8n)

"""
propose_candidate – a fast low‑collision support search.

The idea is simple:
*   The objective for a probability vector `p` of length *n* is
        2·n·max(conv(p,p)) / (sum(p)^2) .
    If the support size is `m` and all non‑zero entries are equal,
    max(conv) = (max ordered pair count) / m², therefore the bound
    becomes 2·n·C / m² where C is the worst ordered‐collision count.
    With C=4 the theoretical optimum is 8·n/m² and the maximal feasible
    `m` satisfies `m² ≤ 8·n`.  For n=4000 this gives m≈179 and a bound ≈1.0.
*   We therefore try to find a large set `S ⊂ {0,…,n−1}` such that each
    ordered sum `i+j` occurs at most 4 times (a B₂[4] set).  The larger the
    set the smaller the bound.
*   A random “over‑filled’’ set is generated, then repaired by repeatedly
    removing elements that partake in a violating sum.  After the repair we
    greedily add further elements while respecting the collision limit.
*   The best support found within the allotted time is turned into a uniform
    probability vector (all mass 1/|S| on the support).  This distribution
    is already optimal for the chosen support, so we skip the heavy
    continuous optimisation phases.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real input, full mode)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Exact linear convolution using a real FFT.
    Returns an array of length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Objective (exact, no clipping) – identical to the one used by the grader.
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """Return 2·n·max(conv(seq,seq)) / (sum(seq)^2)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Low‑collision set utilities
# ----------------------------------------------------------------------
def _conv_counts(mask: np.ndarray) -> np.ndarray:
    """
    Return the ordered‑pair count array for a 0/1 mask.
    The result is a vector of length 2·n‑1 containing exact integers.
    """
    conv = _conv_fft(mask.astype(np.float64), mask.astype(np.float64))
    # rounding removes tiny floating‑point noise
    return np.rint(conv).astype(np.int32)


def _repair_mask(mask: np.ndarray, max_ordered: int, rng: np.random.Generator) -> np.ndarray:
    """
    Remove elements until every ordered sum occurs at most `max_ordered` times.
    This is a simple greedy repair: pick a violating sum, drop one
    offending element, recompute, repeat.
    """
    n = mask.size
    while True:
        counts = _conv_counts(mask)
        bad = np.where(counts > max_ordered)[0]
        if bad.size == 0:
            break                       # feasible
        # pick a random offending sum
        s = rng.choice(bad)
        support = np.nonzero(mask)[0]

        # find elements that contribute to this sum
        # i contributes if j = s - i is also in the support
        j = s - support
        ok = (j >= 0) & (j < n)
        i_candidates = support[ok]
        j_candidates = j[ok]
        offending = i_candidates[mask[j_candidates]]
        if offending.size == 0:          # fallback – drop any support element
            to_remove = rng.choice(support)
        else:
            to_remove = rng.choice(offending)
        mask[to_remove] = False
    return mask


def _greedy_add(mask: np.ndarray, max_ordered: int, rng: np.random.Generator) -> np.ndarray:
    """
    Starting from a feasible mask, try to insert additional indices.
    Uses the same incremental check as the original greedy routine.
    """
    n = mask.size
    # current ordered‑pair counts
    sum_counts = _conv_counts(mask)

    candidates = np.where(~mask)[0]
    rng.shuffle(candidates)

    for cand in candidates:
        # self‑pair check
        if sum_counts[2 * cand] + 1 > max_ordered:
            continue
        # pairs with existing support
        existing = np.nonzero(mask)[0]
        if existing.size:
            sums = cand + existing
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue
        # accept the candidate
        mask[cand] = True
        sum_counts[2 * cand] += 1
        if existing.size:
            sum_counts[sums] += 2
    return mask


def _search_low_collision_set(
    n: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """
    Repeatedly generate a random over‑filled mask, repair it, then greedily
    enlarge it.  Keep the best (largest) feasible mask found before the deadline.
    """
    best_mask = np.zeros(n, dtype=bool)
    best_size = 0
    attempts = 0
    theoretical_max = int(math.sqrt(8.0 * n))

    while time.time() < deadline:
        attempts += 1

        # ---- random over‑filled initial mask ---------------------------------
        # aim slightly above the theoretical bound; the repair will trim it down
        init_sz = min(n, theoretical_max + rng.integers(-5, 6))
        init_sz = max(10, init_sz)                     # guard against tiny sets
        idx = rng.choice(n, size=init_sz, replace=False)
        mask = np.zeros(n, dtype=bool)
        mask[idx] = True

        # ---- repair ---------------------------------------------------------
        mask = _repair_mask(mask, max_ordered, rng)

        # ---- greedy fill ----------------------------------------------------
        mask = _greedy_add(mask, max_ordered, rng)

        cur_sz = mask.sum()
        if cur_sz > best_size:
            best_size = cur_sz
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:4d} size {cur_sz}")

        # early exit if we reach the theoretical ceiling
        if best_size >= theoretical_max:
            break

    return best_mask

# Main entry point expected by the grader
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Construct a large low‑collision support (ordered pair count ≤ 4)
    and return the uniform probability vector on that support.
    The algorithm is pure combinatorial; no continuous optimisation
    phases are required.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for interpreter overhead
    deadline = start_time + budget_s - 5.0

    rng = np.random.default_rng(seed + 1)

    # ------------------------------------------------------------------
    # Problem size – a power of two makes the FFT cheap
    # ------------------------------------------------------------------
    n = 4000
    max_ordered = 4
    print(f"\n=== Low‑collision set search (n={n}) ===")

    # ------------------------------------------------------------------
    # Phase 1 – spend ~70 % of the budget on searching a large feasible set
    # ------------------------------------------------------------------
    set_deadline = start_time + 0.70 * (deadline - start_time)
    best_mask = _search_low_collision_set(
        n=n,
        max_ordered=max_ordered,
        rng=rng,
        deadline=set_deadline,
    )
    support_idx = np.where(best_mask)[0]
    m = support_idx.size
    if m == 0:                                   # safety net – fall back to uniform
        print("[Fallback] no low‑collision set found, using uniform distribution")
        seq = np.ones(n, dtype=np.float64) / n
    else:
        print(f"[Info] best support size = {m}")
        # uniform distribution on the support (optimal for a given support)
        seq = np.zeros(n, dtype=np.float64)
        seq[support_idx] = 1.0 / m

    # ------------------------------------------------------------------
    # Phase 2 – optional short refinement (only if time remains)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        # a very cheap exponentiated‑gradient run; it will keep the vector
        # essentially uniform, but may slightly improve the max‑convolution.
        a = seq.copy()
        best_val = _objective(a)
        for it in range(1, 200_001):
            if time.time() > deadline:
                break
            # compute exact sub‑gradient (as in the previous code)
            conv = _conv_fft(a, a)
            max_val = conv.max()
            active = np.where(conv >= max_val - 1e-12)[0]

            # sub‑gradient (average over active constraints)
            if active.size:
                w = np.zeros(2 * n - 1, dtype=np.float64)
                w[active] = 1.0 / active.size
                rev_a = a[::-1]
                full = _conv_fft(rev_a, w)
                g = 2.0 * full[n - 1:n - 1 + n]
            else:
                g = np.zeros_like(a)

            lr = 0.9 / math.sqrt(it)
            log_a = np.log(a + 1e-30) - lr * g
            log_a -= np.max(log_a)               # keep numbers stable
            a = np.exp(log_a)
            a /= a.sum()

            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                seq = a.copy()
                print(f"[Refine] it={it:6d} best={cur_val:.6f}")

    # ------------------------------------------------------------------
    # Final normalisation & safety checks (the grader will also clip)
    # ------------------------------------------------------------------
    final_seq = np.maximum(seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.ones(n, dtype=np.float64) / n
    else:
        final_seq = final_seq / total

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = final_seq.copy()

    print(f"\n=== Finished ===\nSupport size = {m}, objective ≈ {_objective(final_seq):.6f}")
    return final_seq.tolist()
