def _smooth_grad_desc(start, n, deadline, max_iters_total=5000):
    a = start.copy()
    best = a.copy()
    best_val = _objective(a)

    t = 0
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9
    beta2=0.999
    epsilon=1e-8
    # schedule for alpha as list
    alphas = [0.5, 2., 5., 10., 20., 40.]
    # steps per alpha
    steps_per_alpha = max_iters_total // len(alphas)
    for alpha in alphas:
        for i in range(steps_per_alpha):
            if time.time()>deadline: break outer loops
            t+=1
            conv = _conv_fft(a,a)
            # compute w softmax with alpha
            # shift for numerical stability
            maxc = conv.max()
            scaled = (conv - maxc) * alpha
            exp = np.exp(scaled)
            w = exp / exp.sum()
            # gradient compute via convolution
            a_rev = a[::-1]
            grad_full = _conv_fft(w, a_rev)
            grad = 2.0 * grad_full[(n-1):(n-1+n)]

            # Adam update
            lr = 0.05 / np.sqrt(t)  # decreasing lr
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            a = a - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            # projection onto simplex
            a = _project_simplex(a)

            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best = a.copy()
                print(f"[Smooth] it={t} alpha={alpha:.1f} best={cur_val:.6f}")

    return best, best_val

"""
Hybrid optimiser: analytic seeds → smoothed‐gradient (soft‑max of the max‑convolution)
→ short sub‑gradient refinement → simulated‑annealing fine‑tuning.

The core idea is to replace the non‑smooth max‑operator by a temperature‑controlled
soft‑max.  For a given temperature α the objective
    f_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])
is a smooth upper‑bound of max(a*a).  Its gradient can be computed by a single
convolution, which makes each iteration O(n log n).  By increasing α gradually
the method follows the true non‑smooth objective while keeping stable updates.
After a few passes we polish the result with a cheap sub‑gradient step and a
final simulated‑annealing walk.

The routine respects the 1000 s time‑budget, works on the probability simplex
(∑a = 1) and returns a list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based linear convolution (real → real)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective used by the grader (scale‑invariant)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)**2).  n = len(seq)."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Analytic families (all vectors are already normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_power(n: int, r: float, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = (r ** (idx - 1)) / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sym_geom(n: int, r: float) -> np.ndarray:
    a = r ** np.arange(n, dtype=np.float64)
    a = a + r ** np.arange(n - 1, -1, -1, dtype=np.float64)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    return a / a.sum()


def _gaussian(n: int, sigma_factor: float = 0.18) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = sigma_factor * n
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return a / a.sum()


def _tent_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    return a / a.sum()


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    return a / a.sum()


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    """Tukey window – 0 ≤ α ≤ 1."""
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        return _hann_window(n)
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.ones_like(t)
    first = t < alpha / 2.0
    a[first] = 0.5 * (1 + np.cos(np.pi * (2 * t[first] / alpha - 1)))
    last = t > 1.0 - alpha / 2.0
    a[last] = 0.5 * (1 + np.cos(np.pi * (2 * t[last] / alpha - 2 / alpha + 1)))
    return a / a.sum()


def _beta_window(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on (0,1), discretised on mid‑points."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    a = np.power(t, alpha - 1.0) * np.power(1.0 - t, alpha - 1.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration `alpha`."""
    x = rng.gamma(alpha, 1.0, size=n)
    return x / x.sum()


def _make_extended_seeds(n: int, previous: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors (all normalised to sum = 1)."""
    rng = np.random.default_rng(123456)
    seeds: list[np.ndarray] = []

    seeds.append(_uniform(n))

    # geometric – heavy sweep
    for c in np.linspace(0.5, 12.0, 48):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_geom(n, r))

    # power law – finer sweep
    for p in np.linspace(0.0, 4.0, 41):
        seeds.append(_power(n, p))

    # combined geometric‑power
    for c in np.linspace(0.5, 12.0, 24):
        r = 1.0 - c / n
        if r <= 0.0:
            continue
        for p in np.linspace(0.0, 2.0, 9):
            seeds.append(_geom_power(n, r, p))

    # symmetric geometric
    for c in np.linspace(0.5, 12.0, 24):
        r = 1.0 - c / n
        if r > 0.0:
            seeds.append(_sym_geom(n, r))

    # classic windows
    seeds.append(_sine_window(n))
    seeds.append(_tent_window(n))
    seeds.append(_hann_window(n))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    for alpha in np.linspace(0.2, 0.8, 8):
        seeds.append(_tukey_window(n, alpha))
    for alpha in np.linspace(0.3, 2.5, 23):
        seeds.append(_beta_window(n, alpha))
    for sigma_factor in np.linspace(0.06, 0.30, 13):
        seeds.append(_gaussian(n, sigma_factor))

    # Dirichlet draws (very sparse and very dense)
    for alpha in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(3):
            seeds.append(_random_dirichlet(n, alpha, rng))

    # random soft‑max vectors
    for _ in range(12):
        seeds.append(_random_softmax(n, rng))

    # reuse previous construction (if compatible)
    if previous is not None and previous.shape == (n,):
        seeds.append(previous / previous.sum())

    # final clean‑up
    cleaned = []
    for s in seeds:
        s = np.clip(s, 0.0, None)
        tot = s.sum()
        if tot > 0:
            cleaned.append(s / tot)
    return cleaned


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Smoothed gradient descent (soft‑max of the max‑convolution) + Adam
# ----------------------------------------------------------------------


def _smooth_grad_descent(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 5000) -> tuple[np.ndarray, float]:
    """
    Minimise a smooth upper bound of max(conv(a,a)) using a temperature‑controlled
    soft‑max and Adam updates.  Returns the best sequence seen.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule – start loose, end tight
    alphas = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0]
    it = 0
    outer_break = False

    for alpha in alphas:
        # how many inner iterations for this temperature?
        inner = max_iters // len(alphas)
        for _ in range(inner):
            if time.time() > deadline:
                outer_break = True
                break
            it += 1

            # current convolution and soft‑max weights
            conv = _conv_fft(a, a)                     # length 2n‑1
            maxc = conv.max()
            # stabilised soft‑max
            scaled = alpha * (conv - maxc)
            exp = np.exp(scaled)
            w = exp / exp.sum()                        # positive, sums to 1

            # gradient = 2 * (w * a_rev)[centre … centre+n‑1]
            a_rev = a[::-1]
            grad_full = _conv_fft(w, a_rev)           # length 3n‑2
            grad = 2.0 * grad_full[n - 1:n - 1 + n]

            # Adam step (step size decays slowly with iteration count)
            lr = 0.05 / math.sqrt(it)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # projection onto the simplex (keeps non‑negativity & sum=1)
            a = _project_simplex(a)

            cur_val = _objective(a)
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Smooth] it={it:6d} α={alpha:5.1f} best={cur_val:.6f}")

        if outer_break:
            break

    return best_seq, best_val

# Simulated‑annealing fine‑tuning (as in the original code)
# ----------------------------------------------------------------------


def _anneal_optimize(start_seq: np.ndarray,
                     n: int,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Metropolis–Hastings walk with several move types."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    # initial temperature – a modest fraction of the current objective
    T0 = max(1e-7, cur_val * 0.07)
    decay = 0.9994

    it = 0
    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        rnd = rng.random()
        if rnd < 0.68:                               # mass transfer
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            delta = rng.random() * 0.18 * cur[i]   # up to 18 % of source mass
            new_seq = cur.copy()
            new_seq[i] -= delta
            new_seq[j] += delta
        elif rnd < 0.86:                             # swap two entries
            i, j = rng.integers(0, n, size=2)
            new_seq = cur.copy()
            new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
        else:                                        # global Gaussian jitter
            noise = rng.normal(scale=0.0013, size=n)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            total = new_seq.sum()
            if total > 0:
                new_seq /= total
            else:
                new_seq = _uniform(n)

        new_seq = _project_simplex(new_seq)
        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:8d} best={new_val:.6f}")

        # occasional stronger jump to shake out of plateaus
        if it % 2500 == 0:
            jitter = rng.normal(scale=0.0035, size=n)
            cand = best + jitter
            cand = np.clip(cand, 0.0, None)
            cand = _project_simplex(cand)
            cand_val = _objective(cand)
            if cand_val < best_val:
                best_val = cand_val
                best = cand.copy()
                print(f"[Anneal] (jump) it={it:8d} best={cand_val:.6f}")

    return best, best_val

def propose_candidate(seed: int = 42,
                      budget_s: int = 1000,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.  Returns the best sequence found as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # length – 4000 gives a good trade‑off between quality and speed
    n = 4000

    # possible reuse of the best construction from a previous run
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None

    # ------------------------------------------------------------------
    # 0) evaluate previous construction (if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    if previous is not None and previous.shape == (n,):
        val = _objective(previous)
        best_seq = previous.copy()
        best_val = val
        print(f"[prev] val={val:.6f}")

    # ------------------------------------------------------------------
    # 1) generate a very rich seed pool and keep the top few
    # ------------------------------------------------------------------
    seeds = _make_extended_seeds(n, previous)
    seed_vals = [(s, _objective(s)) for s in seeds]
    seed_vals.sort(key=lambda x: x[1])
    top_k = 8
    top_seeds = [s for s, _ in seed_vals[:top_k]]

    # track best seed
    if seed_vals[0][1] < best_val:
        best_val = seed_vals[0][1]
        best_seq = seed_vals[0][0].copy()
        print(f"[seed] new best={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Smoothed‑gradient (Adam) refinement on the top seeds
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(top_seeds):
        if time.time() > deadline - 12.0:
            break
        print(f"[refine] start smoothed gradient seed {idx}")

        seq, val = _smooth_grad_descent(seed_vec, n, deadline, max_iters=5000)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[refine] smooth best={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Small sub‑gradient polishing (the original mirror‑descent)
    # ------------------------------------------------------------------
    # We reuse a very short version of the former sub‑gradient step to clean
    # up any remaining tiny violations.
    def _quick_subgrad(a0: np.ndarray) -> tuple[np.ndarray, float]:
        a = a0.copy()
        best = a.copy()
        bestv = _objective(a)
        for it in range(1, 20001):
            if time.time() > deadline:
                break
            conv = _conv_fft(a, a)
            maxc = conv.max()
            tol = max(1e-12, 0.001 * maxc)
            active = np.where(conv >= maxc - tol)[0]

            g = np.zeros_like(a)
            if active.size:
                w = np.zeros(2 * n - 1, dtype=np.float64)
                w[active] = 1.0 / active.size
                rev_a = a[::-1]
                full = _conv_fft(rev_a, w)
                g = 2.0 * full[n - 1:n - 1 + n]

            lr = 0.7 / math.sqrt(it)
            log_a = np.log(a + 1e-30) - lr * g
            log_a -= np.max(log_a)
            a = np.exp(log_a)
            a /= a.sum()

            curv = _objective(a)
            if curv < bestv:
                bestv = curv
                best = a.copy()
                print(f"[QuickSub] it={it:5d} best={curv:.6f}")
        return best, bestv

    if best_seq is not None and time.time() < deadline - 6.0:
        seq, val = _quick_subgrad(best_seq)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[post] quick‑subgrad best={val:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning (remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        seq, val = _anneal_optimize(best_seq, n, deadline, max_iters=None)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[overall] anneal best={val:.6f}")

    # ------------------------------------------------------------------
    # 5) Fallback (unlikely)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq)
        print("[fallback] using uniform distribution")

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {len(best_seq)})")
    globals()["height_sequence_1"] = best_seq.copy()
    # convert to plain Python list of floats for the grader
    return best_seq.tolist()
