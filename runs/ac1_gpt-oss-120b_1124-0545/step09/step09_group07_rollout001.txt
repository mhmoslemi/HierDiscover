def _simplex_projection(v):
    # ensures non-negative and sum =1
    # works for any shape
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _anneal_optimize(start_seq, deadline):
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val
    T0 = max(1e-6, cur_val * 0.05)
    dec = 0.9998
    it = 0
    while time.time() < deadline:
        it += 1
        T = T0 * (dec ** it)
        i, j = rng.integers(0, cur.size, size=2)
        if i == j:
            continue
        delta = rng.random() * 0.01 * cur[i]
        delta = min(delta, cur[i])
        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it} best={new_val:.6f}")
        if it % 500 == 0:
            # global jump
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0, None)
            s = new_seq.sum()
            if s > 0:
                new_seq /= s
            else:
                new_seq = np.ones_like(cur) / cur.size
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] global jump best={new_val:.6f}")
    return best, best_val

"""
propose_candidate – Gradient‑Descent with Soft‑max Approximation + Annealing

The routine searches a non‑negative step function (length n=4000) that
minimises

    2·n·max(conv(a,a)) / (sum(a)**2)

Instead of optimising the non‑smooth max directly, we replace it with a
smooth log‑sum‑exp (soft‑max) using a temperature β that is increased
gradually (continuation).  The gradient of the smoothed objective can be
written in closed form and is obtained by a single correlation
(convolution) with the soft‑max weights.  An Adam optimiser updates the
vector, after each step the iterate is projected onto the probability
simplex (non‑negative, sum = 1).  Several β‑stages are run, with tiny
random perturbations between them to escape shallow wells.  When the
time budget is almost exhausted a short simulated‑annealing fine‑tune
is performed.

The algorithm is self‑contained, uses only NumPy and the provided
evaluate_sequence (via the internal _objective) and respects the
1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (FFT‑based convolution)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, sum = 1)
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple simulated‑annealing finetuner (used for the last seconds)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-8, cur_val * 0.05)
    decay = 0.9998
    it = 0

    while time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        # local move: shift a small amount of mass between two positions
        i, j = rng.integers(0, cur.size, size=2)
        if i == j:
            continue
        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)
        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        # occasional global perturbation
        if it % 500 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            s = new_seq.sum()
            if s > 0.0:
                new_seq /= s
            else:
                new_seq = np.ones_like(cur) / cur.size
            new_val = _objective(new_seq)
            if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] global‑jump it={it:7d} best={new_val:.6f}")

    return best, best_val


# ----------------------------------------------------------------------
# Main entry point – gradient descent with soft‑max continuation
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns a plain list of floats.
    The routine respects the supplied time budget (default 1000 s).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 1.0   # safety margin

    # -----------------------------
    # Problem size
    # -----------------------------
    n = 4000
    print(f"\n=== Optimising step function of length {n} ===")

    # -----------------------------
    # Initialise a starting point
    # -----------------------------
    previous = globals().get("height_sequence_1")
    if isinstance(previous, list) and len(previous) == n:
        a = np.asarray(previous, dtype=np.float64)
        a = np.maximum(a, 0.0)
        if a.sum() > 0.0:
            a /= a.sum()
        else:
            a = np.ones(n) / n
    else:
        a = np.ones(n, dtype=np.float64) / n

    best_seq = a.copy()
    best_val = _objective(a)
    print(f"[Init] objective = {best_val:.6f}")

    # -----------------------------
    # Adam state
    # -----------------------------
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # -----------------------------
    # Soft‑max temperature schedule
    # -----------------------------
    betas = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0]

    # -----------------------------
    # Main optimisation loop
    # -----------------------------
    for stage, beta in enumerate(betas):
        if time.time() >= deadline:
            break

        # allocate a rough number of iterations for this stage
        remaining = deadline - time.time()
        # give each remaining stage a fair share of the time
        stages_left = len(betas) - stage
        it_budget = int(min(8000, max(500, remaining * 2.0 / stages_left)))

        print(f"[Beta {beta:.1f}] start, up to {it_budget} iterations, time left {remaining:.1f}s")
        for it in range(it_budget):
            if time.time() >= deadline:
                break

            t += 1

            # ----- convolution a * a -----
            conv = _conv_fft(a, a)                     # length 2n‑1

            # ----- soft‑max weights -----
            scaled = beta * conv
            max_s = np.max(scaled)
            exp_scaled = np.exp(scaled - max_s)        # prevent overflow
            w = exp_scaled / np.sum(exp_scaled)        # shape (2n‑1,)

            # ----- gradient (correlation of w with a) -----
            a_rev = a[::-1]
            corr_full = _conv_fft(w, a_rev)            # length 3n‑2
            corr = corr_full[n - 1: n - 1 + n]         # slice of length n
            grad = 4.0 * n * corr                       # ∇ of the smoothed objective

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = 0.02 / (np.sqrt(v_hat) + eps) * m_hat
            a = a - step

            # ----- project back onto simplex -----
            a = _simplex_projection(a)

            # ----- occasional true‑objective check -----
            if (it + 1) % 100 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Beta {beta:.1f}] it={it+1:5d} new best {val:.6f}")

        # after each β‑stage, reset momentum and add a tiny random jitter
        m.fill(0.0)
        v.fill(0.0)
        if time.time() + 0.5 < deadline:   # keep some slack
            rng = np.random.default_rng(seed + int(time.time() * 1e6) % 2**32)
            jitter = rng.normal(scale=0.001, size=n)
            a = best_seq + jitter
            a = np.maximum(a, 0.0)
            a = _simplex_projection(a)

    # -----------------------------
    # Final simulated‑annealing fine‑tuning (if time permits)
    # -----------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 3] simulated‑annealing fine‑tuning")
        seq_final, val_final = _anneal_optimize(best_seq, deadline)
        if val_final < best_val:
            best_val = val_final
            best_seq = seq_final.copy()
            print(f"[Phase 3] improved to {val_final:.6f}")
        else:
            print("[Phase 3] no improvement from annealing")
    else:
        print("[Phase 3] skipped (time out)")

    # -----------------------------
    # Normalisation & export
    # -----------------------------
    best_seq = np.maximum(best_seq, 0.0)
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.ones(n, dtype=np.float64) / n
    else:
        best_seq = best_seq / total

    final_obj = _objective(best_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = best_seq.copy()
    return best_seq.tolist()
