def _adam_logspace_optimize(start_vec, n, deadline, max_iters=300000):

    m = np.zeros_like(v)
    v_hat = np.zeros_like(v)  # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    # temperature schedule
    temperature = 0.5
    temp_inc = 0.5
    temp_max = 300.0

    best_seq = None
    best_val = float('inf')
    conv_cache = None
    for it in range(1, max_iters+1):
        if time.time()>deadline: break
        a = _softmax(v)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        # smoothed weights
        shift = np.max(conv)
        w = np.exp(temperature * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0:
            continue
        w = w / w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]

        # gradient w.r.t logits
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_v
        v_hat = beta2 * v_hat + (1 - beta2) * (grad_v ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat_corr = v_hat / (1 - beta2**it)

        lr = 0.5 / (np.sqrt(it) + 1)   # can adapt
        v = v - lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        if temperature < temp_max:
            temperature = min(temperature + temp_inc, temp_max)

    return best_seq, best_val

"""
Improved search for a non‑negative step function that minimises

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm consists of three stages:

1. **Seed generation** – a rich collection of analytic families (uniform,
   triangular, cosine‑window, Gaussian, power‑law, beta, exponential,
   sparse Dirichlet, random permutations, mixed shapes) plus a number of
   random Dirichlet draws with many concentration values.  All seeds are
   ℓ₁‑normalised (Σaᵢ = 1).  The best few seeds (according to the cheap
   surrogate 2·n·max(conv)) are kept.

2. **Adam log‑space optimisation** – optimisation on the log‑softmax
   (probability simplex) using a smoothed max (log‑sum‑exp) surrogate.
   An Adam optimiser with a slowly increasing temperature (β) gives a
   stable, fast descent.  Several restarts are performed; the best result
   of each restart is recorded.

3. **Exact sub‑gradient descent** – after the smooth phase we switch to
   the true max‑convolution objective.  A classic sub‑gradient step is
   taken and the iterate is projected back onto the simplex with the
   Euclidean projection.  The step‑size decays like 1/√t.

4. **Stochastic polishing (simulated annealing)** – a final annealing
   loop repeatedly reshuffles the mass of a random subset of coordinates
   (size 2‑20) and occasionally accepts uphill moves according to a
   geometric cooling schedule.

All stages respect the global 1000 s deadline; the best sequence found
is returned as a plain ``list[float]`` and stored in the global name
``height_sequence_1`` for possible reuse.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution using a real FFT (O(N log N))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Analytic shape families (always ℓ₁‑normalised)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    left = i ** (alpha - 1.0)
    right = (n - i + 1.0) ** (beta - 1.0)
    a = left * right
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _exponential_decay(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i) (ℓ₁‑normalised)."""
    if lam <= 0.0:
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_permutation(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    perm = rng.permutation(len(vec))
    return vec[perm]


def _sparse_seed(n: int, m: int, rng: np.random.Generator) -> np.ndarray:
    """m non‑zero entries of equal weight (ℓ₁‑normalised)."""
    if m <= 0 or m > n:
        return _uniform(n)
    idx = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))

# Seed collection
# ----------------------------------------------------------------------


def _collect_seeds(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse set of ℓ₁‑normalised candidate vectors."""
    seeds: list[np.ndarray] = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_gaussian(n, sigma_frac=0.20))
    seeds.append(_gaussian(n, sigma_frac=0.30))
    seeds.append(_gaussian(n, sigma_frac=0.40))

    # power‑law family
    for p in np.linspace(0.0, 4.0, 13):
        seeds.append(_power(n, p))

    # beta family
    for a in np.linspace(0.3, 2.0, 5):
        for b in np.linspace(0.3, 4.0, 5):
            seeds.append(_beta_shape(n, a, b))

    # exponential decays
    for lam in (0.0002, 0.0005, 0.001, 0.002, 0.005):
        seeds.append(_exponential_decay(n, lam))

    # Tukey windows
    for alpha in (0.1, 0.3, 0.5, 0.7, 0.9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws – many concentration values
    for conc in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random sparse seeds (≈√n non‑zero entries)
    sqrt_n = int(np.sqrt(n))
    for mult in (0.8, 1.0, 1.2, 1.5):
        m = max(2, int(sqrt_n * mult))
        for _ in range(2):
            seeds.append(_sparse_seed(n, m, rng))

    # Random permutations of uniform (breaks any hidden symmetry)
    for _ in range(4):
        seeds.append(_random_permutation(_uniform(n), rng))

    # Mixtures of uniform with a low‑frequency random curve
    for mix_alpha in (0.2, 0.5, 0.8):
        curve = _lowfreq_curve(n, nf=4, rng=rng)
        seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        if avg.sum() > 0:
            seeds.append(avg / avg.sum())

    return seeds


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate a smooth non‑negative curve by summing a few random cosine
    components.  The result is ℓ₁‑normalised.
    """
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


# ----------------------------------------------------------------------
# Surrogate objective  (the quantity that the optimiser actually uses)
# ----------------------------------------------------------------------


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate: 2·n·max(conv(seq,seq))  (the denominator Σseq² is 1
    because all vectors are kept on the probability simplex).
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Adam optimisation on the log‑softmax parameters (smooth max surrogate)
# ----------------------------------------------------------------------


def _adam_logspace_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimiser on the logits of a softmax representation.
    A temperature β is used to smooth the max; β is increased slowly so
    the surrogate converges to the true max.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)            # logits (softmax gives simplex)
    m1 = np.zeros_like(v)                  # first moment
    m2 = np.zeros_like(v)                  # second moment

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # temperature schedule for the soft‑max over the convolution entries
    beta = 0.5            # start very smooth
    beta_inc = 0.5
    beta_max = 300.0

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        a = _softmax(v)                      # probability vector (Σa = 1)

        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # ----- smoothed max (log‑sum‑exp) -----
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ----- gradient of weighted sum of convolutions -----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)           # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]  # central n entries

        # gradient w.r.t. logits (softmax chain rule)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m1 = beta1 * m1 + (1 - beta1) * grad_v
        m2 = beta2 * m2 + (1 - beta2) * (grad_v ** 2)

        m1_hat = m1 / (1 - beta1 ** it)
        m2_hat = m2 / (1 - beta2 ** it)

        lr = 0.8 / (np.sqrt(it) + 1.0)            # decaying learning‑rate
        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps_adam)

        # ----- temperature increase (more faithful max) -----
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact sub‑gradient descent on the true max‑convolution objective
# ----------------------------------------------------------------------


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent with Euclidean projection onto the simplex.
    The step‑size follows 1/√t.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))

        cur_val = 2.0 * n * conv[max_idx]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 100_000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient for the active lag
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step‑size decrease ∝ 1/√t
        lr = 0.15 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated annealing polishing – more aggressive moves
# ----------------------------------------------------------------------


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_200_000,
    init_temp_factor: float = 0.03,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simulated annealing that repeatedly redistributes the mass of a random
    subset of coordinates (size 2‑20).  Larger subsets are allowed compared
    to the original implementation, providing a stronger ability to escape
    shallow local minima.
    """
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val

    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # exponential cooling (geometric)
        T = max(T * 0.9995, min_temp)

        # random subset size (2 … 20)
        k = rng.integers(2, 21)
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue

        # redistribute total mass using a Dirichlet draw of size k
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)

        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50_000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective within the allotted time budget.
    Returns the best sequence as a plain list of floats.
    """
    # reproducibility
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # safety margin

    # retrieve previous result if it exists
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------
    # Choose a length – 6000 offers a richer design space while still
    # staying comfortably within the time budget on a modern CPU.
    # ------------------------------------------------------------
    n = 6000

    # ------------------------------------------------------------
    # Stage 1 – generate a rich pool of seeds and keep the most
    # promising candidates (according to the cheap surrogate).
    # ------------------------------------------------------------
    print(f"\n--- Seed generation (length {n}) ---")
    seeds = _collect_seeds(n, rng, previous)
    print(f"Generated {len(seeds)} seeds.")

    evaluated = [(vec, _objective(vec)) for vec in seeds]
    evaluated.sort(key=lambda x: x[1])
    top_k = 6
    best_seeds = [vec for vec, _ in evaluated[:top_k]]
    print(f"Top {top_k} seed surrogate values:")
    for i, (_, val) in enumerate(evaluated[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective(global_best_seq)
    print(f"\n[Init] Best seed surrogate = {global_best_val:.6f}")

    # ------------------------------------------------------------
    # Stage 2 – Adam log‑space SGD (multiple restarts).
    # ------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        print(f"\n--- Adam log‑space SGD run {idx+1}/{len(best_seeds)} ---")
        seq_sgd, val_sgd = _adam_logspace_optimize(
            seed_vec, n, deadline, max_iters=800_000
        )
        if seq_sgd is not None and val_sgd < global_best_val:
            global_best_seq = seq_sgd.copy()
            global_best_val = val_sgd
            print(f"[overall‑SGD] new global best = {val_sgd:.6f}")

        # --------------------------------------------------------
        # Sub‑gradient projection – exact max‑convolution objective
        # --------------------------------------------------------
        seq_proj, val_proj = _subgradient_proj(
            global_best_seq, n, deadline, max_iters=3_000_000
        )
        if val_proj < global_best_val:
            global_best_seq = seq_proj.copy()
            global_best_val = val_proj
            print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------
    # Stage 3 – final stochastic polishing (simulated annealing)
    # ------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=2_000_000
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------
    # Sanity check with the official evaluator (optional)
    # ------------------------------------------------------------
    try:
        _ = evaluate_sequence(global_best_seq.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    print(
        f"\n=== Finished ===\nBest surrogate ≈ {global_best_val:.6f} "
        f"(length {len(global_best_seq)})"
    )

    # expose for possible reuse
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
