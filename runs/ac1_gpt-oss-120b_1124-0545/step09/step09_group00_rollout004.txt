        shift = np.max(conv)

        w_sum = w.sum()
        if w_sum == 0:
            continue
        w /= w_sum

        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)
        grad_a = 2.0 * full[n-1:n-1+n]

        # log gradient
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        vhat = b2 * vhat + (1 - b2) * (grad_v ** 2)
        m_corr = m / (1 - b1**it)
        vhat_corr = vhat / (1 - b2**it)
        step = lr * m_corr / (np.sqrt(vhat_corr) + eps_adam)
        v = v - step

        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)
    return best_seq, best_val

"""
Improved search for a non‑negative step function that minimizes

    2·n·max(conv(seq,seq)) / (sum(seq)²)

The algorithm builds a much richer seed pool (many analytic windows,
Dirichlet samples, low‑frequency random curves and mixtures), keeps the
best few seeds and then refines them with three stages:

1. **Log‑space Adam optimisation** – works on the log‑softmax
   representation of the vector, uses a log‑sum‑exp surrogate for the
   max‑convolution and an Adam‑style adaptive step size.

2. **Exact sub‑gradient descent** – operates on the true max‑convolution,
   projects after each step onto the probability simplex.

3. **Stochastic annealing** – a more aggressive annealing loop that
   reshuffles the mass of a random subset of coordinates, accepts uphill
   moves with a slowly‑cooling schedule and also performs pair‑wise
   transfers.

All stages respect the global deadline (default 1000 s).  The best
sequence found is stored in the global name ``height_sequence_1`` and
returned as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (used only for the Kaiser window)
# ----------------------------------------------------------------------
try:
    from scipy.signal import kaiser as _scipy_kaiser
except Exception:  # pragma: no cover
    _scipy_kaiser = None
try:
    from scipy.special import i0 as _scipy_i0
except Exception:  # pragma: no cover
    _scipy_i0 = None

# ----------------------------------------------------------------------
# Low‑level numeric utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """smallest power of two >= x"""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """full linear convolution using a real FFT (O(n log n))"""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _softmax(v: np.ndarray) -> np.ndarray:
    """numerically stable softmax"""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex"""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_surrogate(seq: np.ndarray) -> float:
    """
    Surrogate objective 2·n·max(conv(seq,seq)) assuming Σseq = 1.
    The returned value respects ordering of candidates; it does not
    perform the division by (sum(seq))².
    """
    n = len(seq)
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Analytic shape families (all return ℓ₁‑normalised vectors)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / c
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int, gamma: float = 1.0) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1)) ** gamma
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _gaussian(n: int, sigma_frac: float) -> np.ndarray:
    if sigma_frac <= 0.0:
        return _uniform(n)
    sigma = sigma_frac * n
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-((i - mu) ** 2) / (2.0 * sigma ** 2))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power(n: int, p: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    if alpha <= 0.0 or beta <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (alpha - 1.0)) * ((n - i + 1) ** (beta - 1.0))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _tukey_window(n: int, alpha: float) -> np.ndarray:
    if alpha <= 0.0:
        return _uniform(n)
    if alpha >= 1.0:
        i = np.arange(n, dtype=np.float64)
        a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
        return a / a.sum()
    a = np.zeros(n, dtype=np.float64)
    per = alpha * (n - 1) / 2.0
    for i in range(n):
        if i < per:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) - 1)))
        elif i <= (n - 1) * (1 - alpha / 2):
            a[i] = 1.0
        else:
            a[i] = 0.5 * (1 + np.cos(np.pi * (2 * i / (alpha * (n - 1)) -
                                            2 / alpha + 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0, a1, a2 = 0.42, 0.5, 0.08
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _nuttall_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a0 = 0.355768
    a1 = 0.487396
    a2 = 0.144232
    a3 = 0.012604
    a = (a0
         - a1 * np.cos(2 * np.pi * i / (n - 1))
         + a2 * np.cos(4 * np.pi * i / (n - 1))
         - a3 * np.cos(6 * np.pi * i / (n - 1)))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    # Use SciPy if available; otherwise fall back to a pure‑numpy
    # implementation based on the modified Bessel function I0.
    if _scipy_kaiser is not None:
        a = _scipy_kaiser(n, beta)
        s = a.sum()
        return a / s if s > 0 else _uniform(n)

    # fallback – use scipy.special.i0 if present, else a crude approximation
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    y = (i - alpha) / alpha
    if _scipy_i0 is not None:
        a = _scipy_i0(beta * np.sqrt(1 - y * y))
    else:
        # series approximation of I0 (first 5 terms)
        t = beta * np.sqrt(1 - y * y)
        a = 1.0 + (t ** 2) / 4.0 + (t ** 4) / 64.0 + (t ** 6) / 2304.0 + (t ** 8) / 147456.0
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _lowfreq_curve(n: int, num_freq: int, rng: np.random.Generator) -> np.ndarray:
    """Sum a few cosine components with random amplitude/phase; result is ℓ₁‑normalised."""
    t = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for k in range(1, num_freq + 1):
        phase = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.exponential(scale=1.0)
        a += amp * np.cos(2.0 * np.pi * k * t / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return _uniform(n)
    return a / s


def _mix_shapes(a: np.ndarray, b: np.ndarray, alpha: float) -> np.ndarray:
    """Convex mixture α·b + (1‑α)·a, re‑normalised."""
    mixed = (1.0 - alpha) * a + alpha * b
    s = mixed.sum()
    return mixed / s if s > 0 else _uniform(len(a))


# ----------------------------------------------------------------------
# Seed collection – richer than the previous version
# ----------------------------------------------------------------------
def _collect_seeds_extended(
    n: int,
    rng: np.random.Generator,
    previous: np.ndarray | None,
) -> list[np.ndarray]:
    """Assemble a diverse pool of ℓ₁‑normalised vectors."""
    seeds: list[np.ndarray] = []

    # Deterministic analytic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n, gamma=1.0))
    seeds.append(_sine_window(n, gamma=2.0))
    seeds.append(_sine_window(n, gamma=3.0))
    seeds.append(_hamming_window(n))
    seeds.append(_blackman_window(n))
    seeds.append(_nuttall_window(n))
    for beta in (2, 4, 6, 8, 10, 12):
        seeds.append(_kaiser_window(n, beta))
    for sigma in np.linspace(0.15, 0.45, 7):
        seeds.append(_gaussian(n, sigma))

    # Power‑law family
    for p in np.linspace(0.0, 5.0, 21):
        seeds.append(_power(n, p))

    # Beta family – systematic grid
    alphas = np.linspace(0.2, 5.0, 13)
    betas = np.linspace(0.2, 5.0, 13)
    for a in alphas:
        for b in betas:
            seeds.append(_beta_shape(n, a, b))

    # Tukey windows
    for alpha in np.linspace(0.1, 0.9, 9):
        seeds.append(_tukey_window(n, alpha))

    # Random Dirichlet draws (different concentration parameters)
    for conc in (0.05, 0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            seeds.append(_random_dirichlet(n, conc, rng))

    # Random exponential (heavy‑tailed)
    for _ in range(5):
        seeds.append(_random_exponential(n, rng))

    # Low‑frequency random curves
    for nf in (2, 3, 4, 5):
        for _ in range(3):
            seeds.append(_lowfreq_curve(n, nf, rng))

    # Mixtures of uniform with a low‑frequency curve
    for mix_alpha in (0.2, 0.5, 0.8):
        for nf in (2, 4):
            curve = _lowfreq_curve(n, nf, rng)
            seeds.append(_mix_shapes(_uniform(n), curve, mix_alpha))

    # Previous solution (if any)
    if previous is not None:
        arr = np.asarray(previous, dtype=np.float64)
        if arr.size == n and arr.sum() > 0:
            seeds.append(arr / arr.sum())

    # Ensure all vectors are normalised (some generators may produce zeros)
    cleaned = []
    for v in seeds:
        v = np.asarray(v, dtype=np.float64)
        s = v.sum()
        if s > 0:
            cleaned.append(v / s)
        else:
            cleaned.append(_uniform(n))
    return cleaned


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _logspace_adam_optimize(
    start_vec: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray | None, float]:
    """
    Adam optimisation on the log‑softmax parametrisation.
    Minimises the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector (ℓ₁‑normalised) found and its surrogate value.
    """
    eps = 1e-30
    v = np.log(start_vec + eps)                 # logits
    m = np.zeros_like(v)                        # first moment
    v_sq = np.zeros_like(v)                     # second moment

    b1, b2 = 0.9, 0.999
    eps_adam = 1e-8

    beta = 0.5               # smoothing temperature
    beta_inc = 0.1
    beta_max = 200.0

    lr0 = 1.0                # initial learning rate (log‑space)
    best_seq = None
    best_val = float("inf")

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # current probability vector on simplex
        a = _softmax(v)

        # surrogate objective (no scaling needed for ordering)
        conv = _conv_fft(a, a)
        cur_val = 2.0 * n * np.max(conv)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 2000 == 0:
                print(f"[LogAdam] it={it:6d} best={cur_val:.6f}")

        # smoothed max via log‑sum‑exp
        shift = np.max(conv)
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # gradient of ∑ w_k·conv_k w.r.t. a
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)                 # length 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]        # central n entries

        # convert to gradient on logits (log‑softmax)
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        # Adam update (learning‑rate decays like 1/√t)
        lr = lr0 / np.sqrt(it)
        m = b1 * m + (1 - b1) * grad_v
        v_sq = b2 * v_sq + (1 - b2) * (grad_v ** 2)

        m_hat = m / (1 - b1 ** it)
        v_hat = v_sq / (1 - b2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        v = v - step

        # temperature schedule
        if beta < beta_max:
            beta = min(beta + beta_inc, beta_max)

    return best_seq, best_val


def _subgradient_proj(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 2_500_000,
) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient descent on the true max‑convolution objective.
    Uses a decreasing step‑size and Euclidean projection onto the simplex.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective_surrogate(a)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        max_idx = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[max_idx]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            if it % 25000 == 0:
                print(f"[SubProj] it={it:7d} best={cur_val:.6f}")

        # sub‑gradient w.r.t. the active index
        g = np.zeros_like(a)
        i_start = max(0, max_idx - (n - 1))
        i_end = min(n - 1, max_idx)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = max_idx - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        # step size (Polyak‑style constant / sqrt(t))
        lr = 0.20 / np.sqrt(it)
        a = a - lr * g
        a = _proj_simplex(a)

    return best_seq, best_val


def _simulated_annealing(
    start_seq: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int = 1_500_000,
    init_temp_factor: float = 0.06,
    min_temp: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """Aggressive annealing that redistributes mass on random subsets."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective_surrogate(cur)
    best_seq = cur.copy()
    best_val = cur_val

    # initial temperature proportional to current objective
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # geometric cooling (very slow)
        T = max(T * 0.9999, min_temp)

        # Two move types – “subset reshuffle” or “pair transfer”
        if rng.random() < 0.7:          # reshape a random subset
            k = rng.integers(2, 12)          # size 2 … 11
            idx = rng.choice(n, size=k, replace=False)
            total = cur[idx].sum()
            if total <= 0.0:
                continue
            new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
            new = cur.copy()
            new[idx] = new_alloc
        else:                           # transfer a small chunk between two positions
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * min(cur[i], 0.05)  # move at most 5 % of source mass
            if eps <= 0.0:
                continue
            new = cur.copy()
            new[i] -= eps
            new[j] += eps

        new_val = _objective_surrogate(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < math.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                if it % 50000 == 0:
                    print(f"[SA] it={it:7d} best={new_val:.6f}")

    return best_seq, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    ``evaluate_sequence`` objective within the given time budget.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # keep a tiny safety margin (the evaluator does a few cheap checks)
    deadline = start_time + budget_s - 2.0

    # ------------------------------------------------------------------
    # Length selection – 4000 gives a good trade‑off between expressivity
    # and speed.  If the remaining time is still large after the first
    # stage we will later try a slightly larger n.
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – seed generation and cheap surrogate ranking
    # ------------------------------------------------------------------
    print(f"\n--- Seed generation (n={n}) ---")
    previous = globals().get("height_sequence_1")
    prev_arr = None
    if previous is not None:
        try:
            prev_arr = np.asarray(previous, dtype=np.float64)
        except Exception:
            prev_arr = None

    seeds = _collect_seeds_extended(n, rng, prev_arr)
    print(f"Generated {len(seeds)} candidate seeds.")

    # evaluate surrogate (very cheap) and keep the top few
    evals = [(vec, _objective_surrogate(vec)) for vec in seeds]
    evals.sort(key=lambda x: x[1])
    top_k = 12
    best_seeds = [vec for vec, _ in evals[:top_k]]

    print(f"Top {top_k} surrogate values:")
    for i, (_, val) in enumerate(evals[:top_k]):
        print(f"  #{i+1}: {val:.6f}")

    global_best_seq = best_seeds[0].copy()
    global_best_val = _objective_surrogate(global_best_seq)
    print(f"\n[Init] Best surrogate after seeding = {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – log‑space Adam optimisation (multiple restarts)
    # ------------------------------------------------------------------
    for idx, seed_vec in enumerate(best_seeds):
        if time.time() > deadline:
            break
        remaining = deadline - time.time()
        # allocate roughly 0.20 · remaining seconds to this run
        max_iters = int(min(400_000, remaining * 0.12 * 1e5))
        print(f"\n--- Log‑space Adam run {idx+1}/{len(best_seeds)} (max_iters={max_iters}) ---")
        seq_sgd, val_sgd = _logspace_adam_optimize(
            seed_vec, n, deadline, max_iters=max_iters
        )
        if seq_sgd is not None and val_sgd < global_best_val:
            global_best_seq = seq_sgd.copy()
            global_best_val = val_sgd
            print(f"[overall‑Adam] new global best = {val_sgd:.6f}")

        # --------------------------------------------------------------
        # Sub‑gradient projection (exact max) – keep a tight budget
        # --------------------------------------------------------------
        remaining = deadline - time.time()
        max_iters_sub = int(min(2_500_000, remaining * 0.45 * 1e5))
        if max_iters_sub > 0:
            seq_proj, val_proj = _subgradient_proj(
                global_best_seq, n, deadline, max_iters=max_iters_sub
            )
            if val_proj < global_best_val:
                global_best_seq = seq_proj.copy()
                global_best_val = val_proj
                print(f"[overall‑SubProj] new global best = {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – simulated annealing polishing
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated annealing polishing ---")
        remaining = deadline - time.time()
        max_iters_sa = int(min(1_500_000, remaining * 1e5))
        seq_sa, val_sa = _simulated_annealing(
            global_best_seq, n, deadline, max_iters=max_iters_sa
        )
        if val_sa < global_best_val:
            global_best_seq = seq_sa.copy()
            global_best_val = val_sa
            print(f"[overall‑SA] final global best = {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Optional – final sanity check with the official evaluator
    # ------------------------------------------------------------------
    try:
        true_val = evaluate_sequence(global_best_seq.tolist())
        print(f"\n=== Finished ===\nSurrogate ≈ {global_best_val:.6f}, true evaluate_sequence = {true_val:.6f}")
    except Exception as exc:  # pragma: no cover
        print("Warning: evaluate_sequence raised an exception:", exc)
        true_val = None

    # expose for possible reuse in a subsequent run
    globals()["height_sequence_1"] = global_best_seq.tolist()
    return global_best_seq.tolist()
