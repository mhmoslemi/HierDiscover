def propose_candidate(seed=42, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.
    rng = np.random.default_rng(seed+1)

    # Determine n
    n = 8000  # may set bigger but compute within time

    # Determine required support size:
    target_rel = 0.99
    target_m = int(0.99 * math.sqrt(8 * n))
    # Use low-collision set search
    t_set_deadline = start_time + 0.6 * (deadline - start_time) # allocate 60% time
    lowcoll_idx = _search_low_collision_set(n=n,
                                            target_m=target_m,
                                            max_ordered=4,
                                            rng=rng,
                                            deadline=t_set_deadline)
    if lowcoll_idx.size == 0:
        # fallback: uniform distribution
        seq = np.full(n, 1.0/n)
    else:
        seq = np.zeros(n)
        seq[lowcoll_idx] = 1.0/lowcoll_idx.size
        seq = seq / seq.sum()
    # Evaluate
    best_seq = seq.copy()
    best_val = _objective(best_seq)
    print(f"[Result] lowcollision support size {lowcoll_idx.size}, obj={best_val:.6f}")

    # If best_val > 1.503, attempt gradient improvements:
    if best_val > 1.503 and time.time() < deadline - 2:
        # Use Adam optimization to perhaps smooth the distribution and lower max.
        # Use smaller number of iterations (20000) within remaining time.
        seq_opt, val_opt = _adam_optimize(best_seq, n, deadline, max_iters=20000)
        if seq_opt is not None and val_opt < best_val:
            best_seq = seq_opt; best_val = val_opt
            print(f"[Adam refined] val={best_val:.6f}")

    # Optional subgradient refine
    if best_val > 1.503 and time.time() < deadline - 2:
        seq_opt2, val_opt2 = _subgradient_refine(best_seq, n, deadline, max_iters=200000)
        if seq_opt2 is not None and val_opt2 < best_val:
            best_seq = seq_opt2; best_val = val_opt2
            print(f"[Subgrad refined] val={best_val:.6f}")

    # global best
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = np.full(n, 1.0/n)
    else:
        final_seq = final_seq / total

    print(f"\n=== Finished ===\nBest objective ≈ {best_val:.6f} (length {n})")

    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()

Propose a step‑function sequence by constructing a large
low‑collision support set (ordered pair‑sum appears at most 4 times)
and using a uniform distribution on that support.  The low‑collision
search occupies the majority of the ~1000 s budget; if the resulting
objective is still above the target a few gradient‑based refinements
(Adam, sub‑gradient) are applied.

import time, math, numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv

def _objective(seq: np.ndarray) -> float:
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)

def _search_low_collision_set(
    n: int,
    target_m: int,
    max_ordered: int,
    rng: np.random.Generator,
    deadline: float,
) -> np.ndarray:
    """Greedy construction of a set S⊂{0,…,n‑1} with ≤max_ordered ordered
    pair‑sum collisions. Returns the index array of the best mask found."""
    best_mask = None
    best_size = 0
    attempts = 0
    while time.time() < deadline:
        attempts += 1
        mask = np.zeros(n, dtype=bool)
        sum_counts = np.zeros(2 * n - 1, dtype=np.uint8)
        order = rng.permutation(n)
        for cand in order:
            # check self‑pair
            if sum_counts[2 * cand] + 1 > max_ordered:
                continue
            existing = np.nonzero(mask)[0]
            if existing.size:
                sums = cand + existing
                # each ordered pair (cand, e) and (e, cand) contributes 2
                if np.any(sum_counts[sums] + 2 > max_ordered):
                    continue
            # accept
            mask[cand] = True
            sum_counts[2 * cand] += 1
            if existing.size:
                sum_counts[sums] += 2
            if mask.sum() >= target_m:
                break
        cur = mask.sum()
        if cur > best_size:
            best_size = cur
            best_mask = mask.copy()
            print(f"[LowColl] attempt {attempts:5d} size {cur}")
        if best_size >= target_m:
            break
    if best_mask is None:
        return np.array([], dtype=int)
    return np.where(best_mask)[0]

    e = np.exp(v - m)
    return e / e.sum()

def _adam_optimize(seed: np.ndarray, n: int, deadline: float, max_iters: int = 20000):
    v = np.log(seed + 1e-30)
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)
    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    beta = 1.5
    beta_mul = 1.007
    best_seq = None
    best_val = float("inf")
    t = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")
        # smoothed max weights
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum
        # gradient of Σ w_k (a*a)_k
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

The algorithm works entirely with NumPy and requires no external files.

# ----------------------------------------------------------------------
# FFT‑based convolution (exact, real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution using a real FFT (full mode).  Returns an array of
    length len(a)+len(b)-1.
    """
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence but without the safety checks)
# ----------------------------------------------------------------------
def _objective(seq: np.ndarray) -> float:
    """2·n·max(conv(seq,seq)) / (sum(seq)²).  Assumes seq is non‑negative."""
    s = seq.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Low‑collision set generator (ordered sums ≤ max_ordered)
# ----------------------------------------------------------------------
def _search_low_collision_set(

    Returns the index array of the best set found.

        if best_size >= target_m:
            break

# Softmax (used by Adam)
# ----------------------------------------------------------------------
def _softmax(v: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(v)
    e = np.exp(v - m)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimisation on a smoothed‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(seed: np.ndarray, n: int, deadline: float,
                   max_iters: int = 20000):
    """
    Adam on a log‑softmax parametrisation of the vector.
    Returns the best sequence seen and its true objective.
    """
    v = np.log(seed + 1e-30)                # logits
    m1 = np.zeros_like(v)
    m2 = np.zeros_like(v)

    lr0 = 0.7
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    beta = 1.5               # temperature for the soft‑max over the convolution
    beta_mul = 1.007         # slowly tighten the approximation

    best_seq = None
    best_val = float("inf")
    t = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        a = _softmax(v)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={it:5d} best={cur_val:.6f}")

        # ---- smoothed‑max weights ----
        beta *= beta_mul
        conv = _conv_fft(a, a)
        shift = conv.max()
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w /= w_sum

        # ---- gradient of Σ w_k·(a*a)_k ----
        rev_a = a[::-1]
        full = _conv_fft(rev_a, w)               # length = 3n‑2
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- chain rule through soft‑max ----
        dot = np.dot(grad_a, a)
        grad_v = a * (grad_a - dot)

        m1 = beta1 * m1 + (1.0 - beta1) * grad_v
        m2 = beta2 * m2 + (1.0 - beta2) * (grad_v ** 2)

        v = v - lr * m1_hat / (np.sqrt(m2_hat) + eps)
