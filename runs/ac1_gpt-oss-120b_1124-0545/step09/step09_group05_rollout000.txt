"""
non‑negative step‑function optimiser – version 3 (evolutionary + local)

This optimiser builds a modest population of candidate step‑functions,
evolves them with a simple (μ+λ) evolutionary strategy that respects the
simplex constraint, and periodically refines promising individuals with
exact sub‑gradient / Euclidean‑projected descent.  The final best vector
is polished by a short simulated‑annealing run.  The whole procedure
honours a user‑supplied time budget (default 1000 s) and returns the
best sequence found as a plain list of floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution via real FFT (O(n log n))."""
    L = len(x) + len(y) - 1
    N = _next_pow2(L)
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """
    Surrogate objective used during the search.
    ``seq`` is assumed non‑negative and already normalised to sum = 1.
    Returns ``2·n·max(conv(seq,seq))`` – exactly the value that
    ``evaluate_sequence`` would produce when the sum equals 1.
    """
    n = seq.shape[0]
    conv = _conv_fft(seq, seq)
    return 2.0 * n * np.max(conv)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Analytic shape families (all normalised to sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = 1.0 - np.abs(i - centre) / centre
    a = np.maximum(a, 0.0)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _geom_decrease(n: int, r: float) -> np.ndarray:
    """Monotone geometric decay a_i = r**i, 0<r<1."""
    if not (0.0 < r < 1.0):
        return _uniform(n)
    i = np.arange(n, dtype=np.float64)
    a = np.power(r, i)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _power_decrease(n: int, p: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^(-p)."""
    if p <= 0.0:
        return _uniform(n)
    i = np.arange(1, n + 1, dtype=np.float64)
    a = np.power(i, -p)
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _sorted_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw sorted in descending order (strongly skewed)."""
    a = rng.dirichlet(np.full(n, conc)).astype(np.float64)
    a.sort()
    a = a[::-1]  # descending
    s = a.sum()
    return a / s if s > 0 else _uniform(n)


def _random_blocks(n: int, max_blocks: int, rng: np.random.Generator) -> np.ndarray:
    """
    Simple block pattern: k equal‑length blocks placed evenly.
    Mass inside each block is uniform.
    """
    k = rng.integers(2, max_blocks + 1)
    # each block must have at least 2 entries
    max_len = max(2, n // (2 * k))
    w = rng.integers(2, max_len + 1)
    total_len = k * w
    if total_len > n:
        w = n // k
        total_len = k * w
    gap = (n - total_len) // (k + 1)
    seq = np.zeros(n, dtype=np.float64)
    pos = gap
    mass_per_block = 1.0 / k
    for _ in range(k):
        seq[pos:pos + w] = mass_per_block / w
        pos += w + gap
    return seq / seq.sum()


def _random_dirichlet(n: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, conc)).astype(np.float64)


def _random_sparse(n: int, k: int, conc: float, rng: np.random.Generator) -> np.ndarray:
    """k‑sparse Dirichlet vector."""
    if k >= n:
        return _random_dirichlet(n, conc, rng)
    idx = rng.choice(n, size=k, replace=False)
    vals = rng.dirichlet(np.full(k, conc))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = vals
    return a / a.sum()

def _subgradient_refine(start_seq: np.ndarray,
                        n: int,
                        deadline: float,
                        max_iters: int = 150000) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑convolution.
    Returns the best sequence seen and its surrogate objective.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    eps = 1e-30
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k]

        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()

        # sub‑gradient of the active entry
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]

        lr = 0.7 / np.sqrt(it)
        log_a = np.log(a + eps) - lr * g
        log_a -= np.max(log_a)          # stabilise
        a = np.exp(log_a)
        s = a.sum()
        if s == 0.0:
            a = _uniform(n)
        else:
            a /= s
    return best_seq, best_val


def _euc_proj_subgradient(start_seq: np.ndarray,
                          n: int,
                          deadline: float,
                          max_iters: int = 120000) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the active convolution entry with Euclidean
    projection onto the simplex after each step.
    """
    a = start_seq.copy()
    best_seq = a.copy()
    best_val = _objective(a)
    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        conv = _conv_fft(a, a)
        k = int(np.argmax(conv))
        cur_val = 2.0 * n * conv[k]
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
        g = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        idx_i = np.arange(i_start, i_end + 1, dtype=int)
        rev_idx = k - idx_i
        g[idx_i] = 2.0 * a[rev_idx]
        lr = 0.6 / np.sqrt(it)
        a_tmp = a - lr * g
        a = _project_simplex(a_tmp)
    return best_seq, best_val


def _simulated_annealing(start_seq: np.ndarray,
                         n: int,
                         deadline: float,
                         max_iters: int = 300000,
                         init_temp_factor: float = 0.07,
                         min_temp: float = 1e-12) -> tuple[np.ndarray, float]:
    """Simple SA that redistributes mass among a few random coordinates."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best_seq = cur.copy()
    best_val = cur_val
    T = max(init_temp_factor * cur_val, 1e-12)

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        T = max(T * 0.9995, min_temp)
        k = rng.integers(3, 8)                     # number of coordinates to touch
        idx = rng.choice(n, size=k, replace=False)
        total = cur[idx].sum()
        if total <= 0.0:
            continue
        new_alloc = rng.dirichlet(np.full(k, 1.0)) * total
        new = cur.copy()
        new[idx] = new_alloc
        new_val = _objective(new)
        delta = new_val - cur_val
        if delta < 0.0 or rng.random() < np.exp(-delta / T):
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_seq = new.copy()
                print(f"[SA] it={it:7d} best={new_val:.6f}")
    return best_seq, best_val

def _generate_initial_population(n: int,
                                 rng: np.random.Generator,
                                 previous: np.ndarray | None,
                                 popsize: int = 30) -> list[np.ndarray]:
    """
    Produce a diverse set of normalised candidate vectors.
    The goal is to fill ``popsize`` entries (duplicates are okay).
    """
    seeds = []

    # deterministic families
    seeds.append(_uniform(n))
    seeds.append(_triangular(n))
    seeds.append(_sine_window(n))

    # geometric grid (fine)
    for r in np.linspace(0.70, 0.98, 9):
        seeds.append(_geom_decrease(n, r))

    # power‑law grid
    for p in np.arange(0.3, 3.01, 0.4):
        seeds.append(_power_decrease(n, p))

    # sorted Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0):
        for _ in range(2):
            seeds.append(_sorted_dirichlet(n, conc, rng))

    # block patterns
    for max_blocks in (3, 5, 7):
        seeds.append(_random_blocks(n, max_blocks, rng))

    # random Dirichlet draws
    for conc in (0.05, 0.20, 0.50, 1.0, 5.0):
        for _ in range(2):
            seeds.append(_random_dirichlet(n, conc, rng))

    # sparse patterns
    for k in (5, 10, 20, 40):
        for conc in (0.20, 0.50):
            seeds.append(_random_sparse(n, k, conc, rng))

    # previous (if any)
    if previous is not None and previous.shape == (n,):
        avg = np.asarray(previous, dtype=np.float64)
        s = avg.sum()
        if s > 0:
            seeds.append(avg / s)

    # Shuffle and cut to the required population size
    rng.shuffle(seeds)
    if len(seeds) >= popsize:
        return seeds[:popsize]
    # pad with uniform if insufficient
    while len(seeds) < popsize:
        seeds.append(_uniform(n))
    return seeds

# Evolutionary optimisation core
# ----------------------------------------------------------------------


def _evolve_population(pop: list[np.ndarray],
                       n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       max_iters: int = 200000) -> tuple[np.ndarray, float]:
    """
    Simple (μ+λ) evolutionary loop.
    ``pop`` is the initial population (list of numpy arrays, already normalised).
    Returns the best vector found and its objective.
    """
    # Evaluate initial population
    scores = [_objective(v) for v in pop]
    best_idx = int(np.argmin(scores))
    best_vec = pop[best_idx].copy()
    best_val = scores[best_idx]

    # Hyper‑parameters
    sigma = 0.07                     # initial mutation strength
    sigma_decay = 0.9995
    min_sigma = 1e-5
    elite = max(2, len(pop) // 5)    # number of elites kept for parent selection

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # 1. parent selection – pick uniformly from the current elite set
        parent_idx = rng.integers(0, elite)
        parent = pop[parent_idx]

        # 2. mutation
        child = parent + sigma * rng.normal(size=n)
        child = _project_simplex(child)

        # 3. evaluation
        child_val = _objective(child)

        # 4. replacement – keep the population sorted by fitness
        # (smaller objective is better)
        worst_idx = int(np.argmax(scores))
        if child_val < scores[worst_idx]:
            pop[worst_idx] = child
            scores[worst_idx] = child_val
            # update elite set boundary if necessary
            # (no explicit re‑sorting – we rely on periodic re‑ordering)
            if child_val < best_val:
                best_val = child_val
                best_vec = child.copy()
                print(f"[EA] it={it:7d} new global best = {best_val:.6f}")

        # 5. adapt mutation strength (simple exponential decay)
        sigma = max(min_sigma, sigma * sigma_decay)

        # 6. occasional local polishing of the current best
        if it % 20000 == 0:
            # run a short sub‑gradient refinement (few thousand steps)
            limit = time.time() + min(5.0, deadline - time.time())
            refined, refined_val = _subgradient_refine(best_vec, n, limit, max_iters=5000)
            if refined_val < best_val:
                best_val = refined_val
                best_vec = refined.copy()
                print(f"[EA‑refine] it={it:7d} improved to {best_val:.6f}")

    # final re‑ordering of the population (not strictly needed)
    return best_vec, best_val

def propose_candidate(seed: int = 123,
                     budget_s: int = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function with minimal ``evaluate_sequence``
    value, respecting the given time budget.
    Returns the best sequence as a list of Python floats.
    """
    # ------------------------------------------------------------------
    # Setup reproducibility and timing
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a small safety margin (5 s)
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load previous solution if available
    # ------------------------------------------------------------------
    previous = globals().get("height_sequence_1")
    if previous is not None:
        try:
            previous = np.asarray(previous, dtype=np.float64)
        except Exception:
            previous = None
    else:
        previous = None

    # ------------------------------------------------------------------
    # Choose a length – 4000 works well and fits comfortably into time budget
    # ------------------------------------------------------------------
    n = 4000

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse initial population
    # ------------------------------------------------------------------
    print(f"\n--- Initial population (n={n}) ---")
    pop = _generate_initial_population(n, rng, previous, popsize=30)
    print(f"Population size: {len(pop)}")

    # ------------------------------------------------------------------
    # Stage 2 – evolutionary optimisation
    # ------------------------------------------------------------------
    print("\n--- Evolutionary optimisation ---")
    best_vec, best_val = _evolve_population(pop, n, deadline, rng,
                                            max_iters=300000)

    # ------------------------------------------------------------------
    # Stage 3 – intensive local polishing of the best candidate
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Sub‑gradient refinement (global) ---")
        limit = time.time() + min(15.0, deadline - time.time())
        best_vec, best_val = _subgradient_refine(best_vec, n, limit,
                                                max_iters=200000)
        print(f"[Polish‑SubGrad] best = {best_val:.6f}")

    if time.time() < deadline:
        print("\n--- Euclidean‑projected gradient descent (global) ---")
        limit = time.time() + min(10.0, deadline - time.time())
        best_vec, best_val = _euc_proj_subgradient(best_vec, n, limit,
                                                   max_iters=150000)
        print(f"[Polish‑EucProj] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated‑annealing polishing (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("\n--- Simulated‑annealing polishing ---")
        limit = time.time() + min(12.0, deadline - time.time())
        best_vec, best_val = _simulated_annealing(best_vec, n, limit,
                                                  max_iters=400000,
                                                  init_temp_factor=0.07)
        print(f"[Polish‑SA] best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (invoke evaluate_sequence to ensure validity)
    # ------------------------------------------------------------------
    try:
        evaluate_sequence(best_vec.tolist())
    except Exception as exc:
        print("Warning: evaluate_sequence raised an exception:", exc)

    # ------------------------------------------------------------------
    # Report and expose the result for possible reuse in subsequent runs
    # ------------------------------------------------------------------
    print(
        f"\n=== Finished ===\nBest surrogate objective ≈ {best_val:.6f} (length {n})"
    )
    globals()["height_sequence_1"] = best_vec.tolist()
    return best_vec.tolist()
