    a = start_seq.copy()

        shift = max_conv

        p = exp_terms / exp_terms.sum()
        # gradient

"""
Propose a step function with a low value of
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of four phases:

1. **Seed generation** – many cheap random sparse candidates are 
   generated (random support size, Dirichlet‐distributed masses).  
   The best few (default 5) are kept.

2. **Smooth exponential‑gradient optimisation** – each kept seed is
   refined by minimising a *smoothed* version of the max‑convolution
   (log‑sum‑exp over the convolution values).  This yields a smooth
   gradient that can be used with an Adam‑style mirror‑descent update
   on the log‑variables (keeps the iterate on the simplex).

3. **Exact sub‑gradient (mirror descent)** – a short run of the
   original non‑smooth sub‑gradient method from the best point found
   so far refines the solution.

4. **Simulated annealing fine‑tuning** – a tiny Metropolis–Hastings
   walk finishes the optimisation.

All phases respect a global 1000 s budget, and the best vector is stored
in the global variable ``height_sequence_1`` for reuse in later runs.
"""

import time
import math
import heapq
import numpy as np

# optional cvxpy – used only for the LP refinement (as in the previous code)

# ----------------------------------------------------------------------
# Convolution utilities (exact, using FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using FFT (full mode)."""
    L = a.size + b.size - 1
    N = _next_pow2(L)
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fa * fb, n=N)[:L]
    return conv


def _objective(seq: np.ndarray) -> float:
    """Exact objective 2·n·max(conv(seq,seq)) / (sum(seq)**2)."""
    s = seq.sum()
    if s < 1e-12:
        return float("inf")
    conv = _conv_fft(seq, seq)
    return 2.0 * seq.size * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple seed families (return a probability vector on the simplex)
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.ones(n, dtype=np.float64) / n


def _random_softmax(n: int, rng: np.random.Generator) -> np.ndarray:
    logits = rng.standard_normal(n)
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def _sidon_like_seed(n: int, m: int) -> np.ndarray:
    """Deterministic Sidon‑like support using triangular numbers."""
    idx = np.mod(np.arange(m) * (np.arange(m) + 1) // 2, n)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / m
    return a / a.sum()


def _greedy_sidon_set(n: int, target_m: int) -> np.ndarray:
    """Greedy construction of a B2 (Sidon) set of size ≈target_m."""
    S = []
    sums = set()
    for cand in range(n):
        if len(S) >= target_m:
            break
        ok = True
        # collisions with existing sums?
        for p in S:
            if (cand + p) in sums:
                ok = False
                break
        if ok and (2 * cand) in sums:
            ok = False
        if ok:
            for p in S:
                sums.add(cand + p)
            sums.add(2 * cand)
            S.append(cand)
    return np.array(S, dtype=int)

# Exact exponentiated‑gradient sub‑gradient descent (non‑smooth)
# ----------------------------------------------------------------------
def _subgradient_refine(start_seq: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the true max‑convolution objective."""
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    rel_tol = 1e-12
    it = 0
    while time.time() < deadline:
        it += 1
        conv = _conv_fft(a, a)
        max_val = conv.max()
        active = np.where(conv >= max_val - rel_tol)[0]

        # sub‑gradient: average over all active constraints
        w = np.zeros(2 * n - 1, dtype=np.float64)
        if active.size:
            w[active] = 1.0 / active.size
            rev_a = a[::-1]
            full = _conv_fft(rev_a, w)
            g = 2.0 * full[n - 1:n - 1 + n]
        else:
            g = np.zeros_like(a)

        lr = 0.9 / math.sqrt(it)
        log_a = np.log(a + 1e-30) - lr * g
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[SubGrad] it={it:8d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Smooth (log‑sum‑exp) gradient optimiser using Adam updates
# ----------------------------------------------------------------------
def _smooth_adam_opt(start_seq: np.ndarray,
                     deadline: float,
                     gamma: float = 8.0,
                     beta1: float = 0.9,
                     beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """
    Minimise a smoothed version of the max‑convolution using an
    Adam‑style mirror descent on the log‑variables (exponentiated
    gradient).  The smoothing is
        M_γ = (1/γ)·log Σ exp(γ·conv).
    """
    a = start_seq.copy()
    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    m1 = np.zeros_like(a)
    m2 = np.zeros_like(a)
    t = 0

    while time.time() < deadline:
        t += 1
        conv = _conv_fft(a, a)
        max_c = conv.max()
        # log‑sum‑exp over the whole convolution vector
        shifted = conv - max_c  # improve numerical stability
        exp_term = np.exp(gamma * shifted)
        p = exp_term / exp_term.sum()  # soft‑max weights over all indices

        # gradient of smoothed max (mirror‑descent on the simplex)
        rev_a = a[::-1]
        full = _conv_fft(rev_a, p)
        g = 2.0 * full[n - 1:n - 1 + n]  # same form as exact sub‑gradient

        # Adam moments on the *log* variables
        m1 = beta1 * m1 + (1 - beta1) * g
        m2 = beta2 * m2 + (1 - beta2) * (g * g)

        m1_hat = m1 / (1 - beta1 ** t)
        m2_hat = m2 / (1 - beta2 ** t)

        # learning‑rate schedule (a small constant base)
        lr = 0.2 / (np.sqrt(t) + 1e-8)

        log_a = np.log(a + 1e-30) - lr * m1_hat / (np.sqrt(m2_hat) + 1e-8)
        log_a -= np.max(log_a)
        a = np.exp(log_a)
        a /= a.sum()

        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[Adam] it={t:7d} best={cur_val:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Linearised LP refinement (optional, needs cvxpy)
# ----------------------------------------------------------------------
def _lp_linearised_refine(a: np.ndarray,
                         deadline: float,
                         max_refine: int = 8,
                         eps_active: float = 1e-9) -> tuple[np.ndarray, float]:
    """Perform a few LP refinements where the quadratic constraints are linearised."""
    if not CVXPY_AVAILABLE:
        return a, _objective(a)

    n = a.size
    best_seq = a.copy()
    best_val = _objective(a)

    x_var = cp.Variable(n)
    M_var = cp.Variable()
    constraints = [x_var >= 0, cp.sum(x_var) == 1]

    for it in range(max_refine):
        if time.time() > deadline - 0.3:
            break

        conv = _conv_fft(a, a)
        M = conv.max()
        active = np.where(conv >= M - eps_active)[0]
        if active.size == 0:
            break

        # Build the linearised constraints
        A = np.zeros((active.size, n), dtype=np.float64)
        for r, k in enumerate(active):
            i_low = max(0, k - (n - 1))
            i_high = min(n - 1, k)
            i_range = np.arange(i_low, i_high + 1, dtype=int)
            A[r, i_range] = a[k - i_range]

        lin_constraints = [2 * (A @ x_var) <= M_var + conv[active]]
        prob = cp.Problem(cp.Minimize(M_var), constraints + lin_constraints)

        try:
            prob.solve(solver=cp.ECOS,
                       warm_start=True,
                       max_iters=2000,
                       abstol=1e-9,
                       reltol=1e-9)
        except Exception:  # pragma: no cover
            break

        if x_var.value is None or M_var.value is None:
            break

        a_new = np.asarray(x_var.value).flatten()
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new /= s

        val_new = _objective(a_new)
        print(f"[LP] iter={it+1:2d} M={M:.6f} new_obj={val_new:.6f}")

        if val_new < best_val - 1e-9:
            best_val, best_seq = val_new, a_new.copy()
            a = a_new
        else:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuner (same as before)
# ----------------------------------------------------------------------
def _anneal_optimize(start_seq: np.ndarray,
                     deadline: float,
                     max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """Tiny Metropolis‑Hastings walk that swaps a small amount of mass."""
    rng = np.random.default_rng()
    cur = start_seq.copy()
    cur_val = _objective(cur)
    best = cur.copy()
    best_val = cur_val

    T0 = max(1e-6, cur_val * 0.04)
    decay = 0.9995
    it = 0

    while (max_iters is None or it < max_iters) and time.time() < deadline:
        it += 1
        T = T0 * (decay ** it)

        i, j = rng.integers(0, cur.size, size=2)
        while i == j:
            j = rng.integers(0, cur.size)

        delta = rng.random() * 0.02 * cur[i]
        delta = min(delta, cur[i])

        new_seq = cur.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_val = _objective(new_seq)

        if new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12)):
            cur = new_seq
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best = new_seq.copy()
                print(f"[Anneal] it={it:7d} best={new_val:.6f}")

        if it % 1000 == 0:
            noise = rng.normal(scale=0.001, size=cur.size)
            new_seq = cur + noise
            new_seq = np.clip(new_seq, 0.0, None)
            tot = new_seq.sum()
            if tot > 0:
                new_seq /= tot
            else:
                new_seq = _uniform(cur.size)
            new_val = _objective(new_seq)
            accept = new_val < cur_val or rng.random() < math.exp(-(new_val - cur_val) / (T + 1e-12))
            if accept:
                cur = new_seq
                cur_val = new_val
                if new_val < best_val:
                    best_val = new_val
                    best = new_seq.copy()
                    print(f"[Anneal] it={it:7d} (jump) best={new_val:.6f}")

    return best, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  Returns the best sequence found as a plain
    Python list of floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin
    rng = np.random.default_rng(seed + 12345)

    # problem size – 4000 was used in the reference solution
    n = 4000
    print(f"\n=== Optimising step function of length {n} ===")

    # ------------------------------------------------------------------
    # Load previous best if it exists
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, np.ndarray)):
        arr = np.asarray(prev, dtype=np.float64)
        if arr.shape == (n,):
            val = _objective(arr)
            best_seq = arr.copy()
            best_val = val
            print(f"[Prev] loaded previous best, objective = {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 0 – cheap seed generation (large random sample)
    # ------------------------------------------------------------------
    top_k = 5                # keep the best 5 seeds for refinement
    heap = []                # max‑heap on negative objective values
    seed_deadline = deadline - 300.0   # reserve ~300 s for later phases
    num_random_seeds = 4000   # cheap enough; each evaluation costs ~0.1 ms
    m_min, m_max = 70, 150   # support size bounds

    # deterministic cheap seeds (uniform, softmax, sidon‑like, greedy sidon)
    cheap_seeds = [
        _uniform(n),
        _random_softmax(n, rng),
        _sidon_like_seed(n, min(120, n)),
        # greedy Sidon set with size close to sqrt(2n)
        (lambda: (
            lambda pos: (np.bincount(pos, minlength=n) / np.bincount(pos).sum())
        ))(np.arange(int(np.sqrt(2 * n))))  # placeholder – replaced below
    ]

    # replace the placeholder with a proper greedy Sidon vector
    greedy_positions = _greedy_sidon_set(n, int(np.sqrt(2 * n)))
    greedy_vec = np.zeros(n, dtype=np.float64)
    greedy_vec[greedy_positions] = 1.0 / greedy_positions.size
    cheap_seeds[3] = greedy_vec

    for s in cheap_seeds:
        val = _objective(s)
        if val < best_val:
            best_val, best_seq = val, s.copy()
            print(f"[Init] improved by deterministic seed, obj = {val:.6f}")
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, s.copy()))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, s.copy()))

    print("[Init] random sparse seed search start")
    for i in range(num_random_seeds):
        now = time.time()
        if now > seed_deadline:
            break
        m = rng.integers(m_min, m_max + 1)
        positions = rng.choice(n, size=m, replace=False)
        # Dirichlet with a low concentration to get varied weights
        alpha = 0.3
        weights = rng.gamma(alpha, 1.0, size=m)
        weights /= weights.sum()
        seq = np.zeros(n, dtype=np.float64)
        seq[positions] = weights
        val = _objective(seq)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Seed] new global best = {val:.6f} (i={i}, m={m})")
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, seq.copy()))
        else:
            if -val > heap[0][0]:
                heapq.heapreplace(heap, (-val, seq.copy()))

    top_seeds = [seq for (_, seq) in heap]   # list of up‑to‑5 best seeds

    # ------------------------------------------------------------------
    # Phase 1 – smooth Adam optimisation on the top seeds
    # ------------------------------------------------------------------
    smooth_deadline = deadline - 150.0  # keep time for later phases
    if top_seeds:
        per_seed_time = (smooth_deadline - time.time()) / len(top_seeds)
        for idx, seed_seq in enumerate(top_seeds):
            if time.time() > smooth_deadline:
                break
            sub_deadline = min(time.time() + per_seed_time, smooth_deadline)
            seq_opt, val_opt = _smooth_adam_opt(seed_seq, sub_deadline, gamma=8.0)
            if val_opt < best_val:
                best_val, best_seq = val_opt, seq_opt.copy()
                print(f"[SmoothAdam] improved global best = {val_opt:.6f} (seed {idx})")

    # ------------------------------------------------------------------
    # Phase 2 – exact sub‑gradient refinement (shortened run)
    # ------------------------------------------------------------------
    sub_deadline = deadline - 80.0   # reserve ~80 s for LP + annealing
    if time.time() < sub_deadline:
        print("[Phase 2] Exact sub‑gradient refinement start")
        seq_sub, val_sub = _subgradient_refine(best_seq, sub_deadline)
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[Phase 2] improved by sub‑gradient = {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – optional LP linearised refinement
    # ------------------------------------------------------------------
    lp_deadline = deadline - 30.0
    if CVXPY_AVAILABLE and time.time() < lp_deadline:
        print("[Phase 3] LP linearised refinement start")
        seq_lp, val_lp = _lp_linearised_refine(best_seq, lp_deadline)
        if val_lp < best_val:
            best_val, best_seq = val_lp, seq_lp.copy()
            print(f"[Phase 3] improved by LP = {val_lp:.6f}")
        else:
            print("[Phase 3] LP did not improve")

    # ------------------------------------------------------------------
    # Phase 4 – simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase 4] Simulated annealing fine‑tuning")
        seq_final, val_final = _anneal_optimize(best_seq, deadline)
        if val_final < best_val:
            best_val, best_seq = val_final, seq_final.copy()
            print(f"[Phase 4] final objective after annealing = {val_final:.6f}")
        else:
            print("[Phase 4] annealing did not improve")

    # ------------------------------------------------------------------
    # Final normalisation and export
    # ------------------------------------------------------------------
    final_seq = np.maximum(best_seq, 0.0)
    total = final_seq.sum()
    if total < 1e-12:
        final_seq = _uniform(n)
        total = final_seq.sum()
    final_seq /= total

    final_obj = _objective(final_seq)
    print(f"\n=== Finished ===\nBest objective ≈ {final_obj:.6f} (length {n})")
    globals()["height_sequence_1"] = final_seq.copy()
    return final_seq.tolist()
