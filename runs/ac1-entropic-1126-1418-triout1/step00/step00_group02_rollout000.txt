    np.random.seed(seed)
    n = len(height_sequence_1)  # 4987
    # Initialize with random Dirichlet
    alpha = 0.5
    seq = np.random.gamma(alpha, 1.0, size=n)
    seq = seq / np.sum(seq)
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    start = time.time()
    # parameters
    it = 0
    while time.time() - start < budget_s - 5:
        # compute convolution
        a = seq
        # compute via FFT
        fft_a = np.fft.rfft(a, n=2*n-1)  # compute required length
        b = np.fft.irfft(fft_a * fft_a, n=2*n-1)
        # find max index
        k = np.argmax(b)
        max_b = b[k]
        # compute partner contribution
        partner = np.zeros_like(a)
        # for i in valid range
        lo = max(0, k - (n-1))
        hi = min(n-1, k)
        # indices i where j = k-i in [0,n-1] => i in [lo,hi]
        idx_range = np.arange(lo, hi+1)
        partner[idx_range] = a[k - idx_range]
        # compute contribution values
        contrib = a * partner
        # pick i_high as index with max contrib
        i_high = int(np.argmax(contrib))
        # pick i_low as index with min a (avoid i_high)
        # random among low values
        low_candidates = np.where(a > 0)[0]
        # maybe pick random low
        i_low = int(np.random.choice(low_candidates))
        # ensure not same
        if i_low == i_high:
            continue
        # transfer amount
        eps = min(a[i_high], 0.01) * np.random.rand()
        # propose new seq
        seq_new = a.copy()
        seq_new[i_high] -= eps
        seq_new[i_low] += eps
        # no need to renorm sum unchanged
        # Clip negative
        seq_new = np.maximum(seq_new, 0)
        # evaluate
        score = evaluate_sequence(seq_new.tolist())
        if score < best_score:
            best_seq = seq_new.copy()
            best_score = score
            seq = seq_new
        else:
            # accept with small prob if improvement moderate? ignore for simplicity

def project_to_simplex(v):
    # v is 1d array
    # reference to implementation from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a better step‐function using simple projected gradient descent.

The goal is to minimise
    2 * n * max(conv(a, a)) / (sum(a) ** 2)
where `conv` is the discrete convolution.
The function is scale invariant, therefore we keep the vector `a`
normalised to have unit L1‑norm (sum == 1).

The algorithm starts from a slightly randomised uniform vector,
then repeatedly:
  * computes the convolution,
  * finds the index `k` where the convolution attains its maximum,
  * builds a sub‑gradient of the maximum term (∂/∂a_i  a_i*a_{k-i}),
  * takes a small step opposite to the gradient,
  * projects the resulting vector back onto the probability simplex.

A simple simulated‑annealing acceptance rule is used to escape
local plateaus.  The whole search runs for the allotted time budget
(≤ budget_s seconds) and the best sequence found is returned.
"""

import time
import numpy as np

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i >= 0, sum(x) = 1 }.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    evaluation function.  Returns a list of floats ready for
    ``evaluate_sequence``.
    """

    a = a / a.sum()                     # keep the sum equal to 1

    best_a = a.copy()

    # hyper‑parameters (hand‑tuned)
    eta = 0.05            # learning‑rate for the gradient step
    temp = 0.02           # temperature for simulated annealing
    decay = 0.9995        # exponential decay for both eta and temp

    iteration = 0
    while time.time() < deadline:
        # ---------- convolution and maximal index ----------
        b = np.convolve(a, a)                # length 2*n-1
        k_max = int(np.argmax(b))
        max_val = b[k_max]

        # ---------- sub‑gradient of the maximal term ----------
        # ∂/∂a_i (a * a)[k_max] = a_{k_max-i} + a_i (if i == k_max-i)
        grad = np.zeros_like(a)
        # vectorised version: for indices where the partner exists
        i_vals = np.arange(n)
        j_vals = k_max - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        # for i != j the derivative is 2*a_j, otherwise a_i
        same = (mask) & (i_vals == j_vals)
        diff = mask & (~same)
        grad[diff] = 2.0 * a[j_vals[diff]]
        grad[same] = a[i_vals[same]]       # self‑pair term

        # ---------- gradient step ----------
        a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)      # enforce non‑negativity
        a_new = _project_to_simplex(a_new)  # keep sum == 1

        # ---------- evaluate ----------
        new_score = evaluate_sequence(a_new.tolist())

        # ---------- acceptance ----------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # occasional uphill moves (simulated annealing)
            prob = np.exp(-(new_score - cur_score) / max(1e-12, temp))
            if np.random.rand() < prob:

        # ---------- schedule update ----------
        eta *= decay

        iteration += 1

    # final sanity check: ensure the returned sequence satisfies the
    # evaluator's requirements (non‑negative, finite, not too large).
    best_a = np.clip(best_a, 0.0, 1000.0)
