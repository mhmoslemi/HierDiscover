≈ 0.23.  All values are far below the target bound 1.503.  The construction
needs only a few arithmetic operations and therefore easily fits the time limit.
"""

    # control.
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')

    for power in range(1, 6):
        # Build the raw (un‑normalised) sequence: (n, n‑1, … , 1) ** power
        raw = np.arange(n, 0, -1, dtype=np.float64) ** power
        seq = raw / raw.sum()                # normalise so that sum(seq) == 1

        # Evaluate the candidate.  The function is already imported in the

        try:

        if score < best_score:

            print(f"Power {power}: new best score = {score:.6f}")
