def propose_candidate(seed=12345, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Choose starting sequence
    if 'height_sequence_1' in globals() and np.random.rand() < 0.7:
        seq = np.array(height_sequence_1, dtype=float)
    else:
        n = np.random.randint(800, 2000)
        seq = np.random.rand(n)  # uniform random in (0,1)
    
    # Ensure sum not too small
    if np.sum(seq) < 0.01:
        seq += 0.1

    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial score: {best_score}")

    # main loop
    step = 0.5  # initial scaling of gradient step (fraction of current values)
    # We'll adapt step based on success
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(seq, seq)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        # compute gradient
        n = len(seq)
        grad = np.zeros(n)
        # grad_i = 2 * seq[max_idx - i] if within bounds
        start_i = max(0, max_idx - (n-1))
        end_i = min(n-1, max_idx)
        # Vectorized
        idxs = np.arange(start_i, end_i+1, dtype=int)
        grad[idxs] = 2.0 * seq[max_idx - idxs]

        # direction opposite of gradient
        direction = -grad

        # choose step size adaptively
        # compute candidate
        alpha = step * 0.01  # small base
        # use line search: try several alphas and keep best
        improvement = False
        for factor in [1.0, 0.5, 0.2, 0.1]:
            cand = seq + alpha * factor * direction
            # clip negatives
            cand = np.maximum(cand, 0.0)
            # renormalize sum to original to avoid sum shrinking too much
            s = np.sum(cand)
            if s < 0.01:
                continue
            cand = cand * (np.sum(seq) / s)
            # evaluate
            try:
                score = evaluate_sequence(cand.tolist())
            except Exception:
                continue
            if score < best_score:
                best_score = score
                best_seq = cand.copy()
                improvement = True
                seq = cand  # accept
                break
        if not improvement:
            # random perturbation to escape local optimum
            i = np.random.randint(len(seq))
            delta = (np.random.rand() - 0.5) * 0.1 * np.mean(seq)
            seq[i] = max(0.0, seq[i] + delta)
            # renormalize sum
            s = np.sum(seq)
            if s < 0.01:
                seq += 0.1  # bump
                s = np.sum(seq)
            seq = seq * (np.sum(best_seq) / s)  # keep sum similar to best
            # shrink step maybe
            step *= 0.99
        else:
            # if we made improvement, maybe increase step a bit
            step = min(step * 1.01, 5.0)

        # Occasionally reinitialize random part if stuck
        # Possibly after many iterations without improvement we can random restart
        # We'll check time for debug.

    return best_seq.tolist()

"""
Search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a,a)) / (sum(a)**2)

The search is based on the analytical observation that the sequence
a_i ∝ (i+offset)^{-0.5} gives a very flat autocorrelation
(its convolution is close to a constant ≈ π).  Starting from such a
power‑law family we perform a cheap gradient‑descent style flattening
of the current maximum convolution entry and occasionally add random
jitter to escape local minima.  Many random parameter samples are tried
within the time budget; the best sequence found is returned.

All helper functions are defined at the top level (no closures or lambdas)
and only standard scientific libraries are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def conv_self(seq: np.ndarray) -> np.ndarray:
    """Return the linear convolution of seq with itself using FFT."""
    n = len(seq)
    size = 1 << ((2 * n - 1).bit_length())          # next power of two
    f = np.fft.rfft(seq, n=size)
    conv = np.fft.irfft(f * f, n=size)[: 2 * n - 1]
    return conv


def gradient_max(seq: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Gradient of the (selected) maximum convolution entry with respect to `seq`.
    For c_k = sum_i a_i a_{k-i} the derivative d c_k / d a_i = 2 * a_{k-i}
    (zero when the index is out of range).
    """
    n = len(seq)
    grad = np.zeros_like(seq)
    start = max(0, max_idx - (n - 1))
    end = min(n - 1, max_idx)
    i = np.arange(start, end + 1, dtype=int)
    j = max_idx - i
    grad[i] = 2.0 * seq[j]
    return grad


def create_power_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """
    Produce a sequence a_i = (i + offset)^{-p}, i = 0 … n‑1.
    The values are clipped to the range accepted by `evaluate_sequence`
    (0 … 1000) but the absolute scale does not affect the objective.
    """
    idx = np.arange(n, dtype=float)
    seq = (idx + offset) ** (-p)
    # numerical safety – keep within the allowed range
    max_val = seq.max()
    if max_val > 1000.0:
        seq *= 1000.0 / max_val
    return seq


def safe_score(seq: np.ndarray) -> float:
    """Call the provided evaluate_sequence safely (returns inf on error)."""
    try:
        return evaluate_sequence(seq.tolist())
    except Exception:          # pragma: no cover
        return float("inf")

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    * start from a power‑law family a_i = (i+offset)^{-p}
    * repeatedly flatten the highest convolution peak by a gradient step
    * occasionally apply random perturbations
    * keep the best sequence seen before the time budget expires
    """
    np.random.seed(seed + int(time.time()) % 2**32)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # small safety margin

    # ------------------------------------------------------------------
    # initialise with the previously known construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

    else:

        best_score = float("inf")
        print("No baseline provided.")

    # ------------------------------------------------------------------
    # main random‑parameter search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # --- random parameters for the power‑law family ----------------
        n = np.random.randint(800, 5000)                 # length of the sequence
        p = np.random.uniform(0.45, 0.55)                 # exponent close to 0.5
        offset = np.random.uniform(0.0, 3.0)              # small shift to avoid singularity

            print(f"New best (raw) {best_score:.6f} (len={n})")

        # ----------------------------------------------------------------
        if cur_score < best_score * 1.2:                 # allow some slack

        print(f"Fallback uniform score: {best_score:.6f}")

    return best_seq.tolist()
