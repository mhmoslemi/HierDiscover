"""
propose_candidate
~~~~~~~~~~~~~~~~~

A simulated‑annealing / hill‑climbing search that tries to flatten the
self‑convolution of a non‑negative step function.

The objective that we minimise is

    2 * n * max_k (a * a)[k] / (sum_i a_i)^2

where `a` is the candidate height vector.  The value is invariant under a
global scaling of `a`, therefore the algorithm keeps the total mass equal
to 1 (the scale does not matter for the final score).

Main ideas
----------

1. **Initialisation** – start from a few simple seeds (uniform, a sparse
   Dirichlet‑like vector and, if available, the previous best `height_sequence_1`).

2. **Fast scoring** – the convolution is computed with an FFT of a power‑of‑2
   length, which is much faster than `np.convolve` for the required
   lengths (≈ 6000).

3. **Targeted moves** – in each iteration we locate the index `k_max`
   where the convolution is maximal.  The contribution of a position `i`
   to that peak is `a[i] * a[k_max‑i]`.  We move a small fraction of mass
   from the most responsible index to a position whose partner
   `k_max‑j` carries the smallest weight (often a border element).  This
   step directly reduces the current peak while keeping the total mass
   constant.

4. **Simulated annealing** – to avoid being trapped in a local minimum,
   moves that increase the objective are occasionally accepted with a
   probability that decays with a simple linear temperature schedule.

5. **Periodic LP refinement** – every few thousand iterations a linear
   program (identical to the one used in the SOTA code) is solved to
   obtain a direction that potentially reduces the bound on the
   convolution.  The new direction is mixed into the current vector with a
   tiny step size.

The loop runs until the allotted time (≈ 1000 s) expires and the best
sequence found is returned as a plain Python list.

The code contains only top‑level functions, no closures or lambdas,
and works with the supplied `evaluate_sequence` function.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _solve_convolution_lp(f_seq, rhs):
    """
    Linear programme used in the original SOTA iteration:
    maximise   Σ b_j
    subject to  (f * b)[k] ≤ rhs   for all k,
                b_j ≥ 0.
    Returns the optimal vector `b` or None on failure.
    """
    n = len(f_seq)
    # maximise → minimise the negative sum
    c = -np.ones(n)

    # build constraints (f * b)[k] ≤ rhs
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # contributions of b_j to (f*b)[k] are f_i with i + j = k
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity of b
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # solve – the default ‘highs’ method is robust and fast
    res = linprog(c, A_ub=a_ub, b_ub=b_ub, method='highs',
                  options={'disp': False, 'presolve': True, 'time_limit': 8.0})
    if res.success:
        return res.x
    return None


def _fast_score(a, size_fft, n):
    """
    Compute the objective 2 * n * max(conv) / (sum(a)^2) using FFT.
    This mirrors `evaluate_sequence` (the clipping inside that function is
    irrelevant for the small numbers we keep).
    """
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return float('inf')
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
    max_b = np.max(conv)
    return 2.0 * n * max_b / (sum_a * sum_a)

            cur_seq[j_tgt] -= delta

        # ----------------------------------------------------------
        # d) occasional LP refinement (similar to the SOTA iteration)
        # ----------------------------------------------------------
        if it % 2000 == 0 and it > 0:
            # normalise current vector (the exact scale does not matter)
            sum_cur = np.sum(cur_seq)
            if sum_cur > 0:
                f_norm = cur_seq * (math.sqrt(2.0 * n) / sum_cur)
                rhs = np.max(np.convolve(f_norm, f_norm))
                g = _solve_convolution_lp(f_norm, rhs)
                if g is not None:
                    sum_g = np.sum(g)
                    if sum_g > 0:
                        g_norm = g * (math.sqrt(2.0 * n) / sum_g)
                        t = 0.01
                        cand_seq = (1 - t) * cur_seq + t * g_norm
                        cand_score = _fast_score(cand_seq, size_fft, n)
                        if cand_score < cur_score or np.random.rand() < 0.05:
                            cur_seq = cand_seq
                            cur_score = cand_score
                            if cand_score < best_score:
                                best_score = cand_score
                                best_seq = cand_seq.copy()
                                print(f"[{it:06d}] LP step improved to {best_score:.6f}")

        it += 1

    # --------------------------------------------------------------
    # 4.  Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    print(f"[final] best score after {it} iterations = {best_score:.6f}")
    # Ensure we return a list of Python floats (the evaluator will clip
    # values larger than 1000 automatically; ours stay well below that)
    return best_seq.tolist()
