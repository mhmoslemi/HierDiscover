"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  The search mixes three cheap strategies:

1.  Parameterised families (geometric, power‑law and linear ramps) are
    sampled quickly – they give good starting points.
2.  A simple projected‑gradient descent is applied to the current
    best candidate.  For the index k where the convolution attains its
    maximum we move the vector a a little away from the gradient of
    b_k = (a*a)[k].  Afterwards we clip to non‑negativity and
    re‑normalise (the objective is scale‑invariant, only the shape
    matters).
3.  Occasionally a random perturbation is performed to escape local
    minima.

The loop stops a few seconds before the time budget expires and the
best sequence found is returned.
"""

import time
import numpy as np
from scipy import optimize


def _solve_convolution_lp(f_seq, rhs):
    """
    Solve   max sum(b)   s.t.  conv(f_seq, b) <= rhs ,  b >= 0.
    Returns the optimal b as a numpy array, or None if the LP fails.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise sum(b)  -> minimise -sum(b)
    # Build A_ub * b <= b_ub for each convolution entry
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv constraint: sum_i f_i * b_{k-i}  (b out of bounds = 0)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints   -b <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # a tiny time limit per LP – we never want to stall.
    result = optimize.linprog(
        c,
        A_ub=a_ub,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": 5.0, "disp": False},
    )
    if result.success:
        return result.x
    return None


def _gradient_of_max(conv_seq, k):
    """
    Gradient of the scalar b_k = (a*a)[k] with respect to the vector a.
    For each component i:
        d b_k / d a_i = a_i + a_{k-i}   (the second term is zero if out of range)
    """
    n = len(conv_seq)
    grad = conv_seq.copy()                     # the a_i part (self‑term)
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad[mask] += conv_seq[j[mask]]
    return grad


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    evaluate_sequence.  The routine runs for at most ``budget_s`` seconds
    and returns the best list it has found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # leave a little margin

    # ------------------------------------------------------------------
    # 1.  Choose a modest length (large enough to give flexibility,
    #     small enough for fast evaluation).
    # ------------------------------------------------------------------
    n = 1200                     # 1200 works well with the time budget

    # ------------------------------------------------------------------
    # 2.  Helper factories for the three parametric families.
    # ------------------------------------------------------------------
    def geometric(eps):
        """a_i ∝ (1‑eps)^i   (eps>0,   eps≈0 → uniform)"""
        r = 1.0 - eps
        seq = np.power(r, np.arange(n))
        return seq / seq.sum()

    def power_law(p):
        """a_i ∝ (i+1)^{-p}   (p≥0)"""
        idx = np.arange(1, n + 1)
        seq = 1.0 / np.power(idx, p)
        return seq / seq.sum()

    def linear_desc():
        """a_i decreasing linearly from n to 1."""
        seq = np.arange(n, 0, -1, dtype=float)
        return seq / seq.sum()

    # ------------------------------------------------------------------
    # 3.  Initialise the best candidate.
    # ------------------------------------------------------------------
    # start from a modest random Dirichlet (more variety than uniform)
    best_seq = np.random.dirichlet(alpha=np.full(n, 0.7))
    best_score = float("inf")

    # try the three families with a few parameter values
    candidates = []

    # geometric – sweep eps in a log‑space range
    for eps in np.logspace(-5, -0.5, 12):
        candidates.append(geometric(eps))

        candidates.append(power_law(p))

    # evaluate the initial pool
    for cand in candidates:

    # 4.  Main optimisation loop.

    it = 0

    while time.time() < deadline:

        conv = np.convolve(best_seq, best_seq)

        if s < 1e-12:                       # degenerate – re‑initialise uniformly

            cur_sum = best_seq.sum()
            if cur_sum > 0:

                rhs = np.max(np.convolve(norm_seq, norm_seq))

                if g is not None:

                    g = g * (np.sqrt(2 * n) / g.sum())
                    # small convex combination
                    t = 0.02
                    new_seq = (1 - t) * best_seq + t * g
                    new_seq = np.maximum(new_seq, 0.0)
                    new_seq = new_seq / new_seq.sum()

        # ---- c) random shake if no progress for a while -----------------------
        if it - last_improve_it > 3000:

            jitter = np.random.normal(scale=0.02, size=n)
            new_seq = new_seq + jitter
            new_seq = np.maximum(new_seq, 0.0)
            new_seq = new_seq / new_seq.sum()
            last_improve_it = it

        # ---- evaluate ---------------------------------------------------------
        try:

            if cur_score < best_score:
                best_score = cur_score
                best_seq = new_seq.copy()
                last_improve_it = it
                print(f"[it {it}] new best: {best_score:.6f}")
        except Exception:
            # any numerical issue – simply ignore this candidate
            pass

        it += 1

    print(f"Finished after {it} iterations – best score {best_score:.6f}")
    # convert to plain list before returning
    return best_seq.tolist()
