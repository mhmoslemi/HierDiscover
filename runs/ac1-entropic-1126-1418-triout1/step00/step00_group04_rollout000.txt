def simplex_projection(v):
    """Projection of vector v onto probability simplex (sum=1, >=0)."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho
    rho = np.where(u - cssv / ind > 0)[0]
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that tries to make the autoconvolution as flat as
possible.  The objective is

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

which is invariant to a global scaling of the vector `a`.  We therefore
work with vectors that are forced to lie on the probability simplex
`sum(a)=1 , a_i≥0`.  Starting from several random initializations we run a
projected gradient descent that reduces the largest entry of the
autoconvolution.

For the current maximal index `k_max` the gradient of `conv_kmax` w.r.t.
`a_i` is `∂conv_kmax/∂a_i = 2 * a_{k_max-i}` (when the index exists,
otherwise 0).  Moving a little opposite to this gradient makes the peak
smaller.  After each step we project back onto the simplex (the standard
Euclidean projection).  The step‑size follows a `1/√t` schedule and a
small random perturbation is added every few thousand iterations to help
escape shallow local minima.

The routine tries a handful of lengths (50 … 800) and a few restarts per
length; the best sequence encountered (according to `evaluate_sequence`)
is returned.  If nothing better than the previously known construction
(`height_sequence_1`) is found the fallback is used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x | x_i >= 0, sum x_i = 1}."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, "Projection onto the probability simplex:
    #   An efficient algorithm with a simple proof, and an application"
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main routine requested by the evaluator.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative sequence that minimises
    `evaluate_sequence`.  The algorithm runs several independent
    projected‑gradient descents on the simplex and keeps the best
    sequence found before the time budget expires.
    """
    np.random.seed(seed)

    # leave a small safety margin so the platform does not kill us
    deadline = time.time() + budget_s - 5.0

    # ------------------------------------------------------------------
    # Keep the best sequence seen so far.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    # If the previous construction is available we start from it as a
    # backup – it yields a score of exactly 2.0.
    try:
        fallback = list(height_sequence_1)  # noqa: F821 (global provided by the platform)
        fallback_score = evaluate_sequence(fallback)
        best_seq = fallback
        best_score = fallback_score
        print(f"Initial fallback score (height_sequence_1): {fallback_score:.6f}")
    except Exception:
        # no fallback – use a trivial uniform vector of length 2
        best_seq = [0.5, 0.5]
        best_score = evaluate_sequence(best_seq)

    # ------------------------------------------------------------------
    # Lengths we will explore.  Very short vectors cannot beat 2.0,
    # very long ones become expensive; the chosen range works well in
    # practice.
    # ------------------------------------------------------------------
    lengths = [50, 100, 200, 400, 800]

    # ------------------------------------------------------------------
    # Main outer loop – one restart per length.
    # ------------------------------------------------------------------
    for n in lengths:
        # number of independent restarts for this length
        restarts = 4
        for r in range(restarts):
            # random Dirichlet start (alpha=0.5 encourages sparsity)
            a = np.random.dirichlet(np.ones(n) * 0.5)

            # simple momentum‑free projected gradient loop
            t = 0
            step0 = 0.5      # initial step size
            while time.time() < deadline:
                # ----------------------------------------------------------
                # evaluate the current vector (score is cheap enough to do
                # every iteration, but we keep the expensive call only for
                # the bookkeeping)
                # ----------------------------------------------------------
                conv = np.convolve(a, a)
                max_conv = conv.max()
                # because we keep sum(a)=1 the denominator is 1
                score = 2.0 * n * max_conv

                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] new best: {best_score:.6f} (n={n})")

                # ----------------------------------------------------------
                # Gradient of the active (largest) convolution entry
                # ----------------------------------------------------------
                k_max = int(conv.argmax())
                # compute gradient g_i = 2 * a_{k_max-i}  (if index valid)
                g = np.zeros_like(a)
                i_start = max(0, k_max - (n - 1))
                i_end = min(n - 1, k_max)
                if i_end >= i_start:          # there is at least one term
                    i_idx = np.arange(i_start, i_end + 1)
                    j_idx = k_max - i_idx
                    g[i_idx] = 2.0 * a[j_idx]

                # ----------------------------------------------------------
                # Take a gradient step and project back onto the simplex.
                # ----------------------------------------------------------
                t += 1
                eta = step0 / np.sqrt(t)      # decreasing step size
                a_new = a - eta * g
                a = _project_onto_simplex(a_new)

                # ----------------------------------------------------------
                # Small random jitter every few thousand steps – helps to
                # escape plateaus.
                # ----------------------------------------------------------
                if t % 5000 == 0:
                    eps = np.random.randn(n) * 0.001
                    a = _project_onto_simplex(a + eps)

    # ------------------------------------------------------------------
    # Final sanity check – make sure the sequence respects the limits
    # (the evaluator does the same clipping, but staying inside the
    # allowed range avoids unnecessary surprises).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return a plain Python list as required by the platform.
    return [float(x) for x in best_seq]
