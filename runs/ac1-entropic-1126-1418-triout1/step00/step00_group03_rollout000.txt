"""
Simple geometric (exponential) construction.

For a given length n we pick a decay parameter `c` and set

    p[i] = r**i / sum_{j=0}^{n-1} r**j ,   r = exp(-c / n)

The vector `p` is a probability distribution (non‑negative and sums to 1).
Its self‑convolution is far flatter than the uniform distribution:
the maximal convolution value is roughly  α / n  with  

    α = c²·e^{-c} / (1‑e^{-c})² .

The evaluated score is 2·n·max(conv) = 2·α, so by choosing `c≈2`
we obtain a score around 1.45 – well below the target 1.503.

The function tries a few `c` values, evaluates the score with
`evaluate_sequence`, and returns the best sequence found.
"""
import time
import numpy as np

def _geometric_sequence(n: int, c: float) -> np.ndarray:
    """Return a length‑n geometric probability vector with decay `c`."""
    # r = exp(-c/n)  (very close to 1, keeps all entries positive)
    r = np.exp(-c / n)
    # denominator of the geometric series: (1 - r**n) / (1 - r)
    # we guard against rounding errors for r≈1 by using np.expm1
    # (1‑r) = -expm1(-c/n), (1‑r**n) = -expm1(-c)
    one_minus_r = -np.expm1(-c / n)
    one_minus_rn = -np.expm1(-c)
    denom = one_minus_rn / one_minus_r
    # p[i] = r**i / denom
    powers = np.exp(-c * np.arange(n) / n)   # r**i  = exp(-c*i/n)
    return powers / denom


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a step‑function height sequence with a low evaluation score.
    The routine tries a handful of decay parameters and keeps the best
    sequence (according to evaluate_sequence).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # keep a second margin

    # Length can be chosen arbitrarily; 5000‑plus entries give a smooth result.
    n = 5037

    # Parameter values to test – they span the region where the score
    # drops from the uniform value 2 down to below the target.
    c_candidates = [1.5, 1.7, 1.8, 1.9, 2.0, 2.2, 2.5]

    best_seq = None
    best_score = float('inf')

    for c in c_candidates:
        seq = _geometric_sequence(n, c)
        # evaluate_sequence expects a plain Python list
        score = evaluate_sequence(seq.tolist())
        print(f"c = {c:.3f} -> score = {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = seq.tolist()
        if time.time() >= deadline:
            break

    # As a tiny safety‑net, if something went wrong fall back to the
    # uniform sequence (which scores exactly 2.0).
    if best_seq is None:
        best_seq = [1.0 / n] * n
        print("Fallback to uniform sequence (score = 2.0)")

    print(f"Best score achieved: {best_score:.6f}")
    return best_seq
