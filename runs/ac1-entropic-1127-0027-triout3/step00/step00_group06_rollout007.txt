def project_onto_simplex(v):
    # v is 1d numpy array
    # returns w with w_i >=0, sum=1
    # Implementation per Wang & Carreira-Perpinan (2013)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def arcsine_shape(n):
    indices = np.arange(n)
    # shift to avoid zero at ends: add epsilon
    eps = 1e-6
    vals = 1.0 / np.sqrt((indices + eps) * (n - 1 - indices + eps))
    # This diverges at ends; we can cap values at some max
    vals = np.clip(vals, None, 1000)
    vals /= np.sum(vals)  # normalize to sum=1
    return vals

def refine_sequence(seq, deadline):
    n = len(seq)
    a = seq.astype(float).copy()
    # ensure sum =1
    a = a / np.sum(a)
    best_a = a.copy()
    # compute objective
    conv = conv_max(a)
    best_obj = 2 * n * conv.max()
    curr_obj = best_obj
    # step factor schedule
    step_factor = 0.5
    no_improve = 0
    while time.time() < deadline:
        conv = conv_max(a)
        max_val = conv.max()
        k = int(np.argmax(conv))
        idx = np.arange(n)
        complement = k - idx
        mask = (complement >=0)&(complement < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[complement[mask]]
        i = np.argmax(contrib)
        # find low contribution indices
        # select candidates where contrib low, maybe below 20th percentile
        threshold = np.percentile(contrib, 20)
        low_idx = np.where(contrib <= threshold)[0]
        # ensure not empty
        if low_idx.size == 0:
            low_idx = np.arange(n)
        # exclude i
        low_idx = low_idx[low_idx != i]
        if low_idx.size == 0:
            j = np.random.randint(n)
        else:
            j = np.random.choice(low_idx)
        # compute max allowed delta based on bound sqrt(max_val) - a[j]
        max_allowed = np.sqrt(max_val) - a[j]
        if max_allowed <= 0:
            # can't increase j without exceeding max, pick another j
            # just random j
            max_allowed = a[i] * 0.5
        delta = min(step_factor * a[i], max_allowed, a[i] * 0.5)
        if delta <= 1e-12:
            # small, reduce step factor
            step_factor *= 0.5
            if step_factor < 1e-6:
                break
            continue
        # apply move
        a[i] -= delta
        a[j] += delta
        # compute new objective
        conv_new = conv_max(a)
        new_obj = 2 * n * conv_new.max()
        # accept if improved else revert
        if new_obj < curr_obj - 1e-12:
            curr_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
            # possibly keep step factor
        else:
            # revert
            a[i] += delta
            a[j] -= delta
            # Optional: reduce step factor
            step_factor *= 0.7
            # Lower bound for step factor
            if step_factor < 1e-6:
                step_factor = 0.5  # restart bigger step
        # per iteration time log maybe
    return best_a

def conv_max(a):
    n = a.shape[0]
    size = 1 << ((2*n - 1).bit_length())
    # Real FFT for speed
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    return conv

"""
Hill‑climbing / simulated‑annealing search for a step‑function (non‑negative
vector) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where conv(a, a) is the discrete self‑convolution.
The algorithm repeatedly moves a small amount of mass from positions that
contribute heavily to the current maximal convolution entry toward positions
that are far from that bound.  The move size is limited by the square‑root of
the current maximal value (since a_i^2 ≤ max(conv)).  Several different
initialisations (uniform, arcsine‑shaped, linear, Dirichlet draws) are tried
and the best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using an FFT of power‑of‑two length."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fft_a = np.fft.rfft(a, size)
    conv = np.fft.irfft(fft_a * fft_a, size)[: 2 * n - 1]
    return conv


def _objective(a: np.ndarray) -> float:
    """Objective value 2 * n * max(conv(a,a)) assuming sum(a)=1."""
    n = a.shape[0]
    conv = _conv_full(a)
    return 2.0 * n * np.max(conv)


def _arcsine_shape(n: int) -> np.ndarray:
    """Heavy‑at‑the‑ends shape a_i ∝ 1/sqrt(i*(n-1-i)).  Normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    eps = 1e-12
    vals = 1.0 / np.sqrt((idx + eps) * (n - 1 - idx + eps))
    # Clip to avoid overflow from the singularities at the ends
    vals = np.clip(vals, 0.0, np.max(vals))
    vals = vals / np.sum(vals)
    return vals


def _linear_shape(n: int) -> np.ndarray:
    """Increasing linear shape, normalised to sum 1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    vals = idx.astype(float)
    vals = vals / np.sum(vals)
    return vals


def _initial_vectors(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a collection of diverse starting vectors (all sum to 1)."""
    vectors = []

    # 1. uniform
    vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. arcsine‑like (heavy at the ends)
    vectors.append(_arcsine_shape(n))

    # 3. linear increase
    vectors.append(_linear_shape(n))

    # 4. Dirichlet with small concentration (spiky)
    vectors.append(rng.dirichlet(alpha=np.full(n, 0.3)))

    # 5. Dirichlet with larger concentration (smooth)
    vectors.append(rng.dirichlet(alpha=np.full(n, 2.0)))

    # 6. If a previous good construction exists, use it (normalised)
    try:
        base = np.array(height_sequence_1, dtype=np.float64)
        if base.size > 0:
            base = np.maximum(base, 0.0)
            base_sum = np.sum(base)
            if base_sum > 0:
                vectors.append(base / base_sum)
    except Exception:
        pass

    return vectors


def _refine_vector(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing / simulated annealing refinement of a single vector.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    n = a.shape[0]

    # ensure exact normalisation
    a = np.maximum(a, 0.0)
    a_sum = np.sum(a)
    if a_sum == 0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a_sum

    best_a = a.copy()
    best_val = _objective(best_a)

    # initial step size (fraction of the source mass)
    step_frac = 0.5
    # simple exponential cooling for occasional uphill moves
    T = best_val * 0.1  # initial temperature
    cooling = 0.9999

    no_improve_iters = 0
    MAX_NO_IMPROVE = 2000

    while time.time() < deadline:
        # current convolution and its peak
        conv = _conv_full(a)
        cur_max = np.max(conv)
        k_max = int(np.argmax(conv))

        # contributions of each index to the current maximal entry
        idx = np.arange(n, dtype=np.int64)
        comp = k_max - idx
        mask = (comp >= 0) & (comp < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[comp[mask]]

        # index that hurts the current max the most
        i_src = int(np.argmax(contrib))

        # pick a destination index where the present value is far below sqrt(cur_max)
        sqrtM = np.sqrt(cur_max)
        low_mask = a < (0.5 * sqrtM)  # fairly low entries
        low_candidates = np.where(low_mask)[0]
        # ensure we have something; fall back to any index except source
        if low_candidates.size == 0:
            low_candidates = np.setdiff1d(np.arange(n), [i_src])
        i_dst = int(rng.choice(low_candidates))

        # maximum admissible transfer without breaking the bound a_j <= sqrtM
        max_transfer = min(step_frac * a[i_src], sqrtM - a[i_dst])
        # also never move more than half of the source entry
        max_transfer = min(max_transfer, 0.5 * a[i_src])

        if max_transfer <= 1e-14:
            # nothing to move – shrink step size and continue
            step_frac *= 0.7
            if step_frac < 1e-5:
                break
            continue

        # propose move
        delta = max_transfer * rng.random()  # a random fraction of the allowed amount
        a[i_src] -= delta
        a[i_dst] += delta

        new_val = _objective(a)

        # decide acceptance
        accept = False
        if new_val < best_val - 1e-12:
            accept = True
        else:
            # simulated‑annealing acceptance for occasional uphill steps
            if T > 0 and rng.random() < np.exp(-(new_val - best_val) / T):
                accept = True

        if accept:
            # keep the change
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                no_improve_iters = 0
                if verbose:
                    print(
                        f"[{time.time():.2f}] Improved to {best_val:.6f} (step {step_frac:.4f})"
                    )
            else:
                no_improve_iters += 1
                # still keep the move (uphill) but do not update best
        else:
            # revert
            a[i_src] += delta
            a[i_dst] -= delta
            no_improve_iters += 1

        # cooling
        T *= cooling
        # occasionally shrink step size after many non‑improving steps
        if no_improve_iters > MAX_NO_IMPROVE:
            step_frac *= 0.7
            no_improve_iters = 0
            if step_frac < 1e-5:
                break

    return best_a, best_val


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for the best non‑negative height sequence within the allotted time.
    The routine tries several diverse initialisations and refines each by a
    greedy / simulated‑annealing procedure that flattens the self‑convolution.
    The best sequence found is returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 2.0  # keep a safety margin

    # length of the step function – use the length of the known construction
    # if it exists, otherwise fall back to a moderate size.
    try:
        n = int(len(height_sequence_1))
    except Exception:
        n = 5276

    # generate starting vectors
    starts = _initial_vectors(n, rng)

    # allocate time roughly equally among the starts
    per_start = max(0.15, (deadline - time.time()) / max(len(starts), 1))
    best_overall_seq = None
    best_overall_val = float("inf")

    for idx, start_vec in enumerate(starts):
        if time.time() >= deadline:
            break
        sub_deadline = time.time() + per_start
        if sub_deadline > deadline:
            sub_deadline = deadline

        if kwargs.get("verbose", False):
            print(f"=== Start #{idx + 1}/{len(starts)} – length {n} ===")

        seq, val = _refine_vector(start_vec, sub_deadline, rng, verbose=kwargs.get("verbose", False))

        if val < best_overall_val:
            best_overall_val = val
            best_overall_seq = seq.copy()
            if kwargs.get("verbose", False):
                print(f"*** New global best: {best_overall_val:.6f}")

    # Safety: if nothing was found (should not happen) fall back to uniform.
    if best_overall_seq is None:
        best_overall_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Convert to ordinary Python list – evaluate_sequence will handle the
    # clipping / normalisation itself.
    return best_overall_seq.tolist()
