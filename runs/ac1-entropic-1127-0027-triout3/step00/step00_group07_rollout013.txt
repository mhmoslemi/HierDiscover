def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # define initial length
    n = np.random.randint(200, 800)  # random length
    # initialize sequence via Dirichlet (sum = 1)
    a = np.random.rand(n)
    a = a / np.sum(a)  # sum to 1
    best_seq = a.copy()
    best_eval = evaluate_sequence(best_seq.tolist())
    print(f'Initial length {n}, evaluation {best_eval:.6f}')
    # parameters for local search
    temperature = 0.1
    cooling = 0.99995
    step_scale = 0.2  # relative step size
    iter = 0
    while time.time() < deadline:
        iter += 1
        # choose move type: local transfer or LP direction
        if np.random.rand() < 0.7:
            # local transfer move
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            # delta proportional to a[i]
            delta = step_scale * a[i]
            # ensure not too large
            if delta > a[i]:
                delta = a[i]
            new_a = a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            # ensure nonnegative
            if new_a[i] < 0:
                new_a[i] = 0.0
            # project onto simplex to keep sum=1
            # Slight error: sum may change due to clipping, so renorm
            sum_new = np.sum(new_a)
            if sum_new == 0:
                continue
            new_a = new_a / sum_new
            # compute evaluation
            cur_eval = evaluate_sequence(new_a.tolist())
            # accept if better or (temperature) 
            if cur_eval < best_eval or np.random.rand() < math.exp((best_eval - cur_eval) / max(best_eval,1e-12) / temperature):
                a = new_a
                if cur_eval < best_eval:
                    best_eval = cur_eval
                    best_seq = new_a.copy()
                    print(f'Iter {iter}, new best eval {best_eval:.6f}')
        else:
            # LP direction move (use existing get_good_direction_to_move_into)
            new_seq = get_good_direction_to_move_into(a.tolist())
            if new_seq is not None:
                new_seq = np.array(new_seq)
                # normalize sum to 1
                if new_seq.sum() == 0:
                    continue
                new_seq = new_seq / new_seq.sum()
                cur_eval = evaluate_sequence(new_seq.tolist())
                if cur_eval < best_eval:
                    a = new_seq
                    best_seq = new_seq.copy()
                    best_eval = cur_eval
                    print(f'Iter {iter}, LP move improved eval {best_eval:.6f}')
        # cooling schedule
        temperature *= cooling
        # occasional length change: try to increase n if stuck
        if iter % 2000 == 0 and np.random.rand() < 0.2:
            # expand length by adding zeros or small random entries
            extra = np.random.randint(10, 50)
            # pad with zeros at random positions
            a = np.concatenate([a, np.zeros(extra)])
            n = len(a)
            # renormalize
            a = a / a.sum()

    best_seq = [float(x * 1000.0) for x in best_seq]  # scale to near bound (non essential)

"""
Propose a candidate sequence that minimises the evaluation function.

The algorithm starts from a very sparse (single‑spike) sequence.
At each iteration we

1. compute the current maximal auto‑convolution value M,
2. solve a linear program that finds a non‑negative vector `g`
   with the largest possible ℓ₁‑sum under the linear constraints
   conv(f, g) ≤ M,
3. determine the largest mixing factor `t` such that the convex
   combination     f' = (1‑t)·f + t·g
   still respects the bound max(conv(f',f')) ≤ M,
4. replace `f` by `f'` and keep the best sequence seen so far.

If the LP fails or the mixing factor is too tiny we perform a tiny
random perturbation to escape possible stalls.  The whole search runs
until the time budget expires (default 1000 s).  The final sequence is
clipped to the allowed range [0, 1000] and returned as a plain Python
list.

The method is a more aggressive version of the “good direction” approach
used in the reference code: we maximise the possible increase in
ℓ₁‑norm, then use a binary line‑search to take the biggest safe step.
This typically reduces the ratio `2·n·max(conv)/sum^2` well below the
previous bound of 2.0.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the linear program  maximise  Σ g_j   s.t.   conv(f,g) ≤ M,  g ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f, M):
    """
    Solve:
        max   sum(g)
        s.t.  (f * g)[k] ≤ M   for all k = 0 … 2·n‑2
              g_j ≥ 0
    Returns the optimal g as a NumPy array or None if infeasible.
    """
    n = len(f)
    # Linear objective (negated because linprog minimises)
    c = -np.ones(n, dtype=float)

    # Build inequality matrix A_ub * g ≤ b_ub
    rows = []
    b = []
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # for each i where 0 ≤ i ≤ n‑1 and 0 ≤ k‑i < n
        for i in range(i_start, i_end + 1):
            j = k - i
            row[j] = f[i]                     # contribution f[i]·g[j]
        rows.append(row)
        b.append(M)

    A_ub = np.vstack(rows)
    b_ub = np.array(b, dtype=float)

    # Bounds for non‑negative variables
    bounds = [(0, None)] * n

    # Solve with HiGHS (fast for medium sized LPs)
    try:
        res = linprog(c,
                      A_ub=A_ub,
                      b_ub=b_ub,
                      bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) yields a smaller
    value of `evaluate_sequence`.  The function respects the 1000‑second
    time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # keep a few seconds margin

    # ------------------------------------------------------------------
    # Initialise a sparse sequence – single spike at the first index.
    # Length is chosen randomly (larger lengths give a larger search space).
    # ------------------------------------------------------------------
    n = np.random.randint(200, 600)       # moderate size, fast LPs
    seq = np.zeros(n, dtype=float)
    seq[0] = 1.0                           # unit mass at the left end

    # ------------------------------------------------------------------
    # Keep the best sequence found so far.
    # ------------------------------------------------------------------
    best_seq = seq.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:
        best_score = float("inf")
    print(f"[INIT] length={n}, initial score={best_score:.6f}")

    # Helper to compute max(conv) quickly
    def max_conv(a):
        return np.max(np.convolve(a, a))

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    stagnation = 0                # counts iterations without improvement
    while time.time() < deadline:
        it += 1
        cur_M = max_conv(seq)

        # --------------------------------------------------------------
        # 1) Attempt an LP step that can increase the ℓ₁‑sum while keeping
        #    the current max‑convolution bound.
        # --------------------------------------------------------------
        g = solve_convolution_lp(seq, cur_M)

        if g is not None and np.sum(g) > np.sum(seq) + 1e-8:
            # ----------------------------------------------------------
            # 2) Binary line‑search for the largest t∈[0,1] such that
            #    max(conv((1‑t)·seq + t·g)) ≤ cur_M
            # ----------------------------------------------------------
            lo, hi = 0.0, 1.0
            for _ in range(20):                 # 20 iterations → ~1e‑6 precision
                mid = (lo + hi) / 2.0
                cand = (1.0 - mid) * seq + mid * g
                if max_conv(cand) <= cur_M * (1.0 + 1e-12):
                    lo = mid
                else:
                    hi = mid
            t_opt = lo

            # If we can move at least a tiny amount, accept the step.
            if t_opt > 1e-4:
                new_seq = (1.0 - t_opt) * seq + t_opt * g
                try:
                    new_score = evaluate_sequence(new_seq.tolist())
                except Exception:
                    new_score = float("inf")

                # Accept any improvement; otherwise keep the old sequence.
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    stagnation = 0
                    print(f"[{it:05d}] improved score={best_score:.6f}  t={t_opt:.4f}")
                else:
                    stagnation += 1

                seq = new_seq
                continue     # go to next iteration

        # --------------------------------------------------------------
        # 3) LP failed or gave a negligible step – perform a tiny random
        #    perturbation to try to escape a plateau.
        # --------------------------------------------------------------
        stagnation += 1
        idx = np.random.randint(n)
        delta = np.random.randn() * 0.01
        seq[idx] = max(0.0, seq[idx] + delta)

        # Re‑normalise to keep the total mass roughly constant (helps
        # numerical stability, but exact normalisation is not required).
        total = np.sum(seq)
        if total > 0:
            seq = seq / total
        else:
            seq = np.zeros_like(seq)

        # Occasionally restart from a fresh random Dirichlet vector if we
        # have been stagnant for a while.
        if stagnation >= 300:
            stagnation = 0
            seq = np.random.rand(n)
            seq = seq / np.sum(seq)
            print(f"[{it:05d}] restart – new random start")

    # ------------------------------------------------------------------
    # Final post‑processing: clip to the allowed range [0,1000] and
    # return as a plain Python list of floats.
    # ------------------------------------------------------------------
    final_seq = np.clip(best_seq, 0.0, 1000.0).tolist()
    print(f"[DONE] best score={best_score:.6f}  length={len(final_seq)}")
    return final_seq
