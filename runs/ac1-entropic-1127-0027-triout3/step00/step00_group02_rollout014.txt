```
def simplex_projection(v):
    # project v onto the simplex {x >=0, sum x = 1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (a non‑negative sequence) whose evaluation value

    2 * n * max(conv) / (sum(a)^2)

is as small as possible.

The method builds a probability‑like vector a (sum(a)=1) and runs a
projected gradient descent on the squared error between its self‑convolution
and a perfectly flat target (the average value of the convolution).  The
objective of the original problem is invariant under scaling, therefore
optimising the shape under the simplex constraint is sufficient.
The gradient of the loss L = Σ_k ( (a*a)[k] – 1/(2n‑1) )² is

    ∂L/∂a_i = 4 * Σ_j ( (a*a)[i+j] – 1/(2n‑1) ) * a_j

which can be computed efficiently with a single correlation.

The algorithm:
  * initialise a uniform vector (optional tiny random perturbation);
  * repeatedly:
        – compute convolution and the gradient;
        – take a step opposite to the gradient;
        – project the result back onto the probability simplex (a_i≥0, Σa_i=1);
        – keep the best sequence seen so far (according to evaluate_sequence).
  * the loop runs until the time budget is exhausted (≈ 1 000 s).

The final sequence is returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex {x ≥ 0, Σx = 1}
def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the rho index
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # renormalise to guard against numerical issues
    w_sum = w.sum()
    if w_sum == 0:
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    return w / w_sum


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    Parameters
    ----------

        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the function stops a few seconds before the limit).

    Returns

    list[float]
        The best sequence found (list of non‑negative floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # leave a small safety margin

    # --------------------------------------------------------------
    # Problem size – a few thousand entries is a good trade‑off.
    n = 2100                     # length of the step function
    # Initialise a uniform probability vector (sum = 1) and break symmetry slightly.
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a += np.random.randn(n) * 1e-4

    a = a / a.sum()

    # Keep track of the best candidate encountered.
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Gradient‑descent hyper‑parameters.
    eta = 0.5            # initial step size
    decay = 0.9995       # geometric decay of the step size
    it = 0

    # Target value of each convolution entry when the convolution is perfectly flat.
    target = 1.0 / (2 * n - 1)   # because sum(a) == 1 → Σ_k (a*a)[k] = 1
