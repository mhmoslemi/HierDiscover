def simplex_projection(v):
    """Project vector v onto probability simplex (nonnegative, sum=1)."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return zero vector
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

                    a = a_tmp
                    improved = True
                    break

                idx = np.random.randint(n)
                a[idx] = max(0.0, a[idx] + (np.random.rand() - 0.5) * 0.01)
                a = a / a.sum()
        iteration += 1
        if iteration % 200 == 0:
            print(f"Iter {iteration}, cur_score={cur_score:.6f}, best_score={best_score:.6f}")
    print(f"Finished after {iteration} iterations, best_score={best_score:.6f}")
    # convert to list
    return best_a.tolist()

"""
Propose a candidate sequence that minimizes the evaluation function

    Q(a) = 2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv(a, a)` is the (full) discrete convolution of a with itself.
The problem is scale‑invariant, so we can fix sum(a)=1 and simply minimise
`max(conv(a,a))`.  The algorithm below uses a projected gradient descent on
the L∞‑norm of the self‑convolution:

* start from a non‑uniform random (exponential) vector and normalise it,
* repeatedly compute the convolution, locate the index/indices where it attains
  its maximum, and form the sub‑gradient of this maximal value w.r.t. the
  entries of `a`,
* take a step opposite to the gradient, then project back onto the probability
  simplex (non‑negative entries that sum to 1).  The projection guarantees the
  feasibility of the sequence and its scale‑invariance keeps the objective
  unchanged,
* if a step does not improve the objective a back‑tracking line‑search reduces
  the step‑size, and occasional random perturbations help escape plateaus.

The method is purely deterministic apart from the initial random seed, runs
within the given time budget, and returns the best sequence found.
"""

import time
import numpy as np

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    # Implementation of the algorithm from:
    # Wang, Weiran, and Miguel A. Carreira‑Perpiñán. "Projection onto the
    # probability simplex: An efficient algorithm with a simple proof, and
    # application." arXiv preprint arXiv:1309.1541 (2013).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Return the full convolution a * a using an FFT (real input)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    # Numerical noise may produce tiny negatives – clip to zero.
    np.maximum(conv, 0, out=conv)
    return conv
