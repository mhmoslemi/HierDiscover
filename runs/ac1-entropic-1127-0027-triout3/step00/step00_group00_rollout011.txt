```
def solve_convolution_lp(f_seq, rhs):
    n = len(f_seq)
    # Build constraint matrix for conv(f, b) <= rhs
    from scipy.sparse import lil_matrix
    rows = 2*n - 1 + n
    A = lil_matrix((rows, n), dtype=float)
    # convolution constraints
    for k in range(2*n - 1):
        # for i yielding j = k - i
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # i runs from i_start to i_end
        for i in range(i_start, i_end+1):
            j = k - i
            A[k, j] = f_seq[i]
    # nonnegativity: -I <= 0
    for i in range(n):
        A[2*n - 1 + i, i] = -1.0
    b_ub = np.concatenate([np.full(2*n-1, rhs), np.zeros(n)])
    # solve with linprog
    res = linprog(c=-np.ones(n), A_ub=A.tocsr(), b_ub=b_ub, method='highs', options={'presolve': True})
    if res.success:
        return np.maximum(0.0, res.x)  # ensure nonnegative due to numeric
    return None
```

```python
def solve_convolution_lp(f, rhs):
    n = len(f)
    from scipy.sparse import lil_matrix
    # number of constraints: 2*n - 1 conv constraints + n nonneg bounds
    total_rows = 2 * n - 1 + n
    A = lil_matrix((total_rows, n), dtype=np.float64)
    # convolution constraints
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i from i_start..i_end
        # j = k - i; set A[k, j] = f[i]
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    # nonnegativity constraints -b_i <= 0
    for i in range(n):
        A[2 * n - 1 + i, i] = -1.0
    b = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])
    # objective: maximize sum b_j => minimize -sum b_j
    c = -np.ones(n)
    # solve using hiGHS
    result = linprog(c, A_ub=A.tocsr(), b_ub=b, method='highs')
    if result.success:
        # clip small negative to zero
        x = np.maximum(result.x, 0.0)
        return x
    return None
```

"""
A hybrid optimizer for the step‑function evaluation problem.

The algorithm works in two phases for a given length `n`:

1. **Sub‑gradient descent** – reduces the maximal convolution entry by moving
   weight away from the active (tight) constraints.  The sequence is always
   re‑normalised to the required sum `sqrt(2*n)`.

2. **LP based refinement** – solves the linear programme described in the
   reference (maximise sum(g) s.t. conv(f,g) ≤ M) and then blends the current
   sequence with the LP solution using a short line‑search.  This phase can
   further improve the bound once the gradient stage has plateaued.

Multiple lengths and random restarts are tried within the given time budget.
The best sequence found (as a plain list of floats) is returned.
"""
import time
import random
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

# ----------------------------------------------------------------------
# fast evaluation (identical to evaluate_sequence but without the safety checks)
def _fast_evaluate(seq: np.ndarray) -> float:
    n = len(seq)
    s = np.sum(seq)
    if s < 0.01:
        return float("inf")
    conv = np.convolve(seq, seq)
    return float(2.0 * n * np.max(conv) / (s * s))


# ----------------------------------------------------------------------
# build the LP constraint matrix for a given sequence `f`
def _build_lp_matrix(f: np.ndarray):
    n = len(f)
    rows = 2 * n - 1 + n                     # conv‑constraints + non‑negativity
    A = lil_matrix((rows, n), dtype=np.float64)

    # convolution constraints:   (f * b)[k] ≤ M
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs from i_start … i_end, j = k‑i
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]

    # b ≥ 0  →  -b ≤ 0
    for i in range(n):
        A[2 * n - 1 + i, i] = -1.0

    return A.tocsr()


# ----------------------------------------------------------------------
# solve   max Σ b_j   s.t.   conv(f,b) ≤ rhs,  b ≥ 0
def _solve_convolution_lp(f: np.ndarray, rhs: float):
    n = len(f)
    A = _build_lp_matrix(f)
    b = np.concatenate([np.full(2 * n - 1, rhs), np.zeros(n)])
    c = -np.ones(n)            # maximise Σ b_j  ≡ minimise -Σ b_j
    res = linprog(c, A_ub=A, b_ub=b, method="highs", options={"presolve": True})
    if res.success:
        # numerical noise may yield tiny negatives
        return np.maximum(res.x, 0.0)
    return None


# ----------------------------------------------------------------------
# sub‑gradient of the max‑convolution function
def _active_set(conv: np.ndarray, M: float, tol: float = 1e-12):
    """indices where the convolution attains (or is within `tol`) the current max."""
    return np.where(conv >= M - tol)[0]


def _subgradient(a: np.ndarray, active: np.ndarray):
    n = len(a)
    g = np.zeros(n, dtype=np.float64)
    # for each active lag k, add a[k‑i] to g[i] (derivative of conv_k w.r.t. a_i)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        g[i] += a[j]
    return g

def propose_candidate(seed: int = 42, budget_s: int = 950, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine respects the 1000 s total budget and
    returns the best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)

    deadline = time.time() + budget_s - 5   # safety margin for the caller
    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # lengths to try – small enough for the LP phase, large enough to allow
    # a good lower bound.  Adjust as needed.
    lengths = [800, 1500, 2400, 3200]

    # ------------------------------------------------------------------
    for n in lengths:
        sqrt_sum = np.sqrt(2.0 * n)

        # ---- 1) geometric start (theoretical optimum for a single exponential)
        x_opt = 1.255  # solution of e^x = 1 + 2x  (≈1.255)
        eps = x_opt / n
        r = 1.0 - eps
        geom = np.array([r ** i for i in range(n)], dtype=np.float64)
        geom *= sqrt_sum / geom.sum()
        cur = geom.copy()
        cur_score = _fast_evaluate(cur)

        # ---- 2) optionally also start from a random exponential vector
        if random.random() < 0.4:
            rnd = np.random.exponential(scale=1.0, size=n).astype(np.float64)
            rnd *= sqrt_sum / rnd.sum()
            if _fast_evaluate(rnd) < cur_score:
                cur = rnd
                cur_score = _fast_evaluate(cur)

        # ---- 3) if the global `height_sequence_1` exists, try a slice of it
        if "height_sequence_1" in globals():
            try:
                base = np.asarray(height_sequence_1, dtype=np.float64)
                if len(base) >= n:
                    cand = base[:n].copy()
                else:
                    cand = np.concatenate([base, np.zeros(n - len(base))])
                if cand.sum() > 0:
                    cand *= sqrt_sum / cand.sum()
                    if _fast_evaluate(cand) < cur_score:
                        cur = cand
                        cur_score = _fast_evaluate(cur)
            except Exception:
                pass

        # keep the best global result found so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.tolist()

        # ------------------------------------------------------------------
        # Phase 1 – sub‑gradient descent (fast, cheap)
        sub_deadline = time.time() + min(30.0, (deadline - time.time()) * 0.3)
        while time.time() < sub_deadline:
            conv = np.convolve(cur, cur)
            M = conv.max()
            act = _active_set(conv, M)
            grad = _subgradient(cur, act)

            # try a few step sizes; keep the best improvement
            improved = False
            for step in (0.0, 0.01, 0.025, 0.05, 0.1, 0.2):
                cand = cur - step * grad
                cand = np.maximum(cand, 0.0)
                if cand.sum() == 0.0:
                    continue
                cand *= sqrt_sum / cand.sum()
                sc = _fast_evaluate(cand)
                if sc < cur_score - 1e-12:
                    cur = cand
                    cur_score = sc
                    improved = True
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.tolist()
                    break
            if not improved:
                # small random jiggle to escape plateaus
                i = np.random.randint(n)
                cur[i] = max(0.0, cur[i] + np.random.randn() * 0.005)
                cur *= sqrt_sum / cur.sum()
                cur_score = _fast_evaluate(cur)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.tolist()

        # ------------------------------------------------------------------
        # Phase 2 – LP refinement with line search
        while time.time() < deadline:
            # current maximum convolution value
            conv = np.convolve(cur, cur)
            M = float(conv.max())

            # solve LP: maximise Σ g_j  subject to conv(cur, g) ≤ M, g ≥ 0
            g = _solve_convolution_lp(cur, M)
            if g is None:
                # fallback: tiny random perturbation
                i = np.random.randint(n)
                cur[i] = max(0.0, cur[i] + np.random.randn() * 0.01)
                cur *= sqrt_sum / cur.sum()
                cur_score = _fast_evaluate(cur)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.tolist()
                continue

            # renormalise g to the prescribed sum
            g_sum = g.sum()
            if g_sum == 0.0:
                break
            g *= sqrt_sum / g_sum

            # line‑search over mixing parameter t ∈ [0,1]
            improvement = False
            for t in (0.0, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, 0.7, 1.0):
                cand = (1.0 - t) * cur + t * g
                sc = _fast_evaluate(cand)
                if sc < cur_score - 1e-12:
                    cur = cand
                    cur_score = sc
                    improvement = True
                    if sc < best_score:
                        best_score = sc
                        best_seq = cand.tolist()
                    break
            if not improvement:
                # no progress – perform a tiny random move
                i = np.random.randint(n)
                cur[i] = max(0.0, cur[i] + np.random.randn() * 0.005)
                cur *= sqrt_sum / cur.sum()
                cur_score = _fast_evaluate(cur)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur.tolist()

        print(f"[{time.strftime('%H:%M:%S')}] length={n}, best_score={best_score:.6f}")

    # final return – a plain Python list (as required by the evaluator)
    return best_seq
