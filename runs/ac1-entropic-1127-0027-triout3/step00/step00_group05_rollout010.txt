def project_to_simplex(v):
    # ensure sum=1, v>=0
    # Shift to have non-negative after projection.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(..., budget=1000):

    start = time.time()
    deadline = start + budget - 0.5

    best_seq = None
    best_score = float('inf')

    # series of attempts
    attempts = 0
    while time.time() < deadline:
        # Choose random n perhaps varying with attempts
        if attempts % 5 ==0:
            n = np.random.choice([256, 512, 1024, 2048, 4096, 8192])
        else:
            n = np.random.choice([1024, 2048, 4096])
        # Random Dirichlet init
        a = np.random.rand(n)
        a = a / a.sum()
        # Maybe add small number of zeros at ends (rare).
        cur_best_local_score = 2*n*np.max(np.convolve(a,a))
        # Local search
        alpha = 0.5
        for _ in range(10000):
            if time.time() > deadline:
                break
            # compute conv and max
            conv = np.convolve(a, a)
            M = conv.max()
            score = 2 * n * M
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best score {best_score:.6f} n={n} iter={_}")
            # compute gradient
            # find active indices
            tol = 1e-12
            active = np.where(conv >= M - tol)[0]
            # Compute gradient g
            g = np.zeros_like(a)
            for k in active:
                i0 = max(0, k - (n-1))
                i1 = min(k, n-1)
                # i runs i0..i1 inclusive
                idx = np.arange(i0,i1+1)
                g[idx] += 2 * a[k - idx]
            # exponentiated step
            # adjust alpha adaptively
            a_new = a * np.exp(-alpha * g)
            a_new = a_new / a_new.sum()
            # optional acceptance if improvement immediate else keep original
            a = a_new
            # occasionally random perturb
            if np.random.rand() < 0.01:
                # add small noise to random elements
                idx = np.random.randint(n)
                delta = np.random.randn() * 0.01
                a[idx] = max(0, a[idx] + delta)
                a = a / a.sum()
        attempts += 1

    # final best seq to list; ensure positive numbers (clip small)
    best_seq = [float(max(0.0,x)) for x in best_seq]

    return best_seq

    size = 1 << ((2*n-1).bit_length())
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math

    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0

    best_seq = None
    best_score = float('inf')
    # Predefine lengths
    lengths = [64, 128, 256, 512, 1024, 2048, 4096]
    # For each length we will perform a certain number of restarts
    while time.time() < deadline:
        # Randomly pick length (prefer larger lengths later)
        if best_seq is None:
            n = np.random.choice(lengths)
        else:
            # bias towards larger lengths gradually
            probs = np.array([0.05, 0.1, 0.15, 0.2, 0.2, 0.2, 0.1])
            probs = probs / probs.sum()
            n = np.random.choice(lengths, p=probs)
        # Initialize a random probability vector
        a = np.random.rand(n)
        a = a / a.sum()
        # Local search parameters
        max_iters = 20000
        # Keep track of last improvement iteration for early stop
        last_improve = 0
        for it in range(max_iters):
            if time.time() >= deadline:
                break
            # Compute convolution via FFT
            conv = _conv_fft(a)
            M = conv.max()
            score = 2.0 * n * M
            # Record global best
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"Global best updated: score={best_score:.6f}, n={n}, it={it}")
                last_improve = it
                # If we beat target
                if best_score < 1.503:
                    # early exit
                    break
            # Compute gradient for active indices
            tol = 1e-12
            active = np.where(conv >= M - tol)[0]
            g = np.zeros_like(a)
            # Vectorized accumulation
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                idx = np.arange(i_start, i_end + 1)
                g[idx] += 2.0 * a[k - idx]
            # Step size decreasing with iteration count
            alpha = 0.5 / math.sqrt(it + 1)
            # Apply exponentiated gradient update
            a_new = a * np.exp(-alpha * g)
            a_new_sum = a_new.sum()
            if a_new_sum == 0:
                # fallback to random perturbation
                a_new = np.random.rand(n)
                a_new = a_new / a_new.sum()
            else:
                a_new = a_new / a_new_sum
            # Optional accept rule: accept if not worse, else occasionally accept
            conv_new = _conv_fft(a_new)
            M_new = conv_new.max()
            if M_new <= M or np.random.rand() < 0.001:
                a = a_new
                # may continue
            # Random perturbation every few iterations to avoid stagnation
            if it % 500 == 0 and np.random.rand() < 0.2:
                # add small Gaussian noise then reproject onto simplex
                noise = np.random.randn(n) * 0.01
                a = a + noise
                a = np.maximum(a, 0.)
                a = a / a.sum()
            # Early stop if no improvement for many iterations
            if it - last_improve > 2000:
                # restart
                break
        # Continue to next restart
        if best_score < 1.503:

"""
A simple projected exponentiated‑gradient search.

The objective to minimise is

    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

Scaling a does not change the score, therefore we can enforce sum(a)=1.
With this normalisation the score is 2·n·max(conv(a,a)).

The algorithm keeps a probability vector a (non‑negative, sum = 1) and
repeatedly:
* evaluates the convolution by FFT,
* identifies the index/indices where the maximum is attained,
* builds a sub‑gradient of the ``max`` (the sum of the gradients of the
  active convolutions),
* performs an exponentiated‑gradient update: a_i ← a_i·exp(−α·g_i) and
  renormalises,
* occasionally adds a small random perturbation to escape plateaus,
* stores the best sequence seen so far.

A decreasing step size α = α0/√t is used.  The search restarts with a new
random initialise when progress stalls.  Several lengths are tried; the
best sequence found before the time budget expires is returned as a plain
list (required by ``evaluate_sequence``).
"""

import time
import math
import numpy as np

def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution a*a using an FFT of size power‑of‑2."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())   # next power of two
    # real FFT is a bit faster
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that gives a low
    ``evaluate_sequence`` score.  The function runs for at most ``budget_s``
    seconds and returns the best sequence found.
    """
    np.random.seed(seed)
    # give a small safety margin to avoid being killed while cleaning up
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float("inf")

    # candidate lengths – we favour larger lengths after the first good result
    lengths = np.array([64, 128, 256, 512, 1024, 2048, 4096, 6144])

    # step‑size control
    alpha0 = 0.8          # initial (large) step size

    while time.time() < deadline:
        # choose a length (bias towards larger after a decent solution is known)
        if best_seq is None:
            n = int(np.random.choice(lengths))
        else:
            # gradually increase probability of longer vectors
            probs = np.array([0.02, 0.04, 0.08, 0.12, 0.2, 0.24, 0.2, 0.1])
            probs = probs / probs.sum()
            n = int(np.random.choice(lengths, p=probs))

        # random start: Dirichlet(1,…,1)
        a = np.random.rand(n)
        a /= a.sum()

        # local optimisation loop
        max_iters = 20000
        last_improve = 0

        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # convolution and current score (sum(a) == 1)
            conv = _conv_fft(a)
            M = conv.max()
            score = 2.0 * n * M

            # store global best
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                last_improve = it
                print(f"[{time.time():.1f}] New best: {best_score:.6f} (n={n}, it={it})")
                # stop early if we already beat the target
                if best_score < 1.503:
                    break

            # -----------------------------------------------------------------
            # sub‑gradient of the max‑convolution (active set)
            # -----------------------------------------------------------------
            tol = 1e-12
            active = np.where(conv >= M - tol)[0]

            g = np.zeros_like(a)      # gradient accumulator
            for k in active:
                i_start = max(0, k - (n - 1))
                i_end = min(k, n - 1)
                idx = np.arange(i_start, i_end + 1)
                # derivative of conv(k) w.r.t. a_i is 2·a_{k-i}
                g[idx] += 2.0 * a[k - idx]

            # -------------------------------------------------------------
            # exponentiated‑gradient step with decreasing step size
            # -------------------------------------------------------------
            alpha = alpha0 / math.sqrt(it + 1.0)
            a_new = a * np.exp(-alpha * g)

            # renormalise to a probability vector (projection onto the simplex)
            s = a_new.sum()
            if s == 0.0 or np.any(np.isnan(a_new)):
                # fall back to a fresh random start if something went wrong
                a_new = np.random.rand(n)
                a_new /= a_new.sum()
            else:
                a_new /= s

            # accept if not worse (or with a tiny probability to allow jumps)
            conv_new = _conv_fft(a_new)
            M_new = conv_new.max()
            if M_new <= M or np.random.rand() < 0.001:
                a = a_new

            # occasional random jitter to escape flat regions
            if (it % 500 == 0) and (np.random.rand() < 0.2):
                noise = np.random.randn(n) * 0.01
                a = a + noise
                a = np.maximum(a, 0.0)
                a_sum = a.sum()
                if a_sum == 0.0:
                    a = np.random.rand(n)
                    a_sum = a.sum()
                a /= a_sum

            # if we have not improved for a while, restart the inner loop
            if it - last_improve > 3000:
                break

        # early exit if target already achieved
        if best_score < 1.503:
            break

    # -------------------------------------------------------------------------
    # Prepare the final answer
    # -------------------------------------------------------------------------
    if best_seq is None:
        # fallback: a short uniform sequence
        n = 100
        best_seq = np.full(n, 1.0 / n)

    # Convert to plain Python list of floats (required by evaluate_sequence)
    result = [float(x) for x in best_seq]
    result = [max(0.0, x) for x in result]   # safety clamp

    # final diagnostic (may be omitted in the returned value)
    try:
        final_score = evaluate_sequence(result)
        print(f"Final sequence length {len(result)} → score {final_score:.6f}")
    except Exception as exc:
        print(f"Error during final evaluation: {exc}")

    return result
