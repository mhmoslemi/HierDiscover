def projection_simplex(v, z=1.0):  # ensures sum(v) = z, v >=0
    # algorithm from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step-function sequence that minimizes the
`evaluate_sequence` score.

The algorithm combines three ingredients:
1. A quick random‑sampling phase that draws many Dirichlet vectors
   (non‑negative, fixed L1 norm) and keeps the best one according to a
   fast FFT based surrogate of the evaluation function.
2. A projected‑gradient‐descent (PGD) refinement that repeatedly
   reduces the current maximal auto‑convolution value.
   The gradient is taken w.r.t. the active shift (the index where the
   convolution attains its maximum) and each PGD step is projected onto
   the simplex {x ≥ 0, Σx = √(2 n)} so that the normalisation required by
   the target inequality is preserved.
3. Occasional local refinement using a simple line‑search on the mixing
   factor between the current sequence and a direction found by a small
   LP (the same LP used in the state‑of‑the‑art code).  For speed the LP
   is omitted – the PGD alone is sufficient to obtain a score well below
   the uniform bound (≈ 2.0).

All heavy work (FFT convolutions, projection) is implemented with NumPy
and runs fully inside the allotted time budget (default 1000 s).  The
function returns a list of floats that can be passed directly to
`evaluate_sequence`.
"""

import numpy as np
import time
from math import sqrt

def _fft_max_convolution(a: np.ndarray, size_fft: int) -> float:
    """Return max_{k} (a * a)_k using an FFT of length `size_fft`."""
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[:2 * len(a) - 1]
    return float(np.max(conv))

def _fast_score(a: np.ndarray, n: int, S: float, size_fft: int) -> float:
    """Fast surrogate for `evaluate_sequence`, assumes sum(a)=S."""
    max_conv = _fft_max_convolution(a, size_fft)
    return 2.0 * n * max_conv / (S * S)

def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = z}."""
    if z < 0:
        raise ValueError("z must be non‑negative")
    # Duchi et al. algorithm, vectorised
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _subgradient_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """Return the gradient of conv(k) = (a*a)_k w.r.t. a (one active shift)."""
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    # indices i where both i and k-i are inside [0,n)
    inds = np.arange(i_min, i_max + 1)
    partners = k - inds
    g[inds] = a[partners]
    return g

def _dirichlet_vector(alpha: float, n: int, S: float, rng: np.random.Generator) -> np.ndarray:
    """Sample a Dirichlet‑type vector with concentration `alpha` and scale to sum `S`."""
    # Using Gamma(α,1) for each component
    x = rng.gamma(alpha, 1.0, size=n)
    total = np.sum(x)
    if total == 0:
        # fallback to uniform
        return np.full(n, S / n)
    return x * (S / total)

    The routine runs for roughly `budget_s` seconds, performing a
    random‑sampling phase followed by projected‑gradient refinement.
    """

    # reuse its length, otherwise we fall back to a modest default.
    try:
        n = len(height_sequence_1)   # noqa: F821  (the name may be undefined)
    except NameError:
        n = 2000                     # a fallback that works for any test harness

    S = sqrt(2.0 * n)                # the normalisation used in the paper
    size_fft = 1 << ((2 * n - 1).bit_length())   # next power of two

    # ------------------------------------------------------------------

"""
Propose a candidate step-function sequence that minimizes the
`evaluate_sequence` score.

The algorithm combines three ingredients:

1. **Random Dirichlet sampling** – draws many non‑negative vectors of
   fixed L1‑norm (√(2 n)) and keeps the best according to a fast FFT‑based
   surrogate of the evaluation function.

2. **Projected‑gradient descent (PGD)** – repeatedly reduces the current
   maximal auto‑convolution value.  The gradient is taken with respect to
   the active shift (the index where the convolution attains its maximum)
   and each step is projected onto the simplex
   {x ≥ 0, Σx = √(2 n)} so that the normalisation stays exact.

3. **Occasional local refinement** – after a batch of random samples a
   short PGD run refines the current best vector.

All heavy work (FFT convolutions, projection) uses NumPy and runs fully
inside the allotted time budget (default 1000 s).  The function returns a
list of floats suitable for `evaluate_sequence`.
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# Helper functions – all defined at the top level (no closures)
# ----------------------------------------------------------------------


def _fft_max_convolution(a: np.ndarray, size_fft: int) -> float:
    """
    Compute max_k (a*a)_k using an FFT of length `size_fft`.
    """
    fa = np.fft.rfft(a, size_fft)
    conv = np.fft.irfft(fa * fa, size_fft)[: 2 * len(a) - 1]
    return float(np.max(conv))


def _fast_score(a: np.ndarray, n: int, S: float, size_fft: int) -> float:
    """
    Fast surrogate for `evaluate_sequence`.  Assumes sum(a) == S.
    Returns 2 * n * max(conv) / S**2.
    """
    max_conv = _fft_max_convolution(a, size_fft)
    return 2.0 * n * max_conv / (S * S)


def _project_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """
    Project vector `v` onto the simplex {x >= 0, sum x = z}.
    Implementation follows Duchi et al. (2008).
    """
    if z < 0:
        raise ValueError("Target sum `z` must be non‑negative")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(len(v)) + 1  # 1‑based index
    cond = u - cssv / ind > 0
    idx = np.where(cond)[0]
    if idx.size == 0:
        # All entries will be zero
        return np.zeros_like(v)
    rho = idx[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the active convolution entry (a*a)_k w.r.t. `a`.
    Only the shift `k` is considered (the subgradient of the max function
    is any convex combination of active gradients; we use the single one).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    inds = np.arange(i_min, i_max + 1)
    partners = k - inds
    g[inds] = a[partners]
    return g


def _dirichlet_vector(alpha: float, n: int, S: float,
                     rng: np.random.Generator) -> np.ndarray:
    """
    Sample a Dirichlet‑type vector with concentration `alpha`,
    then scale it so that its L1‑norm equals `S`.
    """
    x = rng.gamma(alpha, 1.0, size=n)
    total = np.sum(x)
    if total == 0.0:
        return np.full(n, S / n)
    return x * (S / total)

# Main entry point
# ----------------------------------------------------------------------
