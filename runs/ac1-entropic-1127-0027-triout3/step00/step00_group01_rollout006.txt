def project_to_simplex(v, z=1.0):
    """Projects vector v onto the simplex {x | x >= 0, sum x = z}."""
    # Algorithm from Duchi et al. (2008)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # Determine sequence length: use existing height_sequence_1 if present else random
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = len(init_seq)
    else:
        n = np.random.randint(2000, 4000)
        init_seq = np.random.rand(n)
    
    # Ensure non-negative, scale to some sum (keep original sum)
    if np.sum(init_seq) == 0:
        init_seq[0] = 1.0
    # Project onto simplex to have fixed sum (original sum)
    sum_target = np.sum(init_seq)
    init_seq = project_to_simplex(init_seq, sum_target)

    # start from a random perturbation of init to avoid uniform stagnation
    cur = init_seq.copy()
    # add small random noise
    cur = np.maximum(0, cur + np.random.randn(n) * 0.01)
    cur = project_to_simplex(cur, sum_target)

    best = cur.copy()
    best_score = evaluate_sequence(best.tolist())

    # Precompute FFT size
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    # hyperparameters
    eta = 0.2   # learning rate
    alpha = 5.0 # smoothing factor
    iteration = 0
    no_improve = 0

    while time.time() < deadline:
        iteration += 1
        # convolution a * a
        A = np.fft.rfft(cur, size)
        B = A * A
        conv = np.fft.irfft(B, size)[:2 * n - 1]
        max_b = conv.max()
        # compute objective
        score = 2.0 * n * max_b / (sum_target * sum_target)

        # keep best
        if score < best_score:
            best_score = score
            best = cur.copy()
            print(f"Iter {iteration}: improved score {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # compute smoothed weights
        # shift to avoid overflow: b - max_b <= 0
        exp_vals = np.exp(alpha * (conv - max_b))
        w = exp_vals / exp_vals.sum()

        # gradient via correlation with reversed a
        a_rev = cur[::-1]
        W = np.fft.rfft(w, size)
        A_rev = np.fft.rfft(a_rev, size)
        conv_rev = np.fft.irfft(W * A_rev, size)[:2 * n - 1]
        # gradient g_i = 2 * conv_rev[n-1 + i]
        g = 2.0 * conv_rev[n - 1: n - 1 + n]

        # descent step
        step = -eta * g
        cand = cur + step
        # project back to simplex to keep sum and nonnegativity
        cand = project_to_simplex(cand, sum_target)

        # evaluate candidate (optional to compute exact, but we can compute approximate)
        # quick evaluation using convolution again to decide acceptance
        A_c = np.fft.rfft(cand, size)
        B_c = A_c * A_c
        conv_c = np.fft.irfft(B_c, size)[:2 * n - 1]
        cand_score = 2.0 * n * conv_c.max() / (sum_target * sum_target)

        # accept if better or with probability (simulated annealing)
        if cand_score <= score or np.random.rand() < 0.01:
            cur = cand
        else:
            # occasionally inject noise
            if no_improve > 200:
                cur = cur + np.random.randn(n) * 0.005
                cur = np.maximum(cur, 0)
                cur = project_to_simplex(cur, sum_target)
                no_improve = 0

        # adapt step size
        if iteration % 100 == 0:
            if no_improve > 20:
                eta *= 0.7
                alpha *= 1.1
                no_improve = 0
            else:
                eta = min(eta * 1.05, 1.0)

    return best.tolist()

"""
Propose a candidate step‑function by flattening its self‑convolution.

The evaluation function is
    2 * n * max(conv(a, a)) / (sum(a)**2)

For a fixed length `n` the sum does not matter (scaling cancels out), so the
problem reduces to minimizing the maximum coefficient of the convolution.
Uniform coefficients give a value of exactly 2, but the maximum can be lowered
by reshaping the vector so that the pairwise products are spread more evenly.

The algorithm below starts from a (possibly random) non‑negative vector,
projects it onto the simplex (fixed `ℓ1`‑norm), and then performs a
projected gradient descent on a smoothed version of the `max` objective.
The smoothing uses a soft‑max (log‑sum‑exp) with a parameter `alpha`
that is increased slowly.  After each gradient step the vector is projected
back onto the simplex to keep the sum constant and all entries non‑negative.
Occasional random perturbations and simple step‑size adaptation help to avoid
getting stuck at the uniform stationary point.

The loop runs until the allotted time budget is exhausted; the best sequence
found so far is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: projection onto the probability simplex (ℓ1 = const, x >= 0)
def _project_to_simplex(v: np.ndarray, z: float) -> np.ndarray:
    """Project v onto the simplex {x | x_i >= 0, sum x_i = z}."""
    if z < 0:
        raise ValueError("Target sum must be non‑negative")
    # classic O(n log n) algorithm (Duchi et al., 2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Convolution using FFT (linear, not circular)
def _linear_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Return a * a (linear convolution) using FFT of length `size`."""
    A = np.fft.rfft(a, size)
    B = A * A
    conv = np.fft.irfft(B, size)
    return conv[: 2 * len(a) - 1]


# --------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000) -> list[float]:
    """Return a sequence of non‑negative heights that (hopefully) improves the score.

    The routine performs a projected gradient descent on a smoothed max‑convolution
    objective, with occasional random perturbations and simple step‑size control.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Initialise the sequence (use the provided construction if available)
    if "height_sequence_1" in globals():

    else:

    n = len(cur)

    # Ensure a positive total mass and project onto the simplex (fixed ℓ1 norm)
    total = cur.sum()
    if total <= 0:
        cur[0] = 1.0
        total = 1.0
    cur = _project_to_simplex(cur, total)

    # Keep the best solution seen so far
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 2. Pre‑compute FFT length (next power of two)
    size = 1
    while size < 2 * n - 1:
        size <<= 1

    # Hyper‑parameters (will be adapted on the fly)
    eta = 0.25          # initial step size
    alpha = 5.0         # smoothing for the soft‑max
    no_improve = 0      # counter for stagnation
    iteration = 0

    # ------------------------------------------------------------------
    # 3. Main optimisation loop
    while time.time() < deadline:
        iteration += 1

        # ----- convolution and current score
        conv_full = _linear_convolution(cur, size)
        max_b = conv_full.max()
        # exact evaluation (identical to the provided function up to clipping)
        cur_score = 2.0 * n * max_b / (total * total)

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[it {iteration:05d}] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # ----- soft‑max weights (focus on large convolution entries)
        # exp(alpha * (b - max_b)) is safe because the argument is ≤ 0
        weights = np.exp(alpha * (conv_full - max_b))
        weights_sum = weights.sum()
        if weights_sum == 0:
            # numerical safety – fall back to uniform weights
            weights = np.ones_like(conv_full) / len(conv_full)
        else:
            weights /= weights_sum

        # ----- gradient of the smoothed max (via correlation)
        # we need g_i = 2 * Σ_k w_k * a_{k-i}
        a_rev = cur[::-1]                       # reversed current vector
        w_fft = np.fft.rfft(weights, size)
        rev_fft = np.fft.rfft(a_rev, size)
        corr_full = np.fft.irfft(w_fft * rev_fft, size)
        corr_full = corr_full[: 2 * n - 1]       # linear convolution length
        # extract the slice corresponding to lags -i  (see analysis)
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # ----- gradient step  (descent direction = -grad)
        cand = cur - eta * grad
        cand = _project_to_simplex(cand, total)

        cand_conv = _linear_convolution(cand, size)

        cand_score = 2.0 * n * cand_max / (total * total)

        if cand_score <= cur_score or np.random.rand() < 0.005:

                cur = _project_to_simplex(cur, total)
                no_improve = 0

                eta = max(eta * 0.7, 1e-4)
                no_improve = 0
            else:
                eta = min(eta * 1.03, 1.0)

    # ------------------------------------------------------------------
    # Return the best sequence found (as a plain list)
    return best_seq.tolist()
