def propose_candidate(seed=0, budget_s=1000):
    """Search for a non‑negative step sequence that minimizes evaluate_sequence().
    The algorithm performs projected sub‑gradient descent on the L∞‑norm of the
    self‑convolution (which is proportional to the evaluation function when the
    sequence is normalized to sum = 1).  Several random restarts with different
    lengths are tried within the time budget.
    """
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    best_seq = None
    best_score = float('inf')
    lengths = [256, 512, 1024, 2048, 4096]  # increasing sizes, order matters

    # Helper: project onto probability simplex (non‑negative, sum = 1)
    def proj_simplex(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            # fallback to uniform vector
            return np.ones_like(v) / v.size
        return v / s

    for N in lengths:
        # decide how many restart attempts for this length
        restarts = 3
        for r in range(restarts):
            if time.time() >= deadline:
                break

            # initialise p – try a few patterns
            if r == 0:
                p = np.ones(N) / N            # uniform
            elif r == 1:
                # decreasing linear weights (more mass at the left)
                p = np.arange(N, 0, -1, dtype=float)
                p = proj_simplex(p)
            else:
                # random Dirichlet (alpha=1)
                p = np.random.dirichlet(np.ones(N))

            # current best for this start
            cur_score = evaluate_sequence(p.tolist())
            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.tolist()
                print(f"[init] new best {best_score:.6f} (N={N}, restart={r})")

            it = 0
            # hyper‑parameters
            base_lr = 0.03           # controls absolute step size
            decay = 0.9999           # per‑iteration decay for the learning rate
            noise_prob = 0.02
            noise_scale = 0.01

            while time.time() < deadline and it < 200000:
                # compute convolution and its maximum
                conv = np.convolve(p, p)
                max_val = conv.max()
                k_max = conv.argmax()

                # quick score (identical to evaluate_sequence when sum(p)=1)
                cur_score = 2.0 * N * max_val
                # store best
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = p.tolist()
                    print(f"[improve] score={best_score:.6f} (N={N}, it={it})")

                # sub‑gradient of the max‑conv term
                grad = np.zeros_like(p)
                # i where 0 <= k_max - i < N
                i_min = max(0, k_max - (N - 1))
                i_max = min(N - 1, k_max)
                # fill gradient
                # grad[i] = p[k_max - i]
                idx_i = np.arange(i_min, i_max + 1)
                grad[idx_i] = p[k_max - idx_i]

                # scale gradient to a reasonable step size
                norm = np.linalg.norm(grad) + 1e-12
                step = base_lr / norm

                # gradient step (moving opposite to increase of the max value)
                p_new = p - step * grad

                # occasional random perturbation to escape flat regions
                if np.random.rand() < noise_prob:
                    i = np.random.randint(N)
                    delta = np.random.randn() * noise_scale
                    p_new[i] = max(p_new[i] + delta, 0.0)

                # project back onto the simplex
                p = proj_simplex(p_new)

                # geometric decay of learning rate
                base_lr *= decay

                it += 1

    # Final sanity check – evaluate the best candidate with the official function
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] best_score={final_score:.6f} (length={len(best_seq)})")
        best_score = final_score

    # Return a Python list (required by the judge)
    return best_seq if best_seq is not None else [1.0]
```

"""
Propose a candidate sequence for `evaluate_sequence`.

The evaluation function is homogeneous:
    evaluate(sequence) == 2 * n * max(conv(sequence,sequence)) / sum(sequence)^2 .
Therefore we may restrict ourselves to probability vectors
(p_i >= 0, sum(p_i)=1) and minimise 2 * n * max(conv(p,p)).
For a probability vector the average of the convolution is 1/(2n-1), so
the ideal value would be close to this average.  The algorithm below
performs a projected stochastic sub‑gradient descent on a smooth
approximation of the max‑convolution (soft‑max).  The optimisation
variables are kept on the probability simplex, and a few random
perturbations help escape flat regions.  Several restarts with different
lengths are tried within the allotted time budget.

The returned list can be fed directly to `evaluate_sequence`; its scale
does not matter (the function is scale‑invariant) but we keep the
values modest (they lie in [0,1]).
"""
import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        # fallback: uniform distribution
        return np.ones_like(v) / v.size
    return v / s


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The search runs for at most `budget_s` seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a little safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------
    # Helper: evaluate the current probability vector using the official function.
    # For a vector that already sums to 1 the score is simply 2*n*max(conv).
    # ------------------------------------------------------------
    def _score(p: np.ndarray) -> float:
        n = p.shape[0]
        # max convolution (linear convolution, zero‑padding)
        max_conv = np.convolve(p, p).max()
        return 2.0 * n * max_conv

    # ------------------------------------------------------------
    # Search over a few different lengths.
    # ------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]      # increasing sizes; larger => more work
    for n in lengths:
        # several restarts for each length
        for restart in range(3):
            if time.time() >= deadline:
                break

            # ----------- initialise a probability vector -----------------
            if restart == 0:
                # uniform start
                p = np.ones(n) / n
            elif restart == 1:
                # linearly decreasing start (more mass on the left side)
                line = np.arange(n, 0, -1, dtype=float)
                p = _project_simplex(line)
            else:
                # random Dirichlet start
                p = np.random.dirichlet(np.ones(n))

            cur_score = _score(p)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.tolist()
                print(f"[init]  n={n:5d} restart={restart} score={best_score:.6f}")

            # ------------ optimisation parameters -----------------------
            it = 0
            max_iters = 200_000                     # hard limit per restart
            base_lr = 0.02                           # base learning‑rate factor
            decay = 0.9999                           # geometric decay per iteration
            noise_prob = 0.02                        # chance to add a tiny random kick
            noise_scale = 0.01                       # magnitude of the kick
            alpha = 30.0                             # soft‑max temperature (larger → sharper)

            while time.time() < deadline and it < max_iters:
                # compute convolution and its max
                conv = np.convolve(p, p)
                max_val = conv.max()
                # evaluate current candidate only when it improves
                cur_score = 2.0 * n * max_val
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = p.tolist()
                    print(f"[impr]  n={n:5d} it={it:6d} score={best_score:.6f}")

                # ---------- soft‑max weights (smooth approximation of max) ----------
                # shift by the maximum to avoid overflow
                shifted = conv - max_val
                w = np.exp(alpha * shifted)
                w_sum = w.sum()
                if w_sum == 0.0:        # numerical safety
                    w = np.ones_like(w) / w.size
                else:
                    w /= w_sum

                # ---------- gradient of the smoothed objective ----------
                # grad_i = 2 * sum_k w_k * p_{k-i}
                # this is a correlation of w with p (negative shifts)
                corr = np.correlate(w, p, mode='full')
                # correlation indices: offset = len(p)-1 ; shift = -(i)
                # grad[i] = 2 * corr[offset - i]
                offset = n - 1
                indices = offset - np.arange(n)
                grad = 2.0 * corr[indices]                     # shape (n,)

                # ---------- step size -------------------------------------------------
                # limit step size so that nobody becomes negative after the update
                pos = grad > 1e-12
                if np.any(pos):
                    step_limit = np.min(p[pos] / grad[pos])
                    step_limit = 0.9 * step_limit                # safety margin
                else:
                    step_limit = np.inf

                norm = np.linalg.norm(grad) + 1e-12
                step = base_lr / norm
                step = min(step, step_limit)

                # ---------- stochastic perturbation ----------------------------------
                if np.random.rand() < noise_prob:
                    idx = np.random.randint(n)
                    p[idx] = max(p[idx] + np.random.randn() * noise_scale, 0.0)

                # ---------- gradient update + projection -------------------------------
                p = _project_simplex(p - step * grad)

                # decay learning rate for stability
                base_lr *= decay
                it += 1

    # -------------------------------------------------------------------------
    # Final verification (using the official evaluate_sequence) and return.
    # -------------------------------------------------------------------------
    if best_seq is not None:
        final_score = evaluate_sequence(best_seq)
        print(f"[final] best length={len(best_seq)} score={final_score:.6f}")
        best_score = final_score
    else:
        # Fallback: a tiny constant sequence (never used in practice)
        best_seq = [1.0]

    return best_seq
