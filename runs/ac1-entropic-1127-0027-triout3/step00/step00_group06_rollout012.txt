import numpy as np
import time
from scipy.optimize import linprog

def build_convolution_matrix(f):
    n = len(f)
    # matrix A where (A @ b)[k] = sum_i f[i] * b[k-i] (if indices in range)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)
    for k in range(rows):
        # for each i, j such that i + j = k
        # b_j coefficient = f[i]
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # i goes from i_min to i_max inclusive
        # j = k - i
        # assign A[k, j] = f[i]
        # we can vectorize but simple loop fine for n up to few hundred
        for i in range(i_min, i_max + 1):
            j = k - i
            A[k, j] = f[i]
    return A

def solve_convolution_lp(f, bound, max_time=5.0):
    """Solve LP: maximize sum(b) s.t. convolution(f,b) <= bound, b>=0."""
    n = len(f)
    A = build_convolution_matrix(f)
    c = -np.ones(n)  # maximize sum(b)
    b_ub = np.full(2 * n - 1, bound, dtype=float)
    # Use bound on variables
    bounds = [(0.0, None)] * n
    # Use method='highs' which is default, set time limit
    options = {'time_limit': max_time}
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs', options=options)
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def improve_sequence(seq, step=0.1, max_iter=1000, time_limit=5.0):
    """Perform a few LP‐based improvement steps on seq."""
    n = len(seq)
    target_sum = np.sqrt(2.0 * n)
    # normalize once to keep sum fixed
    # we'll keep the sum implicitly by the convex combination
    cur = np.array(seq, dtype=float)
    best = cur.copy()
    best_score = evaluate_sequence(best.tolist())
    for it in range(max_iter):
        s = np.sum(cur)
        if s == 0:
            cur = np.full(n, 1e-3)
            s = np.sum(cur)
        # Normalize to target sum to keep scaling comparable (does not change ratio)
        cur_norm = cur * (target_sum / s)
        # current max bound
        max_conv = np.max(np.convolve(cur_norm, cur_norm))

        b = solve_convolution_lp(cur_norm, max_conv, max_time=time_limit)
        if b is None:

            delta = np.random.randn() * 0.01
            if cur[i] - delta >= 0:
                cur[i] -= delta
                cur[j] += delta
            continue
        # scale b to have the same target sum
        sum_b = np.sum(b)
        if sum_b == 0:

        b_norm = b * (target_sum / sum_b)

"""
Propose a step‑function (list of non‑negative heights) that minimises

    2 * n * max(f*f) / (sum(f)**2)

where * denotes the linear convolution.  The code uses a
LP‑based “increase the L¹‑norm while keeping the convolution bound”
strategy (the same idea that was used in the reference solution)
combined with a simple simulated‑annealing style schedule.
The search runs on several problem sizes (small vectors make each LP cheap)
and keeps the best sequence that was found before the time budget expires.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper that builds the convolution matrix A such that
#    (A @ b)[k] = sum_i f[i] * b[k-i]   (indices out of range are ignored)
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    # Fill only the entries that actually appear in the convolution.
    # For a given shift k the admissible i are max(0,k-(n-1)) … min(k,n-1).
    for k in range(rows):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            j = k - i               # j = k-i is the index of b that pairs with f[i]
            A[k, j] = f[i]
    return A


# ----------------------------------------------------------------------
# Solve:   max sum(b)  s.t.  conv(f,b) <= bound ,  b >= 0
# ----------------------------------------------------------------------
def _solve_convolution_lp(f: np.ndarray, bound: float, time_limit: float = 0.8):
    """Return a feasible vector b (as a NumPy array) or None."""
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, bound, dtype=float)

    # maximise sum(b)  <=>  minimise -sum(b)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    # “highs” is the default method; we give a hard time budget.
    try:
        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit},
        )
        if res.success:
            return res.x
    except Exception:
        pass
    return None


# ----------------------------------------------------------------------
# One improvement step: keep the current maximum convolution value,
# look for a new vector with a larger L¹‑norm but the *same* bound,
# then move a little towards it.
# ----------------------------------------------------------------------
def _lp_improvement_step(seq: np.ndarray, step: float) -> np.ndarray:
    """
    Perform a single LP‑based improvement step.
    Returns a new sequence (the same length) – possibly identical to ``seq``.
    """
    n = seq.shape[0]
    # Normalise to the standard L¹‑norm √(2n); this does not change the score.
    target_sum = np.sqrt(2.0 * n)
    cur_sum = np.sum(seq)
    if cur_sum == 0:
        seq = np.full(n, 1e-3)
        cur_sum = np.sum(seq)
    f = seq * (target_sum / cur_sum)

    # current convolution bound (the value we would like to keep)
    max_conv = np.max(np.convolve(f, f))

    # solve the LP that tries to increase the sum while respecting that bound
    b = _solve_convolution_lp(f, max_conv, time_limit=0.5)
    if b is None:
        # fallback: a tiny random redistribution that preserves the sum
        i, j = np.random.choice(n, 2, replace=False)
        delta = np.random.randn() * 0.01
        if seq[i] - delta >= 0:
            cand = seq.copy()
            cand[i] -= delta
            cand[j] += delta
            return cand
        return seq.copy()

    # Scale the LP solution back to the target L¹‑norm
    b_sum = np.sum(b)
    if b_sum == 0:
        return seq.copy()
    b = b * (target_sum / b_sum)

    # Convex combination: move a fraction ``step`` towards the LP solution
    new_seq = (1.0 - step) * seq + step * b
    # keep everything inside the allowed [0,1000] range
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **_):
    """
    Search for a sequence that yields a low value of

        2 * n * max(conv(seq,seq)) / (sum(seq)**2)

    The routine explores several vector lengths (80 … 500) and,
    for each length, repeatedly applies the LP‑based improvement step
    together with a tiny simulated‑annealing style schedule.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # a deterministic uniform baseline (score = 2.0) – useful as a fallback
    # ------------------------------------------------------------------
    baseline_len = 200
    best_seq = [1.0] * baseline_len
    best_score = evaluate_sequence(best_seq)
    print(f"[init] uniform baseline (len={baseline_len}) → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # try a few problem sizes; smaller sizes make each LP cheap, therefore
    # many iterations can be performed within the time limit.
    # ------------------------------------------------------------------
    lengths = [80, 120, 160, 200, 250, 300, 400, 500]

    # parameters for the outer (simulation) loop
    global_step = 0.08   # initial convex‑combination weight

    for n in lengths:
        if time.time() >= deadline:
            break

        # random positive start (avoid exact zeros which give no gradient)
        cur = np.random.rand(n) + 0.01

        # perform a bounded number of improvement iterations for this size
        max_outer_iters = 300
        for outer in range(max_outer_iters):
            if time.time() >= deadline:
                break

            # one LP improvement step
            new_seq = _lp_improvement_step(cur, step=global_step)

            # evaluate the candidate
            try:
                score = evaluate_sequence(new_seq.tolist())
            except Exception:
                # any numerical mishap – skip this candidate
                continue

            # keep the best globally
            if score < best_score:
                best_score = score
                best_seq = new_seq.tolist()
                print(f"[update] new best → score {best_score:.6f} (len={n}, it={outer})")

            # accept the move if it improves the *local* score, otherwise keep
            # the old vector (standard hill‑climbing).  Occasionally accept
            # non‑improving moves to escape plateaus.
            if score <= evaluate_sequence(cur.tolist()) or np.random.rand() < 0.02:
                cur = new_seq

            # slowly decay the mixing factor – large steps early, fine steps later
            global_step = max(global_step * 0.995, 0.001)

        # after finishing a length we reset the step size a bit
        global_step = min(global_step * 1.05, 0.08)

    # ------------------------------------------------------------------
    # final safety check – make sure the returned sequence is non‑empty
    # ------------------------------------------------------------------
    if not best_seq:
        best_seq = [1.0] * 200

    print(f"[final] best score = {best_score:.6f} (length = {len(best_seq)})")
    return best_seq
