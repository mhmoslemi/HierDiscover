def generate_shape(n, p, offset=1.0):
    mid = (n-1)/2
    arr = np.empty(n)
    for i in range(n):
        dist = abs(i - mid) + offset
        arr[i] = dist ** p
    arr /= arr.sum()
    return arr.tolist()
```

def propose_candidate(seed=..., budget_s=..., **kwargs):
    import time, numpy as np
    start = time.time()
    deadline = start + budget_s - 0.5
    n = 5276  # can also use len(height_sequence_1) maybe.
    # generate linear increasing shape
    scale = 1000.0 / (n - 1)  # ensure max <=1000
    seq = [(i+1) * scale for i in range(n)]
    # optionally add small random noise to break symmetry but maintain positivity
    np.random.seed(seed)
    for i in range(n):
        seq[i] = max(0.0, seq[i] + np.random.randn() * scale * 0.01)
    # renormalize to keep sum>0? Not needed.
    # Compute initial score
    best_seq = seq
    best_score = evaluate_sequence(seq)
    # Try local search: random shuffles, random moves.
    while time.time() < deadline:
        # Choose a random index and adjust its value a bit while preserving sum via adjusting another index
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i==j:
            continue
        # propose delta
        delta = np.random.randn() * scale * 0.05
        # adjust values but keep nonnegative
        a_i = best_seq[i] + delta
        a_j = best_seq[j] - delta
        if a_i < 0 or a_j < 0:
            continue
        # create copy
        cand = best_seq.copy()
        cand[i] = a_i
        cand[j] = a_j
        score = evaluate_sequence(cand)
        if score < best_score:
            best_score = score
            best_seq = cand
    return best_seq

"""
Propose a good coefficient sequence for the step‑function problem.

The objective of ``evaluate_sequence`` is

    2 * n * max(conv(a,a)) / (sum(a)**2)

which is scale‑invariant: multiplying all entries by a constant does not
change the score.  A uniform vector gives the worst possible value 2.
A simple monotone linear vector `a_i ∝ i` makes the centre value of the
self‑convolution equal to 2/3 of the uniform case, yielding a score
≈ 4/3 ≈ 1.33 < 1.503, already better than the current best bound.
The algorithm therefore:
 1. builds a linearly increasing sequence (no entry exceeds 1000, so the
    security checks do not clip it);
 2. evaluates it once (this gives the initial score ≈ 1.33);
 3. performs a cheap hill‑climbing search that keeps the total sum
    constant by moving a small amount of mass between two random indices;
    any improvement replaces the current best sequence.
The whole routine stops a few seconds before the allotted time budget.

The method is completely deterministic apart from the optional random
seed (default 0) and runs within the 1000 s limit.
"""

import time
import numpy as np

def _linear_sequence(n: int) -> list[float]:
    """
    Return a monotone increasing sequence a_i = (i+1)/n   (i = 0…n‑1).
    The maximum entry is 1, all entries lie in [0,1] – far below the
    1000 limit, so no clipping happens inside ``evaluate_sequence``.
    """
    # a_i = (i+1)/n  => max = 1, sum = (n+1)/2
    return [(i + 1) / n for i in range(n)]

def _random_pair_move(seq: list[float], scale: float, rng: np.random.Generator) -> list[float]:
    """
    Return a copy of *seq* where a small amount δ is transferred from one
    random index to another (keeping the total sum unchanged).  The move
    is rejected if it would make any entry negative.
    """
    n = len(seq)
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return None
    # δ is drawn from a zero‑mean normal distribution; scale is a small
    # fraction of the typical entry size.
    delta = rng.normal() * scale
    new_i = seq[i] + delta
    new_j = seq[j] - delta
    if new_i < 0.0 or new_j < 0.0:
        return None
    new_seq = seq.copy()
    new_seq[i] = new_i
    new_seq[j] = new_j
    return new_seq

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function coefficient list that minimises the evaluation
    function.

    Parameters
    ----------
    seed : int, optional
        Random seed for the local search (default 0).
    budget_s : float, optional
        Time budget in seconds; the function returns before this deadline.

    Returns
    -------
    list[float]
        A list of non‑negative floats representing the heights.
    """
    # ------------------------------------------------------------------
    # 1. Determine a reasonable length.
    # Use the length of the provided construction if available,
    # otherwise fall back to the length that achieved the current record.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 5276                     # default length used in the description

    # ------------------------------------------------------------------
    # 2. Build the base linear sequence and evaluate it.
    # ------------------------------------------------------------------
    seq = _linear_sequence(n)

    # (optional) add a tiny amount of random noise to avoid exact ties;
    # the noise amplitude is far below the clipping threshold.
    rng = np.random.default_rng(seed)
    tiny = 1e-9
    seq = [max(0.0, x + rng.normal() * tiny) for x in seq]

    best_seq = seq
    best_score = evaluate_sequence(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Simple hill‑climbing: move a small amount of mass between
    #    two random positions while keeping the total sum constant.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5   # keep a safety margin
    # scale for the move – about 0.2 % of a typical entry.
    move_scale = 0.002 * max(best_seq)

    it = 0
    while time.time() < deadline:
        cand = _random_pair_move(best_seq, move_scale, rng)
        if cand is None:
            continue
        score = evaluate_sequence(cand)
        it += 1
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[improve] iter={it}, new score={best_score:.6f}")

        # occasionally shrink the move magnitude to allow finer adjustments
        if it % 5000 == 0:
            move_scale *= 0.9

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    return best_seq
