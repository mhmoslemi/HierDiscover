def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np
    import time

    # Determine n. If height_sequence_1 global is available, use its length; else default 5000.
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000

    # Build sequence a_i = 1 / sqrt( (i + 0.5) * (n - i - 0.5) )
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt( (idx + 0.5) * (n - idx - 0.5) )
    # Normalize to sum sqrt(2n)
    target_sum = np.sqrt(2.0 * n)
    scale = target_sum / np.sum(a)
    seq = (a * scale).tolist()

    # Optionally, maybe do few gradient descent steps to fine-tune.
    # We'll run simple local optimization for a few seconds.

    start = time.time()
    deadline = start + budget_s - 5.0

    # evaluate initial
    best_seq = seq
    try:
        best_score = evaluate_sequence(best_seq)
    except Exception:
        best_score = float('inf')
    print(f'Initial sequence length {n}, score {best_score:.6f}')

    # simple loop: random minor perturbations, acceptance if improved
    n_iter = 0
    step = 0.01 * np.mean(best_seq)  # initial perturbation amplitude
    while time.time() < deadline:
        # propose perturbation on a random subset of indices
        new_seq = best_seq.copy()
        # choose random number of indices to modify (e.g., 1% of n)
        k = max(1, int(0.01 * n))
        idxs = np.random.choice(n, size=k, replace=False)
        # add gaussian noise
        noise = np.random.randn(k) * step
        for pos, delta in zip(idxs, noise):
            new_seq[pos] = max(0.0, new_seq[pos] + delta)
        # re-normalize sum to target
        s = sum(new_seq)
        if s > 0:
            factor = target_sum / s
            new_seq = [x * factor for x in new_seq]
        else:
            continue
        try:
            score = evaluate_sequence(new_seq)
            if score < best_score:
                best_score = score
                best_seq = new_seq
                print(f'Iter {n_iter}: improved score {best_score:.6f}')
                # adapt step size
                step = max(step * 0.9, 1e-6)
            else:
                # optionally increase step gradually to explore
                step = min(step * 1.01, np.mean(best_seq) * 0.5)
        except Exception:
            pass
        n_iter += 1

    print(f'Finished. Best score {best_score:.6f}')
    return best_seq

"""
Propose a sequence of non‑negative heights that (hopefully) beats the
current bound 2.0.  The algorithm works in three stages:

1.  Initialise a reasonably good shape – a discrete version of the
    1/√x “Sidon” construction.  This yields an evaluation around
    π/2 ≈ 1.57.
2.  Randomly explore many alternative shapes (exponential, Gamma,
    power‑law, noisy Sidon) while keeping the ℓ1‑norm fixed to
    sqrt(2·n).  The best candidate seen is stored.
3.  Refine the current best candidate by a simple sub‑gradient descent
    on the maximum convolution entry.  At each step the index where the
    convolution attains its maximum is identified; the gradient of that
    entry (2·a_{k−i}) is used to move mass away from the offending
    pairs, the vector is re‑normalised and the new score is kept when
    it improves.

All operations respect the non‑negativity and the clipping limits of
``evaluate_sequence``.  The routine runs for the allotted time budget
(~1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level as required)
# ----------------------------------------------------------------------


def _target_sum(n: int) -> float:
    """Desired ℓ1‑norm (the optimizer’s scale is irrelevant, but this choice
    makes the evaluation equal to the maximal convolution entry."""
    return np.sqrt(2.0 * n)


def _normalise(a: np.ndarray, target: float) -> np.ndarray:
    """Scale a non‑negative vector to have the given ℓ1‑norm."""
    s = a.sum()
    if s == 0.0:
        return a
    return a * (target / s)


def _evaluate(a: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    # convert to plain Python list – the evaluator expects a list
    return evaluate_sequence(a.tolist())


def _sidon_like(n: int) -> np.ndarray:
    """Discrete Sidon‑type seed: a_i = 1/√(i+½).  This shape gives a fairly
    flat self‑convolution (value ≈ π/2 after normalisation)."""
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(idx + 0.5)
    a = _normalise(a, _target_sum(n))
    return a


def _random_candidate(n: int, kind: str) -> np.ndarray:
    """Generate a random positive vector of length *n* and normalise it.
    ``kind`` selects the underlying distribution."""
    if kind == "exp":
        raw = np.random.exponential(scale=1.0, size=n)
    elif kind == "gamma_small":
        # shape << 1 → a few large spikes, many tiny values
        raw = np.random.gamma(shape=0.2, scale=1.0, size=n)
    elif kind == "power":
        # power‑law tail: a_i ∝ (i+1)^{-p} with random exponent p∈[0.2,0.8]
        p = np.random.uniform(0.2, 0.8)
        idx = np.arange(n, dtype=np.float64) + 1.0
        raw = idx ** (-p)
    elif kind == "noisy_sidon":
        base = 1.0 / np.sqrt(np.arange(n, dtype=np.float64) + 0.5)
        noise = np.exp(np.random.randn(n) * np.random.uniform(0.2, 0.6))
        raw = base * noise
    else:
        raw = np.random.rand(n) + 0.1
    raw = np.maximum(raw, 0.0)
    return _normalise(raw, _target_sum(n))


def _gradient_step(a: np.ndarray, step_factor: float) -> np.ndarray:
    """
    One sub‑gradient step that tries to lower the current maximal
    convolution entry.
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    k = int(np.argmax(conv))                     # index of the current peak
    # gradient of b_k = Σ a_i a_{k-i} w.r.t a_i is 2·a_{k-i} (if the index exists)
    i = np.arange(n, dtype=np.int64)
    j = k - i
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]

    # step size proportional to the excess over the average (which is 1 after normalisation)
    excess = conv[k] - 1.0
    norm = np.linalg.norm(grad) + 1e-12
    eta = step_factor * excess / norm
    eta = np.clip(eta, 0.0, 0.2)                 # keep the update modest

    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    a_new = _normalise(a_new, _target_sum(n))
    return a_new


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a low‑value step‑function within the allotted time.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs : dict
        Ignored (kept for compatibility).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    np.random.seed(seed)
    # ------------------------------------------------------------------
    # Choose a working length.  If a historic solution is available we
    # reuse its length (longer vectors give more freedom), otherwise we
    # fall back to a moderately sized vector.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)             # type: ignore[name-defined]
    except Exception:
        n = 2000                               # a safe default

    target = _target_sum(n)

    # ------------------------------------------------------------------
    # 1.  Start from the deterministic Sidon‑like construction.
    # ------------------------------------------------------------------
    current = _sidon_like(n)
    best_seq = current.copy()
    best_score = _evaluate(best_seq)

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Random exploration – a handful of different families.
    # ------------------------------------------------------------------
    kinds = ["exp", "gamma_small", "power", "noisy_sidon"]
    deadline = time.time() + budget_s - 5.0   # keep a small safety margin

    # time‑boxed exploration loop
    while time.time() < deadline:
        # occasionally switch to a fresh random start
        if np.random.rand() < 0.03:
            cand = _sidon_like(n)
        else:
            cand = _random_candidate(n, np.random.choice(kinds))

        score = _evaluate(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            current = cand.copy()
            print(f"[rand] improved score={best_score:.6f}")

        # ----------------------------------------------------------------
        # 3.  Local gradient‑descent refinements from the current best.
        # ----------------------------------------------------------------
        # Perform a few cheap gradient steps before the next random candidate.
        for _ in range(3):
            current = _gradient_step(current, step_factor=0.6)
            score = _evaluate(current)
            if score < best_score:
                best_score = score
                best_seq = current.copy()
                print(f"[grad] improved score={best_score:.6f}")
            # early exit if we have already beaten the target
            if best_score < 1.5:
                deadline = time.time() + 1.0   # finish shortly

        # safety break if we already reached a strong bound
        if best_score < 1.45:
            break

    print(f"[final] best score={best_score:.6f}")
    return best_seq.tolist()
