def propose_candidate(seed=42, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_score = float('inf')
    best_seq = None
    # define lengths to test: maybe fixed 2000
    n = 2000
    while time.time() < deadline:
        # random initialization: uniform or Dirichlet
        init_type = np.random.choice(['uniform','dirichlet','exp'])
        if init_type == 'uniform':
            a = np.full(n, 1.0/n)
        elif init_type == 'dirichlet':
            a = np.random.dirichlet(np.ones(n))
        else:
            # exponential decreasing
            r = np.random.uniform(0.9,0.99)
            a = np.array([r**i for i in range(n)], dtype=float)
            a = a / a.sum()
        # local gradient descent
        beta = 1.0
        eta = 500.0
        max_beta = 2000.0
        iter_limit = 2000 # but controlled by time
        i = 0
        while time.time() < deadline and i < 5000:
            b = np.convolve(a, a)
            max_b = b.max()
            score = 2 * n * max_b
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best {best_score:.6f} at iteration {i}")
            # compute softmax
            # to avoid overflow, subtract max
            b_scaled = beta * b
            max_bsc = b_scaled.max()
            expb = np.exp(b_scaled - max_bsc)
            w = expb / expb.sum()
            grad = 2.0 * np.correlate(w, a, mode='valid')
            # exponentiated gradient update
            a = a * np.exp(-eta * grad)
            # renormalize
            sum_a = a.sum()
            if sum_a == 0:
                a = np.full(n, 1.0/n)
            else:
                a = a / sum_a
            # adjust parameters
            beta = min(beta * 1.001, max_beta)
            eta = eta * 0.9995
            # occasional noise
            if i % 1000 == 0 and i>0:
                a = a + np.random.randn(n) * 1e-4
                a = np.maximum(a, 0)
                a = a / a.sum()
            i += 1
        # after local descent, continue to next random start
    # final best_seq as list
    return best_seq.tolist()

"""
Propose a step‑function height sequence by minimizing the maximal
self‑convolution value.  The algorithm uses a smooth L2 objective that
penalises deviation of the convolution from a flat target (the
theoretical lower bound).  Gradient descent with an exponentiated‑gradient
update keeps the vector non‑negative and automatically respects the
normalisation Σa_i = 1, which makes the evaluation score
    score = 2·n·max(conv(a,a))
scale‑invariant.  Multiple random restarts, occasional Gaussian noise and
a simple learning‑rate schedule help escape the uniform stationary point.
The best sequence found before the time budget expires is returned.
"""

import time
import numpy as np

def _init_sequence(method: str, n: int) -> np.ndarray:
    """Generate an initial non‑negative sequence that sums to 1."""
    if method == "uniform":
        return np.full(n, 1.0 / n, dtype=np.float64)
    if method == "dirichlet":
        # Dirichlet(1,…,1) → uniform‑like but random
        return np.random.dirichlet(np.ones(n))
    # exponential‑decay shape
    r = np.random.uniform(0.70, 0.99)
    a = np.power(r, np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _gradient_l2_error(a: np.ndarray, target: float) -> np.ndarray:
    """
    Gradient of  Σ_k ( (a*a)_k - target )^2   w.r.t. a.
    Using the identity ∂/∂a_i (a*a)_k = 2·a_{k-i} (out‑of‑range entries are zero).
    The resulting gradient is 4·correlate( (a*a - target), a ).
    """
    b = np.convolve(a, a)                # length 2·n‑1
    err = b - target                     # deviation from flat target
    # correlation gives g_i = Σ_j err_{i+j} a_j  (valid mode → length n)
    grad = 4.0 * np.correlate(err, a, mode="valid")
    return grad


def _exp_gradient_update(a: np.ndarray, grad: np.ndarray,
                         lr: float, eps: float = 1e-12) -> np.ndarray:
    """
    Exponentiated‑gradient step that guarantees positivity.
    a_new_i ∝ a_i·exp(‑lr·grad_i)   and then renormalise to Σa = 1.
    The learning rate `lr` is scaled automatically to keep the largest
    multiplicative factor reasonable.
    """
    max_abs = np.max(np.abs(grad))
    if max_abs > 0:
        # scale so that the largest exponent lies roughly in [‑lr, +lr]
        scale = lr / max_abs
    else:
        scale = lr
    # avoid overflow/underflow in the exponential
    expo = -scale * grad
    # Clip extreme values before exponentiation (np.exp over ~709 overflows)
    expo = np.clip(expo, -700, 700)
    a = a * np.exp(expo)
    s = a.sum()
    if s < eps or not np.isfinite(s):
        # fallback to uniform if numerical trouble occurs
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that reduces the evaluation
    score.  The function runs for approximately `budget_s` seconds and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 3.0   # keep a tiny safety margin

    # Length of the sequence – larger n gives a lower theoretical bound.
    n = 2130   # the length of the previous best construction
    target_const = 1.0 / (2 * n - 1)          # desired flat convolution height

    best_seq = None
    best_score = float("inf")
    restart = 0

    # parameters for the optimisation loop
    init_methods = ["dirichlet", "exp"]
    base_lr = 12.0          # base learning‑rate for the exp‑gradient update
    lr_decay = 0.9995       # slow decay of the learning‑rate
    max_inner_iters = 2500  # maximum iterations per restart

    while time.time() < deadline:
        # ----- initialise a fresh start ------------------------------------
        method = np.random.choice(init_methods)
        a = _init_sequence(method, n)   # a already normalised (Σa = 1)

        lr = base_lr
        # ----- inner optimisation -----------------------------------------
        for it in range(max_inner_iters):
            if time.time() >= deadline:
                break

            # convolution and current evaluation score
            conv = np.convolve(a, a)                # length 2·n‑1
            max_conv = conv.max()
            cur_score = 2.0 * n * max_conv          # Σa = 1 ⇒ scale‑invariant

            # store the best sequence ever seen
            if cur_score < best_score:
                best_score = float(cur_score)
                best_seq = a.copy()
                print(f"[{time.strftime('%H:%M:%S')}] "
                      f"new best {best_score:.6f} (restart {restart}, iter {it})")

            # ----- gradient of the L2‑error to flat target ------------------
            grad = _gradient_l2_error(a, target_const)

            # ----- exponentiated‑gradient update ----------------------------
            a = _exp_gradient_update(a, grad, lr)

            # occasional tiny random perturbation to avoid stagnation
            if it % 300 == 0 and it > 0:
                a += np.random.randn(n) * 1e-5
                a = np.clip(a, 0.0, None)
                a /= a.sum()

            # decay learning‑rate gently
            lr *= lr_decay

        restart += 1

    # If nothing was found (should not happen), fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Return a plain Python list as required by the evaluation harness.
    return best_seq.tolist()
