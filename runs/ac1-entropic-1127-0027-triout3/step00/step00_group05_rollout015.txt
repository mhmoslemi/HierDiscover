The evaluation function (reading it) calculates:

- Takes a sequence list (potentially any length). Validates non-empty.
- For each element x, if not int/float, returns inf; if NaN or inf, return inf.
- Converts to float.
- Protect against negative numbers: replace x with max(0, x). So any negative is clamped to 0.
- Protect against numbers that are too large: clamp each x to max 1000 (min(1000, x)). So any number >1000 becomes 1000.
- n = len(sequence)
- b_sequence = np.convolve(sequence, sequence)
- max_b = max(b_sequence)
- sum_a = np.sum(sequence)
- If sum_a < 0.01, return inf.
- Returns 2 * n * max_b / (sum_a**2). So we want to minimize that.

import numpy as np
import time
import math

def _fast_score(a):
    """
    Compute the same objective as evaluate_sequence but without the safety clamping.
    a must be a 1‑D numpy array of non‑negative floats.
    """
    n = a.size
    if n == 0:
        return float('inf')
    sum_a = a.sum()
    if sum_a < 0.01:
        return float('inf')
    conv = np.convolve(a, a)
    max_b = conv.max()
    return 2.0 * n * max_b / (sum_a * sum_a)

"""
Simple multiscale projected‑subgradient optimizer.

The objective is
    F(a) = 2·n·max_k (a * a)[k] / (sum_i a_i)^2
with a_i ≥ 0.
Because the function is homogeneous we may fix sum_i a_i = 1, i.e.
    F(a) = 2·n·max_k (a * a)[k] .
For a fixed length n the problem is a non‑convex min‑max one.
We use a sub‑gradient step that reduces the currently maximal
convolution entry, project the iterate onto the simplex
({x ≥ 0, sum x = 1}) and repeat.

To obtain larger sequences without paying the full quadratic cost at
once we start from a short vector (n≈30), optimise it, then repeatedly
double its length by splitting each weight into two equal halves.
After each refinement we run a short optimisation again.
The whole procedure is bounded by the supplied time budget and the
best sequence found is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast evaluation of the target function (identical to
# evaluate_sequence but without the safety clamping).  Used inside the
# search because it is orders of magnitude faster.
def _fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a))**2  (a must be non‑negative)."""
    n = a.size
    if n == 0:
        return float("inf")
    s = a.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex {x ≥ 0, sum x = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex."""
    # implementation of Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# One optimisation run for a fixed length -------------------------------------------------
def _optimize_fixed_length(
    a: np.ndarray, max_iters: int, deadline: float
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the maximal convolution entry.
    Starts from vector a (already on the simplex) and performs at most
    max_iters iterations or until deadline is reached.
    Returns the best vector seen and its score.
    """
    n = a.size
    best_vec = a.copy()
    best_val = _fast_score(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current convolution and the index of the maximal entry
        conv = np.convolve(a, a)
        k_max = int(conv.argmax())

        # gradient of the *selected* maximal entry (sub‑gradient of the max)
        grad = np.zeros_like(a)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * a[j[mask]]

        # step size – simple linear decay
        lr = 0.5 * (1.0 - it / max_iters) + 1e-4
        a_new = a - lr * grad

        # occasional random perturbation to help escaping plateaus
        if np.random.rand() < 0.02:
            a_new += np.random.randn(n) * 1e-3
        a_new = np.maximum(a_new, 0.0)

        # project back onto the simplex (preserves the sum = 1 constraint)
        a_new = _project_simplex(a_new)

        # evaluate
        cur_val = _fast_score(a_new)
        if cur_val < best_val:
            best_val = cur_val
            best_vec = a_new.copy()

        a = a_new

    return best_vec, best_val
