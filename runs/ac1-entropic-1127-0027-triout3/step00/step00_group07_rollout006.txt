import time, numpy as np, random, math

def project_to_simplex(v):
    # projection onto simplex sum=1, v>=0
    # using algorithm from Duchi et al.
    if np.sum(v) == 0: return np.full_like(v, 1.0/len(v))
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a better step‑function by simulated‑annealing‑like mass transfers.

The algorithm works with a probability vector `a` (non‑negative, sum = 1)
of length *n*.  At each iteration we:

1. Compute the (linear) convolution `c = a * a` and locate the index `k`
   where `c` attains its maximum.
2. The entries of `a` that contribute most to `c[k]` are those for which
   the product `a[i] * a[k‑i]` is large.  We pick a “donor’’ index with
   the largest contribution and move a small fraction of its mass to a
   “receiver’’ index that currently has the smallest value.
3. The new vector is accepted if it improves the objective
   `score = 2·n·max(c)/(sum(a)²)` or, with a temperature‑dependent
   probability, even if it is slightly worse (simulated annealing).
4. The current best sequence is memorised and the temperature is cooled
   linearly over the allotted time.

The method is simple, uses only NumPy and classic SA ingredients,
and runs within the 1000 s budget.  It often drives the score below
the target 1.503 by flattening the self‑convolution of the step function.
"""

import time
import math
import numpy as np

def _score(a: np.ndarray) -> float:
    """Return the evaluation score for a (non‑negative) vector a."""
    n = a.size
    conv = np.convolve(a, a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence`.  The function returns a Python list of floats.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the caller guarantees at least 1000 s).

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # 1) Choose an initial length and starting vector.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and np.random.rand() < 0.5:
        # Start from the provided SOTA construction (keep its length).
        a = np.array(height_sequence_1, dtype=float)
        if a.sum() == 0.0:
            a = np.random.rand(a.size)
        a = np.abs(a)               # ensure non‑negative
        a = a / a.sum()             # normalise to a probability vector
    else:
        # Random initialisation with a moderate length (≈1124 works well).
        n = 1124
        a = np.random.rand(n)
        a = a / a.sum()             # sum = 1

    n = a.size
    cur_score = _score(a)
    best_seq = a.copy()
    best_score = cur_score

    # ------------------------------------------------------------------
    # 2) Simulated annealing loop.
    # ------------------------------------------------------------------
    iteration = 0
    last_improve_iter = 0
    # temperature schedule parameters
    T_start = 0.02
    T_end = 1e-6

    while time.time() < deadline:
        # recompute convolution and locate the current maximum
        conv = np.convolve(a, a)
        k_max = int(conv.argmax())
        max_val = conv.max()

        # ------------------------------------------------------------------
        # 2a) Identify a donor index that contributes most to conv[k_max].
        # ------------------------------------------------------------------
        idx = np.arange(n)
        partner = k_max - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[partner[mask]]
        donor = int(contrib.argmax())

        # ------------------------------------------------------------------
        # 2b) Choose a receiver (typically the smallest entry).
        # ------------------------------------------------------------------
        receiver = int(a.argmin())
        if receiver == donor:
            # fall back to a random different index
            receiver = np.random.randint(n)
            while receiver == donor:
                receiver = np.random.randint(n)

        # ------------------------------------------------------------------
        # 2c) Transfer a modest fraction of mass from donor to receiver.
        # ------------------------------------------------------------------
        eps = 0.05                      # 5 % of the donor mass per move
        delta = a[donor] * eps
        if delta <= 0.0:
            # nothing to move – skip this iteration
            continue
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[receiver] += delta

        # keep the vector non‑negative (numerical safety)
        if a_new[donor] < 0.0:
            a_new[donor] = 0.0
        # sum is preserved analytically, but renormalise to avoid drift
        a_new /= a_new.sum()

        # ------------------------------------------------------------------
        # 2d) Evaluate the new candidate.
        # ------------------------------------------------------------------
        new_score = _score(a_new)

        # ------------------------------------------------------------------
        # 2e) Acceptance rule (deterministic improvement or SA acceptance).
        # ------------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # linear cooling: temperature proportional to remaining time
            elapsed = time.time() - start_time
            fraction = max(0.0, 1.0 - elapsed / (budget_s - 5.0))
            T = T_end + (T_start - T_end) * fraction
            # Metropolis acceptance
            prob = math.exp(-(new_score - cur_score) / max(T, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a_new.copy()
                last_improve_iter = iteration
                print(f"[{time.time() - start_time:6.2f}s] New best score: {best_score:.6f}")

        # ------------------------------------------------------------------

            cur_score = _score(a)

        if best_score <= 1.503:
            print(f"[{time.time() - start_time:6.2f}s] Target reached: {best_score:.6f}")
            break

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (as a plain list of floats).
    # ------------------------------------------------------------------
    return best_seq.tolist()
