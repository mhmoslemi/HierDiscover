    # Optionally try hill climb from best_seq:

             idx = np.random.randint(len(seq))
             seq[idx] *= (1 + np.random.randn() * 0.01)  # small perturb
             seq[idx] = max(seq[idx],0)
         val = evaluate_sequence(seq)

              best_seq = seq
              best_eval=val

"""
Propose a good step‑function (non‑negative sequence) for the given
``evaluate_sequence``.  The algorithm works in three stages:

1. **Initialisation** – try a few exponential families a_i = r**i
   (r close to 1) for several lengths N and a few values of the
   parameter c = (1‑r)·N.  The exponential family is known to give
   evaluations ≈1.8, already better than the uniform (value 2).

2. **Gradient descent on the maximal convolution entry** – for the
   current sequence compute the convolution, locate the index k where
   it attains its maximum and move mass away from the pairs (i,
   k‑i) that contribute most to this peak.  The step size is chosen
   adaptively so that all coordinates stay non‑negative; after the
   move the vector is re‑scaled to keep its ℓ₁‑norm constant.

3. **Random local perturbations** – occasional small random redistributions
   of mass are tried to escape possible plateaus.

The whole search runs for the allotted time (default 1000 s) and the
best sequence found is returned as a plain Python list.  All numbers are
kept in the interval [0, 1000] so that ``evaluate_sequence`` does not
clip them.

The evaluation used during the search is identical to the one performed
by ``evaluate_sequence`` (except for the final clipping which is not
relevant for the generated values).

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _evaluate(arr: np.ndarray) -> float:
    """Fast version of ``evaluate_sequence`` for a numpy array."""
    s = arr.sum()
    if s < 0.01:               # matches the protection in evaluate_sequence
        return float("inf")
    conv = np.convolve(arr, arr)
    m = conv.max()
    n = arr.size
    return 2.0 * n * m / (s * s)


def _exp_sequence(N: int, c: float) -> np.ndarray:
    """
    Generate a length‑N exponential sequence a_i = r**i where
    r = 1 - c/N (c > 0).  The sequence is returned as a float64 ndarray.
    """
    eps = c / N
    r = 1.0 - eps
    if r <= 0.0:
        r = 0.0
    return np.power(r, np.arange(N, dtype=np.float64))


def _gradient_step(arr: np.ndarray, idx_max: int, factor: float = 0.5) -> np.ndarray:
    """
    Perform a single projected gradient step aimed at reducing the
    maximal convolution entry located at ``idx_max``.
    The step size is a fraction (``factor``) of the largest admissible
    step that keeps the vector non‑negative.
    """
    N = arr.size
    # gradient of the max‑convolution entry w.r.t. each coordinate
    j = idx_max - np.arange(N)
    mask = (j >= 0) & (j < N)
    grad = np.zeros_like(arr)
    grad[mask] = 2.0 * arr[j[mask]]

    # admissible step for each positive gradient component
    pos = grad > 0.0
    if not np.any(pos):
        return arr
    # avoid division by zero
    step_limits = np.where(pos, arr / grad, np.inf)
    max_step = step_limits[pos].min()
    step = factor * max_step      # be conservative to stay inside the feasible set
    if step <= 0.0:
        return arr

    new_arr = arr - step * grad
    new_arr = np.maximum(new_arr, 0.0)      # projection onto non‑negative orthant
    return new_arr


def _random_redistribute(arr: np.ndarray, frac: float = 0.05) -> np.ndarray:
    """Move a small random amount of mass from one index to another."""
    N = arr.size
    i, j = np.random.randint(N, size=2)
    if i == j or arr[i] <= 0.0:
        return arr
    delta = arr[i] * frac * np.random.rand()
    new_arr = arr.copy()
    new_arr[i] -= delta
    new_arr[j] += delta
    return new_arr


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that gives a
    low value of ``evaluate_sequence``.  The routine respects the given
    time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Stage 1 – coarse grid of exponential families
    # ------------------------------------------------------------------
    best_arr = None
    best_val = float("inf")
    candidate_N = [1500, 2000, 2500, 3000]    # moderate lengths – fast conv

    for N in candidate_N:
        # explore a few values of the shape parameter c
        for c in np.linspace(0.6, 2.2, 9):
            if time.time() > deadline:
                break
            arr = _exp_sequence(N, c)
            val = _evaluate(arr)
            if val < best_val:
                best_val = val
                best_arr = arr.copy()
                print(f"Init – N={N:4d}, c={c:.2f}, eval={val:.6f}")

    if best_arr is None:                     # fallback – uniform sequence
        N = 2000
        best_arr = np.full(N, 1.0, dtype=np.float64)
        best_val = _evaluate(best_arr)

    # ------------------------------------------------------------------
    # Stage 2 – iterative gradient descent + random perturbations
    # ------------------------------------------------------------------
    arr = best_arr.copy()
    target_len = arr.size
    total_sum = arr.sum()                     # will remain constant

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # compute the full convolution and locate the maximal entry
        conv = np.convolve(arr, arr)
        k_max = int(conv.argmax())
        cur_val = 2.0 * target_len * conv.max() / (total_sum * total_sum)

        # ----- gradient step aimed at the current maximum -----
        new_arr = _gradient_step(arr, k_max, factor=0.5)

        # re‑scale to keep the ℓ₁‑norm unchanged (the scaling does not affect
        # the evaluation because it cancels out, but we keep it for safety)
        new_sum = new_arr.sum()
        if new_sum > 0.0:
            new_arr *= (total_sum / new_sum)

        new_val = _evaluate(new_arr)

        # accept if we improve
        if new_val < best_val:
            best_val = new_val
            best_arr = new_arr.copy()
            arr = new_arr
            print(f"[{iteration:05d}] improvement → eval={best_val:.6f}")

        else:
            # with a small probability try a random redistribution
            if np.random.rand() < 0.02:
                pert = _random_redistribute(arr, frac=0.02)
                pert_val = _evaluate(pert)
                if pert_val < best_val:
                    best_val = pert_val
                    best_arr = pert.copy()
                    arr = pert
                    print(f"[{iteration:05d}] random improve → eval={best_val:.6f}")

        # occasional re‑initialisation from a fresh exponential if we are stuck
        if iteration % 2000 == 0 and time.time() < deadline:
            N = target_len
            c = np.random.uniform(0.6, 2.2)
            cand = _exp_sequence(N, c)
            cand_val = _evaluate(cand)
            if cand_val < best_val:
                best_val = cand_val
                best_arr = cand.copy()
                arr = cand
                print(f"[{iteration:05d}] restart → N={N}, c={c:.2f}, eval={best_val:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain list of floats)
    # ------------------------------------------------------------------
    print(f"Finished – best evaluation = {best_val:.6f}, length = {best_arr.size}")
    return best_arr.tolist()
