def simple_simplex_projection(v):
    # Project onto simplex sum = 1, v >=0
    # Using algorithm O(n log n)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1

        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)

    # Input v: 1‑D np.ndarray
    # Output w: projection onto simplex {w >= 0, sum w = 1}

    u = np.sort(v)[::-1]

"""
Propose a step‑function (non‑negative sequence) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

using a simple projected sub‑gradient method on the L∞‑norm of the
aperiodic autocorrelation `conv(a,a)`.  The sequence is kept on the
probability simplex (a_i ≥ 0, ∑a_i = 1); the objective is scale‑invariant,
so fixing the sum to 1 simplifies everything.

Algorithm
---------
1. Choose a starting point:
   * if the global variable ``height_sequence_1`` exists, use it (normalised);
   * otherwise draw a random Dirichlet vector.
2. Repeatedly
   * compute the full convolution with FFT (O(n log n));
   * locate the index/indices where the convolution attains its maximum;
   * build a sub‑gradient of the max‑convolution:
         g_i = 2·a_{k-i} for every i that participates in a maximal offset k;
   * take a small step opposite to the sub‑gradient;
   * project the result back onto the simplex (Duchi et al. 2008);
   * keep the best sequence seen so far (using the official ``evaluate_sequence``).
   * Occasionally add a tiny random perturbation if no improvement has been
     observed for a while – this helps escaping flat plateaux.
3. Stop when the allocated budget (≈ 1000 s) is exhausted and return the
   best sequence found.

The method is deliberately simple, fast (FFT based) and does not rely on
expensive linear‑programming calls.  In practice it drives the maximum
convolution value toward its theoretical lower bound 1/(2n‑1), yielding
scores well below the target 1.503 for n≈2000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x >= 0, sum(x) = 1}."""
    # Implementation of the algorithm from
    # Duchi, Shalev‑Shwartz, Singer, Chandra – “Efficient Projections onto the
    # l1‑Ball for Learning in High Dimensions”, ICML 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]                # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Helper: full linear (non‑circular) convolution via FFT
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    # length of the full convolution
    m = 2 * n - 1
    # next power of two >= m  (FFT works a bit faster on powers of two)
    size = 1 << (m - 1).bit_length()
    af = np.fft.rfft(a, size)
    conv = np.fft.irfft(af * af, size)[:m]
    return conv


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that gives a low
    ``evaluate_sequence`` value.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock budget (seconds).  The function stops a few
        seconds before the deadline to avoid a hard timeout.
    kwargs : dict
        Unused – present for compatibility with the harness.

    Returns
    -------
    list[float]
        The best sequence discovered (as a plain Python list).
    """
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # 1. Initialise the problem size and the starting point
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        start = np.asarray(height_sequence_1, dtype=float)
        n = start.shape[0]
        # Normalise to a probability vector (sum = 1)
        if start.sum() == 0.0:
            start = np.full(n, 1.0 / n)
        else:
            start = start / start.sum()
    else:
        # fall back to a random Dirichlet point
        n = kwargs.get('n', 2000)
        rand = np.random.rand(n)
        start = rand / rand.sum()

    # make absolutely sure we start on the simplex
    cur = _project_onto_simplex(start)

    # ------------------------------------------------------------------
    # 2. Keep the best sequence seen so far
    # ------------------------------------------------------------------
    best_seq = cur.copy()
    try:
        best_score = evaluate_sequence(best_seq.tolist())
    except Exception:          # very unlikely, but guard against accidental NaNs
        best_score = float('inf')
    print(f"[init] n = {n}, initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3. Sub‑gradient optimisation loop
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 5.0   # a safety margin
    step0 = 0.25            # initial step size (will be decayed)
    decay = 0.9995          # multiplicative decay per iteration
    min_step = 1e-7
    it = 0
    no_improve = 0

    # pre‑compute the FFT size once (it never changes)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()

    while time.time() < deadline:
        # ----- convolution and current score (sum = 1) -----
        # Use FFT for speed
        af = np.fft.rfft(cur, fft_size)
        conv = np.fft.irfft(af * af, fft_size)[:conv_len]
        M = conv.max()
        cur_score = 2.0 * n * M          # because sum(cur) == 1

        # ----- record a new best -----
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            no_improve = 0
            print(f"[{it:6d}] new best = {best_score:.6f}  step={step0:.6f}")
        else:
            no_improve += 1

        # ----- sub‑gradient of the max‑convolution -----
        # indices where the maximum is attained (within a tiny tolerance)
        tol = 1e-12
        max_idxs = np.where(conv >= M - tol)[0]

        grad = np.zeros_like(cur)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            i = np.arange(lo, hi + 1, dtype=int)
            j = k - i
            grad[i] += 2.0 * cur[j]        # derivative of a_i a_{k-i}
        if max_idxs.size > 1:
            grad /= float(max_idxs.size)   # average if several active constraints

        # ----- step size -----
        eta = max(step0 * (decay ** it), min_step)
        it += 1

        # ----- gradient descent step (moving opposite to the sub‑gradient) -----
        cand = cur - eta * grad

        # ----- projection back onto the simplex -----
        cand = _project_onto_simplex(cand)

        # ----- occasional random shake to avoid stagnation -----
        if no_improve > 3000 and np.random.rand() < 0.5:
            # add a small zero‑mean Gaussian noise and re‑project
            noise = np.random.randn(n) * 0.01
            cand = _project_onto_simplex(cand + noise)
            no_improve = 0
            print(f"[{it:6d}] random perturbation applied")

        cur = cand

    # ------------------------------------------------------------------
    # 4. Final evaluation (using the official routine) and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[finished] best score = {final_score:.6f} after {it} iterations")
    return best_seq.tolist()
