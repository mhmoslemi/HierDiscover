"""
Propose a better step‑function by a fast projected gradient descent.
The algorithm works directly on the objective 
    R(a)=2·n·max(conv(a,a))/sum(a)^2 .
It repeatedly moves mass away from indices that contribute most to the
current maximal convolution value (the “peak”) and adds it to indices that
contribute the least.  The update is projected onto the simplex (non‑negative,
unit sum) after each step.  Convolutions are performed with an FFT (O(n log n))
so thousands of iterations fit easily into the 1000 s budget.
Multiple random restarts and a few occasional random perturbations are used
to avoid local minima.  The best sequence seen is returned.
"""

import numpy as np
import time


def _fft_convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Return a * a (discrete convolution) using a real FFT.
    The length is 2*len(a)-1.
    """
    n = a.shape[0]
    # smallest power of two >= 2*n-1 for efficient FFT
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = A * A
    conv = np.fft.irfft(B, size)
    return conv[: 2 * n - 1]


def _gradient_of_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of conv_kmax (the maximal entry of the convolution) w.r.t. a.
    For each i,
        d_i = ∂ conv_kmax / ∂ a_i = a_{k_max-i} + a_{i-k_max} .
    Because a is real and the convolution is symmetric this reduces to
        d_i = 2 * a_{k_max-i}   (if the index is inside the support)
    with a single contribution when i = k_max-i.
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = k_max - idx
    mask = (j >= 0) & (j < n)

    grad = np.zeros_like(a)
    grad[mask] = a[j[mask]]
    grad[mask] *= 2.0

    # correct the double‑count on the diagonal (i == j)
    if k_max % 2 == 0:
        i_eq = k_max // 2
        if 0 <= i_eq < n:
            grad[i_eq] -= a[i_eq]   # we added it twice, keep only one copy
    return grad


def _project_onto_simplex(x: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {y | y_i >= 0, sum y_i = 1}.  Standard O(n log n) algorithm.
    """
    # make sure we work with a copy
    u = np.sort(x)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(x)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(x, 1.0 / len(x))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(x - theta, 0.0)
    return w


def propose_candidate(budget_s: float = 1000, seed: int = 42, **kwargs):

    before the time budget expires is returned as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a safety margin

    # length of the sequence – we keep the same length as the previous best
    # (the global variable height_sequence_1 is available if it exists)
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]

    best_seq = None

            except Exception:

        # ----- main optimisation loop ------------------------------------
        it = 0

            # compute convolution and locate the current maximum
            conv = _fft_convolve_self(a)

                # reject the proposal and keep the old a
                a_candidate = a
            else:

                step = min(step * step_inc, 0.5)
                a = a_new

            # occasional true evaluation of the black‑box objective
            if it % eval_every == 0:

                except Exception:
                    score = np.inf
                if score < best_score:
                    best_score = score
                    best_seq = a_candidate.copy()
                    print(f"[{time.time():.1f}] restart {restart+1}/{n_restarts} "

            # occasional random jitter to escape tiny plateaus
            if it % 500 == 0 and np.random.rand() < 0.03:
                jitter = 0.005 * np.random.randn(n)
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = _project_onto_simplex(a)

            it += 1

        # end of one restart
        if time.time() >= deadline:
            break

    # ---------------------------------------------------------------------
    # If nothing was found (highly unlikely) fall back to a simple uniform seq.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    print(f"Finished search. Best score = {best_score:.6f}")
    return best_seq.tolist()
