def compute_gradient(a, gamma):
    # compute convolution b = a * a
    n = len(a)
    size = 1 << ((2*n-2).bit_length())
    A = np.fft.rfft(a, n=size)
    b_full = np.fft.irfft(A * A, n=size)
    b = b_full[:2*n-1]  # length 2n-1
    # compute softmax weights
    # we need stable exponent: subtract max before exponent.
    b_max = np.max(b)
    exp_arg = gamma * (b - b_max) # to avoid overflow

"""
Propose a candidate step‑function sequence.

The algorithm treats the sequence as a probability vector (sum = 1) and
applies a simple mirror‑descent (multiplicative‑weights) scheme that
minimises an smoothed approximation of the objective

    f(a) = max_k (a * a)_k                  (convolution with itself)

The smoothing is performed by a soft‑max with parameter `gamma`.  The
gradient of the smoothed max is a weighted sum of the individual
convolution gradients and can be computed efficiently with FFT‑based
convolutions.  After each update the vector is projected back onto the
simplex (normalised to sum = 1).  The procedure is monotone: a new
iterate is kept only if it does not increase the true objective
`2·n·max(conv(a,a))`.  If the search stagnates a random Dirichlet restart
is performed.  The best sequence seen during the allotted time is
returned.

All operations use only NumPy and the provided `evaluate_sequence`
function; no file‑IO or external solvers are required.
"""

import time
import numpy as np

def _conv_fft(x, y):
    """
    Linear convolution of two real vectors using FFT.
    Returns the first len(x)+len(y)-1 entries.
    """
    m = len(x) + len(y) - 1
    size = 1 << ((m - 1).bit_length())          # next power of two
    X = np.fft.rfft(x, n=size)
    Y = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(X * Y, n=size)
    return conv[:m]

    """

    evaluate_sequence().  The function returns the best sequence found
    before the time budget expires.
    """

    # ------------------------------------------------------------------

        n = len(height_sequence_1)      # use the length of the previous best if available
    except Exception:

    # 2) Hyper‑parameters.
    # ------------------------------------------------------------------
    gamma = 30.0            # smoothing for the max (higher => tighter approximation)
    eta0 = 2.0              # initial step size for the mirror‑descent update
    max_no_improve = 3000  # restart after this many non‑improving iterations

    # ------------------------------------------------------------------
    # 3) Initialise the search.
    # ------------------------------------------------------------------
    a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start (sum = 1)
    b = _conv_fft(a, a)
    best_score = 2.0 * n * np.max(b)            # = evaluate_sequence(a) because sum(a)=1
    best_seq = a.copy()
    curr_score = best_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin
    it = 0
    stagnant = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 4) Compute convolution and the smoothed max (soft‑max) weights.
        # ------------------------------------------------------------------
        b = _conv_fft(a, a)                     # length 2n‑1
        max_b = np.max(b)

        # soft‑max over the convolution values
        b_max = np.max(b)
        # avoid overflow: exp(gamma*(b-b_max)) ∈ (0,1]
        exp_arg = gamma * (b - b_max)
        exp_arg = np.clip(exp_arg, -700, 0)    # np.exp underflows safely
        w = np.exp(exp_arg)
        w_sum = np.sum(w)
        if w_sum == 0.0:                       # extremely unlikely, but be safe
            w = np.full_like(b, 1.0 / b.size)
        else:
            w /= w_sum

        # ------------------------------------------------------------------
        # 5) Gradient of the smoothed max w.r.t. a.
        #    ∂b_k/∂a_i = 2·a_{k-i} (when the index is valid)
        #    Hence grad = 2 * (w * a_rev) shifted appropriately.
        # ------------------------------------------------------------------
        a_rev = a[::-1]
