def local_move(seq, step_frac=0.01):
    """Perform a deterministic move reducing max convolution."""
    a = np.array(seq, dtype=np.float64)
    n = len(a)
    conv = np.convolve(a, a)
    max_idx = np.argmax(conv)
    # compute contributions as vector length n
    low = max(0, max_idx-(n-1))
    high = min(n-1, max_idx)
    idx = np.arange(low, high+1)
    j = max_idx - idx
    contributions = np.zeros(n)
    contributions[idx] = a[idx] * a[j]
    # Determine donor and receiver
    # donor = index with max contribution (and a[i] > 0)
    donor_candidates = np.where(contributions > 0)[0]
    if donor_candidates.size == 0:
        return seq
    donor = donor_candidates[np.argmax(contributions[donor_candidates])]
    # receiver = index with minimal contribution (could be zero)
    receiver = np.argmin(contributions)
    # Determine amount delta to move
    total_sum = np.sum(a)
    delta = step_frac * total_sum  # amount total to move from donor to receiver
    # Bound delta by available at donor
    delta = min(delta, a[donor])
    # Ensure non-negative receiver after addition (no problem)
    a[donor] -= delta
    a[receiver] += delta
    return a.tolist()

"""
Simple hybrid optimizer for the step‑function inequality.

The evaluation to be minimized is

    E(a) = 2 * n * max(conv(a,a)) / (sum(a)**2)

which is homogeneous: scaling the whole vector does not change the score.
The algorithm starts from a random (or the provided) sequence and repeatedly
tries two kinds of moves:

1. **Local mass transfer** – locate the shift where the convolution is
   maximal, compute the contribution a_i * a_{k‑i} of each entry and move a
   small amount of mass from the index with the largest contribution to the
   index with the smallest one.  The total sum of the vector stays unchanged
   and the move is accepted only if it does not increase the score
   (simulated‑annealing style fallback is also used).

2. **LP direction** – solve a linear programme that finds a vector `g` with
   the largest possible sum while keeping `conv(f,g) ≤ max(conv(f,f))`.
   Mixing the current vector `f` with the normalised `g` (a small step `t`)
   strictly reduces the bound on the convolution, as shown in the SOTA
   approach.  The LP is invoked only occasionally because it is relatively
   expensive.

The search runs for the allotted time (default 1000 s).  Several restarts
with different lengths and random seeds are performed; the best sequence
found is returned.  All entries are forced to be non‑negative; the
evaluation function itself also clips negative values to zero.

The procedure usually reduces the score well below the uniform bound `2.0`
and often reaches the target `1.5` in a few hundred seconds.
"""

import time
import random

import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: linear programme used by the LP direction
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs):
    """
    Solve:
        maximise   sum(b)
        subject to conv(f_seq, b) <= rhs   (for every shift)
                   b >= 0
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_seq)
    # objective (minimise -sum(b) to maximise sum(b))
    c = -np.ones(n)

    # Inequalities for each convolution shift
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # fill the anti‑diagonal corresponding to shift k
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # i runs from i_low .. i_high, and j = k - i is within [0,n)
        for i in range(i_low, i_high + 1):
            j = k - i
            row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  -b <= 0
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # solve with the default ‘highs’ method; limit the time spent per LP
    options = {"maxiter": 2000}
    result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method="highs", options=options)

    if result.success:
        return result.x
    return None


# ----------------------------------------------------------------------
# LP based direction (identical to the “good direction” from the SOTA code)
# ----------------------------------------------------------------------
def get_good_direction_to_move_into(seq, t=0.05):
    """
    Given a sequence seq, compute a new sequence by moving a small step `t`
    towards the LP direction that maximises the sum while preserving the
    current convolution bound.
    """
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-9:          # degenerate case – bail out
        return None

    # normalise to the canonical sum sqrt(2n) (the SOTA normalisation)
    norm_factor = np.sqrt(2.0 * n) / sum_seq
    f_norm = np.array(seq) * norm_factor

    # current bound M
    M = np.max(np.convolve(f_norm, f_norm))

    # solve the LP
    g = solve_convolution_lp(f_norm, M)
    if g is None:
        return None

    # normalise g to the same canonical sum
    sum_g = np.sum(g)
    if sum_g < 1e-12:
        return None
    g_norm = g * (np.sqrt(2.0 * n) / sum_g)

    # convex combination
    new_seq = (1.0 - t) * np.array(seq) + t * g_norm
    # clip tiny negatives that may appear due to numerical errors
    new_seq = np.maximum(new_seq, 0.0)
    return new_seq.tolist()


# ----------------------------------------------------------------------
# Pure local move: shift a small amount of mass from the biggest
# contribution to the shift where conv is maximal, to the smallest one.
# ----------------------------------------------------------------------
def local_move(seq, step_frac=0.01):
    """
    Perform a deterministic mass‑transfer move.
    A fraction `step_frac` of the total sum is taken from the index that
    contributes most to the maximal convolution entry and given to the
    index that contributes the least.
    """
    a = np.array(seq, dtype=np.float64)
    n = a.shape[0]

    conv = np.convolve(a, a)
    k_max = int(np.argmax(conv))

    # contributions of each index to conv[k_max]
    low_i = max(0, k_max - (n - 1))
    high_i = min(n - 1, k_max)
    i_idx = np.arange(low_i, high_i + 1)
    j_idx = k_max - i_idx
    contrib = np.zeros(n, dtype=np.float64)
    contrib[i_idx] = a[i_idx] * a[j_idx]

    # donor: index with largest contribution (and positive mass)
    donors = np.where(contrib > 0)[0]
    if donors.size == 0:        # degenerate – fallback to a random move
        donor = np.random.randint(n)
    else:
        donor = donors[np.argmax(contrib[donors])]

    # receiver: index with smallest contribution (could be zero)
    receiver = int(np.argmin(contrib))

    # amount to move – a fraction of the total sum, clipped by donor’s mass
    tot = np.sum(a)
    delta = step_frac * tot
    delta = min(delta, a[donor])
    if delta <= 0:
        return seq  # nothing to do

    a[donor] -= delta
    a[receiver] += delta
    return a.tolist()

        else:
            # occasionally try a slightly different size to explore
            n = max(200, min(1500, int(len(best_seq) * (0.9 + 0.2 * random.random()))))

        # initialise the sequence -------------------------------------------------
        if 'height_sequence_1' in globals() and random.random() < 0.4:
            # use the provided construction as a warm start (trim / pad as needed)
            init = list(height_sequence_1)
            if len(init) > n:
                init = init[:n]
            elif len(init) < n:
                init = init + [0.0] * (n - len(init))
        else:
            # random Dirichlet‑like start (exponential samples)
            init = np.random.exponential(scale=1.0, size=n).tolist()

        cur_seq = init
        cur_score = safe_evaluate(cur_seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[INIT] New best score {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        # inner optimisation loop for this restart
        # ------------------------------------------------------------------
        it = 0
        while time.time() < deadline and it < max_inner_iters:
            if random.random() < lp_probability:
                # LP guided move
                cand = get_good_direction_to_move_into(cur_seq, t=lp_step)
                if cand is None:
                    # fall back to a random local perturbation
                    noise = np.random.normal(scale=0.01, size=n)
                    cand = (np.array(cur_seq) + noise).clip(min=0).tolist()
            else:
                # cheap local move
                cand = local_move(cur_seq, step_frac=move_step_frac)

                # occasional tiny random jitter to escape plateaus
                if random.random() < 0.03:
                    jitter = np.random.normal(scale=0.002, size=n)
                    cand = (np.array(cand) + jitter).clip(min=0).tolist()

            cand_score = safe_evaluate(cand)

            # keep the global best
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[IMPR] Improved best to {best_score:.6f} (n={len(best_seq)})")

            # accept the move if it does not worsen the current score
            # (with a tiny probability to accept a worse move – simple annealing)
            if cand_score <= cur_score or random.random() < 0.005:
                cur_seq = cand
                cur_score = cand_score

            # slowly decay the step size to fine‑tune at the end of the search
            if it % 2000 == 0 and it > 0:
                move_step_frac *= 0.9
                lp_step = max(0.01, lp_step * 0.95)

            it += 1

        # after one restart, try another one with a fresh random seed
        # (the loop condition already checks the global deadline)

    # ------------------------------------------------------------------
    # fallback if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * 500
        best_score = safe_evaluate(best_seq)

    print(f"[DONE] Best score found: {best_score:.6f} (length={len(best_seq)})")
    return best_seq
