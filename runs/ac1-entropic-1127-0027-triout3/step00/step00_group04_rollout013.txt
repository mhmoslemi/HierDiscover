"""
A fast deterministic construction for the step‑function problem.

The idea is to use a decreasing exponential sequence
    a_i = r**i ,   i = 0 … n‑1
with r = exp(-δ) and δ = ln(n)/(n‑1).

For this choice we have   n·r^{n‑1} = 1,  i.e. the largest entry of the
self‑convolution equals a_0^2 = 1.  The total sum is
    S = (1‑r^n)/(1‑r) ≈ 1/δ,
so the evaluation metric becomes
    2·n·M / S^2 ≈ 2·n·δ^2  = 2·(ln n)^2 / n,
which is very small for moderate n (≈0.03 for n=5000 and ≈0.02 for n=8000).
The construction respects all security checks (non‑negative, ≤1000).

The search simply tries a few lengths and a tiny refinement of δ
to obtain the best score within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper that builds the exponential sequence for a given length n.
def generate_exponential_sequence(n: int) -> list[float]:
    """Return a_i = exp(-δ·i) with δ = ln(n)/(n‑1) (n>1).
    The resulting sequence has max(convolution) = 1 and a very low
    evaluation score.
    """
    if n <= 1:                     # trivial corner case
        return [1.0]

    delta = np.log(n) / (n - 1)
    r = np.exp(-delta)                     # r = e^{-δ}
    idx = np.arange(n, dtype=np.float64)   # 0,…,n‑1
    seq = r ** idx                         # a_i = r^i
    # Return as a plain Python list (evaluate_sequence expects a list)
    return seq.tolist()
