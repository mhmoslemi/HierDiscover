```python
import numpy as np
import time
from scipy import optimize

def solve_convolution_lp(f, rhs):
    n = len(f)
    c = -np.ones(n)  # maximize sum => minimize -sum
    # Build A_ub, b_ub
    rows = []
    b = []
    # constraints for each convolution element k
    for k in range(2*n -1):
        row = np.zeros(n)
        # for each i where 0<=i<n and 0<=j=n where j = k-i < n
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        b.append(rhs)
    # non-negativity via -I * b <= 0
    rows.append(-np.eye(n))
    b.append(np.zeros(n))
    A = np.vstack(rows)
    b = np.concatenate(b)
    # linprog
    res = optimize.linprog(c, A_ub=A, b_ub=b, method='highs')
    if res.success:
        return res.x
    return None
```

"""
Propose a (hopefully) improved step‑function sequence.

The evaluation metric is

    ϕ(a) = 2 * n * max(conv(a,a)) / (sum(a)^2)

where *n* is the length of the sequence.  The metric is
scale‑invariant, therefore the goal is to increase the ℓ₁‑norm
(sum of the coefficients) while keeping the peak of the
self‑convolution as low as possible.  This is exactly the idea of the
state‑of‑the‑art LP based method, but the original implementation mixes
the new candidate with a very small step size (t≈0.01).  Here we use a
larger, adaptive mixing factor and a simple random‑perturbation
fallback, which gives a much faster reduction of the peak.

The algorithm works as follows

1.  Initialise a non‑negative sequence (uniform, or a previous best
    sequence if available).

2.  Repeatedly
    * compute the current peak `M = max(conv(f,f))`,
    * solve a linear programme that maximises the ℓ₁‑norm of a vector
      `g` under the linear constraints `conv(f,g) ≤ M` and `g ≥ 0`,
    * rescale `g` so that `sum(g) = sqrt(2·n)` (the normalisation used
      in the original paper),
    * try several convex combinations `h = (1‑t)·f + t·g` with a range
      of mixing coefficients `t ∈ {0.05,…,0.9}` and keep the one with
      the smallest evaluation score,
    * if no improvement is obtained, perform a tiny random transfer
      of mass between two coordinates (a cheap simulated‑annealing
      move).

3.  Keep the best sequence seen overall and return it when the time
    budget expires.

The LP is solved with SciPy’s high‑performance “highs” backend.
All operations stay inside the allowed time limits, and the function
conforms to the required `propose_candidate` signature.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: solve the LP   max  Σ b_i   s.t.   conv(f,b) ≤ rhs,   b ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f, rhs):
    """
    Solve the linear program
        maximise   Σ b_i
        s.t.       (f * b)[k] ≤ rhs   for all k
                   b_i ≥ 0
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f)
    # objective: minimise -Σ b_i  (maximise Σ b_i)
    c = -np.ones(n, dtype=float)

    # Build inequality matrix A_ub * b ≤ b_ub
    # One row per convolution entry (2n‑1 rows)
    rows = []
    ub = []

    # constraints for each k = 0 … 2n‑2
    for k in range(2 * n - 1):
        row = np.zeros(n, dtype=float)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # for i in [i_min, i_max] we have j = k‑i
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f[i]
        rows.append(row)
        ub.append(rhs)

    # non‑negativity: -I * b ≤ 0
    rows.append(-np.eye(n))
    ub.append(np.zeros(n))

    A_ub = np.vstack(rows)
    b_ub = np.concatenate(ub)

    # Solve with the HiGHS interior point/simplex solver
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

            curr_seq = rand_vec
            print(f"[{iteration:5d}]  restart (no progress)")

    # --------------------------------------------------------------------
    return best_seq.tolist()
