"""
Propose a step‐function sequence that (approximately) minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a)**2)

The routine works as follows

1.  Work with sequences that sum to 1 (the objective is scale invariant).
2.  Initialise a random probability vector (Dirichlet) of moderate length.
3.  Perform projected gradient descent on a smooth approximation of the
    ℓ∞‑norm of the self‑convolution.
    * For a sequence a, let b = conv(a, a).
    * Approximate max(b) by a soft‑max:
          η(b) = (1/α)·log Σ exp(α·b_k)      (α > 0, large → true max)
    * The gradient of η w.r.t. a can be written in closed form:
          ∂η/∂a_i = 2· Σ_j w_{i+j}·a_j
      where w_k = exp(α·b_k)/Σ exp(α·b) are the soft‑max weights.
    * After a gradient step we project back onto the probability simplex
      (non‑negative entries that sum to 1).
4.  The step size and the soft‑max temperature α are adapted during the run.
5.  Several random restarts (different lengths, different seeds) are performed
    until the time budget expires.  The best sequence found (lowest score)
    is returned.

The algorithm is lightweight (O(n²) per iteration) and works well for n≈
few hundred, giving a noticeable improvement over the trivial uniform
sequence (score ≈ 2).  The final sequence is a plain Python list of
non‑negative floats, ready to be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper: projection onto the probability simplex {x ≥ 0, sum(x)=1}
# ------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    # From: Wang & Carreira‑Perpiñán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / n)
    return w


# ------------------------------------------------------------
# Gradient of the smoothed max‑convolution objective
# ------------------------------------------------------------
def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∂η/∂a, where η = softmax_α(conv(a,a)).
    The formula is 2 * Σ_j w_{i+j} * a_j .
    """
    n = a.shape[0]
    # indices for broadcasting: i (rows) + j (columns)
    j = np.arange(n)
    i = np.arange(n)[:, None]          # shape (n,1)
    k = i + j                           # shape (n,n), values in [0, 2n-2]
    # weighted sum for each i
    grad = 2.0 * np.sum(w[k] * a[j], axis=1)
    return grad


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """Return a sequence of non‑negative heights with a low evaluation score.

    The function respects the global time budget (budget_s).  It uses several
    random restarts, projected gradient descent on a smooth ℓ∞‑approximation,
    and keeps the best candidate found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = float("inf")

    # if a previous solution is available, seed the best
    if "height_sequence_1" in globals():
        try:
            hs = np.asarray(height_sequence_1, dtype=float)
            if hs.size > 0:
                hs = np.maximum(0.0, hs)          # safety
                s = hs.sum()
                if s > 0:
                    a0 = hs / s                     # normalise to sum=1
                    score0 = 2.0 * len(a0) * np.convolve(a0, a0).max()
                    best_seq = a0.copy()
                    best_score = float(score0)
                    print(f"Initial best from height_sequence_1: {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------
    # Parameters for the optimisation loop
    # --------------------------------------------------------
    min_n, max_n = 200, 800                 # problem size range
    max_inner_iters = 2000                  # per restart iteration budget
    alpha_start = 5.0                       # initial soft‑max temperature
    alpha_max = 50.0
    step_start = 0.5                        # initial step size (in parameter space)

    # --------------------------------------------------------
    # Main search loop (random restarts)
    # --------------------------------------------------------
    while time.time() < deadline:
        # ---- choose a problem size and initialise a random a ----
        n = np.random.randint(min_n, max_n + 1)
        a = np.random.rand(n)
        a = a / a.sum()                      # probability vector (sum = 1)

        alpha = alpha_start
        step = step_start

        # pre‑compute index matrix for the gradient (re‑used each iteration)
        j_idx = np.arange(n)
        i_idx = np.arange(n)[:, None]        # shape (n,1)

        # ---- inner optimisation loop ----
