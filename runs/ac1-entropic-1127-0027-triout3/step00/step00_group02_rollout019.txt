def propose_candidate(seed=42, budget_s=900, **kwargs):
    import numpy as np, time, random
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 2000   # length
    # initial shape
    a = 1.0 / np.sqrt(np.arange(1, n+1))
    # optionally random shuffle initial positions
    # Let's start with a random permutation to avoid bias
    perm = np.random.permutation(n)
    a = a[perm]
    # evaluate
    best_seq = a.tolist()
    best_score = evaluate_sequence(best_seq)
    curr_seq = a.copy()
    curr_score = best_score
    # temperature schedule
    init_temp = 1.0
    while time.time() < deadline:
        # compute convolution and max index
        conv = np.convolve(curr_seq, curr_seq)
        k_max = int(np.argmax(conv))
        # find indices in pair set
        i_min = max(0, k_max - (n-1))
        i_max = min(n-1, k_max)
        # choose i with largest value in this range
        # also choose j outside this range with smallest value
        i_candidates = np.arange(i_min, i_max+1)
        # find i with largest a_i in this set
        i_vals = curr_seq[i_candidates]
        i_max_idx = i_candidates[np.argmax(i_vals)]
        # pick j outside range randomly
        # provide complement indices
        # we can try a handful of attempts to find a beneficial swap
        found = False
        for _ in range(5):
            j = np.random.randint(n)
            if j < i_min or j > i_max:
                # propose swap
                new_seq = curr_seq.copy()
                new_seq[i_max_idx], new_seq[j] = new_seq[j], new_seq[i_max_idx]
                new_score = evaluate_sequence(new_seq.tolist())
                # acceptance
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.tolist()
                    print(f"New best score {best_score:.6f} at time {time.time():.2f}")
                # use Metropolis
                delta = new_score - curr_score
                # temperature decays linearly
                remaining = deadline - time.time()
                T = init_temp * (remaining / budget_s)
                accept = False
                if delta < 0:
                    accept = True
                else:
                    if T > 0 and np.exp(-delta / T) > np.random.rand():
                        accept = True
                if accept:
                    curr_seq = new_seq
                    curr_score = new_score
                found = True
                break
        if not found:
            # fallback random tweak
            idx = np.random.randint(n)
            delta = np.random.randn() * 0.01
            new_seq = curr_seq.copy()
            new_seq[idx] = max(0.0, new_seq[idx] + delta)
            new_score = evaluate_sequence(new_seq.tolist())
            if new_score < best_score:

                best_seq = new_seq.tolist()
                print(f"New best score {best_score:.6f} (random tweak)")
            if new_score < curr_score:
                curr_seq = new_seq
                curr_score = new_score
    return best_seq

"""
propose_candidate

Search for a sequence of non‑negative heights that minimises the
evaluation function

    2 * n * max(conv(sequence,sequence)) / (sum(sequence)**2)

The algorithm starts from a analytically motivated shape
a_i ≈ 1/√(i+1) (the discrete analogue of the continuous
solution whose self‑convolution is uniform).  This gives an
initial value ≈ π/2 ≈ 1.57, already better than the constant
sequence (value 2).

To improve further we perform a simulated‑annealing hill‑climbing
procedure:

* In each iteration the current convolution is computed and the
  index k where it attains its maximum is identified.
* Among the positions that contribute to this maximal sum we pick
  the largest entry and try to swap it with a position outside the
  contributing set (this directly reduces the peak).
* The swap is accepted if it improves the score, otherwise it may
  be accepted with a Metropolis probability that slowly cools.
* Occasionally random perturbations or completely random swaps are
  tried to escape local minima.

All operations stay within the allowed time budget; the best
sequence found is returned as a plain Python list.
"""

import time
import math
import numpy as np

def _fast_evaluate(seq: np.ndarray) -> float:
    """
    Compute the evaluation value without the security wrappers.
    The sequence is assumed to be non‑negative and finite.
    """
    n = len(seq)
    sum_a = np.sum(seq)
    if sum_a < 0.01:
        return float('inf')
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a * sum_a))


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # give a small safety margin so the function finishes cleanly
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # 1.  Build an analytically motivated starting point.
    # ------------------------------------------------------------------
    n = 2000                     # length of the sequence (tunable)
    # a_i ≈ 1 / sqrt(i+1)  (discrete analogue of the continuous optimal shape)
    base = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))

    # Interleave large and small values – a cheap deterministic perturbation.
    seq = np.empty_like(base)
    left, right = 0, n - 1
    for idx in range(n):
        if idx % 2 == 0:
            seq[idx] = base[left]
            left += 1
        else:
            seq[idx] = base[right]
            right -= 1

    # Keep the best sequence we have seen so far.
    best_seq = seq.copy()
    best_score = _fast_evaluate(best_seq)

    # Try a few random permutations of the same multiset – cheap
    # diversification that sometimes yields a better start.
    for _ in range(6):
        perm = np.random.permutation(n)
        cand = base[perm]
        cand_score = _fast_evaluate(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 2.  Simulated‑annealing hill climbing.
    # ------------------------------------------------------------------
    curr_seq = best_seq.copy()
    curr_score = best_score
    init_temp = 1.0

    while time.time() < deadline:
        # current convolution and its maximal index
        conv = np.convolve(curr_seq, curr_seq)
        k_max = int(np.argmax(conv))

        # indices that contribute to the maximal sum (i + j = k_max)
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        contributing = np.arange(i_min, i_max + 1, dtype=int)

        # pick the largest entry among the contributors
        i_large = contributing[np.argmax(curr_seq[contributing])]

        # ------------------------------------------------------------------
        # 2a. Targeted swap: move that large entry out of the offending set.
        # ------------------------------------------------------------------
        swapped = False
        for _ in range(3):
            j = np.random.randint(n)
            if j < i_min or j > i_max:          # j does NOT belong to the offending set
                cand_seq = curr_seq.copy()
                cand_seq[i_large], cand_seq[j] = cand_seq[j], cand_seq[i_large]

                cand_score = _fast_evaluate(cand_seq)

                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = cand_seq.copy()
                    print(f"[{time.time() - start_time:6.2f}s] New best (swap) = {best_score:.6f}")

                # Metropolis acceptance
                delta = cand_score - curr_score
                remaining = max(0.0, deadline - time.time())
                T = init_temp * (remaining / budget_s)
                accept = False
                if delta < 0:
                    accept = True
                elif T > 1e-12 and math.exp(-delta / T) > np.random.rand():
                    accept = True

                if accept:
                    curr_seq = cand_seq
                    curr_score = cand_score
                swapped = True
                break

        if swapped:
            continue

        # ------------------------------------------------------------------
        # 2b. Random perturbation (rare) – helps to escape plateaus.
        # ------------------------------------------------------------------
        if np.random.rand() < 0.02:
            idx = np.random.randint(n)
            # multiplicative Gaussian perturbation, keep non‑negative
            delta = np.random.randn() * 0.05 * max(1e-12, curr_seq[idx])
            cand_seq = curr_seq.copy()
            cand_seq[idx] = max(0.0, cand_seq[idx] + delta)

            cand_score = _fast_evaluate(cand_seq)

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand_seq.copy()
                print(f"[{time.time() - start_time:6.2f}s] New best (perturb) = {best_score:.6f}")

            if cand_score < curr_score:
                curr_seq = cand_seq
                curr_score = cand_score
            continue

        # ------------------------------------------------------------------
        # 2c. Pure random swap as a fallback.
        # ------------------------------------------------------------------
        i, j = np.random.choice(n, 2, replace=False)
        cand_seq = curr_seq.copy()
        cand_seq[i], cand_seq[j] = cand_seq[j], cand_seq[i]

        cand_score = _fast_evaluate(cand_seq)

        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand_seq.copy()
            print(f"[{time.time() - start_time:6.2f}s] New best (rand‑swap) = {best_score:.6f}")

        delta = cand_score - curr_score
        remaining = max(0.0, deadline - time.time())
        T = init_temp * (remaining / budget_s)
        if delta < 0 or (T > 1e-12 and math.exp(-delta / T) > np.random.rand()):
            curr_seq = cand_seq
            curr_score = cand_score

    # ------------------------------------------------------------------
    # 3.  Return the best sequence discovered.
    # ------------------------------------------------------------------
    # Convert to a plain Python list as required by the evaluation harness.
    return best_seq.tolist()
