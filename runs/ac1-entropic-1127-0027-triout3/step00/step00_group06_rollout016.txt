def propose_candidate(seed=123, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    best_score = float('inf')
    best_seq = None
    
    # choose n maybe 2000. Could be adaptively set.
    n = 2000
    target_sum = np.sqrt(2 * n)
    
    # initialize uniform
    a = np.full(n, target_sum / n)
    
    # hyperparameters
    beta = 20.0  # smoothing for softmax
    eta = 0.1    # step size initial
    
    # projection onto simplex function
    def project_simplex(v, s):
        # returns w, projection onto {w>=0, sum w = s}
        # Implementation using sorted v.
        nlen = len(v)
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - s
        ind = np.arange(nlen) + 1
        cond = u - cssv / ind > 0
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        w = np.maximum(v - theta, 0)
        return w

    # iterate
    while time.time() < deadline:
        # compute convolution via FFT
        # pad length for FFT
        L = 1
        while L < 2*n:
            L <<= 1
        fft_a = np.fft.rfft(a, n=L)
        b = np.fft.irfft(fft_a * fft_a, n=L)[:2*n-1]
        M = b.max()
        score = 2 * n * M / (target_sum ** 2)
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"New best score {best_score:.6f} at time {time.time()-start:.1f}s")
            # early exit if good enough
            if best_score < 1.503:
                break
        
        # softmax weights over b
        max_b = M
        # compute probabilities
        # avoid overflow, subtract max_b
        exp_vals = np.exp(beta * (b - max_b))
        p = exp_vals / exp_vals.sum()
        # compute gradient
        # compute a_rev
        a_rev = a[::-1]
        # convolution of a_rev and p
        # we need length L conv size >= n + (2n-1) -1 = 3n-2
        L2 = 1
        while L2 < 3*n:
            L2 <<= 1
        fft_rev = np.fft.rfft(a_rev, n=L2)
        fft_p = np.fft.rfft(p, n=L2)
        conv_rev = np.fft.irfft(fft_rev * fft_p, n=L2)
        # extract g_i (size n)
        g = 2.0 * conv_rev[n-1 : n-1 + n]
        # gradient direction: move opposite to g (since increasing a_i increase M)
        # But need to remove component along all-ones to keep sum unchanged: subtract mean
        # Actually projection will handle sum constraint.
        v = a - eta * g
        # project onto simplex with sum = target_sum
        a = project_simplex(v, target_sum)
        # adapt step size maybe
        # reduce eta after many iterations
        if (np.random.rand() < 0.01):
            # random perturbation to escape local minima
            a += np.random.randn(n) * 0.01 * target_sum / n
            a = np.maximum(a, 0)
            a = project_simplex(a, target_sum)
        # decaying eta
        eta = max(1e-4, eta * 0.9999)
    
    # if best_seq still None (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, target_sum / n)
    return best_seq.tolist()

"""
Propose a better step‑function sequence.

The algorithm is a simple iterative improvement scheme based on the
LP‑direction described in the statement:

1.  Start from the uniform sequence of length ``n`` with the required
    ℓ₁‑norm ``sqrt(2 n)`` (this gives the trivial bound 2).

2.  Given the current sequence ``f`` compute the maximal convolution value
    ``M = max(f*f)``.

3.  Solve the linear program

        maximise   Σ g_j
        subject to  (f * g)_k ≤ M   for k = 0,…,2n‑2
                    g_j ≥ 0

   The LP gives a non‑negative direction ``g`` whose sum can be larger
   than the sum of ``f`` while keeping the mixed convolution bounded by
   ``M``.

4.  Rescale ``g`` so that it has the same ℓ₁‑norm as ``f`` (the value
    required by ``evaluate_sequence``) and move a small step towards it:
    ``f ← (1‑t) f + t g``.  The step size ``t`` is adapted – it is increased
    when the move reduces ``M`` and decreased otherwise.

5.  The whole loop is repeated until the time budget expires.  Whenever a
    better score is observed (according to ``evaluate_sequence``) it is
    saved.  Random perturbations are occasionally performed to escape
    possible plateaus.

The procedure is applied to several problem sizes (n = 256, 512, 1024)
and the best sequence found overall is returned.  All operations respect
the non‑negativity requirement; values never exceed the ``evaluate_sequence``
clipping limit (1000).

The implementation uses only NumPy, SciPy (linprog) and the already
imported ``evaluate_sequence`` function.

def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Project a vector onto the simplex {x ≥ 0, Σ x = s}.
    Implementation of the O(n log n) algorithm by
    Michelot/Condat.
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    # Find the largest rho satisfying the condition
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(seq: np.ndarray) -> float:
    """Maximum value of the auto‑convolution of *seq*."""
    n = seq.shape[0]
    # FFT‑based convolution (real → real)
    L = 1
    while L < 2 * n:
        L <<= 1
    fhat = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fhat * fhat, n=L)[: 2 * n - 1]
    return conv.max()


def _solve_conv_lp(fseq: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the LP  maximise Σ b_j  subject to   (fseq * b)_k ≤ rhs,
    b_j ≥ 0  (Linear constraints only; the product is linear because the
    current ``fseq`` is fixed).

    Returns the optimal vector ``b`` or ``None`` if the solver fails.
    """
    n = fseq.shape[0]
    # Build the (2n‑1) × n constraint matrix A where
    #   (A @ b)[k] = Σ_i f_i * b_{k‑i}
    # positions outside the range are ignored (treated as zero).
    A = np.zeros((2 * n - 1, n), dtype=np.float64)

    # Fill rows – each row corresponds to a concrete convolution indexk.
    for k in range(2 * n - 1):
        # i ranges from max(0, k‑(n‑1)) to min(n‑1, k)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # j = k‑i
        i_vals = np.arange(i_min, i_max + 1, dtype=np.intp)
        j_vals = k - i_vals
        A[k, j_vals] = fseq[i_vals]

    # Linear programme data
    c = -np.ones(n)                    # maximise Σ b_j  ⇔ minimise −Σ b_j
    b_ub = np.full(2 * n - 1, rhs)    # RHS for the convolution constraints
    bounds = [(0.0, None)] * n         # b_j ≥ 0

    # Solve with HiGHS (fast default for SciPy ≥1.6)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

            cur_score = 2.0 * n * cur_max / (target_sum ** 2)

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[{time.time() - start_time:5.1f}s] "
                      f"new best (n={n})  score={best_score:.6f}")
                # stop early if we already beat the target
                if best_score <= 1.503:
                    return best_seq.tolist()

            # --- compute a direction via LP ---------------------------------
            g_raw = _solve_conv_lp(a, cur_max)
            if g_raw is None or g_raw.sum() == 0.0:
                # LP failed – fall back to a small random perturbation
                idx = np.random.randint(n)
                delta = np.random.randn() * a[idx] * 0.02
                a[idx] = max(0.0, a[idx] + delta)
                a = a * (target_sum / a.sum())
                continue

            # Rescale the LP solution so it has the same ℓ₁‑norm as ``a``
            g = g_raw * (target_sum / g_raw.sum())

            # --- try a convex step towards ``g`` ---------------------------
            a_candidate = (1.0 - t) * a + t * g
            a_candidate = np.maximum(a_candidate, 0.0)   # safety
            # (both a and g have the required sum, convex combo preserves it)

            cand_max = _conv_max(a_candidate)

            # Accept the step if it does not increase the maximal convolution.
            if cand_max <= cur_max:
                a = a_candidate
                # slightly increase the step size to accelerate progress
                t = min(0.5, t * 1.07)
                stagnation = 0
            else:
                # step was too aggressive – shrink ``t`` and try again later
                t = t * 0.5
                stagnation += 1

                # If we have been stuck for a while, inject random noise
                if stagnation >= 30:
                    a = np.full(n, target_sum / n, dtype=np.float64)
                    a += np.random.randn(n) * target_sum / n * 0.08
                    a = np.maximum(a, 0.0)
                    a = a * (target_sum / a.sum())
                    t = 0.05
                    stagnation = 0

    # ------------------------------------------------------------------
    # If nothing better than the trivial uniform vector was found, fall
    # back to the uniform construction (score exactly 2.0).
    # ------------------------------------------------------------------
    if best_seq is None:
        # default to a uniform sequence of length 256 (any length works)
        n = 256
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    return best_seq.tolist()
