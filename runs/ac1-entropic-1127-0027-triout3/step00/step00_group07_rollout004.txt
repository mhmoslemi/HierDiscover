    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    # Precompute sum for each iteration
    while time.time() < deadline:
        # compute conv
        conv = np.convolve(seq, seq)
        max_val = conv.max()
        k_max = conv.argmax()
        # gradient
        idx = np.arange(n)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        grad = seq.copy()
        grad[valid] += seq[j[valid]]
        # pick high/low
        i_high = np.argmax(grad)
        i_low = np.argmin(grad)
        if i_high == i_low:
            # random perturb
            i_high = np.random.randint(n)
            i_low = np.random.randint(n)
        delta = min(seq[i_high] * 0.2, 0.1)  # ensure not too big

            seq[np.random.randint(n)] += 1e-4
            continue
        new_seq = seq.copy()
        new_seq[i_high] -= delta
        new_seq[i_low] += delta
        new_seq = np.clip(new_seq, 0, None)
        new_score = evaluate_sequence(new_seq.tolist())
        # accept if better else maybe accept with small probability
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            print(f"Improved to {best_score:.6f}")
        elif new_score < evaluate_sequence(seq.tolist()):
            # accept improve relative to current (if better)
            seq = new_seq
        else:
            # maybe accept with probability decreasing
            if np.random.rand() < 0.001:
                seq = new_seq
        # occasional LP mix

            # compute LP direction
            sum_seq = seq.sum()
            normalized_seq = seq * np.sqrt(2 * n) / sum_seq
            rhs = np.max(np.convolve(normalized_seq, normalized_seq))

"""
Randomized hill‑climbing combined with occasional LP mixing.

The algorithm starts from a (slightly perturbed) uniform step function.
At each iteration it:
  * evaluates the self‑convolution,
  * finds the index k where the convolution attains its maximum,
  * computes the gradient of this maximal entry w.r.t. the coefficients,
  * moves a small amount of mass from the coefficient with the largest
    gradient to the one with the smallest gradient (thus decreasing the
    maximal convolution value),
  * accepts the move if it improves the objective (or with a tiny
    probability to escape plateaus).

Every few hundred iterations a linear‑programming direction is computed:
the LP maximises the total sum of a new sequence `g` under the current
convolution bound.  Several mixing factors `t` are tried and the best
mixture is kept.

All operations keep the coefficients non‑negative.  The objective
`evaluate_sequence` is reproduced locally (it is scale‑invariant) to
avoid the O(n²) cost of the reference implementation.  The routine
stops just before the supplied time budget expires and returns the best
sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: fast convolution using FFT (real inputs)
def _conv_fft(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())  # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Compute the same score as `evaluate_sequence` (scale‑invariant)
def _score(seq: np.ndarray) -> float:
    total = seq.sum()
    if total < 0.01:
        return float("inf")
    conv = _conv_fft(seq)
    max_b = np.max(conv)
    n = seq.shape[0]
    return float(2.0 * n * max_b / (total * total))


# ----------------------------------------------------------------------
# Solve the LP used in the original SOTA approach.
def solve_convolution_lp(f_seq, rhs):
    """
    Maximise sum(b) subject to  b >= 0,
    and (f * b)[k] <= rhs for every k.
    Returns the optimal b (numpy array) or None.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise sum(b)  -> minimise -sum(b)

    # Build the convolution constraints (one row per k = 0 .. 2n-2)
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # Non‑negativity constraints
    a_ub.extend(-np.eye(n))
    b_ub.extend([0.0] * n)

    a_ub = np.vstack(a_ub)
    b_ub = np.array(b_ub)

    # Solve with a short time limit; the default method ("highs") understands it.
    try:
        res = linprog(
            c,
            A_ub=a_ub,
            b_ub=b_ub,
            bounds=(0, None),
            method="highs",
            options={"time_limit": 0.5, "disp": False},
        )
    except Exception:          # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Hill‑climbing search for a step‑function with a small evaluation score.

    Returns a list of non‑negative floats (the heights).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0    # leave a safety margin

    # ------------------------------------------------------------------
    # Initialise the candidate sequence.
    # Prefer the previously known construction if it exists.
    global_vars = globals()
    if "height_sequence_1" in global_vars:
        base = np.array(global_vars["height_sequence_1"], dtype=float)
        n = base.size
        seq = base.copy()
    else:
        n = 1024
        seq = np.ones(n, dtype=float) / n

    # Small random perturbation to break symmetry
    seq += (np.random.rand(n) - 0.5) * 1e-3
    seq = np.clip(seq, 0.0, None)

    # Normalise to unit sum (scale does not affect the objective)
    seq /= seq.sum()

    best_seq = seq.copy()
    best_score = _score(best_seq)

    # Simple temperature schedule for occasional uphill moves
    temperature = best_score * 0.1 + 1e-6
    decay = 0.9995

    it = 0
    while time.time() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) Evaluate current convolution and locate the maximal entry.
        conv = _conv_fft(seq)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # --------------------------------------------------------------
        # 2) Gradient of the maximal entry w.r.t. each coefficient.
        idx = np.arange(n)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        grad = seq.copy()
        grad[valid] += seq[j[valid]]

        # --------------------------------------------------------------
        # 3) Propose a deterministic move: shift a small amount of mass
        #    from a high‑gradient position to a low‑gradient one.
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        if i_high == i_low:
            i_high = np.random.randint(n)
            i_low = np.random.randint(n)

        delta = min(seq[i_high] * 0.3, seq[i_high] + seq[i_low]) * 0.1
        if delta <= 0:
            # fallback random perturbation
            i_high, i_low = np.random.randint(n, size=2)
            delta = min(seq[i_high] * 0.1, 1e-4)

        new_seq = seq.copy()
        new_seq[i_high] -= delta
        new_seq[i_low] += delta
        # keep non‑negativity (very small rounding noise may appear)
        new_seq = np.clip(new_seq, 0.0, None)

        # keep the total sum unchanged (it already is)
        new_score = _score(new_seq)

        # --------------------------------------------------------------
        # 4) Acceptance rule (hill‑climbing + occasional uphill)
        accept = False
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            accept = True
        if new_score < _score(seq):
            accept = True
        else:
            # uphill with small probability governed by temperature
            if np.random.rand() < np.exp(-(new_score - _score(seq)) / temperature):
                accept = True

        if accept:
            seq = new_seq

        # --------------------------------------------------------------
        # 5) Occasionally perform a random jitter to explore new regions.
        if np.random.rand() < 0.003:
            i, j = np.random.randint(n, size=2)
            eps = (np.random.rand() - 0.5) * 1e-3
            # transfer eps from i to j while keeping sum constant
            eps = np.clip(eps, -seq[i], seq[j])
            seq[i] -= eps
            seq[j] += eps
            seq = np.clip(seq, 0.0, None)

        # --------------------------------------------------------------
        # 6) Every few hundred iterations try the LP mixing direction.
        if it % 400 == 0:
            total = seq.sum()
            if total > 0:
                # Normalise to the usual √(2n) factor used by the SOTA method
                norm_factor = np.sqrt(2.0 * n) / total
                f_norm = seq * norm_factor
                rhs = np.max(_conv_fft(f_norm))
                g_raw = solve_convolution_lp(f_norm.tolist(), rhs)
                if g_raw is not None:
                    g_raw = np.array(g_raw, dtype=float)
                    # re‑normalise g to the same √(2n) scale
                    g_norm = g_raw * (np.sqrt(2.0 * n) / g_raw.sum())
                    # try a few blending coefficients
                    for t in (0.02, 0.05, 0.1, 0.2, 0.5):
                        mixed = (1.0 - t) * seq + t * g_norm
                        mixed = np.clip(mixed, 0.0, None)
                        mixed /= mixed.sum()                # keep unit sum
                        mixed_score = _score(mixed)
                        if mixed_score < best_score:
                            best_score = mixed_score
                            best_seq = mixed.copy()
                            seq = mixed
                            print(f"[{it}] LP mix t={t:.2f} improved score to {best_score:.6f}")
                            break

        # --------------------------------------------------------------
        # 7) Cool down the temperature
        temperature = max(temperature * decay, 1e-12)

        # Periodic progress report (does not affect timing)
        if it % 2000 == 0:
            print(f"Iter {it:6d} – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best discovered sequence (as plain Python list)
    return best_seq.tolist()
