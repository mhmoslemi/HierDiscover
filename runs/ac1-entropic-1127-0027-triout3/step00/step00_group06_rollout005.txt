def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, random
    from scipy.optimize import linprog
    from scipy.sparse import csr_matrix, vstack

    np.random.seed(seed)
    random.seed(seed)

    # Determine initial sequence length
    n = None
    try:
        # Use provided global variable if present
        global height_sequence_1
        init_seq = list(height_sequence_1)
        if len(init_seq) > 0:
            n = len(init_seq)
    except Exception:
        init_seq = None

    if init_seq is None or len(init_seq) < 100:
        # generate random length between 500 and 1500
        n = np.random.randint(500, 1500)
        init_seq = np.random.rand(n)
    else:
        n = len(init_seq)

    # Normalize sum to target
    target_sum = np.sqrt(2 * n)
    cur = np.array(init_seq, dtype=float)
    cur_sum = cur.sum()
    if cur_sum == 0:
        cur = np.ones(n) / n
        cur_sum = cur.sum()
    cur = cur * (target_sum / cur_sum)

    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"Initial sequence length {n}, evaluation {best_score:.6f}")

    deadline = time.time() + budget_s - 5

    # Helper to build LP matrix
    def build_lp_matrix(f_vec):
        n_local = len(f_vec)
        # Build data for convolution constraints
        rows = []
        cols = []
        data = []
        for i in range(n_local):
            # coefficient f_i for all b_j where row k = i + j
            # we can compute vector of row indices
            # j from 0 to n-1 => rows = i + np.arange(n_local)
            rows.extend(range(i, i + n_local))
            cols.extend(range(n_local))
            data.extend([f_vec[i]] * n_local)
        A_conv = csr_matrix((data, (rows, cols)), shape=(2 * n_local - 1, n_local))
        # nonneg constraints: -I * b <= 0
        A_nonneg = csr_matrix((-np.ones(n_local), (np.arange(n_local), np.arange(n_local))), shape=(n_local, n_local))
        A = vstack([A_conv, A_nonneg], format='csr')
        return A

    # LP solver wrapper
    def solve_lp(f_vec, M_val):
        n_local = len(f_vec)
        A = build_lp_matrix(f_vec)
        num_conv_constraints = 2 * n_local - 1
        b_ub = np.concatenate([np.full(num_conv_constraints, M_val), np.zeros(n_local)])
        c = -np.ones(n_local)  # maximize sum(b)
        res = linprog(c, A_ub=A, b_ub=b_ub, method='highs', options={'time_limit': 5.0})
        if res.success:
            # ensure nonnegative
            b = np.maximum(res.x, 0.0)
            return b
        else:
            return None

    iteration = 0
    no_improve_counter = 0

    while time.time() < deadline:
        iteration += 1
        conv_f_f = np.convolve(cur, cur)
        M = conv_f_f.max()
        # Solve LP for g
        g = solve_lp(cur, M)
        if g is None:
            # fallback: small random perturbation
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.01)
            # renormalize sum
            cur = cur * (target_sum / cur.sum())
            continue

        # Scale g to target sum (might increase M violation)
        sum_g = g.sum()
        if sum_g == 0:
            continue
        scale_factor = target_sum / sum_g
        g_scaled = g * scale_factor

        # Precompute convolutions
        conv_f_g = np.convolve(cur, g_scaled)
        conv_g_g = np.convolve(g_scaled, g_scaled)

        # Binary search for max feasible t such that max conv <= M * 1.0000001 (tolerance)
        lo = 0.0
        hi = 1.0
        # tolerance for max violation
        tol = 1e-12
        # limit iterations of binary search
        for _ in range(30):
            mid = (lo + hi) / 2.0
            # Compute convex combination convolutions
            combined = (1 - mid) ** 2 * conv_f_f + 2 * mid * (1 - mid) * conv_f_g + mid ** 2 * conv_g_g
            max_comb = combined.max()
            if max_comb <= M * (1 + 1e-12):
                lo = mid
            else:
                hi = mid
        t_opt = lo
        if t_opt < 1e-6:
            # small step, try random perturbation instead
            # random jitter
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.01)
            cur = cur * (target_sum / cur.sum())
            no_improve_counter += 1
            continue

        # Update cur
        cur_new = (1 - t_opt) * cur + t_opt * g_scaled
        # renormalize sum (should already be target_sum but enforce)
        cur_new = cur_new * (target_sum / cur_new.sum())

        cur_score = evaluate_sequence(cur_new.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_new.copy()
            print(f"Iter {iteration}, t={t_opt:.4f}, new score {cur_score:.6f}")
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        cur = cur_new

        # occasional random perturb to escape local optimum
        if no_improve_counter >= 20:
            # apply random perturbation to current and reset counter
            perturb = np.random.rand(n) * 0.001
            cur = np.maximum(cur + perturb, 0.0)
            cur = cur * (target_sum / cur.sum())
            no_improve_counter = 0

    print(f"Best score after {iteration} iterations: {best_score:.6f}")
    return best_seq.tolist()

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, vstack

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‐function that minimises the
    evaluate_sequence() objective.

    The algorithm repeatedly:
        1. normalises the current sequence to sum = sqrt(2 n).
        2. solves a linear programme that maximises the L1‑norm of a
           candidate g under the linear constraints   conv(f,g) ≤ M,
           where M is the current maximal self‑convolution.
        3. rescales g to the same L1‑norm as f and performs a binary
           line‑search to find the largest mixing coefficient t for
           f' = (1‑t) f + t g that keeps the maximal self‑convolution
           of f' below the old bound M.
        4. if the new sequence yields a smaller evaluation score it is
           kept as the current best, otherwise occasional random
           perturbations are applied to escape stagnation.
    The loop runs until ``budget_s`` seconds are exhausted (a few seconds
    are reserved for finalisation).  The best sequence found is returned.
    """
    # ------------------------------------------------------------------
    # Helper functions (must be top level, no closures)
    # ------------------------------------------------------------------
    def _build_lp_matrix(f_vec):
        """Return a CSR matrix encoding the constraints
        sum_i f_i * b_{k-i} ≤ M  (k = 0 … 2n‑2) and  -b_i ≤ 0."""
        n_local = f_vec.shape[0]
        # Convolution constraints – one entry per (i,j) pair.
        rows = []
        cols = []
        data = []
        # For each i (coefficient from f), it contributes to rows i…i+n‑1
        for i in range(n_local):
            rows.extend(range(i, i + n_local))
            cols.extend(range(n_local))
            data.extend([f_vec[i]] * n_local)
        A_conv = csr_matrix((data, (rows, cols)), shape=(2 * n_local - 1, n_local))
        # Non‑negativity constraints: -I * b ≤ 0
        A_nonneg = csr_matrix((-np.ones(n_local), (np.arange(n_local), np.arange(n_local))),
                              shape=(n_local, n_local))
        # Stack vertically
        return vstack([A_conv, A_nonneg], format='csr')

    def _solve_lp(f_vec, M_val, max_time=5.0):
        """Linear programme:
               maximise 1ᵀb
               subject to conv(f,b) ≤ M_val , b ≥ 0.
           Returns the optimal b or None."""
        n_local = f_vec.shape[0]
        A = _build_lp_matrix(f_vec)
        # RHS: first 2n‑1 rows have bound M_val, last n rows have bound 0.
        b_ub = np.concatenate([np.full(2 * n_local - 1, M_val), np.zeros(n_local)])
        c = -np.ones(n_local)         # minimise -sum(b)  → maximise sum(b)
        res = linprog(c, A_ub=A, b_ub=b_ub, method='highs',
                      options={'time_limit': max_time})
        if res.success:
            # numerical noise may produce tiny negatives – clip them.
            return np.maximum(res.x, 0.0)
        return None

    # ------------------------------------------------------------------
    # Initialisation
    # ------------------------------------------------------------------
    np.random.seed(seed)
    # Try to use the provided global sequence; otherwise fall back to random.
    try:
        # height_sequence_1 may be defined in the global namespace.
        init_seq = list(globals()['height_sequence_1'])
    except Exception:
        init_seq = None

    if init_seq is None or len(init_seq) == 0:
        n = np.random.randint(600, 1500)
        cur = np.random.rand(n)
    else:
        # Work with at most 2000 coordinates to keep the LP tractable.
        max_n = 2000
        n = min(len(init_seq), max_n)
        cur = np.array(init_seq[:n], dtype=float)

    # Normalise to the “target” L1‑norm √(2 n) – this is the
    # normalisation used in the reference algorithm and leaves the
    # objective invariant (the evaluation is scale‑free).
    target_sum = np.sqrt(2.0 * n)
    cur_sum = cur.sum()
    if cur_sum == 0.0:
        cur = np.ones(n) / n
        cur_sum = cur.sum()
    cur = cur * (target_sum / cur_sum)

    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, score={best_score:.6f}")

    deadline = time.time() + budget_s - 5.0   # keep a small safety margin
    iteration = 0
    stagnation_counter = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        iteration += 1

        # Current maximal self‑convolution (M) and the full convolution array.
        conv_ff = np.convolve(cur, cur)
        M = conv_ff.max()

        # ----------------------------------------------------------------
        # 1) Linear‑programming step: find a direction g that maximises the
        #    L1‑norm under the linear bound conv(cur, g) ≤ M.
        # ----------------------------------------------------------------
        g = _solve_lp(cur, M, max_time=4.0)
        if g is None:
            # LP failed – apply a tiny random jitter to shake the current state.
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.005)
            cur = cur * (target_sum / cur.sum())
            continue

        # ----------------------------------------------------------------
        # 2) Scale g to the same L1‑norm as cur.
        #    (Scaling may violate the original linear bound, which is why
        #     we will later back‑track with a line‑search on t.)
        # ----------------------------------------------------------------
        sum_g = g.sum()
        if sum_g == 0.0:
            # Degenerate solution – skip this iteration.
            continue
        scale = target_sum / sum_g
        g_scaled = g * scale

        # Pre‑compute the three convolutions needed for the quadratic mixing.
        conv_fg = np.convolve(cur, g_scaled)
        conv_gg = np.convolve(g_scaled, g_scaled)

        # ----------------------------------------------------------------
        # 3) Binary search for the largest admissible mixing coefficient t
        #    such that the self‑convolution of the mixture does not exceed M.
        # ----------------------------------------------------------------
        lo, hi = 0.0, 1.0
        for _ in range(30):               # 30 iterations → ~1e‑9 precision
            mid = (lo + hi) * 0.5
            # convex combination of the three convolution vectors
            mixed = ((1 - mid) ** 2) * conv_ff + 2 * mid * (1 - mid) * conv_fg + (mid ** 2) * conv_gg
            if mixed.max() <= M * (1.0 + 1e-14):
                lo = mid
            else:
                hi = mid
        t_opt = lo

        # If the feasible step is negligible, resort to a random perturbation.
        if t_opt < 1e-5:
            idx = np.random.randint(n)
            cur[idx] = max(0.0, cur[idx] + np.random.randn() * 0.01)
            cur = cur * (target_sum / cur.sum())
            stagnation_counter += 1
            continue

        # ----------------------------------------------------------------
        # 4) Form the new candidate and evaluate it.
        # ----------------------------------------------------------------
        cur_new = (1.0 - t_opt) * cur + t_opt * g_scaled
        # Force the exact target L1‑norm (numerical drift may appear).
        cur_new = cur_new * (target_sum / cur_new.sum())

        new_score = evaluate_sequence(cur_new.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = cur_new.copy()
            print(f"[iter {iteration:04d}] t={t_opt:.4f}, score={new_score:.6f}")
            stagnation_counter = 0
        else:
            stagnation_counter += 1

        cur = cur_new

        # ----------------------------------------------------------------
        # Occasionally break possible stagnation with a tiny random jiggle.
        # ----------------------------------------------------------------
        if stagnation_counter >= 30:
            jitter = np.random.rand(n) * 0.001
            cur = np.maximum(cur + jitter, 0.0)
            cur = cur * (target_sum / cur.sum())
            stagnation_counter = 0

    print(f"[final] best score after {iteration} iterations: {best_score:.6f}")
    return best_seq.tolist()

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, vstack

# ----------------------------------------------------------------------
# Helper functions – must be top‑level (no closures)
# ----------------------------------------------------------------------
def _build_lp_matrix(f_vec: np.ndarray) -> csr_matrix:
    """
    Build the sparse matrix for the linear constraints

        Σ_i f_i * b_{k-i} ≤ M   (k = 0 … 2n‑2)
        -b_i               ≤ 0   (i = 0 … n‑1)

    for a given coefficient vector f_vec of length n.
    The returned matrix has shape ( (2*n-1)+n , n ) and can be used
    directly as ``A_ub`` in ``scipy.optimize.linprog``.
    """
    n = f_vec.shape[0]

    # Convolution part: each pair (i,j) contributes f_i to row i+j, column j.
    rows, cols, data = [], [], []
    for i in range(n):
        # rows i … i+n‑1, one entry per column j = 0 … n‑1
        rows.extend(range(i, i + n))
        cols.extend(range(n))
        data.extend([f_vec[i]] * n)

    A_conv = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # Non‑negativity part: -I * b ≤ 0
    A_nonneg = csr_matrix((-np.ones(n), (np.arange(n), np.arange(n))),
                          shape=(n, n))

    # Stack the two blocks vertically.
    return vstack([A_conv, A_nonneg], format="csr")


def _solve_lp(f_vec: np.ndarray, M_val: float, max_time: float = 5.0) -> np.ndarray | None:
    """
    Linear programme

        maximise   ∑ b_i
        subject to  conv(f, b) ≤ M_val ,  b ≥ 0

    The convolution constraints are linear in b, therefore the problem
    is a genuine LP.  Returns the optimal vector b (or None on failure).
    """
    n = f_vec.shape[0]
    A = _build_lp_matrix(f_vec)

    # RHS: first 2n‑1 rows bound by M_val, last n rows bound by 0.
    b_ub = np.concatenate([np.full(2 * n - 1, M_val), np.zeros(n)])
    c = -np.ones(n)                     # minimise -∑b  →  maximise ∑b

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        method="highs",
        options={"time_limit": max_time},
    )
    if res.success:
        # Clip tiny negatives caused by numerical noise.
        return np.maximum(res.x, 0.0)
    return None
