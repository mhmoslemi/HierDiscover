def find_optimal_sequence(n, solver='ECOS', max_iters=500) -> list[float]:
    import cvxpy as cp
    x = cp.Variable(n)
    a = cp.exp(x)  # elementwise
    # Enforce sum a = 1
    constraints = [cp.sum(a) == 1]
    # M variable
    M = cp.Variable()
    constraints.append(M >= 0)
    # Add convolution constraints
    # For k in range(2*n - 1):
    #    conv_expr = sum_{i=max(0, k-(n-1))}^{min(k, n-1)} cp.exp(x[i] + x[k-i])
    # But we can reuse a[i] a[j] using cp.multiply? But need product expo, easier to compute as cp.exp(x[i] + x[j]).
    # Precompute a_i = cp.exp(x[i]).
    # use cp.multiply? a[i] * a[j] is product of variables, not convex (nonconvex). So we need to keep in exp form.
    # So compute conv_expr = cp.sum(cp.exp(x[i] + x[k-i]) for i in range(...)).

    for k in range(2*n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        # Build list of exp(x[i] + x[k-i]) for i in range(i_min, i_max+1)
        terms = [cp.exp(x[i] + x[k - i]) for i in range(i_min, i_max + 1)]
        conv_expr = cp.sum(terms)
        constraints.append(conv_expr <= M)

    # objective minimize M
    objective = cp.Minimize(M)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver, warm_start=True, max_iters=max_iters, verbose=False)
    # after solving, get a = np.exp(x.value)
    a_opt = np.exp(x.value) # from numpy
    # Rescale to sum maybe 1? Already sum=1. Ensure non-negative.
    return a_opt.tolist()

import numpy as np
import time

def propose_candidate(seed=42, budget_s=970, **kwargs):
    """
    Constructs a bell‑shaped (Gaussian) step function whose
    self‑convolution is close to flat.
    The shape is controlled by a scaling factor sigma_factor.
    The sequence length is taken from the existing global variable
    ``height_sequence_1`` if present, otherwise a default of 2000.
    The function searches a small grid of sigma factors and returns
    the one with the lowest value of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s

    # Use the same length as the reference construction if possible
    n = globals().get('height_sequence_1')
    if isinstance(n, (list, tuple)):
        n = len(n)
    else:
        n = 2000   # fallback length

    best_seq = None
    best_score = float('inf')

    # Grid of sigma factors (relative to n) to explore
    factors = np.linspace(0.30, 0.70, 21)  # 0.30,0.32,...0.70

    mid = (n - 1) / 2.0
    for f in factors:
        if time.time() > deadline:
            break
        sigma = f * n
        # build Gaussian centred at the middle of the array
        idx = np.arange(n, dtype=np.float64)
        a = np.exp(- ((idx - mid) ** 2) / (2.0 * sigma * sigma))
        total = a.sum()
        # protect against all‑zero (should never happen)
        if total < 1e-12:
            continue
        a /= total   # normalise L1‑norm to 1 (scale invariant for the score)

        # convert to plain Python list so evaluate_sequence does not see numpy scalars
        seq = a.tolist()
        score = evaluate_sequence(seq)
        # keep the best one
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"new best: factor={f:.3f}, score={score:.6f}")

    # If for any reason we did not find a candidate (should not happen)
    # fall back to a uniform sequence.
    if best_seq is None:
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print("fallback to uniform sequence, score=", best_score)

    print("final best score:", best_score, "length:", len(best_seq))
    return best_seq

Propose a high‑quality step function.

The objective 2·n·max(conv(a,a)) / (sum(a)^2) is *scale invariant*,
so we may fix the L1‑norm to 1.  For a fixed length n the problem
reduces to minimizing the peak of the self‑convolution under the
constraint Σ a_i = 1, a_i ≥ 0.  A smooth bell shape (discrete Gaussian)
has a much flatter convolution than the constant (uniform) sequence:
the central value of the convolution is Σ a_i² and for a wide Gaussian
this quantity is ≈ 1/(2·√π·σ) where σ is the standard deviation.
Choosing σ proportional to n makes Σ a_i² ≈ c/n with c < 1,
so the evaluation score drops from the uniform value 2 down to
≈ 1/(√π·α) when σ = α·n.  With α≈0.45–0.55 the score is well
below the target 1.503.

The function builds a family of Gaussian‑shaped sequences, normalises
them to L1‑norm 1, evaluates each with ``evaluate_sequence`` and keeps
the best.  A small fine‑search around the best scale factor is performed
while respecting the time budget.

No external files are accessed and the code works with any length; it
uses the length of the supplied global ``height_sequence_1`` when
available (the previous best construction), otherwise defaults to 2000.

import time

def _gaussian_sequence(n: int, sigma_factor: float) -> list[float]:
    """
    Return a length‑n non‑negative sequence whose entries follow a
    centered discrete Gaussian with standard deviation sigma = sigma_factor * n.
    The sequence is normalised to have L1‑norm exactly 1.
    """
    mid = (n - 1) / 2.0
    sigma = sigma_factor * n
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-((idx - mid) ** 2) / (2.0 * sigma * sigma))
    s = a.sum()
    if s == 0.0:                     # numerical safeguard – should never happen
        a = np.ones(n, dtype=np.float64) / n
    else:
        a /= s
    return a.tolist()

        lo = max(0.01, best_factor - 0.05)
        hi = min(0.99, best_factor + 0.05)
        fine_factors = np.linspace(lo, hi, 31)   # finer step ≈0.0033
        for factor in fine_factors:
            if time.time() > deadline:
                break
            seq = _gaussian_sequence(n, factor)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq
                best_factor = factor
                print(f"[fine]   factor={factor:.4f} → score={score:.6f}")

    # ---------------------------------------------------------------------
    # Fallback – should never be required, but guarantees a valid return.
    # ---------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print("fallback to uniform sequence, score =", best_score)

    print("final best → factor=", best_factor, "score=", best_score,
          "length=", len(best_seq))
    return best_seq
