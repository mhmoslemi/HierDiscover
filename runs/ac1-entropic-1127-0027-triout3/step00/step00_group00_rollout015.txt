def propose_candidate(seed=..., budget_s=1000):
    import random, time, numpy as np.

    start_time = time.time()
    best_sequence = None
    best_score = np.inf

    lengths = [2,3,4,5,6,7,8,9,10,12,15,20,30,50,80,120,200]
    for n in lengths:
        # Create initial sequence: random with sum=1.
        a = np.random.rand(n)
        a = a / np.sum(a)
        # Heuristic improvement loop
        for iteration in range(5000):
            if time.time() - start_time > budget_s - 1: break
            # compute convolution and score
            c = np.convolve(a, a)
            maxc = c.max()
            score = 2 * n * maxc / (a.sum()**2) # but sum=1
            if score < best_score:
                best_score = score
                best_sequence = a.copy().tolist()
                print(f"New best score {best_score:.6f} for n={n}")
            # Identify max conv index
            k = int(np.argmax(c))
            # compute contributions
            # pairs i from max(0, k-(n-1)) to min(k,n-1)
            i_low = max(0, k - (n - 1))
            i_high = min(k, n - 1)
            # compute probs for each i in range: p_i = a[i]*a[k-i]
            # choose i with max p_i
            i_candidates = np.arange(i_low, i_high+1)
            p_vals = a[i_candidates] * a[k - i_candidates]
            i_max = i_candidates[np.argmax(p_vals)]
            j = k - i_max
            # move mass
            d = min(a[i_max], a[j]) * 0.1  # step fraction
            if d < 1e-12:
                # fallback random perturb
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.01
                a[idx] = max(0.0, a[idx] + delta)
                # renormalize
                a = a / np.sum(a)
                continue
            # Save copy for revert
            a2 = a.copy()
            a2[i_max] -= d
            a2[j] -= d
            # distribute to ends
            a2[0] += d/2
            a2[-1] += d/2
            # Ensure nonnegatives (should hold)
            # compute new score
            c2 = np.convolve(a2, a2)
            new_maxc = c2.max()
            new_score = 2 * n * new_maxc / (a2.sum()**2)
            if new_score < score:
                a = a2
            else:
                # random small perturb
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.01
                a[idx] = max(0.0, a[idx] + delta)
                # renormalize
                a = a / np.sum(a)

    # After loops, return best_sequence
    # Ensure values not >1000 or <0.
    if best_sequence is None:
        # fallback simple [1,2]
        return [1.0, 2.0]
    else:
        # maybe rescale to keep numbers moderate
        # The evaluation function will clamp to [0,1000].
        # We can output as is.
        return best_sequence

def arcsine_sequence(n):
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for i in range(1,n):
        c[i] = c[i-1] * (2*(i-1)+1) / (2*i)   # because i-1 index in recurrence.
    sum_c = np.sum(c)
    a = c / sum_c

"""
Generate a high‑quality step‑function using the discrete arcsine (central‑binomial)
distribution.

The coefficients a_i are taken proportional to C(2i, i) / 4**i.
These values satisfy
    2*a_0*a_k + Σ_{i=1}^{k-1} a_i a_{k-i} = a_0²   for 0 ≤ k < n,
which equalises the first n convolution entries.  For a fixed length n this
maximises Σ a_i under a bound on max(conv(a,a)), yielding the smallest possible
value of

    2 * n * max(conv(a,a)) / (Σ a_i)² .

As n → ∞ the objective tends to π/2 ≈ 1.571, a substantial improvement over
the trivial uniform construction (value 2.0).  The routine tries a handful of
different lengths, keeps the best result seen before the time budget expires,
and returns the corresponding coefficient list.
"""

import time
import random
import numpy as np

def _arcsine_sequence(n: int) -> list[float]:
    """
    Return the discrete arcsine distribution of length n.
    a_i = C(2i,i)/4**i, normalised so that the sum equals 1.
    """
    # recurrence for c_i = C(2i,i)/4**i
    c = np.empty(n, dtype=float)
    c[0] = 1.0
    for i in range(1, n):
        # c[i] = c[i-1] * (2*(i-1)+1) / (2*i)
        c[i] = c[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    total = c.sum()
    a = c / total
    return a.tolist()


def _improve_sequence(seq: list[float], max_iter: int, deadline: float) -> tuple[list[float], float]:
    """
    Very light local search: repeatedly move a small amount of mass away from the
    pair of indices that currently contributes most to the maximal convolution
    entry.  The sum is re‑normalised after each move.
    """
    a = np.array(seq, dtype=float)
    n = len(a)
    best = a.copy()
    best_score = float('inf')

    for _ in range(max_iter):
        if time.time() > deadline:
            break

        conv = np.convolve(a, a)
        M = conv.max()
        k = int(conv.argmax())                     # index where max occurs

        # indices that participate in this k‑th convolution term
        i_lo = max(0, k - (n - 1))
        i_hi = min(k, n - 1)
        i_vals = np.arange(i_lo, i_hi + 1)
        j_vals = k - i_vals
        products = a[i_vals] * a[j_vals]

        # choose the pair with the largest product (largest contribution)
        idx = i_vals[np.argmax(products)]
        jdx = j_vals[np.argmax(products)]

        # step size – a modest fraction of the smaller of the two entries
        step = min(a[idx], a[jdx]) * 0.12
        if step < 1e-12:                 # nothing to move – random perturbation
            rnd = np.random.randint(n)
            a[rnd] = max(0.0, a[rnd] + (np.random.rand() - 0.5) * 0.01)
            a /= a.sum()
            continue

        a_new = a.copy()
        a_new[idx] -= step
        a_new[jdx] -= step

        # redistribute the removed mass to the two ends – they have the smallest
        # influence on the current maximal convolution entry.
        a_new[0] += step / 2.0
        a_new[-1] += step / 2.0

        # safety – keep non‑negative and re‑normalise
        np.maximum(a_new, 0.0, out=a_new)
        a_new /= a_new.sum()

        # evaluate quickly (the expensive evaluate_sequence call is avoided here;
        # we use the exact formula because scaling does not matter)
        conv_new = np.convolve(a_new, a_new)
        score = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        if score < best_score:
            best_score = score
            best = a_new.copy()
            a = a_new                      # keep the improvement
        # otherwise keep the old a (no uphill moves)

    return best.tolist(), best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function sequence.  Several lengths are tried, each initialised
    with the discrete arcsine construction; a tiny local search is applied to
    each candidate.  The best sequence found before the time budget expires is
    returned.
    """
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 2.0   # a small safety margin
    best_seq: list[float] | None = None
    best_score = float('inf')

    # lengths to try – small ones allow deeper local search, large ones give a
    # value close to the asymptotic limit π/2.
    lengths = [2, 3, 4, 5, 10, 20, 50, 100, 200, 400, 800, 1600, 3000]

    for n in lengths:
        if time.time() > deadline:
            break

        # start from the optimal arcsine distribution for this length
        start_seq = _arcsine_sequence(n)

        # a light optimisation phase; the number of iterations scales down for
        # bigger n to stay within the time budget.
        max_iter = max(200, int(2000 / (n ** 0.5)))
        seq, score = _improve_sequence(start_seq, max_iter, deadline)

        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"New best: score={best_score:.6f}, length={n}")

    # fallback – should never happen, but guarantee a valid output
    if best_seq is None:
        best_seq = [1.0, 2.0]   # simple 2‑element construction (score ≈ 1.777)

    # The evaluation routine clamps values to [0,1000]; our numbers are tiny,
    # well within the allowed range.
    return best_seq
