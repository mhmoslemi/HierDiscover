"""
Generate a low‑value step function for the provided evaluation metric.

The metric is
    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For a geometric sequence a_i = (1-δ)^i of length n one can show that
the maximum of the convolution is attained at the centre (index n‑1) and
that the score depends only on the product x = δ·n :

    score = 2 * x**2 * exp(-x) / (1 - exp(-x))**2      (x = δ·n).

This expression is strictly decreasing for 0 < x < ln n and then grows
again, so the optimal choice is x = ln n, i.e. δ = ln(n)/n.  With this
choice the score simplifies to approximately 2·(ln n)² / n, which rapidly
drops when n increases.

The function searches a few reasonable lengths, builds the corresponding
geometric sequences (using δ = ln(n)/n) and returns the one with the
lowest score that fits into the given time budget.
All numbers are non‑negative and well below the 1000‑limit, so the
security checks in ``evaluate_sequence`` are satisfied.
"""

import math
import time


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) gives a very low
    evaluation score.  The algorithm simply tries geometric sequences of the
    form a_i = (1‑δ)^i with δ = ln(n)/n for several values of n and keeps the
    best one found before the time budget expires.
    """
    start = time.time()
    best_seq = None
    best_score = float("inf")

    # A few lengths spanning a few orders of magnitude.
    # Longer sequences give smaller scores but also cost more time.
    candidate_lengths = [200, 500, 1000, 2000, 5000, 8000]

    for n in candidate_lengths:
        # Stop early if we are out of time.
        if time.time() - start > budget_s - 1.0:
            break

        # Optimal decay parameter for a geometric sequence of length n:
        delta = math.log(n) / n
        if delta >= 1.0:      # safety – would make r ≤ 0
            continue
        r = 1.0 - delta

        # Build the sequence a_i = r**i  (i = 0 … n‑1)
        seq = [r ** i for i in range(n)]

        # Evaluate – this is the expensive part (full convolution).
        try:
            score = evaluate_sequence(seq)
        except Exception:
            # Any unexpected problem (should not happen) → skip this candidate
            continue

        if score < best_score:
            best_score = score
            best_seq = seq
            print(

            )

    # Fallback – should never be needed, but guarantees a valid return.
    if best_seq is None:
        best_seq = [1.0] * 200
        best_score = evaluate_sequence(best_seq)
        print(f"Fallback uniform sequence, score={best_score}")

    print(f"Final best score: {best_score:.6g} (length {len(best_seq)})")
    return best_seq
