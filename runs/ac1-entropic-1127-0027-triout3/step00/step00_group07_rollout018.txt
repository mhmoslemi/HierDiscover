def simplex_projection(v):
    # returns projection onto simplex {x >=0, sum x =1}
    v = np.asarray(v)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def compute_conv_fft(a):
    """Compute self-convolution of a using FFT (real-valued)."""
    n = a.shape[0]
    # length for FFT at least 2*n-1, choose next power of two for speed
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    # Clip small negative numeric noise
    conv[conv < 0] = 0.0
    return conv

def objective_from_conv(conv, n):
    """Return the evaluation value 2*n*max(conv) (sum of a assumed 1)."""
    return 2.0 * n * np.max(conv)

def simplex_projection(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    # Reference: Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.where(u + (1 - cssv) / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = 0.0
    else:
        rho = rho_arr[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed=42, budget_s=1000):
    """
    Simple hill‑climbing guided by the currently maximal autocorrelation.
    Starts from a random Dirichlet vector and repeatedly transfers a small
    amount of probability mass away from the pair of indices that realise
    the maximal convolution value.  The move is directed to an index that
    contributes minimally to that maximal entry, therefore the peak can only
    decrease (or stay equal).  The algorithm stops when the time budget is
    exhausted and returns the best sequence found.
    """

    start = time.time()
    deadline = start + budget_s - 2.0   # leave a little margin

    best_seq = None
    best_score = float('inf')
    iteration = 0

    # try a few different lengths – shorter sequences have a weaker
    # requirement on the maximal convolution.
    candidate_lengths = [200, 400, 800, 1124]
    # Randomly shuffle order to avoid bias.
    np.random.shuffle(candidate_lengths)

    for n in candidate_lengths:
        # initialise with a Dirichlet(1) (uniform random) distribution.
        a = np.random.rand(n)
        a = a / a.sum()          # enforce sum = 1 (the evaluation is scale‑invariant)
        # keep track of the best for this length
        conv = compute_conv_fft(a)
        cur_score = objective_from_conv(conv, n)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
        # heuristic parameters
        step_factor = 0.2
        # main loop for this length
        while time.time() < deadline:
            # compute convolution and locate the peak
            conv = compute_conv_fft(a)
            max_idx = int(np.argmax(conv))
            max_val = conv[max_idx]

            # record improved solution
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f'Improved score {best_score:.6f} with n={n} iteration={iteration}')
            # --- choose a pair that contributes most to the peak ---
            lo = max(0, max_idx - (n - 1))
            hi = min(max_idx, n - 1)
            i_range = np.arange(lo, hi + 1)
            j_range = max_idx - i_range
            contrib = a[i_range] * a[j_range]          # contribution of each i to the current peak
            i_max_idx = i_range[np.argmax(contrib)]
            j_max_idx = max_idx - i_max_idx

            # --- choose a safe destination index t ---

            peak_contrib[j_max_idx] = np.inf
            t = int(np.argmin(peak_contrib))

                new_a = a.copy()
                new_a[i_max_idx] -= delta

        print('Returning fallback sequence')
        return fallback
    else:
        print(f'Final best score {best_score:.6f} (n={len(best_seq)})')
        return best_seq.tolist()

import numpy as np
import time

def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """
    Compute the (aperiodic) self‑convolution a*a using an FFT.
    The result has length 2·len(a)‑1.
    """
    n = a.shape[0]
    # length of the FFT, power of two for speed
    L = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(A * A, n=L)[: 2 * n - 1]
    # eliminate tiny negative artefacts caused by rounding
    conv[conv < 0] = 0.0
    return conv


def _objective(conv: np.ndarray, n: int) -> float:
    """
    Evaluation value for a sequence with sum == 1.
    Since the original function is 2*n*max(conv)/(sum**2) and sum is 1,
    this reduces to 2*n*max(conv).
    """
    return 2.0 * n * np.max(conv)

      * start from a random Dirichlet vector,
      * locate the pair (i, j) that contributes most to the peak of the
        self‑convolution,
      * transfer a small amount of mass from that pair to an index that has
        the smallest contribution to the same peak,
      * accept the move if the peak does not increase (with a small chance
        to accept a worsening move – simulated‑annealing flavour),
      * repeat until the time budget expires.
    The best sequence found (lowest evaluation value) is returned.
    If no improvement over the provided `height_sequence_1` is obtained,
    the fallback sequence is returned.
    """
