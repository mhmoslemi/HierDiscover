"""
Propose a step–function (non‑negative list) that attains a low value of the
evaluation function

    2 * n * max(conv(a,a)) / (sum(a)^2)

The key observation is that the score only depends on the *shape* of the
sequence, not on an overall scaling factor.  If the entries are 0/1 the
convolution counts, for each possible sum k, the number of ordered pairs
(i,j) with i+j = k and a_i = a_j = 1.  Hence the score is

    score = max_count / average_count   where   average = m²/(2n)

with m = number of ones.  To obtain a score ≤ 1.503 we only have to find a
binary set S ⊂ {0,…,n‑1} of size m such that every sum occurs at most

    C = floor(1.503 * m²/(2n))

times (ordered pairs).  This is a pure combinatorial packing problem.
A simple greedy randomized construction works well:

* build an array `cnt[0..2n‑2]` storing the current number of ordered pairs
  that realise each sum,
* scan the positions of the universe in a random order,
* add a position i to S if for all existing j∈S the increments
  `cnt[i+j] + 2` stay ≤ C and also `cnt[2*i] + 1 ≤ C`.

If the construction succeeds we obtain a binary sequence whose score is
guaranteed not to exceed the target.  The outer loop tries many random
choices of n and m (n≈1000…4000, m≈1.5·√(2n) … 3·√(2n)) until the time
budget expires.  The best sequence found is returned.

The algorithm runs only integer arithmetic inside the construction loop,
so many candidates can be examined within the allowed time.
"""

import random
import time
import numpy as np

def attempt_construct_set(n: int, m: int, max_allowed: int) -> list[int] | None:
    """
    Try to build a set S of size m inside {0,…,n-1} such that the ordered
    pair counts for each sum never exceed `max_allowed`.  Returns the list of
    selected indices or None on failure.
    """
    # try a few random orderings
    for _ in range(5):
        cnt = np.zeros(2 * n - 1, dtype=np.int32)   # ordered‑pair counters
        S: list[int] = []
        # random permutation of candidate indices
        cand = list(range(n))
        random.shuffle(cand)
        for i in cand:
            # self‑pair (i,i) would increase cnt[2*i] by 1
            if cnt[2 * i] + 1 > max_allowed:
                continue
            ok = True
            # check the contribution of the new element with all already chosen ones
            for j in S:
                s = i + j
                if cnt[s] + 2 > max_allowed:   # (i,j) and (j,i)
                    ok = False
                    break
            if not ok:
                continue
            # insertion is feasible – update structures
            S.append(i)
            cnt[2 * i] += 1
            for j in S[:-1]:
                s = i + j
                cnt[s] += 2
            if len(S) == m:
                return S
    return None

    The function respects the given time budget and returns the best
    sequence discovered.
    """
    random.seed(seed)
    np.random.seed(seed)

    # a trivial fallback (uniform sequence gives score 2)
    best_seq = [1.0] * 10
    best_score = float(evaluate_sequence(best_seq))

    target_ratio = 1.503                # required upper bound
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin
    it = 0

    while time.time() < deadline:
        # choose problem size
        n = random.randint(1000, 4000)

        # choose a size m for the support; we aim for a density that
        # makes the target ratio reachable.
        factor = random.uniform(1.5, 3.0)           # empirical tuning
        m = int(factor * (2 * n) ** 0.5)
        m = max(2, min(m, n - 1))

        # average number of ordered pairs per sum
        avg = m * m / (2.0 * n)
        max_allowed = int(np.floor(target_ratio * avg))
        max_allowed = max(1, max_allowed)

        # attempt the combinatorial construction
        S = attempt_construct_set(n, m, max_allowed)
        if S is None:               # construction failed – try another (n,m)
            continue

        # build the explicit 0/1 sequence
        seq = [0.0] * n
        for idx in S:

        # compute the exact score (avoiding the heavy security checks of evaluate_sequence)

        score = 2.0 * n * max_b / (sum_a * sum_a)

        # keep the best one

            best_seq = seq

        # early exit when we have already reached the required bound
        if best_score <= target_ratio:
            break

        it += 1

    # final safety: make sure the returned sequence is accepted by evaluate_sequence
    # (it will be, but we call it once more to be sure)
    _ = evaluate_sequence(best_seq)   # may raise if something went wrong
    return best_seq
