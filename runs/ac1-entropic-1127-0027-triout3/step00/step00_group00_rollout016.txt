def propose_candidate(seed=0, budget_s=1000):

    np.random.seed(seed)

    # length: use length of height_sequence_1 if present else 3500

            seq = seq + ((n - 1 - i) + a1) ** (-p1)

                seq = seq + (1 - w) * ((n - 1 - i) + a2) ** (-p2)

    # Coarse grid search

                seq = gen_seq(p, a, sym=sym)

"""
propose_candidate
-----------------

Search for a step‑function that minimizes the custom `evaluate_sequence`
objective.  The algorithm uses a simple parametric family inspired by the
continuous solution of the problem:

    a_i = (i + α) ** ( -p )               (i = 0 … n‑1)

For p≈0.5 the convolution becomes almost flat and the evaluation is about
π/2 ≈ 1.57.  By increasing p above 0.5 and choosing a modest offset α we
reduce the large contributions from the first element while keeping the
interior convolution low; this can bring the score below the target
1.503.  The implementation

* tries a coarse grid over (p, α) and an optional symmetric copy,
* then performs a random refinement phase around the best parameters,
  always keeping the best sequence seen so far,
* respects the time budget (default 1000 s) and the non‑negativity / size
  constraints enforced inside `evaluate_sequence`.

The returned sequence is a list of floats; scaling does not affect the
score, so no explicit normalisation is performed.
"""

import time
import math
import random
import numpy as np

def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    # ------------------------------------------------------------------
    # Helper utilities
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # Length of the sequence – use the known construction if available
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 3500

    deadline = time.time() + budget_s - 2  # leave a couple of seconds margin

    def evaluate(seq):
        """Wrapper that returns a huge value if `evaluate_sequence` fails."""
        try:
            return evaluate_sequence(seq)   # the function is provided by the framework
        except Exception:
            return float('inf')

    # ------------------------------------------------------------------
    # Sequence generator from parameters
    # ------------------------------------------------------------------
    def generate(p, alpha, sym=False, p2=None, alpha2=None, w=1.0):
        """
        Returns a list of length n with entries
            a_i = w * (i+α)^{-p} + (1-w) * (i+α2)^{-p2}
        optionally added to their mirror image if `sym` is True.
        """
        i = np.arange(n, dtype=np.float64)

        # primary term
        seq = (i + alpha) ** (-p)

        # optional second term (adds more flexibility)
        if p2 is not None:
            seq = w * seq + (1.0 - w) * ((i + alpha2) ** (-p2))

        # optional symmetric copy (adds the reversed term)
        if sym:
            rev = ((n - 1 - i) + alpha) ** (-p)
            seq = seq + rev
            if p2 is not None:
                rev2 = ((n - 1 - i) + alpha2) ** (-p2)
                seq = seq + (1.0 - w) * rev2

        # ensure non‑negativity and stay well below the hard 1000 cap
        seq = np.maximum(seq, 0.0)
        seq = np.minimum(seq, 999.0)   # a safety margin

        return seq.tolist()

    # ------------------------------------------------------------------
    # Coarse grid exploration
    # ------------------------------------------------------------------
    best_score = float('inf')
    best_seq = None
    best_params = None

    p_vals = np.linspace(0.40, 0.80, 9)            # 0.40 … 0.80
    alpha_vals = [0.3, 0.7, 1.0, 2.0, 5.0, 10.0]   # diverse offsets
    symmetry_options = (False, True)

    print(f"Starting coarse grid (n={n})")
    for p in p_vals:
        for alpha in alpha_vals:
            for sym in symmetry_options:
                seq = generate(p, alpha, sym=sym)
                score = evaluate(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    best_params = (p, alpha, sym, None, None, 1.0)
                    print(f"[grid] New best {score:.6f}  p={p:.3f} α={alpha} sym={sym}")
                if time.time() > deadline:
                    print("Time limit reached during grid search.")
                    return best_seq

    # ------------------------------------------------------------------
    # Random refinement – mix of one‑ and two‑term families
    # ------------------------------------------------------------------
    print("Entering random refinement phase")
    while time.time() < deadline:
        # Pick whether to use a single power‑law term or a two‑term mixture
        use_two = random.random() < 0.45

        # Parameters for the first term
        p1 = random.uniform(0.45, 0.78)
        a1 = random.uniform(0.1, 20.0)

        if use_two:
            p2 = random.uniform(0.45, 0.78)
            a2 = random.uniform(0.1, 20.0)
            w = random.random()
        else:
            p2 = None
            a2 = None
            w = 1.0

        sym = random.random() < 0.5

        seq = generate(p1, a1, sym=sym, p2=p2, alpha2=a2, w=w)
        score = evaluate(seq)

        if score < best_score:
            best_score = score
            best_seq = seq
            best_params = (p1, a1, sym, p2, a2, w)
            print(f"[rand] New best {score:.6f}  params={best_params}")

        # Local perturbation around the current best (only if already decent)
        if best_score < 1.6 and random.random() < 0.25:
            # Small Gaussian tweaks of the parameters
            for _ in range(3):
                if best_params is None:
                    break
                p1b, a1b, symb, p2b, a2b, wb = best_params
                p1b = max(0.01, min(0.99, p1b + random.gauss(0, 0.02)))

        # Safety break if we have already out‑performed the target
        if best_score < 1.503:
            print("Target surpassed – terminating early.")
            break

    print(f"Search finished. Best score found: {best_score:.6f}")
    return best_seq
