def propose_candidate(seed=42, budget_s=1000):
    import random, time, numpy as np
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2
    # Determine n.
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 2000
    # uniform baseline.
    best_seq = [1.0 / n] * n
    best_score = evaluate_sequence(best_seq)
    print("Initial uniform score:", best_score)
    # maybe also evaluate height_sequence_1 if exists.
    try:
        seq1 = list(height_sequence_1)
        s1 = evaluate_sequence(seq1)
        if s1 < best_score:
            best_seq, best_score = seq1, s1
            print("Better from existing sequence:", s1)
    except NameError:
        pass

    # Precompute distance array for linear/quadratic.
    center = (n-1)/2.0
    d_arr = np.array([abs(i - center) / center for i in range(n)])

    iter = 0
    while time.time() < deadline:
        iter += 1
        # randomly pick shape
        shape_type = random.choice(['linear','quadratic','exponential','piecewise'])
        if shape_type == 'linear':
            gamma = random.uniform(0.0, 10.0)  # control ratio <= 11.
            a = (1.0 + gamma * d_arr)
            total = a.sum()
            seq = (a / total).tolist()
        elif shape_type == 'quadratic':
            gamma = random.uniform(0.0, 10.0)
            delta = random.uniform(-5.0, 5.0)  # can be negative.
            a = 1.0 + gamma * d_arr + delta * (d_arr**2)
            # enforce nonnegativity
            min_val = a.min()
            if min_val < 0:
                a = a - min_val  # shift up to nonnegative (makes sum bigger)
            total = a.sum()
            if total == 0:
                continue
            seq = (a / total).tolist()
        elif shape_type == 'exponential':
            gamma = random.uniform(0.0, 5.0)
            a = np.exp(gamma * d_arr)  # larger at edges
            total = a.sum()
            seq = (a / total).tolist()
        else:  # piecewise constant
            # randomly choose proportion m (edge width)
            m = random.randint(1, n//2 - 1)
            r = random.uniform(1.0, 20.0)
            # interior weight w2 = 1/(2*m*r + n-2*m)
            w2 = 1.0 / (2.0*m*r + (n - 2*m))
            w1 = r * w2
            seq = np.concatenate([np.full(m, w1), np.full(n-2*m, w2), np.full(m, w1)]).tolist()
        # Evaluate score
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"Iter {iter}: New best {score:.6f} shape {shape_type}")
        # Additional local search: on best_seq, randomly adjust a bit
        if random.random() < 0.1:
            # Perform small random perturbation on best_seq
            idx_i = random.randint(0, n-1)
            idx_j = random.randint(0, n-1)
            if idx_i == idx_j:
                continue
            eps = random.uniform(0, 0.01)  # up to 1% of current share
            # shift mass from idx_i to idx_j
            seq2 = best_seq.copy()
            transfer = min(seq2[idx_i], eps)
            seq2[idx_i] -= transfer
            seq2[idx_j] += transfer
            # renormalize just in case small floating error:
            total = sum(seq2)
            if total == 0: # shouldn't happen
                continue
            seq2 = [x/total for x in seq2]
            score2 = evaluate_sequence(seq2)
            if score2 < best_score:
                best_score = score2
                best_seq = seq2
                print(f"Iter {iter} local perturb: improved to {score2:.6f}")
    print("Finished. Best score:", best_score, "time elapsed:", time.time() - start_time)
    return best_seq

"""
A lightweight optimizer for the “max‑convolution” evaluation problem.

The goal is to minimise

    score = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the ordinary (full) discrete convolution.
The function is homogeneous of degree 0, therefore we can keep
`sum(a) == 1` all the time – scaling does not affect the score.

Strategy
--------
1. **Start** from the uniform distribution (known to give a score of 2).
2. **Random parametric families** (linear, quadratic, exponential,
   two‑level) are generated and evaluated – this quickly discovers
   promising “shapes’’ that already beat the uniform bound.
3. **Iterative smoothing** – repeatedly locate the shift `k` that
   attains the current maximal convolution value.  The gradient of the
   term `conv(a, a)[k]` with respect to `a` is simply the reversed slice
   `a_{k-i}`.  Moving probability mass *away* from indices with a
   large gradient (and re‑projecting onto the probability simplex)
   reduces the current maximum.  A simple step‑size schedule
   `η = η₀ / sqrt(iter)` is used.
4. **Occasional random perturbations** keep the search from getting
   trapped in a local minimum.

The whole procedure runs for the allotted wall‑time (default 1000 s)
and returns the best sequence discovered.
"""

import time
import random
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for generating candidate shapes
# ----------------------------------------------------------------------
def linear_shape(n, gamma):
    """a_i ∝ 1 + γ·|i−mid|/mid   (γ ≥ 0)"""
    mid = (n - 1) / 2.0
    d = np.abs(np.arange(n) - mid) / mid
    a = 1.0 + gamma * d
    a /= a.sum()
    return a


def quadratic_shape(n, gamma, delta):
    """a_i ∝ 1 + γ·d + δ·d²   (d as in linear_shape)"""
    mid = (n - 1) / 2.0
    d = np.abs(np.arange(n) - mid) / mid
    a = 1.0 + gamma * d + delta * d ** 2
    a[a < 0] = 0.0                     # keep non‑negative
    if a.sum() == 0:
        return np.full(n, 1.0 / n)
    a /= a.sum()
    return a


def exponential_shape(n, gamma):
    """a_i ∝ exp(γ·d)   (γ ≥ 0)"""
    mid = (n - 1) / 2.0
    d = np.abs(np.arange(n) - mid) / mid
    a = np.exp(gamma * d)
    a /= a.sum()
    return a


def piecewise_two_level(n, m, ratio):
    """
    Two‑level vector:
        first m entries = high, last m entries = high,
        middle block = low.
        ratio = high / low   (≥ 1)
    """
    if m == 0 or m * 2 >= n:
        return np.full(n, 1.0 / n)

    low_weight = 1.0 / (2 * m * ratio + (n - 2 * m))
    high_weight = ratio * low_weight
    a = np.concatenate(
        [np.full(m, high_weight),
         np.full(n - 2 * m, low_weight),
         np.full(m, high_weight)]
    )
    return a


# ----------------------------------------------------------------------
# Core optimisation loop
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (non‑negative list) that yields a low
    value of `evaluate_sequence`.  The routine respects the time budget
    and returns the best sequence found.
    """
    random.seed(seed)
    np.random.seed(seed)

    # --------------------------------------------------------------
    # Problem size – use the length of the provided reference if any
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # pylint: disable=undefined-variable
    except Exception:                      # reference not present
        n = 2000

    # --------------------------------------------------------------
    # Initial uniform candidate (score = 2.0)
    # --------------------------------------------------------------
    cur = np.full(n, 1.0 / n)
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] uniform score = {best_score:.6f}")

    # If a historic construction exists, try it as a warm start
    try:
        seq1 = np.array(height_sequence_1, dtype=float)
        s1 = evaluate_sequence(seq1.tolist())
        if s1 < best_score:
            best_score, best_seq = s1, seq1 / seq1.sum()
            cur = best_seq.copy()
            print(f"[init] using existing sequence, score = {s1:.6f}")
    except Exception:
        pass

    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # small safety margin

    it = 0
    # step‑size for gradient‑style moves
    eta0 = 0.8

    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # 30 % of the time: try a fresh parametric shape (exploration)
        # ----------------------------------------------------------
        if random.random() < 0.3:
            shape_type = random.choice(["linear", "quadratic", "exp", "piecewise"])
            if shape_type == "linear":
                gamma = random.uniform(0.0, 12.0)
                cand = linear_shape(n, gamma)
            elif shape_type == "quadratic":
                gamma = random.uniform(-4.0, 8.0)
                delta = random.uniform(-4.0, 4.0)
                cand = quadratic_shape(n, gamma, delta)
            elif shape_type == "exp":
                gamma = random.uniform(0.0, 6.0)
                cand = exponential_shape(n, gamma)
            else:                               # piecewise two‑level
                m = random.randint(1, n // 2 - 1)
                ratio = random.uniform(1.0, 20.0)
                cand = piecewise_two_level(n, m, ratio)

            # evaluate the new candidate
            try:
                cand_score = evaluate_sequence(cand.tolist())
            except Exception:
                continue

            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                cur = cand.copy()
                print(f"[{it}] new best via shape ({shape_type}) "
                      f"score={cand_score:.6f}")

            continue   # next iteration

        # ----------------------------------------------------------
        # 70 % of the time: gradient‑style reduction of the current max
        # ----------------------------------------------------------
        conv = np.convolve(cur, cur)
        k_max = int(np.argmax(conv))

        # build the gradient w.r.t. the term conv[k_max]
        # derivative of Σ_i a_i a_{k_max-i} w.r.t. a_i is a_{k_max-i}
        w = np.zeros_like(cur)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        w[idx] = cur[k_max - idx]

        # simple decreasing step size
        eta = eta0 / math.sqrt(it)

        # move probability mass away from indices with large gradient
        nxt = cur - eta * w
        nxt = np.maximum(nxt, 0.0)

        # re‑normalise to stay on the simplex (sum = 1)
        s = nxt.sum()
        if s < 1e-12:                         # degenerate – re‑initialise
            nxt = np.full(n, 1.0 / n)
        else:
            nxt /= s

        # occasional tiny random jitter to avoid stagnation
        if random.random() < 0.02:
            j = random.randint(0, n - 1)
            delta = (random.random() - 0.5) * 0.02   # ±1 %
            nxt[j] = max(0.0, nxt[j] + delta)
            nxt /= nxt.sum()

        # evaluate
        try:
            score = evaluate_sequence(nxt.tolist())
        except Exception:
            # on any numerical mishap just keep the old vector
            continue

        if score < best_score:
            best_score = score
            best_seq = nxt.copy()
            print(f"[{it}] improvement: score={score:.6f}")

        # always accept the mutation for the next iteration (it is a
        # descent step in the objective we are trying to reduce)
        cur = nxt

    print(f"[done] iter={it} best_score={best_score:.6f}")
    return best_seq.tolist()
