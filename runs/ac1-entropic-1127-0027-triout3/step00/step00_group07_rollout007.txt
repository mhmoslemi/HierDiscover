smooth‐max gradient descent and the LP‑based direction from the

import numpy as np
import time
from scipy import optimize

linprog = optimize.linprog


def _conv(seq: np.ndarray) -> np.ndarray:
    """Linear convolution of a non‑negative sequence with itself."""
    return np.convolve(seq, seq)


def _smooth_max(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights for the smooth‑max approximation."""
    w = np.exp(beta * conv)
    # avoid overflow – the values are tiny (conv≤1) but keep it safe
    w = np.clip(w, a_min=0, a_max=np.finfo(w.dtype).max)
    return w / w.sum()


def _gradient_step(seq: np.ndarray, beta: float, eta: float) -> np.ndarray:
    """
    One multiplicative‑gradient step for the smooth‑max objective.
    The gradient of the smoothed max of the convolution is
        g_i = 2 * Σ_k p_k * a_{k‑i},
    where p_k are the soft‑max weights.
    The update is a_i ← a_i * exp(‑η g_i) followed by simplex
    projection (renormalisation).
    """
    n = len(seq)
    conv = _conv(seq)
    p = _smooth_max(conv, beta)                # probabilities over k

    # correlation p ⋆ a   (g_i = 2 * Σ_k p_k a_{k-i})
    a_rev = seq[::-1]
    corr = np.convolve(p, a_rev)               # length  (2n‑1)+n‑1 = 3n‑2
    # extract the n components corresponding to i = 0 … n‑1
    offset = n - 1
    g = 2.0 * corr[offset:offset + n]

    # multiplicative update + projection onto the simplex
    seq = seq * np.exp(-eta * g)
    seq = np.maximum(seq, 0.0)
    total = seq.sum()
    if total <= 0.0:                           # numerical safety
        seq = np.random.rand(n)
        total = seq.sum()
    seq /= total
    return seq


def _lp_improve(seq: np.ndarray) -> np.ndarray:
    """
    Apply one LP‑based improvement (the routine from the reference
    solution).  It returns a new sequence; if the LP fails the
    original sequence is returned unchanged.
    """
    # The helper from the reference code is reproduced locally.
    n = len(seq)
    sum_seq = np.sum(seq)
    if sum_seq < 1e-12:
        return seq
    # normalise to the “sqrt(2 n)’’ scaling used there
    norm_seq = seq * np.sqrt(2.0 * n) / sum_seq
    rhs = np.max(np.convolve(norm_seq, norm_seq))

    # ----- solve the LP: maximise Σ b_j  s.t. conv(f,b) ≤ rhs , b ≥ 0 -----
    f = norm_seq
    c = -np.ones(n)                     # maximise Σ b_j
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f[i]
        a_ub.append(row)
        b_ub.append(rhs)
    a_ub = np.vstack([a_ub, -np.eye(n)])   # non‑negativity: –b ≤ 0
    b_ub = np.hstack([b_ub, np.zeros(n)])

    try:
        res = linprog(c, A_ub=a_ub, b_ub=b_ub,
                      method='highs',
                      options={'disp': False, 'time_limit': 5.0})
    except Exception:                     # any linprog trouble → no move
        return seq

    if not res.success:
        return seq

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return seq
    # renormalise g to the original scale (sum = 1)
    g = g * (np.sqrt(2.0 * n) / sum_g)
    # simple convex combination – a small step size works well
    t = 0.03
    new_seq = (1.0 - t) * seq + t * g
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


def _initial_sequence(n: int, kind: str, rng: np.random.Generator) -> np.ndarray:
    """Create a non‑negative start vector of length n."""
    if kind == "geometric":
        # r close to 1 gives a slowly decaying profile
        delta = rng.uniform(0.5, 5.0) / n        # δ ≈ Δ / n
        r = 1.0 - delta
        seq = np.array([r ** i for i in range(n)], dtype=float)
    elif kind == "power":
        p = rng.uniform(0.4, 2.2)
        seq = np.array([1.0 / ((i + 1) ** p) for i in range(n)], dtype=float)
    elif kind == "random":
        seq = rng.random(n)
    else:                     # uniform
        seq = np.ones(n, dtype=float)
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list:
    """
    Search for a step‑function (list of non‑negative heights) whose
    evaluation score  2·n·max(conv)/sum²  is as small as possible.
    The routine stays within the given time budget, mixing
    stochastic starts, smooth‑max gradient descent and occasional LP
    refinements.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a safety margin

    # start from the existing construction if it exists
    best_seq = None
    best_score = np.inf
    if "height_sequence_1" in globals():
        try:
            cand = list(height_sequence_1)
            sc = evaluate_sequence(cand)
            if sc < best_score:
                best_score = sc
                best_seq = cand
                print(f"Loaded height_sequence_1: score {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Main stochastic search
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose a random length and start type
        n = rng.integers(300, 1200)                # moderate size
        kind = rng.choice(["geometric", "power", "random", "uniform"])
        seq = _initial_sequence(n, kind, rng)

        # a few gradient‑descent epochs
        it = 0
        local_best_seq = seq.copy()
        local_best_score = evaluate_sequence(seq.tolist())

        while time.time() < deadline and it < 2500:
            # schedule for beta (smoothness) and eta (step size)
            beta = 6.0 + 0.001 * it                # gradually sharper
            eta = 0.25                               # fairly aggressive

            seq = _gradient_step(seq, beta, eta)

            # occasional random jitter to escape plateaus
            if rng.random() < 0.012:
                idx = rng.integers(n)
                seq[idx] = max(0.0, seq[idx] + rng.standard_normal() * 0.02)
                seq /= seq.sum()

            # LP‑based refinement every ~200 iterations
            if it % 200 == 0:
                seq = _lp_improve(seq)

            # evaluate
            cur_score = evaluate_sequence(seq.tolist())
            if cur_score < local_best_score:
                local_best_score = cur_score
                local_best_seq = seq.copy()
                # update global best immediately
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = seq.tolist()
                    elapsed = time.time() - start_time
                    print(f"New global best: {best_score:.6f} (n={n}) @ {elapsed:.1f}s")
            it += 1

        # after finishing this trial, compare its local optimum
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = local_best_seq.tolist()
            print(f"Trial finished – new best: {best_score:.6f} (len={len(best_seq)})")

    # final safety check – if nothing was found fall back to uniform
    if best_seq is None:
        best_seq = [1.0] * 100
        best_score = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence")

    print(f"Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    return best_seq
