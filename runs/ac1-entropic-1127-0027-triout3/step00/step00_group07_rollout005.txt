def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_score = float('inf')
    # try multiple lengths
    for n in [50, 100, 200, 400, 800]:
        # start some restarts
        for restart in range(10):
            # random Dirichlet
            seq = np.random.dirichlet(alpha=np.ones(n)*0.5).tolist()  # moderate concentration
            curr_score = evaluate_sequence(seq)
            if curr_score < best_score:
                best_score = curr_score
                best_seq = seq[:]
            # local search loop
            temp = 0.5
            while time.time() < deadline:
                # propose new sequence via random perturbation of log-space
                i,j = np.random.randint(n), np.random.randint(n)
                if i==j:
                    continue
                # shift δ from i to j
                delta = (np.random.rand()*0.1) * seq[i]  # up to 10% of i
                new_seq = seq.copy()
                new_seq[i] = max(0.0, new_seq[i] - delta)
                new_seq[j] = new_seq[j] + delta
                # optionally add small Gaussian noise to all entries
                # then renormalize to sum (not needed because sum unchanged)
                # evaluate
                new_score = evaluate_sequence(new_seq)
                if new_score < curr_score or np.random.rand() < np.exp((curr_score - new_score)/temp):
                    seq = new_seq
                    curr_score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = new_seq[:]
                # cooling
                temp *= 0.9999
                if temp < 1e-6:
                    break
                # restart local search after some steps or if no improvement
    # final: return best_seq if found else fallback uniform
    if best_seq is None:
        n=200
        best_seq = [1.0/n]*n
    return best_seq

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_score = float('inf')
    # define list of lengths
    lengths = [20, 50, 100, 200, 400, 800]  # adapt to time
    for n in lengths:
        # do a number of restarts limited by time
        restart = 0
        while time.time() < deadline and restart < 20:
            # init random Dirichlet (alpha=0.5 may give more diversity)
            seq = np.random.dirichlet(np.ones(n) * 0.5).tolist()
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq[:]
            # simple annealing loop
            t0 = 1.0
            t = t0
            step = 0
            while time.time() < deadline and step < 5000:
                # propose modification
                i, j = np.random.randint(n), np.random.randint(n)
                if i==j:
                    continue
                # modify proportionally
                delta = (np.random.rand() - 0.5) * 0.2 * seq[i] # could be +-20% of i
                # ensure positivity
                new_seq = seq[:]
                new_seq[i] = max(0.0, new_seq[i] - delta)
                new_seq[j] = max(0.0, new_seq[j] + delta)
                # renormalize to keep sum constant (optional)
                s = sum(new_seq)
                if s <= 0:
                    continue
                inv_s = 1.0 / s
                new_seq = [x * inv_s for x in new_seq]  # keep sum 1
                new_score = evaluate_sequence(new_seq)
                # acceptance criterion using simulated annealing
                if new_score < score or np.random.rand() < np.exp((score - new_score) / t):
                    seq = new_seq
                    score = new_score
                    if new_score < best_score:
                        best_score = new_score
                        best_seq = new_seq[:]
                # temperature schedule
                t = max(1e-4, t * 0.9995)
                step += 1
            restart += 1
    # if best_seq is None fallback to uniform length 100
    if best_seq is None:
        best_seq = [1.0/100] * 100
    # optionally convert to list of floats
    return [float(x) for x in best_seq]

def gradient_step(seq, eta):
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))
    n = len(seq)
    # compute gradient (subgradient) for max conv
    grad = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            grad[i] = 2 * seq[j]  # since derivative w.r.t a_i is 2 * a_{k-i}
    # step opposite gradient
    new_seq = seq - eta * grad
    # set negative to 0
    new_seq = np.maximum(new_seq, 0.0)
    # renormalize to keep sum ~1 to avoid inf case
    s = new_seq.sum()
    if s < 1e-8:
        # fallback
        return seq
    new_seq = new_seq / s
    return new_seq

"""
Propose a good step‐function sequence.

The evaluation we want to minimise is
    E(a) = 2 * n * max(conv(a,a)) / (sum(a)^2)

Scale of the vector does not matter, therefore we keep the sum close to 1.
The algorithm is a simple stochastic local search:

* Several problem sizes (n) are tried.  Smaller n gives fast iterations,
  larger n gives a better asymptotic bound.
* For each size a number of random restarts are performed.
* Each restart starts from a Dirichlet sample (α≈0.3 gives a
  moderately “spiky’’ distribution) and is then improved by a
  simulated–annealing hill‑climber.
* At every iteration a random pair of indices (i , j) is chosen and a
  small amount of mass is transferred from i to j.  After the move the
  vector is re‑normalised (sum → 1) and the new score is evaluated.
* Improvements are always kept; occasionally a worse move is accepted
  with a tiny probability to escape shallow local minima.
* The step size slowly decays, and every few thousand iterations a
  small global Gaussian perturbation is applied to shake the current
  point.
* If a global variable ``height_sequence_1`` exists (the previous best
  construction) it is also used as a start‑point.

The routine respects the 1000 s budget and returns the best sequence
found so far.  Even a modest reduction of the evaluation below the
trivial uniform value 2.0 is valuable; the method is fast enough to
reach considerably better values on typical hardware.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
def _normalize(vec: np.ndarray) -> np.ndarray:
    """Scale the vector so that its L1‑norm equals 1 (avoids the sum<0.01 safeguard)."""
    total = vec.sum()
    # Guard against degenerate vectors
    if total < 1e-12:
        total = 1.0
    return vec / total


def _initial_sequence(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random start vector of length n (Dirichlet with α<1 gives diversity)."""
    # α = 0.3 makes the distribution reasonably non‑uniform
    seq = rng.dirichlet(np.full(n, 0.3))
    # add a tiny epsilon to avoid exact zeros (helps the perturbation step)
    seq += 1e-6
    return _normalize(seq)


def _score(seq: List[float]) -> float:
    """Wrapper around the provided evaluate_sequence (assumed to be globally imported)."""
    return evaluate_sequence(seq)


def _local_search(
    seq: np.ndarray,
    deadline: float,
    max_iters: int = 30000,
    start_temp: float = 1.0,
    accept_prob: float = 0.001,
) -> np.ndarray:
    """
    Stochastic hill‑climber with a slowly decreasing step size.
    Returns the best vector encountered during the search.
    """
    n = len(seq)
    # keep a python list for evaluate_sequence (fast conversion)
    cur = seq.tolist()
    cur_score = _score(cur)
    best = cur[:]
    best_score = cur_score

    # exponential cooling for the “transfer fraction’’
    step_frac = 0.2  # maximum fraction of the donor mass that may be moved
    temp = start_temp
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_iters:
        # pick two distinct indices
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # amount taken from i – a random fraction of its current mass
        delta = rng.random() * step_frac * cur[i]
        if delta <= 0.0:
            it += 1
            continue

        new_vec = cur[:]
        new_vec[i] = max(0.0, new_vec[i] - delta)
        new_vec[j] = new_vec[j] + delta

        # re‑normalise to keep the sum close to 1 (scale does not matter)
        total = sum(new_vec)
        if total < 1e-12:
            it += 1
            continue
        new_vec = [v / total for v in new_vec]

        new_score = _score(new_vec)

        # accept if better, otherwise with a tiny probability (helps escaping)
        if new_score < cur_score or rng.random() < accept_prob:
            cur = new_vec
            cur_score = new_score
            if new_score < best_score:
                best = new_vec[:]
                best_score = new_score
                print(
                    f"  *** new best (len={n}) iter={it} score={best_score:.6f}"
                )
        # slowly shrink the step size – this is a cheap “cooling’’ schedule
        step_frac *= 0.99997
        temp *= 0.99999
        it += 1

        # occasional global perturbation to escape plateaus
        if it % 5000 == 0:
            perturb = rng.normal(scale=0.02, size=n)
            perturbed = np.maximum(np.array(cur) + perturb, 0.0)
            perturbed = _normalize(perturbed)
            perturbed = perturbed.tolist()
            pert_score = _score(perturbed)
            if pert_score < cur_score:
                cur = perturbed
                cur_score = pert_score
                if pert_score < best_score:
                    best = perturbed[:]
                    best_score = pert_score
                    print(
                        f"  *** global shake improved (len={n}) iter={it} score={best_score:.6f}"
                    )
    return np.array(best)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """Search for a low‑evaluation step‑function sequence."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    deadline = time.time() + budget_s - 0.5  # small safety margin
    best_seq: List[float] = []
    best_score = float("inf")

    # ------------------------------------------------------------------
    # optional start from the previously known construction if present
    try:
        prev = globals()["height_sequence_1"]
        if isinstance(prev, (list, tuple)) and len(prev) > 0:
            prev_arr = np.array(prev, dtype=float)
            prev_arr = _normalize(prev_arr)
            print("Including previous best construction as a start point.")
            cand = _local_search(prev_arr, deadline, max_iters=20000)
            cand_score = _score(cand.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.tolist()
                print(f"  --> improvement from previous best: {best_score:.6f}")
    except Exception:
        # no previous construction available – ignore
        pass

    # ------------------------------------------------------------------
    # try several lengths; the longer the better asymptotically,
    # but very long vectors are slower per iteration.
    lengths = [200, 400, 800]  # adjust based on remaining time
    for n in lengths:
        if time.time() >= deadline:
            break
        # decide how many restarts we can afford for this n
        restarts = 6
        for r in range(restarts):
            if time.time() >= deadline:
                break
            init_vec = _initial_sequence(n, rng)
            # quick evaluation – keep the best of the random starts
            init_score = _score(init_vec.tolist())
            if init_score < best_score:
                best_score = init_score
                best_seq = init_vec.tolist()
                print(
                    f"Random start (len={n} restart={r}) score={init_score:.6f}"
                )
            # local improvement
            improved = _local_search(init_vec, deadline, max_iters=25000)
            imp_score = _score(improved.tolist())
            if imp_score < best_score:
                best_score = imp_score
                best_seq = improved.tolist()
                print(
                    f"Improved sequence (len={n} restart={r}) score={imp_score:.6f}"
                )
            # early exit if we already beat the target
            if best_score < 1.55:  # a comfortable margin below the target 1.503
                print("Target region reached – terminating early.")
                deadline = time.time()  # force exit from outer loops
                break

    # ------------------------------------------------------------------
    # Fallback to a uniform vector if nothing better was found (should never happen)
    if not best_seq:
        n_fallback = 200
        best_seq = [1.0 / n_fallback] * n_fallback
        best_score = _score(best_seq)
        print("Fallback to uniform sequence.")

    print(f"Final best score: {best_score:.6f} (length={len(best_seq)})")
    # ensure plain python floats
    return [float(v) for v in best_seq]
