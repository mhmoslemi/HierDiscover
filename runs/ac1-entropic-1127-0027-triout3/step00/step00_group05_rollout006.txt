def flatten_step(seq):
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    # compute contributions
    n = len(seq)
    # consider all i where j=k-i in range
    i_vals = np.arange(max(0, k-(n-1)), min(n-1, k)+1)
    j_vals = k - i_vals
    # compute product
    products = seq[i_vals] * seq[j_vals]
    idx_max_local = np.argmax(products)
    i_max = i_vals[idx_max_local]
    j_max = j_vals[idx_max_local]
    # define epsilon
    eps_factor = 0.02
    eps_i = seq[i_max] * eps_factor
    eps_j = seq[j_max] * eps_factor
    # ensure we don't go negative
    eps_i = min(eps_i, seq[i_max])
    eps_j = min(eps_j, seq[j_max])
    # choose edge targets
    # put removed mass to edges with min current values to improve flattening
    # find index with smallest seq
    idx_target = np.argmin(seq)
    # distribute mass
    seq[i_max] -= eps_i
    seq[j_max] -= eps_j
    seq[idx_target] += eps_i + eps_j
    # not needed to re-normalize sum, sum unchanged
    # ensure small negative rounding
    seq = np.maximum(seq, 0.0)
    return seq
```

def solve_convolution_lp(f_sequence, rhs):
    n = len(f_sequence)
    c = -np.ones(n)  # maximize sum(b), equivalently minimize -sum
    # Build A_ub matrix: each row for each k has f_i at column j such that i+j=k
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # For i in [i_min, i_max]:
        # j = k - i
        # set row[j] = f_i
        # Equivalent: For j from max(0, k-(n-1)) to min(n-1,k):
        # row[j] = f_sequence[k-j]  (since i = k - j)
        # So we can fill using vectorized assignment:
        # Actually easiest: loop.
        for i in range(i_min, i_max+1):
            j = k - i
            row[j] = f_sequence[i]
        rows.append(row)
    # Nonnegativity constraints: b >= 0
    A_ub = np.vstack([rows, -np.eye(n)])  # b >= 0 -> -b <= 0
    b_ub = np.hstack([np.full(len(rows), rhs), np.zeros(n)])
    # Equality: sum(b) == 1 (or any target). Use A_eq.
    A_eq = np.ones((1, n))
    b_eq = np.array([1.0])
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=(0, None), method='highs', options={'presolve': True, 'time_limit': 1.0})
    if result.success:
        return result.x
    else:
        return None

"""
Propose a better step‑function sequence.

The core idea is an iterative improvement based on the linear‑program
direction used in the state‑of‑the‑art method, but with an adaptive
line‑search on the mixing parameter `t`.  A LP is solved that maximises
the total mass of a new sequence `g` under the constraint
`conv(f,g) <= M`, where `M` is the current maximal auto‑convolution of
the normalised sequence `f`.  Because the LP pushes mass into
positions where `f` is small, mixing `f` and `g` tends to flatten the
auto‑convolution and reduces the evaluation score
`2·n·max(conv(f,f))/sum(f)^2`.  Several mixing factors are tried and
the best improvement is kept.  When the LP does not yield improvement
a small random perturbation is applied.  The routine periodically
re‑starts from a fresh random sequence to avoid being stuck in a local
minimum.

The algorithm runs until the allotted time (default 1000 s) expires and
returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def normalize(seq: np.ndarray) -> np.ndarray:
    """Scale `seq` so that its sum equals sqrt(2·len(seq))."""
    total = seq.sum()
    if total == 0.0:
        return seq
    scale = np.sqrt(2.0 * len(seq)) / total
    return seq * scale


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Linear programme:
        maximise   sum(b)
        subject to  conv(f_seq, b) <= rhs   for every shift
                    b_i >= 0

    Returns the optimal b (as a numpy array) or None on failure.
    """
    n = len(f_seq)
    # objective: maximise sum(b)  -> minimise -sum(b)
    c = -np.ones(n)

    # build A_ub (one row per convolution index)
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # conv_k = sum_i f_i * b_{k-i}
        for i in range(i_min, i_max + 1):
            j = k - i
            row[j] = f_seq[i]
        rows.append(row)

    # non‑negativity: -b_i <= 0
    A_ub = np.vstack([rows, -np.eye(n)])
    b_ub = np.hstack([np.full(len(rows), rhs), np.zeros(n)])

    # No equality constraints – the objective will drive the sum as high
    # as possible while respecting the bound.
    try:
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={"presolve": True, "time_limit": 2.0},
        )
    except Exception:
        return None

    if res.success:
        return np.maximum(0.0, res.x)   # safety
    return None


def direction_via_lp(seq: np.ndarray):
    """
    Return a new sequence `g` that (after normalisation) satisfies
    conv(f,g) ≤ M, where M is the current maximal auto‑convolution of f.
    The returned `g` is already normalised to sqrt(2·len(seq)).
    """
    n = len(seq)
    f_norm = normalize(seq)
    conv_f = np.convolve(f_norm, f_norm)
    M = np.max(conv_f)

    b = solve_convolution_lp(f_norm, M)
    if b is None:
        return None
    g_norm = normalize(b)
    return g_norm


def fast_score(seq: np.ndarray) -> float:
    """Quick evaluation of the objective (identical to evaluate_sequence
    for non‑negative, non‑clipped inputs)."""
    n = len(seq)
    total = seq.sum()
    if total < 0.01:
        return float("inf")
    max_c = np.max(np.convolve(seq, seq))
    return float(2.0 * n * max_c / (total ** 2))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the outer loop
    deadline = start_time + budget_s - 5.0

    # ----------------------------------------------
    # initialise the current best solution
    # ----------------------------------------------
    try:
        # height_sequence_1 may be provided by the evaluation harness
        best_seq = np.array(height_sequence_1, dtype=float)
        best_seq = np.maximum(best_seq, 0.0)      # safety
    except Exception:
        best_seq = None

    best_score = float("inf")
    t_candidates = [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7]

    # -------------------------------------------------
    # outer loop – occasional restarts to escape plateaus
    # -------------------------------------------------
    while time.time() < deadline:
        # decide whether to restart from a fresh random sequence
        if best_seq is None or np.random.rand() < 0.03:
            n = np.random.randint(300, 1200)            # moderate length
            seq = np.random.rand(n)
            # occasionally enforce sparsity (helps diversity)
            if np.random.rand() < 0.3:
                mask = np.random.rand(n) < 0.6
                seq[~mask] = 0.0
            seq = normalize(seq)
        else:
            # continue from the current best
            seq = best_seq.copy()

        cur_score = fast_score(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[init] score={best_score:.6f}, n={len(seq)}")

        # -------------------------------------------------
        # inner optimisation loop (LP direction + line search)
        # -------------------------------------------------
        inner_deadline = time.time() + min(30.0, deadline - time.time())
        stagnation = 0

        while time.time() < inner_deadline:
            # ----- LP direction -------------------------------------------------
            g = direction_via_lp(seq)
            improved = False

            if g is not None:
                # try several mixing factors
                for t in t_candidates:
                    cand = (1.0 - t) * seq + t * g
                    cand = normalize(cand)          # protect against drift
                    cand_score = fast_score(cand)

                    if cand_score + 1e-12 < best_score:
                        best_score = cand_score
                        best_seq = cand.copy()
                        seq = cand.copy()
                        improved = True
                        stagnation = 0
                        print(
                            f"[mix] t={t: .2f} score={best_score:.6f} time={time.time()-start_time: .1f}s"
                        )
                        break  # accept the first improving t
                if not improved:
                    # no t yielded improvement – make a tiny random move
                    idx = np.random.randint(len(seq))
                    delta = (np.random.rand() - 0.5) * 0.02 * max(seq[idx], 1e-12)
                    seq[idx] = max(0.0, seq[idx] + delta)
                    seq = normalize(seq)
            else:
                # LP failed – fallback to a random perturbation
                idx = np.random.randint(len(seq))
                seq[idx] = max(0.0, seq[idx] + (np.random.rand() - 0.5) * 0.05)
                seq = normalize(seq)

            # ---- check progress -------------------------------------------------
            cur_score = fast_score(seq)
            if cur_score + 1e-12 < best_score:
                best_score = cur_score
                best_seq = seq.copy()
                improved = True
                stagnation = 0
                print(
                    f"[pert] score={best_score:.6f} time={time.time()-start_time: .1f}s"
                )
            else:
                stagnation += 1

            # stop inner loop if we see no improvement for a while
            if stagnation > 250:
                break

    # final sanity check – evaluate with the official routine
    # (the harness will do this as well, but we keep a copy handy)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluated score={final_score:.6f}")
    except Exception:
        pass

    return best_seq.tolist()
