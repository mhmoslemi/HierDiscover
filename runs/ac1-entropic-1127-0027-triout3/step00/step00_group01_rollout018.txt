def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Generates a step-height sequence based on the continuous arcsine distribution,
    which has the property that its self-convolution is (asymptotically) constant.
    This yields an evaluation close to the theoretical lower bound (≈1).
    The function optionally tries a few sequence lengths and keeps the best.
    """
    import numpy as np, time, math
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5
    best_score = float('inf')
    best_seq = None

    # lengths to try, bounded by a reasonable max (e.g., 10000) to keep runtime low.
    lengths = [500, 1000, 2000, 4000, 6000, 8000]   # we can adjust if deadline runs out

    for n in lengths:
        if time.time() > deadline:
            break
        # compute arcsine-shaped base values
        i = np.arange(n, dtype=np.float64)
        # use midpoint rule to avoid zero denominator
        x = i + 0.5
        y = n - i - 0.5
        base = 1.0 / np.sqrt(x * y)
        # normalise to sum = 1 (scale invariance of the objective)
        s = base.sum()
        a = base / s
        # evaluate
        try:
            score = evaluate_sequence(a.tolist())
        except Exception as e:
            # safety: treat as INF
            score = float('inf')
        print(f"Length {n}, score {score:.6f}")
        if score < best_score:
            best_score = score
            best_seq = a.tolist()
            # early exit if we reach very good bound close to 1.
            if best_score < 1.05:
                # we have a near optimal solution; continue trying maybe larger n for tiny improvements
                pass
    # If we still have time, run a few local refinement steps on best_seq
    if best_seq is not None and time.time() < deadline:
        a = np.array(best_seq, dtype=np.float64)
        # simple iterative redistribution to reduce max_b a bit
        for it in range(2000):
            if time.time() > deadline:
                break
            b = np.convolve(a, a)
            k_max = int(np.argmax(b))
            # construct contribution weights
            n = len(a)
            idx = np.arange(n)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[idx[mask]] * a[partner[mask]]
            total = contrib.sum()
            if total == 0:
                break
            # compute how much mass to move away from heavy indices
            # choose a small step factor relative to total sum (which is 1 after normalisation)
            step = 0.005  # 0.5% of the total mass each iteration
            # amount to take from each index proportional to its contribution
            delta = np.minimum(step * contrib / total, a * 0.5)   # never take more than half of current value
            removed = delta.sum()
            if removed == 0:
                break
            a -= delta
            # distribute removed mass uniformly
            a += removed / n
            # enforce non‑negativity and renormalise exactly to keep sum = 1
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0:
                break
            a /= s
        # final evaluation
        final_score = evaluate_sequence(a.tolist())
        print(f"After refinement length {len(a)} score {final_score:.6f}")
        if final_score < best_score:
            best_seq = a.tolist()
            best_score = final_score

    print(f"Best score returned: {best_score:.6f} (length {len(best_seq)})")
    return best_seq

def flatten_sequence(a, iterations, step):
    for it in range(iterations):
        b = np.convolve(a, a)
        k_max = np.argmax(b)
        # compute product contributions
        idx = np.arange(len(a))
        partner = k_max - idx
        mask = (partner >= 0) & (partner < len(a))
        contribution = np.zeros_like(a)
        contribution[mask] = a[idx[mask]] * a[partner[mask]]
        total = contribution.sum()
        if total == 0:
            break
        # decide amount to shift: delta = step * (contribution / total) * a (or something)
        # but we need to keep a nonnegative
        delta = step * (contribution / total)  # sum of delta = step
        # ensure we don't subtract more than a's value
        delta = np.minimum(delta, a) * 0.5   # at most half of each a_i
        # subtract
        a_new = a - delta
        # total removed
        removed = delta.sum()
        # distribute removed mass uniformly
        a_new += removed / len(a)
        # enforce nonneg
        a_new = np.maximum(a_new, 0)
        # renormalize sum to 1
        a_new /= a_new.sum()
        a = a_new
    return a

"""
Propose a step‑function sequence that minimizes the evaluation function.

The evaluation is scale‑invariant:
 E(a) = 2·n·max(a∗a) / (sum(a)²).

The optimal (theoretical) lower bound is 2·n·(average of the convolution) /
(sum²) = 2·n/(2n‑1) → 1 when n is large.
Thus we try to make the self‑convolution as flat as possible.
The algorithm works as follows:

1. Choose a moderate length (≈4000–6000).  Larger n gives a better
   theoretical bound but makes each convolution more expensive.
2. Initialise the coefficient vector with a random Dirichlet draw
   (uniform on the simplex).  This breaks the symmetry of the uniform
   sequence (which gives the trivial bound 2.0).
3. Repeatedly “flatten’’ the worst convolution entry:
   * Compute the full linear convolution b = a∗a.
   * Locate the index k_max where b is maximal.
   * For each position i that participates in this sum, the contribution
     is c_i = a[i]·a[k_max‑i] (zero if the partner is out of range).
   * Remove a small fraction of mass from the heavily contributing
     positions (proportional to c_i) and redistribute the removed mass
     uniformly over the whole vector.
   * Renormalise to keep sum(a)=1.
4. Occasionally (10 % of the steps) inject a tiny random perturbation
   to avoid getting stuck.
5. Keep the best sequence found (lowest evaluation) and, when time runs
   out, return it.

The whole search runs inside the allotted budget (default 1000 s) and
does not rely on any external files.  All numbers stay well below the
security limits of ``evaluate_sequence``.
"""

import numpy as np
import time
import math


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Generate a non‑negative step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : int, optional
        Time budget in seconds for the search.
    **kwargs :
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best coefficient list discovered before the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # a tiny safety margin

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate and return a large number on error.
    # ------------------------------------------------------------------
    def safe_evaluate(seq):
        try:
            return evaluate_sequence(seq)
        except Exception:
            return float("inf")

    # ------------------------------------------------------------------
    # Parameters of the search
    # ------------------------------------------------------------------
    # Lengths to try – larger lengths give a tighter theoretical bound but
    # cost more per iteration.  We keep the list short to stay well inside the
    # time budget.
    length_candidates = [2000, 3000, 4000, 5000, 6000]
    best_score = float("inf")
    best_seq = None

    # Core optimisation loop for a single length --------------------------------
    def optimise_length(n):
        nonlocal best_score, best_seq, deadline

        # Initialise with a Dirichlet(1,…,1) draw – uniform on the simplex.
        a = np.random.dirichlet(np.ones(n))
        # Ensure we stay strictly positive (required for the redistribution step).
        a = np.maximum(a, 1e-12)
        a /= a.sum()

        # Keep the best sequence found for this length.
        local_best_score = safe_evaluate(a.tolist())
        local_best_seq = a.copy()
        if local_best_score < best_score:
            best_score = local_best_score
            best_seq = a.tolist()

        # Parameters of the flattening loop.
        max_iter = 200                 # absolute upper bound
        iter_cnt = 0
        step = 0.02                    # fraction of the total mass moved per step

        while time.time() < deadline and iter_cnt < max_iter:
            iter_cnt += 1

            # Occasionally add a tiny random perturbation to escape plateaus.
            if np.random.rand() < 0.10:
                noise = np.random.randn(n) * 0.001
                a = np.maximum(a + noise, 0.0)
                a /= a.sum()
                continue

            # Full linear convolution (numpy's implementation is C‑level and fast).
            b = np.convolve(a, a)
            k_max = int(np.argmax(b))

            # Compute the contribution of each index to the maximal entry.
            idx = np.arange(n)
            partner = k_max - idx
            mask = (partner >= 0) & (partner < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[idx[mask]] * a[partner[mask]]

            total_contrib = contrib.sum()
            if total_contrib == 0.0:
                # Degenerate case – break out.
                break

            # Amount of mass to relocate away from heavy contributors.
            # We move a fraction `step` of the total mass (i.e. step∈(0,1)).
            delta = step * contrib / total_contrib           # sum(delta) == step
            # Do not remove more than half of any entry (keeps non‑negativity safe).
            delta = np.minimum(delta, a * 0.5)

            removed = delta.sum()
            a = a - delta
            # Uniform redistribution of the removed mass.
            a += removed / n

            # Numerical safety – clip and renormalise.
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                break
            a /= s

            # Evaluate and keep the best found.
            cur_score = safe_evaluate(a.tolist())
            if cur_score < local_best_score:
                local_best_score = cur_score
                local_best_seq = a.copy()
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = a.tolist()
                    print(f"[{time.time() - start_time:.1f}s] New global best: "
                          f"{best_score:.6f} (len={n})")

        # After the loop, keep the best sequence discovered for this length.
        return local_best_seq

    # ------------------------------------------------------------------
    # Main driver – try the different lengths.
    # ------------------------------------------------------------------
    for n in length_candidates:
        if time.time() >= deadline:
            break
        print(f"--- Optimising length {n} ---")
        final_seq = optimise_length(n)
        # final evaluation for this length (already computed inside)
        # but we keep the printed best scores from inside the loop.

    # ------------------------------------------------------------------
    # In the very unlikely case nothing was found, fall back to uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 3000
        uniform = [1.0 / n] * n
        best_seq = uniform
        best_score = safe_evaluate(best_seq)

    print(f"Search finished after {time.time() - start_time:.1f}s – "
          f"best score {best_score:.6f} with length {len(best_seq)}")
    return best_seq
