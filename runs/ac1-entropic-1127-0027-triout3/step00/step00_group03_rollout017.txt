    # perform random Dirichlet draws

    eps = np.random.rand() * min(seq[i], 0.01)  # lose at most 0.01

"""Propose a low‑evaluation step sequence.

Algorithm
---------
* Sample many probability vectors using a Dirichlet distribution with
  different concentration parameters (α).  Small α gives a few large
  coefficients and many tiny ones – a shape that can reduce the
  peak of the self‑convolution.
* For each sampled vector sort it descending and interleave the largest
  and smallest values (largest‑smallest permutation).  This arrangement
  minimises the dot‑product of the sequence with its reverse, i.e. the
  central convolution value.
* Evaluate the candidate.  Keep the best one seen.
* Perform a cheap local search on the current candidate:
    – random swaps of two positions,
    – small mass transfers between two positions.
  Accept only improving moves (hill‑climbing).  This refines the
  ordering and the exact values.
* Run the above loop until the time budget (≈ 1000 s) is exhausted.

The evaluation function `evaluate_sequence` is scale‑invariant, so all
generated sequences are normalised to sum 1 (the Dirichlet draws already
have this property).  The final best sequence is returned as a plain
list of floats.
"""

import time
import random
import numpy as np

def _score(seq: np.ndarray) -> float:
    """Fast equivalent of evaluate_sequence for internal use."""
    n = len(seq)
    conv = np.convolve(seq, seq)
    maxb = np.max(conv)
    s = np.sum(seq)
    if s < 0.01:                     # protect division by zero
        return float("inf")
    return 2.0 * n * maxb / (s * s)


def _interleave_desc_asc(weights: np.ndarray) -> np.ndarray:
    """
    Return a permutation of `weights` where the largest and smallest
    values are placed at opposite ends (largest‑smallest interleaving).
    This minimises the central convolution term.
    """
    sorted_w = np.sort(weights)[::-1]          # descending
    n = len(weights)
    out = np.empty_like(weights)
    left, right = 0, n - 1
    idx = 0
    while left <= right:
        out[left] = sorted_w[idx]
        idx += 1
        left += 1
        if left <= right:
            out[right] = sorted_w[idx]
            idx += 1
            right -= 1
    return out


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence that minimises the evaluation
    function.  The function respects the 1000 s time budget and returns
    the best sequence (list of floats) found.
    """
    # ------------------------------------------------------------------
    # basic setup
    # ------------------------------------------------------------------
    np.random.seed(seed)
    random.seed(seed)

    # use the length of the previously known construction if available
    n = len(globals().get("height_sequence_1", [])) or 5300

    # start from a trivial uniform sequence (all ones); its score is 2.0
    best_seq = np.ones(n, dtype=float)
    best_score = _score(best_seq)

    # optionally try the previous state‑of‑the‑art sequence
    if "height_sequence_1" in globals():
        try:
            cand = np.array(height_sequence_1, dtype=float)
            sc = _score(cand)
            if sc < best_score:
                best_seq = cand
                best_score = sc
        except Exception:
            pass

    print(f"Initial best score: {best_score:.6f}")

    # ------------------------------------------------------------------
    # search parameters
    # ------------------------------------------------------------------
    alpha_choices = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0]   # concentration values
    deadline = time.time() + budget_s - 5.0           # leave a few seconds margin

    # ------------------------------------------------------------------
    # main search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # 1) draw a random probability vector (Dirichlet)
        # ------------------------------------------------------------------
        alpha = random.choice(alpha_choices)
        # Dirichlet via Gamma – works for any positive alpha
        gamma_samples = np.random.gamma(shape=alpha, scale=1.0, size=n)
        if np.sum(gamma_samples) == 0:
            continue
        prob_vec = gamma_samples / np.sum(gamma_samples)

        # ------------------------------------------------------------------
        # 2) arrange it with the large‑small interleaving
        # ------------------------------------------------------------------
        seq = _interleave_desc_asc(prob_vec)

        # ------------------------------------------------------------------
        # 3) evaluate the candidate
        # ------------------------------------------------------------------
        cur_score = _score(seq)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = seq.copy()
            print(f"[{time.time():.1f}] New best from init (α={alpha}): {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4) cheap local hill‑climbing on this candidate
        # ------------------------------------------------------------------
        # number of local attempts is kept modest to stay within the time budget
        local_iters = 40
        curr_seq = seq.copy()
        curr_score = cur_score

        for _ in range(local_iters):
            # decide between a swap (70 %) and a tiny mass transfer (30 %)
            if random.random() < 0.7:
                # ----- swap two positions -------------------------------------------------
                i, j = np.random.randint(n, size=2)
                if i == j:
                    continue
                new_seq = curr_seq.copy()
                new_seq[i], new_seq[j] = new_seq[j], new_seq[i]
            else:
                # ----- move a small amount of mass from i to j ---------------------------
                i, j = np.random.randint(n, size=2)
                if i == j or curr_seq[i] <= 0:
                    continue
                # size of the transfer – up to 2 % of the source entry
                eps = curr_seq[i] * random.random() * 0.02
                new_seq = curr_seq.copy()
                new_seq[i] -= eps
                new_seq[j] += eps

            # quick evaluation
            new_score = _score(new_seq)

            # accept only improvements (pure hill‑climbing)
            if new_score < curr_score:
                curr_seq = new_seq
                curr_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq.copy()
                    print(f"[{time.time():.1f}] Improved to {best_score:.6f}")

        # occasionally break early if we already reached a very good target
        # (target 1.5 is far below the trivial bound 2.0)
        if best_score < 1.51:
            # keep searching a bit longer to see if we can go even lower
            pass

    # ------------------------------------------------------------------
    # return the best sequence found (as a plain list of floats)
    # ------------------------------------------------------------------
    print(f"Finished search – best score: {best_score:.6f}")
    return best_seq.tolist()
