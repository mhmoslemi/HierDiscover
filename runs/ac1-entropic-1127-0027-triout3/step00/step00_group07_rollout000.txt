def solve_convolution_lp(f, rhs):
    n = len(f)
    # build convolution matrix A (shape (2n-1, n))
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        # for each j in [0, n-1], coefficient = f[k - j] if valid else 0
        low = max(0, k - (n-1))
        high = min(n-1, k)
        # j from low to high
        # coefficient for variable b_j is f_{k - j}
        # vector slice
        A[k, low:high+1] = f[k - np.arange(low, high+1)]
    # Linear program
    c = -np.ones(n)
    b_ub = np.full(2*n - 1, rhs, dtype=float)
    result = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n,
                     method='highs', options={'time_limit': 0.8})
    if result.success:
        return result.x
    return None

Propose a candidate step‑function sequence that aims to lower the value of
evaluate_sequence (i.e. minimise 2·n·max(conv(a,a))/sum(a)^2).

Algorithm (high‑level)
----------------------
1. Choose a length n (by default the length of the known sequence
   `height_sequence_1`).  Initialise a feasible non‑negative sequence a of
   length n and enforce the normalisation
        sum(a) = sqrt(2·n)
   – this makes the evaluation equal to the peak of the self‑convolution.

2. Iteratively improve the sequence.
   * Compute the current peak M = max(conv(a,a)).
   * Solve a linear programme that finds a non‑negative vector g with maximal
     sum(g) under the linear constraints   conv(a,g) ≤ M   (component‑wise).
     The LP is:
         maximise   1ᵀ g
         subject to A·g ≤ M·1 ,   g ≥ 0
     where A is the (2n‑1)×n convolution matrix built from a.
   * Normalise g to the same total mass (sqrt(2·n)).
   * Form a convex combination  a₊ = (1‑t)·a + t·g   with a small step size t.
     Several values of t (0.05,0.1,0.2,0.4) are tried and the best one (lowest
     evaluation) is kept.
   * If the LP fails (rare) perform a cheap random perturbation of a few
     entries and re‑normalise.

3. Keep the best sequence seen so far.  Periodically perform a random
   “shake‑up’’ of 10 % of the coordinates to escape possible local minima.

The whole loop stops a few seconds before the allotted budget and returns the
best sequence found.  All operations respect the non‑negativity and the
normalisation required by `evaluate_sequence`.  The LP uses SciPy’s HiGHS
solver which is fast for the problem sizes (n≈10³) we consider.

The method is a simple deterministic+stochastic hybrid and improves upon the

* Trying several step sizes (line‑search) instead of a fixed small t.
* Adding occasional random re‑initialisation to explore new basins.
* Normalising after each operation to keep the evaluation scale‑invariant.

import time
import numpy as np
from scipy.optimize import linprog

def _normalise(seq: np.ndarray) -> np.ndarray:
    """
    Scale the non‑negative sequence so that its sum equals sqrt(2·len(seq)).
    Values are clipped to the interval [0, 1000] (the limits of evaluate_sequence).
    """
    s = np.sum(seq)
    # Guard against an (almost) zero sum – this would lead to evaluate_sequence = inf.
    if s < 1e-12:
        seq = np.ones_like(seq) * 1e-6
        s = np.sum(seq)
    target = np.sqrt(2.0 * len(seq))
    seq = seq * (target / s)
    # enforce the hard caps used inside evaluate_sequence
    seq = np.clip(seq, 0.0, 1000.0)
    return seq


def _build_convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1)×n) such that A @ g = conv(f, g) for any g.
    Row k corresponds to the k‑th entry of the discrete convolution.
    """
    n = len(f)
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=float)

    # For each row k, the coefficient of variable g[j] is f[k‑j] if 0 ≤ k‑j < n.
    for k in range(rows):
        # j runs over indices where k-j is a valid index of f.
        j_min = max(0, k - (n - 1))
        j_max = min(k, n - 1)
        # Vector of j values: j_min … j_max
        js = np.arange(j_min, j_max + 1, dtype=int)
        # Corresponding i = k - j
        is_ = k - js
        A[k, js] = f[is_]
    return A


def _solve_convolution_lp(f: np.ndarray, rhs: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   1ᵀ g
        subject to conv(f, g) ≤ rhs   (component‑wise)
                   g ≥ 0
    Returns the optimal g (as a NumPy array) or None if the solver fails.
    """
    n = len(f)
    A = _build_convolution_matrix(f)          # shape (2n‑1, n)
    b = np.full(2 * n - 1, rhs, dtype=float)  # RHS of each inequality

    # Objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=float)

    # Use the HiGHS solver; give it a modest time limit.
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"time_limit": 0.8, "presolve": True},
    )
    if res.success:
        return res.x
    return None
