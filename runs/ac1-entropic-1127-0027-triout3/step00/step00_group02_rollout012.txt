"""
Propose a near‑optimal step‑function (non‑negative height sequence).

The evaluation metric (after normalising the sum to √(2 n)) is simply the
maximum entry of the aperiodic auto‑convolution of the sequence.
The algorithm therefore tries to **flatten the convolution**.

Strategy
--------
* Fix the length `n` (use the length of the supplied `height_sequence_1`
  if possible).
* Normalise the sum to `S = sqrt(2*n)` – with this normalisation the
  score returned by `evaluate_sequence` equals the maximal convolution
  value.
* Initialise the sequence with a uniform distribution plus a tiny random
  perturbation.
* Repeatedly perform a simulated‑annealing style move that moves a small
  amount of mass from an index that participates strongly in the current
  maximal convolution entry to a random other index.
* Accept the move if it improves the score, otherwise accept it with a
  small probability that gradually cools down.
* Keep the best sequence ever seen.
* The loop stops after the allotted time budget (default 1000 s).

The code is written with only top‑level functions, uses NumPy, and avoids
any filesystem or network access.

"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast maximal aperiodic auto‑convolution using FFT
# ----------------------------------------------------------------------
def _max_convolution(a: np.ndarray) -> float:
    """
    Return max_k (a * a)[k] using a power‑of‑two FFT.
    The function works for a 1‑D non‑negative array ``a``.
    """
    n = a.shape[0]
    # size is the next power of two >= 2*n-1 (FFT length)
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence with a low value of ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the
    best sequence found (as a ``list`` of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem dimensions and normalisation
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, list):
        n = len(height_sequence_1)
    else:
        n = 2000               # a safe default if the global is missing
    S = np.sqrt(2.0 * n)      # required sum; with this sum score == max conv

    # ------------------------------------------------------------------
    # 2. FFT parameters (constant for a given n)
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 3. Function to compute the score (max convolution) quickly
    # ------------------------------------------------------------------
    def score(seq: np.ndarray) -> float:
        """Return the current evaluation (max convolution)."""
        # seq already has the correct sum, no scaling needed
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        return float(conv.max())

    # ------------------------------------------------------------------
    # 4. Initial sequence (uniform + tiny noise)
    # ------------------------------------------------------------------
    a = np.full(n, S / n, dtype=np.float64)
    a += rng.normal(scale=0.001 * S / n, size=n)   # small perturbation
    a = np.clip(a, 0.0, None)
    a *= S / a.sum()                               # renormalise exactly

    cur_seq = a.copy()
    cur_score = score(cur_seq)

    best_seq = cur_seq.copy()
    best_score = cur_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Simulated‑annealing style optimisation
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a tiny safety margin

    # temperature schedule: start relatively high, exponential cooling
    temperature = cur_score * 0.5
    cooling_rate = 0.99995

    # main optimisation loop
    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Identify the current most critical convolution index
        # --------------------------------------------------------------
        fa = np.fft.rfft(cur_seq, fft_len)
        conv_full = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_val = float(conv_full.max())
        k_max = int(conv_full.argmax())

        # --------------------------------------------------------------
        # Choose a pair (i, j) that contributes to the maximal entry
        # --------------------------------------------------------------
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx_range = np.arange(i_low, i_high + 1, dtype=np.int64)
        j_range = k_max - idx_range

        # product contributed by each admissible i
        contrib = cur_seq[idx_range] * cur_seq[j_range]
        # pick the pair with the largest contribution (greedy)
        best_pair_idx = int(np.argmax(contrib))
        i = int(idx_range[best_pair_idx])
        j = int(j_range[best_pair_idx])

        # --------------------------------------------------------------
        # Decide which index will donate a little mass
        # --------------------------------------------------------------
        if cur_seq[i] >= cur_seq[j]:
            donor = i
            partner = j
        else:
            donor = j
            partner = i

        # Choose a recipient that is **not** the partner (to avoid undoing)
        recipient = rng.integers(0, n)
        while recipient == donor or recipient == partner:
            recipient = rng.integers(0, n)

        # Amount to move – up to 10 % of the donor value
        delta = rng.random() * 0.1 * cur_seq[donor]
        if delta <= 0.0:
            continue

        # --------------------------------------------------------------
        # Apply the move (mass transfer)
        # --------------------------------------------------------------
        new_seq = cur_seq.copy()
        new_seq[donor] -= delta
        new_seq[recipient] += delta

        # (numerical safety)
        if new_seq[donor] < 0:
            new_seq[donor] = 0.0

        # --------------------------------------------------------------
        # Evaluate the new sequence
        # --------------------------------------------------------------
        new_score = score(new_seq)

        # --------------------------------------------------------------
        # Acceptance rule (simulated annealing)
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score - 1e-12:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = new_seq
            cur_score = new_score
            # Keep the best ever seen
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_seq = new_seq.copy()
                elapsed = time.time() - start_time
                print(
                    f"[{elapsed:6.1f}s] new best score = {best_score:.6f} "
                    f"(iteration {iteration})"
                )

        # --------------------------------------------------------------
        # Cool down
        # --------------------------------------------------------------
        temperature *= cooling_rate
        if temperature < 1e-8:            # avoid stagnation
            temperature = 1e-8

        # Occasionally inject a tiny random jitter to escape plateaus
        if rng.random() < 0.001:
            idx = rng.integers(0, n)
            cur_seq[idx] = max(
                0.0, cur_seq[idx] + rng.normal(scale=0.005 * S / n)
            )
            cur_seq *= S / cur_seq.sum()   # renormalise to keep the sum exact

    # ------------------------------------------------------------------
    # 6. Final sanity check (renormalise once more)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, S / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, None)
    total = best_seq.sum()
    if total <= 0:
        best_seq = np.full(n, S / n, dtype=np.float64)
    else:
        best_seq *= S / total

    # Convert to plain Python list as required by the framework
    return best_seq.tolist()
