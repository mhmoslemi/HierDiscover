import cvxpy as cp

def min_max_convolution_opt(n, time_limit=60):
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()
    constraints = [cp.sum(a) == 1] # can also set some normalization
    # We're interested in minimizing max_b = t.
    # Build conv constraints
    rows = []
    for k in range(2*n-1):
        # Build matrix Q_k
        # Use dictionary to store entries
        data = []
        rows_idx = []
        cols_idx = []
        # loop i from max(0,k-(n-1)) to min(n-1,k)
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        for i in range(i_start, i_end+1):
            j = k - i
            # i <= j? we add contributions accordingly.
            if i < j:
                # contribution 0.5 * (a_i * a_j + a_j * a_i) = a_i * a_j
                # using outer product for both entries
                rows_idx.append(i); cols_idx.append(j); data.append(0.5)
                rows_idx.append(j); cols_idx.append(i); data.append(0.5)
            else: # i == j
                rows_idx.append(i); cols_idx.append(i); data.append(1.0)
        Qk = sp.csc_matrix((data, (rows_idx, cols_idx)), shape=(n, n))
        constraints.append(cp.quad_form(a, Qk) <= t)
    # also need t >= 0
    constraints.append(t >= 0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.SCS, max_iters=5000, verbose=False, eps=1e-5, acceleration_lookback=0)
    except cp.error.SolverError:
        return None
    if a.value is None:
        return None
    return a.value, t.value

"""
Propose a step‑function (non‑negative sequence) with a very small
evaluation value.

The evaluation metric is

    E = 2 * n * max(conv(a,a)) / (sum(a))**2 .

For a fixed L1‑norm of the sequence, the numerator is just the maximal
entry of the auto‑convolution.  The problem

    minimise   t
    s.t.       a_i ≥ 0,
               sum_i a_i = 1,
               (a * a)_k ≤ t   for all k = 0,…,2n‑2

is a convex quadratically‑constrained program (QCQP).  Each constraint
has the form aᵀ Q_k a ≤ t with a PSD matrix Q_k (the anti‑diagonal
matrix that picks the terms i+j=k).  Solving the QCQP makes the
convolution as flat as possible; the theoretical lower bound of the
objective is 2 n/(2 n‑1)≈1, so for moderate n we can easily beat the
target 1.503.

The function `solve_flat_convolution` builds the QCQP with CVXPY (using
the SCS solver) and returns the optimal sequence.  `propose_candidate`
tries a few problem sizes, keeps the best result found within the
time budget and finally returns the corresponding sequence.
"""

import time
import numpy as np
import scipy.sparse as sp
import cvxpy as cp

def _build_Qk(n, k):
    """
    Construct the PSD matrix Q_k (in sparse format) such that
        a.T @ Q_k @ a = sum_{i+j=k} a_i a_j .
    For i=j the coefficient is 1, for i≠j the coefficient is 1/2
    (because the product appears twice in the full quadratic form).
    """
    rows = []
    cols = []
    data = []

    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    for i in range(i_start, i_end + 1):
        j = k - i
        if i < j:
            # off‑diagonal entry appears twice → factor 0.5 each
            rows.extend([i, j])
            cols.extend([j, i])
            data.extend([0.5, 0.5])
        else:  # i == j
            rows.append(i)
            cols.append(i)
            data.append(1.0)

    return sp.csc_matrix((data, (rows, cols)), shape=(n, n))


def solve_flat_convolution(n, max_iters=15000):
    """
    Solve the convex QCQP that minimises the maximum entry of the
    auto‑convolution of a non‑negative sequence of length n with unit
    L1‑norm.

    Parameters
    ----------
    n : int
        Length of the sequence.
    max_iters : int, optional
        Maximum number of SCS iterations (default 15000).

    Returns
    -------
    a_opt : np.ndarray or None
        Optimal sequence (length n) if the solver succeeded, else None.
    t_opt : float or None
        The optimal upper bound on the convolution (max entry).
    """
    # decision variables
    a = cp.Variable(n, nonneg=True)
    t = cp.Variable()

    constraints = [cp.sum(a) == 1, t >= 0]

    # add one quadratic constraint per convolution index
    for k in range(2 * n - 1):
        Qk = _build_Qk(n, k)
        constraints.append(cp.quad_form(a, Qk) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    try:
        prob.solve(solver=cp.SCS, max_iters=max_iters, verbose=False)
    except Exception:          # pragma: no cover
        return None, None

    if a.value is None:
        return None, None

    a_opt = np.asarray(a.value).flatten()
    # numerical noise can produce tiny negatives – clip them
    a_opt = np.maximum(a_opt, 0.0)
    t_opt = float(t.value) if t.value is not None else None
    return a_opt, t_opt


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a sequence with a low evaluation score.

    The routine tries a few problem sizes (n = 150, 200, 300, 400) and
    solves the convex QCQP described above.  The best sequence found
    within the remaining time budget is returned.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 2.0     # leave a few seconds margin

    best_seq = None
    best_score = float('inf')

    # problem sizes to try – larger n gives a tighter theoretical bound
    candidate_ns = [150, 200, 300, 400]

    for n in candidate_ns:
        if time.time() >= deadline:
            break

        # allow a modest number of SCS iterations; increase if time permits
        remaining = deadline - time.time()
        max_iters = 12000
        if remaining < 30:
            max_iters = 5000

        print(f"[{time.time():.2f}] Solving for n={n} (max_iters={max_iters})")
        a_opt, t_opt = solve_flat_convolution(n, max_iters=max_iters)

        if a_opt is None or t_opt is None:
            print(f"  → Solver failed for n={n}")
            continue

        # Scale sequence arbitrarily – scaling does not affect the score.
        # Keep values safely below the 1000‑clip used in evaluate_sequence.
        seq = a_opt.tolist()
        score = evaluate_sequence(seq)

        print(f"  → n={n}, t_opt≈{t_opt:.6e}, score={score:.6f}")

        if score < best_score:
            best_score = score
            best_seq = seq

        # stop early if we have already beaten the target
        if best_score < 1.503:
            print("Target achieved – stopping early.")
            break

    # Fallback: if nothing succeeded, use a simple uniform sequence.
    if best_seq is None:
        n = 200
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence.")

    print(f"Returning sequence of length {len(best_seq)} with score {best_score:.6f}")
    return best_seq
