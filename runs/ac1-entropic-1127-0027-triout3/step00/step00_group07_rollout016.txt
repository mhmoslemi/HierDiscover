def hill_climb(seq_initial, time_limit):
    seq = np.array(seq_initial, dtype=np.float64)
    n = len(seq)
    # ensure sum=1
    s = seq.sum()
    if s == 0:
        seq = np.full(n, 1.0/n)
    else:
        seq /= s
    best_seq = seq.copy()
    best_val = evaluate_sequence(best_seq.tolist())
    cur_seq = seq.copy()
    cur_val = best_val
    start = time.time()
    # step size as fraction of typical a_i ~ 1/n
    step = 0.5 / n
    while time.time() < time_limit:
        # compute convolution for direction
        conv = np.convolve(cur_seq, cur_seq)
        k_max = np.argmax(conv)
        # compute contributions
        contrib = np.zeros(n)
        # vectorized: indices i where k_max-i in range
        # j = k_max - i
        # compute mask
        i_idx = np.arange(n)
        j_idx = k_max - i_idx
        valid = (j_idx >= 0) & (j_idx < n)
        contrib[valid] = cur_seq[i_idx[valid]] * cur_seq[j_idx[valid]]
        # find source index i_src with max contribution
        i_src = int(np.argmax(contrib))
        # find target index i_tgt with small contribution
        # pick random among bottom 30%
        low_mask = contrib < np.percentile(contrib, 30)
        # ensure not empty
        if np.any(low_mask):
            i_tgt = int(np.random.choice(np.where(low_mask)[0]))
        else:
            i_tgt = np.random.randint(n)
            while i_tgt == i_src:
                i_tgt = np.random.randint(n)
        # compute max delta allowed
        # we will move mass from src to tgt: delta positive (increase tgt, decrease src)
        max_delta = min(step, cur_seq[i_src])
        # ensure we don't make target exceed 1000 or cause sum too large, not necessary
        if max_delta <= 0:
            continue
        # random fraction up to max_delta
        delta = max_delta * np.random.rand()
        # apply move
        new_seq = cur_seq.copy()
        new_seq[i_src] -= delta
        new_seq[i_tgt] += delta
        # evaluate
        new_val = evaluate_sequence(new_seq.tolist())
        if new_val < cur_val:
            cur_seq, cur_val = new_seq, new_val
            if new_val < best_val:
                best_seq, best_val = new_seq.copy(), new_val
                print(f"New best {best_val:.6f} (n={n}) at {time.time()-start:.1f}s")
        # occasional random perturbation to avoid stagnation
        if np.random.rand() < 0.001:
            # random Dirichlet perturbation
            perturb = np.random.dirichlet(np.ones(n))
            # combine with small weight
            cur_seq = 0.999 * cur_seq + 0.001 * perturb
            cur_seq /= cur_seq.sum()
            cur_val = evaluate_sequence(cur_seq.tolist())
        # adapt step size slowly
        # decrease step to fine-tune near end
        elapsed = time.time() - start
        remaining = time_limit - time.time()
        if remaining < time_limit * 0.1:
            step *= 0.99
    return best_seq.tolist(), best_val

"""Propose a low‑score step‑function.

The algorithm builds on the known near‑optimal shape
a_i ∝ C(2i,i)/4**i  (which behaves like 1/√i) and then
performs a fast hill‑climbing search that repeatedly moves a
tiny amount of mass from the indices that contribute most to the
current maximal convolution value to indices that contribute
least.  All moves keep the total sum unchanged, stay non‑negative
and are accepted only if they improve the evaluation score.
The search is time‑limited (default 1000 s) and returns the best
sequence found.
"""

import time
import numpy as np

def _analytic_sequence(n: int) -> np.ndarray:
    """Return the “1/√i’’ sequence a_i ∝ binom(2i,i)/4**i, normalized to sum=1."""
    # recurrence: b[0]=1, b[i] = b[i-1] * (2*i-1)/(2*i)
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for i in range(1, n):
        b[i] = b[i - 1] * (2 * i - 1) / (2 * i)
    b /= b.sum()
    return b


def _evaluate(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence (expects a list)."""
    return evaluate_sequence(seq.tolist())


def _hill_climb(start_seq: np.ndarray, deadline: float) -> tuple[list[float], float]:
    """
    Simple hill‑climbing optimizer.

    Parameters
    ----------
    start_seq : np.ndarray
        Initial non‑negative sequence (already normalised to sum=1).
    deadline : float
        Absolute time (seconds since epoch) after which the search must stop.

    Returns
    -------
    best_seq : list[float]
        The best sequence found (as a plain list).
    best_score : float
        Its evaluation score.
    """
    seq = start_seq.copy()
    n = len(seq)

    # ensure a valid distribution
    if seq.sum() == 0:
        seq[:] = 1.0 / n
    else:
        seq /= seq.sum()

    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    cur_seq = seq.copy()
    cur_score = best_score

    # typical magnitude of an entry ≈ 1/n
    step = 0.6 / n          # initial step size (absolute amount moved)
    rng = np.random.default_rng()

    while time.time() < deadline:
        # --- direction from the current maximal convolution ---
        conv = np.convolve(cur_seq, cur_seq)
        k_max = int(np.argmax(conv))

        # contribution of each index to the peak at k_max
        # only indices i with 0 ≤ k_max-i < n participate
        idx = np.arange(n)
        j = k_max - idx
        valid = (j >= 0) & (j < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = cur_seq[idx[valid]] * cur_seq[j[valid]]

        # pick a source index (large contribution) and a sink (small contribution)
        src = int(np.argmax(contrib))
        # choose among the lowest 30 % contributions
        low_thr = np.percentile(contrib, 30)
        low_candidates = np.where(contrib <= low_thr)[0]
        if low_candidates.size == 0:
            # fallback to a random different index
            tgt = src
            while tgt == src:
                tgt = rng.integers(n)
        else:
            tgt = int(rng.choice(low_candidates))
            # make sure src≠tgt
            if tgt == src:
                tgt = (src + 1) % n

        # maximum amount we may transfer without breaking non‑negativity
        max_delta = min(step, cur_seq[src])
        if max_delta <= 0:
            continue

        # random proportion of max_delta (helps escaping plateaux)
        delta = max_delta * rng.random()
        new_seq = cur_seq.copy()
        new_seq[src] -= delta
        new_seq[tgt] += delta

        # keep the sum exactly 1 (numerical drift may appear)
        new_seq /= new_seq.sum()

        # evaluate
        new_score = _evaluate(new_seq)

        # accept only improving moves
        if new_score < cur_score:
            cur_seq, cur_score = new_seq, new_score
            if new_score < best_score:
                best_seq, best_score = new_seq.copy(), new_score
                print(
                    f"    New best {best_score:.6f} (n={n})  "
                    f"time={time.time() - (deadline - (deadline - time.time())):.2f}s"
                )

        # occasional random perturbation to avoid stagnation
        if rng.random() < 0.001:
            pert = rng.dirichlet(np.ones(n))
            cur_seq = 0.998 * cur_seq + 0.002 * pert
            cur_seq /= cur_seq.sum()
            cur_score = _evaluate(cur_seq)

        # slowly shrink the step size as time runs out
        if deadline - time.time() < 0.2 * (deadline - (deadline - (deadline - time.time()))):
            step *= 0.995

    return best_seq.tolist(), best_score


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Generate a step‑function sequence with a low evaluation score.

    The routine:
        1. builds an analytic starting sequence (≈1/√i, optimal in theory);
        2. evaluates it;
        3. runs a short hill‑climbing search to improve the score;
        4. repeats the process for a few different lengths (including the
           length of the previously known construction, if available);
        5. returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # keep a safety margin

    # Determine which lengths to try.
    lengths = []
    try:
        # ``height_sequence_1`` may be present in the global namespace.
        prev = height_sequence_1
        if isinstance(prev, (list, tuple, np.ndarray)):
            lengths.append(len(prev))
    except NameError:
        pass
    # add a few hand‑picked sizes (small enough to stay fast, large enough to be expressive)
    lengths.extend([200, 500, 1124])

    best_seq = None
    best_score = float('inf')

    for n in lengths:
        if time.time() >= deadline:
            break

        # --- analytic start point -------------------------------------------------
        base_seq = _analytic_sequence(n)

        try:
            s = _evaluate(base_seq)
        except Exception:
            s = float('inf')
        if s < best_score:
            best_score, best_seq = s, base_seq.tolist()
            print(f"[len={n}] analytic start score = {s:.6f}")

        # allocate a slice of remaining time for local optimisation
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # at most 30 s per length, but never consume more than half of the remaining budget
        slice_time = min(30.0, remaining * 0.5)
        if slice_time < 1.0:   # not enough time left for a meaningful search
            continue

        # --- hill‑climbing -------------------------------------------------------
        seq, score = _hill_climb(base_seq, time.time() + slice_time)
        if score < best_score:
            best_score, best_seq = score, seq
            print(f"[len={n}] improved score = {score:.6f}")

    # Fallback: uniform distribution if everything failed
    if best_seq is None:
        n = 100
        best_seq = [1.0 / n] * n
        best_score = _evaluate(best_seq)

    print(f"Best score found: {best_score:.6f} (length {len(best_seq)})")
    return best_seq
