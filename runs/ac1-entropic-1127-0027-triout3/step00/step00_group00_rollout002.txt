import time
import numpy as np

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Gradient‑descent search on the simplex using a smoothed max‑convolution
    objective.  The algorithm keeps the sum of the heights equal to 1 (the
    scale‑invariance of the evaluation function) and iteratively reduces the
    peak of the self‑convolution by moving in the negative gradient of a
    soft‑max of the convolution entries.  Multiple random restarts with
    different lengths are performed until the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # leave a tiny margin

    # ------------------------------------------------------------------
    # Helpers
    # ------------------------------------------------------------------
    def next_pow2(x):
        """Return the smallest power of two >= x."""
        return 1 << (x - 1).bit_length()

    # ------------------------------------------------------------------
    # Baseline using the provided global sequence if available.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    try:
        # height_sequence_1 is injected by the evaluation harness.
        init_seq = list(height_sequence_1)
        # Force sum to 1 for fair comparison (evaluation normalises anyway).
        init_arr = np.array(init_seq, dtype=float)
        if init_arr.sum() == 0:
            init_arr = np.full_like(init_arr, 1.0 / len(init_arr))
        else:
            init_arr = np.maximum(init_arr, 0.0)
            init_arr /= init_arr.sum()
        init_score = 2.0 * len(init_arr) * np.max(np.convolve(init_arr, init_arr))
        best_seq = init_arr.copy()
        best_score = init_score
        print(f"Baseline score (height_sequence_1): {best_score:.6f}")
    except Exception:
        # No baseline available.
        pass

    # ------------------------------------------------------------------
    # Main search – repeated random restarts with different lengths.
    # ------------------------------------------------------------------
    # Lengths to try (they should be moderate to keep each iteration cheap).
    candidate_lengths = [500, 1000, 2000, 3000]

    while time.time() < deadline:
        # Randomly pick a length for this restart.
        n = int(np.random.choice(candidate_lengths))
        # ----- initialise a random point on the simplex -----
        a = np.full(n, 1.0 / n)                         # start from uniform
        a += 0.02 * np.random.rand(n)                  # tiny random perturbation
        a = np.maximum(a, 0.0)
        a /= a.sum()

        # Pre‑compute FFT sizes (they stay constant for this restart).
        size_conv = next_pow2(2 * n - 1)                # for a * a
        size_corr = next_pow2(3 * n - 2)                # for w ⋆ a

        # Hyper‑parameters – they slowly decay so the method stabilises.
        lr = 0.7                # learning rate
        beta = 4.0              # soft‑max sharpness
        decay_lr = 0.9999
        decay_beta = 1.00005

        # Number of inner steps – limited only by the remaining time.
        step = 0
        while time.time() < deadline:
            # ---------- compute self‑convolution (linear) ----------
            A_fft = np.fft.rfft(a, size_conv)
            b_full = np.fft.irfft(A_fft * A_fft, size_conv)[: 2 * n - 1]
            max_b = np.max(b_full)
            # Current objective (exact, because sum(a)=1)
            score = 2.0 * n * max_b

            # Keep the best ever seen.
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"New best: {best_score:.6f}  (n={n}, step={step}, lr={lr:.5f})")

            # ---------- build a smooth approximation of the max ----------
            # soft‑max weights over the convolution entries
            m = max_b
            exp_weights = np.exp(beta * (b_full - m))
            w = exp_weights / exp_weights.sum()

            # ---------- gradient of the soft‑max objective ----------
            # grad_i = 2 * Σ_j a_j * w_{i+j}   (linear correlation w ⋆ a)
            W_fft = np.fft.rfft(w, size_corr)
            A_fft2 = np.fft.rfft(a, size_corr)
            corr_full = np.fft.irfft(W_fft * np.conj(A_fft2), size_corr)
            # length of correlation = len(w)+len(a)-1 = 3n-2; we need first n values.
            grad = 2.0 * corr_full[:n]

            # ---------- take a projected gradient step ----------
            a -= lr * grad
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s <= 1e-12:                     # numerical safety net
                a = np.full(n, 1.0 / n)
                s = 1.0
            a /= s

            # ---------- schedule updates ----------
            lr *= decay_lr
            beta *= decay_beta
            step += 1

            # Occasionally add a tiny random jitter to avoid stagnation.
            if step % 250 == 0:
                a += 0.001 * np.random.randn(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()

            # Break the inner loop if we have performed a large number of steps.
            if step >= 5000:
                break

    # Fallback – should never happen, but keep the contract.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.array([1.0])

    return best_seq.tolist()

    # Convert to plain Python list as required by the harness.
    return best_seq.tolist()
