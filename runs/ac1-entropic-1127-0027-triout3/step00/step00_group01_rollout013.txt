def proj_simplex(v):
    # projection onto simplex sum=1, v_i >=0
    n = len(v)
    u = sorted(v, reverse=True)
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def simplex_projection(v):
    # Implementation of onto simplex {x : x_i >=0, sum x_i = 1}
    n = v.shape[0]
    # Sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # fallback
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

and prints occasional progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x | x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the largest rho where u_i - cssv_i / (i+1) > 0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast local evaluation (scale = 1, i.e. sum(a) = 1)
def _local_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a with sum = 1."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# One optimisation run (sub‑gradient descent)
def _optimize_from(
    a0: np.ndarray,
    deadline: float,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """Run projected sub‑gradient descent starting from a0 until deadline."""
    a = _project_simplex(a0.astype(float))
    n = a.shape[0]
    best_a = a.copy()
    best_score = _local_score(a)
    it = 0

    while time.time() < deadline - 0.05:
        conv = np.convolve(a, a)
        k = int(np.argmax(conv))
        # gradient of the maximal convolution entry
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        inds_i = np.arange(lo, hi + 1, dtype=int)
        inds_j = k - inds_i
        grad[inds_i] = 2.0 * a[inds_j]

        # diminishing step size
        eta = 0.5 / np.sqrt(it + 1.0)
        # small random jitter to escape the uniform fixed point
        noise = np.random.randn(n) * eta * 0.03
        a_candidate = a - eta * grad + noise
        a_candidate = _project_simplex(a_candidate)

        cand_score = _local_score(a_candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            if verbose:
                print(f"[iter {it:6d}] new best score = {best_score:.6f}")

        # accept improvement always, otherwise keep the current point
        if cand_score <= _local_score(a) or np.random.rand() < 0.01:
            a = a_candidate

        it += 1

    return best_a, best_score

def propose_candidate(budget_s: int = 1000, seed: int = 42, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The function respects the 1000 s time budget,
    prints progress and finally returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

        init = np.random.exponential(1.0, size=n)
        init /= init.sum()
        seq, val = _optimize_from(init, deadline, verbose=True)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
        print(f"[stage 1] best score so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Uniform with tiny perturbation
    if time.time() < deadline:
        init = np.full(n, 1.0 / n)
        init += np.random.randn(n) * 0.001
        init = np.maximum(init, 0.0)
        init /= init.sum()
        seq, val = _optimize_from(init, deadline, verbose=True)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
        print(f"[stage 2] best score so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Arcsine‑like shape (large mass at the ends, small in the centre)
    if time.time() < deadline:
        i = np.arange(n, dtype=float)
        eps = 0.5
        raw = 1.0 / np.sqrt((i + eps) * (n - i - 1 + eps))
        init = raw / raw.sum()
        seq, val = _optimize_from(init, deadline, verbose=True)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
        print(f"[stage 3] best score so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Final refinement starting from the overall best vector
    if best_seq is not None and time.time() < deadline:

        if val < best_val:

        best_seq = np.full(n, 1.0 / n)
