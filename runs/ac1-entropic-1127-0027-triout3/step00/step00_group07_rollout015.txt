def simulated_annealing(n, budget, start_seq=None):
    if start_seq is None:
        seq = np.random.exponential(1.0, size=n)
    else:
        seq = np.array(start_seq)
    seq = seq / np.sum(seq)  # normalize sum to 1

    best_seq = seq.copy()
    best_val = evaluate_sequence(seq.tolist())
    current_val = best_val

    start_time = time.time()
    deadline = start_time + budget
    iteration = 0

    while time.time() < deadline:
        iteration += 1
        # temperature schedule
        T = max(0.01, (deadline - time.time()) / (deadline - start_time))  # decays from 1 to 0.01

        # propose move
        i, j = np.random.choice(n, size=2, replace=False)
        # amount to move: choose fraction of seq[i]
        delta = seq[i] * np.random.rand() * 0.5  # up to half of i's mass
        # apply move
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        # evaluate
        val = evaluate_sequence(new_seq.tolist())
        # acceptance
        if val < current_val or np.random.rand() < np.exp((current_val - val) / max(T, 1e-8)):
            seq = new_seq
            current_val = val
            if val < best_val:
                best_val = val
                best_seq = new_seq.copy()
        # occasional random restart
        if iteration % 1000 == 0 and np.random.rand() < 0.1:
            seq = np.random.exponential(1.0, size=n)
            seq = seq / np.sum(seq)
            current_val = evaluate_sequence(seq.tolist())
    return best_seq, best_val

def hill_climb(seq, time_budget):
    n = len(seq)
    best_seq = seq.copy()
    best_score = evaluate_sequence(seq.tolist())
    start = time.time()
    step = 0.1
    while time.time() - start < time_budget:
        i, j = np.random.choice(n, 2, replace=False)
        delta = step * np.random.rand() * seq[i]
        if delta == 0:
            continue
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure non-negative
        if new_seq[i] < 0:
            continue
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            seq = new_seq
            step = max(step * 0.9, 1e-6)
        else:
            # maybe accept if improvement small
            step *= 0.99
        # occasional random restart
        if np.random.rand() < 0.001:
            seq = np.random.exponential(scale=1.0, size=n)
            seq = seq / np.sum(seq)
            # update best score if necessary

"""
Simple stochastic optimizer for the convolution‑maximisation problem.

The objective returned by ``evaluate_sequence`` (after normalising the sum to
sqrt(2·n)) is exactly the maximum entry of the discrete self‑convolution.
A uniform sequence gives a value of 2 for any length, but non‑uniform
sequences can achieve a smaller maximum.  The optimiser creates many random
shapes (exponential, power‑law, symmetric variants, pure random) of modest
length, normalises them to unit sum and then refines the best candidates by
a cheap hill‑climbing step that moves a small amount of mass from one index
to another.  The search runs until the allocated time budget is exhausted and
returns the best sequence found.

The code obeys the platform restrictions:
* no lambdas, no closures, all helpers are top‑level,
* only ``numpy`` and the standard library are used,
* the provided ``evaluate_sequence`` is called directly.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def shape_uniform(n: int) -> np.ndarray:
    """Uniform shape – all entries equal."""
    return np.ones(n)


def shape_exponential(n: int) -> np.ndarray:
    """a_i = exp(-lam * i) with a random rate."""
    lam = np.random.exponential(scale=0.2)  # typical rates 0‑0.5
    return np.exp(-lam * np.arange(n))


def shape_symmetric_exponential(n: int) -> np.ndarray:
    """U‑shaped exponential: large at the edges, small near the centre."""
    lam = np.random.exponential(scale=0.2)
    d = np.minimum(np.arange(n), np.arange(n)[::-1])
    return np.exp(-lam * d)


def shape_powerlaw(n: int) -> np.ndarray:
    """a_i = (i+1)^(-p) with a random exponent."""
    p = np.random.uniform(0.5, 5.0)
    return (np.arange(1, n + 1, dtype=float) ** (-p))


def shape_symmetric_powerlaw(n: int) -> np.ndarray:
    """U‑shaped power law: (distance to edge +1)^(-p)."""
    p = np.random.uniform(0.5, 5.0)
    d = np.minimum(np.arange(n), np.arange(n)[::-1]) + 1
    return (d.astype(float) ** (-p))


def shape_random(n: int) -> np.ndarray:
    """Completely random positive vector (exponential distribution)."""
    return np.random.exponential(scale=1.0, size=n)


def normalise_to_unity(v: np.ndarray) -> np.ndarray:
    """
    Normalise a non‑negative vector to have sum exactly 1.
    (The evaluation function is homogeneous, so any scale works,
    but fixing the sum simplifies the search.)
    """
    s = np.sum(v)
    if s == 0:
        # fallback to a tiny uniform vector
        v[:] = 1.0 / v.size
        return v
    return v / s


def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    # ``evaluate_sequence`` expects a plain list of Python floats.
    return evaluate_sequence(seq.tolist())


def hill_climb(start_seq: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing local optimiser.
    It repeatedly moves a small amount of mass from one index to another,
    accepting the move only when the objective improves.
    The step size shrinks gradually, and occasional random restarts are
    performed to escape shallow local minima.
    """
    n = start_seq.size
    seq = start_seq.copy()
    best_seq = seq.copy()
    best_score = evaluate(seq)
    # step controls the maximum fraction of mass moved in one operation
    step = 0.2
    start = time.time()
    while time.time() - start < time_budget:
        # choose two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # amount to transfer (a fraction of the source entry)
        delta = step * np.random.rand() * seq[i]
        if delta <= 0:
            continue
        # propose new sequence
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure non‑negativity (clip tiny negatives caused by rounding)
        if new_seq[i] < 0:
            continue
        new_score = evaluate(new_seq)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            seq = new_seq
            # when we improve, tighten the step a little
            step = max(step * 0.9, 1e-6)
        else:
            # slight decay of the step size, allowing finer moves later
            step *= 0.995
        # occasional random restart to diversify the search
        if np.random.rand() < 0.001:
            seq = normalise_to_unity(np.random.exponential(scale=1.0, size=n))
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the framework
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Stochastic search for a step‑function that minimises the evaluation function.
    The routine explores several short random shapes, refines them locally,
    and keeps the best result seen before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0  # small safety margin
    best_sequence = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1.  A trivial deterministic 2‑element improvement (known optimal for n=2)
    # ------------------------------------------------------------------
    seq2 = np.array([1.0 / 3.0, 2.0 / 3.0])
    seq2 = normalise_to_unity(seq2)          # sum = 1, scaling does not matter
    score2 = evaluate(seq2)
    if score2 < best_score:
        best_score = score2
        best_sequence = seq2.tolist()
        print(f"[Init] 2‑element sequence score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2.  Main random‑search loop
    # ------------------------------------------------------------------
    shape_functions = [
        shape_uniform,
        shape_exponential,
        shape_symmetric_exponential,
        shape_powerlaw,
        shape_symmetric_powerlaw,
        shape_random,
    ]

    # length sampling – short vectors are cheap to evaluate and often give
    # good improvements; we also try a few larger sizes.
    length_options = [8, 10, 12, 15, 20, 30, 40, 50, 60, 80, 100]

    while time.time() < deadline:
        # pick a random length and a random shape generator
        n = int(np.random.choice(length_options))
        shape_func = shape_functions[int(np.random.randint(len(shape_functions)))]

        # generate a raw shape and normalise it
        raw = shape_func(n)
        seq = normalise_to_unity(np.abs(raw))

        # a few local refinement steps (time‑budgeted per candidate)
        remaining = deadline - time.time()
        # use at most a few seconds per candidate, but never exceed the overall budget
        local_budget = min(2.0, max(0.1, remaining * 0.05))
        seq, score = hill_climb(seq, local_budget)

        if score < best_score:
            best_score = score
            best_sequence = seq.tolist()
            print(f"[Update] New best score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 3.  Fallback – should never be needed, but guarantees a return value
    # ------------------------------------------------------------------
    if best_sequence is None:
        best_sequence = [1.0 / 5] * 5
        print("[Fallback] Returning a trivial uniform sequence.")
    else:
        print(f"[Result] Final best score = {best_score:.6f}")

    return best_sequence
