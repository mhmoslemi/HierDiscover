def project_simplex(v):
    # Projects onto the probability simplex {x >=0, sum(x)=1}
    v = np.asarray(v)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a candidate step function sequence that minimises the
evaluation function:

    2 * n * max(convolve(a, a)) / (sum(a) ** 2)

The algorithm consists of three stages:

1. **Exploratory seeds** – generate a few simple families of
   sequences (uniform, geometric, U‑shaped Beta‑type) and keep the
   best one.

2. **Projected sub‑gradient descent** – starting from the best seed,
   repeatedly move the vector in the direction that reduces the
   currently maximal convolution coefficient.  The move is a
   sub‑gradient of the maximum‑convolution functional, after which the
   vector is projected back onto the probability simplex (non‑negative,
   unit sum).  A decreasing step size and occasional random noise help
   escaping shallow plateaus.

3. **Periodic full evaluation** – every few iterations the true
   `evaluate_sequence` function is called (it applies the required
   clipping).  The best sequence found so far is stored and finally
   returned.

All operations are O(n log n) per iteration thanks to FFT‑based
convolution, which easily fits into the 1000 s budget for n ≈ 5400.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def simplex_projection(v):
    """Project a vector onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=float)
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen, but fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def max_convolution(seq):
    """Return max_{k} (seq * seq)[k] using FFT for speed."""
    a = np.asarray(seq, dtype=float)
    n = a.size
    size = 2 * n - 1
    # real FFT of length `size`
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size)
    return float(conv.max())

def fast_score(seq):
    """Fast evaluation for sequences already normalised to sum == 1."""
    n = len(seq)
    m = max_convolution(seq)
    # denominator = 1 because sum(seq) == 1 (after projection)
    return 2.0 * n * m

# Candidate generators
# ------------------------------------------------------------

def uniform_seq(n):
    """Uniform (all‑ones) sequence of length n."""
    return np.ones(n, dtype=float)

def geometric_seq(n, r):
    """Geometric decay a_i = r**i, i = 0 … n‑1, then normalised to sum 1."""
    a = r ** np.arange(n, dtype=float)
    return simplex_projection(a)

def u_shaped_seq(n, p, delta=0.5):
    """U‑shaped Beta‑type: a_i ∝ ((i+δ)*(n-1-i+δ))^{-p}."""
    i = np.arange(n, dtype=float)
    denom = (i + delta) * (n - 1 - i + delta)
    # avoid divide by zero (should not happen because delta>0)
    a = denom ** (-p)
    return simplex_projection(a)

def random_seq(n):
    """Random positive vector normalised to unit sum."""
    a = np.random.rand(n)
    return simplex_projection(a)

    else:
        current = best_seq.copy()

    # Ensure we start on the simplex (sum == 1)
    current = simplex_projection(current)

    iteration = 0
    print("=== Sub‑gradient descent phase ===")
    while time.time() < deadline:
        a = np.asarray(current, dtype=float)
        # convolution via FFT
        size = 2 * n - 1
        A = np.fft.rfft(a, n=size)
        conv = np.fft.irfft(A * A, n=size)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active max‑convolution term
        # g_i = a_{k_max - i} when the index is valid, else 0
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(a)
        grad[idx] = a[k_max - idx]          # the vectorised gradient

        # step‑size schedule (decreasing)
        eta = 0.10 / np.sqrt(iteration + 1)

        # gradient step (moving opposite to the gradient)
        new_a = a - eta * grad

        # occasional random jitter to escape plateaus
        if iteration % 200 == 0:
            new_a += np.random.randn(n) * 0.001

        # keep non‑negative and re‑project onto the simplex
        new_a = simplex_projection(new_a)

        current = new_a.tolist()
        iteration += 1

        # periodic full evaluation (cheap fast score for early‑stop)
        if iteration % 10 == 0:
            # fast score (sum is 1 by construction)
            fast_sc = fast_score(current)
            # keep a running best based on the fast score as well
            if fast_sc < best_score:
                # compute the official score for safety before accepting
                official_sc = evaluate_sequence(current)
                if official_sc < best_score:
                    best_score = official_sc
                    best_seq = current.copy()
                    print(f"[iter {iteration}] improved (fast) score = {official_sc:.9f}")

        # occasional full evaluation for more reliability
        if iteration % 100 == 0:
            official_sc = evaluate_sequence(current)
            if official_sc < best_score:
                best_score = official_sc
                best_seq = current.copy()
                print(f"[iter {iteration}] improved (official) score = {official_sc:.9f}")

    # --------------------------------------------------------
    # 3 – final sanity check and return
    # --------------------------------------------------------
    # Ensure the sequence respects the clipping limits imposed by
    # evaluate_sequence (values >1000 are automatically clipped there,
    # but we keep them modest to avoid any surprises).
    final_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]
    final_score = evaluate_sequence(final_seq)
    print(f"Final reported score = {final_score:.9f}")
    return final_seq
