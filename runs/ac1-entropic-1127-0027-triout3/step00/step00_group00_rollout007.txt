"""
Propose a step‑function (non‑negative heights) that tries to minimise
evaluate_sequence.  The optimisation is performed by a smooth‑max
subgradient method on the convolution a*a.  At each iteration we
  1. compute the convolution b = a∗a,
  2. form a soft‑max weighting w_k ∝ exp(β·b_k) (β slowly grows,
     thus the algorithm starts with a smooth objective and ends with a
     true max‑objective),
  3. obtain the gradient of the weighted sum  Σ w_k·b_k w.r.t. a using a
     single convolution (cross‑correlation),
  4. apply an exponentiated‑gradient (mirror‑descent) update
     a_i←a_i·exp(−η·grad_i) and renormalise to the simplex.
We start from many random initial vectors (Dirichlet samples) and keep the
best sequence found within the time budget.  The algorithm works on the
same length as the previous best construction (≈3500) but can be
instantiated with any length ≤ the length of ``height_sequence_1``.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper: fast linear convolution using FFT (real valued)
# --------------------------------------------------------------
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return linear convolution of real vectors x and y using FFT."""
    n = len(x) + len(y) - 1
    # size as next power of two for speed
    L = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, n=L)
    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)[:n]
    return conv


# --------------------------------------------------------------
# Subgradient / mirror‑descent step for a given sequence a
# --------------------------------------------------------------
def _step(a: np.ndarray, beta: float, eta: float) -> np.ndarray:
    """
    Perform one iteration of the smooth‑max mirror‑descent.

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current probability vector (non‑negative, sums to 1).
    beta : float
        Soft‑max temperature. Larger -> approximation of true max.
    eta : float
        Step size for the exponentiated‑gradient update.

    Returns
    -------
    ndarray
        Updated vector, again a probability vector.
    """
    n = a.size

    # 1) convolution b = a * a
    b = _conv_fft(a, a)                     # length 2n-1

    # 2) soft‑max weights w_k ∝ exp(beta * b_k)
    #    (stable formulation: subtract the maximum)
    max_b = b.max()
    exp_arg = beta * (b - max_b)
    # clip to avoid overflow in extreme cases
    exp_arg = np.clip(exp_arg, -700, 700)
    w = np.exp(exp_arg)
    w /= w.sum()                            # shape (2n-1,)

    # 3) gradient of Σ w_k·b_k w.r.t. a.
    #    g_i = Σ_k w_k * ∂b_k/∂a_i .
    #    For unordered pair (i,j) (i≠j) the derivative contributes 2*a_j,
    #    for i=j the contribution is a_i.
    #    The summed expression can be written as
    #        g = 2 * (w ⋆ a)  –  diag(w_{2i}) * a_i,
    #    where ⋆ is the cross‑correlation.
    a_rev = a[::-1]                         # reversed version of a
    cross = _conv_fft(w, a_rev)             # length (2n-1)+n-1 = 3n-2
    # slice the part that corresponds to i = 0 .. n-1
    cross = cross[n - 1:n - 1 + n]           # shape (n,)

    g = 2.0 * cross

    # diagonal correction: when k = 2*i the derivative should be a_i,
    # not 2*a_i, therefore subtract w_{2i} * a_i once.
    diag_idx = 2 * np.arange(n)             # 0,2,4,...
    mask = diag_idx < w.size
    if mask.any():
        g[mask] -= w[diag_idx[mask]] * a[mask]

    # 4) exponentiated‑gradient (mirror descent) update
    #    a_i ← a_i * exp( -η * g_i )
    #    Clip the exponent to avoid under/overflow.
    update_factor = np.exp(-eta * np.clip(g, -50, 50))
    a = a * update_factor
    # renormalise to the simplex (sum = 1).  Zero vector is avoided
    # by resetting to a fresh random point.
    total = a.sum()
    if total <= 0.0 or not np.isfinite(total):
        a = np.random.rand(n)
        a /= a.sum()
    else:
        a /= total
    return a


# --------------------------------------------------------------
# Main entry point required by the framework
# --------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the outer runner reserves a few seconds,
        we stop slightly earlier).
    **kwargs : dict
        Ignored – kept for compatibility with the original interface.

    Returns
    -------
    list[float]
        The best sequence of heights discovered within the time limit.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # keep a safety margin

    # ------------------------------------------------------------------
    # Choose problem size.
    # ``height_sequence_1`` is a global created elsewhere; we reuse its
    # length (≈3500).  If it is not present we fall back to 2000.
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000

    # --------------------------------------------------------------
    # Parameters of the optimiser
    # --------------------------------------------------------------
    eta0 = 0.5          # initial step size for exponentiated gradient
    beta0 = 2.0         # start with a very smooth soft‑max
    beta_max = 120.0    # final temperature (almost exact max)

    # ------------------------------------------------------------------
    # Keep the best result found so far.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Main optimisation loop: we keep restarting with fresh random
    # vectors as long as we have time.  Each restart runs a fixed number
    # of inner iterations (or stops earlier if the global deadline is
    # reached).
    # ------------------------------------------------------------------
    restart = 0
    while time.time() < deadline:
        restart += 1
        # random start on the simplex (Dirichlet(1) == uniform random)
        a = np.random.rand(n)
        a /= a.sum()

        # inner iteration counters (dynamic, stop when time is out)
        it = 0
        while time.time() < deadline:
            # slowly increase beta towards beta_max (geometric schedule)
            beta = min(beta_max, beta0 * (1.001 ** it))

            # step size decays like 1/sqrt(iter) – standard for SGD
            eta = eta0 / np.sqrt(it + 1)

            a = _step(a, beta, eta)

            # evaluate only every few iterations (cheap evaluation)
            if it % 25 == 0:
                # evaluate_sequence is imported by the runner; we can call it directly
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[{time.time():.1f}] restart {restart}, "
                          f"it {it:5d}, new best {best_score:.6f}")

                # optional early exit if we already beat the target
                if best_score < 1.5:
                    # we have already surpassed the required bound – stop early
                    print("Target achieved, exiting.")
                    return best_seq.tolist()

            it += 1

            # safety break in case a single restart would run too long
            if it > 50000:      # generous upper bound
                break

    # ------------------------------------------------------------------
    # Return the best sequence discovered; if nothing was found (unlikely)
    # fall back to a uniform sequence of length n.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(n) / n
        best_score = evaluate_sequence(best_seq.tolist())
        print("Fallback to uniform sequence, score:", best_score)

    print("Finished. Best score:", best_score, "Length:", n)
    return best_seq.tolist()
