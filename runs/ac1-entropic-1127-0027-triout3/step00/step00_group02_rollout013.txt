def propose_candidate(seed=0, budget_s=1000, max_n=5000):
    """
    Search for a step function (list of non-negative heights) minimizing the evaluation.
    Uses a parametric power-law family a_i = (i+beta)^(-alpha) scaled to satisfy the
    prescribed L1 norm. A coarse grid is searched first, then a simple stochastic
    refinement (random walk on the α, β, n parameters) is performed until the time budget
    expires.
    Returns the best sequence found.
    """
    # imports
    import math, time, random, numpy as np

    # set random seed
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 1.0 # leave margin

    best_seq = None
    best_score = float('inf')
    best_params = None

    # define possible lengths (use odd/even maybe)
    lengths = [500, 1000, 1500, 2000, 3000, 4000]   # start with moderate lengths

    # coarse grid
    for n in lengths:
        # compute target sum sqrt(2*n)
        target_sum = math.sqrt(2.0 * n)
        alphas = np.linspace(0.2, 0.9, 15)
        # shift beta values: from 0.5 to 5 (ensuring >0)
        betas = np.linspace(0.5, 5.0, 10)
        for alpha in alphas:
            for beta in betas:
                # generate raw vector
                i = np.arange(n, dtype=np.float64)
                raw = (i + beta) ** (-alpha)
                raw_sum = raw.sum()
                if raw_sum == 0:
                    continue
                scale = target_sum / raw_sum
                seq = (raw * scale).tolist()
                # quickly evaluate
                score = evaluate_sequence(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq
                    best_params = (n, alpha, beta)
                    # print progress
                    print(f"New best: score={best_score:.6f}, n={n}, alpha={alpha:.4f}, beta={beta:.4f}")
    # Random refinement
    n, alpha, beta = best_params
    # Use simulated annealing style
    temperature = 1.0
    while time.time() < deadline:
        # propose new parameters by Gaussian perturbation
        n_new = int(max(100, min(8000, n + int(random.gauss(0, 100)))))
        # keep n even maybe
        if n_new % 2 != 0:
            n_new += 1
        # restrict n range
        n_new = max(100, min(8000, n_new))

        alpha_new = min(0.999, max(0.001, alpha + random.gauss(0, 0.02)))
        beta_new = max(0.01, beta + random.gauss(0, 0.2))

        target_sum = math.sqrt(2.0 * n_new)
        i = np.arange(n_new, dtype=np.float64)
        raw = (i + beta_new) ** (-alpha_new)
        raw_sum = raw.sum()
        if raw_sum == 0:
            continue
        scale = target_sum / raw_sum
        seq_candidate = (raw * scale).tolist()
        score_candidate = evaluate_sequence(seq_candidate)

        if score_candidate < best_score:
            best_score = score_candidate
            best_seq = seq_candidate
            n, alpha, beta = n_new, alpha_new, beta_new
            print(f"Improve via refinement: score={best_score:.6f} (n={n}, α={alpha:.4f}, β={beta:.4f})")
            temperature = 1.0  # reset temperature
        else:
            # optionally accept worse move with small probability (annealing)
            # probability = exp(-(score_candidate - best_score)/temperature)
            if random.random() < math.exp(-(score_candidate - best_score) / max(1e-9, temperature)):
                n, alpha, beta = n_new, alpha_new, beta_new
                temperature *= 0.99
        # slowly cool temperature
        temperature *= 0.9999

    return best_seq

"""
search for a low‑value step function.

The programme explores a parametric family
    a_i = (i + β)^(-α)   ,  i = 0,…,n‑1
scaled so that Σ a_i = √(2n).  For this family the
maximum of the auto‑convolution can be made much smaller
than the uniform construction (value 2).  A coarse grid
over (n,α,β) is evaluated, then a simple simulated‑annealing
loop refines the parameters.  The best sequence found is
returned as a plain list of floats (non‑negative as required).
"""

import math
import time
import random
import numpy as np

# ----------------------------------------------------------------------
def make_powerlaw_sequence(n: int, alpha: float, beta: float = 1.0):
    """
    Return a list of length n with entries (i + beta)**(-alpha),
    scaled to have L1‑norm = sqrt(2*n).  All entries are non‑negative.
    """
    # avoid pathological parameters
    if n <= 0 or alpha <= 0 or beta <= 0:
        return None
    i = np.arange(n, dtype=np.float64)
    raw = (i + beta) ** (-alpha)
    total = raw.sum()
    if total == 0.0:
        return None
    scale = math.sqrt(2.0 * n) / total
    seq = (raw * scale).tolist()
    return seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises evaluate_sequence().  The routine first checks whether a

    refines the parameters with a cheap simulated‑annealing loop.
    The function returns the best sequence discovered before the time
    budget expires.
    """
    random.seed(seed)
    np.random.seed(seed)

    # ------------------------------------------------------------------
    # time management
    deadline = time.time() + budget_s - 0.5   # 0.5 s safety margin

            raise NameError

        best_params = None

    # ------------------------------------------------------------------
