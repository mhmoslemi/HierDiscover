def propose_candidate(seed=42, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5

    n = 5300
    target_sum = math.sqrt(2*n)
    # start with random sequence
    f = np.random.rand(n).astype(float)
    f = f * (target_sum / f.sum())

    best_f = f.copy()
    best_score = evaluate_sequence(best_f.tolist())
    print("Initial score:", best_score)

    iteration = 0
    while time.time() < deadline and iteration < 200:
        iteration += 1
        # compute M
        conv_f = np.convolve(f, f)
        M = conv_f.max()
        # Solve LP to get g
        g = solve_convolution_lp(f.tolist(), float(M))
        if g is None:
            # random perturbation
            idx = np.random.randint(n)
            f[idx] = max(0.0, f[idx] + np.random.randn()*0.01)
            continue
        # Normalize g to target sum
        sum_g = np.sum(g)
        if sum_g == 0:
            continue
        g = g * (target_sum / sum_g)

        # try different mixing fractions
        improved = False
        for t in [0.5, 0.25, 0.75, 0.1, 0.9]:
            candidate = (1 - t) * f + t * g
            # ensure non-negative
            candidate = np.maximum(candidate, 0.0)
            # evaluate
            score = evaluate_sequence(candidate.tolist())
            if score < best_score - 1e-9:
                best_score = score
                best_f = candidate.copy()

    deadline = time.time() + budget_s - 5

    n = 5300

    best_score = float('inf')

    p = 0.5
    alpha = 0.0

        i = np.arange(n, dtype=float)

        return evaluate_sequence(a.tolist())

    # hill climbing with random perturbations
    while time.time() < deadline:
        # propose new parameters
        p_new = p + np.random.randn()*0.01  # small change
        alpha_new = alpha + np.random.randn()*5.0  # change in alpha
        # keep within bounds
        if p_new <= 0.1: p_new = 0.1
        if p_new >= 2.0: p_new = 2.0
        if alpha_new < 0.0: alpha_new = 0.0

        score = evaluate_params(p_new, alpha_new)
        if score < best_score:
            best_score = score
            best_params = (p_new, alpha_new)
            p, alpha = p_new, alpha_new
            print(f"New best p={p:.4f}, alpha={alpha:.2f} score={best_score:.6f}")
        else:
            # optionally accept with probability to escape local minima
            # adapt temperature
            pass

    # after search produce final sequence with best params
    p_opt, alpha_opt = best_params
    i = np.arange(n, dtype=float)
    best_seq = (1.0 / np.power(i + alpha_opt + 1.0, p_opt)).tolist()
    print(f"Final best params p={p_opt:.6f} alpha={alpha_opt:.6f} score={best_score:.6f}")

def gradient_descent_step(f, step):
    n = len(f)
    conv = np.convolve(f, f)
    k_max = np.argmax(conv)
    M = conv[k_max]
    S = np.sum(f)
    grad = np.empty_like(f)
    # compute partner index: j = k_max - i
    # For each i, if j within 0..n-1:
    partner = k_max - np.arange(n)
    valid = (partner >= 0) & (partner < n)
    # s_fj = np.where(valid, f[partner], 0.0)
    s_fj = np.zeros_like(f)
    s_fj[valid] = f[partner[valid]]
    dJ = (4 * n / S**3) * (S * s_fj - M)   # derivative
    # gradient descent direction: -dJ
    new_f = f - step * dJ
    new_f = np.clip(new_f, 0, None)
    # renormalize sum to original sum (scale)
    new_f = new_f * (S / np.sum(new_f))
    return new_f
```
