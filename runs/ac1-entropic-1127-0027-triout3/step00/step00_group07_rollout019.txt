    np.random.seed(seed)

        a = a / a.sum()

            # compute convolution
            b = np.convolve(a, a)
            # compute softmax weights
            max_b = b.max()
            exp_weights = np.exp(gamma * (b - max_b))
            w = exp_weights / exp_weights.sum()
            # compute term1 via convolution

            # compute t via sliding sum of w of length n

            # update
            a = a - lr * grad_E

"""
Propose a candidate step‑function sequence.

The evaluation objective is
    E(a) = 2 * n * max(a*a) / (sum(a)**2)

We look for a family of monotone power‑law sequences
    a_i = (i + shift) ** (-p)   (i = 0 … n‑1)

Because the objective is homogeneous, any global scaling does not matter,
so we keep the raw values (they are all ≤ 1 and avoid the 0…1000 clamp).

For each pair (p, shift) we evaluate the sequence, keep the best one,
and then perform a few random local perturbations around the best found
to try to improve it further.  The whole search runs inside the given
time budget and always returns a valid non‑negative list.
"""

import time
import numpy as np

def _evaluate(seq):
    """Wrapper for the provided evaluate_sequence (imported in the runner)."""
    # The runner guarantees evaluate_sequence is already in the globals.
    return evaluate_sequence(seq)


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search over simple power‑law families a_i = (i + shift)^{-p}
    and a few random perturbations to find a sequence with a low
    evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    **kwargs : dict
        Ignored (kept for compatibility with the harness).

    Returns
    -------
    list[float]
        The best sequence found within the time limit.
    """
    np.random.seed(seed)
    start_time = time.time()
    # length of the construction – can be any reasonable size.
    # 1124 is the length of the current best construction.
    n = 1124

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 1) Grid search over the (p, shift) family.
    # ------------------------------------------------------------------
    p_vals = np.linspace(0.30, 0.70, 9)          # 0.30, 0.35, … 0.70
    shift_vals = np.linspace(0, 20, 41)         # 0, 0.5, 1.0, … 20

    for p in p_vals:
        for shift in shift_vals:
            # early exit if we run out of time
            if time.time() - start_time > budget_s - 5:
                break

            # a_i = (i + shift)^{-p}  (i starts at 0)
            idx = np.arange(n, dtype=float)
            seq = 1.0 / np.power(idx + shift + 1.0, p)   # +1 avoids division by zero

            # Normalise to keep the sum comfortably away from zero.
            # Scaling does not affect the score, but it prevents the
            # “sum < 0.01” guard.
            if seq.sum() < 1.0:
                seq = seq * (1.0 / seq.sum())

            # Evaluate
            score = _evaluate(seq.tolist())
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[grid] new best: score={best_score:.6f}, p={p:.3f}, shift={shift:.2f}")

        # break outer loop if out of time
        if time.time() - start_time > budget_s - 5:
            break

    # ------------------------------------------------------------------
    # 2) Small stochastic refinements around the current best.
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple uniform sequence (never happens in practice)
        best_seq = np.ones(n, dtype=float)
        best_score = _evaluate(best_seq.tolist())

    # parameters for the random walk
    max_refine_iters = 2000
    lr = 0.02                     # step size (relative)
    for it in range(max_refine_iters):
        if time.time() - start_time > budget_s - 2:
            break

        # propose a random perturbation that preserves the sum
        cand = best_seq.copy()
        # pick two distinct indices and move a small amount of mass
        i, j = np.random.choice(n, size=2, replace=False)
        delta = lr * np.random.randn() * cand[i]   # proportional change
        # keep non‑negativity
        delta = np.clip(delta, -cand[i] + 1e-12, cand[j] - 1e-12)
        cand[i] -= delta
        cand[j] += delta

        # optional normalisation (keeps the scale stable)
        total = cand.sum()
        if total > 0:
            cand = cand * (best_seq.sum() / total)

        # evaluate the candidate
        score = _evaluate(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[refine] iter {it:04d} improved: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence as a plain Python list.
    # ------------------------------------------------------------------
    return best_seq.tolist()
