def propose_candidate(seed=..., budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1
    best_sequence = None
    best_score = np.inf

    while time.time() < deadline:
        # random n size
        n = np.random.randint(200, 2000)
        a = np.random.rand(n)
        a = a / np.sum(a)   # normalize
        # maybe initial random perturbation to avoid uniform
        # run gradient descent for some steps
        step = 0.1
        for it in range(200):
            b = np.convolve(a, a)
            max_b = b.max()
            k = b.argmax()
            grad = np.zeros_like(a)
            # compute gradient: 2 * a[k - i] where valid
            # Use vectorized: indices = k - np.arange(n)
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] = 2 * a[idx[mask]]
            # update
            a_new = a - step * grad
            # project onto simplex to keep non-negative and sum=1
            a = project_simplex(a_new)
            # optional step size decay
            step *= 0.99
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_sequence = a.tolist()
            print(f"New best: {best_score:.6f} n={n}")
    return best_sequence

"""
propose_candidate
~~~~~~~~~~~~~~~~~

Search for a non‑negative step‑function (a list of heights) that minimizes the
provided ``evaluate_sequence`` metric.

The search combines several cheap heuristics that are known to improve the
self‑convolution supremum:

1. **Random start** – a random positive vector (normalised to sum = 1) of a
   random length.

2. **Large‑small pairing** – the entries are reordered so that the largest
   values are placed opposite the smallest ones (indices i and n‑1‑i).  This
   reduces the dominant term ``a_i*a_{n‑1‑i}`` in the centre of the
   convolution.

3. **Projected sub‑gradient descent** – for the current sequence ``a`` the
   index ``k`` at which ``a*a`` attains its maximum is found.  The (sub‑)
   gradient of ``max(conv(a,a))`` at that point is
   ``g_i = 2*a_{k‑i}`` (zero when the index is out of range).  A step
   ``a ← a - η·g`` followed by a projection onto the probability simplex
   (non‑negative, sum = 1) strictly reduces the supremum unless the sequence
   is already optimal.

4. **LP direction (optional)** – a linear programme is solved to obtain a
   direction ``g`` that respects the current bound on the convolution and has
   maximal total mass.  Mixing a little of this direction into the current
   sequence can give a further small improvement.

The algorithm repeatedly creates candidates, improves them with the steps
above and keeps the best one seen before the time budget expires (the caller
provides ``budget_s`` seconds).  If no improvement is found the global
fallback ``height_sequence_1`` is returned.

All operations respect the security checks of ``evaluate_sequence`` – the
sequence is always normalised to have a non‑zero sum, entries are kept
non‑negative and are projected onto the simplex after each update.
"""

import time
import numpy as np
from scipy import optimize

linprog = optimize.linprog


def _project_onto_simplex(v):
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Uses the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(a, eta):
    """
    Perform a sub‑gradient step that reduces the maximal self‑convolution
    value.  ``a`` is assumed to sum to 1.  Returns the projected vector.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    k = int(b.argmax())
    # gradient: g_i = 2 * a_{k-i}  (if the index is valid)
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[idx[mask]]
    # take a step against the gradient and re‑project onto the simplex
    a_new = a - eta * g
    a_proj = _project_onto_simplex(a_new)
    return a_proj


def _pair_large_small(a):
    """
    Reorder the entries of ``a`` so that the largest values are placed opposite
    the smallest ones (indices i and n‑1‑i).  The vector is returned unchanged
    in magnitude but with a new ordering.
    """
    n = a.shape[0]
    order = np.argsort(-a)                # descending order
    sorted_vals = a[order]                # largest → smallest
    out = np.empty_like(a)
    left, right = 0, n - 1
    i = 0
    while left <= right:
        out[left] = sorted_vals[i]
        i += 1
        left += 1
        if left <= right:
            out[right] = sorted_vals[i]
            i += 1
            right -= 1
    return out


def _solve_lp_direction(f, rhs):
    """
    Solve the LP used in the original approach:

        maximise  Σ g_j
        s.t.      (f * g)(k) ≤ rhs    for all k
                  g_j ≥ 0

    Returns the optimal g (numpy array) or ``None`` if infeasible.
    """
    n = f.shape[0]
    c = -np.ones(n)                     # maximise sum -> minimise -sum
    # inequality matrix A_ub @ g ≤ b_ub
    A_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # (f * g)(k) = Σ_i f_i * g_{k-i}
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                row[j] = f[i]
        A_ub.append(row)
        b_ub.append(rhs)
    # non‑negativity constraints  -g ≤ 0   (i.e. g ≥ 0)
    A_ub.extend(-np.eye(n))
    b_ub.extend([0.0] * n)

    A_ub = np.vstack(A_ub)
    b_ub = np.array(b_ub)

    # No equality constraints; bound variables to [0, ∞)
    bounds = [(0, None)] * n

    # use HiGHS (fast interior‑point) with a modest time limit
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method='highs',
        options={"time_limit": 8.0, "presolve": True, "dual_feasibility_tolerance": 1e-9}
    )
    if res.success:
        g = np.maximum(res.x, 0.0)      # numerical safety
        # normalise to sum = 1 (scale‑invariant)
        if g.sum() > 0:
            g = g / g.sum()
        return g
    return None


def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The function runs for at most ``budget_s`` seconds and returns the best
    sequence found.  It never raises – if something goes wrong the global
    fallback ``height_sequence_1`` (if available) is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0, budget_s - 2)   # keep a tiny safety margin

    # start with the known construction as a fall‑back
    try:
        best_seq = list(height_sequence_1)
    except Exception:
        best_seq = None
    best_score = np.inf if best_seq is None else float(evaluate_sequence(best_seq))

    # helper to update the global best
    def _try_update(seq):
        nonlocal best_seq, best_score
        try:
            sc = evaluate_sequence(seq.tolist())
        except Exception:
            return
        if sc < best_score:
            best_score = sc
            best_seq = seq.tolist()
            print(f"[{time.time():.2f}]  New best = {sc:.6f}  length = {len(seq)}")
    
    # --------------------------------------------------------------------- #
    # Main stochastic search loop
    # --------------------------------------------------------------------- #
    while time.time() < deadline:
        # 1. pick a random length (moderate size – large enough to give freedom,
        #    small enough for fast LP solves)
        n = np.random.randint(300, 1200)         # can be tuned
        # 2. initialise with random positive numbers and normalise
        a = np.random.rand(n)
        a = a / a.sum()

        # 3. improve the ordering – pair large with small
        a = _pair_large_small(a)

        # 4. a few sub‑gradient steps (fixed step size, simple schedule)
        eta = 0.05
        for _ in range(30):
            a = _gradient_step(a, eta)
            eta *= 0.97          # slowly decay the step
            # occasional tiny random jitter to avoid stagnation
            if np.random.rand() < 0.07:
                i, j = np.random.randint(n, size=2)
                delta = 0.001 * a[i]
                a[i] = max(0.0, a[i] - delta)
                a[j] = a[j] + delta
                a = _project_onto_simplex(a)

        # 5. LP direction – try to find a better neighbour while keeping the
        #    current maximal convolution unchanged.
        conv_self = np.convolve(a, a)
        M = float(conv_self.max())
        g = _solve_lp_direction(a, M)
        if g is not None:
            # mix a little of the LP direction (small learning rate)
            tau = 0.03
            a = (1.0 - tau) * a + tau * g

        # 6. final simple refinement with a few more gradient steps
        eta = 0.02
        for _ in range(15):
            a = _gradient_step(a, eta)
            eta *= 0.95

        # 7. evaluate and possibly keep the result
        _try_update(a)

        # --------------------------------------------------------------- #
        # Occasionally restart from the global best (to explore locally)
        if best_seq is not None and np.random.rand() < 0.15:
            a = np.array(best_seq, copy=True)
            # normalise again (security check strips scaling)
            a = a / a.sum()
        # --------------------------------------------------------------- #
    
    # If nothing succeeded, fall back to the provided construction
    if best_seq is None:
        try:
            best_seq = list(height_sequence_1)
        except Exception:
            best_seq = [1.0]          # a trivial fallback

    print(f"Finished search – best score = {best_score:.6f}")
    return best_seq
