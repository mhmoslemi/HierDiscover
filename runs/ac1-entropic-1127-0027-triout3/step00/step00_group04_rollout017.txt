    # start with uniform sequence length 1000 maybe
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 500
    # to avoid large compute maybe set n=500, but we can use existing length.
    # Use same length as height_sequence_1 for fairness.
    best_seq = list(height_sequence_1) if 'height_sequence_1' in globals() else [1.0]*n
    best_score = evaluate_sequence(best_seq)
    print(f"Initial score: {best_score}")

    # main loop
    cur_seq = best_seq.copy()
    cur_score = best_score
    while time.time() < deadline:
        # choose perturbation type
        if random.random() < 0.7:
            # transfer epsilon from i to j
            i = random.randrange(len(cur_seq))
            j = random.randrange(len(cur_seq))
            if i == j:

            if eps <= 0:

            new_seq[j] += eps

            # random shuffle of a subsegment
            new_seq = cur_seq.copy()
            l = random.randrange(len(new_seq))
            r = random.randrange(l, len(new_seq))
            segment = new_seq[l:r]
            random.shuffle(segment)
            new_seq[l:r] = segment

        # optionally apply scaling to keep numbers in range, but evaluate handles.
        try:
            new_score = evaluate_sequence(new_seq)
        except Exception:
            continue

        # accept if better or with probability
        if new_score < cur_score:
            cur_seq, cur_score = new_seq, new_score
            if new_score < best_score:
                best_seq, best_score = new_seq, new_score
                print(f"New best {best_score:.6f} at time {time.time():.2f}")
        else:
            # simulated annealing acceptance
            T = max(0.01, (deadline - time.time()) / budget_s)  # decreasing temperature

            length = random.randint(200, 2000)

"""
Propose a candidate step‑function sequence.

The function performs a simple stochastic local search (simulated annealing)
starting from a good known construction (the global ``height_sequence_1`` if
available) and tries to reduce the objective
    C = 2 * n * max(conv(a, a)) / (sum(a)**2)
as returned by ``evaluate_sequence``.
The search consists of small mass‑transfer moves, random shuffles of
sub‑segments and occasional random restarts.  Accepted moves are those
that improve the score, otherwise they are accepted with a temperature
dependent probability to escape local minima.  The loop runs until the
budget (default 1000 s) is exhausted and the best sequence found is
returned.
"""

import random
import time
import math
import numpy as np

def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function (list of non‑negative floats) that yields a
    low value of ``evaluate_sequence``.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds. The function returns before the budget is
        exhausted (a few seconds are left for the caller).
    kwargs : dict
        Ignored – kept for compatibility with the surrounding framework.

    Returns
    -------
    list[float]
        The best sequence encountered during the search.
    """
    # ------------------------------------------------------------------
    # initialise RNGs
    # ------------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed + 1)

    # ------------------------------------------------------------------
    # time management
    # ------------------------------------------------------------------
    # keep a small safety margin so that the caller does not kill us
    deadline = time.time() + budget_s - 2.0

    # ------------------------------------------------------------------
    # start from a known good construction if it exists,
    # otherwise fall back to a uniform vector of moderate size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        cur_seq = list(height_sequence_1)          # copy, may be large
    else:
        # a modest default: 400 entries, each 1.0  (score = 2)
        cur_seq = [1.0] * 400

    # normalise only for numerical stability – the objective is scale
    # invariant, but keeping the sum around 1 avoids extreme numbers.
    s = sum(cur_seq)
    if s == 0:
        cur_seq = [1.0] * len(cur_seq)
        s = len(cur_seq)
    cur_seq = [x / s for x in cur_seq]

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    try:
        best_score = evaluate_sequence(cur_seq)
    except Exception:
        # should never happen, but fall back to a safe value
        best_score = float('inf')
    best_seq = cur_seq.copy()
    print(f"[init] length={len(cur_seq)} score={best_score:.6f}")

    # ------------------------------------------------------------------
    # main stochastic search loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        n = len(cur_seq)

        # --------------------------------------------------------------
        # 1) small mass‑transfer move: take ε from i and give it to j
        # --------------------------------------------------------------
        if random.random() < 0.70:
            i = random.randrange(n)
            j = random.randrange(n)
            if i == j:
                continue
            # take at most 10 % of the source entry
            eps = random.random() * cur_seq[i] * 0.10
            if eps <= 0:
                continue
            new_seq = cur_seq.copy()
            new_seq[i] -= eps
            new_seq[j] += eps

        # --------------------------------------------------------------
        # 2) random shuffle of a short sub‑segment (reordering can affect
        #    the convolutions because the indices change)
        # --------------------------------------------------------------
        else:
            new_seq = cur_seq.copy()
            # pick a random segment of length at most 0.2 · n
            seg_len = max(2, int(random.random() * 0.2 * n))
            start = random.randrange(0, n - seg_len + 1)
            segment = new_seq[start:start + seg_len]
            random.shuffle(segment)
            new_seq[start:start + seg_len] = segment

        # --------------------------------------------------------------
        # evaluate the candidate (the evaluate_sequence routine contains all
        # the required clipping and safety checks)
        # --------------------------------------------------------------
        try:
            cand_score = evaluate_sequence(new_seq)
        except Exception:
            # any numerical mishap – simply ignore this move
            continue

        # --------------------------------------------------------------
        # acceptance rule
        # --------------------------------------------------------------
        improve = cand_score < best_score
        if improve or cand_score < float('inf'):
            # temperature decays linearly to a small floor
            remaining = max(0.0, deadline - time.time())
            T = max(0.001, remaining / budget_s)   # in [~0,1]
            # classic simulated‑annealing acceptance
            accept = cand_score < best_score or \
                     (random.random() < math.exp(-(cand_score - best_score) / T))
        else:
            accept = False

        if accept:
            cur_seq = new_seq
            if cand_score < best_score:
                best_score = cand_score
                best_seq = new_seq.copy()
                print(f"[{iteration:07d}] new best = {best_score:.6f} (len={len(best_seq)})")

        # --------------------------------------------------------------
        # occasional random restart to diversify the search
        # --------------------------------------------------------------
        if random.random() < 0.001:
            # pick a fresh length (between 200 and 2000) and random payload
            new_len = random.randint(200, 2000)
            cur_seq = [random.random() for _ in range(new_len)]
            # renormalise to keep the sum close to 1
            total = sum(cur_seq)
            cur_seq = [x / total for x in cur_seq]
            # evaluate the fresh point (ignore if it fails)
            try:
                cur_score = evaluate_sequence(cur_seq)
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = cur_seq.copy()
                    print(f"[{iteration:07d}] restart gave new best = {best_score:.6f}")
            except Exception:
                pass

        # --------------------------------------------------------------
        # occasionally enforce a tiny amount of smoothness – this is a cheap
        # heuristic that keeps the sequence from becoming too spiky.
        # --------------------------------------------------------------
        if iteration % 5000 == 0:
            arr = np.array(cur_seq)
            # simple moving‑average filter (window 3)
            arr = np.convolve(arr, np.ones(3) / 3.0, mode='same')
            cur_seq = arr.tolist()
            # re‑normalise to preserve the scale used by the objective
            tot = sum(cur_seq)
            cur_seq = [x / tot for x in cur_seq]

    # ------------------------------------------------------------------
    # final report
    # ------------------------------------------------------------------
    print(f"[final] best score = {best_score:.6f} with length {len(best_seq)}")
    return best_seq
