def propose_candidate(seed=42, budget_s=1000):

    deadline = time.time() + budget_s - 5

    best_seq = None
    best_score = np.inf

    n = 4000  # or can try variable
    for attempt in range(5):
        # random start: maybe uniform + small noise
        a = np.random.rand(n)
        a = a / a.sum()
        # optionally start from height_sequence_1
        if attempt == 0 and 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=float)
            # pad or truncate to n
            if len(cand) < n:
                cand = np.pad(cand, (0, n-len(cand)), constant_values=0)
            else:
                cand = cand[:n]
            # renorm
            cand = cand / cand.sum()
            a = cand

        # gradient descent loop
        p = 2
        lr = 0.1
        while time.time() < deadline:
            # compute conv via FFT
            N = 1 << ((2*n-1).bit_length())
            # compute fft of a
            fft_a = np.fft.rfft(a, n=N)  # using rfft for real
            conv = np.fft.irfft(fft_a * fft_a, n=N)[:2*n-1]
            # compute weight w = conv**(p-1)
            w = np.power(conv, p-1, where=conv>0, out=np.zeros_like(conv))
            # compute gradient
            a_rev = a[::-1]
            fft_rev = np.fft.rfft(a_rev, n=N)
            fft_w = np.fft.rfft(w, n=N)
            corr = np.fft.irfft(fft_rev * fft_w, n=N)  # length N
            # slice
            start = n-1
            grad = 2 * p * corr[start:start+n]
            # gradient step
            a = a - lr * grad
            # project
            a = np.maximum(a, 0)
            sum_a = a.sum()
            if sum_a == 0:
                a = np.random.rand(n)
                a /= a.sum()
            else:
                a /= sum_a

            # optional adjust lr, p schedule
            if p < 10:
                p += 0.001   # slowly increase
                # adapt lr downward
                lr *= 0.9999

            # evaluate occasionally
            if int(time.time()*1000) % 10 == 0:  # but we can evaluate every N iterations
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"New best {score:.6f} at time {time.time():.2f}")

            # stop condition maybe if improvement small.
        # after each attempt compute final evaluation and possibly update best

import numpy as np, time, math

def proj_nonneg_normalize(v):
    v = np.maximum(v, 0)
    s = v.sum()
    if s == 0:
        v = np.ones_like(v)
        s = v.sum()
    v /= s
    return v

def next_pow2(x):
    return 1 << (x-1).bit_length()

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    best_seq = None
    best_score = float('inf')
    # We'll try a few different lengths
    for n in [2000, 3000, 4000]:
        # initial vector: uniform plus small noise
        a = np.random.rand(n)
        a = proj_nonneg_normalize(a)
        # optionally also try start from height_sequence_1 if length matches or larger
        if 'height_sequence_1' in globals():
            h = np.array(height_sequence_1, dtype=float)
            # pad or truncate
            if len(h) >= n:
                a = proj_nonneg_normalize(h[:n])
            else:
                pad = np.zeros(n - len(h))
                a = proj_nonneg_normalize(np.concatenate([h, pad]))
        # algorithm parameters
        p = 2.0
        lr = 0.5
        # compute N once
        N = next_pow2(3*n - 1)
        # precompute convolution length indices
        slice_start = n-1
        # compute initial conv and objective
        fft_a = np.fft.rfft(a, N)
        conv = np.fft.irfft(fft_a * fft_a, N)[:2*n-1]
        obj = np.sum(conv**p)
        # store best
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"Initial best score {best_score:.6f} with n={n}")
        iter_count = 0
        while time.time() < deadline:
            # weight
            w = p * np.power(conv, p-1, where=conv>0, out=np.zeros_like(conv))
            # gradient via correlation
            fft_w = np.fft.rfft(w, N)
            # correlation (linear) = ifft(conj(fft_a) * fft_w)
            corr = np.fft.irfft(np.conj(fft_a) * fft_w, N)
            grad = 2.0 * corr[slice_start:slice_start+n]
            # gradient step
            a_new = a - lr * grad
            a_new = proj_nonneg_normalize(a_new)
            # compute new conv and obj
            fft_a_new = np.fft.rfft(a_new, N)
            conv_new = np.fft.irfft(fft_a_new * fft_a_new, N)[:2*n-1]
            obj_new = np.sum(conv_new**p)
            # Acceptance test
            if obj_new <= obj:
                # accept
                a, conv, obj, fft_a = a_new, conv_new, obj_new, fft_a_new
                # maybe increase learning rate slightly
                lr = min(lr * 1.02, 1.0)
                # gradually increase p
                if p < 8.0:
                    p += 0.01
                # occasionally evaluate real score
                if iter_count % 200 == 0:
                    score = evaluate_sequence(a.tolist())
                    if score < best_score:
                        best_score = score
                        best_seq = a.copy()
                        print(f"New best score {best_score:.6f} at n={n} iter={iter_count}")
            else:
                # reject, reduce LR
                lr *= 0.5
                # also maybe reset a to previous best in this run?
                # continue with same a
                # maybe also increase p less aggressively
                # no change to a
            iter_count += 1
            # safeguard: if lr becomes too small, break
            if lr < 1e-6:
                # reinitialize a random to escape local minima
                a = proj_nonneg_normalize(np.random.rand(n))
                fft_a = np.fft.rfft(a, N)
                conv = np.fft.irfft(fft_a * fft_a, N)[:2*n-1]
                p = 2.0
                lr = 0.5
                obj = np.sum(conv**p)
        # after time spent on this n loop, check if time left maybe break
        if time.time() >= deadline:
            break
    print(f"Best score found: {best_score}")
    return best_seq.tolist()

def simplex_projection(v):
    # Project onto the simplex {x >=0 , sum=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Optimiser for the step‑function bound.

The task is to find a non‑negative vector `a` (the “heights”) that minimises

    2 * n * max(conv(a, a)) / (sum(a)**2)

where `conv` is the linear convolution.  The value is scale invariant, so the
vector can be normalised to have unit sum – we therefore keep `sum(a) == 1`
throughout the search.

The algorithm is a simple projected gradient descent on a smooth
approximation of the `max` operator (a soft‑max).  At each iteration we:

1. Compute the convolution with an FFT (O(N log N) time, N≈3·len(a)).
2. Build soft‑max weights `w` that concentrate on the largest convolution
   entries.
3. Compute the gradient of the smoothed objective via a correlation
   (again using FFTs).
4. Take a step, project back onto the probability simplex
   (non‑negative entries that sum to one) and repeat.

A “temperature” parameter `alpha` starts small (so the soft‑max is smooth) and
is slowly increased – the optimiser first gets a coarse view of the landscape
and later focuses on the true maximum.  Occasionally we evaluate the exact
score with the provided `evaluate_sequence` and keep the best vector seen.

The search is bounded by the supplied time budget (≈ 1000 s).  Several
restarts with different lengths (`n`) are tried; the best result over all
restarts is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex {x>=0, sum(x)=1}
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    # Shift by the smallest value to avoid negative issues (optional)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]  # the last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: int = 1000, **kwargs):
    """
    Search for a height sequence that beats the current bound 2.0.
    The function runs for at most `budget_s` seconds and returns a list of
    floats (the best sequence discovered).  No external I/O is performed.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0          # a safety margin

    best_seq = None
    best_score = float("inf")

    # try a few different lengths – larger n gives a tighter theoretical
    # lower bound but also costs more per iteration
    lengths = [1500, 3000, 5300]                     # 5300 is the reference size

    for n in lengths:
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # initialise: either from the known construction or a random start
        if "height_sequence_1" in globals():
            hs = np.array(height_sequence_1, dtype=float)
            if len(hs) >= n:
                a = hs[:n].copy()
            else:
                a = np.concatenate([hs, np.zeros(n - len(hs))])
        else:
            a = np.random.rand(n)

        a = _proj_simplex(a)                         # ensure sum == 1

        # FFT size – must accommodate convolution (length 2n‑1) and the
        # correlation (length up to 3n‑2).  Choose a power of two for speed.
        N = _next_pow2(3 * n - 1)

        # optimisation hyper‑parameters
        alpha = 2.0                                  # start soft‑max temperature
        lr = 0.5                                      # learning rate
        next_eval = time.time() + 5.0                # periodic exact evaluation

        # pre‑allocate buffers (avoid repeated allocations)
        fft_a = np.fft.rfft(a, N)

        while time.time() < deadline:
            # ---- convolution -------------------------------------------------
            conv_full = np.fft.irfft(fft_a * fft_a, N)
            conv = conv_full[: 2 * n - 1]            # linear convolution
            # ---- soft‑max weights (smooth approximation of the max) ---------
            scaled = alpha * conv
            max_scaled = scaled.max()
            exp_scaled = np.exp(scaled - max_scaled)  # stability trick
            w = exp_scaled / exp_scaled.sum()        # probability weights
            # ---- gradient (correlation of a with w) ------------------------
            fft_w = np.fft.rfft(w, N)
            corr = np.fft.irfft(np.conj(fft_a) * fft_w, N)
            grad = 2.0 * corr[:n]                    # d/d a of Σ w_i * conv_i

            # ---- gradient step + simplex projection -------------------------
            a_new = a - lr * grad
            a_new = _proj_simplex(a_new)

            # ---- recompute convolution for the candidate --------------------
            fft_a_new = np.fft.rfft(a_new, N)
            conv_new_full = np.fft.irfft(fft_a_new * fft_a_new, N)
            conv_new = conv_new_full[: 2 * n - 1]

            # ---- simple acceptance test (reduce the smoothed objective) ---
            # we compare the soft‑max objective, not the raw max
            obj_cur = (max_scaled + np.log(exp_scaled.sum())) / alpha
            scaled_new = alpha * conv_new
            max_scaled_new = scaled_new.max()
            exp_scaled_new = np.exp(scaled_new - max_scaled_new)
            obj_new = (max_scaled_new + np.log(exp_scaled_new.sum())) / alpha

            if obj_new <= obj_cur:
                # accept the move
                a, fft_a = a_new, fft_a_new
                conv = conv_new
                # slowly tighten the soft‑max and increase the step size
                if alpha < 30.0:
                    alpha *= 1.006
                lr = min(lr * 1.005, 2.0)
            else:
                # reject – reduce the step size
                lr *= 0.5
                if lr < 1e-5:                     # re‑initialise if we get stuck
                    a = _proj_simplex(np.random.rand(n))
                    fft_a = np.fft.rfft(a, N)
                    alpha = 2.0
                    lr = 0.5

            # ---- exact evaluation from time to time -------------------------
            if time.time() >= next_eval:
                # exact score (the objective we ultimately minimise)
                score = evaluate_sequence(a.tolist())
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(
                        f"[n={n}] New best score {best_score:.6f}  "
                        f"time {time.time():.1f}"
                    )
                next_eval = time.time() + 5.0

        # final check for this length
        final_score = evaluate_sequence(a.tolist())
        if final_score < best_score:
            best_score = final_score
            best_seq = a.copy()
            print(f"[n={n}] End‑run best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Ensure the returned values respect the clipping limits of evaluate_sequence
    if best_seq is None:
        # fallback – uniform vector (guaranteed to be valid)
        n = lengths[0]
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"Fallback uniform sequence, score {best_score:.6f}")

    # Convert to plain Python list as required by the evaluator
    return best_seq.tolist()
