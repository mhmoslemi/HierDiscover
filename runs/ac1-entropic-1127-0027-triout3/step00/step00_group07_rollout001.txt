def next_pow2(x):
    return 1 << (x - 1).bit_length()

def full_convolution(a):
    n = len(a)
    size = 2 * n - 1
    m = next_pow2(size)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    return conv

def gradient_update(a, target, eta):
    n = len(a)
    # compute full b
    b = full_convolution(a)
    diff = b - target
    diff = np.where(diff > 0, diff, 0.0)
    # compute gradient via convolution of reversed a and diff
    a_rev = a[::-1]
    size = 2 * n - 1
    conv_len = n + size - 1  # 3n - 2
    m2 = next_pow2(conv_len)
    fr = np.fft.rfft(a_rev, n=m2)
    fd = np.fft.rfft(diff, n=m2)
    conv_full = np.fft.irfft(fr * fd, n=m2)[:conv_len]
    grad = 4.0 * conv_full[n-1:n-1+n]
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)
    # renormalize
    sum_a = np.sum(a_new)
    if sum_a > 0:
        a_new = a_new * np.sqrt(2 * n) / sum_a
    else:
        a_new = np.full(n, np.sqrt(2 * n) / n)
    return a_new, b

"""
Projected gradient descent on the autocorrelation peak.

The objective of evaluate_sequence for a sequence a of length n with
∑a = √(2n) is simply max(conv(a,a)).  The algorithm tries to
flatten the convolution by iteratively moving mass from positions that
contribute to large convolution entries to other positions.
Each iteration:

1. Compute the full convolution b = a * a (FFT).
2. Set a target value t = 2n/(2n‑1) (the theoretical lower bound of the
   maximum convolution when the sum is fixed).
3. For every index where b_k > t we keep the excess diff_k = b_k‑t,
   otherwise diff_k = 0.
4. The gradient of the loss ½∑max(0,b_k‑t)² w.r.t. a is
   g = 4 * (a_rev ⊛ diff)_{center},
   where ⊛ is a linear convolution and a_rev is a reversed.
5. Perform a projected gradient step a ← a – η g,
   clip to non‑negative, and renormalise to keep the sum exactly
   √(2n).

The step size η is adapted: it is increased when a step improves the
max‑convolution and decreased otherwise.  Random jitter is added
occasionally to avoid stagnation.  The best sequence found before the
time budget expires is returned.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Return the aperiodic convolution a * a (full length 2*len(a)-1)
    using an FFT of size a power of two.
    """
    n = a.shape[0]
    size = 2 * n - 1
    m = _next_pow2(size)
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)[:size]
    return conv


def _gradient_step(a: np.ndarray, target: float, eta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Perform one projected gradient update.

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current non‑negative sequence (already normalised to the required sum).
    target : float
        Desired upper bound on the convolution entries.
    eta : float
        Step size.

    Returns
    -------
    a_new : ndarray, shape (n,)
        Updated sequence after the gradient step and projection.
    b_new : ndarray, shape (2n‑1,)
        Full convolution of a_new with itself.
    """
    n = a.shape[0]

    # 1) current convolution
    b = _full_convolution(a)

    # 2) excess over the target (only where b > target)
    diff = b - target
    diff = np.where(diff > 0, diff, 0.0)

    # 3) gradient = 4 * (a_rev ⊛ diff) centred on length n
    a_rev = a[::-1]
    conv_len = n + (2 * n - 1) - 1           # = 3n‑2
    m2 = _next_pow2(conv_len)
    fr = np.fft.rfft(a_rev, n=m2)
    fd = np.fft.rfft(diff, n=m2)
    full = np.fft.irfft(fr * fd, n=m2)[:conv_len]
    grad = 4.0 * full[n - 1 : n - 1 + n]    # slice the centred part

    # 4) gradient descent step, projection to non‑negative
    a_new = a - eta * grad
    a_new = np.maximum(a_new, 0.0)

    # 5) renormalise to keep the prescribed sum = sqrt(2n)
    s_target = np.sqrt(2.0 * n)
    total = a_new.sum()
    if total > 0.0:
        a_new = a_new * s_target / total
    else:
        # degenerate case – fall back to uniform distribution
        a_new = np.full(n, s_target / n)

    # 6) compute the new convolution (needed for the acceptance test)
    b_new = _full_convolution(a_new)

    return a_new, b_new


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) with a low
    evaluate_sequence value.

    The routine runs for at most ``budget_s`` seconds and returns the
    best sequence found.  No external files or network access are used.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # leave a margin for the final return

    # ----------------------------------------------------------------------
    # Problem size – we use a few thousand entries; larger n improves the
    # theoretical lower bound but also increases work per iteration.
    # ----------------------------------------------------------------------
    n = 1500                     # you may increase this value if time permits
    s_target = np.sqrt(2.0 * n)  # required sum of the sequence

    # ----------------------------------------------------------------------
    # Initialisation – start from the uniform distribution with a tiny
    # random jitter (helps to escape the exact symmetry of the uniform case).
    # ----------------------------------------------------------------------
    cur = np.full(n, s_target / n, dtype=np.float64)
    cur += (np.random.rand(n) - 0.5) * 1e-4 * s_target / n
    cur = np.maximum(cur, 0.0)
    cur = cur * s_target / cur.sum()

    # Keep the best (lowest) score seen so far.
    best_seq = cur.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] score = {best_score:.6f}")

    # The target value for the maximum convolution (theoretical lower bound).
    target = 2.0 * n / (2.0 * n - 1.0)

    # Gradient‑descent parameters.
    eta = 0.05            # initial step size
    min_eta = 1e-6
    max_eta = 1.0
    iter_count = 0
    stagnation = 0       # counts iterations without improvement

    # Main optimisation loop.
    while time.time() < deadline:
        # Perform a single gradient step and evaluate the resulting max‑convolution.
        new_seq, new_conv = _gradient_step(cur, target, eta)
        new_max = new_conv.max()                     # equals evaluate_sequence value
        cur_max = _full_convolution(cur).max()       # current max before the step

        # Accept the step only if it improves (decreases) the maximum.
        if new_max < cur_max:
            cur = new_seq
            cur_max = new_max
            # If the new point is also better than the global best, store it.
            if new_max < best_score:
                best_score = new_max
                best_seq = cur.copy()
                stagnation = 0
                print(f"[iter {iter_count:04d}] new best = {best_score:.6f}  eta={eta:.4g}")
            else:
                stagnation += 1
            # On successful steps we may increase the learning rate a little.
            eta = min(max_eta, eta * 1.02)
        else:
            # Unsuccessful step – reject it and shrink the step size.
            eta = max(min_eta, eta * 0.5)
            stagnation += 1

        # Occasionally inject a small random perturbation to avoid being stuck.
        if stagnation >= 200:
            jitter = (np.random.rand(n) - 0.5) * 1e-3 * best_seq.mean()
            cur = best_seq + jitter
            cur = np.maximum(cur, 0.0)
            cur = cur * s_target / cur.sum()
            stagnation = 0
            eta = max(min_eta, eta * 0.8)
            print(f"[iter {iter_count:04d}] random perturbation (eta={eta:.4g})")

        # Periodically recompute the true score with the provided evaluate_sequence
        # (the function also applies the clipping to [0,1000] and the sum‑check).
        if iter_count % 20 == 0:
            try:
                true_score = evaluate_sequence(cur.tolist())
                if true_score < best_score:
                    best_score = true_score
                    best_seq = cur.copy()
                    print(f"[iter {iter_count:04d}] true eval improved to {best_score:.6f}")
            except Exception:
                # Should never happen because we keep the sequence non‑negative
                # and its sum ≈ √(2n) > 0.01.
                pass

        iter_count += 1

    # Return the best sequence found (as a plain Python list of floats).
    print(f"[final] best score = {best_score:.6f} after {iter_count} iterations")
    return best_seq.tolist()
