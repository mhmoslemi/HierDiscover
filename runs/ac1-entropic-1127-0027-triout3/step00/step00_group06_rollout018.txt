def project_simplex(v):
    # Implementation (Duchi et al.)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

import numpy as np
import time

def project_simplex(v, s=1.0):
    """Projects vector v onto the probability simplex {x >= 0, sum x = s}."""
    # based on Duchi et al. (2008)
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero
        return np.maximum(v, 0)
    rho = ind[cond][-1] - 1  # zero-based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Simulated-annealing / coordinate-descent search for a non‑negative
    sequence that minimises `evaluate_sequence`.  Starts from a (slightly
    perturbed) uniform distribution and repeatedly moves a small amount of
    mass from one index to another, accepting moves that improve the score
    and occasionally accepting worse moves (temperature schedule).  The
    algorithm also performs greedy moves that directly target the largest
    convolution entry by decreasing the most contributing coordinate.
    The best sequence found before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # keep a safety margin

    # Choose a working length (short enough for fast conv, but enough freedom)
    n = 300

    # Initial sequence: uniform + tiny noise, then projected onto simplex.
    a = np.full(n, 1.0 / n, dtype=np.float64)
    a += np.random.normal(scale=1e-3, size=n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    # Optionally start from previous best if available
    hs1 = globals().get('height_sequence_1')
    if hs1 is not None:
        # Resize to our n by interpolation if needed
        hs1_arr = np.array(hs1, dtype=float)
        if len(hs1_arr) != n:
            # simple resampling with linear interpolation
            x_old = np.linspace(0, 1, len(hs1_arr))
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, hs1_arr)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        else:
            a = np.maximum(hs1_arr, 0.0)
            a = a / a.sum()

    best_a = a.copy()
    try:
        best_score = evaluate_sequence(best_a.tolist())
    except Exception:
        best_score = float('inf')
        best_a = np.full(n, 1.0 / n)

    # Parameters for the stochastic search
    temperature = 0.5          # initial temperature for Metropolis acceptance
    min_temp = 1e-4
    cooling = 0.9995           # geometric cooling per iteration
    step_factor = 0.02 / n     # initial fraction of a[i] that can be moved
    min_step = 1e-6
    iteration = 0
    last_print = time.time()

    while time.time() < deadline:
        iteration += 1

        # ---- 1) greedy move that attacks the current maximal convolution entry
        conv = np.convolve(a, a)
        k_max = int(np.argmax(conv))
        # compute contribution product a[i] * a[k_max - i] for all valid i
        indices = np.arange(n)
        valid = (k_max - indices >= 0) & (k_max - indices < n)
        partners = np.where(valid, a[k_max - indices], 0.0)
        contribution = a * partners
        i_src = int(np.argmax(contribution))          # most harmful coordinate
        i_dst = int(np.argmin(contribution))          # least harmful coordinate
        # move a small amount from src to dst
        eps = min(a[i_src] * step_factor, a[i_src])
        if eps > 0:
            a_candidate = a.copy()
            a_candidate[i_src] -= eps
            a_candidate[i_dst] += eps
            a_candidate = project_simplex(a_candidate)
            try:
                cand_score = evaluate_sequence(a_candidate.tolist())
            except Exception:
                cand_score = float('inf')
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                # shrink step size a little after successful improvement
                step_factor = max(min_step, step_factor * 0.99)
                temperature = max(min_temp, temperature * 0.99)
                continue
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(temperature, 1e-12)):
                a = a_candidate

        # ---- 2) random coordinate move (to explore other directions)
        i, j = np.random.randint(0, n, size=2)
        if i != j and a[i] > 0:
            eps = min(a[i] * step_factor, a[i])
            a_candidate = a.copy()
            a_candidate[i] -= eps
            a_candidate[j] += eps
            a_candidate = project_simplex(a_candidate)
            try:
                cand_score = evaluate_sequence(a_candidate.tolist())
            except Exception:
                cand_score = float('inf')
            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                step_factor = max(min_step, step_factor * 0.99)
                temperature = max(min_temp, temperature * 0.99)
                continue
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(temperature, 1e-12)):
                a = a_candidate

        # ---- cooling schedule
        temperature = max(min_temp, temperature * cooling)
        step_factor = max(min_step, step_factor * 0.9999)

        # occasional progress report
        if time.time() - last_print > 30.0:
            print(f"Iter {iteration}, best_score={best_score:.6f}, temperature={temperature:.6f}")
            last_print = time.time()

    # Final safety: ensure non‑negative and reasonable magnitude
    final_seq = best_a.tolist()
    # make sure we return a list of python floats
    final_seq = [float(x) for x in final_seq]
    return final_seq

                # keep the step size adaptive
                step_factor = max(min_step, step_factor * 0.99)
                temperature = max(min_temp, temperature * 0.99)
                continue
            # Metropolis acceptance for non‑improving moves
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(temperature, 1e-12)):
                a = a_candidate

        # --------------------------------------------------------------
        # 2) random coordinate move – helps escape plateaus
        # --------------------------------------------------------------
        i, j = np.random.randint(0, n, size=2)
        if i != j and a[i] > 0:
            eps = min(a[i] * step_factor, a[i])
            a_candidate = a.copy()
            a_candidate[i] -= eps
            a_candidate[j] += eps
            a_candidate = _project_simplex(a_candidate)

            try:
                cand_score = evaluate_sequence(a_candidate.tolist())
            except Exception:
                cand_score = float('inf')

            if cand_score < best_score:
                best_score = cand_score
                best_a = a_candidate.copy()
                a = a_candidate
                step_factor = max(min_step, step_factor * 0.99)
                temperature = max(min_temp, temperature * 0.99)
                continue
            if np.random.rand() < np.exp(-(cand_score - best_score) / max(temperature, 1e-12)):
                a = a_candidate

        # --------------------------------------------------------------
        # cooling schedule
        # --------------------------------------------------------------
        temperature = max(min_temp, temperature * cooling)
        step_factor = max(min_step, step_factor * 0.9999)

        # occasional progress report (useful when running interactively)
        if time.time() - last_report > 30.0:
            print(f"Iter {iteration:,d} – best_score = {best_score:.6f} – temp = {temperature:.6f}")
            last_report = time.time()

    # ------------------------------------------------------------------
    # return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    final_seq = [float(x) for x in best_a.tolist()]
    return final_seq
