"""
Propose a step‑function (list of non‑negative heights) that improves the
evaluation score of `evaluate_sequence`.  By analysing the objective

    score = 2 * n * max(conv(a, a)) / sum(a)^2

it can be shown that for a 2‑element sequence the minimal possible value is
obtained when the two heights are in the ratio 2:1, i.e. a = (2/3, 1/3) (or the
reverse order).  The resulting score is 16/9 ≈ 1.77778, which is strictly
better than the uniform construction that yields a score of 2.0.

The function below simply evaluates both orders and returns the one with the
lower score.  It runs in negligible time, well within the 1000‑second budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a sequence of non‑negative heights that gives a low value of
    `evaluate_sequence`.  The optimal 2‑element construction (2/3, 1/3) is used.
    """
    # Analytically optimal 2‑element vector (sum = 1)
    base_seq = [2.0 / 3.0, 1.0 / 3.0]

    # Evaluate both possible orderings – they give the same sum but
    # the convolution peak can differ due to the asymmetry of the
    # linear (non‑circular) convolution.
    seq1 = base_seq

    score1 = evaluate_sequence(seq1)
    score2 = evaluate_sequence(seq2)

    # Choose the ordering with the smaller score.
    if score2 < score1:

    else:

    # Log the result for visibility (the platform prints stdout).
    print(f"Best sequence found: {best_seq}")
