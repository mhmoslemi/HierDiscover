"""
Proposal:  Heuristic search for a step–function (non‑negative vector) that
minimises  2·n·max(conv(a,a)) / (sum(a)^2).  Because the score is invariant under
global scaling we keep the vector on the probability simplex (sum = 1).

The method works as follows:

1.  Choose a length n (we explore many small to medium sizes).
2.  Initialise a from a diverse set of distributions
    – Dirichlet random,
    – geometric decay,
    – a few handcrafted patterns.
3.  Perform a simple projected gradient‑like descent.
    * At each iteration compute the self‑convolution and identify the shift(s)
      attaining the current maximum.
    * For those shifts compute for every index i the total contribution
      a[j] (j = shift−i) – this is a sub‑gradient of the *max* term.
    * Move a modest amount of mass from the index with the largest contribution
      to the index with the smallest contribution (projected onto the simplex).
    * Occasionally inject a tiny random perturbation when the gradient is
      (almost) uniform to escape flat plateaus (e.g. the uniform vector).
4.  Keep the best vector seen across all restarts and all lengths.
5.  Return the best sequence before the time budget expires.

The algorithm is fast (O(n) per iteration) and works well on the
hard‑coded `evaluate_sequence` function.  It repeatedly restarts with
different lengths and initialisations, so it is unlikely to be trapped in a
single local minimum.
"""

import time
import numpy as np

# The evaluation function is provided in the environment.
# from <environment> import evaluate_sequence


def _contributions(a: np.ndarray, ks: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a, a)) w.r.t. a.
    For each shift k in ks (the active set where conv is maximal) we add
    a[k‑i] to the contribution of index i.
    """
    n = a.shape[0]
    contrib = np.zeros(n, dtype=float)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        j = k - i
        contrib[i] += a[j]
    return contrib


def _improve(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Heuristic improvement of a single vector a.
    Returns the best vector found and its score.
    """
    # Ensure the vector lies on the simplex.
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full_like(a, 1.0 / len(a))
    else:
        a = a / s

    best_seq = a.copy()
    best_score = float('inf')
    step = 0.05           # initial step size (fraction of the source entry)
    it = 0

    while time.time() < deadline:
        conv = np.convolve(a, a)
        max_val = conv.max()
        ks = np.where(conv >= max_val - 1e-12)[0]

        # sub‑gradient wrt the active shifts
        contrib = _contributions(a, ks)

        # If all contributions are (almost) equal we are on a plateau
        # (e.g. the uniform vector).  Apply a tiny random perturbation.
        if contrib.max() - contrib.min() < 1e-9:
            i, j = np.random.choice(len(a), size=2, replace=False)
            delta = min(a[i], step * a[i] if a[i] > 0 else 0.0)
            # give it a tiny random twitch if the entries are tiny
            if delta == 0:
                delta = step * 1e-3
            a[i] -= delta
            a[j] += delta
        else:
            i_max = int(np.argmax(contrib))
            i_min = int(np.argmin(contrib))
            # Move a modest amount of mass from i_max to i_min.
            delta = min(step * a[i_max], a[i_max] * 0.5)
            if delta <= 0:
                # fallback: tiny random move
                delta = step * 1e-4
            a[i_max] -= delta
            a[i_min] += delta

        # Project back onto the simplex (non‑negative & sum = 1)
        a = np.maximum(a, 0.0)
        total = a.sum()
        if total == 0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a = a / total

        # Evaluate current candidate
        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            # Reduce step size slowly when we make progress
            step = max(step * 0.99, 1e-4)

        # Periodically decay the step even without improvement
        it += 1
        if it % 200 == 0:
            step = max(step * 0.98, 1e-5)

    return best_seq, best_score

    """
    Main entry point used by the grader.
    Returns a list of non‑negative heights (the step‑function) that yields a low score.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # keep a small safety margin

    # If the historical construction is available we keep it as a fallback.
    global height_sequence_1
    fallback_seq = list(height_sequence_1) if 'height_sequence_1' in globals() else None

    # Lengths to explore – small to medium values give the best chance
    # to beat the current bound with very little computational effort.
    lengths = list(range(2, 31)) + [40, 50, 80, 120]

    best_overall_seq = None
    best_overall_score = float('inf')
    total_restarts = 0

    for n in lengths:
        # Limit time per length to keep the whole run within budget.
        per_len_deadline = min(deadline, time.time() + 8.0)

        # A few different initialisations for each length.
        for restart in range(6):
            total_restarts += 1
            if time.time() >= deadline:

            if restart == 0:

                a = np.array([(1 - r) * (r ** i) for i in range(n)], dtype=float)

            elif restart == 3:

            elif restart == 4:

                low = np.random.rand() * 0.1

                a = a / a.sum()
            else:

                a = np.random.random(n)
                a = a / a.sum()

            # Run the local improvement heuristic
            improved_seq, score = _improve(a.copy(), per_len_deadline)

        # As a last resort produce a tiny uniform vector (will give score = 2)

    # Convert to plain Python list before returning.
    return best_overall_seq.tolist()
