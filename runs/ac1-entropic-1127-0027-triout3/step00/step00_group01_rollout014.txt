def evaluate_obj(arr):
    arr = np.maximum(0, arr)
    arr = np.minimum(1000., arr)
    n = len(arr)
    # Use FFT convolution
    # zero-pad to next power of two
    m = 2*n - 1
    size = 1 << (m-1).bit_length()
    fft_a = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fft_a * fft_a, n=size)[:m]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a**2))

import numpy as np
import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a non‑negative step‑function sequence that minimizes
    evaluate_sequence.  The algorithm combines a small grid of parametric
    families (power‑law, exponential, two‑block piecewise) with a fast
    hill‑climbing / simulated‑annealing phase.  Convolution is evaluated
    by FFT, exploiting the scaling invariance of the objective.
    """
    np.random.seed(seed)
    # Determine length – use the known construction if available.
    try:
        n = len(height_sequence_1)
    except NameError:
        n = 5000    # fallback length
    # Pre‑compute FFT size for a given length
    fft_size = 1 << ((2 * n - 2).bit_length())
    # Helper that returns the evaluation function value (objective = 2*n*max(conv) )
    def objective(arr):
        arr = np.maximum(arr, 0.0)
        total = arr.sum()
        if total < 0.01:
            return np.inf
        p = arr / total                     # normalised probability distribution
        wp = np.fft.rfft(p, n=fft_size)
        conv = np.fft.irfft(wp * wp, n=fft_size)[: 2 * n - 1]
        return float(2.0 * n * conv.max())
    # ------------------------------------------------------------------
    # Baseline – uniform heights (the current “upper bound” of 2.0)
    baseline = np.ones(n, dtype=np.float64)
    best_seq = baseline.copy()
    best_score = objective(best_seq)
    print(f'Initial uniform score: {best_score:.6f}')
    # ------------------------------------------------------------------
    # 1️⃣  Parametric families -------------------------------------------------
    # a) Symmetric power‑law shape   a_i ∝ ((i+1)*(n‑i))^{‑α}
    alphas = np.linspace(0.0, 2.0, 41)          # α = 0 gives the uniform case
    idx = np.arange(n, dtype=np.float64)
    for alpha in alphas:
        a = ((idx + 1.0) * (n - idx)) ** (-alpha)
        sc = objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f'alpha {alpha:.3f} improved score to {best_score:.6f}')
    # b) One‑sided power‑law   a_i ∝ (i+1)^{‑β}  (plus a symmetric copy)
    betas = np.linspace(0.0, 2.0, 41)
    for beta in betas:
        a = (idx + 1.0) ** (-beta) + (n - idx) ** (-beta)
        sc = objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f'beta {beta:.3f} improved score to {best_score:.6f}')
    # c) Exponential decay from the left edge
    rs = np.linspace(0.90, 0.9999, 30)
    for r in rs:
        a = r ** idx
        sc = objective(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f'r {r:.5f} improved score to {best_score:.6f}')
    # d) Two‑block piecewise constant (large mass on a short prefix)
    for k_frac in np.linspace(0.001, 0.05, 20):   # very short prefix
        k = max(1, int(n * k_frac))
        for mass_frac in np.linspace(0.2, 0.9, 8):
            w1 = mass_frac / k
            w2 = (1.0 - mass_frac) / (n - k)
            a = np.empty(n, dtype=np.float64)
            a[:k] = w1
            a[k:] = w2
            sc = objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f'prefix {k} mass {mass_frac:.2f} improved score {best_score:.6f}')
    # also try a few very tiny prefixes (k = 1 … 10)
    for k in range(1, min(11, n)):
        for mass_frac in np.linspace(0.1, 0.9, 9):
            w1 = mass_frac / k
            w2 = (1.0 - mass_frac) / (n - k)
            a = np.empty(n, dtype=np.float64)
            a[:k] = w1
            a[k:] = w2
            sc = objective(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f'prefix {k} mass {mass_frac:.2f} improved score {best_score:.6f}')
    # ------------------------------------------------------------------
    # 2️⃣  Local stochastic refinement (simulated‑annealing style) ----------
    deadline = time.time() + budget_s - 2.0   # keep a small safety margin
    seq = best_seq.copy()
    # initial perturbation magnitude – a fraction of the typical entry size
    step = max(seq.mean(), 1e-3) * 0.1
    it = 0
    while time.time() < deadline:
        # generate a candidate by adding Gaussian noise to a random subset
        cand = seq.copy()
        # we perturb a random fraction of the coordinates (≈10 %)
        mask = np.random.rand(n) < 0.1
        noise = np.random.randn(np.count_nonzero(mask)) * step
        cand[mask] += noise
        # enforce non‑negativity (the evaluation function also truncates negatives)
        cand = np.maximum(cand, 0.0)
        cand_score = objective(cand)
        if cand_score < best_score:
            best_score = cand_score
            best_seq = cand.copy()
            seq = cand                     # continue the walk from the improvement
            # gradually shrink the step size so we can fine‑tune
            step *= 0.95
            print(f'Iter {it:6d}  new best {best_score:.6f}  step {step:.6e}')
        else:
            # occasionally reduce the step even if we do not improve
            if it % 1000 == 0 and step > 1e-6:
                step *= 0.99
        it += 1
    print('Finished search – best score', best_score)
    # Convert to plain Python list before returning
    return best_seq.tolist()

    #    very short prefixes (k = 1 … 10) are tried explicitly
    for k in range(1, min(11, n)):
        for mass_frac in np.linspace(0.1, 0.9, 9):
            w1 = mass_frac / k
            w2 = (1.0 - mass_frac) / (n - k)
            a = np.empty(n, dtype=np.float64)
            a[:k] = w1
            a[k:] = w2
            sc = objective(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f'prefix {k} mass {mass_frac:.2f} -> score {best_score:.6f}')

    # g)  Longer prefixes (up to 5 % of the length)
    for k_frac in np.linspace(0.001, 0.05, 20):
        k = max(1, int(n * k_frac))
        for mass_frac in np.linspace(0.2, 0.9, 8):
            w1 = mass_frac / k
            w2 = (1.0 - mass_frac) / (n - k)
            a = np.empty(n, dtype=np.float64)
            a[:k] = w1
            a[k:] = w2
            sc = objective(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f'prefix {k} mass {mass_frac:.2f} -> score {best_score:.6f}')

    # ----------------------------------------------------------------------
    # 2️⃣  Stochastic local refinement (mass‑transfer moves)
    deadline = time.time() + budget_s - 2.0  # reserve a tiny safety margin
    seq = best_seq.copy()
    step = 0.01                             # typical transfer amount (will shrink)

    it = 0
    while time.time() < deadline:
        # pick two distinct indices
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # propose moving a random amount delta from j to i
        delta = (np.random.rand() - 0.5) * 2.0 * step   # uniform in [-step, step]
        # enforce non‑negativity
        delta = np.clip(delta, -seq[i], seq[j])

        cand = seq.copy()
        cand[i] += delta
        cand[j] -= delta

        # evaluate the candidate
        sc = objective(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            seq = cand
            # shrink the step a little to allow finer adjustments
            step *= 0.95
            print(f'iter {it:6d}  improved score {best_score:.6f}  step {step:.6e}')
        else:
            # occasionally decay the step even without improvement
            if it % 2000 == 0 and step > 1e-7:
                step *= 0.99

        it += 1

    print(f'final best score = {best_score:.6f}')
    # Return a plain Python list (the framework expects a list)
    return best_seq.tolist()
