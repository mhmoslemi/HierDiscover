"""
Optimization of a non‑negative step function for the objective

    score = 2 * n * max( a * a ) / (sum(a) ** 2)

where `a * a` is the linear convolution of the sequence with itself.
The score is scale‑invariant, therefore we fix the ℓ₁‑norm to
sqrt(2·n) (the same normalisation used in the reference algorithm).  

The algorithm works in two phases:

1. **Parametric sweep** –  a one‑parameter family of symmetric shapes
   ``a_i ∝ ((i+1)*(n-i))**α`` is examined for many values of α
   (α∈[‑0.5,0.5]).  After normalisation each candidate is evaluated and
   the best one is kept.

2. **Simulated‑annealing refinement** – starting from the best candidate,
   a simple Metropolis walk is performed:
   * pick two indices i≠j;
   * move a small amount ε from i to j (ε is a random fraction of a_i);
   * re‑normalise the whole vector to the target ℓ₁‑norm;
   * accept the move if the score improves, otherwise accept with a
     temperature‑dependent probability.
   The temperature decreases linearly with the remaining time.

Both phases are bounded by the allotted wall‑clock budget (default 1000 s).
If the time runs out the best sequence found so far is returned.

The implementation avoids any heavy external optimisation libraries;
only NumPy is used for fast convolution and arithmetic.
"""

import math
import time
import random
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast score (does NOT perform the security checks of the
# official `evaluate_sequence`; we only use it during the search).
def _fast_score(seq: np.ndarray) -> float:
    """Return the evaluation score for a non‑negative sequence."""
    s = seq.sum()
    if s < 0.01:                      # same guard as evaluate_sequence
        return float('inf')
    max_b = np.max(np.convolve(seq, seq))
    return 2.0 * len(seq) * max_b / (s * s)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The function runs for at most `budget_s` seconds
    and returns the best sequence discovered.
    """
    # --------------------------------------------------------------
    # General set‑up
    random.seed(seed)
    np.random.seed(seed)

    deadline = time.time() + budget_s - 2.0   # small safety margin

    # ------------------------------------------------------------------
    # Choose a problem size – a few thousand entries give a good
    # theoretical lower bound while still keeping a convolution cheap.
    n = np.random.randint(800, 2000)      # 800 ≤ n < 2000
    target_sum = math.sqrt(2.0 * n)       # fixed ℓ₁‑norm (as in the reference)

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # Phase 1 – parametric sweep over a simple symmetric family.
    #   a_i(α) ∝ ((i+1)*(n-i)) ** α
    alphas = np.linspace(-0.5, 0.5, 51)    # 51 values, cheap to evaluate

    for alpha in alphas:
        # build the raw (non‑normalised) shape
        i = np.arange(n, dtype=np.float64)
        raw = ((i + 1.0) * (n - i)) ** alpha   # symmetric, non‑negative
        if raw.sum() == 0.0:                    # numerical safety
            continue
        seq = raw * (target_sum / raw.sum())

        # quick score for the candidate
        val = _fast_score(seq)

        if val < best_val:
            best_val = val
            best_seq = seq.copy()

        if time.time() >= deadline:
            return best_seq.tolist()

    # ------------------------------------------------------------------
    # Phase 2 – simulated‑annealing style local optimisation.
    # The score is monotone with `max_b` when the ℓ₁‑norm is fixed, so we
    # use the fast score for the Metropolis acceptance test.
    seq = best_seq.copy()
    best_val = _fast_score(seq)
    best_seq = seq.copy()

    # Temperature schedule: start relatively hot, cool down linearly.
    T_start = 0.8
    T_end   = 1e-4
    total_iters = 0

    while time.time() < deadline:
        # linear cooling depending on the fraction of time left
        fraction = (deadline - time.time()) / budget_s
        T = T_end + (T_start - T_end) * fraction

        # propose a simple mass‑transfer move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # move at most 1 % of a_i (sign randomly chosen)
        eps = (np.random.rand() - 0.5) * 0.02 * seq[i]
        if eps > seq[i]:                         # would make a_i negative
            continue

        new_seq = seq.copy()
        new_seq[i] -= eps
        new_seq[j] += eps

        # re‑normalise to keep the ℓ₁‑norm exactly `target_sum`
        new_seq *= (target_sum / new_seq.sum())

        new_val = _fast_score(new_seq)

        # Metropolis acceptance
        if new_val < best_val or np.random.rand() < math.exp(-(new_val - best_val) / (T + 1e-12)):
            seq = new_seq
            if new_val < best_val:
                best_val = new_val
                best_seq = new_seq.copy()

        total_iters += 1

        # occasional random restart if we seem stuck
        if total_iters % 50000 == 0 and time.time() < deadline:
            # 10 % chance to re‑initialise from a fresh random Dirichlet draw
            if np.random.rand() < 0.1:
                raw = np.random.rand(n) + 1e-6
                seq = raw * (target_sum / raw.sum())
                # keep the best sequence seen so far
                if _fast_score(seq) < best_val:
                    best_seq = seq.copy()
                    best_val = _fast_score(seq)

    # ------------------------------------------------------------------
    # Return the best sequence discovered (as a plain Python list)
    return best_seq.tolist()
