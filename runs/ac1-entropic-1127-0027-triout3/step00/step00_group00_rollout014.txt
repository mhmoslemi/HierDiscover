def project_to_simplex(v):
    # v: numpy array
    # returns projected onto simplex {x: x_i >=0, sum x_i = 1}
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a candidate step‑function (non‑negative heights) that beats the
baseline value 2.0.  The algorithm works with any length `n` (by default the
length of the previously found sequence `height_sequence_1`).  It builds a
non‑uniform, antisymmetric start‑up and then performs a projected
gradient‑descent on a smooth approximation of the objective

        f(a) = 2·n·max(conv(a,a)) / (sum(a)²)

using a log‑sum‑exp smoothing of the `max`.  After each step the vector is
projected back onto the simplex (non‑negative, fixed total mass) so the
search stays in the admissible region.  Random perturbations are injected
when the line‑search cannot find an improving step.  The best sequence seen
during the allotted time is returned.
"""

import numpy as np
import time
from typing import List

# ----------------------------------------------------------------------
# Helpers ---------------------------------------------------------------
# ----------------------------------------------------------------------


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        { x | x_i >= 0 , sum(x) = 1 }.
    Returns the projected vector (same shape as *v*).
    """
    if np.all(v <= 0):
        # all non‑positive – the projection is the unit vector on the largest entry
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w

    u = np.sort(v)[::-1]                # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def smoothed_max(conv: np.ndarray, beta: float = 30.0) -> (float, np.ndarray):
    """
    Returns a smooth approximation of max(conv) together with the
    (unnormalised) softmax weights that will be used for the gradient.
    """
    m = conv.max()
    # shift to avoid overflow, the maximal element gets weight 1
    shifted = conv - m
    # the exponent is negative for all non‑max entries
    w = np.exp(beta * shifted)          # w[max] == 1
    return m, w


def gradient_of_smoothed_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max (log‑sum‑exp) with respect to the
    vector *a*.  Uses correlation to avoid an explicit O(n²) loop.
    """
    L = len(w)                         # = 2·n‑1
    # np.correlate returns the full correlation (lags –(L‑1)…+(n‑1))
    corr = np.correlate(a, w, mode='full')
    # lag 0 corresponds to index L‑1; we need entries for lags 0…n‑1
    g = corr[L - 1 : L - 1 + len(a)]
    return g


def evaluate_fast(a: np.ndarray) -> float:
    """Fast computation of the evaluation function (avoids the import)."""
    conv = np.convolve(a, a)
    m = conv.max()
    s = a.sum()
    n = len(a)
    # protect against numerical zero – the original evaluate_sequence
    # returns np.inf in that case.
    if s < 0.01:
        return np.inf
    return float(2.0 * n * m / (s * s))


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: int = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a sequence of non‑negative heights that yields a small
    evaluation value.  The routine runs for up to *budget_s* seconds
    (default 1000) and returns the best sequence discovered.
    """
    # ------------------------------------------------------------------
    # 1. Basic data (length, baseline)
    # ------------------------------------------------------------------
    try:
        # Use the length of the previously found construction if it exists.
        n = len(height_sequence_1)          # type: ignore[name-defined]
        base_seq = np.asarray(height_sequence_1, dtype=float)  # type: ignore[name-defined]
    except Exception:
        n = 1500          # a reasonable default if the global is missing
        base_seq = None

    # ------------------------------------------------------------------
    # 2. Initialise a few starting points
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)

    def make_random_start():
        """Return a random antisymmetric start‑vector of length n."""
        base = 1.0
        a = np.full(n, base, dtype=float)

        # half‑size for the antisymmetric perturbation
        half = n // 2
        eps = rng.uniform(-0.4, 0.4, size=half)   # keep values positive after perturb
        a[:half] += eps
        a[-half:] -= eps
        if n % 2 == 1:
            # middle entry stays exactly at `base`
            a[half] = base

        # clip negatives and re‑scale to keep total mass = n (any scaling works)
        a = np.clip(a, 0.0, None)
        a = a * (n / a.sum())
        return a

    # start from the known construction (if any) and from a random one
    candidates = []
    if base_seq is not None:
        # re‑scale the known sequence to the same total mass `n`
        seq = np.clip(base_seq, 0.0, None)
        seq = seq * (n / seq.sum())
        candidates.append(seq)
    candidates.append(make_random_start())

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (projected gradient on smoothed max)
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 2.0   # a little margin for cleanup
    best_score = np.inf
    best_seq = None

    # algorithm parameters
    beta = 30.0            # smoothing parameter for log‑sum‑exp
    step_factor = 1.0     # initial step size (will be back‑tracked)
    min_step = 1e-6
    max_no_progress = 30  # after this many iterations without improvement we randomise a bit

    # we keep the current vector from the best of the initial candidates
    cur = candidates[0].copy()
    cur_score = evaluate_fast(cur)

    # keep the best of the initial seeds as a starting point
    for cand in candidates[1:]:
        sc = evaluate_fast(cand)
        if sc < cur_score:
            cur = cand.copy()
            cur_score = sc

    it = 0
    no_progress = 0

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # a) compute convolution, smoothed max and gradient
        # ------------------------------------------------------------------
        conv = np.convolve(cur, cur)
        max_val, w = smoothed_max(conv, beta=beta)
        grad = gradient_of_smoothed_max(cur, w)

        # ------------------------------------------------------------------
        # b) line‑search along the negative gradient direction
        # ------------------------------------------------------------------
        step = step_factor
        improved = False
        while step > min_step:
            # propose a new vector, move opposite to the gradient
            new = cur - step * grad
            # enforce non‑negativity
            new = np.clip(new, 0.0, None)
            # project onto the simplex keeping the total mass unchanged
            tot = new.sum()
            if tot == 0.0:
                # fall back to a random start if we collapsed to zero
                new = make_random_start()
                tot = new.sum()
            new = new * (cur.sum() / tot)          # scaling does not affect the ratio

            # evaluate
            new_score = evaluate_fast(new)

            if new_score < cur_score:
                # accept
                cur = new
                cur_score = new_score
                improved = True
                break
            else:
                step *= 0.5   # back‑track

        if improved:
            no_progress = 0
            # occasionally tighten the smoothing (focus more on the true max)
            if beta < 200.0:
                beta *= 1.02
        else:
            no_progress += 1
            # if we are stuck, inject a small random perturbation
            if no_progress >= max_no_progress:
                # random shuffle of a small sub‑set of indices
                idx = rng.choice(n, size=min(n, 20), replace=False)
                perturb = rng.normal(scale=0.05 * cur.mean(), size=idx.shape)
                cur[idx] = np.clip(cur[idx] + perturb, 0.0, None)
                cur = cur * (cur.sum() / cur.sum())  # re‑normalise (no change)
                cur_score = evaluate_fast(cur)
                no_progress = 0

        # ------------------------------------------------------------------
        # c) keep global best
        # ------------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur.copy()
            print(f"[{it:05d}] new best = {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 4. Return the best sequence found (as a plain Python list)
    # ------------------------------------------------------------------
    if best_seq is None:
        # safety fallback – return the baseline if everything failed
        return (base_seq if base_seq is not None else make_random_start()).tolist()
    return best_seq.tolist()
