"""
Propose a candidate step function using a simple LP‑based improvement loop.

The algorithm starts from a very sparse random sequence (many zeros, a few
positive entries).  For a fixed sequence `f` we compute the current
convolution maximum `M = max(f * f)`.  A linear program then finds a
non‑negative vector `g` with the largest possible ℓ₁‑sum subject to the
pointwise bound

    (f * g)[k] ≤ M   for all k                         (1)

Because `g` is constructed while never exceeding the current convolution
peak, adding a small fraction of `g` to `f` can only increase the total
sum while keeping the peak essentially unchanged.  The ratio

        2 * n * max(f*f) / (sum(f) ** 2)

therefore decreases.  By repeatedly solving the LP and mixing the new
direction we drive the evaluation value below the trivial constant‑function
bound of 2.

The whole procedure runs for the allotted time (≈ 1000 s) and returns the
best sequence encountered.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: solve the LP   max 1ᵀb   s.t.  (f * b)[k] ≤ rhs  (∀k),  b ≥ 0
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq, rhs, time_limit=5.0):
    """
    Solve the linear program:

        maximize    sum(b)
        subject to  convolution(f_seq, b)[k] ≤ rhs   (k = 0 … 2n‑2)
                    b_j ≥ 0                         (j = 0 … n‑1)

    Parameters
    ----------
    f_seq : list[float]
        Fixed non‑negative sequence `f`.
    rhs : float
        Upper bound for every convolution entry.
    time_limit : float, optional
        Upper bound on the solver time (seconds).

    Returns
    -------
    np.ndarray or None
        Optimal `b` (as a NumPy array) if the solver succeeds,
        otherwise ``None``.
    """
    n = len(f_seq)
    # objective: maximize sum(b)  →  minimize -sum(b)
    c = -np.ones(n)

    # build the inequality matrix A_ub * b ≤ b_ub
    # one row per convolution index (2n‑1 rows)
    A_ub = []
    b_ub = []

    for k in range(2 * n - 1):
        row = np.zeros(n)
        # row[j] = f_i where i = k‑j  (only if 0 ≤ i < n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        A_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity constraints  –b ≤ 0
    A_ub.append(-np.eye(n))
    b_ub.append(np.zeros(n))

    A_ub = np.vstack(A_ub)
    b_ub = np.hstack(b_ub)

    opts = {"max_iter": 2000, "disp": False}
    # recent SciPy versions support a `time_limit` argument via `options`.
    # We pass it through the generic `options` dict; if the solver does not
    # honour it, the iteration limit will still keep us safe.
    try:
        res = optimize.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            method="highs",
            options={**opts, "time_limit": time_limit},
        )
    except TypeError:  # older SciPy without `time_limit`
        res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, method="highs", options=opts)

    if res.success:
        return res.x
    return None

    low value of `evaluate_sequence`.

    The routine creates a sparse random start, then repeatedly
    improves it with the LP described above.  The best sequence found
    before the time budget expires is returned.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5   # a safety margin

    # --------------------------------------------------------------
    # configuration
    # --------------------------------------------------------------
    n = 800                     # length of the step function (≤ 1000 to avoid clamping)
    sparsity = max(1, n // 30)  # number of non‑zero entries in the initial vector
    step_size = 0.03            # mixing proportion (t in the description)
    lp_time = 4.0               # seconds per LP solve

    # start from a sparse random vector
    f = np.zeros(n)
    idx = np.random.choice(n, size=sparsity, replace=False)
    f[idx] = np.random.rand(sparsity) * 5.0 + 0.5   # modest positive values
    f = np.clip(f, 0, 1000)

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, sparsity={sparsity}, initial score={best_score:.6f}")

    iteration = 0
    while time.time() < deadline:
        iteration += 1

        # current convolution maximum
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        # solve the LP for a better direction
        g = solve_convolution_lp(f.tolist(), M, time_limit=lp_time)

        if g is None:
            # LP failed – perform a tiny random perturbation and continue
            j = np.random.randint(n)
            f[j] = max(0.0, f[j] + np.random.randn() * 0.02)
            f = np.clip(f, 0, 1000)
            continue

        # mix a fraction of the LP solution into the current sequence
        f = (1.0 - step_size) * f + step_size * g
        # honour the hard caps enforced by `evaluate_sequence`
        f = np.clip(f, 0, 1000)

        # evaluate the new candidate
        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[{iteration:5d}] improved score = {best_score:.6f}")

        # occasional aggressive restart if progress stalls
        if iteration % 300 == 0:
            # re‑introduce sparsity – keep the best part and zero out a fraction
            mask = np.random.rand(n) < 0.7
            f = best_seq * mask
            f += np.random.rand(n) * 0.01            # tiny seed to keep LP feasible
            f = np.clip(f, 0, 1000)

    print(f"[final] best score after {iteration} iterations = {best_score:.6f}")
    return best_seq.tolist()
