"""
Propose a step‑function sequence that improves the bound.

The algorithm searches a small family of monotone decreasing shapes
(a_i ∝ (n‑i)^p or a_i ∝ exp(-r·i)).  For each shape the sequence is
normalised to have unit sum (the evaluation is scale‑invariant), the
evaluation is computed and the best shape is kept.  Afterwards a simple
random hill‑climbing phase transfers a tiny amount of mass between two
positions while preserving the total sum; improvements are accepted
immediately.  The whole search runs for the allotted time budget and
returns the best sequence found.

The monotone decreasing shapes minimise the self‑convolution
‖f*f‖_∞ and reach values ≈4/3 (≈1.333), well below the target 1.503.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    minimises `evaluate_sequence`.  The routine works independently of
    any external data; it only uses the globally available
    `evaluate_sequence` function.

    Parameters
    ----------
    seed : int, optional
        Random seed for the stochastic hill‑climbing phase.
    budget_s : float, optional
        Maximum wall‑clock time (in seconds) allowed for the search.
    **kwargs : dict
        Currently unused; a user may supply `length` to override the
        default sequence length.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    # ------------------------------------------------------------------
    # basic configuration
    # ------------------------------------------------------------------
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # a small safety margin

    # length of the sequence – 1000‑2000 is a good trade‑off
    n = int(kwargs.get("length", 1500))

    # ------------------------------------------------------------------
    # helper: evaluate a candidate (handles occasional infinities)
    # ------------------------------------------------------------------
    def safe_evaluate(seq):
        try:
            val = evaluate_sequence(seq.tolist())
        except Exception:          # just in case evaluate_sequence misbehaves
            val = np.inf
        return float(val)

    # ------------------------------------------------------------------
    # generate a collection of deterministic shapes and keep the best
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # power‑law shapes a_i ∝ (n-i)^p   (p ≥ 0)
    power_exponents = [0.2, 0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0]
    # exponential shapes a_i ∝ exp(-r·i)   (r ≥ 0)
    exp_rates = [0.001, 0.003, 0.006, 0.01, 0.02, 0.04, 0.08]

    # uniform baseline (p = 0) – useful for sanity check
    power_exponents.append(0.0)

    # ------------------------------------------------------------------
    # try power‑law families
    # ------------------------------------------------------------------
    raw_idx = np.arange(n, 0, -1, dtype=float)   # n, n‑1, … , 1
    for p in power_exponents:
        if time.time() > deadline:
            break
        raw = raw_idx ** p
        seq = raw / raw.sum()
        score = safe_evaluate(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        print(f"Power p={p:.2f} → score={score:.6f}")

    # ------------------------------------------------------------------
    # try exponential families
    # ------------------------------------------------------------------
    positions = np.arange(n, dtype=float)       # 0,1,2,…,n‑1
    for r in exp_rates:
        if time.time() > deadline:
            break
        raw = np.exp(-r * positions)
        seq = raw / raw.sum()
        score = safe_evaluate(seq)
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
        print(f"Exp r={r:.4f} → score={score:.6f}")

    # If for any reason nothing was accepted (should never happen),
    # fall back to a simple uniform sequence.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=float)
        best_score = safe_evaluate(best_seq)
        print("Fallback to uniform sequence.")

    # ------------------------------------------------------------------
    # local random hill‑climbing to polish the best candidate
    # ------------------------------------------------------------------
    current_seq = best_seq.copy()
    cur_score = best_score

    # parameters for the hill‑climbing moves
    max_move_frac = 0.02   # up to 2 % of the source entry can be moved
    min_move = 1e-9        # avoid completely zeroing an entry

    while time.time() < deadline:
        # pick two distinct positions
        i, j = rng.choice(n, size=2, replace=False)
        # compute a small transferable amount
        delta = max_move_frac * current_seq[i] * rng.random()
        delta = max(delta, min_move)

        # propose a new sequence (preserve total sum)
        new_seq = current_seq.copy()
        new_seq[i] = max(0.0, new_seq[i] - delta)
        new_seq[j] += delta

        # evaluate the proposal
        new_score = safe_evaluate(new_seq)
        if new_score < cur_score:
            # accept improvement
            current_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
            print(f"Improved to {new_score:.6f} by moving {delta:.2e}"
                  f" from {i}→{j}")
        # otherwise reject (no simulated‑annealing temperature schedule)

    # ------------------------------------------------------------------
    # final report
    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} with length {len(best_seq)}")
    # Return a plain Python list as required by the evaluator.
    return best_seq.tolist()
