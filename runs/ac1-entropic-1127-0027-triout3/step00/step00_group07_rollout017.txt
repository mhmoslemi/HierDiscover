def lp_step(seq):
    n = len(seq)
    s = sum(seq)
    if s < 1e-6:
        return None
    # normalize f as per algorithm
    f = [x * np.sqrt(2 * n) / s for x in seq]
    rhs = max(np.convolve(f, f))
    g = solve_convolution_lp(f, rhs)
    if g is None:
        return None
    sum_g = np.sum(g)
    g_norm = [x * np.sqrt(2 * n) / sum_g for x in g]
    # Mix

    # try multiple t
    best = None
    best_score = float('inf')
    for t in [0.005, 0.01, 0.05, 0.1]:
        new_seq = [(1 - t) * x + t * y for x, y in zip(seq, g_norm)]
        score = evaluate(new_seq)
        if score < best_score:
            best_score = score
            best = new_seq
    return best

"""
Simple stochastic optimizer for the step‑function evaluation problem.

The objective 2·n·max(conv(a,a)) / (sum(a)^2) is homogeneous:
scaling all coefficients does not change the score.
Therefore we keep the sum of the current candidate equal to 1
(normalisation) – this avoids numerical issues and makes the
objective simply 2·n·max(conv).

The optimizer keeps the best sequence found so far and repeatedly
tries two kinds of moves:

1. **LP‑driven move** – solve a linear program that finds a vector
   `g` with a larger ℓ₁‑norm while guaranteeing that the convolution
   bound does not increase.  The current sequence is mixed with `g`
   using several mixing coefficients `t` and the best resulting
   candidate is kept.

2. **Random local perturbation** – pick a random index and add a
   small Gaussian perturbation, then re‑normalise.

Occasionally a fresh random start is generated to escape local minima.
The loop runs until the allocated time budget expires and finally
returns the best sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programming utilities
# ----------------------------------------------------------------------
def solve_convolution_lp(f_sequence, rhs):
    """
    Solve the LP:
        maximise   sum(b_j)
        subject to   conv(f, b) <= rhs  (for every shift)
                     b_j >= 0

    Returns the optimal b vector (as a NumPy array) or None if the LP
    fails.
    """
    n = len(f_sequence)
    # objective: maximise sum(b) -> minimise -sum(b)
    c = -np.ones(n)

    # Build the convolution constraints A_ub @ b <= rhs
    a_rows = []
    b_vals = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # conv coefficient for shift k: sum_i f_i * b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_rows.append(row)
        b_vals.append(rhs)

    # non‑negativity constraints: -b <= 0
    a_rows.append(-np.eye(n))
    b_vals.append(np.zeros(n))

    A_ub = np.vstack(a_rows)
    b_ub = np.hstack(b_vals)

    # Use the high‑performance HiGHS solver
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs',
                  options={'presolve': True, 'time_limit': 8.0})
    if res.success:
        return res.x
    return None


def lp_direction(seq):
    """
    Given a non‑negative sequence `seq`, compute a direction `g`
    that (approximately) increases the ℓ₁‑norm while keeping the
    convolution bound unchanged.
    Returns the normalised vector `g` (list) or None if the LP fails.
    """
    n = len(seq)
    total = float(np.sum(seq))
    if total < 1e-12:
        return None
    # Normalise to the standard scale used by the original SOTA code
    f = [x * np.sqrt(2.0 * n) / total for x in seq]
    rhs = max(np.convolve(f, f))
    g_raw = solve_convolution_lp(f, rhs)
    if g_raw is None:
        return None
    # Re‑normalise g to the same ℓ₁‑norm as the original sequence
    g_sum = float(np.sum(g_raw))
    if g_sum < 1e-12:
        return None
    g = [float(x) * np.sqrt(2.0 * n) / g_sum for x in g_raw]
    return g


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def normalise(seq):
    """Scale the sequence so that its ℓ₁‑norm equals 1 (or leave it unchanged if sum≈0)."""
    s = float(np.sum(seq))
    if s < 1e-12:
        return seq
    return [float(x) / s for x in seq]


def max_convolution(seq):
    """Return max(conv(seq, seq)) – a fast wrapper."""
    return float(np.max(np.convolve(seq, seq)))
