"""
propose_candidate
-----------------
A mixed heuristic that searches for a step‑function (a list of non‑negative heights)
with a small value of

    2·n·max(conv(a,a)) / (sum(a)**2)

The search consists of three stages

1. **Structured initial families** – power‑law sequences
   a_i = (i+1)^(-β) for a few choices of length *n* and exponent β.
   Each sequence is randomly permuted; the best permutation is kept.

2. **Local improvement** – random pairwise swaps (hill climbing).  
   While time permits we repeatedly try to swap two positions; a swap is kept
   if it strictly lowers the evaluation score.

3. **LP‑based refinement (optional)** – using the same linear program as the
   published “state‑of‑the‑art” method.  From the current best sequence we
   solve a LP that finds a direction *g* with the same convolution bound.
   A 1‑D line search in the mixture (1‑t)·f + t·g is performed and the best
   mixture is taken as the new current sequence.

All operations respect the 1000 s time budget; intermediate progress and the
best score are printed.  The function finally returns the best list it has
found.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper: linear program used by the original SOTA method
# ----------------------------------------------------------------------
def solve_convolution_lp(f_sequence, rhs, time_limit=5.0):
    """
    Solve
        maximize   Σ b_j
        subject to ∀k : (f * b)[k] ≤ rhs
                  b_j ≥ 0
    The LP is written in the standard “c, A_ub, b_ub” form.
    Returns the optimal b as a NumPy array or None on failure.
    """
    n = len(f_sequence)
    c = -np.ones(n)                     # maximise Σb ⇔ minimise -Σb
    # build convolution constraints
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # (f * b)[k] = Σ_i f_i * b_{k-i}
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)

    # non‑negativity of b
    a_ub = np.vstack([a_ub, -np.eye(n)])
    b_ub = np.hstack([b_ub, np.zeros(n)])

    # use the interior‑point method; give a small time budget
    opts = {"time_limit": time_limit, "disp": False}
    result = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub, method="highs", options=opts)
    if result.success:
        return result.x
    return None


# ----------------------------------------------------------------------
# Helper: generate a power‑law sequence
# ----------------------------------------------------------------------
def build_power_sequence(n, beta):
    """
    Return a NumPy array a_i = (i+1)^{-beta}, i = 0 … n‑1.
    The values are positive and already inside the allowed range.
    """
    idx = np.arange(1, n + 1, dtype=float)
    seq = idx ** (-beta)
    return seq


# ----------------------------------------------------------------------
# Helper: hill‑climbing (random swaps) on a sequence
# ----------------------------------------------------------------------
def hill_climb(seq, score, deadline, rng, max_no_improve=2000):
    """
    Perform a simple random‑swap hill‑climber.
    *seq*   – NumPy array (will be modified in place)
    *score* – current evaluation of *seq*
    Returns the (possibly) improved sequence and its score.
    """
    n = len(seq)
    no_improve = 0
    while time.time() < deadline and no_improve < max_no_improve:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # try the swap
        seq[i], seq[j] = seq[j], seq[i]
        new_score = evaluate_sequence(seq.tolist())
        if new_score < score:
            score = new_score
            no_improve = 0
        else:
            # revert
            seq[i], seq[j] = seq[j], seq[i]
            no_improve += 1
    return seq, score


# ----------------------------------------------------------------------
# Helper: LP‑based refinement of a sequence (optional)
# ----------------------------------------------------------------------
def lp_refine(seq, deadline):
    """
    Starting from *seq* (list or array) repeatedly apply the LP direction
    improvement described in the statement.  A line‑search in the mixing
    parameter t∈[0,1] is performed each iteration.
    Returns the best sequence seen during the refinement.
    """
    f = np.array(seq, dtype=float)
    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    while time.time() < deadline:
        n = len(f)
        sum_f = f.sum()
        if sum_f == 0:
            break
        # normalise to the “standard” L1‑norm √(2n)
        f_norm = f * np.sqrt(2.0 * n) / sum_f
        # current convolution bound
        M = np.max(np.convolve(f_norm, f_norm))

        # solve LP for a direction g
        g = solve_convolution_lp(f_norm.tolist(), M, time_limit=3.0)
        if g is None:
            break

        g = np.array(g, dtype=float)
        sum_g = g.sum()
        g_norm = g * np.sqrt(2.0 * n) / sum_g

        # line‑search: try a few t values and keep the best
        best_local_score = np.inf
        best_t = 0.0
        # we never try t=0 (no change) – start at a small positive value
        for t in np.linspace(0.05, 1.0, 20):
            cand = (1.0 - t) * f_norm + t * g_norm
            sc = evaluate_sequence(cand.tolist())
            if sc < best_local_score:
                best_local_score = sc
                best_t = t
        if best_t == 0.0:
            break      # no improvement possible

        # update current sequence
        f = (1.0 - best_t) * f_norm + best_t * g_norm
        cur_score = evaluate_sequence(f.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
        # continue the loop – the bound M tends to drop

    return best_seq.tolist()
