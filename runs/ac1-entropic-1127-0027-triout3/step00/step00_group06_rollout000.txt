"""
Propose a candidate sequence that minimizes the evaluation function.

Idea
----
The evaluation function is  

    2 * n * max_k (a * a)[k] / (sum_i a_i)^2

with a_i ≥ 0.  Since the expression is homogeneous, we can fix the
normalisation sum_i a_i = 1 and minimise   t = max_k (a * a)[k].
The constraints

    (a * a)[k] = Σ_{i+j=k} a_i a_j ≤ t        (k = 0,…,2n−2)

are convex after the change of variables a_i = exp(u_i)
(because a_i a_j = exp(u_i+u_j) and the sum of exponentials is a
convex function).  Hence we obtain a convex programme

    minimise   t
    subject to Σ_i exp(u_i) = 1,
               Σ_{i+j=k} exp(u_i+u_j) ≤ t   for all k,
               t ≥ 0.

The problem is a geometric programme and can be solved with CVXPY
(using the exponential cone).  For a fixed length n the optimal value
t* satisfies   t* ≥ 1/(2n‑1)   (the average of the convolution), and the
resulting evaluation score is 2·n·t*.  By taking n ≈ 300–400 we can get
scores very close to the theoretical lower bound 2n/(2n‑1) ≈ 1.

The function below tries a few values of n (as long as time permits),
solves the GP with ECOS, and keeps the best sequence found.
If the solver fails or time runs out, it falls back to the uniform
sequence (which gives the current bound 2.0).

No external files are accessed; only numpy, time, and cvxpy are used.
"""

import time
import numpy as np
import math

def _solve_for_n(n: int, time_limit: float):
    """
    Solve the geometric programme for a given length n.

    Returns
    -------
    seq : list[float] | None
        The optimal sequence (normalised to sum 1) or None on failure.
    score : float
        The evaluation score of the sequence (inf if failure).
    """
    import cvxpy as cp

    # Create variables: u (log‑amplitudes) and the upper bound t
    u = cp.Variable(n)
    t = cp.Variable(nonneg=True)

    constraints = []

    # Normalise the sum of a_i = exp(u_i) to 1
    constraints.append(cp.sum(cp.exp(u)) == 1)

    # Convolution constraints: for each possible sum index k
    for k in range(2 * n - 1):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # build the list of exp(u_i + u_{k-i}) terms
        terms = [cp.exp(u[i] + u[k - i]) for i in range(i_min, i_max + 1)]
        constraints.append(cp.sum(cp.hstack(terms)) <= t)

    prob = cp.Problem(cp.Minimize(t), constraints)

    # Solver options – ECOS handles exponential cones efficiently.
    # We let ECOS run for at most the remaining time (rounded down).
    max_iters = 20000

    try:
        start = time.time()
        prob.solve(solver=cp.ECOS,
                   max_iters=max_iters,
                   abstol=1e-9,
                   reltol=1e-9,
                   feastol=1e-9,
                   verbose=False)
        elapsed = time.time() - start
        # If the solver spent too much time, treat as failure.
        if elapsed > time_limit:
            print(f"  n={n}: solver exceeded time limit ({elapsed:.1f}s).")
            return None, math.inf
    except Exception as e:  # pragma: no cover
        print(f"  n={n}: solver error – {e}")
        return None, math.inf

    if prob.status not in ("optimal", "optimal_inaccurate"):
        print(f"  n={n}: solver did not converge (status = {prob.status}).")
        return None, math.inf

    # Recover the sequence (it is already normalised to sum 1)
    a = np.exp(u.value)          # a_i = exp(u_i) ≥ 0
    seq = a.tolist()
    try:
        score = evaluate_sequence(seq)
    except Exception:  # safety net – should never happen
        score = math.inf
    return seq, score
