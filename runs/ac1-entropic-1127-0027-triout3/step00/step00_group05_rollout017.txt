def improve_sequence(seq):
    n = len(seq)
    sum_seq = seq.sum()
    # normalized the seq for LP as per earlier: scaled to sqrt(2n)
    normalized_seq = seq * np.sqrt(2 * n) / sum_seq
    rhs = np.max(np.convolve(normalized_seq, normalized_seq))
    g = solve_convolution_lp(normalized_seq, rhs)
    if g is None: return seq
    sum_g = np.sum(g)
    normalized_g = g * np.sqrt(2 * n) / sum_g
    # compute convs
    a = np.convolve(seq, seq)   # current conv values
    b = np.convolve(seq, normalized_g)   # cross conv
    c = np.convolve(normalized_g, normalized_g)  # conv of g
    M = np.max(a)
    # binary search t from 0 to 1
    low, high = 0.0, 1.0
    best_t = 0.0
    for _ in range(30):
        mid = (low + high) / 2
        # compute new max
        new_vals = (1 - mid)**2 * a + 2*mid*(1 - mid) * b + mid**2 * c
        if np.max(new_vals) <= M + 1e-12:

            low = mid
        else:
            high = mid
    # Update seq

"""
Propose a high‑quality step‑function (non‑negative height sequence) for the
evaluation function:

    eval(seq) = 2 * n * max(conv(seq, seq)) / (sum(seq)**2)

The algorithm starts from the analytically optimal “Catalan” sequence,
which makes the first half of the auto‑convolution exactly constant.
From this point a projected gradient descent is performed:

* The index k* where the convolution attains its maximum is found.
* The sub‑gradient of max(conv) w.r.t. the coefficients is d_i = a_{k*-i}
  (or 2·a_i when i = k*-i, 0 otherwise).
* We move mass from indices with large d_i (they contribute heavily to the
  peak) to indices with small d_i (they do not), while keeping the total
  sum unchanged (projected step).  The step size is chosen so that no entry
  becomes negative.
* The move is accepted only if the evaluation value decreases.
* Occasionally a small random perturbation is tried to escape possible
  plateaus.

The loop runs until the time budget is exhausted (default 1000 s).  The best
sequence found is returned as a plain Python list, ready for the external
`evaluate_sequence` routine.

This method is completely self‑contained, uses only NumPy and the provided
`evaluate_sequence`, and respects all the problem constraints.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper: generate the Catalan (central‑binomial) sequence
# -------------------------------------------------------------
def catalan_sequence(length: int) -> np.ndarray:
    """
    Return an array a[0..length-1] where
        a[i] = C(2i, i) / 4**i  (central binomial coefficient divided by 4^i)

    The sequence satisfies  a_0 = 1 and a_{i+1} = a_i * (2*i+1)/(2*i+2).
    """
    a = np.empty(length, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, length):
        a[i] = a[i - 1] * (2 * i - 1) / (2 * i)
    return a


# -------------------------------------------------------------
# Core optimisation step: projected gradient move
# -------------------------------------------------------------
def projected_gradient_move(seq: np.ndarray, step_factor: float) -> np.ndarray:
    """
    Perform one projected gradient step that tries to lower the peak
    of the auto‑convolution while keeping the L1‐norm (the sum) unchanged.

    Parameters
    ----------
    seq : np.ndarray
        Current non‑negative coefficient vector.
    step_factor : float
        Multiplicative factor (0 < step_factor ≤ 1) that scales the maximal
        admissible step length.

    Returns
    -------
    np.ndarray
        New sequence after the move (may be identical to the input if no
        feasible move was found).
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))          # shift where the peak occurs

    # Sub‑gradient of the peak w.r.t. each coefficient
    i = np.arange(n)
    j = k_max - i
    mask = (j >= 0) & (j < n)

    d = np.zeros_like(seq)                # derivative of the peak
    same = mask & (i == j)                # terms a_i * a_i
    d[same] = 2.0 * seq[i[same]]
    diff = mask & (i != j)                # terms a_i * a_{k-i}
    d[diff] = seq[j[diff]]

    # Projection that keeps the total sum unchanged:
    #   delta = mean(seq) - d  (so that sum(delta)=0)
    avg = np.mean(seq)
    delta = avg - d

    # Determine the largest step that keeps all entries non‑negative
    negative = delta < 0
    if np.any(negative):
        eta_max = np.min(seq[negative] / (-delta[negative]))
    else:                                 # delta is everywhere non‑negative
        eta_max = np.inf

    # Use a fraction of the admissible step (step_factor) and also bound it
    # so that we do not move too far in a single iteration.
    eta = min(eta_max * step_factor, avg * 0.1)   # 0.1·average is a safe upper bound
    if eta <= 0.0:                         # nothing can be moved safely
        return seq

    new_seq = seq + eta * delta
    # Numerical safety – clip to problem domain
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq


# -------------------------------------------------------------
# Main entry point required by the competition harness
# -------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimizes the
    provided evaluation function.  The routine runs for at most
    `budget_s` seconds and returns the best sequence discovered.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds (the wrapper reserves a few seconds for
        clean‑up, so we stop slightly earlier).

    Returns
    -------
    list[float]
        The best height sequence found.
    """
    np.random.seed(seed)
    # Reserve a small safety margin before the hard timeout.
    deadline = time.time() + budget_s - 5.0

    # ---------------------------------------------------------
    # Initialise: use the Catalan sequence with the same length as
    # the previously found construction (if available) or fall back
    # to a moderate default length.
    # ---------------------------------------------------------
    try:
        n = len(height_sequence_1)          # noqa: F821  (global supplied by the platform)
    except NameError:
        n = 2000                           # a reasonable default if the variable is missing

    # Catalan start is already very good (evaluation ≈ π/2 ≈ 1.571)
    current = catalan_sequence(n)

    # Keep track of the best solution seen so far
    best_seq = current.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # Parameters for the projected gradient descent
    step_factor = 0.9          # initial fraction of the maximal admissible step
    decay = 0.9995             # slowly shrink the step size
    min_step = 1e-6            # never go below this magnitude

    iteration = 0
    while time.time() < deadline:
        # -----------------------------------------------------
        # 1) deterministic projected‑gradient move
        # -----------------------------------------------------
        new_seq = projected_gradient_move(current, step_factor)

        # Evaluate only if a change actually occurred
        if not np.allclose(new_seq, current):
            new_score = evaluate_sequence(new_seq.tolist())
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                # small log line – helps to monitor progress when run locally
                print(f"Iter {iteration:5d}: improved score = {best_score:.6f}")
            # Accept the move if it improves the current candidate
            if new_score < evaluate_sequence(current.tolist()):
                current = new_seq
        else:
            # No feasible deterministic move – fall back to a tiny random tweak
            idx = np.random.randint(n)
            tweak = (np.random.rand() - 0.5) * 0.01 * max(1.0, current[idx])
            candidate = current.copy()
            candidate[idx] = max(0.0, candidate[idx] + tweak)
            cand_score = evaluate_sequence(candidate.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = candidate.copy()
                print(f"Iter {iteration:5d}: random improvement = {best_score:.6f}")
            if cand_score < evaluate_sequence(current.tolist()):
                current = candidate

        # -----------------------------------------------------
        # 2) occasional larger random reshuffle (helps escape plateaus)
        # -----------------------------------------------------
        if np.random.rand() < 0.001:
            # pick a small subset of positions and redistribute their total mass
            idxs = np.random.choice(n, size=5, replace=False)
            total = current[idxs].sum()
            # generate a random Dirichlet distribution for the mass split
            weights = np.random.rand(5)
            weights /= weights.sum()
            candidate = current.copy()
            candidate[idxs] = total * weights
            cand_score = evaluate_sequence(candidate.tolist())
            if cand_score < best_score:
                best_score = cand_score
                best_seq = candidate.copy()
                print(f"Iter {iteration:5d}: reshuffle improvement = {best_score:.6f}")
            if cand_score < evaluate_sequence(current.tolist()):
                current = candidate

        # -----------------------------------------------------
        # 3) slowly reduce the step size to refine the solution
        # -----------------------------------------------------
        step_factor = max(step_factor * decay, min_step)

        # occasional status output (every few thousand iterations)
        if iteration % 5000 == 0 and iteration > 0:
            print(f"[{iteration}] current best = {best_score:.6f}, step_factor={step_factor:.6e}")

        iteration += 1

    # Final report
    print("Search finished.")
    print(f"Best evaluation value found: {best_score:.6f}")
    return best_seq.tolist()
