def propose_candidate(seed=0, budget_s=1000):
    """Search for a step sequence minimizing evaluate_sequence.
    Uses a Gaussian mixture parametric family and differential evolution,
    followed by a local hill‑climbing refinement on the raw sequence."""
    import numpy as np, math, time, random
    from scipy.optimize import differential_evolution

    np.random.seed(seed)
    random.seed(seed)

    # Determine length
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, list):
        n = len(height_sequence_1)
    else:
        n = 3000

    deadline = time.time() + max(0, budget_s - 5)

    # precompute x grid
    x = np.linspace(0, 1, n)

    # utility sigmoid
    def sigmoid(z):
        return 1.0/(1.0+math.exp(-z))

    # generate a sequence from a parameter vector
    def gen_seq(params):
        m = 4   # number of Gaussian components
        seq = np.zeros(n)
        for k in range(m):
            w = math.exp(params[3*k])
            mu = sigmoid(params[3*k + 1])
            sigma = math.exp(params[3*k + 2])
            # avoid sigma too tiny
            if sigma < 1e-6:
                sigma = 1e-6
            seq += w * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
        # make non‑negative (already)
        total = seq.sum()
        if total <= 0:
            seq = np.ones_like(seq) * 1e-6
            total = seq.sum()
        seq = seq / total
        # optional scaling (not needed)
        seq = seq * 10.0  # keep values modest
        # convert to python list for evaluate_sequence
        return seq.tolist()

    # objective for optimizer (minimize)
    def obj_func(params):
        seq = gen_seq(params)
        try:
            val = evaluate_sequence(seq)
        except Exception:
            val = float('inf')
        return val

    bounds = []

    w_low, w_high = -5, 5      # exp => approx 0.0067 to 148

    sigma_low, sigma_high = math.log(0.001), math.log(0.5)

    for _ in range(m):
        bounds.append((w_low, w_high))
        bounds.append((mu_low, mu_high))
        bounds.append((sigma_low, sigma_high))

    best_score = float('inf')
    best_seq = None
    best_params = None

    # Run several DE restarts while time permits
    restart = 0
    while time.time() < deadline - 1 and restart < 5:
        restart += 1
        # For each restart we can randomize initial popsize based on remaining time
        remaining = deadline - time.time()
        max_iter = max(5, int( (remaining / budget_s) * 30 ))  # adapt
        popsize = 8
        print(f"DE restart {restart}, max_iter={max_iter}")
        result = differential_evolution(
            obj_func,
            bounds,
            maxiter=max_iter,
            popsize=popsize,
            polish=False,

            disp=False,
            workers=1,

            seq = gen_seq(result.x)
            score = evaluate_sequence(seq)
            print(f"  DE result score {score:.6f}")
            if score < best_score:
                best_score = score
                best_seq = seq
                best_params = result.x
                print(f"  New global best {best_score:.6f}")

    if best_seq is None:
        # fallback: uniform
        best_seq = [1.0] * n
        best_score = evaluate_sequence(best_seq)
        print("Fallback to uniform sequence")

        seq = best_seq[:]
        score = best_score

            new_seq = np.array(seq, copy=True)

            if new_score < score:

                if score < best_score:
                    best_score = score
                    best_seq = seq[:]
                    print(f"  HC improvement to {best_score:.6f} at iter {iter_hc}")
            # else reject
        # final best_seq already set

    print("Final best score:", best_score)
    return best_seq

import numpy as np
import math
import time
import random
from scipy.optimize import differential_evolution

def sigmoid(z):
    """Sigmoid mapping from R to (0,1)."""
    # Use math.exp for speed in scalar context.
    return 1.0 / (1.0 + math.exp(-z))

def generate_sequence(params, n, x):
    """Create a non‑negative step sequence from a Gaussian‑mixture parameter vector.

    The parameter vector is interpreted as m=4 triples (log_weight, logit_mu, log_sigma).
    The returned object is a plain Python list suitable for evaluate_sequence.
    """
    m = 4
    seq = np.zeros(n, dtype=np.float64)
    for k in range(m):
        w = math.exp(params[3 * k])                    # weight >0
        mu = sigmoid(params[3 * k + 1])               # centre in [0,1]
        sigma = math.exp(params[3 * k + 2])           # width >0
        # avoid absurdly small sigma that would create huge spikes
        if sigma < 1e-6:
            sigma = 1e-6
        seq += w * np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    total = seq.sum()
    if total <= 0.0:
        # Degenerate case – fall back to a tiny flat sequence.
        seq = np.full(n, 1e-6, dtype=np.float64)
        total = seq.sum()
    # Normalise to make the sum independent of the scale.
    seq = seq / total
    # Multiply by a modest constant to avoid zeroes after the evaluating code
    # (the objective is scale‑invariant, the factor does not matter).
    seq = seq * 10.0
    # Convert to a plain list of floats, as required by evaluate_sequence.
    return seq.tolist()

class ObjectiveEvaluator:
    """Callable object that evaluates the objective for a given parameter vector."""
    def __init__(self, n):
        self.n = n
        self.x = np.linspace(0.0, 1.0, n)

    def __call__(self, params):
        seq = generate_sequence(params, self.n, self.x)
        try:
            return evaluate_sequence(seq)
        except Exception:
            # Any problem (e.g. overflow) is treated as a very bad score.
            return float('inf')

    """
    Search for a non‑negative step sequence with a low evaluation score.

    The search uses:
      * A 4‑component Gaussian‑mixture family to parametrise candidate sequences.
      * Differential Evolution (SciPy) to explore the parameter space.
      * A short hill‑climbing phase that tweaks the best raw sequence.
    The function respects the time budget and returns the best sequence found.
    """

    np.random.seed(seed)
    random.seed(seed)

    # Choose a length.  If a previous construction is available we reuse its size.
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, list):
        n = len(height_sequence_1)
    else:
        n = 3000  # a reasonable default when no prior data is present.

    deadline = time.time() + max(0.0, budget_s - 5.0)  # leave a small safety margin

    # ----- Differential‑Evolution stage -----
    evaluator = ObjectiveEvaluator(n)

    # Bounds for the 4 Gaussian components (weight, mu, sigma) expressed in log‑space.
    # weight   : log w  ∈ [−5, 5]   → w ≈ [0.0067, 148]
    # mu       : logit μ ∈ [−10, 10] → μ ∈ (≈0, ≈1)
    # sigma    : log σ  ∈ [ln(0.001), ln(0.5)] → σ ≈ [0.001, 0.5] (relative to unit interval)
    m = 4
    bounds = []
    for _ in range(m):
        bounds.append((-5.0, 5.0))          # log‑weight
        bounds.append((-10.0, 10.0))        # logit‑mu
        bounds.append((math.log(0.001), math.log(0.5)))  # log‑sigma

    best_score = float('inf')
    best_seq = None
    best_params = None

    restart = 0
    while time.time() < deadline - 2.0 and restart < 5:
        restart += 1
        # Dynamically shrink the allowed generations if time is running out.
        remaining = deadline - time.time()
        max_iter = max(5, int(30 * remaining / budget_s))

            evaluator,

            maxiter=max_iter,

            disp=False,

            continue

        # Evaluate.

            new_score = evaluate_sequence(new_seq.tolist())
        except Exception:
            continue

        if new_score < current_score:
            seq = new_seq
            current_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.tolist()
                print(f"HC improvement: iteration {iteration}, score {best_score:.6f}")

    print("Search finished. Best score:", best_score)
    return best_seq
