def smooth_max_grad(p, alpha):
    # compute conv = p conv p
    n = len(p)
    nconv = 2*n-1
    nfft = 1 << (nconv-1).bit_length()
    p_pad = np.zeros(nfft)
    p_pad[:n] = p
    fft = np.fft.rfft(p_pad)
    conv = np.fft.irfft(fft * fft, nfft)[:nconv]

    # compute weights w via softmax
    scaled = alpha * conv
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / exp_scaled.sum()

    # compute gradient using convolution with reversed w
    w_rev = w[::-1]
    # pad w_rev to nfft
    w_pad = np.zeros(nfft)
    w_pad[:nconv] = w_rev
    fft_w = np.fft.rfft(w_pad)
    grad_full = np.fft.irfft(fft * fft_w, nfft)[:n + nconv - 1] # length n + nconv -1 = 3n -2

    offset = nconv - 1  # 2n-2
    grad = 2 * grad_full[offset - np.arange(n)]
    return conv, grad

def project_to_simplex(v):
    # v is array of shape (n,)
    v = np.asarray(v, dtype=float)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_conv_and_grad(p, alpha):
    n = len(p)
    nconv = 2 * n - 1
    nfft = 1 << (nconv - 1).bit_length()
    p_pad = np.zeros(nfft)
    p_pad[:n] = p
    fft_p = np.fft.rfft(p_pad)
    conv = np.fft.irfft(fft_p * fft_p, nfft)[:nconv]
    # compute softmax weights
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)
    # gradient using convolution with reversed w
    w_rev = w[::-1]
    w_pad = np.zeros(nfft)
    w_pad[:nconv] = w_rev
    fft_w = np.fft.rfft(w_pad)
    grad_full = np.fft.irfft(fft_p * fft_w, nfft)[:n + nconv - 1]  # length 3n-2
    offset = nconv - 1
    grad = 2.0 * grad_full[offset - np.arange(n)]
    return conv, grad

"""
Simple fast search for step‑function heights minimizing the evaluation
function 2*n*max(conv(p,p))/sum(p)^2.

The algorithm parametrises the non‑negative sequence as a probability
distribution p (sum = 1).  p is expressed as a soft‑max of unconstrained
logits q, which guarantees non‑negativity and the simplex constraint.
A smooth approximation of the max (log‑sum‑exp) is used and its gradient
is obtained analytically via FFT‑based convolutions.  Stochastic
gradient descent with an adaptive learning rate explores the space,
while the best sequence found is kept.
The method runs until the allocated budget (≈1000 s) expires.
"""

import time
import numpy as np

def _softmax(q: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    max_q = np.max(q)
    exp_q = np.exp(q - max_q)
    return exp_q / np.sum(exp_q)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_and_grad(p: np.ndarray, alpha: float, nfft: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the auto‑convolution of p and the gradient of the smoothed
    max‑operator  (log‑sum‑exp with parameter ``alpha``) with respect to p.

    Returns:
        conv (np.ndarray)  – length ``2*len(p)-1`` array of p*p.
        grad (np.ndarray)  – length ``len(p)`` gradient vector.
    """
    n = p.shape[0]
    nconv = 2 * n - 1

    # ----- convolution p * p -----
    p_pad = np.zeros(nfft, dtype=np.float64)
    p_pad[:n] = p
    fft_p = np.fft.rfft(p_pad)

    conv_full = np.fft.irfft(fft_p * fft_p, nfft)[:nconv]

    # ----- smooth max weights -----
    scaled = alpha * conv_full
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)               # shape (nconv,)

    # ----- gradient of smooth max -----
    w_rev = w[::-1]
    w_pad = np.zeros(nfft, dtype=np.float64)
    w_pad[:nconv] = w_rev
    fft_w = np.fft.rfft(w_pad)

    # cross‑correlation between p and w (implemented via convolution)
    grad_full = np.fft.irfft(fft_p * fft_w, nfft)[: n + nconv - 1]

    offset = nconv - 1                # equals 2*n-2
    grad = 2.0 * grad_full[offset - np.arange(n)]

    return conv_full, grad

    The function runs for at most ``budget_s`` seconds and returns the
    best sequence (list of floats) found.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 2.0   # small safety margin

    np.random.seed(seed)

    # ---- initialise probability vector p (sum = 1) ----

        else:
            p = init / total
    else:
        n = np.random.randint(500, 1500)
        p = np.random.dirichlet(np.ones(n))

    n = p.shape[0]                     # length of the sequence
    nconv = 2 * n - 1
    nfft = 1 << (nconv - 1).bit_length()   # next power of two for FFT

    # log‑itensity parameters (soft‑max gives p)
    q = np.log(p + 1e-15)

    # best solution tracking
    best_p = p.copy()
    best_score = 2.0 * n * np.max(np.convolve(p, p))
    print(f"Start – length {n}, initial score {best_score:.6f}")

    # optimisation hyper‑parameters
    lr = 0.5                               # learning rate
    lr_up = 1.02
    lr_down = 0.5
    min_lr = 1e-6

    alpha_max = 60.0
    alpha_inc = 0.1
    iteration = 0

        conv, grad_p = _conv_and_grad(p, alpha, nfft)

        # keep the best found so far

        dot = np.dot(p, grad_p)

                q += noise

                p = _softmax(q)

    # ---- return best sequence, scaled to sqrt(2n) for aesthetic ----
    scale = np.sqrt(2.0 * n)
    best_seq = (best_p * scale).tolist()
    final_score = evaluate_sequence(best_seq)
    print(f"Finished after {iteration} iterations. Best score {best_score:.6f}, "

"""
Fast stochastic search for a step‑function (non‑negative heights) that
minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

where `conv` is the discrete convolution.  The value is invariant under
scaling, therefore we may restrict ourselves to probability vectors
p (p_i ≥ 0, Σp_i = 1) and minimise `2 * n * max(conv(p, p))`.

Algorithm
---------

* Parameterise p as a soft‑max of unconstrained logits `q`:
      p_i = exp(q_i) / Σ_j exp(q_j)
  This automatically enforces the simplex constraints.

* Approximate the non‑smooth maximum with a smooth log‑sum‑exp:
      M_α(p) = (1/α) log Σ_k exp(α·(p*p)[k])
  for a large smoothing parameter α.  Its gradient w.r.t. p can be
  written in closed form and is computed with FFT‑based convolutions,
  which makes each iteration O(n log n).

* Perform stochastic gradient descent on the logits `q`.  The learning
  rate is adapted (increase on successful steps, decrease otherwise)
  and α is slowly raised so that the optimisation finally targets the
  true maximum.

* Keep the best sequence seen so far and occasionally inject a small
  random perturbation to escape plateaus.

The search runs until the allocated time budget (≈ 1000 s) expires and
returns the best sequence found, scaled by √(2 n) only for aesthetic
reasons – scaling does not change the objective value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _softmax(q: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(q)
    e = np.exp(q - m)
    return e / np.sum(e)


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_and_grad(p: np.ndarray, alpha: float, nfft: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the auto‑convolution of a probability vector `p` and the gradient
    of the smoothed max (log‑sum‑exp with parameter `alpha`) w.r.t. `p`.

    Returns
    -------
    conv : np.ndarray
        Length `2*len(p)-1` array containing (p * p).
    grad : np.ndarray
        Length `len(p)` gradient vector.
    """
    n = p.shape[0]
    nconv = 2 * n - 1

    # ----- p * p -------------------------------------------------------
    p_pad = np.zeros(nfft, dtype=np.float64)
    p_pad[:n] = p
    fft_p = np.fft.rfft(p_pad)

    conv_full = np.fft.irfft(fft_p * fft_p, nfft)[:nconv]

    # ----- smooth‑max weights (log‑sum‑exp) ---------------------------
    scaled = alpha * conv_full
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)          # shape (nconv,)

    # ----- gradient of the smooth max -------------------------------
    # cross‑correlation between p and w (implemented via convolution)
    w_rev = w[::-1]
    w_pad = np.zeros(nfft, dtype=np.float64)
    w_pad[:nconv] = w_rev
    fft_w = np.fft.rfft(w_pad)

    grad_full = np.fft.irfft(fft_p * fft_w, nfft)[: n + nconv - 1]

    offset = nconv - 1                # = 2*n-2
    grad = 2.0 * grad_full[offset - np.arange(n)]

    return conv_full, grad
