def propose_candidate(seed=123, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1

    # Starting sequence: uniform of length n, n = 3000
    n = 3000
    best_seq = np.ones(n, dtype=float) # sum = n
    # ensure within bound of 1000; max values zero? fine.
    best_score = evaluate_sequence(best_seq.tolist())
    
    current = best_seq.copy()
    # Precompute target sum sqrt(2*n)
    target_sum = np.sqrt(2 * n)
    
    # helper functions
    def normalize(v):
        s = np.sum(v)
        if s == 0:
            return v
        return v * target_sum / s
    
    # compute conv using FFT
    def conv_full(v):
        # length needed 2*n-1
        fft_len = 2 * n - 1
        # next power of 2 for speed?
        N = 1 << (fft_len - 1).bit_length()
        fv = np.fft.rfft(v, n=N)
        prod = fv * fv
        conv = np.fft.irfft(prod, n=N)[:2*n-1]
        return conv
    
    # main loop
    step = 0.5  # initial step size for gradient updates
    while time.time() < deadline:
        # Compute convolution and max
        c = conv_full(current)
        max_idx = int(np.argmax(c))
        max_val = c[max_idx]
        S = np.sum(current)
        # gradient of M
        gradM = np.zeros_like(current)
        # we can compute vector of a[max_idx - i] plus a[i]
        # Determine i range where j in [0, n)
        # For i from max(0, max_idx-(n-1)) to min(n-1, max_idx)
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        # Use slice view
        js = np.arange(lo, hi+1)
        gradM[js] = current[max_idx - js] + current[js]
        # compute direction
        lam = 2 * max_val / S
        direction = lam - gradM  # ones * lam - gradM (broadcast)
        # apply step
        new = current + step * direction
        # enforce nonnegative
        new = np.maximum(new, 0.0)
        # normalize sum
        new = normalize(new)
        # evaluate
        try:
            score = evaluate_sequence(new.tolist())
        except Exception:
            score = np.inf
        if score < best_score:
            best_score = score
            best_seq = new.copy()
            current = new
            # optionally increase step size a bit
            step = min(step * 1.1, 1.0)
            print(f"Improved score: {best_score:.6f} step {step:.4f}")
        else:
            # no improvement: reduce step size
            step *= 0.5
            # occasionally random restart
            if step < 1e-5:
                # random reinitialize with small perturb
                current = best_seq + 0.01 * np.random.randn(n)
                current = np.maximum(current, 0.0)
                current = normalize(current)
                step = 0.5
        # occasionally run LP to get a better g
        if time.time() < deadline - 5:
            # compute normalized f for LP (as in given code)
            f_norm = normalize(current)  # sum = target_sum
            M = np.max(conv_full(f_norm))
            # Solve LP to get b (g)
            g = solve_convolution_lp(f_norm.tolist(), M)
            if g is not None:
                g = np.array(g)
                g = normalize(g)
                # line search for alpha in [0, 1] 5 points
                alphas = np.linspace(0.0, 1.0, 6)[1:]  # exclude 0
                best_alpha_score = best_score
                best_alpha_seq = current
                for a in alphas:
                    cand = (1 - a) * current + a * g
                    cand = normalize(cand)
                    try:
                        cand_score = evaluate_sequence(cand.tolist())
                    except Exception:
                        cand_score = np.inf
                    if cand_score < best_alpha_score:
                        best_alpha_score = cand_score
                        best_alpha_seq = cand
                if best_alpha_score < best_score:
                    best_score = best_alpha_score
                    best_seq = best_alpha_seq.copy()
                    current = best_alpha_seq.copy()
                    print(f"LP blend improved score: {best_score:.6f}")
    
    return best_seq.tolist()

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    # Choose length
    n = min(5000, len(height_sequence_1) if 'height_sequence_1' in globals() else 5000)
    
    # define target sum
    target_sum = np.sqrt(2 * n)
    
    # helper functions...
    
    best_seq = None
    best_score = np.inf
    
    # number of restarts
    restarts = 5
    while time.time() < deadline and restarts > 0:
        # initialize
        seq = np.ones(n)
        # random perturb initial to break symmetry
        for _ in range(10):
            i = np.random.randint(n)
            delta = np.random.randn() * 0.5
            seq[i] = max(seq[i] + delta, 0.0)
        seq = normalize(seq)
        curr_score = eval_est(seq)
        # local search:
        step = 0.2
        no_improve = 0
        while time.time() < deadline:
            # gradient step
            c = conv_full(seq)
            k_max = int(np.argmax(c))
            M = c[k_max]
            S = np.sum(seq)
            gradM = np.zeros(n)
            lo = max(0, k_max - (n-1))
            hi = min(n-1, k_max)
            js = np.arange(lo, hi+1)
            gradM[js] = seq[k_max - js] + seq[js]
            lam = 2 * M / S
            direction = lam - gradM
            # apply step
            new_seq = seq + step * direction
            # random jitter to avoid plateau
            if np.random.rand() < 0.1:
                j = np.random.randint(n)
                new_seq[j] += (np.random.rand() - 0.5) * step
            new_seq = np.maximum(new_seq, 0.0)
            new_seq = normalize(new_seq)
            est = eval_est(new_seq)
            if est < curr_score - 1e-9:
                seq = new_seq
                curr_score = est
                no_improve = 0
                step = min(step * 1.05, 1.0)
            else:
                step *= 0.5
                no_improve += 1
                if no_improve > 100:
                    # restart local search with random perturbation
                    break
        # after local search
        exact_score = evaluate_sequence(seq.tolist())
        if exact_score < best_score:
            best_score = exact_score
            best_seq = seq.copy()
        # reduce remaining restarts
        restarts -= 1
    
    # as fallback, if no improvement, return uniform of length n
    if best_seq is None:
        best_seq = np.ones(n)
    
    print(f"Final best score: {best_score}")
    return best_seq.tolist()

"""
A hybrid stochastic/gradient/LP optimizer for the step‑function bound.

The objective to minimise is
    2 * n * max(conv(a,a)) / (sum(a)**2)

where `conv(a,a)` is the full discrete convolution.
The algorithm proceeds in three phases:

1.  A short random “symmetry‑breaking’’ phase – the uniform sequence is a
    stationary point, so we need a small perturbation to get off it.
2.  A gradient descent phase that uses the exact gradient of the current
    maximum convolution entry (piecewise‑linear).  The update direction is
    `λ·1 – ∇M`, where `M` is the current maximum and `λ = 2·M / sum(a)`.
    After each step the sequence is projected back onto the non‑negative
    orthant and renormalised to the target sum `sqrt(2·n)`.  A simple
    back‑tracking line‑search on the step size is performed.
3.  Occasionally (every 20‑30 gradient steps) a linear‑programming step is
    executed.  It finds a direction `g` that increases the total sum while
    keeping the convolution bound ≤ the current maximum.  A short line‑search
    over the blend factor `α` further reduces the score.

All heavy convolution work is done with an FFT (O(n log n)) so the method
scales to several thousand coefficients within the 1000 s budget.
Only the final best candidate is sent to `evaluate_sequence`; intermediate
scores are estimated with the same formula for speed.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Small helper returning the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    The length of the result is 2*len(a)-1.
    """
    n = a.shape[0]
    length = 2 * n - 1
    N = _next_pow2(length)
    fa = np.fft.rfft(a, n=N)
    prod = fa * fa
    full = np.fft.irfft(prod, n=N)[:length]
    return full


def normalize_to_target(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a non‑negative vector to have the given sum."""
    s = np.sum(a)
    if s == 0:
        return a
    return a * (target_sum / s)


def surrogate_score(a: np.ndarray) -> float:
    """Fast estimate of evaluate_sequence (identical formula without clipping)."""
    n = a.shape[0]
    conv = conv_full_fft(a)
    max_c = np.max(conv)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def solve_convolution_lp(f_seq: list, rhs: float):
    """
    Linear programme:
        maximise  sum(b)
        subject to conv(f, b) <= rhs  (for every convolution index)
                   b >= 0
    Returns the optimal b as a NumPy array, or None on failure.
    """
    n = len(f_seq)
    c = -np.ones(n)                     # maximise sum(b)  <=> minimise -sum(b)

    # build convolution constraints “conv(f,b) <= rhs”
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # for each i, if 0 <= k-i < n then f[i] contributes to b[k-i]
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        i_range = np.arange(lo, hi + 1)
        row[i_range] = f_seq[k - i_range]
        rows.append(row)
    A_ub = np.vstack([rows, -np.eye(n)])   # also enforce b >= 0  => -b <= 0
    b_ub = np.hstack([np.full(2 * n - 1, rhs), np.zeros(n)])

    # use the default HiGHS solver – fast and reliable
    res = optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          method='highs',
                          options={'time_limit': 5.0, 'presolve': True})
    if res.success:
        return np.array(res.x, copy=False)
    return None


# ----------------------------------------------------------------------
# Main candidate generator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function sequence with a low value of the evaluation
    function.  The routine runs for at most ``budget_s`` seconds and returns
    the best sequence found (as a plain Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Choose a working length.
    # Prefer the length of the known construction if it exists,
    # otherwise fall back to a moderate size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        start_seq = np.array(height_sequence_1, dtype=float)
    else:
        n = 3000
        start_seq = np.ones(n, dtype=float)   # uniform start

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Normalise the start and compute its initial (surrogate) score.
    # ------------------------------------------------------------------
    current = normalize_to_target(start_seq, target_sum)
    best_seq = current.copy()
    best_score = surrogate_score(best_seq)

    # ------------------------------------------------------------------
    # Small random perturbation to break the uniform symmetry.
    # ------------------------------------------------------------------
    for _ in range(10):
        idx = np.random.randint(n)
        delta = (np.random.rand() - 0.5) * 0.1 * target_sum
        current[idx] = max(current[idx] + delta, 0.0)
    current = normalize_to_target(current, target_sum)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    step_size = 0.3          # initial gradient step scale
    grad_no_improve = 0      # counter for stagnation
    it_since_lp = 0          # number of gradient steps since last LP

    while time.time() < deadline:
        # ---- Gradient descent step ------------------------------------
        conv_full = conv_full_fft(current)
        k_max = int(np.argmax(conv_full))
        M = conv_full[k_max]
        S = np.sum(current)

        # gradient of the *active* convolution entry M
        gradM = np.zeros_like(current)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idxs = np.arange(lo, hi + 1)
        gradM[idxs] = current[k_max - idxs] + current[idxs]

        lam = 2.0 * M / S                         # λ = 2·M / Σa
        direction = lam - gradM                    # λ·1 - ∇M

        # back‑tracking line‑search on the surrogate score
        improved = False
        for _ in range(8):                         # at most a few reductions
            cand = current + step_size * direction
            # occasional tiny random jitter to avoid flat plateaux
            if np.random.rand() < 0.07:
                j = np.random.randint(n)
                cand[j] += (np.random.rand() - 0.5) * step_size
            cand = np.maximum(cand, 0.0)
            cand = normalize_to_target(cand, target_sum)

            cand_score = surrogate_score(cand)
            if cand_score < best_score - 1e-12:
                # Accept the improvement
                current = cand
                best_seq = cand.copy()
                best_score = cand_score
                improved = True
                # enlarge step a little – we are making progress
                step_size = min(step_size * 1.2, 2.5)
                grad_no_improve = 0
                print(f"[{time.time():.1f}] New best (grad) : {best_score:.6f}")
                break
            else:
                # try a smaller step
                step_size *= 0.5

        if not improved:
            grad_no_improve += 1
            if grad_no_improve > 120:   # stagnation -> random perturbation
                idx = np.random.randint(n)
                delta = (np.random.rand() - 0.5) * 0.2 * target_sum
                current[idx] = max(current[idx] + delta, 0.0)
                current = normalize_to_target(current, target_sum)
                step_size = 0.3
                grad_no_improve = 0

        # ---- Periodic LP‑based improvement ----------------------------
        it_since_lp += 1
        if it_since_lp >= 25 and time.time() < deadline - 2:
            # Build the LP with the *current* normalised sequence
            f_norm = normalize_to_target(current, target_sum)
            M_f = np.max(conv_full_fft(f_norm))
            g_vec = solve_convolution_lp(f_norm.tolist(), float(M_f))
            if g_vec is not None:
                g_vec = normalize_to_target(g_vec, target_sum)

                # simple line search on the blend α ∈ (0,1]
                alphas = np.linspace(0.05, 1.0, 10)
                for a in alphas:
                    cand = (1.0 - a) * current + a * g_vec
                    cand = normalize_to_target(cand, target_sum)
                    cand_score = surrogate_score(cand)
                    if cand_score < best_score - 1e-12:
                        current = cand
                        best_seq = cand.copy()
                        best_score = cand_score
                        print(f"[{time.time():.1f}] LP blend α={a:.2f} → {best_score:.6f}")
                        break
            it_since_lp = 0

    # ------------------------------------------------------------------
    # Return the best sequence found (converted to plain Python list)
    # ------------------------------------------------------------------
    print(f"Finished. Best surrogate score: {best_score:.6f}")
    # final, true evaluation – safe to call once at the end
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final true score reported by evaluate_sequence: {final_score:.6f}")
    except Exception as exc:
        print("Final evaluation failed:", exc)
    return best_seq.tolist()
