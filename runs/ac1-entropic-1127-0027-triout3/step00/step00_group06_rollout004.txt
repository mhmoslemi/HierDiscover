def phase_retrieval_sequence(n, iterations=500, seed=None):
    rng = np.random.default_rng(seed)
    # initialize random real non-negative sequence that sums to S.
    a = rng.random(n)
    a = a / a.sum() * np.sqrt(2 * n)  # sum = sqrt(2n)
    # Precompute constant magnitude.
    # We'll aim for magnitude sqrt(C), where C = np.mean(np.abs(np.fft.fft(a))**2) (maybe).
    # Actually for constant magnitude we need |A| = sqrt(const) for all frequencies. Let's compute target magnitude M_target = np.sqrt(np.mean(np.abs(A)**2)).
    # We'll enforce each iteration.
    for it in range(iterations):
        A = np.fft.fft(a)
        # compute average power
        avg_power = np.mean(np.abs(A)**2)
        # target magnitude = sqrt(avg_power)
        target_mag = np.sqrt(avg_power)
        # adjust magnitude to constant while preserving phase
        A = target_mag * np.exp(1j * np.angle(A))
        a = np.real(np.fft.ifft(A))
        # enforce nonnegativity
        a = np.maximum(a, 0)
        # enforce sum constraint
        total = a.sum()
        if total > 0:
            a = a * (np.sqrt(2 * n) / total)
        else:
            a = rng.random(n)
            a = a / a.sum() * np.sqrt(2 * n)
    return a.tolist()

def generate_sequence_by_phase(n, max_iter=500, seed=None):
    rng = np.random.default_rng(seed)
    # initial random nonnegative a.
    a = rng.random(n)
    # compute required sum S = sqrt(2 * n)
    S = np.sqrt(2 * n)
    a = a / a.sum() * S
    best_score = evaluate_sequence(a.tolist())
    best_a = a.copy()
    # iterate
    no_improve = 0
    for it in range(max_iter):
        A = np.fft.fft(a)
        # keep DC magnitude unchanged? Usually we set DC to sum S.
        # Compute current magnitudes
        # Use target magnitude = sqrt(np.mean(np.abs(A)**2))
        target_mag = np.sqrt(np.mean(np.abs(A)**2))
        # Build new A with constant magnitude
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        # IFFT to time domain
        a_new = np.real(np.fft.ifft(A_new))
        # Clip negatives
        a_new = np.maximum(a_new, 0)
        # Rescale to sum S
        total = a_new.sum()
        if total > 0:
            a_new = a_new * (S / total)
        else:
            a_new = rng.random(n)
            a_new = a_new / a_new.sum() * S
        # evaluate
        score = evaluate_sequence(a_new.tolist())
        if score < best_score:
            best_score = score
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        a = a_new
        if no_improve >= 100:
            # early stop if not improving for many iterations
            break
    return best_a.tolist(), best_score

def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    start_time = time.time()
    deadline = start_time + budget_s - 2
    best_seq = None
    best_score = float('inf')
    rng = np.random.default_rng(seed)
    lengths = [256, 512, 1024, 2048]  # choose based on time budget
    while time.time() < deadline:
        n = int(rng.choice(lengths))
        max_iter = max(300, n // 4)  # adapt
        seq, score = generate_sequence_by_phase(n, max_iter=max_iter, seed=int(rng.integers(0, 2**31-1)))
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f"New best {best_score:.6f} with length {n}")
        # optionally, try LP refinement using get_good_direction_to_move_into if best_seq not None
        if best_seq is not None:
            new_seq = get_good_direction_to_move_into(best_seq)
            if new_seq is not None:
                new_score = evaluate_sequence(new_seq)
                if new_score < best_score:
                    best_score = new_score
                    best_seq = new_seq
                    print(f"Improved via LP to {best_score:.6f}")
        # maybe also random perturbation (add small noise), re-evaluate
        if best_seq is not None:
            # random perturbation
            pert_seq = best_seq.copy()
            # choose some indices
            for _ in range(5):
                i = rng.integers(0, len(pert_seq))
                pert_seq[i] = max(0.0, pert_seq[i] + rng.normal() * 0.01 * np.sqrt(2*len(pert_seq)))
            # re-normalize sum
            total = sum(pert_seq)
            if total > 0:
                pert_seq = [x * (np.sqrt(2*len(pert_seq))/total) for x in pert_seq]
            pert_score = evaluate_sequence(pert_seq)
            if pert_score < best_score:
                best_score = pert_score
                best_seq = pert_seq
                print(f"Improved via perturbation to {best_score:.6f}")
    return best_seq

"""
Propose a low‑value step‑function using a phase‑retrieval style projection:
1. Randomly initialise a non‑negative sequence of a chosen length n with
   the required L1‑norm √(2 n) (the evaluation function is scale‑invariant).
2. Iteratively enforce two constraints:
   * Constant magnitude of the discrete Fourier transform (makes the
     self‑convolution close to flat).
   * Non‑negativity in the time domain.
   After each iteration we clip negatives and renormalise the sum.
   The best sequence seen during the iterations is kept.
3. Run many independent restarts with several lengths (256…2048 and the
   length of the provided `height_sequence_1`) while the overall time
   budget (<1000 s) permits.
4. Occasionally improve the current best candidate with a tiny random
   perturbation and with a linear‑programming direction (the LP from the
   reference code).  The best sequence found before the timeout is
   returned.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# Linear‑programming helper (re‑used from the reference code)
def solve_convolution_lp(f_sequence, rhs):
    """Solve  max Σ b_j   s.t.  (f * b)_k ≤ rhs,  b ≥ 0.
    Returns the optimal b as a NumPy array or None if LP failed."""
    n = len(f_sequence)
    c = -np.ones(n)                               # maximise Σ b_j  → minimise −Σ b_j
    # Build convolution constraints: for each k the dot‑product of b with
    # the shifted f gives (f*b)_k.
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_sequence[i]
        a_ub.append(row)
        b_ub.append(rhs)
    a_ub = np.vstack([a_ub, -np.eye(n)])          # also enforce b ≥ 0 via –b ≤ 0
    b_ub = np.hstack([b_ub, np.zeros(n)])
    # Solve with the default HiGHS solver
    res = optimize.linprog(c, A_ub=a_ub, b_ub=b_ub,
                           bounds=[(0, None)] * n,
                           method='highs')
    if res.success:
        return res.x
    return None


def get_good_direction_to_move_into(sequence):
    """Return a new sequence obtained by moving a small step towards the LP
    direction that increases Σ b while keeping the convolution bound."""
    n = len(sequence)
    sum_seq = np.sum(sequence)
    if sum_seq < 1e-12:        # avoid division by zero
        return None
    # Normalise to the standard L1‑norm √(2 n)
    norm_seq = np.array(sequence) * np.sqrt(2 * n) / sum_seq
    rhs = np.max(np.convolve(norm_seq, norm_seq))
    g_fun = solve_convolution_lp(norm_seq, rhs)
    if g_fun is None:
        return None
    sum_g = np.sum(g_fun)
    if sum_g == 0:
        return None
    norm_g = g_fun * np.sqrt(2 * n) / sum_g
    t = 0.01                         # small mixing factor
    new_seq = (1 - t) * np.array(sequence) + t * norm_g
    return new_seq.tolist()


# ----------------------------------------------------------------------
# Phase‑retrieval style generator
def generate_sequence_phase(n, max_iter=500, seed=None):
    """
    Produce a non‑negative sequence of length n whose self‑convolution is
    close to constant.  The algorithm alternates between:
      * forcing the DFT magnitude to be constant,
      * clipping to non‑negative values,
      * renormalising the L1‑norm to √(2 n).

    Returns (best_sequence, best_score) observed during the iterations.
    """
    rng = np.random.default_rng(seed)
    target_sum = np.sqrt(2 * n)

    # Random start, then enforce the target L1‑norm
    a = rng.random(n)
    a = a / a.sum() * target_sum

    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    no_improve = 0

    for it in range(max_iter):
        # DFT of the current sequence
        A = np.fft.fft(a)

        # Desired magnitude: the root‑mean‑square over all frequencies
        target_mag = np.sqrt(np.mean(np.abs(A) ** 2))

        # Replace magnitude while keeping phase
        A = target_mag * np.exp(1j * np.angle(A))

        # Back to time domain, enforce non‑negativity
        a_new = np.real(np.fft.ifft(A))
        a_new = np.maximum(a_new, 0.0)

        # Renormalise the L1‑norm (scale invariance of the objective)
        total = a_new.sum()
        if total > 0:
            a_new = a_new * (target_sum / total)
        else:
            # Very unlikely, but safeguard: fall back to a fresh random start
            a_new = rng.random(n)
            a_new = a_new / a_new.sum() * target_sum

        # Gentle mixing to stabilise the dynamics
        mix = 0.8
        a = (1 - mix) * a + mix * a_new

        # Evaluate
        score = evaluate_sequence(a.tolist())
        if score < best_score:
            best_score = score
            best_seq = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Early stop if we have stagnated for a while
        if no_improve > 50:
            break

    return best_seq.tolist(), best_score


# ----------------------------------------------------------------------
def propose_candidate(seed=0, budget_s=1000, **kwargs):

    Main entry point for the optimizer.

    Uses many random restarts of the projection algorithm (different lengths,
    different seeds) inside the allotted time budget, keeps the best sequence
    found, and occasionally tries small random perturbations and an LP‑based
    refinement.
    """
    start_time = time.time()
    deadline = start_time + budget_s - 2      # keep a small safety margin

    rng = np.random.default_rng(seed)

    # Initial best candidate = the provided SOTA sequence (if available)
    best_seq = None
    best_score = float('inf')
    try:
        if height_sequence_1:                # global from the surrounding harness
            best_seq = list(height_sequence_1)
            best_score = evaluate_sequence(best_seq)

    except Exception:
        best_seq = None
        best_score = float('inf')

    # Lengths to explore – smaller lengths are cheap and often give good
    # improvements fast; we also include the length of the known sequence.
    lengths = [256, 512, 1024, 2048]
    try:
        if height_sequence_1:
            lengths.append(len(height_sequence_1))
    except Exception:
        pass

    # Main search loop
    while time.time() < deadline:
        # --------------------------------------------------------------
        # 1) Random restart of the phase‑retrieval projection
        n = int(rng.choice(lengths))
        max_iter = max(200, n // 5)          # a few hundred iterations are usually enough
        seq, score = generate_sequence_phase(
            n,
            max_iter=max_iter,
            seed=int(rng.integers(1, 2**31 - 1))
        )
        if score < best_score:
            best_score = score
            best_seq = seq
            print(f'New best score {best_score:.6f} (len={n})')

        # --------------------------------------------------------------

        a = [np.sqrt(2 * n) / n] * n
        best_seq = a
        print('Fallback to uniform sequence.')
