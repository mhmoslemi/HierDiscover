def fast_evaluate(seq):
    n = len(seq)
    # Clip negative to zero
    arr = np.maximum(seq, 0)
    arr = np.minimum(arr, 1000.0)
    sum_a = np.sum(arr)
    if sum_a < 0.01:
        return np.inf
    # Compute convolution using FFT
    L = 1 << (2*n - 1).bit_length()
    fft = np.fft.rfft(arr, n=L)
    conv = np.fft.irfft(fft * fft, n=L)
    max_b = np.max(conv[:2*n-1])  # only relevant part
    return 2.0 * n * max_b / (sum_a * sum_a)
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import numpy as np, time, math, random
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # set length
    n = 2000  # can be adjusted based on time
    # Initialize uniform
    seq = np.full(n, 1.0 / n, dtype=np.float64)
    # Evaluate
    def fast_score(arr):
        s = arr.sum()
        if s < 0.01:
            return np.inf
        # compute convolution via FFT
        L = 1 << ((2 * n - 1).bit_length())
        fft = np.fft.rfft(arr, n=L)
        conv = np.fft.irfft(fft * fft, n=L)[:2*n-1]
        max_b = conv.max()
        return 2.0 * n * max_b / (s * s)
    cur_seq = seq.copy()
    cur_score = fast_score(cur_seq)
    best_seq = cur_seq.copy()
    best_score = cur_score
    # Temperature schedule
    temp0 = 1.0
    temp = temp0
    it = 0
    while time.time() < deadline:
        # random move
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random fraction
        frac = np.random.uniform(-0.3, 0.3)  # up to 30% of a_i
        delta = cur_seq[i] * frac
        new_ai = cur_seq[i] + delta
        new_aj = cur_seq[j] - delta
        if new_ai < 0 or new_aj < 0:
            continue
        new_seq = cur_seq.copy()
        new_seq[i] = new_ai
        new_seq[j] = new_aj
        # keep other entries unchanged
        new_score = fast_score(new_seq)
        # accept if better or with probability
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Simulated annealing
            prob = math.exp(-(new_score - cur_score) / (temp + 1e-12))
            if np.random.rand() < prob:
                accept = True
        if accept:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f'Iter {it}: new best score {best_score:.6f}')
        # temperature cool down
        it += 1
        temp = temp0 * (0.9995 ** it)
        # occasional random restart
        if it % 10000 == 0:
            # small shuffle
            cur_seq = np.random.dirichlet(np.ones(n))
            cur_score = fast_score(cur_seq)
    # Return best_seq as list
    return best_seq.tolist()

"""
Propose a low‑value step‑function using simulated‑annealing.

The evaluation function is            2·n·max(a∗a) / (∑a)² .
For a fixed length n the scale of the vector does not matter, therefore the
search works with a probability vector (∑a = 1) and tries to minimise the
maximum entry of the self‑convolution.  The algorithm starts from a smoothly
decreasing distribution (1/(i+1) normalised) which is already better than the
uniform step and then performs a temperature‑controlled random walk that moves
mass between two coordinates while keeping the total mass constant.  The fast
convolution is computed with an FFT (O(n log n)) so thousands of candidates can
be evaluated inside the time budget.  The best sequence found is returned as a
plain Python list – it will be judged by the supplied `evaluate_sequence`
function.
"""

import time
import math
import random
import numpy as np

def _fast_score(seq: np.ndarray) -> float:
    """
    Fast replacement for evaluate_sequence.
    Works on a non‑negative NumPy vector that already respects the clipping
    limits (0 ≤ x ≤ 1000).  Returns the same value as evaluate_sequence.
    """
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    n = seq.size
    # Length for FFT – next power of two ≥ 2n‑1
    L = 1 << ((2 * n - 1).bit_length())
    fft = np.fft.rfft(seq, n=L)
    conv = np.fft.irfft(fft * fft, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function with a low evaluation score.
    The routine runs a simple simulated‑annealing loop for as long as time
    permits (budget_s seconds) and returns the best sequence discovered.
    """
    # -----------------------------------------------------------------
    # 1.  Deterministic seeding & time limit
    # -----------------------------------------------------------------
    random.seed(seed)
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # keep a tiny margin

    # -----------------------------------------------------------------
    # 2.  Problem size – a few thousand entries give a good trade‑off
    # -----------------------------------------------------------------
    n = 2000                     # length of the step function
    # start from a smooth decreasing distribution (better than uniform)
    init = 1.0 / (np.arange(1, n + 1) ** 0.7)   # exponent 0.7 works well
    init = init / init.sum()                    # normalise to unit mass
    cur = init.astype(np.float64, copy=True)

    cur_score = _fast_score(cur)
    best = cur.copy()
    best_score = cur_score

    print(f"initial score (should be ≤ 2.0) : {best_score:.6f}")

    # -----------------------------------------------------------------
    # 3.  Simulated‑annealing parameters
    # -----------------------------------------------------------------
    T0 = 0.5               # initial temperature
    decay = 0.9996         # exponential cooling factor per iteration
    temperature = T0
    it = 0

    # -----------------------------------------------------------------
    # 4.  Main optimisation loop
    # -----------------------------------------------------------------
    while time.time() < deadline:
        # pick two distinct indices and propose a mass transfer
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue

        # transfer a random fraction (up‑to 30 %) of the mass at i to j
        frac = np.random.uniform(-0.3, 0.3)
        delta = cur[i] * frac
        new_i = cur[i] + delta
        new_j = cur[j] - delta
        if new_i < 0.0 or new_j < 0.0:
            continue

        # build the candidate vector (all other entries unchanged)
        cand = cur.copy()
        cand[i] = new_i
        cand[j] = new_j

        cand_score = _fast_score(cand)

        # Acceptance test (Metropolis criterion)
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # probability of accepting a worse move
            prob = math.exp(-(cand_score - cur_score) / (temperature + 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur = cand
            cur_score = cand_score
            # keep the best ever seen
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"Iter {it:07d} – new best score: {best_score:.6f}")

        # temperature schedule
        temperature *= decay
        it += 1

        # occasional small “shake‑up” to avoid stagnation
        if it % 20000 == 0:
            # re‑normalise a tiny random perturbation
            perturb = np.random.exponential(scale=1.0, size=n)
            perturb = perturb / perturb.sum()
            cur = 0.99 * cur + 0.01 * perturb
            cur = cur / cur.sum()
            cur_score = _fast_score(cur)

    # -----------------------------------------------------------------
    # 5.  Return the best sequence found (plain Python list)
    # -----------------------------------------------------------------
    print(f"finished – best score obtained: {best_score:.6f}")
    return best.tolist()
