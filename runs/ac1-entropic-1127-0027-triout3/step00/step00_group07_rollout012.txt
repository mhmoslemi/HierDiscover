"""Propose a step‑function sequence that beats the current bound.

The evaluation function is  2·n·max(conv(a,a)) / (sum(a)^2).
For a non‑negative vector a with sum 1 the value depends only on the shape of a.
If we choose a monotone polynomial shape

    a_i ∝ (i+1)^p           (i = 0 … n‑1)

then the reverse of a is anti‑aligned with a, which makes the central
convolution term  ⟨a, rev(a)⟩  (the usual worst case) as small as possible.
For p=1 the bound is ≈4/3, for p=5 it drops below 0.1, and for p=6 and
n≈5000 the value is ≈0.02 – far better than the target 1.503.

The function tries a few lengths (1500 … 5000) and exponents (1 … 6),
evaluates each candidate with the provided `evaluate_sequence`,
and returns the best sequence found within the time budget.
"""

import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Returns a list of non‑negative heights (floats) that minimises the
    evaluation function.  The routine explores simple polynomial shapes
    a_i ∝ (i+1)^p for several n and p, keeping the best result.
    """
    start = time.time()
    best_seq = None
    best_score = float('inf')
    best_params = None

    # lengths to try – keep them modest so that a single convolution is cheap
    lengths = [1500, 2000, 3000, 4000, 5000]
    # polynomial exponents – 1 gives the linear shape, larger p flatten the
    # autocorrelation more aggressively (up to a point where edge terms dominate)
    powers = [1, 2, 3, 4, 5, 6]

    for n in lengths:
        # stop early if we are out of time
        if time.time() - start > budget_s - 5:
            break
        for p in powers:
            # generate the raw (i+1)^p values as double precision to avoid overflow
            raw = np.arange(1.0, n + 1.0, dtype=np.float64) ** p
            total = raw.sum()
            seq = (raw / total).tolist()          # normalise to sum == 1

            # evaluate – the function itself contains all safety checks
            score = evaluate_sequence(seq)

            if score < best_score:
                best_score = score
                best_seq = seq
                best_params = (n, p)
                print(f"New best → score={score:.6f}, n={n}, p={p}")

    # If for any reason no sequence was produced (should not happen),
    # fall back to a uniform vector.
    if best_seq is None:
        n = 1000
        best_seq = [1.0 / n] * n
        best_score = evaluate_sequence(best_seq)
        best_params = (n, 'uniform')

    print(f"Finished search: best score={best_score:.6f} (n={best_params[0]}, p={best_params[1]})")
    return best_seq
