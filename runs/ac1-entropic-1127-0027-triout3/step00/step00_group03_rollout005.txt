def generate_linear_decreasing(length: int) -> list[float]:
    """
    Create a length‑`length` non‑negative step function

        a_i = 2·(length – i) / (length·(length+1))

    The vector sums to 1.  It is the discrete analogue of the linear

    current best 2.0 and the target 1.503).
    """
    denom = length * (length + 1)
    return [(2.0 * (length - i)) / denom for i in range(length)]
