    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5  # margin
    best_seq = None
    best_score = float('inf')
    # Determine lengths to try - maybe list lengths small to moderate
    lengths = [200, 400, 800, 1200, 2000]  # choose some.
    # Parameter loops
    for n in lengths:
        # Uniform
        seq = [1.0] * n
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score; best_seq = seq.copy()
        # arcsine
        seq = arcsine_sequence(n)
        seq = normalize(seq)  # sum = 1
        score = evaluate_sequence(seq)
        if score < best_score:
            best_score = score; best_seq = seq.copy()
        # power law beta values
        for beta in [0.3, 0.5, 0.7, 0.9]:
            seq = power_law_edge(n, beta)
            seq = normalize(seq)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score; best_seq = seq.copy()
        # exponential
        for lam in [0.003,0.01,0.03,0.1]:
            seq = exponential_edge(n, lam)
            seq = normalize(seq)
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score; best_seq = seq.copy()
        # Dirichlet random draws
        for _ in range(5):
            alpha = np.random.uniform(0.05, 2.0)
            seq = list(np.random.dirichlet([alpha]*n))
            score = evaluate_sequence(seq)
            if score < best_score:
                best_score = score; best_seq = seq.copy()
    # Local refinement using projected gradient / random perturbations
    seq = best_seq.copy()
    while time.time() < deadline:
        # compute convolution
        conv = np.convolve(seq, seq)
        k = int(np.argmax(conv))
        # gradient of conv at k
        nlen = len(seq)
        grad = np.zeros(nlen)
        for i in range(nlen):
            j = k - i
            if 0 <= j < nlen:
                grad[i] = seq[j]
        grad = 2.0 * grad
        # project gradient onto tangent of simplex: subtract mean
        grad_proj = grad - np.mean(grad)
        # step size adaptively
        step = 0.01
        new_seq = seq - step * grad_proj
        # ensure nonnegative
        new_seq = np.maximum(new_seq, 0)
        # project onto simplex (sum=1)
        new_seq = project_simplex(new_seq)
        # evaluate
        new_score = evaluate_sequence(new_seq.tolist())
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.tolist()
            seq = new_seq.tolist()
        else:
            # occasional random shake
            if np.random.rand() < 0.01:
                i = np.random.randint(len(seq))
                j = np.random.randint(len(seq))
                eps = np.random.rand() * 0.05
                seq[i] = max(0, seq[i] - eps)
                seq[j] = seq[j] + eps
                seq = project_simplex(seq)
            else:
                seq = new_seq.tolist()
    print("Best score", best_score, "length", len(best_seq))
    return best_seq

def project_simplex(v):
    # algorithm from Duchi et al.
    v = np.asarray(v)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    best_seq = seq[:]

    n = len(seq)
    while time.time() < deadline:
        # determine step magnitude relative to max of seq

            continue
        # evaluate

            best_score = score
            best_seq = new_seq
    return best_seq, best_score

def hill_climbing(seq, deadline):
    n = len(seq)
    best_seq = np.array(seq)
    best_score = evaluate_sequence(best_seq.tolist())
    while time.time() < deadline:
        # select two distinct indices
        i, j = np.random.choice(n, 2, replace=False)
        # move a fraction of the larger one to the smaller one
        if best_seq[i] < 1e-12 and best_seq[j] < 1e-12:
            continue
        # choose step as fraction of source quantity
        if best_seq[i] > best_seq[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        amount = np.random.rand() * 0.1 * best_seq[src]  # up to 10% of source
        if amount <= 0:
            continue
        new_seq = best_seq.copy()
        new_seq[src] -= amount
        new_seq[dst] += amount
        # ensure non-negative
        if new_seq[src] < 0:
            continue
        score = evaluate_sequence(new_seq.tolist())
        if score < best_score:
            best_score = score
            best_seq = new_seq
    return best_seq.tolist(), best_score

probability vector `p` (sum(p)=1) that makes the maximal
entry of the aperiodic self‑convolution `p*p` as small as
possible.  Uniform weights give a triangular convolution with
peak 1/n and evaluation 2, but the optimum is slightly
better (e.g. for n=4 the vector [0.4,0.2,0.2,0.2] yields
evaluation 1.92).  The optimiser starts from several simple
families (uniform, Dirichlet draws, power‑law “edge‑heavy’’)
and then performs a very cheap random‑pair mass‑transfer
local search while keeping the sum exactly 1.  Because each
move changes only two coordinates the convolution can be
recomputed quickly, and the method works for many different
lengths within the allotted time.

The function returns the best sequence found before the time
budget expires.  All numbers are non‑negative; the final
sequence is a plain Python list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------

def project_simplex(v):
    """Project a vector onto the probability simplex (sum=1, v_i>=0)."""
    v = np.asarray(v, dtype=float)
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    # Duchi et al. "Efficient projections onto the l1‑ball"
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w


def uniform_sequence(n):
    """Uniform probability vector of length n (sum = 1)."""
    return np.full(n, 1.0 / n)


def power_law_edge(n, beta):
    """Edge‑heavy shape: p_i ∝ (i+1)^{-beta} + (n-i)^{-beta}."""
    idx = np.arange(n)
    w = (idx + 1) ** (-beta) + (n - idx) ** (-beta)
    return w / w.sum()


def exponential_edge(n, lam):
    """Edge‑heavy shape with exponential decay from both ends."""
    idx = np.arange(n)
    w = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    return w / w.sum()


def arcsine_sequence(n):
    """Discrete arcsine‑like shape: p_i ∝ 1 / sqrt((i+0.5)*(n-i-0.5))."""
    idx = np.arange(n) + 0.5
    w = 1.0 / np.sqrt(idx * (n - idx))
    return w / w.sum()


def random_dirichlet(n):
    """Draw a Dirichlet vector with a random concentration parameter."""
    alpha = np.random.uniform(0.05, 5.0)          # wide range: sparse ↔ uniform
    return np.random.dirichlet([alpha] * n)


def random_mixture(n):
    """Mixture of a uniform vector and a random Dirichlet draw."""
    u = uniform_sequence(n)
    d = random_dirichlet(n)
    t = np.random.rand()                       # mixing weight
    return (t * u + (1 - t) * d) / (t + (1 - t))


# --------------------------------------------------------------
# Local search primitive
# --------------------------------------------------------------

def hill_climb(seq, deadline):
    """
    Small random mass‑transfer hill climbing.
    The sum of the vector stays exactly 1, all entries stay ≥0.
    """
    a = np.array(seq, dtype=float)
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    n = len(a)

    # pre‑compute the index list once for speed
    indices = np.arange(n)

    while time.time() < deadline:
        # pick two distinct indices uniformly
        i, j = np.random.choice(indices, size=2, replace=False)

        # move a small fraction from the larger entry to the smaller one
        if best[i] > best[j]:
            src, dst = i, j
        else:
            src, dst = j, i

        # step size = a random fraction (up to 5 %) of the source mass
        step = np.random.rand() * 0.05 * best[src]
        if step <= 0:
            continue

        cand = best.copy()
        cand[src] -= step
        cand[dst] += step

        # numerical safety – keep exact sum = 1
        cand = project_simplex(cand)

        # evaluate
        try:
            sc = evaluate_sequence(cand.tolist())
        except Exception:
            continue

        if sc < best_score:
            best_score = sc
            best = cand
    return best.tolist(), best_score


# --------------------------------------------------------------
# Main entry point
# --------------------------------------------------------------

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that
    minimizes the provided `evaluate_sequence` function.
    The routine tries a handful of constructive families on several
    lengths, then refines the current best candidate with a cheap
    stochastic hill‑climber.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2  # a small safety margin

    # ------------------------------------------------------------------
    # 1.  Initialise a pool of candidate sequences of various lengths.
    # ------------------------------------------------------------------
    lengths = [30, 50, 80, 120, 200, 400, 800]      # modest sizes – cheap to evaluate
    best_seq = None
    best_score = float('inf')
    pool = {}                                        # length → (seq, score)

    for n in lengths:
        # a) uniform
        seq = uniform_sequence(n)
        sc = evaluate_sequence(seq.tolist())
        pool[n] = (seq, sc)
        if sc < best_score:
            best_score, best_seq = sc, seq.tolist()

        # b) arcsine‑like
        seq = arcsine_sequence(n)
        sc = evaluate_sequence(seq.tolist())
        if sc < best_score:
            best_score, best_seq = sc, seq.tolist()
        if sc < pool[n][1]:
            pool[n] = (seq, sc)

        # c) a few edge‑heavy power‑law shapes
        for beta in (0.3, 0.5, 0.7, 0.9):
            seq = power_law_edge(n, beta)
            sc = evaluate_sequence(seq.tolist())
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
            if sc < pool[n][1]:
                pool[n] = (seq, sc)

        # d) exponential edge shapes
        for lam in (0.001, 0.005, 0.02, 0.08):
            seq = exponential_edge(n, lam)
            sc = evaluate_sequence(seq.tolist())
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
            if sc < pool[n][1]:
                pool[n] = (seq, sc)

        # e) a few random Dirichlet draws
        for _ in range(3):
            seq = random_dirichlet(n)
            sc = evaluate_sequence(seq.tolist())
            if sc < best_score:
                best_score, best_seq = sc, seq.tolist()
            if sc < pool[n][1]:
                pool[n] = (seq, sc)

    print(f"Initial best score: {best_score:.6f} (length {len(best_seq)})")

    # ------------------------------------------------------------------
    # 2.  Main stochastic search loop – repeatedly improve a random
    #     candidate from the pool (or the global best) by hill‑climbing.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # Pick a length – occasionally try a completely new random draw
        if np.random.rand() < 0.15:
            n = np.random.choice(lengths)
            seq = random_mixture(n)
        else:
            # reuse the current global best, or a random entry from the pool
            if np.random.rand() < 0.5 and best_seq is not None:
                seq = np.array(best_seq, copy=True)
                n = len(seq)
            else:
                n = np.random.choice(lengths)
                seq = pool[n][0].copy()

        # Run a short hill‑climb on this sequence
        sub_deadline = time.time() + min(0.15, deadline - time.time())
        new_seq, new_score = hill_climb(seq.tolist(), sub_deadline)

        # Update global best
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq
            print(f"New global best: {best_score:.6f} (len {len(best_seq)})")

        # Keep the pool entry up‑to‑date (helps future restarts)
        if new_score < pool.get(len(new_seq), (None, float('inf')))[1]:
            pool[len(new_seq)] = (np.array(new_seq), new_score)

    # ------------------------------------------------------------------
    # 3.  Return the best sequence discovered.
    # ------------------------------------------------------------------
    print(f"Final best score: {best_score:.6f} (length {len(best_seq)})")
    # evaluate_sequence expects a Python list of floats
    return [float(x) for x in best_seq]
