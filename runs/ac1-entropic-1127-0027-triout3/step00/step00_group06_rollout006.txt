"""
Propose a candidate sequence of non‑negative heights that minimizes the
evaluation function

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm works on a fixed length `n` (taken from the global
`height_sequence_1` if it exists).  It normalises the sequence to unit sum,
then iteratively reduces the maximum entry of the self‑convolution using a
smooth approximation of the max (log‑sum‑exp) and an exponentiated‑gradient
(update in the simplex).  The smoothness parameter `alpha` is annealed so
that early iterations move the distribution globally while later iterations
focus on the true maximum.  Multiple random restarts are performed until the
time budget expires.  The best sequence seen is returned as a plain Python
list (required by `evaluate_sequence`).

The routine uses FFT‑based convolutions (real FFT) so each iteration is
approximately O(n log n) and easily fits into the 1000 s budget for
n≈5000.
"""

import time
import numpy as np

def _fft_convolve(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """Convolution of two real vectors using a real FFT of length `size`."""
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    # The full convolution length is len(a) + len(b) - 1
    return conv[: len(a) + len(b) - 1]


def _gradient_via_fft(a: np.ndarray, w: np.ndarray, size: int) -> np.ndarray:
    """
    Compute the gradient of  Σ_k w_k * (a * a)_k  w.r.t. `a`.
    The gradient entry i is   Σ_k w_k * a_{k-i}  (zero when out of range).
    This is exactly the sliding dot‑product of `a` with `w`,
    i.e. the convolution of the reversed `a` with `w`.
    """
    # reversed a
    a_rev = a[::-1]
    grad_full = _fft_convolve(a_rev, w, size)
    n = len(a)
    # the needed slice starts at (len(w) - 1) = (2*n-2) and has length n
    start = len(w) - 1
    return grad_full[start:start + n]

    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)    # use the provided construction if possible
    except NameError:
        n = 2000                     # fallback length
    # FFT length – needs to accommodate the longest convolution (a_rev * w)
    fft_len = 1 << int(np.ceil(np.log2(3 * n)))   # >= 3n

    # ------------------------------------------------------------------
    # Helper to compute the current objective (fast, avoids calling evaluate_sequence)
    # ------------------------------------------------------------------
    def _score_from_conv(conv: np.ndarray) -> float:
        # sum of a is 1 (we always keep it normalized)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')
    restart_counter = 0

    # Main outer loop – multiple random restarts until time runs out
    while time.time() < deadline:
        # ----- initialise a new random start ---------------------------------
        a = np.random.exponential(scale=1.0, size=n)
        a /= a.sum()

        # optionally blend with the existing construction on the first restart
        if restart_counter == 0 and 'height_sequence_1' in globals():
            base = np.asarray(height_sequence_1, dtype=float)
            base /= base.sum()
            a = 0.5 * a + 0.5 * base
            a /= a.sum()

        # Hyper‑parameters for this run
        eta = 0.3                     # step size for exponentiated update
        alpha = 0.5                   # initial smoothing (inverse temperature)
        alpha_target = 8.0            # final smoothing strength
        alpha_inc = (alpha_target - alpha) * 0.001   # geometric increase per iteration
        max_iters = 200_000           # safety cap per restart
        no_improve = 0                # counter to trigger occasional random kicks

        for it in range(max_iters):
            if time.time() >= deadline:
                break

            # ----- self‑convolution (FFT) ------------------------------------
            conv = _fft_convolve(a, a, fft_len)          # length 2n‑1
            cur_score = _score_from_conv(conv)

            # ----- keep best -------------------------------------------------
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                no_improve = 0
                print(f"[{restart_counter}] iter {it:6d}  new best = {best_score:.6f}")
                # stop early if target already reached
                if best_score <= 1.503:
                    print("Target reached – exiting early.")
                    return best_seq.tolist()
            else:
                no_improve += 1

            # ----- smooth max (log‑sum‑exp) ----------------------------------
            max_val = conv.max()
            # weights w_k ∝ exp(alpha * (conv_k - max_val))  (numerically stable)
            exp_scaled = np.exp(alpha * (conv - max_val))
            w = exp_scaled / exp_scaled.sum()

            # ----- gradient via FFT (convolution of reversed a with w) -------
            grad = _gradient_via_fft(a, w, fft_len)      # length n, non‑negative

            # ----- exponentiated gradient update ----------------------------
            a = a * np.exp(-eta * grad)

            # ----- projection onto the simplex (non‑negativity + unit sum) ---
            a = np.maximum(a, 1e-12)                     # keep strict positivity
            a /= a.sum()

            # ----- anneal the smoothing parameter ----------------------------
            alpha = min(alpha + alpha_inc, alpha_target)

            # ----- occasional random perturbation to escape flat regions -----
            if no_improve >= 500:
                # add tiny Gaussian noise, re‑normalise
                a += np.random.normal(scale=1e-3, size=n)
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                no_improve = 0

        restart_counter += 1

    # ----------------------------------------------------------------------
    # If nothing better than uniform was found, fall back to uniform
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)
        best_score = _score_from_conv(_fft_convolve(best_seq, best_seq, fft_len))
        print(f"Fallback uniform sequence, score = {best_score:.6f}")

    print(f"Search finished. Best score = {best_score:.6f} after {restart_counter} restarts.")
    return best_seq.tolist()
