"""
Propose a candidate sequence for the convolution‑based objective.

The algorithm starts from the analytically motivated shape
    a_i = 1 / sqrt(i+1)  (i = 0 .. n‑1)
which yields a self‑convolution that is almost constant and already
gives a score ≈ π/2 ≈ 1.57.  To improve this we iteratively solve a
linear program that finds a direction g which maximises the ℓ1‑norm
while keeping the convolution with the current sequence bounded by the
current maximum value M = max(a*a).  A small step towards g reduces the
objective (the denominator grows while the numerator stays almost the
same).  The LP uses a sparse representation so that moderate lengths
(n≈1500) are fast.  Random perturbations are used as a fallback and a
tiny simulated‑annealing acceptance helps to escape plateaus.

The function runs for the allotted time (default 1000 s) and returns the
best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix, vstack, diags


def solve_convolution_lp(f_seq, rhs):
    """
    Linear program:
        maximise   Σ b_j
        subject to  (f * b)(k) ≤ rhs   for all k = 0 … 2n‑2
                    b_j ≥ 0
    Returns the optimal b vector (numpy array) or None on failure.
    """
    n = len(f_seq)

    # Build sparse matrix A_ub such that A_ub @ b ≤ rhs is exactly (f*b) ≤ rhs
    rows = []
    cols = []
    data = []
    # convolution constraints
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f_seq[i])
    A_conv = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    # non‑negativity constraints: -I * b ≤ 0
    A_nonneg = -diags(np.ones(n), format='csr')

    # combine
    A_ub = vstack([A_conv, A_nonneg])

    b_ub = np.concatenate([np.full(2 * n - 1, rhs, dtype=float), np.zeros(n, dtype=float)])

    # maximise sum(b)  → minimise -sum(b)
    c = -np.ones(n, dtype=float)

    # use the HiGHS solver (handles sparse matrices efficiently)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True, 'time_limit': 5.0})

    if res.success:
        return res.x
    return None


def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  The routine runs for at most *budget_s* seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0  # a small safety margin

    # ------------------------------------------------------------------
    # 1) initialise with the 1/√i shape (a classical near‑optimal solution)
    # ------------------------------------------------------------------
    n = 1500                     # moderate length – fast LP, decent quality
    a = np.array([1.0 / np.sqrt(i + 1) for i in range(n)], dtype=float)
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] length={n}, initial score={best_score:.6f}")

    curr_seq = a.copy()
    step = 0.05                 # step size towards the LP direction
    it = 0

    # ------------------------------------------------------------------
    # 2) main optimisation loop
    # ------------------------------------------------------------------
    while time.time() < deadline:

        conv = np.convolve(curr_seq, curr_seq)
        M = np.max(conv)
