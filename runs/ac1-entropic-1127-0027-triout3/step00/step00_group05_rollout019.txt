def generate_gaussian(n, sigma):
    center = (n-1)/2
    i = np.arange(n)
    a = np.exp(-0.5 * ((i - center)/sigma)**2)
    a = np.maximum(a, 0)
    a /= a.sum()
    return a

cosine bump:

```
def generate_cosine_bump(n, w):
    center = (n-1)/2
    i = np.arange(n)
    d = np.abs(i - center)
    mask = d <= w
    a = np.zeros(n)
    a[mask] = np.cos(np.pi * d[mask] / (2 * w))  # shape from 1 down to 0
    a /= a.sum()
    return a
```

def generate_poly(n, alpha):

def generate_triangle(n):
    i = np.arange(n)
    a = np.minimum(i+1, n-i)
    a = a.astype(float)
    a /= a.sum()
    return a

"""
Propose a better coefficient sequence for the evaluation function.

The algorithm:
  * Initialise a non‑negative vector a (sum = 1).  Uniform starts are used,
    optionally taking the provided `height_sequence_1`.
  * Keep the full linear convolution `conv = a * a` (numpy convolve).
  * Perform a simulated‑annealing / random-walk search:
        – pick two indices i,j (i ≠ j) and transfer a small amount δ
          from i to j (δ ≤ 0.5·a[i]).
        – update the convolution exactly with O(n) slice operations
          (including the necessary quadratic correction term).
        – compute the new score  = 2·n·max(conv).
        – accept the move if the score improves or with a probability
          exp(−Δ/temperature) (temperature decays linearly).
        – keep the best sequence ever seen.
  * The move updates preserve the ℓ₁‑norm (sum = 1) and non‑negativity,
    so the candidate is always admissible for `evaluate_sequence`.
  * The search runs until the allotted time expires and finally returns
    the best sequence found.
"""

import numpy as np
import time
import random
import math

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    # Implementation of the algorithm from
    #   Wang & Carreira‑Perpiñán, “Projection onto the probability simplex”
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a coefficient sequence with a low evaluation score.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : int
        Time budget in seconds.
    kwargs :
        Unused, kept for compatibility.

    Returns
    -------
    list[float]
        The best sequence discovered within the time limit.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Initialise the sequence.
    # ------------------------------------------------------------------
    # Use the provided construction if it exists; otherwise start uniform.
    if "height_sequence_1" in globals():
        init = np.array(height_sequence_1, dtype=float)
        # Normalise to unit ℓ₁‑norm (scale does not affect the score)
        if init.sum() == 0:
            init = np.full_like(init, 1.0)
        a = init / init.sum()
    else:
        # default choice – uniform of a moderate length
        a = None

    # If the initial sequence is very long, trim it for speed.
    # A length of ~2000 gives a good trade‑off between quality and cost.
    TARGET_LEN = 2000
    if a is None or a.size < 4:
        n = TARGET_LEN
        a = np.full(n, 1.0 / n)
    else:
        n = a.size
        if n > TARGET_LEN:
            # keep the first TARGET_LEN entries and renormalise
            a = a[:TARGET_LEN]
            a = a / a.sum()
            n = TARGET_LEN
        elif n < TARGET_LEN:
            # pad with small uniform mass to reach TARGET_LEN
            extra = TARGET_LEN - n
            a = np.concatenate([a, np.full(extra, 1.0 / TARGET_LEN)])
            a = a / a.sum()
            n = TARGET_LEN

    # Ensure strict non‑negativity and unit sum
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    # Initial convolution
    conv = np.convolve(a, a)

    # Scoring function (exactly the same as evaluate_sequence for sum=1)
    def _score_from_conv(c):
        return float(2.0 * n * c.max())

    current_score = _score_from_conv(conv)
    best_score = current_score
    best_a = a.copy()

    print(f"Initial length {n}, score {current_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing random walk.
    # ------------------------------------------------------------------
    deadline = time.time() + budget_s - 0.5  # small safety margin
    iteration = 0
    max_iters = 500000  # just a large safety bound
    # temperature will linearly decay to near zero
    T0 = 0.05 * current_score if current_score > 0 else 0.001
    temperature = T0

    # Pre‑compute some constants for speed
    max_transfer_factor = 0.5   # at most half of the source mass is moved

    while time.time() < deadline and iteration < max_iters:
        # ------------------------------------------------------------------
        # Propose a move: transfer δ from i → j
        # ------------------------------------------------------------------
        i, j = random.sample(range(n), 2)
        if a[i] <= 0.0:
            continue
        # Choose δ ≤ max_transfer_factor * a[i]
        delta = np.random.rand() * a[i] * max_transfer_factor
        if delta == 0.0:
            continue

        # Keep copies for possible rejection
        a_old = a.copy()
        conv_old = conv.copy()
        # Store the original vector (used for the exact convolution update)
        a_orig = a.copy()

        # ---------- update convex ----------
        # apply change at i (decrease by delta)
        #   conv[k] += 2*Δ*a_orig[k-i]   for k∈[i, i+n-1]
        #   and quadratic term at 2*i
        conv[i:i + n] -= 2.0 * delta * a_orig
        conv[2 * i] += delta * delta

        # apply change at j (increase by delta)
        conv[j:j + n] += 2.0 * delta * a_orig
        conv[2 * j] += delta * delta

        # cross quadratic term (product a_i * a_j appears twice)
        conv[i + j] -= 2.0 * delta * delta

        # update the vector a
        a[i] -= delta
        a[j] += delta

        # ------------------------------------------------------------------
        # Evaluate the new state
        # ------------------------------------------------------------------
        new_score = _score_from_conv(conv)

        # Acceptance criterion (simulated annealing)
        accept = False
        if new_score < current_score:
            accept = True
        else:
            # probability = exp(−Δ / T)
            if temperature > 0:
                prob = math.exp(-(new_score - current_score) / temperature)
                if random.random() < prob:
                    accept = True

        if accept:
            current_score = new_score
            # keep the best ever seen
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                print(f"Iter {iteration:6d}: new best {best_score:.6f}")
        else:
            # revert to the previous state
            a = a_old
            conv = conv_old

        # --------------------------------------------------------------
        # Temperature schedule (linear decay)
        # --------------------------------------------------------------
        iteration += 1
        # linear cooling
        temperature = T0 * max(0.0, 1.0 - iteration / max_iters)

        # Occasionally recompute the convolution from scratch to avoid drift
        if iteration % 10000 == 0:
            conv = np.convolve(a, a)

    # ------------------------------------------------------------------
    # Return the best sequence found.
    # ------------------------------------------------------------------
    print(f"Finished after {iteration} iterations – best score {best_score:.6f}")
    # Small safety: ensure the sequence is non‑negative and sum > 0.01
    best_a = np.maximum(best_a, 0.0)
    if best_a.sum() < 0.01:
        # rescale to a safe magnitude
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()
    return best_a.tolist()
