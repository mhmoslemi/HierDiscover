    n = len(x) + len(y) - 1
    size = 1 << (n - 1).bit_length()
    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    Z = X * Y
    result = np.fft.irfft(Z, size)
    return result[:n]

import numpy as np
import time

def conv_full_fft(a):
    n = len(a)
    size = 1 << ((2*n - 1) - 1).bit_length()
    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * A, size)[:2*n - 1]
    return conv

def softmax_weights(b, beta):
    maxb = b.max()
    exps = np.exp(beta * (b - maxb))
    return exps / exps.sum()

def corr_valid(p, a):
    n = len(a)
    m = len(p)
    size = 1 << ((m + n - 1) - 1).bit_length()
    P = np.fft.rfft(p, size)
    A = np.fft.rfft(a, size)
    Z = P * np.conj(A)
    corr_full = np.fft.irfft(Z, size)[:m + n - 1]
    start = n - 1
    valid = corr_full[start:start + m - n + 1]
    return valid

def gradient_update(a, beta, eta):
    b = conv_full_fft(a)
    p = softmax_weights(b, beta)
    g = corr_valid(p, a)
    grad = 2.0 * g  # shape (n,)
    # multiplicative update
    a = a * np.exp(-eta * grad)
    # enforce small epsilon
    a = np.maximum(a, 1e-12)
    # renormalize to sum 1
    a = a / a.sum()
    return a, b

   scale‑invariant, this fixes the denominator).
3. Repeatedly:
   * Compute the full self‑convolution using an FFT (fast O(n log n)).
   * Form a soft‑max distribution p over the convolution values
     (β controls the sharpness, it grows slowly).
   * Compute the correlation of p with a (again via FFT).  This
     gives the sub‑gradient of the current maximal convolution bin.
   * Update a multiplicatively:  a_i ← a_i·exp(−η·grad_i) and renormalise.
   * Keep the best sequence seen (lowest evaluation score).
4. Run several independent restarts (different random seeds) within the
   allotted time budget and return the best sequence found.

The method pushes mass away from indices that contribute most to the
current peak of the convolution, thereby reducing the peak while
preserving the total sum.  With enough iterations the convolution becomes
almost flat, driving the evaluation score well below the previous bound
of 2.0 and approaching the theoretical limit 2·n/(2n−1) ≈ 1.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper FFT utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_full_fft(a: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Full self‑convolution of a (length n) using FFT.
    Returns array of length 2*n‑1.
    """
    A = np.fft.rfft(a, fft_size)
    conv = np.fft.irfft(A * A, fft_size)[: 2 * len(a) - 1]
    return conv


def _valid_correlation_fft(p: np.ndarray, a: np.ndarray, fft_size: int) -> np.ndarray:
    """
    Compute the "valid" correlation of p (length 2*n‑1) with a (length n).
    Result length is n and corresponds to
        g_i = Σ_j p_{i+j} * a_j   (0 ≤ i < n)
    """
    # FFT of both sequences (zero‑padded to fft_size)
    P = np.fft.rfft(p, fft_size)
    A = np.fft.rfft(a, fft_size)
    # Correlation in frequency domain: ifft(P * conj(A))
    corr_full = np.fft.irfft(P * np.conj(A), fft_size)
    # Length of full correlation = len(p)+len(a)-1 = 3n‑2
    # The "valid" part starts at index n‑1 and has length n.
    start = len(a) - 1
    return corr_full[start : start + len(a)]


# ------------------------------------------------------------
# Core optimisation step
# ------------------------------------------------------------

def _gradient_step(a: np.ndarray, beta: float, eta: float, fft_size: int):
    """
    Perform one exponentiated‑gradient iteration.
    Returns the updated a and the full convolution array.
    """
    # 1) convolution a*a
    conv = _conv_full_fft(a, fft_size)

    # 2) soft‑max over convolution entries
    maxc = conv.max()
    # shift before exponentiation for numerical stability
    shifted = conv - maxc
    # the exponential never overflows because shifted ≤ 0
    weights = np.exp(beta * shifted)
    weights /= weights.sum()                       # p_k, Σ p = 1

    # 3) correlation p ⋆ a (valid part)
    corr = _valid_correlation_fft(weights, a, fft_size)

    # gradient of max‑convolution w.r.t. a (sub‑gradient when several k tie)
    grad = 2.0 * corr  # each term appears twice in the convolution sum

    # 4) multiplicative update on the simplex
    a = a * np.exp(-eta * grad)
    # keep a strictly positive to avoid division‑by‑zero later
    a = np.maximum(a, 1e-12)
    a /= a.sum()
    return a, conv


# ------------------------------------------------------------
# Single optimisation run (one random start)
# ------------------------------------------------------------

def _run_one_instance(n: int, rng: np.random.Generator,
                      time_limit: float):
    """
    Optimise a sequence of length n for at most `time_limit` seconds.
    Returns (best_sequence, best_score).
    """
    # Initialise a random positive vector (Dirichlet‑like)
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()                     # enforce sum == 1

    # pre‑allocate FFT size that works for both convolution and correlation
    fft_size = _next_pow2(3 * n - 2)

    best_a = a.copy()
    # score = 2 * n * max(conv) because sum(a) == 1
    best_score = 2.0 * n * _conv_full_fft(a, fft_size).max()

    start = time.time()
    it = 0
    # hyper‑parameters – they are annealed during the run
    beta = 2.0          # soft‑max sharpness
    eta = 0.5           # step size

    while time.time() - start < time_limit:
        a, conv = _gradient_step(a, beta, eta, fft_size)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[run] iter={it:5d}  score={cur_score:.6f}  beta={beta:.3f}  eta={eta:.4f}")

        # simple schedule: increase sharpness, slowly decay step size
        if it % 200 == 0 and it > 0:
            beta = min(beta * 1.15, 80.0)
            eta *= 0.94
        it += 1

    return best_a.tolist(), best_score


# ------------------------------------------------------------
# Main entry point required by the evaluator
# ------------------------------------------------------------

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a low‑evaluation step‑function sequence.
    Uses several independent gradient‑descent restarts and returns the
    best sequence found within the allocated time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 2.0   # reserve a couple of seconds

    # -----------------------------------------------------------------
    # Choose the length of the vector.
    # Prefer the length of the previously known construction if it exists.
    # -----------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        print(f"[info] Using length n = {n} (taken from height_sequence_1).")
    else:
        # a reasonable fallback length (must be >1 to make the problem non‑trivial)
        n = 3000
        print(f"[info] No previous sequence – using default length n = {n}.")

    best_overall_seq = None
    best_overall_score = float('inf')
    run_id = 0

    # -----------------------------------------------------------------
    # Main time‑controlled loop: start several optimisation runs.
    # -----------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate at most 25 s per restart (or the remaining time)
        run_time = min(25.0, remaining)

        rng = np.random.default_rng(seed + run_id * 12345)

        # start from the known construction with probability 0.4
        if run_id % 5 == 0 and 'height_sequence_1' in globals():
            init = np.array(height_sequence_1[:n], dtype=float)
            # ensure positivity and normalise
            init = np.maximum(init, 0.0)
            if init.sum() < 1e-12:
                init = rng.exponential(scale=1.0, size=n)
            init /= init.sum()
            a_start = init
            # run a short warm‑up phase with the same gradient routine
            best_seq, best_score = _run_one_instance(
                n, rng, run_time, )
        else:
            best_seq, best_score = _run_one_instance(n, rng, run_time)

        if best_score < best_overall_score:
            best_overall_score = best_score
            best_overall_seq = best_seq
            print(f"[global] New best after run {run_id}: score={best_score:.6f}")

        run_id += 1

    # -----------------------------------------------------------------
    # As a safety net, fall back to a uniform sequence if nothing was found.
    # -----------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = [1.0] * n
        print("[fallback] Returning uniform sequence.")

    # The evaluator will apply its own clipping and scaling, so we simply
    # return the raw floats.
    return best_overall_seq
