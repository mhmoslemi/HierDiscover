def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, random
    np.random.seed(seed)
    # get base sequence length: maybe use length of height_sequence_1 if defined, else 2000
    n = len(height_sequence_1) if 'height_sequence_1' in globals() else 2000
    # initial sequence: random positive numbers
    a = np.random.rand(n)
    a = a / a.sum()  # sum to 1
    best_seq = a.copy()
    best_score = evaluate_sequence(a.tolist())
    start = time.time()
    deadline = start + budget_s - 5
    # parameters
    lr = 0.1
    decay = 0.99
    min_lr = 0.001
    # loop
    while time.time() < deadline:
        # compute convolution
        b = np.convolve(a, a)
        max_idx = int(np.argmax(b))
        max_val = b[max_idx]
        # compute gradient for that max index
        # g[i] = 2 * a[max_idx - i] if index in range else 0
        idxs = np.arange(n)
        rev_idx = max_idx - idxs
        # mask valid idx
        mask = (rev_idx >= 0) & (rev_idx < n)
        g = np.zeros_like(a)
        g[mask] = a[rev_idx[mask]]
        # gradient step
        a_new = a - lr * g
        # ensure non-negative
        a_new = np.maximum(a_new, 0)
        # renormalize
        sum_a = a_new.sum()
        if sum_a < 1e-12:
            # if vanish, reinitialize random
            a_new = np.random.rand(n)
            a_new = a_new / a_new.sum()
        else:
            a_new = a_new / sum_a
        a = a_new
        # evaluate
        try:
            score = evaluate_sequence(a.tolist())
            if score < best_score:
                best_score = score
                best_seq = a.copy()
                # optional print
                print(f"New best {best_score:.6f} at t={time.time()-start:.1f}s")
        except Exception:
            pass
        # adjust learning rate occasionally
        lr = max(min_lr, lr * decay)
    # return best_seq as list
    return best_seq.tolist()

"""
Propose a sequence of non‑negative heights that minimises
    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm mixes two inexpensive heuristics:
*  a projected gradient step that reduces the current maximal
   convolution entry,
*  an LP‑based step (the same idea used in the reference solution)
   that finds a direction with a larger L1‑mass while keeping the
   convolution bounded.

The sequence is always kept on the probability simplex (sum = 1)
so the objective is scale‑invariant.  Several restarts and a
time‑budget guard guarantee that a valid list is returned before the
global deadline (≈1000 s).  The best sequence found is returned.
"""

import time
import random
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the linear program
# ----------------------------------------------------------------------
def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Solve:
        max   sum(b_j)
        s.t.  (f * b)_k <= rhs   for all k = 0 .. 2n-2
              b_j >= 0

    Returns the optimal b (as a 1‑D numpy array) or None if the LP fails.
    """
    n = f_seq.shape[0]
    # objective: maximise sum(b)  → minimise –sum(b)
    c = -np.ones(n)

    # Build the convolution constraints A_ub b <= rhs
    # For each k we need sum_i f_i * b_{k-i} <= rhs.
    rows = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        # i runs over indices where both i and k-i are inside [0,n)
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        # j = k - i  →  i = k - j
        for j in range(j_start, j_end + 1):
            i = k - j
            row[j] = f_seq[i]
        rows.append(row)

    # combine with non‑negativity constraints: -I * b <= 0
    A_ub = np.vstack([np.asarray(rows), -np.eye(n)])
    b_ub = np.hstack([np.full(2 * n - 1, rhs), np.zeros(n)])

    # Use the HiGHS solver (fast and robust)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs')
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Gradient descent step (projected onto the simplex)
# ----------------------------------------------------------------------
def grad_step(seq: np.ndarray, lr: float = 0.1, top_k: int = 1) -> np.ndarray:
    """
    One projected gradient step that tries to reduce the current
    maximal convolution entry.

    seq   : current probability vector (sum = 1, non‑negative)
    lr    : learning rate (step size)
    top_k : number of largest convolution positions to use for the gradient
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)                 # full convolution, length 2n‑1
    # pick the `top_k` indices with largest values
    if top_k == 1:
        ks = [int(conv.argmax())]
    else:
        ks = np.argpartition(conv, -top_k)[-top_k:]

    # accumulate gradient contributions from the selected indices
    grad = np.zeros_like(seq)
    for k in ks:
        rev_idx = k - np.arange(n)               # array of j = k‑i
        mask = (rev_idx >= 0) & (rev_idx < n)
        grad[mask] += seq[rev_idx[mask]]

    # gradient of the max‑entry (ignoring the factor 2, irrelevant for the direction)
    new_seq = seq - lr * grad
    new_seq = np.clip(new_seq, 0.0, None)        # keep non‑negative
    s = new_seq.sum()
    if s < 1e-12:                                # protect against degenerate zero vector
        new_seq = np.random.rand(n)
        s = new_seq.sum()
    return new_seq / s                           # re‑project onto simplex


# ----------------------------------------------------------------------
# LP based improvement step
# ----------------------------------------------------------------------
def lp_step(seq: np.ndarray, mix_t: float = 0.2) -> np.ndarray:
    """
    Perform the LP direction step described in the reference solution:
    1. Compute the current maximal convolution value M.
    2. Solve the LP to maximise the L1‑mass of a new vector g
       while keeping conv(seq, g) ≤ M.
    3. Mix the current seq towards the normalised g.

    Returns the mixed sequence (still on the simplex) or None if the LP fails.
    """
    M = np.max(np.convolve(seq, seq))
    g = solve_convolution_lp(seq, M)
    if g is None:
        return None
    # normalise g to be a probability vector
    if g.sum() < 1e-12:
        return None
    g = g / g.sum()

    # convex combination, then re‑project onto the simplex (exact because both lie on it)
    new_seq = (1.0 - mix_t) * seq + mix_t * g
    # numerical safety: tiny negatives may appear
    new_seq = np.clip(new_seq, 0.0, None)
    new_seq = new_seq / new_seq.sum()
    return new_seq

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a good height sequence within the allotted time.
    The routine alternates cheap gradient steps with occasional LP steps,
    keeps the best sequence seen, and returns it as a plain Python list.
    """
    np.random.seed(seed)
    random.seed(seed)

    # ------------------------------------------------------------------
    # Choose a problem size (large enough to give freedom, small enough to be fast)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2000                       # fallback size

    # ------------------------------------------------------------------
    # Initialise a random probability vector
    # ------------------------------------------------------------------
    a = np.random.rand(n)
    a = a / a.sum()                    # enforce sum = 1  (scale‑invariant)

    best_seq = a.copy()
    try:
        best_score = evaluate_sequence(a.tolist())
    except Exception:
        best_score = float('inf')

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # leave a safety margin

    it = 0
    lr = 0.8
    min_lr = 0.02
    decay = 0.9995                      # slowly shrink learning rate

    while time.time() < deadline:
        # Occasionally try the more expensive LP step
        if it % 250 == 0:
            cand = lp_step(a, mix_t=0.3)
            if cand is not None:
                a = cand
        else:
            #otherwise cheap gradient descent on the worst convolution entry
            a = grad_step(a, lr=lr, top_k=1)
            lr = max(min_lr, lr * decay)   # decay learning rate slowly

        # Evaluate the candidate (protect against accidental NaNs etc.)
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:

            a = np.random.rand(n)
            a = a / a.sum()

            best_seq = a.copy()
            print(f"[{time.time() - start_time:6.1f}s] iter {it:5d} – new best = {best_score:.6f}")

            # early exit if we reach the target bound
            if best_score < 1.503:
                break

        it += 1

    # Return a plain Python list (evaluate_sequence expects that)
    return best_seq.tolist()
