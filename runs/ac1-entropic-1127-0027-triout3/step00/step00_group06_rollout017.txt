                for t in [0.2, 0.5, 0.8]:

                    cand = np.maximum(cand, 0)
                    cand = cand / cand.sum()
                    score = compute_score(cand)
                    if score < best_score:
                        best_score = score
                        best_seq = cand.copy()
                        seq = cand.copy()

        conv = np.convolve(seq, seq)
        k_max = np.argmax(conv)

"""
Propose a candidate sequence that beats the uniform bound (evaluation = 2).

Algorithm
---------
1. Initialise a non‑negative sequence of length *n* (default = length of the
   provided `height_sequence_1`).  The initial sequence is uniform and scaled
   to have sum 1 (the objective is homogeneous, so the absolute scale does not
   matter).

2. Repeatedly improve the sequence while the overall runtime is below the
   provided budget.
   * **Local targeted move** – Find the convolution index *k* where the
     auto‑convolution attains its maximum.  Among the pairs (i, k‑i) that
     contribute to this maximum, select the pair with the largest product
     a_i·a_{k‑i}.  Reduce both entries by a small fraction and redistribute the
     removed mass to two smallest entries.  This directly lowers the dominant
     term of the maximum convolution.
   * **LP‑guided direction** – Occasionally (every few thousand iterations)
     compute a direction using the linear programme from the state‑of‑the‑art
     method.  Mix the current sequence with the LP solution using a line
     search on the mixing coefficient *t* and keep the best mixture.
   * **Cooling / random perturbations** – Simulated‑annealing style acceptance
     allows uphill moves early on; the temperature is slowly decreased.  Random
     perturbations are injected periodically to escape shallow local minima.

3. Keep track of the best sequence (lowest evaluation) found so far and return
   it at the end.

The implementation uses only NumPy, SciPy's `linprog`, and the already‑imported
`evaluate_sequence` for final correctness checks.  No file or network I/O is
performed, and all helper functions are defined at the top level.
"""

import time
import numpy as np
from scipy import optimize

# --- Helper functions ---------------------------------------------------------

def compute_score(seq: np.ndarray) -> float:
    """Fast evaluation of the objective (scale‑invariant)."""
    s = np.sum(seq)
    if s < 1e-8:
        return float("inf")
    conv = np.convolve(seq, seq)
    max_b = np.max(conv)
    return float(2.0 * len(seq) * max_b / (s * s))


def solve_convolution_lp(f_seq: np.ndarray, rhs: float):
    """
    Linear programme used in the SOTA approach:
    maximise sum(b)  subject to  conv(f, b) <= rhs,  b >= 0.
    Returns the optimal b as a NumPy array or ``None`` on failure.
    """
    n = len(f_seq)
    # maximise sum(b)  <=> minimise -sum(b)
    c = -np.ones(n)
    # Build the matrix for the (2n‑1) convolution inequalities.
    a_ub = []
    b_ub = []
    for k in range(2 * n - 1):
        row = np.zeros(n)
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                row[j] = f_seq[i]
        a_ub.append(row)
        b_ub.append(rhs)
    # Non‑negativity constraints:  -b <= 0
    a_ub_nonneg = -np.eye(n)
    b_ub_nonneg = np.zeros(n)
    A = np.vstack([a_ub, a_ub_nonneg])
    B = np.hstack([b_ub, b_ub_nonneg])
    # Small time limit per LP call to avoid stagnation.
    res = optimize.linprog(
        c,
        A_ub=A,
        b_ub=B,
        method="highs",
        options={"presolve": True, "time_limit": 5.0},
    )
    if res.success:
        return res.x
    return None


def lp_direction(seq: np.ndarray):
    """Return a new sequence obtained from the LP direction, or None."""
    n = len(seq)
    total = np.sum(seq)
    if total == 0:
        return None
    # Normalise to the usual scale √(2n); the exact scale does not matter.
    norm_seq = seq * np.sqrt(2.0 * n) / total
    rhs = np.max(np.convolve(norm_seq, norm_seq))
    g = solve_convolution_lp(norm_seq, rhs)
    if g is None:
        return None
    # Rescale back to the original sum (here we keep the sum = 1).
    g = np.maximum(g, 0.0)
    g_sum = np.sum(g)
    if g_sum == 0:
        return None
    return g / g_sum


def local_targeted_move(seq: np.ndarray, step_frac: float = 0.1):
    """
    Reduce the dominant term of the maximum convolution.
    Returns a new sequence (already normalised) or the original if no move
    improves the objective.
    """
    n = len(seq)
    conv = np.convolve(seq, seq)
    k_max = int(np.argmax(conv))

    # Find the pair (i, j) with i+j = k_max that gives the largest product.
    best_i = None
    best_prod = -1.0
    for i in range(n):
        j = k_max - i
        if 0 <= j < n:
            prod = seq[i] * seq[j]
            if prod > best_prod:
                best_prod = prod
                best_i = i
    if best_i is None:
        return seq

    i = best_i
    j = k_max - i

    # Take a small fraction of the smaller entry.
    delta = step_frac * min(seq[i], seq[j])
    if delta <= 0:
        return seq

    new_seq = seq.copy()
    new_seq[i] -= delta
    new_seq[j] -= delta

    # Redistribute the removed mass to the two smallest entries
    # (they are the least likely to increase the max convolution).
    small_idx = np.argsort(new_seq)[:2]
    new_seq[small_idx[0]] += delta
    new_seq[small_idx[1]] += delta

    # Guard against negative values caused by numerical noise.
    new_seq = np.maximum(new_seq, 0.0)
    # Renormalise to keep the sum = 1 (scale‑invariant objective).
    s = np.sum(new_seq)
    if s == 0:
        return seq
    new_seq /= s
    return new_seq


# --- Main entry point ---------------------------------------------------------

def propose_candidate(budget_s: float = 1000, seed: int = 42, **kwargs) -> list:
    """
    Search for a sequence with a low evaluation value.

    Parameters
    ----------
    budget_s : float
        Seconds allocated for the whole search (default 1000).
    seed : int
        Random seed for reproducibility.

    Returns
    -------
    list[float]
        A list of non‑negative heights (the best sequence found).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # a tiny safety margin

    # ----------------------------------------------------------------------
    # Initialise sequence --------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(height_sequence_1, list):
        n = len(height_sequence_1)
        current = np.array(height_sequence_1, dtype=float)
        # Normalise to sum = 1 for a scale‑invariant start.
        current = np.maximum(current, 0.0)
        if current.sum() == 0:
            current = np.full(n, 1.0 / n)
        else:
            current /= current.sum()
    else:
        # Fallback: a modest size (≈2000) uniform start.
        n = 2000
        current = np.full(n, 1.0 / n)

    best_seq = current.copy()
    best_score = compute_score(best_seq)

    print(f"[INIT] length={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Search loop -----------------------------------------------------------
    iteration = 0
    temperature = 1.0
    while time.time() < deadline:
        # ----- local targeted move -----------------------------------------
        cand = local_targeted_move(current, step_frac=0.1)
        cand_score = compute_score(cand)

        # Simulated‑annealing acceptance
        if cand_score < best_score or np.random.rand() < np.exp(
            -(cand_score - best_score) / max(temperature, 1e-8)
        ):
            current = cand
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[{iteration:06d}] new best (local) = {best_score:.6f}")

        # ----- occasional LP guided direction -------------------------------
        if iteration % 2500 == 0 and iteration > 0:
            direction = lp_direction(current)
            if direction is not None:
                # line search on mixing coefficient t
                for t in (0.2, 0.4, 0.6, 0.8):
                    mixed = (1.0 - t) * current + t * direction
                    mixed = np.maximum(mixed, 0.0)
                    mixed_sum = mixed.sum()
                    if mixed_sum == 0:
                        continue
                    mixed /= mixed_sum
                    mixed_score = compute_score(mixed)
                    if mixed_score < best_score:
                        best_score = mixed_score
                        best_seq = mixed.copy()
                        current = mixed
                        print(
                            f"[{iteration:06d}] new best (LP t={t:.2f}) = {best_score:.6f}"
                        )
                        break

        # ----- random perturbations to avoid stagnation --------------------
        if iteration % 1000 == 0 and np.random.rand() < 0.1:
            idx = np.random.randint(n)
            # Gaussian jitter (small compared with typical entries)
            current[idx] = max(0.0, current[idx] + np.random.randn() * 0.005)
            # re‑normalise
            s = current.sum()
            if s == 0:
                current = np.full(n, 1.0 / n)
            else:
                current /= s

        # ----- temperature schedule -----------------------------------------
        iteration += 1
        if iteration % 500 == 0:
            temperature *= 0.97  # exponential cooling

    print(f"[DONE] iterations={iteration}, best score={best_score:.6f}")
    # Return a plain Python list (as required by the evaluator).
    return best_seq.tolist()
