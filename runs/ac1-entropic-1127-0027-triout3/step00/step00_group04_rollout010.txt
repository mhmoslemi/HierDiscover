import numpy as np
import time

def compute_score(mask, n):
    """Return the evaluation score for a zero/one mask.
    mask: 1-D numpy array of 0/1 of length n.
    Returns (score, max_count, m) where score = 2*n*max_count/m^2
    """
    m = int(mask.sum())
    if m == 0:
        return np.inf, None, m
    conv = np.convolve(mask, mask)
    max_count = int(conv.max())
    score = 2.0 * n * max_count / (m * m)
    return score, max_count, m

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Random search for a sparse step function that minimises the evaluation
    score.  The algorithm draws binary masks of varying density, scales the
    non‑zero entries so that the L1 norm equals sqrt(2*n) and keeps the mask
    with the smallest max‑convolution value (the actual score).  Simple
    hill‑climbing swaps are also attempted when time permits.
    """
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 5.0  # keep a safety margin

    best_sequence = None
    best_score = np.inf
    best_mask = None
    best_n = None

    # parameters for the search
    min_n = 1500
    max_n = 3000
    # we will try a few different densities; keep them in a list
    densities = [0.35, 0.45, 0.55, 0.65]

    # --- random sampling phase -------------------------------------------------
    while time.time() < deadline:
        # choose a length and a density at random
        n = int(rng.integers(min_n, max_n + 1))
        p = float(rng.choice(densities))
        # ensure at least one element and at most n-1
        m = max(1, int(p * n))
        if m >= n:
            m = n - 1

        # generate a mask with exactly m ones
        idx = rng.choice(n, size=m, replace=False)
        mask = np.zeros(n, dtype=np.int8)
        mask[idx] = 1

        score, _, _ = compute_score(mask, n)
        if score < best_score:
            best_score = score
            best_mask = mask.copy()
            best_n = n
            w = np.sqrt(2.0 * n) / m
            best_sequence = [float(w) if mask[i] else 0.0 for i in range(n)]
            print(f"[{time.time():.1f}] New best: n={n}, m={m}, density={p:.2f}, score={best_score:.6f}")

        # Occasionally try a tiny local move on the current best mask
        if best_mask is not None and rng.random() < 0.02:
            # attempt a single swap of a 1 and a 0
            n_swap = best_n
            mask_swap = best_mask.copy()
            one_positions = np.flatnonzero(mask_swap)
            zero_positions = np.flatnonzero(1 - mask_swap)
            if len(one_positions) > 0 and len(zero_positions) > 0:
                i = int(rng.choice(one_positions))
                j = int(rng.choice(zero_positions))
                mask_swap[i] = 0
                mask_swap[j] = 1
                m_swap = mask_swap.sum()
                # recompute score for the swapped mask
                score_swap, _, _ = compute_score(mask_swap, n_swap)
                if score_swap < best_score:
                    best_score = score_swap
                    best_mask = mask_swap
                    best_n = n_swap
                    w = np.sqrt(2.0 * n_swap) / m_swap
                    best_sequence = [float(w) if mask_swap[k] else 0.0 for k in range(n_swap)]
                    print(f"[{time.time():.1f}] Improved via swap: n={n_swap}, m={m_swap}, score={best_score:.6f}")

    # fallback to a uniform sequence if nothing was found
    if best_sequence is None:
        best_sequence = [1.0, 1.0]

    return best_sequence

"""
Random sparse step‑function search.

The evaluation score is 2·n·max(b)/sum(a)² where b = a∗a.
Because the code normalises the L¹ norm to √(2·n), the score
reduces to the maximal convolution value max(b).  By using a
binary mask (0/1) and scaling the non‑zero entries so that the
total sum equals √(2·n), the score becomes

    score = 2·n·max_count / m²

where m is the number of non‑zero entries and max_count is the
largest ordered‑pair count produced by the mask (the convolution
of the mask with itself).  For a random mask of density p=m/n this
quantity is close to 1, which is far better than the uniform
baseline (score = 2).  The algorithm repeatedly draws random masks
of several densities, keeps the best one and occasionally performs
single‑swap hill‑climbing steps to improve it.  The whole search
runs within the allotted time budget and finally returns the
corresponding float sequence (non‑negative heights).
"""

import time

def _score_from_mask(mask: np.ndarray, n: int):
    """
    Compute the evaluation score for a 0/1 mask.
    Returns (score, max_count, m) where:
      - m is the number of ones,
      - max_count = max_{k} (mask * mask)[k],
      - score = 2*n*max_count / m**2  (identical to evaluate_sequence).
    """
    m = int(mask.sum())
    if m == 0:
        return np.inf, None, m
    # ordered‑pair counts via linear convolution
    conv = np.convolve(mask, mask)
    max_count = int(conv.max())
    score = 2.0 * n * max_count / (m * m)
    return score, max_count, m

    best_seq = None
    best_score = np.inf
    best_mask = None
    best_n = None

    # search parameters
    min_n = 1500
    max_n = 3000
    # a few densities that work well in practice
    densities = [0.35, 0.45, 0.55, 0.65]

    # ------------------------------------------------------------------
    # Random sampling loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a random length and a density
        n = int(rng.integers(min_n, max_n + 1))
        p = float(rng.choice(densities))

        # number of ones; keep it strictly between 1 and n‑1
        m = max(1, int(p * n))
        if m >= n:
            m = n - 1

        # generate a mask with exactly m ones
        idx = rng.choice(n, size=m, replace=False)
        mask = np.zeros(n, dtype=np.int8)
        mask[idx] = 1

        score, _, _ = _score_from_mask(mask, n)

        if score < best_score:
            best_score = score
            best_mask = mask.copy()
            best_n = n
            w = np.sqrt(2.0 * n) / m          # scaling to meet the normalisation
            best_seq = [float(w) if mask[i] else 0.0 for i in range(n)]
            print(f"[{time.time():.1f}] New best → n={n}, m={m}, density={p:.2f}, score={best_score:.6f}")

        # Occasionally try a tiny local improvement on the current best mask
        if best_mask is not None and rng.random() < 0.02:
            # perform a single swap (1 ↔ 0) and keep it if it improves the score
            one_pos = np.flatnonzero(best_mask)
            zero_pos = np.flatnonzero(1 - best_mask)
            if one_pos.size and zero_pos.size:
                i = int(rng.choice(one_pos))
                j = int(rng.choice(zero_pos))
                trial = best_mask.copy()
                trial[i] = 0
                trial[j] = 1
                trial_score, _, _ = _score_from_mask(trial, best_n)
                if trial_score < best_score:
                    best_score = trial_score
                    best_mask = trial
                    m_trial = trial.sum()
                    w = np.sqrt(2.0 * best_n) / m_trial
                    best_seq = [float(w) if trial[k] else 0.0 for k in range(best_n)]
                    print(f"[{time.time():.1f}] Swap improved → n={best_n}, m={m_trial}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen but keeps the contract safe
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = [1.0, 1.0]   # trivial safe sequence

    return best_seq
