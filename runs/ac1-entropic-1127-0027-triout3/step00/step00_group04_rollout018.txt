Implementation: simple function:

def project_simplex(v):
    # project onto simplex sum=1, v>=0
    # using sorting O(n log n)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

This ensures sum to 1.

def fast_score(arr):
    n = arr.shape[0]
    # compute convolution using FFT with zero-padding to length 2*n-1
    size = 1 << ((2*n - 1).bit_length())
    # Use np.fft.rfft for real input
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))

"""
Propose a sequence that beats the previous bound (2.0) by using the known
continuous‐optimiser a(x)=1/√(x+α).  For a discrete step function the
convolution a*a becomes almost constant and its maximal value can be
reduced by choosing a small positive shift α.  The ratio

    2·n·max(a*a)/sum(a)^2

is then ≈ (π‑4√α/√n)/2 + (π‑4√α/√n)/√n,
which for n≈6600 is minimised around α≈20‑30, giving a value ≈1.48.
The code searches α in a coarse‑to‑fine manner using a fast FFT‑based
evaluation (no calls to the provided evaluate_sequence) and finally
returns the best sequence found.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Fast evaluation of the objective (identical to evaluate_sequence)
# but using FFT for speed.
def _fast_score(arr: np.ndarray) -> float:
    """Return 2*n*max(conv(arr,arr))/sum(arr)^2 using FFT."""
    n = arr.shape[0]
    # size for convolution: next power of two ≥ 2*n-1
    size = 1 << ((2 * n - 1).bit_length())
    # real FFT of the zero‑padded array
    fa = np.fft.rfft(arr, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = arr.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a * sum_a))

    # Build the final sequence with the best α.
    final_alpha = best_alpha if best_alpha is not None else 1.0
    idx = np.arange(n, dtype=np.float64)
    final_seq = (1.0 / np.sqrt(idx + final_alpha)).tolist()

    # Optional logging (the platform prints these lines back to you).
    print(f"Chosen length n = {n}")
    print(f"Best shift α = {final_alpha:.6f}")
    print(f"Score (fast estimate) = {best_score:.6f}")

    # Return the sequence as a plain Python list of floats.
    return final_seq
