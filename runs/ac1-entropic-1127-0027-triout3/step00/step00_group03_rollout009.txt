"""
Propose a candidate step function sequence.

The algorithm searches for a probability distribution a (sum = 1)
that makes its self‑convolution as flat as possible.
A simple gradient descent on the squared error
    E(a) = Σ_k ( (a*a)_k - μ )²   with μ = 1/(2n‑1)
is performed for several lengths and random restarts.
The current best sequence (lowest value of evaluate_sequence)
is kept and returned.
"""

import numpy as np
import time

def _flatten_gradient(a):
    n = a.shape[0]
    b = np.convolve(a, a)                # linear convolution, length 2n‑1
    mu = 1.0 / (2 * n - 1)               # average of b (because Σb = 1)
    e = b - mu
    # correlation a with e: corr[k] = Σ_i a[i] * e[i+k]
    corr = np.correlate(a, e, mode='full')
    start = len(e) - 1                   # = 2n‑2
    grad = 4.0 * corr[start:start + n]   # derivative of Σ(e²) w.r.t a
    return grad, b

def _single_run(n, rng, max_iters, deadline):
    # start from a random Dirichlet vector
    a = rng.dirichlet(np.ones(n))
    # add a tiny jitter to break perfect symmetry
    a += 1e-6 * rng.randn(n)
    a = np.clip(a, 0, None)
    a /= a.sum()
    best_local = a.copy()
    best_score = float('inf')
    lr = 0.2          # base learning‑rate (will be scaled each step)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        grad, conv = _flatten_gradient(a)
        # evaluate current sequence
        try:
            score = evaluate_sequence(a.tolist())
        except Exception:
            score = float('inf')
        if score < best_score:
            best_score = score
            best_local = a.copy()
        # gradient step with norm‑based scaling
        max_abs = np.max(np.abs(grad))
        if max_abs > 0:
            step = lr / max_abs
        else:
            step = 0.0
        a = a - step * grad
        # keep feasibility
        a = np.clip(a, 0, None)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            # all zero – re‑sample
            a = rng.dirichlet(np.ones(n))
        # occasional random jitter to escape shallow minima
        if it % 100 == 0:
            a += 5e-4 * rng.randn(n)
            a = np.clip(a, 0, None)
            a /= a.sum()
    return best_local, best_score

def propose_candidate(seed=123, budget_s=1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine performs a few gradient‑descent runs on different
    lengths (256 … 2048) and keeps the best sequence found before the
    time budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # few seconds margin
    best_overall = None
    best_overall_score = float('inf')
    # lengths are chosen so that the theoretical lower bound
    #   2·n·(1/(2n‑1)) ≈ 1
    # becomes reachable when n is a few thousand.
    lengths = [256, 512, 1024, 2048]
    # number of independent restarts per length – adapt to remaining time.
    for n in lengths:
        # a very cheap upper bound: at most 20000 iterations per length,
        # but stop earlier if we run out of time.
        max_iters = 20000
        # we perform three independent restarts per length.
        for restart in range(3):
            if time.time() > deadline:
                break
            seq, seq_score = _single_run(n, rng, max_iters, deadline)
            if seq_score < best_overall_score:
                best_overall_score = seq_score
                best_overall = seq.copy()
                print(f"[{time.time()-start_time:6.2f}s] New global best "
                      f"score={best_overall_score:.6f} (n={n})")
            # early stop if we already beat the target
            if best_overall_score < 1.503:
                break
        if best_overall_score < 1.503:
            break
    # As a fallback we keep a simple uniform sequence if the search
    # failed to produce a better one.
    if best_overall is None:
        best_overall = np.ones(256) / 256.0
    return best_overall.tolist()

"""
search for a step‑function (non‑negative list) that minimises
    evaluate_sequence(seq) = 2·n·max(conv(seq,seq))/sum(seq)^2 .

The idea is to treat the coefficients as a probability vector a
(sum(a)=1) and to make its self‑convolution as flat as possible.
For a flat convolution the maximum value is close to the average
µ = 1/(2n‑1) and the objective 2·n·max(b) approaches the theoretical
limit 1.  We minimise the squared deviation

    E(a) = Σ_k ( (a*a)_k – µ )² ,

by a simple gradient descent with projection onto the simplex.
Several independent runs for different lengths (256 … 2048) are
performed within the time budget; the best sequence found is returned.
"""

import time
import numpy as np

def _conv_and_error(a: np.ndarray):
    """return (b, e, µ) where b = a*a, µ = 1/(2n‑1) and e = b‑µ."""
    n = a.shape[0]
    b = np.convolve(a, a)                     # linear convolution, length 2n‑1
    mu = 1.0 / (2 * n - 1)                    # average of b (∑b = 1)
    e = b - mu
    return b, e, mu

def _gradient(a: np.ndarray, e: np.ndarray):
    """
    Gradient of Σ e_k² w.r.t. a.
    For each i   g_i = 4· Σ_j a_j·e_{i+j},
    which is a correlation of a and e.
    """
    n = a.shape[0]
    # full correlation: corr[k] = Σ_i a[i]·e[i+k-(len(e)-1)]
    corr = np.correlate(a, e, mode='full')
    start = len(e) - 1                       # index that corresponds to shift 0
    grad = 4.0 * corr[start:start + n]
    return grad

def _single_run(n: int, rng: np.random.Generator,
                max_iters: int, deadline: float):
    """
    Gradient‑descent optimisation for a fixed length n.
    Returns the best sequence (as ndarray) and its score.
    """
    # start from a random Dirichlet vector (guarantees sum=1, a_i>0)
    a = rng.dirichlet(np.ones(n))
    # a tiny jitter helps to break the symmetry of the Dirichlet draw
    a += 1e-7 * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    best_local = a.copy()
    # score = 2·n·max(conv) because sum(a)=1
    b, _, _ = _conv_and_error(a)
    best_score = 2.0 * n * np.max(b)

    lr = 0.2          # base learning‑rate (will be scaled each iteration)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current convolution and error
        b, e, _ = _conv_and_error(a)

        # current score (no need to call evaluate_sequence)
        cur_score = 2.0 * n * np.max(b)
        if cur_score < best_score:
            best_score = cur_score
            best_local = a.copy()
            # occasional progress report
            print(f"[{time.time():.2f}] n={n} iter={it}  score={cur_score:.6f}")

        # gradient step on the L2‑flattening objective
        grad = _gradient(a, e)
        max_abs = np.max(np.abs(grad))
        if max_abs > 0:
            step = lr / max_abs                # scale so max change ≈ lr
        else:
            step = 0.0

        a = a - step * grad
        # project back to the probability simplex
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:                                 # numerical safety – re‑sample
            a = rng.dirichlet(np.ones(n))

        # every few iterations add a tiny random jitter to escape shallow minima
        if it % 100 == 0:
            a += 5e-4 * rng.standard_normal(n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()

    return best_local, best_score


def propose_candidate(seed: int = 123, budget_s: int = 1000, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Several gradient‑descent runs are performed for different lengths.
    The best sequence discovered before the time budget expires is returned.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    best_seq = None
    best_score = float('inf')

    # lengths are powers of two that keep the theoretical bound close to 1.
    lengths = [256, 512, 1024, 2048]

    for n in lengths:
        # three independent restarts per length (adjusted to remaining time)
        for restart in range(3):
            if time.time() > deadline:
                break
            seq, score = _single_run(n, rng, max_iters=20000, deadline=deadline)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()
                print(f"[{time.time():.2f}] NEW GLOBAL BEST: n={n} "
                      f"score={best_score:.6f}")

            # stop early if we have already beaten the target bound
            if best_score < 1.503:
                break
        if best_score < 1.503:
            break

    # Fallback: a plain uniform distribution (always feasible)
    if best_seq is None:
        n = 256
        best_seq = np.ones(n) / float(n)

    return best_seq.tolist()
