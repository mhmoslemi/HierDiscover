"""
Simple sub‑gradient descent for the L∞‐norm of a self‑convolution.

The evaluation metric is scale‑invariant: 2·n·max(conv(a,a))/(∑a)².
We fix the L¹–norm to 1 (simplex) and minimise the maximal entry of the
convolution.  The algorithm works as follows

1.  Initialise a decreasing geometric sequence a_i = r**i (r≈1‑c/n) and
    normalise to the simplex.
2.  Repeatedly compute the convolution, locate the (maybe several)
    indices attaining the maximum and form the sub‑gradient of the
    max‑function:
        g_i = Σ_{k∈K_max} a_{k‑i}   (zero if the index is out of range)
       (the true gradient of (a*a)_k is 2·a_{k‑i}; the factor 2 is irrelevant
        for the descent direction).
3.  Take a step a ← a – α·g with a diminishing step size α and project the
    result back onto the probability simplex (non‑negative, sum = 1).
4.  Keep the best vector seen so far (according to ``evaluate_sequence``).
    Occasionally perform a tiny random perturbation to escape plateaus.

The whole search runs for the allotted time (default 1000 s) and finally
returns the best sequence found.
"""

import time
import numpy as np
from typing import List

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # Find the rho value
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of  max_k (a*a)_k  w.r.t. a.
    For all k attaining the maximum, the gradient component is
        g_i = Σ a_{k-i}  (i.e. half of the true gradient 2·a_{k-i}).
    """
    n = a.shape[0]
    M = conv.max()
    # tolerance to catch numerical ties
    idxs = np.where(np.abs(conv - M) < 1e-12)[0]
    g = np.zeros_like(a)
    for k in idxs:
        # indices i for which k-i lies inside [0,n)
        i = np.arange(n)
        j = k - i
        mask = (j >= 0) & (j < n)
        g[i[mask]] += a[j[mask]]
    # Scale by 2 because the true derivative of (a*a)_k is 2*a_{k-i}
    # The constant factor does not affect the descent direction.
    return 2.0 * g


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Return a list of non‑negative heights (a sequence) that (hopefully)
    gives a low value of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5.0   # a small safety margin

    # ------------------------------------------------------------------
    # Determine the problem size.  Prefer the length of the previously
    # discovered construction if it exists.
    # ------------------------------------------------------------------
    try:
        init_len = len(height_sequence_1)   # type: ignore[name-defined]
    except Exception:
        init_len = 5300                     # fallback length used in the contest

    n = init_len

    # ------------------------------------------------------------------
    # 1) Start from a geometric sequence a_i = r^i .  The parameter c
    #    controls the decay rate: r = 1 - c/n .  A moderate c (≈2) works
    #    well as a warm start for the sub‑gradient phase.
    # ------------------------------------------------------------------
    c0 = 2.0
    r0 = max(0.0, 1.0 - c0 / n)
    a = np.array([r0 ** i for i in range(n)], dtype=float)
    a /= a.sum()                     # project onto the simplex (L¹ = 1)

    # Keep the best sequence seen so far
    best_a = a.copy()
    best_score = evaluate_sequence(best_a.tolist())

    # ------------------------------------------------------------------
    # Main optimisation loop (sub‑gradient descent with simplex projection)
    # ------------------------------------------------------------------
    it = 0
    step0 = 0.5                     # initial step size (will decay)
    while time.time() < deadline:
        # Compute convolution and the sub‑gradient of its maximum
        conv = np.convolve(a, a)
        grad = _subgradient_max_conv(a, conv)

        # Diminishing step size (simple 1/√t schedule)
        step = step0 / np.sqrt(it + 1)

        # Gradient step and projection onto the simplex
        a_candidate = _project_onto_simplex(a - step * grad)

        # Occasionally try a tiny random jitter to escape flat regions
        if np.random.rand() < 0.001:
            jitter = np.random.randn(n) * 1e-4
            a_candidate = _project_onto_simplex(a_candidate + jitter)

        # Evaluate – keep the better vector
        try:
            score = evaluate_sequence(a_candidate.tolist())
            if score < best_score:
                best_score = score
                best_a = a_candidate.copy()
                a = a_candidate               # continue from the improvement
            else:
                # With a small probability accept a non‑improving move
                # (simulated‑annealing style, temperature falls with time)
                temperature = max(0.01, (deadline - time.time()) / (budget_s))
                if np.random.rand() < 0.01 * temperature:
                    a = a_candidate
        except Exception:
            # In the unlikely event that the candidate is rejected by the
            # security checks we simply ignore it.
            pass

        # Periodically print progress (useful for debugging / monitoring)
        if it % 1000 == 0:
            print(f"[Iter {it:>6}] best score = {best_score:.6f}")

        # Simple restart mechanism: if we appear stuck for a long time,
        # re‑initialise a with a fresh random point on the simplex.
        if it % 20000 == 0 and it > 0:
            a = np.random.rand(n)
            a /= a.sum()

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python list)
    # ------------------------------------------------------------------
    return best_a.tolist()
