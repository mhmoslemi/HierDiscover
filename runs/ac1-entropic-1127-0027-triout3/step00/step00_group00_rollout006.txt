def project_simplex(v):
    # ensure sum(v)=1, v_i >= 0
    # from literature
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=12345, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5
    # Choose lengths for search
    lengths = [200, 500, 1000, 2000]  # adapt based on time
    best_seq = None
    best_score = np.inf

    for n in lengths:
        # start with uniform probability distribution
        p = np.full(n, 1.0/n)
        # run subgradient descent
        iter = 0
        while time.time() < deadline and iter < 5000:
            # compute convolution
            conv = np.convolve(p, p)
            max_conv = conv.max()
            # compute score (2*n*max_conv / (sum(p)^2) but sum(p)=1)
            score = 2 * n * max_conv
            if score < best_score:
                best_score = score
                best_seq = p.copy()
                print(f"New best n={n} iter={iter} score={score}")
            # find argmax index
            k = int(conv.argmax())
            # compute gradient
            g = np.zeros_like(p)
            lo = max(0, k - (n-1))
            hi = min(n-1, k)
            idx = np.arange(lo, hi+1, dtype=int)
            g[idx] = 2 * p[k - idx]
            # step size schedule
            eta = 0.5 / (1 + iter*0.001)  # decreasing
            # multiplicative update
            p = p * np.exp(-eta * g)
            # project onto simplex
            p = project_simplex(p)
            iter += 1
        # Optionally run random restarts
        for restart in range(5):
            # random Dirichlet init
            p = np.random.dirichlet(np.ones(n))
            for it in range(2000):
                # similar descent
                conv = np.convolve(p, p)
                k = int(conv.argmax())
                g = np.zeros_like(p)
                lo = max(0, k - (n-1))
                hi = min(n-1, k)
                idx = np.arange(lo, hi+1)
                g[idx] = 2 * p[k - idx]
                eta = 0.1 / (1 + it*0.01)
                p = p * np.exp(-eta * g)
                p = project_simplex(p)
                # evaluate
                if (it % 100) == 0:
                    score = 2 * n * conv.max()
                    if score < best_score:
                        best_score = score
                        best_seq = p.copy()
                        print(f"New best (restart) n={n} restart={restart} it={it} score={score}")
                if time.time() >= deadline:
                    break
            if time.time() >= deadline:
                break
        if time.time() >= deadline:
            break

    # Convert best_seq to raw heights (scale arbitrary)
    if best_seq is None:
        # fallback uniform
        best_seq = np.full(100, 1.0)
    # Scale up to avoid very small numbers (optional)
    best_seq = (best_seq * 1000).tolist()

    best_score = np.inf

    # define sequence lengths to try
    lengths = [200, 500, 1000, 1500, 2000]
    # allocate time per length proportionally
    for n in lengths:
        # compute remaining time
        remaining = deadline - time.time()
        if remaining < 2:
            break

        # number of restarts for this n
        restarts = 5
        for r in range(restarts):
            # init distribution: either uniform or Dirichlet or geometric
            if r == 0:
                p = np.full(n, 1.0 / n)
            elif r == 1:
                # random Dirichlet
                p = np.random.dirichlet(np.ones(n))
            else:

                q = np.random.rand() * 0.9 + 0.05
                weights = np.power(q, np.arange(n))

            # iterate subgradient descent

def _project_to_simplex(v):
    # Returns projection of v onto simplex {x >= 0, sum x = 1}
    # v may be numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function height sequence that minimizes the evaluation score.

The algorithm treats the sequence as a probability distribution (sum = 1) and
optimises the ℓ∞‑norm of its aperiodic self‑convolution:

    score = 2 * n * max(conv(p, p))

since the evaluation function is scale‑invariant.  We minimise the maximum
convolution value by a projected sub‑gradient (multiplicative) descent on the
simplex.  The method repeatedly:

1. computes the convolution,
2. identifies the index where it attains its maximum,
3. builds the sub‑gradient of that particular quadratic term,
4. applies a multiplicative update `p ← p * exp(-η * g)`,
5. projects back onto the probability simplex.

Multiple random restarts with different initialisations (uniform, Dirichlet,
geometric) and several sequence lengths are tried within the allotted time.
The best sequence found (scaled to the range [0, 1000] – scaling does not affect
the score) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: projection of a vector onto the probability simplex
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    # find the largest index rho where u_i > cssv_i / (i+1)
    ind = np.arange(1, n + 1)
    condition = u - cssv / ind > 0
    if not np.any(condition):
        # all entries will be zero; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(condition)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# Core optimisation routine for a given length n
# ----------------------------------------------------------------------
def _optimise_length(n: int, deadline: float, best_score: float, best_seq: np.ndarray):

    # maximum number of restarts per length

    # maximum iterations per restart (adjusted to time budget)
    max_iters = 5000

        if restart == 0:
            p = np.full(n, 1.0 / n)                     # uniform

        else:

        # Ensure numerical sanity

    deadline = time.time() + budget_s - 2.0  # small safety margin

    # ------------------------------------------------------------------
    # Try to start from the existing construction if it exists
    # ------------------------------------------------------------------
    try:
        # height_sequence_1 is provided by the evaluation harness
        init = np.array(height_sequence_1, dtype=float)
        if init.size > 0:
            init = np.maximum(init, 0.0)
            if init.sum() == 0:
                init = np.full_like(init, 1.0 / init.size)
            else:
                init = init / init.sum()
            best_seq = init.copy()
            best_score = float(2 * len(init) * np.convolve(init, init).max())
            print(f"[i] Started from height_sequence_1, score={best_score:.6f}")
        else:
            raise NameError
    except Exception:
        best_seq = None
        best_score = np.inf

    # ------------------------------------------------------------------
    # Main search over a few different lengths
    # ------------------------------------------------------------------
    # Lengths are chosen to give a good trade‑off between quality and speed.
    lengths = [200, 400, 800, 1200, 1600]
    for n in lengths:
        if time.time() >= deadline:
            break
        # Ensure we have enough time for at least one short run
        if time.time() + 0.2 > deadline:
            break
        best_score, best_seq = _optimise_length(
            n, deadline, best_score, best_seq
        )

    # ------------------------------------------------------------------
    # Fallback: uniform distribution if nothing better was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 500
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    # Scale the sequence – scaling does not affect the score, but keeping the
    # values within the allowed [0, 1000] range is convenient.
    best_seq = (best_seq * 1000.0).tolist()
    return best_seq
