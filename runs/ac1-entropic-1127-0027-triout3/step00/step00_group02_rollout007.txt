"""
Gradient‐descent search for a non‑negative step‑function with a flat
auto‑convolution.

The algorithm works on a normalized copy of the sequence (sum = 1),
minimises a smooth approximation of the max convolution value
`max_k (f*f)_k` by using a soft‑max (log‑sum‑exp) and the analytic gradient
of the soft‑max w.r.t. the sequence.  The update respects the scale‑invariance
of the original objective, keeps the total mass constant (up to clipping) and
projects the variables back to the feasible non‑negative orthant.
Periodically the current sequence is evaluated with ``evaluate_sequence`` and
the best one found is kept.

The method is fast (FFT based convolutions) and typically improves the
uniform “all equal heights’’ construction (score ≈ 2.0) to a score well below
2.0 within the allotted time.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _fft_convolve(a: np.ndarray, b: np.ndarray, size: int) -> np.ndarray:
    """Convolution of real vectors a and b using FFT of given size."""
    A = np.fft.fft(a, size)
    B = np.fft.fft(b, size)
    conv = np.fft.ifft(A * B).real
    return conv[: a.size + b.size - 1]

    Search for a good height sequence.

    Parameters

    seed : int
        Random seed (default 42).
    budget_s : int

    Returns

    list[float]
        The best sequence found within the time budget.
    """

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # use the known SOTA length
    except Exception:                     # fallback: pick a moderate size
        n = 800

    # 2. Initial sequence (uniform + tiny noise)
    # ------------------------------------------------------------------
    x = np.full(n, 1.0, dtype=np.float64)
    x += 0.001 * np.random.rand(n)        # break perfect symmetry

    # Keep the best solution seen so far
    best_seq = x.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    # ------------------------------------------------------------------
    # 3. FFT size pre‑computations
    # ------------------------------------------------------------------
    # size for the (f*f) convolution (length 2n‑1)
    conv_size = _next_pow2(2 * n - 1)
    # size for the gradient convolution (length (2n‑1) + n‑1 = 3n‑2)
    grad_size = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
