import numpy as np, time

def convolve_fft(a):
    n = len(a)
    m = 1 << ((2*n - 1).bit_length())
    A = np.fft.rfft(a, m)
    C = np.fft.irfft(A*A, m)[:2*n-1]
    return C

def compute_gradient(a, max_indices):
    n = len(a)
    grad = np.zeros(n)
    for k in max_indices:
        start = max(0, k - (n-1))
        end = min(k, n-1)
        # i from start to end
        for i in range(start, end+1):
            j = k - i
            grad[i] += 2*a[j]  # as derived
    return grad

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 5 # margin

    # choose length based on seed, maybe random in range 500-4000

    best_score = evaluate_sequence(best_a.tolist())
    print(f"Initial n={n}, score={best_score}")

    # parameters
    iteration = 0
    temperature = 0.1  # for simulated annealing acceptance probability
    temp_decay = 0.9995  # per iteration

    # choose step scaling factor relative to target sum
    while time.time() < deadline:
        # compute convolution and key stats
        conv = convolve_fft(a)
        max_val = conv.max()
        # identify indices where conv close to max (within tolerance)
        tol = 1e-10 * max_val + 1e-12
        max_indices = np.where(conv >= max_val - tol)[0]

        grad = compute_gradient(a, max_indices)

        # compute direction to move mass from high gradient to low gradient
        # subtract mean to zero-sum direction
        grad_mean = np.mean(grad)
        direction = -(grad - grad_mean)

        # step size: adapt based on magnitude
        # Ensure step does not exceed a fraction of a to avoid negativity
        step_scale = 0.02 * target_sum / np.linalg.norm(direction) if np.linalg.norm(direction) != 0 else 0
        # direction normalized to have L2=1 times step_scale

        a_new = a + delta

        a_new = np.maximum(a_new, 0.0)

        # renormalize sum to target_sum to keep scaling invariant
        sum_new = a_new.sum()
        if sum_new == 0:
            # fallback random restart
            a_new = np.full(n, target_sum / n)
        else:
            a_new *= target_sum / sum_new

        # evaluate candidate
        try:
            cand_score = evaluate_sequence(a_new.tolist())

            cand_score = np.inf

        if cand_score < best_score:

        if cand_score < best_score or np.random.rand() < np.exp(-(cand_score - best_score) / temperature):

def convolve_full(a):
    n = len(a)
    size = 1 << ((2*n - 1).bit_length())
    fft_a = np.fft.rfft(a, size)
    con = np.fft.irfft(fft_a * fft_a, size)[:2*n-1]
    return con

def compute_max_and_indices(conv, rel_tol=1e-9):
    max_val = conv.max()
    tol = max_val * rel_tol + 1e-12
    indices = np.where(conv >= max_val - tol)[0]
    return max_val, indices

def gradient_for_max(a, max_inds):
    n = len(a)
    grad = np.zeros(n)
    for k in max_inds:
        start = max(0, k - (n-1))
        end = min(k, n-1)
        i = np.arange(start, end+1)
        j = k - i  # reversed indices
        grad[i] += 2 * a[j]
    return grad
