def propose_candidate(seed=0, budget_s=1000, **kwargs):
    """
    Search for a step function that minimizes the ratio 2*n*max(conv)/sum^2.
    The algorithm first samples parametrized power‑law sequences a_i = (i+α)^‑p
    (with an optional linear tilt). The best sampled sequence is then refined
    by a simple hill‑climbing that swaps a small amount of mass between two
    random positions while keeping the total sum constant. The procedure stops
    after the allotted time and returns the best sequence found.
    """
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    # Determine length
    try:
        n = len(height_sequence_1)
    except Exception:
        n = 2000
    # Helper to compute the evaluation score (without the security checks)
    def compute_score(arr):
        conv = np.convolve(arr, arr)
        return 2.0 * n * conv.max() / (arr.sum() ** 2)
    best_score = float('inf')
    best_seq = None
    deadline = time.time() + budget_s - 0.5   # keep a little margin
    # ----------------------------------------------------
    # 1) Random parametric sampling
    # ----------------------------------------------------
    while time.time() < deadline:
        # stop after a reasonable number of samples to leave time for refinement
        # (controlled by remaining time)
        # sample parameters
        p = rng.uniform(0.35, 0.65)            # exponent around the critical 0.5
        # offset α sampled logarithmically
        log_alpha = rng.uniform(-2.0, 2.0)     # α in [1e‑2, 1e2]
        alpha = 10.0 ** log_alpha
        tilt = rng.uniform(-0.5, 0.5)          # linear tilt coefficient
        idx = np.arange(n, dtype=np.float64)
        a = (idx + alpha) ** (-p)
        if abs(tilt) > 1e-12:
            a = a * (1.0 + tilt * idx / (n - 1))
            a = np.clip(a, 0.0, None)
        # quick discard if sum too small (should never happen)
        if a.sum() < 1e-3:
            continue
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[search] new best score {sc:.6f}  p={p:.4f}  α={alpha:.4f}  tilt={tilt:.4f}")
        # break after a few thousand samples to reserve time
        # Estimate remaining time and number of samples done
        # We'll just continue until we have < 15% of time left
        if time.time() > deadline * 0.85:
            break
    # ----------------------------------------------------
    # 2) Local hill‑climbing refinement
    # ----------------------------------------------------
    if best_seq is not None:
        a = best_seq.copy()
        curr_score = best_score
        # we keep a separate copy of the current best for fast access
        # number of refinement steps determined by remaining budget
        while time.time() < deadline:
            # pick two distinct indices
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            total = a[i] + a[j]
            if total == 0.0:
                continue
            # propose moving a random fraction of the total mass between the two slots
            delta = (rng.random() - 0.5) * 0.2 * total   # |delta| ≤ 0.1 * total
            new_i = a[i] + delta
            new_j = a[j] - delta
            if new_i < 0.0 or new_j < 0.0:
                continue
            a_candidate = a.copy()
            a_candidate[i] = new_i
            a_candidate[j] = new_j
            sc = compute_score(a_candidate)
            # simple greedy acceptance (could be replaced by simulated annealing)
            if sc < curr_score:
                a = a_candidate
                curr_score = sc
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[refine] improved score {sc:.6f} at iter {int(time.time())}")
    # If nothing was found fall back to the provided sequence
    if best_seq is None:
        try:
            best_seq = np.array(height_sequence_1, dtype=np.float64)
        except Exception:
            best_seq = np.ones(1000, dtype=np.float64) / 1000.0
    # Return as plain Python list (as required by evaluate_sequence)
    return best_seq.tolist()

"""
Propose a step‑function that minimizes

    2 · n · max(conv(a,a)) / (sum(a))²

The algorithm works in two phases.

1.  A parametric family a_i = (i+α)^‑p·(1+w·i/(n‑1)) is explored.
    The base (i+α)^‑p with p≈0.5 (≈1/√i) gives an almost flat self‑convolution.
    Adding a linear tilt w raises the later entries, which increases the total
    sum faster than it raises the maximal convolution, thereby lowering the
    ratio.  A coarse grid over (α,p,w) is searched and the best parameters are
    kept.

2.  The best sequence is refined by a cheap hill‑climbing: a small amount of
    mass is moved between two random positions while preserving the total
    sum.  Moves that improve the score are kept.

The whole search respects the 1000 s budget and never returns a negative,
NaN or infinite value.  The returned list can be fed directly to
`evaluate_sequence`.  No external I/O is performed.
"""

import time
import numpy as np

def _score(arr: np.ndarray) -> float:
    """Fast evaluation of the objective (without the security wrapper)."""
    conv = np.convolve(arr, arr)
    return 2.0 * len(arr) * conv.max() / (arr.sum() ** 2)


def _make_sequence(n: int, p: float, alpha: float, tilt: float) -> np.ndarray:
    """
    Build a candidate sequence of length `n`.

    a_i = (i + alpha) ** (-p) * (1 + tilt * i/(n-1))
    Negative values are clipped to zero.
    """
    idx = np.arange(n, dtype=np.float64)
    a = (idx + alpha) ** (-p)
    if abs(tilt) > 1e-12:
        a = a * (1.0 + tilt * idx / (n - 1))
        a = np.clip(a, 0.0, None)
    return a

    """
    Search for a (non‑negative) height sequence that yields a low
    `evaluate_sequence` score.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length of the problem (use the provided reference)
    # --------------------------------------------------------------
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
    except Exception:
        n = 2000                            # fallback size if the variable is missing

    # --------------------------------------------------------------
    # 1) Coarse parametric search
    # --------------------------------------------------------------
    best_score = float("inf")
    best_seq = None
    best_params = (0.5, 0.0, 0.0)           # (p, alpha, tilt)

    # grids (chosen to be cheap enough to stay well within the time limit)
    p_grid = [0.45, 0.5, 0.55]             # around the optimal 0.5
    alpha_grid = np.logspace(-2, 2, 40)    # 0.01 … 100
    tilt_grid = np.linspace(-0.85, 0.85, 71)  # avoid extreme negatives

    deadline = time.time() + budget_s - 5.0   # reserve a few seconds for refinement

    for p in p_grid:
        for alpha in alpha_grid:
            # pre‑compute the base part (independent of tilt)
            base = (np.arange(n, dtype=np.float64) + alpha) ** (-p)
            for tilt in tilt_grid:
                if time.time() > deadline:
                    break
                if abs(tilt) > 1e-12:
                    seq = base * (1.0 + tilt * np.arange(n) / (n - 1))
                    seq = np.clip(seq, 0.0, None)

                    seq = base
                # quick sanity check (sum must be > 0.01 to avoid INF)
                if seq.sum() < 1e-2:

                sc = _score(seq)
                if sc < best_score:
                    best_score = sc
                    best_seq = seq.copy()
                    best_params = (p, alpha, tilt)
                    print(f"[grid] new best {sc:.6f}  p={p:.3f} α={alpha:.3f} tilt={tilt:.3f}")

    # --------------------------------------------------------------
    # 2) Local refinement of the continuous parameters (p, alpha, tilt)
    # --------------------------------------------------------------
    # simple random walk around the best point found above
    p, alpha, tilt = best_params
    step_scales = np.array([0.02, max(alpha * 0.1, 0.02), 0.05])  # step sizes for p, alpha, tilt

    for _ in range(4000):
        if time.time() > deadline:
            break
        # propose a small perturbation
        cand_p = np.clip(p + rng.normal() * step_scales[0], 0.3, 0.8)
