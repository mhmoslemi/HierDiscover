"""
Propose a step‑function minimizing 2 n·max(a*a)/(∑a)².

Algorithm (≈1000 s)
-------------------
* **Initialisation** – start from the best known sequence (global
  `height_sequence_1` if present) padded to a power‑of‑two length,
  otherwise from a uniform distribution.

* **Phase 1 – smoothed‑max gradient descent** (≈70 % of the budget)  
  – minimise a smooth approximation of the peak:
        Lε(a)=ε·log(∑k exp((a*a)[k]/ε)) .
    Gradient of Lε is
        ∇Lε = 2·a * w_rev  with  w_k ∝ exp((a*a)[k]/ε) .
    The gradient is obtained with two FFT‑based convolutions.
    An Adam optimiser updates the vector, then a Euclidean projection
    onto the simplex (Σa=1, a≥0) keeps the feasibility.

* **Phase 2 – stochastic fine‑tuning** (≈30 % of the budget)  
  – classic mass‑transfer moves (donor→recipient) with a Metropolis
    acceptance rule and a linearly cooling temperature.
    The self‑convolution is updated incrementally (O(n)) to keep the
    inner loop cheap.  Every few hundred accepted moves a full FFT
    recomputation eliminates drift.  Occasional log‑normal block
    jitter helps to escape plateaux.

* **Restarts** – if no improvement for 30 s the current best is
  perturbed with moderate log‑normal noise and the search continues.

The best feasible vector seen during the whole run is returned as a
plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new a and c."""
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # update linear part: 2·δ·(shift_j – shift_i)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ part
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("input must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_lse(a: np.ndarray, eps: float,
                  fft_len_self: int, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smoothed max Lε(a)=ε·log∑exp((a*a)/ε).
    Returns a vector of shape a.shape.
    """
    n = a.size

    # 1) self‑convolution a*a
    conv = _self_conv(a, fft_len_self)          # length 2n‑1

    # 2) softmax weights over convolution entries
    w = np.exp(conv / eps)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # 3) compute 2·a * w_rev via FFT (full convolution length 3n‑2)
    w_rev = w[::-1]                             # reversed vector
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    conv2 = np.fft.irfft(fa * fw, n=fft_len_grad)[: 3 * n - 2]

    # gradient extraction: g_j = 2·conv2[2n‑2‑j]
    idx = 2 * n - 2 - np.arange(n)
    grad = 2.0 * conv2[idx]
    return grad
