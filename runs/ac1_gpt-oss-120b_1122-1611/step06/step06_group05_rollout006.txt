def project_simplex(v):
    # v is ndarray
    # ensures non-negative after projection
    # Using algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho = max {j | u_j > cssv_j/j}
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Will ensure sum=1.

"""
Propose a non‑negative step‑function (vector of heights) that minimizes
`evaluate_sequence`.  

Approach
--------
1. Initialise a candidate from the best known global vector (if present) or
   from the classical √‑difference shape.  The vector is normalised to sum‑1.
2. Scan a small family of analytic shapes (power‑law, exponential decay,
   random Dirichlet draws) and keep the best one.
3. Refine the best candidate by a projected sub‑gradient method that directly
   targets the lag where the self‑convolution attains its maximum.
   * For the current vector `a` compute the convolution `c = a*a`.
   * Let `k` be an index where `c[k]` is maximal.
   * The sub‑gradient of `max_k c[k]` w.r.t. `a` is
         g_i = 2·a_{k‑i}   (if 0 ≤ k‑i < n, else 0).
   * Perform an additive step `a ← a – η·g`, clamp negatives to 0 and
     project back onto the probability simplex (sum = 1).  The step size
     η decays slowly with elapsed time.
4. Occasionally add a small random jitter to the logits (log a) and restart
   from the current best if no improvement is seen for a while.
5. The loop runs until the allotted wall‑time (≈ 1000 s) expires.
   The best sequence encountered is returned (clipped to the limits enforced
   by `evaluate_sequence`).

The algorithm uses only FFT‑based convolutions (O(n log n)) and a fast
simplex projection, allowing many thousands of iterations within the time
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold a linear convolution
# of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    # round up to the next power of two
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Projection of a vector onto the probability simplex {x≥0, Σx=1}.
# Uses the algorithm of Duchi et al. (ICML 2008).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex."""
    # work on a copy to avoid side‑effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.shape[0] + 1, dtype=v.dtype)
    # find the rho (last index where the condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.where(cond)[0][-1]
    theta = cssw = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Score used during the search: 2·n·max(conv(a,a)).
# This equals the value returned by `evaluate_sequence` for a vector with
# unit sum (the search always keeps the sum equal to 1).
# ----------------------------------------------------------------------
def _score(a: np.ndarray, fft_len: int) -> float:
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())
