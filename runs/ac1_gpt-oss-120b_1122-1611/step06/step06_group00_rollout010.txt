from scipy.optimize import linprog

def _conv_full(a):
    return np.convolve(a, a)

def _build_conv_matrix(a, A):
    n = a.size
    # assumes A shape (2*n - 1, n)
    # zero whole matrix first (not needed if we always fill all rows column wise)
    # but we must fill from start to end; we will fill each column
    for i in range(n):
        A[i:i+n, i] = a

def _solve_lp_for_g(a, M, A, rng):
    # fill matrix with current a
    _build_conv_matrix(a, A)
    # constraints: A @ g <= M (vector)
    b = np.full(A.shape[0], M, dtype=np.float64)
    # objective: maximize sum g -> minimize -1 vector
    c = -np.ones(a.size, dtype=np.float64)
    bounds = [(0, None)] * a.size

    return res.x

Propose a step‑function that improves the bound of `evaluate_sequence`.

Algorithm
---------
1. **Initialisation** – If a previous solution exists in the global
   variable `height_sequence_1`, use it (normalised to sum = 1).  Otherwise
   start from a uniform distribution of a moderate length (≈2000).

2. **Linear‑programming refinement** – For the current vector `a` (size *n*,
   sum = 1) we compute the peak `M = max(a*a)`.  The constraints
   ``(a * g)_k ≤ M`` are linear in `g`; they can be written as
   `A @ g ≤ M` where `A` is the (2n−1)×n convolution matrix built from `a`.
   We solve the LP  

        maximise   Σ g_j
        subject to A·g ≤ M , g ≥ 0

   L₁‑norm while still respecting the current peak.

3. **Convex move** – Scale `g₀` to the same L₁‑norm as `a` (which is 1) and
   try a line‑search over step sizes `t ∈ {0.5,0.3,0.2,0.1,0.05,0.02,0.01,0.005}`
   for the update `a ← (1‑t)·a + t·g`.  The new peak is recomputed; the
   first `t` that yields a strict improvement is accepted.

4. **Stagnation handling** – If no improvement is found we apply a mild
   log‑normal jitter to a random contiguous block of `a`.  After 30 s of
   no progress we restart from the best solution seen so far, adding a
   moderate noise perturbation.

5. **Loop** – Repeat steps 2‑4 until the time budget (≈1000 s, 5 s safety
   margin) is exhausted or the target bound 1.5025 is reached.

The routine returns the best sequence discovered (clipped to the limits

from scipy.optimize import linprog


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a."""
    return np.convolve(a, a)


def _build_conv_matrix(a: np.ndarray, A: np.ndarray) -> None:
    """
    Fill the matrix ``A`` with the convolution constraints for a given
    vector ``a``.  After the call ``A @ g`` equals ``a * g`` for any ``g``.
    ``A`` must have shape (2*len(a)-1, len(a)) and be initialised to zero.
    """
    n = a.size
    # each column i is the vector a placed at rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = a


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
