def _optimise_one_length(a_init, deadline, rng):
    a = a_init.copy()
    n = a.shape[0]
    # precompute fft lengths
    fft_len_self = _next_pow2(2*n - 1)
    fft_len_grad = _next_pow2(3*n - 2)

    # compute initial score
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    last_improve = time.monotonic()
    it = 0

    # schedule
    start = time.monotonic()
    total = deadline - start
    phase1_end = start + 0.3 * total  # 30% for softmax
    # we can also make a warmup for smoothing not needed.

    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        if now < phase1_end:
            # softmax gradient
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()
            # Temperature schedule
            frac = (now - start) / (phase1_end - start)
            beta = 0.5 + (4000.0 - 0.5) * frac   # increase
            # softmax weights:
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum
            # gradient via convolution with rev_a
            rev_a = a[::-1]
            full = _linear_conv(p, rev_a, fft_len_grad)
            grad = 2.0 * full[n-1:n-1+n]
            # step size schedule decreasing linearly
            eta = 1.0 * (1 - frac) + 0.01 * frac
            a = a - eta * grad
        else:
            # subgradient with Adam
            conv = _self_conv(a, fft_len_self)
            k = int(conv.argmax())
            # subgradient g_i: 2 * a_{k-i} where indices valid
            g = np.zeros_like(a)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k - idx_i
            g[idx_i] = 2.0 * a[idx_j]

            # update moments
            beta1 = 0.9
            beta2 = 0.999
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g * g)
            m_hat = m / (1 - beta1 ** it)
            v_hat = v / (1 - beta2 ** it)
            # step size
            eta0 = 0.2
            eta = eta0 / np.sqrt(it)
            step = eta * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = a - step

        # projection
        a = _project_simplex(a)

        # evaluation
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:6d} new best {best_score:.6f}")

        s = a0.sum()
        if s > 0:

        else:

def _interpolate_to_length(arr, new_n):
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)

"""
Propose‑candidate algorithm
---------------------------

The routine searches for a non‑negative step function (a probability vector on the
simplex) that minimizes the evaluation function

    2·n·max(conv(a,a)) / (sum(a)**2)

Because the objective is scale‑invariant we keep the sum equal to 1 and minimise

    score = 2·n·max(conv(a,a))

The optimizer works in two phases for each chosen length `n`:

* **Phase 1 – smoothed soft‑max gradient (≈ 60 % of the allocated time).**
  A temperature `β` is increased from 0.5 to 6000 while a learning rate `η`
  is decreased from 2.0 to 0.02.  The gradient of the soft‑max approximation of
  the maximum is computed by a convolution in the frequency domain.  An
  additive step is taken and the vector is projected onto the probability
  simplex.

* **Phase 2 – exact sub‑gradient descent with Adam‑style moments.**
  The true peak index `k` of the self‑convolution is located; the sub‑gradient
  `g_i = 2·a_{k‑i}` is formed and an Adam update with a stepsize decaying as
  `η₀/√t` (`η₀ = 0.3`) is applied.  The iterate is again projected onto the
  simplex.

* **Stagnation handling** – if no improvement occurs for 20 s the current
  iterate is jittered by a multiplicative log‑normal perturbation of
  decreasing variance and the Adam moments are reset.

The method is applied to several lengths (including the previous construction
if available).  Each length receives a time slice; the best sequence found over
all lengths is returned.  The final vector is clipped to the limits enforced by
`evaluate_sequence` (0 ≤ x ≤ 1000).

The implementation uses only NumPy and the standard library, respects the
1000 s budget, and avoids closures as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to `x`."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length `fft_len_self`."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)
    return conv[: 2 * a.size - 1]


def _linear_conv(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution (len(x)+len(y)-1) via FFT of given length."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    return np.fft.irfft(fx * fy, n=fft_len)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _interpolate_to_length(arr: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a 1‑D array to a new length."""
    old_n = arr.shape[0]
    if old_n == new_n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    return np.interp(x_new, x_old, arr)


# ----------------------------------------------------------------------
# Core optimisation routine for a fixed length
# ----------------------------------------------------------------------


def _optimise_one_length(a_init: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Optimise a probability vector `a_init` (sum = 1) for a given length.
    Returns the best vector found and its score (2·n·max(conv)).
    """
    a = a_init.copy()
    n = a.size

    # FFT lengths (powers of two, fixed for the whole run)
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # initial evaluation
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    cur_score = best_score

    start = time.monotonic()
    total = deadline - start
    # 60 % of the time for the smoothed phase
    phase1_end = start + 0.60 * total

    # hyper‑parameters for phase 1
    eta_start = 2.0
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 6000.0

    # Adam moments for phase 2
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eta0_sub = 0.30

    it = 0
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        if now < phase1_end:
            # ----- Phase 1 – smoothed soft‑max gradient -----
            frac = (now - start) / (phase1_end - start)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()

            # soft‑max probabilities (stable formulation)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            # gradient = 2·(p * rev(a))
            rev_a = a[::-1]
            full = _linear_conv(p, rev_a, fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * full[n - 1 : n - 1 + n]

            a = a - eta * grad
        else:
            # ----- Phase 2 – exact sub‑gradient with Adam -----
            conv = _self_conv(a, fft_len_self)
            k = int(conv.argmax())

            # sub‑gradient g_i = 2·a_{k-i} where indices exist
            g = np.zeros_like(a)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k - idx_i
            g[idx_i] = 2.0 * a[idx_j]

            # Adam update
            m = beta1 * m + (1.0 - beta1) * g
            v = beta2 * v + (1.0 - beta2) * (g * g)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            eta = eta0_sub / np.sqrt(it)
            step = eta * m_hat / (np.sqrt(v_hat) + 1e-8)
            a = a - step

        # ----- Projection onto the simplex -----
        a = _project_simplex(a)

        # ----- Evaluation of the current candidate -----
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}")

        # ----- Stagnation handling (jitter) -----
        if now - last_improve > 20.0:
            # reset to the best seen, apply a small log‑normal perturbation
            a = best.copy()
            sigma = max(0.02, 0.2 * (1.0 - (now - start) / total))
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            a = a * noise
            a = _project_simplex(a)
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} jitter (stagnation)")

    return best, best_score

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    overall_deadline = start_time + total_budget

    # ------------------------------------------------------------------
    # 1) Determine lengths to explore
    # ------------------------------------------------------------------
    lengths = []

    # keep the previous construction (if any)
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
            lengths.append(seq0.size)

    # a range of candidate sizes (mix of powers of two and the historic size)
    for n in (1024, 2048, 3000, 3809, 4096, 6000, 8000, 10000, 12000):
        if n not in lengths:
            lengths.append(n)

    best_overall = None
    best_score_overall = float("inf")

    # ------------------------------------------------------------------
    # 2) Optimise each length while time permits
    # ------------------------------------------------------------------
    while lengths and time.monotonic() < overall_deadline:
        n = lengths.pop(0)
        now = time.monotonic()
        remaining = overall_deadline - now

        # ensure we keep at least 30 s for any future length
        min_per_length = 30.0
        if remaining < min_per_length:
            break
        # allocate a slice: at most 180 s, otherwise keep enough for the rest
        slice_time = min(180.0, remaining - (len(lengths) * min_per_length))
        deadline = now + slice_time

        # initialise a probability vector of size n
        if best_overall is not None:
            a0 = _interpolate_to_length(best_overall, n)
        elif "height_sequence_1" in globals():
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            a0 = _interpolate_to_length(seq, n)
        else:
            a0 = rng.dirichlet(np.ones(n))

        # normalise onto the simplex (sum = 1)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s > 0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / n)

        # run the optimiser for the chosen length
        vec, score = _optimise_one_length(a0, deadline, rng)

        # track the global best
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # 3) Final fine‑tuning on the best length (if time permits)
    # ------------------------------------------------------------------
    now = time.monotonic()
    if best_overall is not None and now < overall_deadline - 5.0:
        # give the remaining budget to a last sub‑gradient run
        extra_deadline = overall_deadline
        vec, score = _optimise_one_length(best_overall, extra_deadline, rng)
        if score < best_score_overall - 1e-12:
            best_overall = vec.copy()
            best_score_overall = score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] *** FINAL TUNE BEST score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback – should never happen but guards against empty result.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    # 5) Clip to the limits enforced by ``evaluate_sequence`` and return.
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    return [float(x) for x in best_overall]
