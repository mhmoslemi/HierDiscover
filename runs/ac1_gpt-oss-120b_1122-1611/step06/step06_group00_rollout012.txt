            a = a / a_sum if a_sum>0 else np.full_like(a, 1.0 / a.size)
        else:
            n = 2048
            a = np.full(n, 1.0/n, dtype=np.float64)
    else:
        n = 2048
        a = np.full(n, 1.0/n, dtype=np.float64)

    # Ensure positivity: small epsilon.
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()

    # Precompute FFT length for convolution
    def conv_self(v):
        fft_len = 1 << ((2*len(v)-2).bit_length())
        fa = np.fft.rfft(v, n=fft_len)
        fc = fa*fa
        conv = np.fft.irfft(fc, n=fft_len)[:2*len(v)-1]
        return conv

    # Evaluate objective directly (no need to call evaluate_sequence)
    # But for final score we might want to use evaluate_sequence for final check.
    def score_of(v):
        c = conv_self(v)
        M = c.max()
        return 2.0 * len(v) * M

    # Main stats
    c = conv_self(a)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # parameters
    move_counter = 0
    RECOMPUTE_EVERY = 300
    last_lp = start_time
    lp_interval = 30.0
    alpha_lp = 0.4  # mixing factor after LP

    # Main loop
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # Compute k_max and contributions
        k_max = int(c.argmax())
        M = c.max()
        # occasional LP refinement
        if now - last_lp > lp_interval and n <= 3000:
            # Build Toeplitz matrix A using sparse diags
            # each column i has value a_i on diagonal offset i (length n)
            diags_data = [np.full(n, a[i]) for i in range(n)]
            offsets = np.arange(n)
            A = sp.diags(diags_data, offsets, shape=(2*n-1, n), format='csr')
            # Build bounds: (0, 1000)
            bounds = [(0.0, 1000.0) for _ in range(n)]
            # Objective: maximize sum(g) -> minimize -sum(g)
            c_obj = -np.ones(n)
            b_ub = np.full(2*n-1, M)
            res = opt.linprog(c=c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs',
                              options={'presolve': True, 'time_limit': 10})
            if res.success:
                g0 = res.x
                sum_g0 = g0.sum()
                if sum_g0 > 1.0 + 1e-9:
                    # Normalise g to have sum=1
                    g = g0 / sum_g0
                    # Mix a with g
                    a = (1 - alpha_lp) * a + alpha_lp * g
                    a = np.clip(a, 0.0, None)
                    a /= a.sum()
                    # recompute convolution
                    c = conv_self(a)
                    cur_score = 2.0 * n * c.max()
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best = a.copy()
                        last_improve = now
                        print(f"[{elapsed:6.2f}s] LP step improved score = {best_score:.6f}")
                else:
                    # No improvement possible via LP
                    pass
            else:
                # LP didn't succeed
                pass
            last_lp = now

        # Perform stochastic move (similar to algorithm above)
        # compute contribution for donor selection
        partners = k_max - np.arange(n)
        mask = (partners >= 0) & (partners < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[partners[mask]]
        total_contrib = contrib.sum()
        # donor selection
        if total_contrib > 0.0:
            prob = contrib / total_contrib
            donor = rng.choice(np.arange(n), p=prob)
        else:
            donor = rng.integers(0, n)
        if a[donor] < 1e-14:
            continue
        # recipient selection, bias to small mass
        recv_weights = 1.0 - a
        recv_weights[donor] = 0.0
        sum_rw = recv_weights.sum()
        if sum_rw > 0.0:
            recv_weights /= sum_rw
            recipient = rng.choice(np.arange(n), p=recv_weights)
        else:
            recipient = (donor + rng.integers(1, n)) % n
        # delta fraction
        max_delta = 0.5 * a[donor]
        delta = rng.random() * max_delta
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta
        # incremental convolution update
        c_new = c.copy()
        c_new[donor:donor+n] -= 2.0 * delta * a
        c_new[recipient:recipient+n] += 2.0 * delta * a
        # quadratic terms
        ii = 2 * donor
        jj = 2 * recipient
        ij = donor + recipient
        if ii < c_new.size:
            c_new[ii] += delta*delta
        if jj < c_new.size:
            c_new[jj] += delta*delta
        if ij < c_new.size:
            c_new[ij] -= 2.0 * delta*delta
        new_score = 2.0 * n * c_new.max()
        # temperature schedule
        total_time = deadline - start_time
        t_frac = (now - start_time) / total_time
        temperature = 0.07 * (1 - t_frac) + 1e-6 * t_frac
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (temperature+1e-12)):
            a = a_new
            c = c_new
            cur_score = new_score
            move_counter += 1
            if move_counter % RECOMPUTE_EVERY == 0:
                c = conv_self(a)
                cur_score = 2.0 * n * c.max()
        # occasional global jitter
        if rng.random() < 0.025:
            # pick random block width up to n//8
            width = rng.integers(1, max(2, n // 8))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.2))
            a_block = a.copy()
            a_block[start:start+width] *= factor
            a_block = np.clip(a_block, 0.0, None)
            a_block /= a_block.sum()
            a = a_block
            c = conv_self(a)
            cur_score = 2.0 * n * c.max()
        # record best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (stochastic) = {best_score:.6f}")

        # Stagnation restart
        if now - last_improve > 35.0:
            # restart from best with random perturb
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = conv_self(a)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{elapsed:6.2f}s] restart after stagnation, best {best_score:.6f}")

    # final best sequence as list, clip to safety
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

"""
Propose a step‑function that improves the bound on
    2·n·max(conv(a,a)) / (sum(a))²

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1).  If a previous
   construction exists (`height_sequence_1`) it is used, otherwise a set
   of hand‑crafted candidates is evaluated and the best taken.

2. Two alternating phases (whole budget ≈1000 s):
   * a) *LP‑refinement* (≈30 % of the time).  
       – Build the Toeplitz matrix `A` that maps a vector `g` to
         the linear convolution `a*g`.  
       – Solve a linear programme  

            maximise   Σ g_i
            subject to A·g ≤ M               (M = current peak of a*a)
                       0 ≤ g_i ≤ 1000

         (`scipy.optimize.linprog`, HiGHS).  
       – If the optimum has a larger total mass than the current
         distribution, normalise it to sum = 1 and mix it with `a`
         (`a ← (1‑α)·a + α·g_norm`, α≈0.4).  This never makes the peak
         larger because `a*a ≤ M`, `g_norm*g_norm` is tiny (the scaling
         makes the entries ≪ M) and the mixed cross‑term is ≤ M.

   * b) *fast stochastic mass transfer* (≈70 % of the time).  
       – Pick a “donor’’ index with probability proportional to its
         contribution to the current peak (`a_i·a_{k_max‑i}`).  
       – Pick a “recipient’’ that currently carries little mass.  
       – Move a random fraction `δ ≤ 0.5·a_donor` and update the
         self‑convolution incrementally (O(n)).  
       – Accept the move if it improves the score, otherwise with a
         Metropolis rule based on a linearly cooling temperature.

3. Restarts: if no improvement is seen for ~35 s, re‑initialise from
   the best known vector with a mild log‑normal jitter.

4. The best vector seen throughout the run is returned (clipped to the
   limits enforced by `evaluate_sequence`).  All helper functions are
   top‑level, no closures, and the routine respects the 5‑second safety
   margin.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(v: np.ndarray) -> np.ndarray:
    """Linear self‑convolution `v*v` using an FFT of appropriate length."""
    n = v.size
    fft_len = _next_pow2(2 * n - 1)
    fv = np.fft.rfft(v, n=fft_len)
    fc = fv * fv
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * n - 1]
    return conv


def _apply_move(a: np.ndarray,
                c: np.ndarray,
                i: int,
                j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Transfer `delta` from index `i` to `j` (both in `a`) and update
    the self‑convolution `c` incrementally.  Returns the new `a` and `c`."""
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # linear part of the convolution change (2·δ·a shifted)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic part (δ²) – three positions may be affected
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _build_toeplitz(a: np.ndarray) -> sp.csr_matrix:
    """Return the (2n‑1)×n Toeplitz matrix `A` such that `A @ g`
    equals the linear convolution `a * g`.  Uses `scipy.sparse.diags`."""
    n = a.size
    # each column i contributes a_i on the diagonal with offset i
    data = [np.full(n, a[i]) for i in range(n)]
    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _lp_refine(a: np.ndarray, M: float, alpha: float = 0.4) -> tuple[np.ndarray, bool]:
    """
    Perform one LP refinement step.
    Returns (new_a, improved_flag).
    """
    n = a.size
    # Build the Toeplitz matrix for the current a
    A = _build_toeplitz(a)

    # Upper‑bound for each convolution entry: M
    b_ub = np.full(2 * n - 1, M)

    # Bounds for the variables: 0 ≤ g_i ≤ 1000  (the evaluation function clips)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # Linear programme: maximise Σ g_i  → minimise –Σ g_i
    c_obj = -np.ones(n)

    # Solve; time limit ~10 s is enough for these modest sizes
    res = opt.linprog(c_obj,
                      A_ub=A,
                      b_ub=b_ub,
                      bounds=bounds,
                      method="highs",
                      options={"presolve": True, "time_limit": 10.0})

    if not res.success:
        return a, False

    g0 = res.x
    sum_g0 = g0.sum()
    # Only accept a direction that genuinely adds mass
    if sum_g0 <= 1.0 + 1e-9:
        return a, False

    # Normalise to sum = 1 and convexly mix with the current a
    g_norm = g0 / sum_g0
    a_new = (1.0 - alpha) * a + alpha * g_norm
    a_new = np.clip(a_new, 0.0, None)
    a_new /= a_new.sum()
    return a_new, True


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Hand‑crafted starting vectors (all sum to 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        # reuse a previous construction if it exists
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0 and np.isfinite(a0).all():
            a = np.clip(a0, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            n = 2048
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # --- No prior construction – try a few hand‑crafted starts
        n = 2048                          # a comfortable size for LP + FFT
        candidates = _initial_candidates(n, rng)

        best_score = np.inf
        best_vec = None
        for v in candidates:
            c = _self_conv(v)
            sc = 2.0 * n * c.max()
            if sc < best_score:
                best_score = sc
                best_vec = v.copy()
        a = best_vec

    n = a.size
    # guarantee strictly positive entries (avoids unbounded LP directions)
    eps = 1e-12
    a = np.maximum(a, eps)
    a /= a.sum()

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the current state
    # ------------------------------------------------------------------
    c = _self_conv(a)                # current self‑convolution
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # LP‑refinement schedule
    lp_interval = 30.0               # seconds between LP calls
    lp_alpha = 0.4                   # mixing coefficient after LP

    # Stochastic refinement schedule
    temp_start = 0.08
    temp_end = 1e-6

    # Re‑compute full convolution after this many accepted moves
    RECOMPUTE_EVERY = 300
    move_counter = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    last_lp = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # a) Occasionally run the LP refinement step
        # --------------------------------------------------------------
        if now - last_lp > lp_interval:
            M = c.max()
            a_candidate, improved = _lp_refine(a, M, alpha=lp_alpha)
            if improved:
                a = a_candidate
                c = _self_conv(a)
                cur_score = 2.0 * n * c.max()
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] LP refinement → score {best_score:.6f}")
            last_lp = now

        # --------------------------------------------------------------
        # b) Fast stochastic mass‑transfer moves (simulated annealing)
        # --------------------------------------------------------------
        # temperature (linear cooling)
        t_frac = (now - start_time) / total_time
        temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
        temperature = max(temperature, 1e-12)

        # current peak location
        k_max = int(c.argmax())

        # contribution of each index to the current peak
        partners = k_max - np.arange(n)
        mask = (partners >= 0) & (partners < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[partners[mask]]
        total_contrib = contrib.sum()

        # donor index (biased by contribution)
        if total_contrib > 0.0:
            prob = contrib / total_contrib
            donor = rng.choice(np.arange(n), p=prob)
        else:
            donor = rng.integers(0, n)

        # recipient index (biased towards small current mass)
        recv_w = 1.0 - a
        recv_w[donor] = 0.0
        if recv_w.sum() > 0.0:
            recv_w /= recv_w.sum()
            recipient = rng.choice(np.arange(n), p=recv_w)
        else:
            recipient = (donor + rng.integers(1, n)) % n

        # amount to move
        max_delta = 0.5 * a[donor]
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        # apply move incrementally
        a_new, c_new = _apply_move(a, c, donor, recipient, delta)
        new_score = 2.0 * n * c_new.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            a = a_new
            c = c_new
            cur_score = new_score
            move_counter += 1

            if move_counter % RECOMPUTE_EVERY == 0:
                c = _self_conv(a)
                cur_score = 2.0 * n * c.max()

        # occasional global jitter (≈2 % of iterations)
        if rng.random() < 0.02:
            # pick a random contiguous block and perturb multiplicatively
            width = rng.integers(1, max(2, n // 8))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.18))
            a_tmp = a.copy()
            a_tmp[start:start + width] *= factor
            a_tmp = np.clip(a_tmp, 0.0, None)
            a_tmp /= a_tmp.sum()
            a = a_tmp
            c = _self_conv(a)
            cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Keep track of the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation (≈35 s)
        # --------------------------------------------------------------
        if now - last_improve > 35.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
