def _project_simplex(v):
    # Efficient O(n log n) projection onto simplex {x>=0, sum=1}
    # Code similar to Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
propose_candidate
-----------------

A fast, deterministic search for a non‑negative step‑function (probability
vector) that minimizes the evaluation function

    score = 2·n·max_k (a ∗ a)_k / (sum a)^2

The algorithm proceeds in three phases:

1. **Analytic families** – generate candidates from three simple
   parametrised families (power‑difference, power‑law‑difference and
   exponential).  The best candidate of this cheap sweep is kept as the
   starting point.

2. **Projected sub‑gradient descent** – the true ‑∞ objective is
   non‑smooth, but its sub‑gradient at the current maximum convolution
   index can be written in closed‑form.  At each iteration we:
        • compute the full convolution by a single FFT,
        • locate the index `k*` where the convolution attains its maximum,
        • evaluate the sub‑gradient `g = 2·a ⋆ e_{k*}` (a simple
          O(n) operation),
        • take an additive step `a ← a – η·g`, clip negatives and project
          back onto the simplex (sum = 1, a_i ≥ 0).

   The step size `η` follows a linear schedule from a large initial value
   (`≈200`) down to a small final value (`≈5`).  Random jitter and a
   soft‑restart mechanism keep the search from stagnating.

3. **Return** – the best vector found (already normalised and clipped) is
   returned as a plain list, ready for `evaluate_sequence`.

The whole routine respects a user‑provided time budget (default 1000 s)
and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors of length `n` (full length = 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value for a probability vector (sum = 1)."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Analytic shape generators
# ----------------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^r – i^r, 0<r<1 (decreasing)."""
    if not (0.0 < r < 1.0):
        return None
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_shape_p(p: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^{-p} – (i+2)^{-p},  p>0."""
    if p <= 0.0:
        return None
    k = np.arange(1, n + 2, dtype=np.float64)          # 1 … n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_geo(r: float, n: int) -> np.ndarray | None:
    """Exponential family a_i ∝ r^i, 0<r<1."""
    if not (0.0 < r < 1.0):
        return None
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


# ----------------------------------------------------------------------
# Sub‑gradient for the max‑convolution index
# ----------------------------------------------------------------------
def _gradient_max_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k at the active index `k`.
    For the active index the gradient is
        g_i = 2 * a_{k-i}  (if 0 ≤ k-i < n, else 0)
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------

    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` while respecting the 1000 s time budget.

    n = int(kwargs.get("n", 3800))

    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n)               # uniform baseline
