    n = a.size

    end = start + n

"""
Propose a step‑function (non‑negative heights on the simplex) that beats the
previous bound (~1.5132).  

The algorithm works directly on the simplex by representing the sequence
`a` as a soft‑max of unconstrained log‑weights `x`.  It minimises a smooth
approximation of the objective  

    f(a) = max_k (a*a)_k

by using the log‑sum‑exp surrogate

    Fγ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)                (γ → ∞ ⇒ max)

The gradient of `Fγ` w.r.t. the raw weights `x` is obtained analytically.
A standard Adam optimizer (with a simple learning‑rate schedule) updates the
log‑weights.  The smoothing parameter `γ` is increased linearly from a small
value (γ≈0.5) to a larger one (γ≈30) so that the optimisation first sees a
well‑behaved landscape and later focuses on the true maximum.

Additional ingredients:

* **Restart** – if no improvement is observed for 30 s the current best
  solution is re‑initialised with moderate log‑normal noise.
* **Occasional jitter** – a multiplicative log‑normal perturbation of a random
  block (≈2 % of the iterations) helps escaping shallow local minima.
* **Best‑ever tracking** – the best sequence seen so far (according to the
  exact evaluation `2·n·max(conv)`) is kept and printed.

All convolutions are performed with the FFT (O(N log N)) and the gradient
is obtained with a single FFT‑based correlation, keeping the per‑iteration
cost low enough to run many thousands of steps within the 1000 s budget.

The function returns the best sequence found as a plain Python list of
floats (each clipped to the limits imposed by `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient ∂F/∂a for a smoothed max‑objective,
    where ``w`` is the soft‑max weight vector (length 2·n‑1).
    Result: vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    # slice the part that corresponds to Σ_j a_j w_{i+j}
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initial log‑weights
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)                     # start from log‑weights
        else:
            n = 3840
            x = np.zeros(n, dtype=np.float64)         # uniform on the simplex
    else:
        n = 3840
        x = np.zeros(n, dtype=np.float64)             # uniform log‑weights

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)               # for a*a
    fft_len_corr = _next_pow2(3 * n - 2)               # for correlation a_rev * w

    # ------------------------------------------------------------------
    # 3) Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # 4) Tracking best solution
    # ------------------------------------------------------------------
    def _a_from_x(x_vec: np.ndarray) -> np.ndarray:
        return _softmax(x_vec)

    a = _a_from_x(x)
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 30.0                      # final γ → true max
    lr_start = 0.5
    lr_end = 0.02                         # linear decay of the base learning‑rate

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_time = elapsed / total_time

        # ---------- current iterate ----------
        a = _a_from_x(x)                     # probability vector (sum = 1)

        # ---------- self‑convolution ----------
        c = _self_conv(a, fft_len_self)      # length 2·n‑1

            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 7) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
