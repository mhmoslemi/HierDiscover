"""
Improved step‑function optimizer.

The goal is to minimize
    score = 2·n·max_k (a * a)_k / (sum a)^2
where `*` is a linear convolution.
For a non‑negative vector with unit sum the denominator disappears and the
objective reduces to `2·n·max(conv)`.  The code explores a variety of analytic
families (power‑differences, power‑law differences, geometric, logistic and
truncated uniform), keeps the best candidate and then refines it with an
Adam‑style exponentiated‑gradient on a soft‑max surrogate of the max‑convolution.
Additional random perturbations and restarts keep the search from getting stuck.
The routine respects the 1000 s time budget and returns a list of heights ready
for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of
    two vectors of length `n` (full length = 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max with temperature `beta`."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k (a*a)_k  w.r.t. `a`.
    Returns 2·a ⋆ w (correlation) of length |a|.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a unit‑sum vector `a`."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


def _gen_shape_r(r: float, n: int) -> np.ndarray:
    """a_i = (i+1)^r - i^r  (decreasing for 0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_shape_p(p: float, n: int) -> np.ndarray:
    """a_i = (i+1)^(-p) - (i+2)^(-p)  (p>0)."""
    k = np.arange(1, n + 2, dtype=np.float64)  # 1 .. n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_shape_geo(lam: float, n: int) -> np.ndarray:
    """Geometric shape a_i ∝ lam**i  (0<lam<1)."""
    i = np.arange(n, dtype=np.float64)
    a = lam ** i
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_shape_logistic(k: float, n: int) -> np.ndarray:
    """Logistic decreasing shape centred at n/2."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(k * (i - n / 2.0)))
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_shape_trunc(K: int, n: int) -> np.ndarray:
    """Uniform on the first K positions, zeros elsewhere."""
    if K <= 0:
        return None
    a = np.zeros(n, dtype=np.float64)
    a[:K] = 1.0 / K
    return a


def _project_decreasing(a: np.ndarray) -> np.ndarray:
    """Simple projection onto the cone of non‑increasing vectors."""
    # enforce a[i] >= a[i+1] by iterating forward
    out = a.copy()
    for i in range(1, out.size):
        if out[i] > out[i - 1]:
            out[i] = out[i - 1]
    # renormalise to keep sum = 1
    s = out.sum()
    if s <= 0.0:
        return a
    return out / s

    # 2) Random convex combinations of the top few candidates
    # ---------------------------------------------------------------

        cand = mix * a1 + (1.0 - mix) * a2
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand = cand / s
            try_candidate(cand, "mix")
            pool.append(cand)

    # ------------------------------------------------------------------
    # 3) Refinement with Adam‑style mirror descent on a soft‑max surrogate
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # small safety margin

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # schedules (linear in time)
    lr_start, lr_end = 0.8, 0.005
    beta_start, beta_end = 0.2, 10000.0

    # state for Adam
    logits = np.log(best_a + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    cur_score = best_score
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # linear schedules for learning rate and soft‑max temperature
        lr = lr_start * (1.0 - elapsed / total) + lr_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # current probability vector
        a = np.exp(logits)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # convolution and exact objective (for bookkeeping)
        conv = _conv_self(a, fft_len)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # soft‑max weights over the full convolution (smooth approx of max)
        w = _softmax(conv, beta)

        # gradient of the smooth surrogate w.r.t. a
        grad_a = _gradient_via_correlation(a, w)

        # map gradient to logits (the natural gradient of the KL‑mirror)
        avg = np.dot(a, grad_a)
        grad_logit = a * (grad_a - avg)

        # Adam update on logits
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_logit
        v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        logits -= step

        # occasional random perturbations to escape plateaus
        if rng.random() < 0.03:                     # block perturbation
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.30))
            a[start:start + width] *= mult
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            logits = np.log(a + eps)

        if rng.random() < 0.04:                     # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-14:
                delta = rng.random() * a[i] * 0.5
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                logits = np.log(a + eps)

        # restart if no improvement for a while
        if now - last_improve > 150.0:
            jitter = np.exp(rng.normal(0.0, 0.2, size=n))
            a = best_a * jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            logits = np.log(a + eps)
            m[:] = 0.0
            v[:] = 0.0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

        # optional monotonic projection every few hundred steps
        if iteration % 500 == 0:
            a = _project_decreasing(a)
            logits = np.log(a + eps)

    # ------------------------------------------------------------------
    # Return the best vector (already normalised, clipped to security limits)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
