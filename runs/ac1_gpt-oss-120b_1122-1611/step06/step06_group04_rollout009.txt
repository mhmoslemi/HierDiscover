"""
Propose a non‑negative step function that (approximately) minimizes
evaluate_sequence().

Algorithm
---------
1. Choose a length n (defaults to the length of the previously found
   construction if it exists, otherwise 3800).

2. Coarse parametric search:
   - Try families a_i = (i+γ+1)^p – (i+γ)^p for a grid of (p,γ).
   - Normalise each candidate to unit sum.
   - Evaluate 2·n·max(a∗a) using an FFT‑based linear convolution.
   - Keep the best candidate.

3. Simulated‑annealing refinement:
   - Maintain the current vector a (unit sum) and its convolution conv.
   - Repeatedly move a small amount ε from a randomly chosen index i
     to another index j (ε = step_frac·a_i, with a decreasing step_frac).
   - The effect on conv is updated in O(n) by slice‑adds; the new
     maximum is inspected.
   - Accept moves that lower the peak, otherwise accept with a
     Metropolis probability based on a temperature schedule.
   - Every `restart_gap` seconds without improvement a jittered restart
     from the best solution is performed.

4. Final Adam‑style optimisation on a smooth soft‑max surrogate
   of the max‐convolution (the same surrogate used in the reference
   implementation but with a more aggressive learning‑rate schedule).

The routine respects the 1000 s budget, prints intermediate scores,
and finally returns the best sequence found (clipped to the limits of
evaluate_sequence).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length for a linear convolution of size n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for FFT convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (used in the final Adam phase).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w  (correlation).
    ``a`` is length n, ``w`` length 2·n‑1.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("Simplex projection requires a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Score (the quantity returned by evaluate_sequence for a unit‑sum vector).
# ----------------------------------------------------------------------
def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv) – the objective for a unit‑sum vector."""
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Grid search over the (p,γ) family a_i = (i+γ+1)^p – (i+γ)^p.
# ----------------------------------------------------------------------
def _grid_search(n: int, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Return the best sequence found in a modest grid search."""
    best_score = np.inf
    best_seq = None
    fft_len = _fft_len_for_n(n)

    # coarse grids – a few dozen candidates are cheap
    p_vals = np.linspace(0.40, 0.60, 11)      # exponent p ∈ [0.4,0.6]
    gamma_vals = np.linspace(0.0, 4.0, 17)    # shift γ ∈ [0,4]

    idx = np.arange(n, dtype=np.float64)
    for p in p_vals:
        # pre‑compute (idx+γ)^p for each γ in a vectorised way
        for gamma in gamma_vals:
            a = (idx + gamma + 1.0) ** p - (idx + gamma) ** p
            total = a.sum()
            if total == 0:
                continue
            a /= total                         # unit‑sum normalisation
            conv = _conv_self(a, fft_len)
            cur_score = _score_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
    return best_seq, best_score


# ----------------------------------------------------------------------
# Simulated‑annealing refinement (mass‑transfer moves).
# ----------------------------------------------------------------------
def _simulated_annealing(a: np.ndarray,
                         conv: np.ndarray,
                         best_seq: np.ndarray,
                         best_score: float,
                         start: float,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Perform a Metropolis‑style random walk on the simplex.
    Returns the best (a, score) pair observed.
    """
    n = a.size
    cur_max = np.max(conv)
    cur_score = 2.0 * n * cur_max
    last_improve = start

    # preparation for the incremental convolution update
    conv_len = conv.size
    # schedules
    restart_gap = 120.0            # seconds without improvement → jitter restart
    step_start = 0.20
    step_end = 0.001
    temp_start = cur_max * 0.1    # initial temperature proportional to peak
    temp_end = 1e-12

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        ratio = min(1.0, elapsed / total)

        # temperature and step‑size schedules
        T = temp_start * (1.0 - ratio) + temp_end * ratio
        step_frac = step_start * (1.0 - ratio) + step_end * ratio

        # pick donor i (must have some mass) and recipient j ≠ i
        i = rng.integers(0, n)
        while a[i] <= 1e-15:                     # avoid near‑zero donors
            i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        eps = step_frac * a[i]
        if eps < 1e-15:
            continue

        # --------------------------------------------------------------
        # Build a candidate convolution = conv + Δ (Δ computed via slices)
        # --------------------------------------------------------------
        cand_conv = conv.copy()
        # contribution of the cross‑terms a * d (both orders → factor 2)
        cand_conv[i:i + n] += -2.0 * eps * a
        cand_conv[j:j + n] +=  2.0 * eps * a
        # d*d term (only 3 distinct indices)
        ii = 2 * i
        jj = 2 * j
        ij = i + j
        cand_conv[ii] += eps * eps
        cand_conv[jj] += eps * eps
        cand_conv[ij] -= 2.0 * eps * eps

        new_max = np.max(cand_conv)
        new_score = 2.0 * n * new_max

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability of uphill move
            prob = np.exp(-(new_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # commit the move
            a[i] -= eps
            a[j] += eps
            conv = cand_conv
            cur_max = new_max
            cur_score = new_score

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best SA score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Restart (jitter) if stuck for a while
        # ------------------------------------------------------------------
        if now - last_improve > restart_gap:
            a = best_seq.copy()
            # small multiplicative jitter
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            a *= jitter
            a = _proj_simplex(a)
            conv = _conv_self(a, _fft_len_for_n(n))
            cur_max = np.max(conv)
            cur_score = 2.0 * n * cur_max
            last_improve = now
            print(f"[{elapsed:6.2f}s] SA restart (jitter) – best score {best_score:.6f}")

    return best_seq, best_score


# ----------------------------------------------------------------------
# Adam optimisation on a smooth soft‑max surrogate of the max.
# ----------------------------------------------------------------------
def _adam_refine(a: np.ndarray,
                 best_seq: np.ndarray,

                 deadline: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:

    n = a.size
    fft_len = _fft_len_for_n(n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # schedule parameters
    beta_start = 0.5
    beta_end = 2500.0
    lr_start = 0.25
    lr_end = 0.001

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        ratio = min(1.0, elapsed / total)

        beta = beta_start + (beta_end - beta_start) * ratio
        lr = lr_start * (1.0 - ratio) + lr_end * ratio

        conv = _conv_self(a, fft_len)
        cur_max = np.max(conv)
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{elapsed:6.2f}s] new best Adam score = {best_score:.6f}")

        # soft‑max weights over the convolution entries
        diff = conv - cur_max               # ≤ 0
        w_raw = np.exp(beta * diff)
        w_sum = w_raw.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            mask = conv == cur_max
            w_raw = np.where(mask, 1.0, 0.0)
            w_sum = w_raw.sum()
        w = w_raw / w_sum

        # gradient of the smooth surrogate
        grad = _gradient_correlation(a, w)

        # Adam step
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _proj_simplex(a)

        # occasional noise to avoid stagnation
        if rng.random() < 0.02:
            factor = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * factor
            a = _proj_simplex(a)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point expected by the outer harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Length of the step sequence (defaults to 3800 or to the length of the
        previously discovered construction).

    Returns
    -------
    list[float]
        The best sequence found (values clipped to the limits imposed by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    # --------------------------------------------------------------
    # choose problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    # if a previous solution exists, adopt its length (it is usually good)
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand /= s
                    n = cand.size
                    # use this candidate as a warm start
                    best_seq = cand.copy()
                    best_score = _score_from_conv(_conv_self(best_seq, _fft_len_for_n(n)), n)
                else:
                    best_seq = None
                    best_score = np.inf
            else:
                best_seq = None
                best_score = np.inf
        except Exception:
            best_seq = None
            best_score = np.inf
    else:
        best_seq = None
        best_score = np.inf

    # --------------------------------------------------------------
    # 1) parameter grid search (fast)
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        seq_grid, score_grid = _grid_search(n, rng)
        if score_grid < best_score:
            best_seq, best_score = seq_grid, score_grid

    # --------------------------------------------------------------
    # 2) simulated‑annealing refinement
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback: a simple sqrt‑difference start
        idx = np.arange(n, dtype=np.float64)
        best_seq = np.sqrt(idx + 1) - np.sqrt(idx)
        best_seq /= best_seq.sum()
        best_score = _score_from_conv(_conv_self(best_seq, _fft_len_for_n(n)), n)

    # initialise convolution for the SA phase
    conv = _conv_self(best_seq, _fft_len_for_n(n))
    if time.monotonic() < deadline:
        best_seq, best_score = _simulated_annealing(
            best_seq,
            conv,
            best_seq,
            best_score,
            start_time,
            deadline,
            rng,
        )

    # --------------------------------------------------------------
    # 3) final Adam‑type refinement on the soft‑max surrogate
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        # reuse the best sequence as starting point
        best_seq, best_score = _adam_refine(
            best_seq,
            best_seq,
            best_score,
            start_time,
            deadline,
            rng,
        )

    # --------------------------------------------------------------
    # 4) Return the best sequence (apply evaluate_sequence clipping).
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
