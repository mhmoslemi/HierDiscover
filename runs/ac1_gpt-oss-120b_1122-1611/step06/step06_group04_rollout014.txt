"""
A fast stochastic search for a step‐function that minimises
``evaluate_sequence``.  The idea is simple:

* The objective for a non‑negative sequence a (after normalising
  ``sum(a)=1``) is   score = 2·n·max_k (a*a)_k .
* We explore a few low‑dimensional families of sequences
  (power‑law, shifted power‑law, the classical “sqrt‑difference’’)
  and keep the best one found.
* Around the current best we perform cheap random perturbations
  (multiplicative log‑normal noise) and accept only improvements.
  This refines the solution without any heavy gradient machinery.
* All operations are O(n log n) thanks to an FFT based convolution,
  so many thousands of candidates can be evaluated well inside the
  1000 s budget.

The returned sequence is clipped to the limits imposed by
``evaluate_sequence`` (non‑negative, ≤1000) and is therefore ready
to be scored directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length required for a linear convolution.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Fast evaluation of the objective (identical to evaluate_sequence
# after normalising the sum to 1 and assuming all entries are ≤1000).
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a normalised to unit sum."""
    total = a.sum()
    if total < 0.01:          # matches the protection inside evaluate_sequence
        return float("inf")
    a = a / total
    n = a.size
    fft_len = _fft_len_for_n(n)
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Families of candidate sequences.
# ----------------------------------------------------------------------
def _power_law(p: float, offset: float, n: int) -> np.ndarray:
    """a_i ∝ (i + offset + 1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + offset + 1.0
    a = np.power(idx, -p)
    s = a.sum()
    if s == 0.0:
        return None
    return a / s


def _sqrt_diff(n: int) -> np.ndarray:
    """Classic a_i = √(i+1)‑√i, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _uniform(n: int) -> np.ndarray:
    """Flat distribution (unit‐sum)."""
    a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


# ----------------------------------------------------------------------
# Main entry point required by the challenge infrastructure.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low
    ``evaluate_sequence`` score.  The routine runs for at most
    ``budget_s`` seconds and returns the best vector discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (use the length of the previously found best if it
    # exists, otherwise fall back to a reasonable default).
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3809))
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # State variables.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    best_params = None          # (p, offset) for the power‑law family

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # slight safety margin

    # ------------------------------------------------------------------
    # Helper to test a candidate and keep the best.
    # ------------------------------------------------------------------
    def _try(seq: np.ndarray, params=None):
        nonlocal best_seq, best_score, best_params
        if seq is None:
            return
        score = _fast_score(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            best_params = params
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Simple baselines.
    # ------------------------------------------------------------------
    _try(_uniform(n))
    _try(_sqrt_diff(n))

    # ------------------------------------------------------------------
    # 2) Coarse grid over the power‑law family.
    # ------------------------------------------------------------------
    p_grid = np.linspace(0.05, 1.00, 45)                     # 45 points
    off_grid = np.linspace(0.0, n * 0.9, 30)                # 30 points
    for p in p_grid:
        for off in off_grid:
            if time.monotonic() >= deadline:
                break
            cand = _power_law(p, off, n)
            _try(cand, params=(p, off))
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 3) Refine around the best power‑law parameters (if any).
    # ------------------------------------------------------------------
    if best_params is not None:
        p0, off0 = best_params
        sigma_p = 0.02
        sigma_off = n * 0.02
        for _ in range(2500):
            if time.monotonic() >= deadline:
                break
            p = np.clip(rng.normal(p0, sigma_p), 0.01, 1.20)
            off = np.clip(rng.normal(off0, sigma_off), 0.0, n * 0.99)
            cand = _power_law(p, off, n)
            _try(cand, params=(p, off))
            # shrink the exploration radius slowly
            sigma_p *= 0.9995
            sigma_off *= 0.9995

    # ------------------------------------------------------------------
    # 4) Random Dirichlet draws – a blind global search.
    # ------------------------------------------------------------------
    for _ in range(1200):
        if time.monotonic() >= deadline:
            break
        conc = rng.uniform(0.05, 5.0)           # concentration of the Dirichlet
        cand = rng.dirichlet(np.full(n, conc))
        _try(cand)

    # ------------------------------------------------------------------
    # 5) Local stochastic refinement of the current best solution.
    #    Multiplicative log‑normal jitter is cheap and respects the
    #    simplex constraint after renormalisation.
    # ------------------------------------------------------------------
    if best_seq is None:
        # very unlikely, but keep a safe fallback.
        best_seq = _uniform(n)

    cur = best_seq.copy()
    cur_score = best_score
    jitter_sigma = 0.12
    while time.monotonic() < deadline:
        # propose a jittered version
        factor = np.exp(rng.normal(0.0, jitter_sigma, size=n))
        proposal = cur * factor
        proposal /= proposal.sum()
        prop_score = _fast_score(proposal)
        if prop_score < cur_score:
            cur = proposal
            cur_score = prop_score
            if prop_score < best_score:
                best_seq = proposal.copy()
                best_score = prop_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] improved via jitter → {best_score:.6f}")
        # slowly tighten the jitter
        jitter_sigma *= 0.9998

    # ------------------------------------------------------------------
    # 6) Return the best sequence (apply the clipping that evaluate_sequence
    #    performs internally).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
