def _optimise_one_length_adam(a_init, deadline, rng):
    a = a_init.copy()
    n = a.size
    fft_self = _next_pow2(2*n - 1)
    fft_grad = _next_pow2(3*n - 2)
    # Initialize Adam states
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # hyperparams for schedule
    beta_start = 1.0
    beta_end = 5000.0
    alpha_start = 0.4
    alpha_end = 0.001

    # Timing control
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.55 * total_time  # 55% for softmax

    best = a.copy()
    conv = _self_conv(a, fft_self)
    best_score = 2.0 * n * conv.max()
    best_score_time = start_time

    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        t += 1

        if now < phase1_end:
            # softmax gradient
            frac = (now - start_time) / (phase1_end - start_time)
            beta = beta_start + (beta_end - beta_start) * frac
            alpha = alpha_start * (1.0 - frac) + alpha_end * frac

            conv = _self_conv(a, fft_self)
            max_c = conv.max()
            # softmax
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            # compute gradient
            rev_a = a[::-1]
            fp = np.fft.rfft(p, n=fft_grad)
            fra = np.fft.rfft(rev_a, n=fft_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_grad)[: 3*n - 2]
            grad = 2.0 * grad_full[n-1 : n-1 + n]
        else:
            # subgradient of max entry
            conv = _self_conv(a, fft_self)
            k_max = int(conv.argmax())
            g = np.zeros_like(a)
            i_min = max(0, k_max - (n - 1))
            i_max = min(n - 1, k_max)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k_max - idx_i
            g[idx_i] = 2.0 * a[idx_j]
            grad = g
            # step size diminishing: use alpha schedule
            # we can keep alpha same as last frac?
            # compute fraction of total after phase1_end
            frac2 = (now - phase1_end) / (deadline - phase1_end)
            # decaying alpha principle
            alpha = alpha_end * (1.0 - frac2) + 0.0005 * frac2  # small

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # projection onto simplex
        a = _project_simplex(a)

        # evaluate
        conv = _self_conv(a, fft_self)
        cur_score = 2.0 * n * conv.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] n={n:5d} it={t:6d} new best {best_score:.6f}")

        # stagnation: if no improvement for 20 sec, jitter
        if now - last_improve > 20.0:
            # jitter best
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = _project_simplex(a)
            # reset Adam moments to zero to avoid stale momentum
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] n={n:5d} jitter (stagnation)")

    return best, best_score

"""
Propose a step‑function with a lower bound than the previous best (~1.5136).

Algorithm
---------
* **Multi‑length search** – try a handful of lengths (including the length of the
  previous construction).  The bulk of the time budget is spent on the most
  promising length, the rest is split among the others.

* **Adam‑accelerated projected optimisation** – for a fixed length we minimise
  the smoothed objective
        f_β(a) = Σ_i p_i(β)·(a*a)_i  ,   p_i(β) = softmax_β((a*a)_i)
  which converges to the true max as the temperature β→∞.  The gradient of
  f_β is obtained by a single FFT‑based convolution.  An Adam update is applied
  and the iterate is projected onto the probability simplex.

* **Exact sub‑gradient phase** – after half of the allocated time the temperature
  β has become large; we switch to the true sub‑gradient of the peak
  convolution entry.  Adam is still used (with a decaying step‑size) to obtain
  smoother progress.

* **Stagnation handling** – if no improvement is observed for 20 s we
  jitter the current best solution with multiplicative log‑normal noise,
  reset the Adam moments and continue.

* **Final polishing** – a short random‑perturbation hill‑climb is performed
  after the main optimisation.

All iterates are kept on the simplex (∑a_i = 1, a_i ≥ 0); consequently the
objective value returned by this routine equals the value that
`evaluate_sequence` would compute (since the denominator is fixed to 1).
The function respects the 1000 s budget and returns the best sequence found.

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length *fft_len*."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a 1‑D array onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  O(n log n) implementation.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # Phase 2 – exact sub‑gradient of the current peak
        # --------------------------------------------------------------
        else:
            conv = _self_conv(a, fft_len_self)
            k_max = int(conv.argmax())

            # sub‑gradient g_i = 2·a_{k_max‑i} when the index exists
            g = np.zeros_like(a)
            i_min = max(0, k_max - (n - 1))
            i_max = min(n - 1, k_max)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k_max - idx_i
            g[idx_i] = 2.0 * a[idx_j]
            grad = g

            # slowly shrinking step size after the switch
            frac2 = (now - phase1_end) / (deadline - phase1_end)
            alpha = alpha_end * (0.5 + 0.5 * (1.0 - frac2))

        # ---------------------- Adam update ---------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step

        # ---------------------- Simplex projection --------------------
        a = _project_simplex(a)

        # ---------------------- Evaluation ----------------------------
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] n={n:5d} it={t:7d} new best {best_score:.6f}")

        # ---------------------- Stagnation handling -----------------
        if now - last_improve > 20.0:
            # jitter around the best solution, reset Adam moments
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] n={n:5d} jitter (stagnation)")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (list of floats) within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose candidate lengths.
    #    We always include the length of the previous construction (if any)
    #    and a few other sizes that were competitive in earlier runs.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 3000, 3809, 5000, 6000, 8000]
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
            if seq0.size not in lengths:
                lengths.insert(0, seq0.size)
            else:
                # move it to the front (most time will be spent on it)
                lengths.remove(seq0.size)
                lengths.insert(0, seq0.size)

    best_overall = None
    best_score_overall = float("inf")

    # ------------------------------------------------------------------
    # 2) Optimise each length in turn while time permits.
    #    The first (most promising) length receives at most 70 % of the
    #    remaining budget; the others share what is left.
    # ------------------------------------------------------------------
    for n in lengths:
        now = time.monotonic()
        if now >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a probability vector for this length.
        # --------------------------------------------------------------
        if "height_sequence_1" in globals():
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq.size == n:
                a0 = seq.copy()
            else:
                # linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, seq.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(x_new, x_old, seq)
        else:
            a0 = rng.dirichlet(np.ones(n)).astype(np.float64)

        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / n)

        # --------------------------------------------------------------
        # Allocate a slice of the remaining time for this length.
        # --------------------------------------------------------------
        remaining = deadline - now
        # at most 70 % of what is left, but never more than 300 s
        max_for_this = min(remaining * 0.70, 300.0)
        length_deadline = now + max_for_this

        # --------------------------------------------------------------
        # Run the optimiser.
        # --------------------------------------------------------------
        vec, score = _optimise_one_length_adam(a0, length_deadline, rng)

        # --------------------------------------------------------------
        # Track the best overall solution.
        # --------------------------------------------------------------
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - (deadline - (budget_s - 5.0))
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # 3) Short random‑perturbation polishing if time still remains.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # use the current best length (or fall back to 3809)
        n = best_overall.size if best_overall is not None else 3809
        a = best_overall.copy() if best_overall is not None else rng.dirichlet(np.ones(n))
        # small multiplicative noise
        noise = np.exp(rng.normal(0.0, 0.03, size=n))
        a = a * noise
        a = _project_simplex(a)
        conv = _self_conv(a, _next_pow2(2 * n - 1))
        score = 2.0 * n * conv.max()
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = a.copy()
            elapsed = time.monotonic() - (deadline - (budget_s - 5.0))
            print(f"[{elapsed:6.2f}s] *** POLISH BEST n={n} score={best_score_overall:.6f}")
        if time.monotonic() > deadline - 0.1:
            break

    # ------------------------------------------------------------------
    # 4) Fallback – if nothing was found (should never happen).
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    # 5) Clip to limits enforced by ``evaluate_sequence`` and return.
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    return [float(x) for x in best_overall]
```

This implementation uses an Adam‑based projected gradient descent with a soft‑max smoothing of the max‑convolution, switches to the exact sub‑gradient after half the allotted time, restarts with jitter on stagnation, explores several sequence lengths, and finishes with a brief random‑perturbation polishing phase. It stays within the 1000 s budget and returns the best non‑negative step‑function found.
