```
def _project_simplex(v):
    # v is array
    # algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

assistantfinal**Solution Overview**

The optimizer builds a population of candidate step‑functions (vectors that sum to 1 and are non‑negative) and evolves it with a simple evolutionary strategy:

* **Initialization** – the current best construction (`height_sequence_1`) is inserted into the population and the remaining members are drawn from a Dirichlet distribution.
* **Evaluation** – the objective is  
  `score(a) = 2·n·max(conv(a,a))`  
  (exactly the value returned by `evaluate_sequence` for a normalized vector).  
  Convolution is performed with a real FFT for speed.
* **Variation operators** – each generation creates a new offspring by
  1. **Crossover** – convex combination of two parents,
  2. **Block perturbation** – multiplicative log‑normal perturbation of a random block,
  3. **Pairwise mass transfer** – move a small fraction of mass from one index to another,
  4. **Sub‑gradient descent step** – a few additive sub‑gradient moves followed by projection onto the simplex.
* **Selection** – the union of parents and offspring is sorted by score and the best `popsize` individuals survive.
* **Polishing** – when the time budget is almost exhausted a short deterministic sub‑gradient descent run (projected gradient) refines the current best candidate.

The algorithm is completely self‑contained, uses only `numpy` (and `time` for the budget) and respects the 1000 s limit.  It has been tuned to favour exploitation (the population quickly concentrates around good solutions) while still keeping a healthy amount of exploration.

```python
"""
Evolutionary optimisation of a non‑negative step function.
The objective is  score(a) = 2·n·max(conv(a,a))   (for ∑a = 1).
The routine builds a population, applies crossover, block‑perturbations,
pairwise mass transfers and a few sub‑gradient steps, and keeps the best
individuals.  A final short projected‑gradient polish is applied before
returning the candidate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector that already sums to 1."""
    n = a.size
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    The worst index i_max contributes 2·a_{i_max‑j} for admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


# ----------------------------------------------------------------------
# Random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    The block width is at most n//30 (but at least 1).
    """
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out = np.full_like(out, 1.0 / n)
    else:
        out /= out_sum
    return out


def _pairwise_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Transfer a random fraction of mass from one index to another.
    The move is kept only if it reduces the objective; otherwise it is discarded.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    # at most 20 % of the source mass may be moved
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    # renormalise (still sums to 1)
    fft_len = _fft_len_for_len(2 * n - 1)
    old_score = _score(a, fft_len)
    new_score = _score(a2, fft_len)
    if new_score < old_score:
        return a2
    return a


def _subgrad_refine(a: np.ndarray,
                    steps: int,
                    rng: np.random.Generator,
                    deadline: float) -> np.ndarray:
    """
    A few additive sub‑gradient steps with projection onto the simplex.
    Used as a cheap local polish inside the evolutionary loop.
    """
    n = a.size
    fft_len = _fft_len_for_len(2 * n - 1)
    for t in range(1, steps + 1):
        if time.time() >= deadline:
            break
        conv = _conv_self(a, fft_len)
        grad = _subgradient_step(a, conv)
        lr = 0.5 / np.sqrt(t)            # decreasing learning rate
        a = a - lr * grad
        a = _project_simplex(a)
    return a

        combined = population + offspring
        combined_scores = np.concatenate([scores, off_scores])
        # keep the best `popsize` individuals
        best_keep = np.argsort(combined_scores)[:popsize]
        population = [combined[i] for i in best_keep]
        scores = combined_scores[best_keep]

        # ---- update global best ----
        if scores[0] < best_score:
            best_score = float(scores[0])
            best_vec = population[0].copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Occasionally run a short polishing sub‑gradient descent on the current best.
        # This costs almost nothing and often nudges the score a bit lower.
        # ------------------------------------------------------------------
        if time.time() < deadline - 5.0 and rng.random() < 0.12:
            polished = _subgrad_refine(best_vec.copy(),
                                        steps=120,
                                        rng=rng,
                                        deadline=deadline)
            polished_score = _score(polished, fft_len_conv)
            if polished_score < best_score:
                best_score = float(polished_score)
                best_vec = polished.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.2f}s] Polished best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final deterministic polish (as much time as we still have)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        final = _subgrad_refine(best_vec.copy(),
                                steps=2000,
                                rng=rng,
                                deadline=deadline)
        final_score = _score(final, fft_len_conv)
        if final_score < best_score:
            best_score = float(final_score)
            best_vec = final.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:6.2f}s] Final polish: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety clipping (evaluate_sequence does the same, but we keep it explicit)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # The harness expects a plain Python list of floats
    return [float(x) for x in best_vec]
```
