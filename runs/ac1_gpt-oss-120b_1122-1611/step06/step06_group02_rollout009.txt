def _init_arcsine(rng, n):
    idx = np.arange(n, dtype=float)
    values = 1.0 / np.sqrt( (idx + 0.5) * (n - 0.5 - idx) )
    # values become large near ends; clip to maybe top 1e4? But after normalization they become fine.
    values = np.clip(values, 0, None)
    # Normalize
    if values.sum() == 0.0:
        return np.full(n, 1.0 / n)
    else:
        values /= values.sum()
    return values

def _local_optimise(initial, rng, deadline, max_iters=2000):
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2*n-1)
    fft_len_grad = _fft_len_for_len(3*n-2)
    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)
    cur_score = best_score
    beta = 0.5
    beta_mul = 1.05
    # temperature schedule for SA
    temp_start = cur_score * 0.05  # 5% of current score
    for t in range(1, max_iters+1):
        if time.time() >= deadline: break
        # Choose method: gradient or SA
        if rng.random() < 0.7:
            # gradient step
            conv = _conv_self(a, fft_len_conv)
            # subgradient or softmax
            if rng.random() < 0.5:
                raw_grad = _subgradient_step(a, conv)
            else:
                beta = min(1e5, beta * beta_mul)
                raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)
            lr = 0.6 / np.sqrt(t)
            grad_clip = np.clip(raw_grad, -20.0, 20.0)
            a = a * np.exp(-lr * grad_clip)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:

"""
Improved stochastic optimisation for the self‑convolution max‑norm.

The algorithm mixes many complementary ideas:

*   Simple initialisation families (Dirichlet, sine‑power, arcsine‐like,
    and interpolation from the current best).
*   A fast “exponentiated‑gradient” optimiser that alternates exact
    sub‑gradients of the worst convolution entry with a smooth
    log‑sum‑exp (soft‑max) gradient whose temperature β grows
    exponentially.
*   A Simulated‑Annealing (SA) layer that performs small random
    mass‑transfer moves and sometimes accepts a worsening move,
    helping the search escape plateaus.
*   Occasional random block perturbations and cheap pairwise moves
    for extra exploration.
*   A multi‑scale search: many restarts with varying lengths (up to
    20000 entries).  Larger n reduces the theoretical lower bound
    (≈2·n/(2·n‑1)≈1), so the search favours longer vectors when time permits.
*   A “polish” phase that runs a longer optimiser on the current global
    best after each successful improvement.

All heavy work uses real FFTs (O(N log N)) and stays well within the 1000 s
budget.  The returned list contains non‑negative floats; the harness’
`evaluate_sequence` performs the final clipping and scoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Scoring utilities (fast, avoid evaluate_sequence overhead)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return the value 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. ``a``.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max
        f_β(a) = (1/β)·log Σ_i exp(β·conv_i)
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))          # safe: max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    grad_part = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Simple random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a random fraction of mass from i to j (accept only if improving)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a


def _mass_transfer_candidate(a: np.ndarray,
                             rng: np.random.Generator) -> np.ndarray:
    """Create a new candidate by moving a small random mass from i to j."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    return a2


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponentials."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """Shape a_i ∝ sin^p(π·(i+0.5)/n), p∈[0.5,4]."""
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_arcsine(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Discrete analogue of the arcsine density:
        a_i ∝ 1 / sqrt((i+0.5)*(n-0.5-i))
    which puts more weight near the ends.
    """
    idx = np.arange(n, dtype=np.float64)
    eps = 1e-12
    vals = 1.0 / np.sqrt((idx + 0.5 + eps) * (n - 0.5 - idx + eps))
    vals = np.clip(vals, 0.0, None)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linearly interpolate an existing distribution to a different length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Local optimiser – exponentiated gradient + simulated annealing
# ----------------------------------------------------------------------


def _local_optimise(initial: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float,
                    max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Refine ``initial`` on the probability simplex.
    The method mixes exponentiated‑gradient steps (with sub‑ or soft‑max
    gradients) and a simulated‑annealing mass‑transfer move.
    Returns the best vector seen and its score.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    beta = 0.5
    beta_mul = 1.07

    # temperature for SA – start as a few percent of the current score
    temp_start = max(1e-3, 0.05 * best_score)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ------------------------------------------------------------------
        # Choose an elementary move
        # ------------------------------------------------------------------
        if rng.random() < 0.65:                     # gradient‑type move
            conv = _conv_self(a, fft_len_conv)
            if rng.random() < 0.55:
                raw_grad = _subgradient_step(a, conv)
            else:
                beta = min(1e5, beta * beta_mul)
                raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

            lr = 0.65 / np.sqrt(t)                 # 1/√t decay
            grad_clip = np.clip(raw_grad, -20.0, 20.0)
            a = a * np.exp(-lr * grad_clip)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

        else:                                       # simulated‑annealing move
            a_cand = _mass_transfer_candidate(a, rng)
            conv_cand = _conv_self(a_cand, fft_len_conv)
            cand_score = 2.0 * n * conv_cand.max()
            cur_score = 2.0 * n * _conv_self(a, fft_len_conv).max()
            if cand_score < cur_score:
                a = a_cand
            else:
                # linear cooling
                T = temp_start * max(0.0, 1.0 - t / max_iters)
                if T > 0.0 and rng.random() < np.exp(-(cand_score - cur_score) / T):
                    a = a_cand

        # ------------------------------------------------------------------
        # Occasional exploratory perturbations
        # ------------------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.02:
            a = _pairwise_move(a, rng, fft_len_conv)

        # ------------------------------------------------------------------
        # Keep the best ever seen
        # ------------------------------------------------------------------
        cur_score = 2.0 * n * _conv_self(a, fft_len_conv).max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

    return best_vec, best_score
