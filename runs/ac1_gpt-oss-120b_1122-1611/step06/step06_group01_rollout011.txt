from scipy.optimize import linprog

def solve_lp_max_sum(f, M):
    n = f.size
    # Build matrix A
    # Use zeros
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        # i ranges max(0, k-(n-1)) to min(k, n-1)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for i in [i_start,i_end], j = k - i
        A[k, i_start:i_end+1] = f[k - np.arange(i_start, i_end+1)]
        # This uses vectorized assignment.
    # Bound vector b = M * np.ones(2*n-1)
    # c = -np.ones(n) for maximizing sum(g)
    # bounds = [(0, None)] * n
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=M*np.ones(2*n-1), bounds=[(0,None)]*n, method='highs')
    if res.success:
        return res.x
    else:
        return None

"""
Propose a step‑function with a lower evaluation score.

Algorithm
---------
* The search works on a probability simplex (the coefficients are
  normalised to sum = 1, scaling does not affect the score).

* **Phase 1 – smoothed gradient descent** (≈ 50 % of the remaining time)
  – Approximate `max(a*a)` by a soft‑max with temperature β.
  – Compute the exact gradient of the smoothed objective via FFT:
        g = 2·(p * rev(a))                      (p = soft‑max weights)
  – Perform an *additive* projected gradient step and project back onto the
    simplex.

* **Phase 2 – exact sub‑gradient descent** (≈ 30 % of the time)
  – Locate the current peak index `k = argmax(a*a)`.
  – The sub‑gradient of the peak is `g_i = 2·a_{k-i}` (zero when out of range).
  – Update additively with a decreasing step size `η = η₀/√t` and project.

* **Phase 3 – linear‑programming polishing** (the rest of the budget)
  – For the current best vector `a` compute `M = max(a*a)`.
  – Solve the LP  
        maximise   Σ g_i
        subject to (a * g)_k ≤ M  (for all k) ,   g_i ≥ 0
    which finds a distribution that spreads the mass as much as possible
    while keeping the convolution below the present peak.
  – Normalise the LP solution to sum = 1 and mix it with the current best
    (`a ← (1−τ)·a + τ·ĝ`) with a small mixing factor τ.
  – Repeat while the score improves.

* **Stagnation handling** – after 30 s without improvement a
  multiplicative log‑normal jitter (≈ 10 % σ) is applied to the best
  vector and optimisation restarts.

* The routine tries several lengths (including the previous construction
  if available) and keeps the overall best sequence.  All helper
  functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# optional linear‑programming imports
try:
    from scipy.optimize import linprog
    from scipy.sparse import lil_matrix
except Exception:                     # pragma: no cover – scipy may be missing
    linprog = None
    lil_matrix = None


# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of prescribed length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# LP helper (only used for moderate lengths)
# ----------------------------------------------------------------------


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   maximise Σ g_i   subject to (f * g)_k ≤ M, g ≥ 0.
    Returns the optimal g (may have arbitrary sum) or None on failure.
    Uses a dense constraint matrix – safe for n ≤ ~1800.
    """
    if linprog is None:                        # pragma: no cover
        return None
    n = f.size
    rows = 2 * n - 1
    # build dense matrix A where A[k,i] = f[k-i] (zero when out of range)
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_arr = np.arange(i_start, i_end + 1, dtype=np.int32)
        A[k, i_arr] = f[k - i_arr]

    # linear programme (maximise sum(g) → minimise -sum(g))
    res = linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=M * np.ones(rows),
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        return res.x
    return None

def _optimise_one_length(a_init: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Runs three optimisation phases for a fixed length `n`.
    Returns the best vector (sum = 1) and its score
    (2 · n · max(a*a)).
    """
    a = a_init.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # initial statistics
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    last_improve = time.monotonic()
    it = 0

    start = time.monotonic()
    total = deadline - start
    # time slices
    phase1_end = start + 0.50 * total          # smoothed gradient
    phase2_end = start + 0.80 * total          # exact sub‑gradient

    # hyper‑parameters
    eta_start = 1.5
    eta_end = 0.02
    beta_start = 0.5
    beta_end = 5000.0
    sub_eta0 = 0.30

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        # --------------------------------------------------------------
        # Phase 1 – smoothed soft‑max gradient
        # --------------------------------------------------------------
        if now < phase1_end:
            frac = (now - start) / (phase1_end - start)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # self‑convolution
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()

            # soft‑max weights (stable)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            # gradient = 2·(p * rev(a))
            rev_a = a[::-1]
            fp = np.fft.rfft(p, n=fft_len_grad)
            fra = np.fft.rfft(rev_a, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            a = a - eta * grad

        # --------------------------------------------------------------
        # Phase 2 – exact sub‑gradient of the current peak
        # --------------------------------------------------------------
        elif now < phase2_end:
            conv = _self_conv(a, fft_len_self)
            k_max = int(conv.argmax())

            g = np.zeros_like(a)
            i_min = max(0, k_max - (n - 1))
            i_max = min(n - 1, k_max)
            i_idx = np.arange(i_min, i_max + 1)
            j_idx = k_max - i_idx
            g[i_idx] = 2.0 * a[j_idx]

            eta = sub_eta0 / np.sqrt(it)
            a = a - eta * g

        # --------------------------------------------------------------
        # Phase 3 – LP‑based polishing (only for modest n)
        # --------------------------------------------------------------
        else:
            # we exit the main loop and switch to a lightweight LP loop
            break

        # ---------------------------------
        # Projection onto simplex
        # ---------------------------------
        a = _project_simplex(a)

        # ---------------------------------
        # Evaluate current candidate (fast, no call to evaluate_sequence)
        # ---------------------------------
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}")

        # stagnation handling
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * noise
            a = _project_simplex(a)
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    # ------------------------------------------------------------------
    # Phase 3 – LP polishing (executed only if we have time left)
    # ------------------------------------------------------------------
    # we only attempt LP for n that are not too large (≈1800) – otherwise the
    # dense constraint matrix would be excessive.
    LP_MAX_N = 1800
    while time.monotonic() < deadline and n <= LP_MAX_N:
        conv = _self_conv(best, fft_len_self)
        M = conv.max()

        g_raw = _solve_lp_max_sum(best, M)
        if g_raw is None:
            break

        # Normalise the LP vector to sum = 1
        s = g_raw.sum()
        if s <= 0.0:
            break
        g_norm = g_raw / s

        # Mix a small amount of the LP solution into the current best.
        tau = 0.20
        cand = (1.0 - tau) * best + tau * g_norm
        cand = _project_simplex(cand)

        conv_cand = _self_conv(cand, fft_len_self)
        cand_score = 2.0 * n * conv_cand.max()

        if cand_score + 1e-12 < best_score:
            best_score = cand_score
            best = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] n={n:5d} LP‑improved score {best_score:.6f}")
        else:
            # no further improvement – break out of LP loop
            break

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence (list of floats) found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose candidate lengths.
    # ------------------------------------------------------------------
    lengths = []
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
            lengths.append(seq0.size)

    # additional lengths – mix a few mid‑size values
    for n in (1024, 2048, 3000, 3809, 4096):
        if n not in lengths:
            lengths.append(n)

    # randomise order a little to avoid systematic bias
    rng.shuffle(lengths)

    best_overall = None
    best_score_overall = float("inf")
    start_global = time.monotonic()

    # ------------------------------------------------------------------
    # 2) Optimise each length while time permits.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # ---------- initialise a probability vector ----------
        if "height_sequence_1" in globals():
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq0.size == n:
                a0 = seq0.copy()
            else:
                # linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, seq0.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(x_new, x_old, seq0)
        else:
            a0 = rng.dirichlet(np.ones(n)).astype(np.float64)

        # enforce simplex
        a0 = a0.clip(min=0.0)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / n)

        # ---------- run optimiser for this length ----------
        vec, score = _optimise_one_length(a0, deadline, rng)

        # ---------- update global best ----------
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - start_global
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – never happens, but guard against empty result.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    # 4) Clip to limits enforced by ``evaluate_sequence`` and return.
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    return [float(v) for v in best_overall]
