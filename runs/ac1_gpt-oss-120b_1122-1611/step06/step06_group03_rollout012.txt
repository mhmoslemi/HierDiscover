def _build_constraint_matrix(a, active, n):
   # returns csr matrix shape (len(active), n)
   data = []
   rows = []
   cols = []
   for idx, k in enumerate(active):
       start = max(0, k - (n - 1))
       end = min(n - 1, k)
       # for j from start..end:
       # coeff = a[k - j]
       # Note length L = end - start + 1
       # Using slices: we can get reversed a segment.
       # a segment = a[k-end : k-start+1]? Let's compute.
       # Example: j = start => coeff = a[k - start]
       # j = end => coeff = a[k - end]
       # So array of coeffs = a[k - np.arange(start, end+1)]
       # This segment may be reversed relative to j.
       segment = a[k - np.arange(start, end+1)]
       # Append each coefficient and corresponding column j
       # For each col j:
       for offset, coeff in enumerate(segment):
           j = start + offset
           data.append(coeff)
           rows.append(idx)
           cols.append(j)
   return csr_matrix((data, (rows, cols)), shape=(len(active), n))

"""
Gradient‑descent on a smooth approximation of the L∞‑objective.

The score to minimise is 2·n·max_k (a∗a)_k  (the sum of the
coefficients cancels because the objective is homogeneous, so we
normalise the vector to sum = 1).

The algorithm works in three phases

1. **Initialisation** – start from the analytically optimal
   “√‑difference’’ shape (a_i = √(i+1)−√i) or from a previously saved
   candidate (`height_sequence_1`) if it is better.

2. **Soft‑max mirror descent** – the non‑smooth max is replaced by a
   soft‑max:
        f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k) .
   Its gradient is a weighted correlation
        ∇f_β(a) = 2·corr(a, w) ,   w_k ∝ exp(β·(a∗a)_k) .
   At each iteration we
        * compute the full self‑convolution by FFT,
        * form the soft‑max weights,
        * obtain the gradient via another FFT,
        * perform an exponentiated‑gradient (mirror‑descent) step,
        * re‑project onto the probability simplex.
   The temperature‑like parameter β grows from 5 to 2000, while the
   step size η decays from 0.25 to 0.001, both on a linear schedule
   with respect to the elapsed time.

3. **Exploratory jitter** – with a small probability we apply a
   multiplicative log‑normal perturbation to avoid getting stuck in
   local minima.

The loop runs until the time budget (≈ 1000 s) is exhausted.  The
best vector seen during the search is returned (clipped to the limits
required by ``evaluate_sequence``).  All code is written with NumPy
and SciPy only and respects the 2‑CPU limit.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the smallest power‑of‑two length that can hold a linear
    convolution of two vectors of length *n*."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective value for a vector that is already normalised (sum = 1)."""
    return 2.0 * n * float(conv.max())


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a normalised vector."""
    conv = _conv_self(a, fft_len)
    return _score_from_conv(conv, n)


# ----------------------------------------------------------------------
# Analytic start and possible reuse of a previously stored vector
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)−√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _maybe_use_global() -> np.ndarray | None:
    """If a global variable ``height_sequence_1`` exists, return it
    normalised to sum = 1, otherwise None."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolutions (correlation)
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·corr(a, w), i.e. the gradient of Σ_k w_k·(a∗a)_k.
    The implementation uses real FFTs and works for any length.
    """
    n = a.size
    L = w.size                     # should be 2·n‑1
    # size for the FFT that can hold the linear convolution of lengths n and L
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3809))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Initial vector (analytic start or a previously stored one)
    # --------------------------------------------------------------
    a = _analytic_start(n)

    cand_global = _maybe_use_global()
    if cand_global is not None:
        # choose the better starting point according to the fast internal score
        score_global = _compute_score(cand_global, _fft_len_for_n(cand_global.size), cand_global.size)
        score_analytic = _compute_score(a, fft_len, n)
        if score_global < score_analytic:
            a = cand_global
            n = a.size
            fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    best_score = _compute_score(best, fft_len, n)

    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 0.5      # small safety margin

    # --------------------------------------------------------------
    # Hyper‑parameters for the soft‑max mirror descent
    # --------------------------------------------------------------
    beta_start = 5.0       # soft‑max temperature at the beginning
    beta_end = 2000.0      # temperature at the end (closer to true max)
    eta_start = 0.25       # step size at the beginning
    eta_end = 0.001        # step size at the end
    jitter_prob = 0.02     # probability of a multiplicative jitter
    jitter_sigma = 0.07    # log‑normal jitter width

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ----- convolution and current score -----
        conv = _conv_self(a, fft_len)          # length 2·n‑1
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (iter {iteration}) = {best_score:.6f}")

        # ----- schedule of the two hyper‑parameters -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        if total <= 0:
            break
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # ----- soft‑max weights -----
        # shift by max to avoid overflow, all entries ≤ 1
        w_raw = np.exp(beta * (conv - cur_max))
        w = w_raw / w_raw.sum()                # shape (2·n‑1,)

        # ----- gradient (2·correlation) -----
        grad = _gradient_correlation(a, w)      # shape (n,)

        # ----- exponentiated‑gradient update -----
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ----- occasional random jitter to escape plateaux -----
        if rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = a * jitter
            a /= a.sum()

        # ----- early termination safeguard -----
        if time.monotonic() > deadline - 0.1:
            break

    # --------------------------------------------------------------
    # Final polishing – a few cheap pairwise mass‑transfer moves
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 2.0:
        cur_a = best.copy()
        cur_score = best_score
        T0 = cur_score * 0.15
        Tmin = 1e-6
        sa_start = time.monotonic()
        while time.monotonic() < deadline:
            elapsed = time.monotonic() - sa_start
            total = deadline - sa_start
            T = T0 * (Tmin / T0) ** (elapsed / total)

            a_new = cur_a.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 1e-15:
                delta = rng.random() * a_new[i] * 0.5
                a_new[i] -= delta
                a_new[j] += delta
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()

            new_score = _compute_score(a_new, fft_len, n)
            if new_score < cur_score or rng.random() < np.exp((cur_score - new_score) / max(T, 1e-12)):
                cur_a = a_new
                cur_score = new_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = cur_a.copy()
                    elapsed_tot = time.monotonic() - start_time
                    print(f"[{elapsed_tot:6.2f}s] SA refinement: best = {best_score:.6f}")

            if T < 1e-12:
                break

    # --------------------------------------------------------------
    # Return the best vector (respect the bounds enforced by evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
