def _self_conv(a, fft_len_self):
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:2*a.size-1]
    return conv

def _gradient_softmax(a, beta, fft_len_self, fft_len_grad):
    conv = _self_conv(a, fft_len_self)
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        p = np.full_like(w, 1.0/w.size)
    else:
        p = w / w_sum
    # compute gradient wrt a
    rev_a = a[::-1]
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[:3*a.size-2]
    grad = 2.0 * grad_full[a.size-1 : a.size-1 + a.size]
    return conv, p, grad

"""
Fast gradient‑based search for a non‑negative step function that minimizes
evaluate_sequence().

Key ideas
---------
* Parameterise the step‑function with a small set of smooth basis functions
  (triangular “hats”).  The high‑dimensional vector a (length n) is obtained
  as a positive linear combination of the bases, then normalised to lie on
  the probability simplex (Σ a_i = 1).  This reduces the optimisation
  dimension from O(n) to O(K≈50).

* Optimise the parameters u_j = log w_j (where w_j > 0 are the basis
  coefficients) with Adam.  The objective is the smooth approximation of the
  max–convolution:
      L(u) = (1/β)·log Σ_i exp(β·(a*a)_i)   (soft‑max of the self‑convolution)
      f(u) = 2·n·L(u)                     (the quantity returned by
                                            evaluate_sequence when Σa=1)

  The gradient of L w.r.t. a is
      ∂L/∂a = 2·(p * rev(a)),   p_i = exp(β·c_i)/Σ_j exp(β·c_j),
  where c = a*a.  Using the chain rule we obtain ∂L/∂w_j and then
  ∂L/∂u_j = w_j·∂L/∂w_j.

* A simple schedule linearly raises the soft‑max temperature β from 0.5 to
  5000 and simultaneously lowers the learning‑rate.  If no improvement is
  seen for 30 s a multiplicative log‑normal jitter is applied to w and the
  Adam moments are reset.

* Several lengths are tried (including the previous construction from the
  global variable ``height_sequence_1``).  For each length the basis optimiser
  runs for a fraction of the remaining time budget.  The best full‑length
  vector found is stored.

* After all lengths a final refinement on the full vector (no basis) is
  performed with the same Adam scheme for the remaining time.

The code respects the 1000 s budget (a 5 s safety margin) and never writes
files or uses network I/O.  It returns the best sequence as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of prescribed length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation for a single length using a low‑dimensional basis
# ----------------------------------------------------------------------


def _optimise_length_basis(
    n: int,
    deadline: float,
    rng: np.random.Generator,
    init_u: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimises a step‑function of length `n` using a triangular‑hat basis
    and Adam.  Returns the best full‑length vector found and its
    evaluate_sequence score.
    """
    start = time.monotonic()
    # basis size (keep it modest to stay fast)
    K = min(64, n)
    # centres of the triangular hats
    centres = np.linspace(0, n - 1, K)
    sigma = max(1.0, n / (2.0 * K))  # width of each hat
    i_idx = np.arange(n)[:, None]  # (n,1)
    phi = np.maximum(0.0, 1.0 - np.abs(i_idx - centres[None, :]) / sigma)
    phi_sum = phi.sum(axis=0)  # (K,)

    # initialise log‑weights u
    if init_u is None:
        u = rng.normal(loc=0.0, scale=0.1, size=K)
    else:
        # pad or truncate to K if necessary
        if init_u.size != K:
            # simple linear interpolation of the old weights onto the new basis
            old_w = np.exp(init_u)
            old_a = phi @ old_w
            old_a = old_a / old_a.sum()
            # re‑fit to new basis via a single least‑squares step (non‑neg.)
            # solve min ||phi·w - old_a||²  with w≥0; use np.linalg.lstsq and clip
            w0, *_ = np.linalg.lstsq(phi, old_a, rcond=None)
            w0 = np.clip(w0, 1e-12, None)
            u = np.log(w0 + 1e-12)
        else:
            u = init_u.copy()

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # schedule parameters
    lr_start, lr_end = 0.8, 0.02
    beta_start, beta_end = 0.5, 5000.0

    best_seq = None
    best_score = float("inf")
    last_improve = start
    step = 0

    while time.monotonic() < deadline:
        step += 1
        now = time.monotonic()
        # linear schedules based on elapsed fraction
        frac = (now - start) / (deadline - start)
        lr = lr_start * (1.0 - frac) + lr_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- map u → a -------------------------------------------------
        w = np.exp(u)                     # positive basis weights
        s = phi @ w                       # unnormalised a
        S = s.sum()
        a = s / S                         # lies on simplex, Σa=1

        # ----- compute soft‑max convolution and gradient -----------------
        conv = _self_conv(a, fft_len_self)                # length 2n‑1
        max_c = conv.max()
        # soft‑max probabilities (stable)
        exp_term = np.exp(beta * (conv - max_c))
        p = exp_term / exp_term.sum()
        rev_a = a[::-1]
        fp = np.fft.rfft(p, n=fft_len_grad)
        fra = np.fft.rfft(rev_a, n=fft_len_grad)
        grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
        grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]        # ∂L/∂a

        # ----- chain rule: ∂L/∂w ---------------------------------------
        dot_g_phi = phi.T @ grad_a                         # (K,)
        dot_g_a_a = grad_a @ a                             # scalar
        grad_w = (dot_g_phi - phi_sum * dot_g_a_a) / S     # (K,)
        grad_u = grad_w * w                                 # (K,)

        # ----- Adam update ----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)
        u = u - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- evaluate (using the exact objective) --------------------
        cur_score = 2.0 * n * max_c
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_seq = a.copy()
            last_improve = now
            print(
                f"[{now - start:6.2f}s] n={n:5d} step={step:7d} "
                f"new local best {best_score:.6f}"
            )

        # ----- stagnation handling --------------------------------------
        if now - last_improve > 30.0:
            # log‑normal jitter around the current best
            if best_seq is not None:
                a = best_seq.copy()
            else:
                a = a  # keep current
            w = np.exp(u)
            noise = np.exp(rng.normal(0.0, 0.10, size=K))
            w *= noise
            u = np.log(w + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start:6.2f}s] n={n} step={step} restart (stagnation)")

        # early exit if we already beat the target
        if best_score <= 1.5025:
            break

    # final exact evaluation using evaluate_sequence (if available)
    if best_seq is not None:
        try:
            eval_score = evaluate_sequence(best_seq.tolist())
        except Exception:
            eval_score = best_score
        best_score = eval_score
    else:
        # fallback to a uniform vector
        best_seq = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_seq.tolist())

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # 5 s safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Choose lengths to explore (include previous construction if present)
    # ------------------------------------------------------------------
    lengths = []
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
            lengths.append(seq0.size)

    # a handful of diverse sizes (powers of two and a “medium” size)
    for n in (512, 1024, 2048, 3809, 6000, 8192):
        if n not in lengths:
            lengths.append(n)

    # ------------------------------------------------------------------
    # Optimise each length in turn, dividing the remaining time fairly
    # ------------------------------------------------------------------
    best_overall_seq = None
    best_overall_score = float("inf")
    start_all = time.monotonic()

    for idx, n in enumerate(lengths):
        now = time.monotonic()
        if now >= deadline:
            break

        remaining_lengths = len(lengths) - idx
        # allocate at most 90 % of the remaining budget, split equally
        time_for_this = (deadline - now) * 0.90 / remaining_lengths
        local_deadline = now + time_for_this

        # If a previous construction matches the size, initialise u close to it
        init_u = None
        if "height_sequence_1" in globals():
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq0.size == n:
                # Fit the old sequence to the new basis (cheap least‑squares)
                K = min(64, n)
                centres = np.linspace(0, n - 1, K)
                sigma = max(1.0, n / (2.0 * K))
                i_idx = np.arange(n)[:, None]
                phi = np.maximum(0.0, 1.0 - np.abs(i_idx - centres[None, :]) / sigma)
                # solve min ||phi·w - seq0||²  with non‑negativity
                w0, *_ = np.linalg.lstsq(phi, seq0, rcond=None)
                w0 = np.clip(w0, 1e-12, None)
                init_u = np.log(w0 + 1e-12)

        # Run the low‑dimensional optimiser
        vec, score = _optimise_length_basis(n, local_deadline, rng, init_u)

        # Update global best
        if score < best_overall_score - 1e-12:
            best_overall_score = score
            best_overall_seq = vec.copy()
            elapsed = time.monotonic() - start_all
            print(
                f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_overall_score:.6f}"
            )
        # If we already reached the target we can stop early
        if best_overall_score <= 1.5025:
            break

    # ------------------------------------------------------------------
    # Final refinement on the full vector (if time permits)
    # ------------------------------------------------------------------
    now = time.monotonic()
    if now < deadline and best_overall_seq is not None:
        # give the remaining budget to refinement
        refined_vec, refined_score = _refine_full_vector(
            best_overall_seq, deadline, rng
        )
        if refined_score < best_overall_score - 1e-12:
            best_overall_seq = refined_vec
            best_overall_score = refined_score
            elapsed = time.monotonic() - start_all
            print(
                f"[{elapsed:6.2f}s] *** GLOBAL BEST AFTER REFINEMENT score={best_overall_score:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback – should never happen but guards against empty result.
    # ------------------------------------------------------------------
    if best_overall_seq is None:
        best_overall_seq = np.array([1.0], dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence and return as list
    best_overall_seq = np.clip(best_overall_seq, 0.0, 1000.0)
    return [float(x) for x in best_overall_seq]
