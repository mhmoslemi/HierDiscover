def _gradient_corr_fft(a, w):
    n = a.size
    L = w.size
    conv_len = L + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)
    # take appropriate slice
    start = n - 1
    return 2.0 * corr_full[start:start + n]
```

def _normalize(v):
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:

"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

The routine works as follows:

1.  Build a diverse pool of analytic shapes (sqrt‑difference, power‑difference,
    exponential, triangular, discretised Gaussian) together with many random
    Dirichlet samples and, if available, the previously best construction.
    All vectors are normalised to an L1‑norm of 1.

2.  Pick the best vector from the pool using a fast FFT‑based evaluation of
    ``2·n·max(a*a)`` (the objective for a normalised sequence).

3.  Refine this vector with a mirror‑descent / exponentiated‑gradient scheme.
    At each iteration we compute the self‑convolution, locate the (few)
    indices attaining the current maximum and obtain the exact sub‑gradient of
    the L∞‑norm via a correlation performed with FFTs.  A linear schedule
    drives the step‑size from 0.5 down to 5·10⁻⁴, while momentum stabilises the
    updates.  Small random perturbations occasionally help to escape plateaus.

4.  The remaining time is spent in a simulated‑annealing phase that proposes
    random mass‑transfer moves (and occasional smoothing) on the current best
    candidate, accepting improvements or uphill moves with a temperature that
    cools exponentially.

5.  The best vector found is clipped to the limits enforced by
    ``evaluate_sequence`` and returned as a plain list of floats.

The whole procedure respects a 1000 s budget.  No closures are used – all
helpers are defined at module level.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level (no closures).
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution
    of two vectors of length n (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution of a (real) vector using FFT.  Returns length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _score_of_fast(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast objective for a normalised (sum = 1) non‑negative vector:
    2·n·max(a∗a)."""
    conv = _conv_self_fft(a, fft_len)
    return 2.0 * n * conv.max()


def _gradient_corr_fft(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_k w_k·(a∗a)_k with respect to a.
    Gradient = 2·a ⋆ w (cross‑correlation).
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    conv_len = L + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)
    start = n - 1
    return 2.0 * corr_full[start:start + n]


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (can be overridden via kwargs)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 5000))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 1) Assemble a diverse pool of starting points
    # ------------------------------------------------------------------
    candidates = []

    # a) Analytic √‑difference start
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    candidates.append(_normalize(a_sqrt))

    # b) Triangular (linear decreasing) shape
    a_tri = np.linspace(1.0, 0.0, n)
    candidates.append(_normalize(a_tri))

    # c) Gaussian‑like (discretised normal) shape
    sigma = n / 6.0
    x = np.arange(n, dtype=np.float64)
    a_gauss = np.exp(-0.5 * ((x - (n - 1) / 2) / sigma) ** 2)
    candidates.append(_normalize(a_gauss))

    # d) Exponential families
    for lam in np.linspace(0.001, 0.020, 12):
        a = (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))
        candidates.append(_normalize(a))

    # e) Power‑difference families
    for p in np.linspace(0.2, 2.0, 12):
        a = ((np.arange(n, dtype=np.float64) + 1) ** (-p) -
             (np.arange(n, dtype=np.float64) + 2) ** (-p))
        candidates.append(_normalize(a))

    # f) Random Dirichlet samples
    for _ in range(30):
        alpha = rng.uniform(0.1, 1.5)
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(_normalize(a))

    # g) Simple mixtures of √‑difference and exponential families
    for lam in (0.005, 0.010, 0.015):
        for w in (0.3, 0.5, 0.7):
            mix = (w * (np.sqrt(np.arange(n) + 1) - np.sqrt(np.arange(n))) +
                   (1.0 - w) * ((1.0 - np.exp(-lam)) *
                               np.exp(-lam * np.arange(n, dtype=np.float64))))
            candidates.append(_normalize(mix))

    # h) Previous best construction, if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                candidates.append(_normalize(prev))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) Pick the best among the initial pool (fast evaluation)
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    for a in candidates:
        sc = _score_of_fast(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    print(f"[init] best score from pool = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 3) Mirror‑descent refinement (exponentiated‑gradient)
    # ------------------------------------------------------------------
    cur = best.copy()
    grad_prev = np.zeros_like(cur)

    start_time = time.monotonic()
    # 80 % of the budget for the deterministic refinement
    descent_deadline = start_time + 0.8 * budget_s

    # Step‑size schedule (linear decay)
    eta_start, eta_end = 0.5, 5e-4
    momentum = 0.9
    # tolerance for selecting an active set around the maximum
    rel_tol = 0.999
    max_active = 15

    while time.monotonic() < descent_deadline:
        # --- convolution and current score ---
        conv = _conv_self_fft(cur, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # --- keep global best ---
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.9f}")

        # --- schedule step size ---
        now = time.monotonic()
        frac = (now - start_time) / (descent_deadline - start_time)
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # --- active set (indices within rel_tol of the current maximum) ---
        active = np.where(conv >= cur_max * rel_tol)[0]
        if active.size > max_active:
            active = np.argsort(conv)[-max_active:]

        # --- build weight vector w on the full convolution domain ---
        w = np.zeros_like(conv)
        if active.size == 0:
            w[np.argmax(conv)] = 1.0
        else:
            w[active] = 1.0 / active.size

        # --- gradient of the soft‑max surrogate (exact for the chosen w) ---
        grad = _gradient_corr_fft(cur, w)

        # --- momentum ---
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # --- exponentiated‑gradient update (mirror descent) ---
        # Clip gradient to avoid overflow in the exponent
        grad_clip = np.clip(grad, -10.0, 10.0)
        factor = np.exp(-eta * grad_clip)
        cur = cur * factor
        cur = np.clip(cur, 0.0, None)
        s = cur.sum()
        if s > 0.0:
            cur = cur / s
        else:
            cur = np.full_like(cur, 1.0 / n)

        # --- occasional random log‑perturbation to escape stagnation ---
        if rng.random() < 0.02:
            log_cur = np.log(cur + 1e-30)
            log_cur += rng.normal(scale=0.15, size=n)
            cur = np.exp(log_cur)
            cur = cur / cur.sum()

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning (remaining time)
    # ------------------------------------------------------------------
    sa_deadline = start_time + float(budget_s)
    cur_sa = best.copy()
    cur_sa_score = best_score
    T_start = 5e-3
    T_end = 1e-6
    iteration = 0

    while time.monotonic() < sa_deadline:
        now = time.monotonic()
        frac = (now - descent_deadline) / (sa_deadline - descent_deadline)
        T = T_start * (1.0 - frac) + T_end * frac

        # --- propose a random mass‑transfer move ---
        i, j = rng.integers(0, n, size=2)
        if i != j and cur_sa[i] > 1e-15:
            delta = rng.random() * cur_sa[i] * 0.5
            new_a = cur_sa.copy()
            new_a[i] -= delta
            new_a[j] += delta

            new_score = _score_of_fast(new_a, n, fft_len)

            improve = new_score < cur_sa_score
            if improve or rng.random() < np.exp(-(new_score - cur_sa_score) / max(T, 1e-12)):
                cur_sa = new_a
                cur_sa_score = new_score
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best = new_a.copy()
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        iteration += 1

        # --- occasional gentle smoothing (helps flatten local peaks) ---
        if iteration % 4000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            cur_sa = smoothed
            cur_sa_score = _score_of_fast(cur_sa, n, fft_len)

    # ------------------------------------------------------------------
    # 5) Return the best sequence (respect clipping constraints)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
