    _fft_len_conv = _fft_len_for_len(2*n -1)

    best_vec = None

    init_weights = [0.5,0.3,0.2]

    while time.time() < deadline:

"""
Hybrid optimisation for the self‑convolution upper‑bound problem.

The algorithm proceeds in three stages:
1. Random initialisations (Dirichlet or sine‑power) for a chosen length n.
2. A continuation of smooth‑max optimisation:
   – The objective f_β(a) = (1/β)·log Σ_i exp(β·(a*a)_i) approximates max(a*a).
   – For a sequence of increasing β (5 → 20 → 80 → 300) we minimise f_β
     using L‑BFGS‑B on the log‑softmax parametrisation a_i = softmax(x_i).
3. The best candidate from all runs is polished with a fast exponentiated‑gradient
   (EG) routine (the same scheme used in the reference implementation).

All heavy work is done with real FFTs, so the cost per iteration is
O(n log n).  The search respects the supplied time budget and returns a list
of non‑negative heights ready for ``evaluate_sequence``.

        x0 = np.log(np.maximum(a0, eps))

        # temperature continuation
        for beta in (5.0, 20.0, 80.0, 300.0):
            _current_beta = beta
            # a modest iteration budget for each stage (fast enough)
            max_iter_stage = 150
            if time.time() >= deadline:
                break
            try:
                res = opt.minimize(f_and_grad_lb,
                                   x0,
                                   method='L-BFGS-B',
                                   jac=True,
                                   options={'maxiter': max_iter_stage,
                                            'gtol': 1e-12,
                                            'disp': False})
                x0 = res.x
            except Exception:
                # if anything goes wrong just abort this run
                break

            # evaluate the current candidate
            a = np.exp(x0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s
            score = _score(a, _fft_len_conv)
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                elapsed = time.time() - start_global
                print(f"[{elapsed:5.2f}s] Length {n} new best {best_score:.6f}")

        # stop if we are within a few seconds of the global deadline
        if time.time() >= deadline - 5.0:
            break

    return best_vec, best_score


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Hybrid optimisation: random starts → smooth‑max L‑BFGS continuation
    → final polishing with exponentiated‑gradient.
    The function respects the given time budget and returns a list of
    non‑negative heights.
    """
    start = time.time()
    # safety margin for the harness bookkeeping
    deadline = start + max(0.0, float(budget_s) - 0.3)

    rng = np.random.default_rng(seed)

    best_vec = None
    best_score = float('inf')

    # Lengths to explore – a mix of medium and large sizes.
    lengths = [1500, 2500, 3809, 5000, 8000]

    for n in lengths:
        if time.time() >= deadline:
            break
        vec, score = run_optim_len(rng, n, deadline, start)
        if vec is not None and score < best_score:
            best_score = score
            best_vec = vec
            print(f"[{time.time() - start:5.2f}s] Global best → {best_score:.6f} (n={n})")

    # Polishing phase – use the fast EG optimiser for a few thousand steps
    if best_vec is not None and time.time() < deadline - 1.0:
        best_vec, best_score = _local_optimise(best_vec.copy(),
                                               rng,
                                               deadline,
                                               max_iters=3000)
        print(f"[{time.time() - start:5.2f}s] Polished final best: {best_score:.6f}")

    # Fallback in the extremely unlikely case that nothing was found
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform sequence, score = {best_score:.6f}")

    # Final clipping (evaluate_sequence also performs the same clipping)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to a plain Python list of floats
    return [float(v) for v in best_vec]
