"""
Propose a step‑function (non‑negative heights) that minimises ``evaluate_sequence``.

Algorithm
---------
1. Build a diverse pool of normalised candidates (sqrt‑diff, power‑diff,
   exponential, random Dirichlet, and the previous best if available).
2. Select the best candidate from the pool using a fast FFT‑based
   self‑convolution score ``2·n·max(conv)`` (the objective for sum‑1 vectors).
3. Refine the selected candidate by a *projected sub‑gradient* method:
   * At each iteration compute the exact self‑convolution (FFT).
   * Identify the active index set where the convolution attains the current
     maximum (within a tiny tolerance).
   * The sub‑gradient of the maximal convolution w.r.t. the coefficients is
     ``g_i = 2·∑_{k∈active} a_{k‑i}``.  It is obtained by simple
     O(|active|·n) operations (the active set is usually ≤ 2).
   * Transfer a small amount of mass from the coefficient with the largest
     gradient to the one with the smallest gradient.  The step size decays
     linearly from 0.5 to 0.001 over the allotted time.
   * Accept the move only if the maximal convolution value decreases.
   * Occasionally apply random block scaling or pair‑wise mass swaps to
     escape plateaus.
4. When the remaining time is ≤ 20 s, run a short simulated‑annealing phase
   with exponential cooling to fine‑tune the solution.
5. Return the best sequence found, clipped to the limits imposed by
   ``evaluate_sequence`` (0 ≤ value ≤ 1000).

The routine respects a 1000 s budget, prints progress information and
returns a plain list of ``float`` values.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power‑of‑two length that can contain a full linear convolution
# of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Score for a probability vector (sum = 1).
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) for a normalised non‑negative vector."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Normalise a raw vector to a probability distribution (non‑negative,
# sum = 1).  Zero‑vectors are replaced by the uniform distribution.
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray) -> np.ndarray:
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Gradient of the maximal convolution (sub‑gradient) w.r.t. a.
# active is a list/array of indices where conv == current maximum.
# ----------------------------------------------------------------------
def _subgradient_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Compute g_i = 2·∑_{k∈active} a_{k‑i} for i = 0..n‑1."""
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi + 1, dtype=np.int64)
        grad[i_range] += 2.0 * a[k - i_range]
    return grad


# ----------------------------------------------------------------------
# Build a collection of diverse initial candidates.
# ----------------------------------------------------------------------
def _initial_candidates(n: int) -> list[np.ndarray]:
    candidates = []

    # a) sqrt‑difference (classic)
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    candidates.append(_normalise(a_sqrt))

    # b) previous best if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                candidates.append(_normalise(prev[:n]))
        except Exception:
            pass

    # c) power‑difference families
    for p in np.linspace(0.2, 2.0, 12):
        a = ((np.arange(n, dtype=np.float64) + 1) ** (-p)
             - (np.arange(n, dtype=np.float64) + 2) ** (-p))
        candidates.append(_normalise(a))

    # d) exponential families
    for lam in np.linspace(0.0005, 0.02, 12):
        a = (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))
        candidates.append(_normalise(a))

    # e) random Dirichlet samples
    rng = np.random.default_rng(12345)
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0)
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(_normalise(a))

    return candidates

# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Search for a non‑negative step‑function that minimises evaluate_sequence().
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (default 5000, can be overridden via kwargs)
    # ------------------------------------------------------------------

        n = 2
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 1) Initialise a diverse pool of candidates and pick the best.
    # ------------------------------------------------------------------
    pool = _initial_candidates(n)
    best = None
    best_score = np.inf
    for a in pool:
        sc = _score_of(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    print(f"[init] best score from pool = {best_score:.9f}")

    # 2) Projected sub‑gradient descent with occasional random moves.
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_conv = _conv_self(cur, fft_len)
    cur_max = float(cur_conv.max())
    cur_score = 2.0 * n * cur_max

    deadline_main = start_time + max(0.0, float(budget_s) - 20.0)  # reserve 20 s for SA
    best_global = cur.copy()
    best_global_score = cur_score

    step_start, step_end = 0.5, 0.001          # maximal mass‑transfer fraction
    rel_tol = 1e-12                           # tolerance for “active” indices
    prob_block = 0.025                        # probability of a block scaling move
    prob_swap = 0.07                          # probability of a pairwise mass swap

        # ------------------------------------------------------------------
        # Identify active indices where convolution attains its maximum.
        # ------------------------------------------------------------------
        cur_max = float(cur_conv.max())
        active = np.where(np.isclose(cur_conv, cur_max, atol=rel_tol))[0]

        # ------------------------------------------------------------------
        # Sub‑gradient for the current active set.
        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

        tgt = int(np.argmin(grad))

                    cur[i] -= delta
                    cur[j] += delta

                    cur_conv = _conv_self(cur, fft_len)

        new_a[tgt] += delta
        # Numerical safety

            improved = False

                new_a[tgt] += delta

                        cur_conv = _conv_self(cur, fft_len)

    # ------------------------------------------------------------------
    deadline_total = start_time + float(budget_s)

    T_end = 1e-6

        now = time.monotonic()
        frac = (now - deadline_main) / (deadline_total - deadline_main) if deadline_total > deadline_main else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        delta = rng.random() * cur_sa[i] * 0.5

        cand[i] -= delta
        cand[j] += delta

        cand_max = float(cand_conv.max())

    # 4) Return the best sequence found (respect clipping limits).
    # ------------------------------------------------------------------
    best_global = np.clip(best_global, 0.0, 1000.0)
