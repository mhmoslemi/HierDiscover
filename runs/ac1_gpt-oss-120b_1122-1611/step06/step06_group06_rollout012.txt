"""
Propose a step‑function (non‑negative heights) that minimises evaluate_sequence().

Algorithm
---------
1. **Analytic families** – the convolution of the discrete derivative of x^c
   (a_i = (i+2)^c – (i+1)^c) has a provably low L∞‑norm for c≈0.4.
   We scan a fine grid of exponents (c) and a few classical families
   (power‑difference, exponential, √‑difference) for several problem sizes.
2. **Pick the best candidate** according to the fast FFT based score
   (2·n·max(self‑convolution)).
3. **Mirror‑descent refinement** – exponentiated‑gradient updates driven by a
   soft‑max surrogate of the maximum convolution.  Momentum, a linear step‑size
   schedule and occasional random mass‑transfer moves keep the iterate from
   stagnating.
4. **Simulated‑annealing fine‑tuning** for the remaining time budget.
   Simple pairwise mass transfers are accepted with a Metropolis rule.
5. Return the best sequence seen, clipped to the limits imposed by
   `evaluate_sequence` (0 ≤ value ≤ 1000).

The search focuses on the *c‑family* because its theoretical limit
   2·c²·B(c,c)  (B – beta function) is ≤ π/2 for c=0.5 and drops to ≈ 1.35
   for c≈0.40, well below the target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Fast (exact) L∞‑objective for a probability vector.
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a*a) for a normalised (sum=1) non‑negative vector."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (used only when the active set is empty).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Grad of Σ_k w_k (a∗a)_k = 2·a ⋆ w (correlation)."""
    n = a.size
    L = w.size                # = 2·n‑1
    # length needed for convolution a_rev * w
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

# ----------------------------------------------------------------------

    Search for a non‑negative step‑function that minimises evaluate_sequence().
    Returns the best sequence found before the time budget expires.

    # comfortable medium size (≈5 k) that keeps FFT cheap but still large
