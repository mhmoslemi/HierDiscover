def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    n = kwargs.get('n', 5000)
    # Ensure n not too large; maybe clamp to max 8000 due to time.
    n = max(2, int(n))
    # 1) Grid search over p and alpha
    # Define param ranges
    p_vals = np.linspace(0.45, 0.55, 11)
    alpha_vals = np.linspace(0.0, 5.0, 11)
    best_score = np.inf
    best_a = None

    fft_len = _fft_len_for_n(n)
    for p in p_vals:
        for alpha in alpha_vals:
            i = np.arange(n, dtype=np.float64)
            a = (i + alpha) ** (-p)
            s = a.sum()
            if s == 0:
                continue
            a = a / s
            # compute max convolution
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * np.max(conv)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
    # Log best found
    print(f"Initial grid best score {best_score:.6f}")

    # 2) Local Adam refinement using softmax surrogate
    # Convert best_a to logits
    x = np.log(best_a + 1e-30)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    # Precompute FFT lengths for conv and correlation (as before)
    fft_len_conv = _fft_len_for_n(n)
    fft_len_corr = 1 << ((3 * n - 2 - 1).bit_length())
    # Schedules:
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)
    # schedule parameters:
    beta_start = 5.0
    beta_end = 3000.0
    lr_start = 0.5
    lr_end = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_score = best_score
    best_a_opt = best_a.copy()
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # schedule
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        lr = lr_start * (1.0 - elapsed / total) + lr_end * (elapsed / total)
        # compute a from logits
        a = np.exp(x - np.max(x))
        a /= a.sum()
        # evaluate objective
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a_opt = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")
        # compute softmax weights for surrogate
        exp_arg = beta * (conv - cur_max)
        exp_arg = np.clip(exp_arg, -700, 0)
        w = np.exp(exp_arg)
        w /= w.sum()
        # gradient w.r.t. a
        grad_a = _grad_corr(a, w, fft_len_corr)
        # gradient w.r.t logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        it = (it if 'it' in locals() else 0) + 1
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step
        # occasional jitter
        if rng.random() < 0.02:
            x += rng.normal(0.0, 0.01, size=n)
        # block perturb
        if rng.random() < 0.02:
            width = rng.integers(1, max(2, n // 50))
            start = rng.integers(0, n - width + 1)
            delta = rng.normal(0.0, 0.1)
            x[start:start+width] += delta
        # restart if stuck
        if now - last_improve > 60.0:
            # reinitialize from best with jitter
            x = np.log(best_a_opt + 1e-30)
            x += rng.normal(0.0, 0.05, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # after loop, return best_a_opt
    best_a_opt = np.clip(best_a_opt, 0.0, 1000.0)
    return [float(v) for v in best_a_opt]

"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Strategy
--------
1. Grid‑search a two‑parameter family a_i ∝ (i+α)^{-p} (i starts at 1)
   with p∈[0.5,0.7] and α∈[0,3].  For each pair we normalise the
   vector to sum to one and compute the exact objective using FFT‑based
   convolution (O(n log n)).  The best pair gives a strong initial guess.
2. Refine the initial guess with a soft‑max / Adam optimiser that
   directly minimises the surrogate  (1/β)·log Σ_k exp(β·(a∗a)_k)
   while keeping the vector on the probability simplex.
   The optimisation runs until the remaining time budget is exhausted.
3. The best sequence observed (both from the grid and the local search)
   is returned.  The sequence is clipped to the limits enforced by
   evaluate_sequence() (≥0, ≤1000).

The whole routine respects a 1000‑second wall‑clock budget, prints
progress timestamps and finally yields a candidate list of heights.
"""

import time
import numpy as np

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution
    of two vectors of length n (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return a* a as linear self‑convolution (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _grad_corr(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """Correlation a ⋆ w used for the gradient of Σ w_k·(a∗a)_k.
    Result is a vector of the same length as a."""
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: 3 * a.size - 2]
    start = a.size - 1
    return 2.0 * corr_full[start:start + a.size]

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence() within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # Problem dimension – a few thousand entries give a good trade‑off.
    n = int(kwargs.get("n", 4000))
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 1) Coarse grid search over the family a_i ∝ (i+α)^{-p}
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # leave a small safety margin

    # Parameter ranges – chosen after a quick theoretical analysis.
    p_vals = np.linspace(0.50, 0.70, 9)          # 0.50, 0.525, … , 0.70
    alpha_vals = np.linspace(0.0, 3.0, 13)       # 0.0, 0.25, … , 3.0

    fft_len = _fft_len_for_n(n)

    best_score = np.inf
    best_a = None
    best_params = None

    # indices start at 1 to avoid a_i = 0⁻ᵖ singularities when α = 0.
    indices = np.arange(1, n + 1, dtype=np.float64)

    for p in p_vals:
        for alpha in alpha_vals:
            # generate the candidate shape
            a = (indices + alpha) ** (-p)
            s = a.sum()
            if s <= 0.0:
                continue
            a /= s
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * np.max(conv)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_params = (p, alpha)

    print(f"[grid] best score {best_score:.6f} with p={best_params[0]:.3f}, α={best_params[1]:.3f}")

    # ------------------------------------------------------------------
    # 2) Local refinement with Adam on a soft‑max parametrisation
    # ------------------------------------------------------------------
    # Initialise logits from the best grid point
    x = np.log(best_a + 1e-30)          # logits (soft‑max will enforce sum=1)
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # FFT lengths that stay constant throughout the optimisation
    fft_len_conv = _fft_len_for_n(n)
    fft_len_corr = 1 << ((3 * n - 2 - 1).bit_length())

    # Schedules
    beta_start = 5.0
    beta_end = 3000.0
    lr_start = 0.7
    lr_end = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_score_refined = best_score
    best_a_refined = best_a.copy()
    last_improve = time.monotonic()
    it = 0

    # Random‑jitter configuration
    jitter_prob = 0.015
    block_prob = 0.015
    restart_gap = 60.0   # seconds without improvement → restart

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ----- schedules ------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        lr = lr_start * (1.0 - elapsed / total) + lr_end * (elapsed / total)

        # ----- current probability vector a ----------------------------
        a = np.exp(x - np.max(x))
        a /= a.sum()

        # ----- evaluate current objective --------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score_refined - 1e-12:
            best_score_refined = cur_score
            best_a_refined = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (refined) score = {best_score_refined:.6f}")

        # ----- soft‑max surrogate weights --------------------------------
        exp_arg = beta * (conv - cur_max)          # keep values ≤ 0
        exp_arg = np.clip(exp_arg, -700, 0)
        w_raw = np.exp(exp_arg)
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)

        # ----- gradient w.r.t. a (smooth surrogate) ----------------------
        grad_a = _grad_corr(a, w, fft_len_corr)   # shape (n,)

        # ----- chain rule → gradient w.r.t. logits -----------------------
        dot = np.dot(grad_a, a)                   # scalar <a, grad_a>
        grad_x = a * (grad_a - dot)               # shape (n,)

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- occasional stochastic perturbations ------------------------
        if rng.random() < jitter_prob:
            x += rng.normal(0.0, 0.03, size=n)

        if rng.random() < block_prob:
            width = rng.integers(1, max(2, n // 40))
            start = rng.integers(0, n - width + 1)
            delta = rng.normal(0.0, 0.15)
            x[start:start + width] += delta

        # ----- soft restart if stuck ------------------------------------
        if now - last_improve > restart_gap:
            # Re‑initialise from the current best with a modest jitter
            x = np.log(best_a_refined + 1e-30)
            x += rng.normal(0.0, 0.08, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score_refined:.6f}")

    # ------------------------------------------------------------------
    # 3) Return the best sequence discovered (clipped to the limits
    #    enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best_a_refined = np.clip(best_a_refined, 0.0, 1000.0)
    return [float(v) for v in best_a_refined]

   evaluate_sequence()).

The routine respects the 1000 s wall‑clock budget, prints progress
timestamps and finally yields a candidate list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (FFT‑based convolution / correlation)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution
    of two vectors of length n (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_corr(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """Correlation a ⋆ w (gradient of Σ w_k·(a∗a)_k).  Returns a vector
    of the same length as a."""
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: 3 * a.size - 2]
    start = a.size - 1
    return 2.0 * corr_full[start:start + a.size]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence() within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (a few thousand entries give a good trade‑off)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 1) Coarse grid search over a_i ∝ (i+α)^{-p}
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # keep a tiny safety margin

    p_vals = np.linspace(0.50, 0.70, 9)          # 0.50, 0.525, … , 0.70
    alpha_vals = np.linspace(0.0, 3.0, 13)       # 0.0, 0.25, … , 3.0

    fft_len = _fft_len_for_n(n)
    indices = np.arange(1, n + 1, dtype=np.float64)    # i = 1 … n (avoids 0⁻ᵖ)

    best_score = np.inf
    best_a = None
    best_params = None

    for p in p_vals:
        for alpha in alpha_vals:
            a = (indices + alpha) ** (-p)
            s = a.sum()
            if s <= 0.0:
                continue
            a /= s
            conv = _conv_self(a, fft_len)
            cur_score = 2.0 * n * np.max(conv)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                best_params = (p, alpha)

    print(f"[grid] best score {best_score:.6f} with p={best_params[0]:.3f}, α={best_params[1]:.3f}")

    # ------------------------------------------------------------------
    # 2) Local refinement with Adam on a soft‑max parametrisation
    # ------------------------------------------------------------------
    # initialise logits from the best grid point
    x = np.log(best_a + 1e-30)          # logits; the soft‑max will enforce Σ a = 1
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    fft_len_conv = _fft_len_for_n(n)
    fft_len_corr = 1 << ((3 * n - 2 - 1).bit_length())

    # schedule parameters
    beta_start = 5.0
    beta_end = 3000.0
    lr_start = 0.7
    lr_end = 0.02
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_score_ref = best_score
    best_a_ref = best_a.copy()
    last_improve = time.monotonic()
    it = 0

    jitter_prob = 0.015
    block_prob = 0.015
    restart_gap = 60.0   # seconds without improvement → soft restart

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ----- schedules -------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        lr = lr_start * (1.0 - elapsed / total) + lr_end * (elapsed / total)

        # ----- current probability vector a -----------------------------
        a = np.exp(x - np.max(x))
        a /= a.sum()

        # ----- evaluate current objective --------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score_ref - 1e-12:
            best_score_ref = cur_score
            best_a_ref = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (refined) score = {best_score_ref:.6f}")

        # ----- soft‑max surrogate weights --------------------------------
        exp_arg = beta * (conv - cur_max)          # values ≤ 0
        exp_arg = np.clip(exp_arg, -700, 0)
        w_raw = np.exp(exp_arg)
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)

        # ----- gradient w.r.t. a (smooth surrogate) ----------------------
        grad_a = _grad_corr(a, w, fft_len_corr)   # shape (n,)

        # ----- chain rule → gradient w.r.t. logits -----------------------
        dot = np.dot(grad_a, a)                   # scalar ⟨a, grad_a⟩
        grad_x = a * (grad_a - dot)               # shape (n,)

        # ----- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- occasional stochastic perturbations ------------------------
        if rng.random() < jitter_prob:
            x += rng.normal(0.0, 0.03, size=n)

        if rng.random() < block_prob:
            width = rng.integers(1, max(2, n // 40))
            start = rng.integers(0, n - width + 1)
            delta = rng.normal(0.0, 0.15)
            x[start:start + width] += delta

        # ----- soft restart if stuck ------------------------------------
        if now - last_improve > restart_gap:
            x = np.log(best_a_ref + 1e-30)
            x += rng.normal(0.0, 0.08, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score_ref:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (clipped to the limits enforced
    # by evaluate_sequence()).
    # ------------------------------------------------------------------
    best_a_ref = np.clip(best_a_ref, 0.0, 1000.0)
    return [float(v) for v in best_a_ref]
