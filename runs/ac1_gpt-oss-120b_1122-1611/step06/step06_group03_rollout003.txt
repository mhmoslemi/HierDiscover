"""
Propose a high‑quality step‑function (non‑negative list of heights).

Algorithm
---------
1. **Problem size** – use the length `n` supplied in ``kwargs`` (default 3800).
   The vector is always kept on the probability simplex (sum = 1) because the
   objective is homogeneous.

2. **Initialisation** – start from the analytically known √‑difference shape
   (`_analytic_start`).  If a previous construction is available in the global
   variable ``height_sequence_1`` it is used when it gives a lower internal
   score.

3. **Fast parametric sweep** – evaluate a family of simple parametric shapes:
   * pure power‑law   a_i ∝ (i+offset)^(−p)
   * pure exponential a_i ∝ exp(−c·i)
   * combined power‑law × exponential
   The sweep explores a grid of exponents `p`, offsets and decay rates `c`
   (the whole sweep finishes well within the budget).  The best candidate
   becomes the new current best.

4. **Mirror‑descent polishing** – minimise the smoothed max of the
   self‑convolution using exponentiated‑gradient (mirror descent) on the
   simplex.  At each iteration
        • the full convolution `conv = a*a` is obtained by a single FFT,
        • a soft‑max weight vector `w ∝ exp(β·(conv−max(conv)))` approximates
          the sub‑gradient of the max,
        • the gradient of the weighted sum is computed via FFT
          (`_gradient_correlation`),
        • the update `a ← a·exp(−η·grad)` is performed and the vector is
          normalised.
   The step‑size `η` decays linearly, while the soft‑max temperature `β`
   grows linearly, giving a coarse‑to‑fine optimisation.  The loop runs
   until the time budget is almost exhausted, keeping the best vector seen.

5. **Return** – the best vector (clipped to the limits of
   ``evaluate_sequence``) is returned as a plain Python list.

The whole routine respects the 1000 s time limit, prints progress stamps,
and works for any `n`.  Only NumPy from the standard scientific stack is
required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helper utilities
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via real FFT (returns the full 2·n‑1 result)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a normalised vector a (sum(a)=1).
    Returns 2·n·max(conv) – the exact value returned by evaluate_sequence.
    """
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _maybe_use_global() -> np.ndarray | None:
    """Return a normalised version of the global height_sequence_1 if it exists and is valid."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolutions
# ----------------------------------------------------------------------


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k with respect to a.
    Implements the correlation a ⋆ w (see the original code).
    """
    n = a.size
    L = w.size                      # should be 2·n‑1
    size = L + n - 1
    fft_len = 1 << ((size - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: size]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that approximately minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    n = max(2, n)
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Starting point (analytic or global)
    # ------------------------------------------------------------------
    a = _analytic_start(n)
    cand_global = _maybe_use_global()
    if cand_global is not None:
        cand_score = _compute_score(cand_global,
                                    _fft_len_for_n(cand_global.size),
                                    cand_global.size)
        start_score = _compute_score(a, fft_len, n)
        if cand_score < start_score:
            a = cand_global
            n = a.size
            fft_len = _fft_len_for_n(n)

    best = a.copy()
    best_score = _compute_score(best, fft_len, n)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # a small safety margin

    # ------------------------------------------------------------------
    # 1) Fast parametric sweep
    # ------------------------------------------------------------------
    # Power‑law exponents
    p_vals = np.linspace(0.35, 0.65, 13)          # 0.35 … 0.65
    offset_vals = np.linspace(0.0, 5.0, 11)       # 0 … 5
    # Exponential decay rates
    c_vals = np.linspace(2e-4, 1e-2, 15)         # 0.0002 … 0.01

    # Sweep pure power‑law
    for p in p_vals:
        for off in offset_vals:
            if time.monotonic() > deadline - 20.0:
                break
            i = np.arange(n, dtype=np.float64)
            a_candidate = (i + off) ** (-p)
            a_candidate /= a_candidate.sum()
            score = _compute_score(a_candidate, fft_len, n)
            if score < best_score:
                best_score = score
                best = a_candidate.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] "
                      f"pow‑law p={p:.3f} off={off:.1f} → {best_score:.6f}")

    # Sweep pure exponential
    for c in c_vals:
        if time.monotonic() > deadline - 20.0:
            break
        i = np.arange(n, dtype=np.float64)
        a_candidate = np.exp(-c * i)
        a_candidate /= a_candidate.sum()
        score = _compute_score(a_candidate, fft_len, n)
        if score < best_score:
            best_score = score
            best = a_candidate.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] "
                  f"exp c={c:.5f} → {best_score:.6f}")

    # Sweep combined power‑law × exponential
    for p in p_vals:
        for off in offset_vals:
            for c in c_vals:
                if time.monotonic() > deadline - 20.0:
                    break
                i = np.arange(n, dtype=np.float64)
                a_candidate = (i + off) ** (-p) * np.exp(-c * i)
                a_candidate /= a_candidate.sum()
                score = _compute_score(a_candidate, fft_len, n)
                if score < best_score:
                    best_score = score
                    best = a_candidate.copy()
                    print(f"[{time.monotonic() - start_time:6.2f}s] "
                          f"combo p={p:.3f} off={off:.1f} c={c:.5f} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Mirror‑descent polishing (soft‑max approximation of the max)
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score
    momentum = 0.85
    grad_prev = np.zeros_like(cur)

    while time.monotonic() < deadline - 2.0:   # keep a tiny buffer
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ---------- schedules ----------
        # Linear decay of learning rate η from 0.5 → 0.01
        eta = 0.5 * (1.0 - elapsed / total) + 0.01 * (elapsed / total)
        # Linear increase of soft‑max temperature β from 20 → 5000
        beta = 20.0 + (5000.0 - 20.0) * (elapsed / total)

        # ---------- convolution ----------
        conv = _conv_self(cur, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---------- soft‑max weights ----------
        # w_k ∝ exp(β·(conv_k − cur_max))
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient ----------
        grad = _gradient_correlation(cur, w)

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---------- exponentiated‑gradient update ----------
        factor = np.exp(-eta * grad)
        cur = cur * factor
        # projection onto the simplex
        cur = np.clip(cur, 0.0, None)
        s = cur.sum()
        if s == 0.0:
            cur = np.full_like(cur, 1.0 / n)
        else:
            cur /= s

        # ---------- keep best ----------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # occasional tiny random jitter to avoid stagnation
        if int(elapsed * 10) % 47 == 0:
            jitter = np.exp(rng.normal(0.0, 0.001, size=n))
            cur = cur * jitter
            cur = cur / cur.sum()

    # ------------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)

    # Return plain Python list
    return [float(v) for v in best]
