   # schedules

   eta_start = 1.0

   beta_end = 5000.0

   momentum = 0.9

        now = time.monotonic()

        conv = _self_conv(a, fft_len_self)
        max_c = conv.max()
        # compute gradient
        if now < phase1_end:
            # softmax gradient
            frac = (now - start) / (phase1_end - start)

"""
Propose a step‑function that improves the bound on
    evaluate_sequence(sequence) = 2·n·max(a*a)/(sum a)^2 .
The algorithm uses three optimisation phases for a fixed length:
    1️⃣ Softmax‑smoothed gradient (β grows → exact max).
    2️⃣ Active‑set gradient (all near‑max convolution entries).
    3️⃣ Exact sub‑gradient on the current peak.
Projected additive updates with Nesterov‑style momentum are applied,
and stagnation triggers a small random jitter around the current best.
A short warm‑start on a smaller size is performed, then the solution
is interpolated to the target size (3809 by default) and refined until
the time budget is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length) using FFT."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimisation of a single length
# ----------------------------------------------------------------------


def _optimise_one_length(a_init: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Optimise a vector on the simplex to minimise
        2·n·max(a*a)   (sum(a) is kept = 1)
    using three phases:
        • soft‑max gradient (β ↑)
        • active‑set gradient (ε ↓)
        • exact sub‑gradient
    Returns the best vector seen and its score.
    """
    a = a_init.copy()
    n = a.size

    # FFT lengths (powers of two, fixed for the whole run)
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # initialise best
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * conv.max()

    start = time.monotonic()
    total_time = deadline - start

    # --------------------------------------------------------------
    # Phase schedule (fractions of the total remaining time)
    # --------------------------------------------------------------
    phase1_end = start + 0.40 * total_time   # soft‑max phase
    phase2_end = start + 0.80 * total_time   # active‑set phase

    # hyper‑parameters
    eta_start = 1.0          # start step size (soft‑max)
    eta_end = 0.01           # end step size (soft‑max)
    beta_start = 0.5
    beta_end = 5000.0        # final smoothing temperature

    # active‑set parameters
    eps_start_factor = 0.10      # start with 10 % of the current peak
    eps_end_factor = 1e-8        # end with a tiny relative tolerance

    momentum = 0.9
    v = np.zeros_like(a)         # momentum buffer

    it = 0
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        # --------------------------------------------------------------
        # current convolution and peak
        # --------------------------------------------------------------
        conv = _self_conv(a, fft_len_self)
        max_c = conv.max()

        # --------------------------------------------------------------
        # Choose gradient according to the current phase
        # --------------------------------------------------------------
        if now < phase1_end:
            # ----------- phase 1 – soft‑max gradient -----------
            frac = (now - start) / (phase1_end - start)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # soft‑max weights (stable)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            # gradient = 2·(p * rev(a))  (FFT)
            a_rev = a[::-1]
            fp = np.fft.rfft(p, n=fft_len_grad)
            fra = np.fft.rfft(a_rev, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        elif now < phase2_end:
            # ----------- phase 2 – active‑set gradient ----------
            frac = (now - phase1_end) / (phase2_end - phase1_end)

            # step‑size – shrink a bit more than in phase 1
            eta = eta_end * (0.5 + 0.5 * (1.0 - frac))

            # relative tolerance for the active set
            eps = max_c * (eps_start_factor * (1.0 - frac) + eps_end_factor * frac)

            mask = (conv >= max_c - eps).astype(float)

            a_rev = a[::-1]
            fp = np.fft.rfft(mask, n=fft_len_grad)
            fra = np.fft.rfft(a_rev, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        else:
            # ----------- phase 3 – exact sub‑gradient ----------
            k_max = int(conv.argmax())
            grad = np.zeros_like(a)
            i_min = max(0, k_max - (n - 1))
            i_max = min(n - 1, k_max)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k_max - idx_i
            grad[idx_i] = 2.0 * a[idx_j]

            # classic √t diminishing step‑size
            eta = 0.20 / np.sqrt(it)

        # --------------------------------------------------------------
        # Momentum update and projected step
        # --------------------------------------------------------------
        v = momentum * v + grad
        a = a - eta * v
        a = _project_simplex(a)

        # --------------------------------------------------------------
        # Score of the current iterate
        # --------------------------------------------------------------
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}")

        # --------------------------------------------------------------
        # Stagnation handling – mild jitter around the current best
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            a = best.copy()
            # 5 % multiplicative log‑normal noise (fine‑grained)
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * noise
            a = _project_simplex(a)
            v = np.zeros_like(a)     # reset momentum
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine performs a brief warm‑start on a small size,
    interpolates to the target length (≈3800) and then refines it.
    Returns the best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Full time‑budget (5 s safety margin)
    # ------------------------------------------------------------------
    start_global = time.monotonic()
    deadline = start_global + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a target length (the best known length)
    # ------------------------------------------------------------------
    target_n = 3809          # default – the length that gave the previous best

    # ------------------------------------------------------------------
    # 2) Optional quick warm‑start on a smaller problem
    # ------------------------------------------------------------------
    warmup_n = 1024
    warmup_time = 30.0       # seconds allotted for the warm‑up phase
    if deadline - time.monotonic() > warmup_time + 10.0:
        warmup_deadline = time.monotonic() + warmup_time
        if "height_sequence_1" in globals():
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq0.size == warmup_n:
                a_warm = seq0.copy()
            else:
                # linear interpolation to the warm‑up size
                x_old = np.linspace(0.0, 1.0, seq0.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, warmup_n, endpoint=False)
                a_warm = np.interp(x_new, x_old, seq0)
                a_warm = a_warm.clip(min=0.0)
                s = a_warm.sum()
                a_warm = a_warm / s if s > 0 else np.full_like(a_warm, 1.0 / warmup_n)
        else:
            a_warm = rng.dirichlet(np.ones(warmup_n)).astype(np.float64)

        # run optimisation on the small size
        vec_warm, _ = _optimise_one_length(a_warm, warmup_deadline, rng)

        # interpolate the warm solution to the target size
        x_small = np.linspace(0.0, 1.0, warmup_n, endpoint=False)
        x_target = np.linspace(0.0, 1.0, target_n, endpoint=False)
        a_init = np.interp(x_target, x_small, vec_warm)
    else:
        # No warm‑up – initialise directly at the target size
        if "height_sequence_1" in globals():
            seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq0.size == target_n:
                a_init = seq0.copy()
            else:
                x_old = np.linspace(0.0, 1.0, seq0.size, endpoint=False)
                x_target = np.linspace(0.0, 1.0, target_n, endpoint=False)
                a_init = np.interp(x_target, x_old, seq0)
                a_init = a_init.clip(min=0.0)
                s = a_init.sum()
                a_init = a_init / s if s > 0 else np.full_like(a_init, 1.0 / target_n)
        else:
            a_init = rng.dirichlet(np.ones(target_n)).astype(np.float64)

    # ensure we are on the probability simplex
    a_init = a_init.clip(min=0.0)
    s = a_init.sum()
    a_init = a_init / s if s > 0 else np.full_like(a_init, 1.0 / target_n)

    # ------------------------------------------------------------------
    # 3) Main optimisation on the target length
    # ------------------------------------------------------------------
    vec_best, best_score = _optimise_one_length(a_init, deadline, rng)

    # ------------------------------------------------------------------
    # 4) (optional) try a slightly larger length if time permits
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        larger_n = 4096
        # initialise from the current best by interpolation
        x_src = np.linspace(0.0, 1.0, target_n, endpoint=False)
        x_lg = np.linspace(0.0, 1.0, larger_n, endpoint=False)
        a_lg = np.interp(x_lg, x_src, vec_best)
        a_lg = a_lg.clip(min=0.0)
        s = a_lg.sum()
        a_lg = a_lg / s if s > 0 else np.full_like(a_lg, 1.0 / larger_n)

        vec_lg, score_lg = _optimise_one_length(a_lg, deadline, rng)
        if score_lg < best_score:
            vec_best, best_score = vec_lg, score_lg

    # ------------------------------------------------------------------
    # 5) Clip to the limits enforced by ``evaluate_sequence`` and return
    # ------------------------------------------------------------------
    vec_best = np.clip(vec_best, 0.0, 1000.0)

    # Debug line – final score (will also be printed by the harness)
    final_score = 2.0 * vec_best.size * np.max(np.convolve(vec_best, vec_best))
    print(f"FINAL SCORE ≈ {final_score:.6f} (length {vec_best.size})")

    return [float(x) for x in vec_best]
