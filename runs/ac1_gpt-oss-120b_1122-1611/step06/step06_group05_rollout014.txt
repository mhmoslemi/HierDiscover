"""
Propose a step‑function (non‑negative heights) that (approximately) minimises
`evaluate_sequence`.  The algorithm is a multi‑phase search:

1.  Random sampling of Dirichlet‑distributed vectors (many cheap candidates).
2.  A sub‑gradient descent on the true ℓ∞‑objective
    (max‑convolution), using the exact gradient
    ∇‖a∗a‖∞ = 2·(a ⋆ w) where w is a probability vector supported on the
    indices attaining the maximal convolution value.
3.  Adam‑style momentum, learning‑rate annealing and occasional jitter.
4.  Restarts after a period without improvement and occasional re‑initialisation
    with a different length.

All heavy work is performed with FFT‑based convolution/correlation
( O(n log n) ).  The routine respects the 1000 s budget, prints progress
stamps and finally returns the best sequence found (clipped to the limits
enforced by `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that holds the full linear convolution
    of two vectors of length n (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution via FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_corr(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """Correlation a ⋆ w needed for the sub‑gradient of max(a∗a).
    Returns a vector of length len(a); factor 2 accounts for the two
    symmetric terms in the convolution."""
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: 3 * a.size - 2]
    start = a.size - 1
    return 2.0 * corr_full[start:start + a.size]


def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of evaluate_sequence (without the safety checks)."""
    conv = _conv_self_fft(a, _fft_len_for_n(n))
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    `evaluate_sequence` within the allotted time.  The function returns the
    best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ----- problem size -------------------------------------------------
    n = int(kwargs.get("n", 3000))
    if n < 2:
        n = 2

    # ----- FFT lengths (fixed for a given n) ----------------------------
    fft_len_conv = _fft_len_for_n(n)                     # for a∗a
    fft_len_corr = 1 << ((3 * n - 2 - 1).bit_length())   # for correlation

    # ----- bookkeeping --------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # safety margin
    best_a = None
    best_score = np.inf
    last_improve = start_time

    # ------------------------------------------------------------------
    # Phase 1 – cheap random sampling (Dirichlet draws)
    # ------------------------------------------------------------------
    samples_per_sec = 2000  # heuristic, enough to fill the first ~30 s
    sample_deadline = start_time + min(30.0, budget_s * 0.1)

    while time.monotonic() < sample_deadline:
        # draw a Dirichlet vector (concentration=1 → uniform on the simplex)
        a = rng.dirichlet(np.ones(n))
        cur_score = _objective(a, n)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] random sample improvement → {best_score:.6f}")

    # If we managed to load a previously discovered construction, give it priority
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                # normalise to sum = 1 (the scale does not affect the objective)
                if cand.sum() > 0.0:
                    cand = cand / cand.sum()
                    cand_score = _objective(cand, cand.size)
                    if cand_score < best_score:
                        best_score = cand_score
                        best_a = cand.copy()
                        n = cand.size
                        fft_len_conv = _fft_len_for_n(n)
                        fft_len_corr = 1 << ((3 * n - 2 - 1).bit_length())
                        print(f"[{time.monotonic() - start_time:6.2f}s] "
                              f"adopt previous best → {best_score:.6f}")
        except Exception:
            pass

    # If we still have no candidate (should not happen), initialise with a simple shape
    if best_a is None:
        a0 = np.sqrt(np.arange(n, dtype=np.float64) + 1.0) - np.sqrt(np.arange(n, dtype=np.float64))
        a0 /= a0.sum()
        best_a = a0.copy()
        best_score = _objective(best_a, n)

    # ------------------------------------------------------------------
    # Phase 2 – sub‑gradient (Adam) optimisation of the true max‑convolution
    # ------------------------------------------------------------------
    # Adam state
    x = np.log(best_a + 1e-30)          # logits – we optimise in log‑space for positivity
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_start = 0.8
    lr_end = 0.02
    restart_gap = 45.0    # seconds without improvement → re‑initialise
    jitter_prob = 0.03
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ----- schedules -------------------------------------------------
        lr = lr_start + (lr_end - lr_start) * (elapsed / total)

        # ----- recover a from logits ------------------------------------
        a = np.exp(x - np.max(x))       # numerical stability
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.ones_like(a) / a.size
            a_sum = 1.0
        a /= a_sum

        # ----- compute exact convolution --------------------------------
        conv = _conv_self_fft(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (a_sum ** 2)

        # ----- record improvement ---------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (Adam) → {best_score:.6f}")

        # ----- construct sub‑gradient (active indices) -----------------
        # consider all indices whose value is within a tiny tolerance of the max
        tol = max(1e-12, cur_max * 1e-12)
        active = conv >= (cur_max - tol)
        w = active.astype(np.float64)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[0] = 1.0
            w_sum = 1.0
        w /= w_sum

        # ----- gradient w.r.t. a (exact sub‑gradient) -------------------
        grad_a = _grad_corr(a, w, fft_len_corr)   # shape (n,)

        # ----- convert to gradient w.r.t. logits -----------------------
        # Jacobian of soft‑max: grad_x = a * (grad_a - <a,grad_a>)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- occasional jitter ----------------------------------------
        if rng.random() < jitter_prob:
            x += rng.normal(0.0, 0.06, size=n)

        # ----- restart if stuck -----------------------------------------
        if now - last_improve > restart_gap:
            # re‑initialise from the current best (plus small noise)
            x = np.log(best_a + 1e-30)
            x += rng.normal(0.0, 0.15, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (respect clipping limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
