def _softmax_weights(conv, beta):
    max_conv = conv.max()
    diff = conv - max_conv  # non-positive
    # compute exp(beta*diff) with overflow protection
    # to avoid underflow, we can shift by max of diff (0) is fine.
    w_raw = np.exp(beta * diff)
    sum_raw = w_raw.sum()
    if sum_raw == 0 or not np.isfinite(sum_raw):
        w = (conv == max_conv).astype(float)
        sum_raw = w.sum()
    else:
        w = w_raw / sum_raw
    return w
```

"""
propose_candidate
=================

Search for a non‑negative step‑function (probability vector) that
minimises ``evaluate_sequence``.  The algorithm consists of three stages:

1. **Analytic seeding** – generate a collection of candidate shapes
   (sqrt‑differences, power laws, symmetric power laws, discretised
   arcsine density) for a single length *n* (default 4000).  The best
   candidate is kept.

2. **Smooth Adam optimisation** – minimise a soft‑max surrogate of the
   L∞‑norm of the auto‑convolution using Adam on the simplex.
   The temperature ``β`` is increased from 0.5 to about 4000, turning
   the surrogate gradually into the true max.  Occasional multiplicative
   log‑normal noise and small mass‑transfer moves help escape plateaus.
   If no improvement is seen for a while a jittered restart is performed.

3. **Simulated‑annealing refinement** – a short stochastic hill‑climb
   that moves a random amount of mass between two coordinates,
   accepting uphill moves with a temperature schedule.

All steps respect the 1000 s time budget.  The best vector found is
clipped to the limits of ``evaluate_sequence`` and returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length for a linear convolution of size n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for use with FFT."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return linear convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Projection onto the probability simplex (non‑negative, unit sum).
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("input must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Normalize to eliminate numeric drift
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (correlation a ⋆ w, multiplied by 2).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector.
    w : np.ndarray, shape (2·n‑1,)
        Non‑negative weights (soft‑max or indicator).

    Returns
    -------
    grad : np.ndarray, shape (n,)
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # linear correlation length = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Soft‑max weights for the surrogate objective.
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute w_k ∝ exp(beta·(conv_k - max(conv))).
    Guarantees a proper distribution even if the exponent underflows.
    """
    max_conv = conv.max()
    diff = conv - max_conv               # ≤ 0
    # exponent may under‑flow; that's fine.
    w_raw = np.exp(beta * diff)
    w_sum = w_raw.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        mask = conv == max_conv
        w_raw = np.where(mask, 1.0, 0.0)
        w_sum = w_raw.sum()
    w = w_raw / w_sum
    return w


# ----------------------------------------------------------------------
# Objective for a probability vector (sum = 1): 2·n·max(a∗a).
# ----------------------------------------------------------------------
def _compute_score(a: np.ndarray) -> float:
    n = a.size
    fft_len = _fft_len_for_n(n)
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic shape generators (return *unnormalised* vectors).
# ----------------------------------------------------------------------
def _shape_sqrtdiff(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return np.sqrt(i + 1.0) - np.sqrt(i)


def _shape_power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    return i ** (-p)


def _shape_arcsine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))


def _shape_sym_power(n: int, p: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    return (i * (n - i + 1)) ** (-p)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Desired length of the sequence (default = 4000).

    Returns
    -------
    list[float]
        The best sequence found (values already clipped to the limits of
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)   # a tiny safety margin

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 4000))
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 2. Analytic seeding – try a handful of families and some random draws.
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # Helper to evaluate a raw (non‑normalised) shape.
    def _eval_shape(v_raw):
        nonlocal best_a, best_score
        if v_raw.ndim != 1 or v_raw.size == 0:
            return
        # Clip negatives (should not happen) and normalise.
        v = np.maximum(v_raw, 0.0)
        s = v.sum()
        if s <= 0.0:
            return
        v /= s
        sc = _compute_score(v)
        if sc < best_score:
            best_score = sc
            best_a = v.copy()

    # -- deterministic families -------------------------------------------------
    _eval_shape(_shape_sqrtdiff(n))

    p_vals = [0.5, 0.6, 0.7, 0.8, 0.9, 1.0, 1.2, 1.5]
    for p in p_vals:
        _eval_shape(_shape_power(n, p))
        _eval_shape(_shape_sym_power(n, p))

    _eval_shape(_shape_arcsine(n))

    # -- random Dirichlet‑type draws -------------------------------------------
    for _ in range(25):
        v_raw = rng.exponential(scale=1.0, size=n)   # all positive
        _eval_shape(v_raw)

    # -- optionally incorporate a previously known construction -----------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    cand = cand / cand.sum()
                    _eval_shape(cand)
        except Exception:
            pass

    # If, for some reason, we still have no candidate, fall back to uniform.
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_score(best_a)

    # ------------------------------------------------------------------
    # 3. Smooth Adam optimisation on the soft‑max surrogate.
    # ------------------------------------------------------------------
    a = best_a.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    lr_start = 0.5
    lr_end = 0.001
    beta_start = 0.5
    beta_end = 4000.0

    last_improve = start
    restart_gap = 120.0                     # seconds without improvement → jittered restart

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start
        total = deadline - start
        ratio = min(1.0, elapsed / total)

        # schedule
        lr = lr_start * (1.0 - ratio) + lr_end * ratio
        beta = beta_start + (beta_end - beta_start) * ratio

        # convolution & score
        fft_len = _fft_len_for_n(a.size)
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * a.size * conv.max()

        # best‑track
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # choose weights: soft‑max while beta is moderate; otherwise pure max‑indicator
        if beta < beta_end * 0.7:
            w = _softmax_weights(conv, beta)
        else:
            mask = conv == conv.max()
            w = mask.astype(float)
            w_sum = w.sum()
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full_like(conv, 1.0 / conv.size)

        # gradient
        grad = _gradient_correlation(a, w)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _proj_simplex(a)

        # occasional multiplicative log‑normal jitter
        if rng.random() < 0.032:
            factor = np.exp(rng.normal(0.0, 0.20, size=a.size))
            a = a * factor
            a = _proj_simplex(a)

        # occasional small mass‑transfer between two coordinates
        if rng.random() < 0.07:
            i, j = rng.integers(0, a.size, size=2)
            if i != j and a[i] > 1e-12:
                delta = rng.random() * a[i] * 0.5
                a[i] -= delta
                a[j] += delta
                a = _proj_simplex(a)

        # restart if stuck for a while
        if now - last_improve > restart_gap:
            a = best_a.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=a.size))
            a = a * jitter
            a = _proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best still {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Simulated‑annealing refinement (uses remaining time, if any).
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        a = best_a.copy()
        cur_score = best_score
        T0 = 0.015        # initial temperature
        while time.monotonic() < deadline:
            now = time.monotonic()
            elapsed = now - start
            # linear cooling
            T = T0 * max(0.0, 1.0 - elapsed / total)

            # propose a random mass shift
            i, j = rng.integers(0, a.size, size=2)
            if i == j or a[i] < 1e-14:
                continue
            delta = rng.random() * a[i] * 0.6
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            a_new = _proj_simplex(a_new)
            new_score = _compute_score(a_new)

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new
                cur_score = new_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{elapsed:6.2f}s] SA improved best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (apply the clipping imposed by
    #    evaluate_sequence).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
