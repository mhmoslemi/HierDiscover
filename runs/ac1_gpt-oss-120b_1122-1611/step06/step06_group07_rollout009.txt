"""
Propose a non‑negative step function that (approximately) minimises
    score = 2·n·max_k (a∗a)_k / (sum a)^2 .
The code combines three ideas that were not all used together before:

1.  Analytic families – in addition to the previous (r‑, p‑) families we also
    scan a two‑parameter Beta‑distribution family a_i ∝ x^{α‑1}(1‑x)^{β‑1},
    which is known to produce very “U‑shaped’’ step‑functions that lower the
    peak of the auto‑convolution.

2.  Adaptive gradient descent – a fast Adam‑style optimizer works on the
    log‑it (soft‑max) parametrisation a_i = exp(x_i)/∑exp(x_j).  The gradient
    is taken w.r.t. the exact max‑convolution (a one‑hot weight), which gives a
    true sub‑gradient of the L∞ objective.

3.  Local stochastic refinement – after the Adam phase a cheap simulated‑anneal
    moves small amounts of mass between two random indices; this helps to
    escape the last few plateaus.

All steps respect the given 1000 s budget.  The best sequence found is
returned as a plain Python list of floats (already clipped to the limits of
`evaluate_sequence`).
"""

import time
import numpy as np
from scipy.special import gammaln

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two that can hold a linear convolution of length 2·n‑1."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a (length 2·n‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of (a*a)·w with respect to a.
    Implements 2·a ⋆ w   (correlation) using a single FFT‑based convolution.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective (sum a_i is assumed to be 1)."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^r - i^r   (decreasing for 0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_shape_p(p: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^(-p) - (i+2)^(-p)."""
    k = np.arange(1, n + 2, dtype=np.float64)   # 1 .. n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_beta_shape(alpha: float, beta: float, n: int) -> np.ndarray | None:
    """
    Discrete Beta‑pdf shape.
        a_i ∝ ( (i+0.5)/n )^{α‑1} * ( 1‑(i+0.5)/n )^{β‑1}
    The result is normalised to sum 1, clipped to [0,1000] and returned.
    """
    if alpha <= 0.0 or beta <= 0.0:
        return None
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # work in log‑space for stability
    log_a = (alpha - 1.0) * np.log(x) + (beta - 1.0) * np.log(1.0 - x)
    log_a -= np.max(log_a)           # avoid overflow
    a = np.exp(log_a)
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    # clip before returning – matches `evaluate_sequence`'s safety net
    a = np.clip(a, 0.0, 1000.0)
    return a


# ----------------------------------------------------------------------
# Optimiser utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, sum x = 1}."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _adam_optimize(
    init_a: np.ndarray,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the logits of the probability vector.
    Uses the exact sub‑gradient of the max‑convolution.
    Returns the best vector seen during the run and its score.
    """
    n = init_a.size
    # logits (log‑softmax parameters)
    x = np.log(np.clip(init_a, 1e-16, None))
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    best_vec = init_a.copy()
    best_sc = _score(best_vec, fft_len)
    start = time.time()

    while time.time() < deadline:
        t += 1
        # ---- probabilities from logits ----
        a = np.exp(x)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ---- evaluate ----
        conv = _conv_self(a, fft_len)
        cur_sc = float(2.0 * n * conv.max())
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_vec = a.copy()
            if verbose:
                elapsed = time.time() - start
                print(f"[{elapsed:6.2f}s] Adam best = {best_sc:.6f}")

        # ---- gradient of the exact max (one‑hot) ----
        k = int(conv.argmax())
        w = np.zeros_like(conv)
        w[k] = 1.0
        grad_a = _gradient_via_correlation(a, w)  # ∂max/∂a

        # ---- chain rule to logits ----
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = 0.1 / (np.sqrt(v_hat) + eps)
        x -= step * m_hat

        # occasional jitter to avoid stagnation
        if rng.random() < 0.01:
            x += rng.normal(0.0, 0.05, size=n)

    return best_vec, best_sc


def _local_anneal(
    init_a: np.ndarray,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
    verbose: bool = False,
) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing style hill‑climbing.
    Moves a random fraction of mass from one index to another.
    """
    cur = init_a.copy()
    cur_sc = _score(cur, fft_len)
    best_vec = cur.copy()
    best_sc = cur_sc
    n = cur.size
    start = time.time()
    while time.time() < deadline:
        # linear cooling
        elapsed = time.time() - start
        tfrac = elapsed / (deadline - start)
        temp = max(0.001, 0.05 * (1.0 - tfrac))

        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] < 1e-15:
            continue
        # take up to 10 % of the source mass
        delta = cur[i] * rng.random() * 0.1
        new = cur.copy()
        new[i] -= delta
        new[j] += delta
        new_sc = _score(new, fft_len)

        accept = False
        if new_sc < cur_sc:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp((cur_sc - new_sc) / (temp + 1e-12)):
                accept = True

        if accept:
            cur = new
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_vec = new.copy()
                if verbose:
                    elapsed = time.time() - start
                    print(f"[{elapsed:6.2f}s] anneal new best = {best_sc:.6f}")

    return best_vec, best_sc

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence`.  The algorithm respects the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 1) Scan analytic families (r, p, Beta)
    # --------------------------------------------------------------
    best_a = None
    best_score = np.inf

    # (a) r‑family
    for r in np.linspace(0.5, 0.95, 91):
        a = _gen_shape_r(r, n)
        if a is None:
            continue
        sc = _score(a, fft_len)
        if sc < best_score:
            best_a, best_score = a, sc

    # (b) p‑family
    for p in np.linspace(0.1, 2.0, 96):
        a = _gen_shape_p(p, n)
        if a is None:
            continue
        sc = _score(a, fft_len)
        if sc < best_score:
            best_a, best_score = a, sc

    # (c) Beta‑family (two parameters)
    alphas = np.linspace(0.25, 2.0, 18)          # step ≈0.1
    betas = np.linspace(0.25, 5.0, 30)          # step ≈0.16
    for alpha in alphas:
        for beta in betas:
            if alpha + beta <= 1.02:            # avoid singularities
                continue
            a = _gen_beta_shape(alpha, beta, n)
            if a is None:
                continue
            sc = _score(a, fft_len)
            if sc < best_score:
                best_a, best_score = a, sc

    # --------------------------------------------------------------
    # 2) Keep a few random Dirichlet seeds (they often discover new
    #    basins of attraction).  Store the best 5 of them.
    # --------------------------------------------------------------
    random_inits = []
    for _ in range(30):
        conc = rng.uniform(0.1, 5.0)                     # concentration parameter
        a = rng.gamma(conc, 1.0, size=n)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a = a / a_sum
        sc = _score(a, fft_len)
        random_inits.append((sc, a))
        if sc < best_score:
            best_a, best_score = a, sc

    # keep only the top 5 random starts for the heavy optimiser
    random_inits.sort(key=lambda tup: tup[0])
    top_random = [vec for _, vec in random_inits[:5]]

    # --------------------------------------------------------------
    # 3) Adam optimisation – start from the best analytic shape and also
    #    from the top random seeds.  Give each run a time slice.
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # a few seconds margin

    # allocate about half of the remaining budget to Adam, the rest to
    # the cheap annealing phase.
    remaining = deadline - time.time()
    adam_budget = remaining * 0.55
    anneal_budget = remaining - adam_budget

    # a list of starting points for Adam
    starters = [best_a] + top_random
    adam_deadline = time.time() + adam_budget

    for start_vec in starters:
        cur_deadline = min(adam_deadline, time.time() + adam_budget / max(1, len(starters)))
        refined, refined_sc = _adam_optimize(start_vec, fft_len, cur_deadline, rng, verbose=False)
        if refined_sc < best_score:
            best_a, best_score = refined, refined_sc

    # --------------------------------------------------------------
    # 4) Final cheap local annealing to polish the best vector.
    # --------------------------------------------------------------
    anneal_deadline = time.time() + anneal_budget
    polished, polished_sc = _local_anneal(best_a, fft_len, anneal_deadline, rng, verbose=False)
    if polished_sc < best_score:
        best_a, best_score = polished, polished_sc

    # --------------------------------------------------------------
    # 5) Return the best vector found (already normalised, safe for the
    #    security checks in `evaluate_sequence`).
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)      # respect the hard caps
    return [float(v) for v in best_a]
```
