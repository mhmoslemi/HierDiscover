Algorithm:

def project_simplex(v):
    # v: numpy array, arbitrary
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1 - cssv) / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Algorithm
---------
1. Build a large, diverse pool of analytic candidates.
   * Power‑difference families a_i = (i+α+1)^β − (i+α)^β  (β≈0.3–0.7,
     α∈[0,5]).
   * Exponential families a_i = (1−e^{−λ})·e^{−λ·(i+δ)} (λ≈0.001–0.05,
     δ∈[0,5]).
   * The classic √‑difference start.
   * Random Dirichlet draws.
   * The previous best construction if present.
   All vectors are normalised to sum = 1.

2. Exhaustively evaluate the pool (FFT‑based convolution) and keep the
   best candidate.

3. Refine the best candidate with projected gradient descent on a smooth
   surrogate of the L∞‑objective:
      – soft‑max over the full convolution (temperature β increases).
      – gradient = 2·a⋆w (correlation, obtained by FFT).
      – additive update followed by Euclidean projection onto the
        probability simplex (Duchi et al. 2008).
      – learning‑rate η linearly decays, momentum stabilises updates.
      – occasional random block perturbations and pairwise mass‑transfer
        moves keep the iterate from stagnating.

4. In the last ~20 s a short simulated‑annealing phase performs random
   mass‑transfer moves, accepting uphill steps with a decreasing temperature.
   The best vector seen throughout the whole run is retained.

5. Return the best vector (clipped to the limits of evaluate_sequence()).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length sufficient for a full self‑convolution.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Fast exact L∞‑objective for a probability vector (sum = 1).
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Gradient of the soft‑max surrogate (weights w of length 2·n‑1).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Returns ∇_a  Σ_k w_k (a∗a)_k = 2·a⋆w,
    where ⋆ denotes the (full) correlation.
    """
    n = a.size
    L = w.size                # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # a reversed because correlation = convolution with reversed kernel
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean projection).
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx=1}."""
    # reference: Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = u + (1.0 - cssv) / np.arange(1, v.size + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are zero – fall back to uniform
        return np.full_like(v, 1.0 / v.size)
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (allow a user supplied n, default 3800)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Helper to normalise a raw vector to a probability distribution.
    # ------------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # ------------------------------------------------------------------
    # 1) Exhaustive search over analytic families.
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    start_time = time.monotonic()

    # a) Classic √‑difference start (β = 0.5, α = 0)
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    a = _normalise(a_sqrt)
    sc = _score_of(a, n, fft_len)
    best, best_score = a, sc
    print(f"[init] √‑diff score = {sc:.9f}")

    # b) Previous construction if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                a = _normalise(prev)
                sc = _score_of(a, n, fft_len)
                if sc < best_score:
                    best, best_score = a, sc
                    print(f"[init] previous construction score = {sc:.9f}")
        except Exception:
            pass

    # c) Power‑difference families   a_i = (i+α+1)^β – (i+α)^β
    betas = np.linspace(0.35, 0.65, 61)          # fine grid around 0.5
    alphas = np.linspace(0.0, 5.0, 21)           # shift the origin
    for beta in betas:
        for alpha in alphas:
            idx = np.arange(n, dtype=np.float64) + alpha
            a = ((idx + 1.0) ** beta) - (idx ** beta)
            a = _normalise(a)
            sc = _score_of(a, n, fft_len)
            if sc < best_score:
                best, best_score = a, sc
                print(f"[grid] β={beta:.4f} α={alpha:.2f} → {sc:.9f}")

    # d) Exponential families   a_i = (1‑e^{‑λ})·e^{‑λ·(i+δ)}
    lambdas = np.linspace(0.001, 0.04, 40)       # approx 0.001 … 0.04
    deltas = np.linspace(0.0, 5.0, 11)           # offset the start
    for lam in lambdas:
        coeff = 1.0 - np.exp(-lam)
        for delta in deltas:
            a = coeff * np.exp(-lam * (np.arange(n, dtype=np.float64) + delta))
            a = _normalise(a)
            sc = _score_of(a, n, fft_len)
            if sc < best_score:
                best, best_score = a, sc
                print(f"[exp] λ={lam:.4f} δ={delta:.2f} → {sc:.9f}")

    # e) Random Dirichlet samples (more exploration)
    for _ in range(200):
        alpha = rng.uniform(0.1, 2.0)
        a = rng.gamma(alpha, 1.0, size=n)
        a = _normalise(a)
        sc = _score_of(a, n, fft_len)
        if sc < best_score:
            best, best_score = a, sc
            print(f"[rand] Dirichlet α={alpha:.4f} → {sc:.9f}")

    print(f"[init] best initial score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 2) Refine with projected gradient descent on a soft‑max surrogate.
    # ------------------------------------------------------------------
    cur = best.copy()
    grad_prev = np.zeros_like(cur)

    # timing
    total_budget = float(budget_s)
    now = time.monotonic()
    deadline = start_time + max(0.0, total_budget - 20.0)   # reserve 20 s for SA

    # schedules
    eta_start, eta_end = 0.5, 0.001
    beta_start, beta_end = 0.5, 5000.0
    momentum = 0.85

    # exploration probabilities
    prob_block = 0.04
    prob_pair = 0.10

    while time.monotonic() < deadline:
        # ---- current convolution & objective ----
        conv = _conv_self(cur, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- keep global best ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.9f}")

        # ---- linear schedules for step‑size and soft‑max temperature ----
        now = time.monotonic()
        elapsed = now - start_time
        frac = (elapsed / (deadline - start_time)) if deadline > start_time else 0.0
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ---- soft‑max weights over the *full* convolution ----
        # (shifted by the current max for numerical stability)
        w_raw = np.exp(beta * (conv - cur_max))
        w = w_raw / w_raw.sum()

        # ---- gradient of the surrogate ----
        grad = _gradient_correlation(cur, w)

        # ---- momentum smoothing ----
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ---- additive update & projection onto simplex ----
        cur = cur - eta * grad
        cur = _project_simplex(cur)

        # ---- occasional random block perturbation (exploration) ----
        if rng.random() < prob_block:
            width = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - width + 1)
            blk_factor = np.exp(rng.normal(0.0, 0.2))
            cur[start_idx:start_idx + width] *= blk_factor
            cur = _project_simplex(cur)

        # ---- occasional pairwise mass transfer ----
        if rng.random() < prob_pair:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur[i] > 1e-14:
                delta = rng.random() * cur[i] * 0.5
                cur[i] -= delta
                cur[j] += delta
                cur = _project_simplex(cur)

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing fine‑tuning (remaining time)
    # ------------------------------------------------------------------
    sa_deadline = start_time + total_budget
    cur_sa = best.copy()
    cur_sa_score = best_score
    T_start = 5e-3
    T_end = 1e-6
    it = 0

    while time.monotonic() < sa_deadline:
        # exponential cooling based on elapsed fraction
        now = time.monotonic()
        frac = (now - deadline) / (sa_deadline - deadline) if sa_deadline > deadline else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        # propose a random mass‑transfer move
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_sa[i] < 1e-14:
            continue
        delta = rng.random() * cur_sa[i] * 0.5
        new_a = cur_sa.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # evaluate
        new_score = _score_of(new_a, n, fft_len)

        # Metropolis acceptance
        if new_score < cur_sa_score or rng.random() < np.exp(-(new_score - cur_sa_score) / max(T, 1e-12)):
            cur_sa = new_a
            cur_sa_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = new_a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        it += 1

        # every few thousand iterations apply a tiny smoothing filter
        if it % 5000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            cur_sa = smoothed
            cur_sa_score = _score_of(cur_sa, n, fft_len)

    # ------------------------------------------------------------------
    # 4) Return the best vector (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # optional sanity check (does not affect the returned value)
    try:
        final_val = evaluate_sequence(best.tolist())
        print(f"[final] evaluate_sequence = {final_val:.9f}")
    except Exception:
        pass
    return [float(v) for v in best]
