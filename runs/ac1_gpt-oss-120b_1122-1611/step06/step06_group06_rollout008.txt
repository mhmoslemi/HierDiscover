def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Grad of Σ_k w_k (a∗a)_k = 2·a ⋆ w (correlation)."""
    n = a.size
    L = w.size                # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)

    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

"""
Propose a non‑negative step‑function (a list of heights) that minimizes
the provided ``evaluate_sequence`` objective.

Algorithm
---------
1. **Broad initial sampling** – generate a diverse pool of candidates:
      * Analytic √‑difference shape.
      * Power‑difference families a_i = (i+1)^‑p – (i+2)^‑p for a dense grid of p.
      * Exponential families a_i = (1‑e^{‑λ}) e^{‑λ i} for a log‑spaced grid of λ.
      * Random Dirichlet samples with a variety of concentration parameters.
      * The previous best construction (if the variable ``height_sequence_1``
        exists in the global namespace).

   Every candidate is normalised to unit ℓ₁‑norm and scored with a fast
   FFT‑based evaluation of  
   `score = 2·n·max(a∗a)/(sum(a)²)`.  The best **K** candidates (K≈7) are kept.

2. **Refinement by Adam on the log‑simplex** – each of the retained candidates
   is refined independently using projected Adam on the log‑odds (logits) of
   the simplex:
      * logits → soft‑max → probability vector `a`.
      * The smoothed objective uses a temperature‑scaled soft‑max over the
        self‑convolution (`β` grows linearly from 0.1 to 5000).
      * Gradient of the smooth surrogate is computed via the correlation
        `∇_a Σ w_k (a∗a)_k = 2·a⋆w`.
      * The gradient is transformed to logits (`∇_ℓ = a·(∇_a – ⟨a,∇_a⟩)`),
        and Adam updates (`m`, `v`) are applied.
      * The best vector seen throughout the whole run (all candidates,
        all iterations) is recorded.

   The optimisation runs until the main budget is exhausted, leaving a short
   (~30 s) window for the final phase.

3. **Simulated‑annealing fine‑tuning** – starting from the globally best
   vector, perform cheap random mass‑transfer moves (`a_i ← a_i – δ`,
   `a_j ← a_j + δ`).  Moves are accepted with the Metropolis criterion using an
   exponentially cooling temperature.  Every few thousand iterations a mild
   three‑point smoothing kernel is applied to help escape shallow plateaus.

4. **Return** the best sequence (clipped to the limits required by
   ``evaluate_sequence``) as a plain list of floats.

The routine respects the global time budget (default 1000 s), prints periodic
progress messages and never relies on file I/O or external services.
"""

import time
import numpy as np
import heapq

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can accommodate a linear convolution of length 2·n‑1."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Exact linear self‑convolution via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k.
    Returns 2·a ⋆ w, i.e. the correlation of a with w.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear convolution of w (len L) with a_rev (len n)
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Fast objective (scale‑invariant)
# ----------------------------------------------------------------------
def _fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return the exact value of ``evaluate_sequence`` for a non‑negative vector."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _conv_self(a, fft_len)
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    n = max(2, n)
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 2) normalisation helper (projects onto the ℓ₁‑simplex)
    # --------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # --------------------------------------------------------------
    # 3) generate a large, diverse pool of starting vectors
    # --------------------------------------------------------------
    K_TOP = 7                     # number of elite candidates to keep
    top_heap = []                 # min‑heap storing (‑score, vector) -> max‑heap behaviour

    def _push_candidate(vec: np.ndarray):
        sc = _fast_score(vec, n, fft_len)
        # store negative score so that heap[0] is the *worst* among the kept ones
        entry = (-sc, vec.copy())
        if len(top_heap) < K_TOP:
            heapq.heappush(top_heap, entry)
        else:
            if entry[0] > top_heap[0][0]:          # better (less negative) score
                heapq.heapreplace(top_heap, entry)

    # a) analytic √‑difference start
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    _push_candidate(_normalise(a_sqrt))

    # b) previous construction, if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                _push_candidate(_normalise(prev))
        except Exception:
            pass

    # c) power‑difference families (dense p‑grid)
    for p in np.linspace(0.05, 2.5, 150):
        base = np.arange(1, n + 2, dtype=np.float64)  # size n+1
        a = base[:-1] ** (-p) - base[1:] ** (-p)
        _push_candidate(_normalise(a))

    # d) exponential families (log‑spaced λ)
    for lam in np.logspace(np.log10(5e-5), np.log10(0.03), 70):
        a = (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))
        _push_candidate(_normalise(a))

    # e) random Dirichlet samples
    for _ in range(500):
        alpha = rng.uniform(0.1, 5.0)
        a = rng.gamma(alpha, 1.0, size=n)
        _push_candidate(_normalise(a))

    # --------------------------------------------------------------
    # 4) extract elite vectors (sorted from best to worst)
    # --------------------------------------------------------------
    top_candidates = [vec for (_, vec) in sorted(top_heap, key=lambda x: -x[0])]
    best_score = np.inf
    best_vec = None

    # --------------------------------------------------------------
    # 5) Adam‑based refinement on the log‑simplex
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # reserve ~30 s for later simulated‑annealing
    main_deadline = start_time + max(0.0, float(budget_s) - 30.0)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.5                       # initial learning rate (on logits)

    # schedule for the temperature β in the soft‑max surrogate
    beta_start, beta_end = 0.1, 5000.0

    for init_idx, init_vec in enumerate(top_candidates):
        # stop early if the global deadline is reached
        if time.monotonic() >= main_deadline:
            break

        # initialise logits (log‑softmax representation)
        logits = np.log(init_vec + 1e-12)

        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        t_iter = 0

        # give each elite candidate roughly the same share of remaining time
        # (simple round‑robin; the inner loop checks the deadline anyway)
        while time.monotonic() < main_deadline:
            t_iter += 1

            # ---- probability vector (simplex) ----
            a = _softmax(logits)                     # guarantees sum(a)=1, a_i≥0

            # ---- self‑convolution and current score ----
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max               # sum(a)=1 ⇒ denominator =1

            # ---- global best tracking ----
            if cur_score < best_score:
                best_score = cur_score
                best_vec = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (Adam #{init_idx}) = {best_score:.9f}")

            # ---- temperature schedule ----
            now = time.monotonic()
            frac = (now - start_time) / (main_deadline - start_time)
            beta = beta_start + (beta_end - beta_start) * frac

            # ---- smooth soft‑max weights over the convolution ----
            # subtract cur_max for numerical stability before scaling by β
            w_raw = np.exp(beta * (conv - cur_max))
            w_sum = w_raw.sum()
            if w_sum == 0.0:
                w = np.full_like(w_raw, 1.0 / w_raw.size)
            else:
                w = w_raw / w_sum

            # ---- gradient w.r.t. a (smooth surrogate) ----
            grad_a = _gradient_correlation(a, w)       # shape (n,)

            # ---- transform gradient to logits (mirror‑descent on simplex) ----
            dot = np.dot(a, grad_a)                     # scalar ⟨a,∇_a⟩
            grad_logit = a * (grad_a - dot)            # ∇_ℓ

            # ---- Adam update on logits ----
            m = beta1 * m + (1.0 - beta1) * grad_logit
            v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)

            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)

            lr = lr0 / (1.0 + t_iter * 1e-5)           # very mild decay
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- occasional tiny random perturbation (helps escape flat regions) ----
            if t_iter % 1000 == 0:
                logits += rng.normal(0.0, 5e-5, size=logits.shape)

    # --------------------------------------------------------------
    # 6) Simulated‑annealing fine‑tuning (remaining time)
    # --------------------------------------------------------------
    sa_deadline = start_time + float(budget_s)
    if best_vec is None:
        # fallback – should never happen, but guard against it
        best_vec = np.full(n, 1.0 / n)
        best_score = _fast_score(best_vec, n, fft_len)

    cur_sa = best_vec.copy()
    cur_score = best_score

    # SA schedule: temperature drops from ~5·10⁻³ to ~1·10⁻⁶ (geometric)
    T_start, T_end = 5e-3, 1e-6

    iteration = 0
    while time.monotonic() < sa_deadline:
        # ---- temperature (linear in remaining fraction) ----
        now = time.monotonic()
        frac = (now - main_deadline) / (sa_deadline - main_deadline) if sa_deadline > main_deadline else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        # ---- propose a random mass‑transfer move ----
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_sa[i] < 1e-15:
            continue
        delta = rng.random() * cur_sa[i] * 0.5   # move at most half of a[i]
        trial = cur_sa.copy()
        trial[i] -= delta
        trial[j] += delta

        trial_score = _fast_score(trial, n, fft_len)

        # ---- Metropolis acceptance ----
        improve = trial_score < cur_score
        if improve or rng.random() < np.exp(-(trial_score - cur_score) / max(T, 1e-12)):
            cur_sa = trial
            cur_score = trial_score
            if trial_score < best_score:
                best_score = trial_score
                best_vec = trial.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        iteration += 1

        # ---- gentle smoothing every few thousand steps (helps flatten tiny spikes) ----
        if iteration % 5000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = _normalise(smoothed)
            cur_sa = smoothed
            cur_score = _fast_score(cur_sa, n, fft_len)

    # --------------------------------------------------------------
    # 7) Return the best sequence (clip to the limits enforced by evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]
