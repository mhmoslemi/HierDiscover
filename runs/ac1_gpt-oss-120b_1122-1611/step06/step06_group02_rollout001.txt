"""
Propose a low‑max‑self‑convolution step function.

Algorithm overview
------------------
* Work on the probability simplex (∑a = 1, a_i ≥ 0).  The evaluation        *
  function `evaluate_sequence` is scale‑invariant, therefore we only need *
  to minimise the maximum entry of the self‑convolution `conv(a,a)`.      *
* A fast FFT based convolution is used throughout.
* Local optimisation is performed with an Adam optimiser on the            *
  *logits* `z` (softmax parametrisation).  The objective is a           *
  smoothed max‑convolution (`log‑sum‑exp`) whose gradient is obtained by *
  a second convolution, both O(n log n).                                  *
* The temperature β of the smooth‑max is increased gradually, so that the *
  optimisation smoothly switches from a global guide to an exact sub‑   *
  gradient on the worst index.
* Several families of initial vectors are tried (Dirichlet, sinusoidal‑   *
  power, interpolation from the current best).  Restarts are performed   *
  until the time budget (≈1000 s) is exhausted.
* The best vector found is returned as a plain Python list.               *
"""

import time
import numpy as np
from math import sqrt

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a (length 2·len(a)‑1) using a real FFT.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``x`` and ``y`` using a real FFT.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. ``a``.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max
        f_β(a) = (1/β)·log Σ_i exp(β·conv_i)
    w.r.t. ``a`` (no extra “+a” term).  Returns a vector of length len(a).
    """
    cur_max = conv.max()
    # the exponent is zero at the maximum -> numerically safe
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    # correlation of w with reversed a (gives the derivative)
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    # the central n entries correspond to the derivative w.r.t each a_i
    n = a.size
    grad_part = conv_w_rev[n - 1: n - 1 + n] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Simple random moves (used as occasional escapes)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out = np.full_like(out, 1.0 / n)
    else:
        out /= out_sum
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a small random fraction of mass from i to j (accept if better)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2_sum = a2.sum()
    if a2_sum == 0.0:
        a2 = np.full_like(a2, 1.0 / n)
    else:
        a2 /= a2_sum
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a


# ----------------------------------------------------------------------
# Fast score (does not invoke the heavy evaluate_sequence)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponential variables."""
    a = rng.exponential(scale=1.0, size=n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Shape a_i ∝ sin^p( π·(i+0.5)/n ), with p sampled uniformly in [0.5, 4].
    """
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linear interpolation of an existing simplex vector to a new length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Adam‑based local optimiser on the simplex (log‑softmax parametrisation)
# ----------------------------------------------------------------------


def _adam_optimize(initial: np.ndarray,
                   rng: np.random.Generator,
                   deadline: float,
                   max_iters: int = 2000,
                   prob_soft: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Optimise a simplex vector using Adam on logits with a smooth‑max
    surrogate for the max‑self‑convolution.  Returns the best vector seen
    and its (exact) score.
    """
    a = initial.astype(np.float64, copy=False)
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # logits initialisation (stable log‑softmax)
    eps = 1e-12
    z = np.log(a + eps)

    # Adam moments
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.12

    # smooth‑max temperature schedule
    beta = 0.5
    beta_mul = 1.025

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)
    no_improve = 0

    for t in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        # softmax of logits → current simplex vector
        max_z = np.max(z)
        exp_z = np.exp(z - max_z)      # stability
        a = exp_z / exp_z.sum()

        # convolution and exact score
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # ------------------------------------------------------------------
        # Gradient (soft‑max or exact sub‑gradient)
        # ------------------------------------------------------------------
        if rng.random() < prob_soft:
            # smooth max gradient
            grad = _softmax_gradient(a, conv, beta, fft_len_grad)
            # increase temperature (slowly)
            beta = min(beta * beta_mul, 1e6)
        else:
            # exact sub‑gradient at the worst index
            grad = _subgradient_step(a, conv)

        # chain rule: dL/dz = a * (grad - <a,grad>)
        avg = np.dot(a, grad)
        grad_z = a * (grad - avg)

        # Adam update (learning rate decays like 1/√t)
        lr = lr0 / np.sqrt(t)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # Apply the Adam step in log‑space
        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ------------------------------------------------------------------
        # Occasional random exploratory moves to escape plateaus
        # ------------------------------------------------------------------
        if no_improve >= 500 and rng.random() < 0.4:
            # apply a perturbation and reset the momentum
            a = _random_block_perturb(a, rng)
            z = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            no_improve = 0

        if rng.random() < 0.03:
            a = _pairwise_move(a, rng, fft_len_conv)
            z = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)

    # final evaluation of the best‑seen candidate
    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score

                print(f"[{time.time() - start:5.2f}s] Seeded from previous best: {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths we will explore – a mix of modest and large vectors
    # ------------------------------------------------------------------
    lengths = [500, 800, 1200, 1800, 2500, 3500, 4500, 5500, 7500, 10000]

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (biased slightly towards the larger end)
        n = int(rng.choice(lengths, p=np.ones(len(lengths)) / len(lengths)))

        # pick an initialiser
        which = rng.integers(0, 3)
        if which == 0:
            a0 = _init_random_dirichlet(rng, n)
        elif which == 1:
            a0 = _init_sine_power(rng, n)
        else:
            if best_vec is not None:
                a0 = _init_interpolate(rng, best_vec, n)
            else:
                a0 = _init_random_dirichlet(rng, n)

        # Allocate iteration budget – more iterations for smaller n
        iters = int(2500 * (5000.0 / max(n, 500)))
        a_opt, score_opt = _adam_optimize(a0, rng, deadline, max_iters=iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:5.2f}s] New global best: {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Polishing phase – a longer dedicated run on the current best
        # ------------------------------------------------------------------
        if best_vec is not None and time.time() < deadline - 1.0:
            # give the polishing a generous iteration budget
            polish_iters = int(4000 * (5000.0 / max(best_vec.size, 500)))
            a_polished, score_polished = _adam_optimize(best_vec.copy(),
                                                        rng,
                                                        deadline,
                                                        max_iters=polish_iters,
                                                        prob_soft=0.95)
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:5.2f}s] Polished best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (mirrors evaluate_sequence's clipping)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the outer harness
    return [float(x) for x in best_vec]
