    # conv self:
    conv = np.fft.irfft(fa * fa, n=L)[:2*n-1]  # conv length
    cur_max = conv.max()

def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    n = int(kwargs.get('n', 3800))
    if n < 2: n = 2

    # compute FFT length for correlation (>=3n-2)
    L = 1 << ((3*n - 2 - 1).bit_length())
    # Precompute constant for conv length
    conv_len = 2*n - 1

    # Helper: compute conv and gradient given a and beta
    def compute(a, beta, max_active):
        # forward FFT of a
        fa = np.fft.rfft(a, n=L)
        conv_full = np.fft.irfft(fa * fa, n=L)
        conv = conv_full[:conv_len]
        cur_max = conv.max()
        # select active indices (largest values)
        if max_active < conv_len:
            idx = np.argpartition(conv, -max_active)[-max_active:]
        else:
            idx = np.arange(conv_len)
        vals = conv[idx]
        w_vals = np.exp(beta * (vals - cur_max))
        w_sum = w_vals.sum()
        if w_sum == 0:
            w_vals[:] = 1.0 / w_vals.size
        else:
            w_vals /= w_sum
        # build w vector
        w = np.zeros_like(conv)
        w[idx] = w_vals
        # compute correlation gradient via FFT
        fw = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(fa * np.conj(fw), n=L)
        start = conv_len - 1
        grad = 2.0 * corr_full[start:start + n]
        return conv, grad, cur_max

    # function to evaluate score from conv max
    def score_from_max(m):
        return 2.0 * n * m

    # generate candidate initial shapes
    candidates = []

    # analytic sqrt-diff shape
    i = np.arange(n, dtype=np.float64)
    a_root = np.sqrt(i+1) - np.sqrt(i)
    a_root /= a_root.sum()
    candidates.append(a_root)

    # exponential shape
    for c in np.linspace(0.0, 0.001, 5):  # small c
        a_exp = np.exp(-c * i)
        a_exp /= a_exp.sum()
        candidates.append(a_exp)

    # power-diff shapes
    for p in np.linspace(0.3, 0.8, 11):
        a_p = (i+1)**p - i**p
        a_p /= a_p.sum()
        candidates.append(a_p)

    # include global if exists
    try:
        if "height_sequence_1" in globals():
            cand_glob = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand_glob.ndim == 1 and cand_glob.size >= 2:
                cand_glob = np.clip(cand_glob, 0.0, None)
                s = cand_glob.sum()
                if s > 0:
                    cand_glob = cand_glob / s
                    candidates.append(cand_glob)
    except Exception:
        pass

    # evaluate candidates quickly using conv (no full run)
    best_score = np.inf
    best_a = None
    for a in candidates:
        # compute conv for candidate using FFT length conv_len only; use same compute function with beta placeholder (not needed).
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[:conv_len]
        cur_max = conv.max()
        cur_score = score_from_max(cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    print(f"Initial best score after exponent search: {best_score:.6f}")

    cur_a = best_a.copy()
    cur_score = best_score
    best = best_a.copy()
    best_score = best_score

    # gradient descent parameters
    max_active = 30
    beta_start = 5.0
    beta_end = 3000.0
    eta_start = 0.25
    eta_end = 0.001
    momentum = 0.9

    grad_prev = np.zeros_like(cur_a)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    iteration = 0
    next_print = start_time + 5.0

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start * (1.0 - frac) + eta_end * frac

        conv, grad, cur_max = compute(cur_a, beta, max_active)
        cur_score = score_from_max(cur_max)

        # record best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur_a.copy()
            print(f"[{elapsed:6.2f}s] new best: {best_score:.6f}")

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # update
        factor = np.exp(-eta * grad)
        # clamp factor to avoid overflow/underflow
        factor = np.clip(factor, 1e-12, 1e12)
        cur_a = cur_a * factor
        # optional jitter to escape plateaus
        if rng.random() < 0.001:
            jitter = np.exp(rng.normal(0.0, 0.0005, size=n))
            cur_a = cur_a * jitter

        cur_a = np.clip(cur_a, 0.0, None)
        s = cur_a.sum()
        if s == 0.0:
            cur_a = np.full_like(cur_a, 1.0 / n)
        else:
            cur_a /= s

        iteration += 1

        # occasional print
        if now >= next_print:
            print(f"[{elapsed:6.2f}s] iter {iteration}, cur_score={cur_score:.6f}, best={best_score:.6f}, beta={beta:.1f}, eta={eta:.5f}")
            next_print = now + 5.0

    # final clipping and return list
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

"""
Fast gradient‑descent optimizer for the step‑function problem.

The algorithm:
1. Build a few analytically motivated initial vectors (sqrt‑difference,
   power‑differences, exponential decay, and a previously stored global
   candidate if available) and keep the one with the smallest objective.
2. Run a long “soft‑max’’ gradient descent on the L∞‑objective:
   – each iteration computes the self‑convolution a*a by a single FFT;
   – the indices that are near the current maximum form an active set;
   – a soft‑max (temperature β) on these values yields a smooth weight
     vector w;
   – the gradient of the smoothed objective is the cross‑correlation
     a⋆w, obtained with a second FFT;
   – an exponentiated‑gradient update (with momentum) stays on the simplex.
3. β is increased from a small to a large value, η (step size) is
   decreased, and occasional tiny random jitter is added to escape flat
   plateaus.
4. The best vector seen during the run is returned (clipped to the limits
   imposed by ``evaluate_sequence``).

The whole routine respects the time budget (default 1000 s) and prints
periodic progress reports.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n_extended(n: int) -> int:
    """
    Length needed for a full correlation of a (len=n) with a vector of
    length 2*n‑1.  The required size is (n + (2*n‑1) - 1) = 3*n‑2.
    """
    needed = 3 * n - 2
    return _next_pow2(needed)


def _conv_self_fft(a: np.ndarray, L: int, n: int) -> np.ndarray:
    """
    Linear self‑convolution of ``a`` using FFT length ``L``.
    Returns the first 2*n‑1 entries (the valid linear convolution).
    """
    fa = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fa * fa, n=L)
    return conv_full[: 2 * n - 1]


def _conv_and_grad(
    a: np.ndarray,
    n: int,
    L: int,
    beta: float,
    max_active: int,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Simultaneously compute
        * the self‑convolution (length 2*n‑1),
        * the gradient of the β‑soft‑max of the convolution,
        * the current maximum of the convolution.
    The gradient is 2·(a ⋆ w) where w is the soft‑max weight vector.
    """
    # forward FFT of a (re‑used for both convolution and correlation)
    fa = np.fft.rfft(a, n=L)

    # ----- convolution a*a ------------------------------------------------
    conv_full = np.fft.irfft(fa * fa, n=L)
    conv = conv_full[: 2 * n - 1]
    cur_max = float(conv.max())

    # ----- active set & soft‑max weights ---------------------------------
    if max_active < conv.size:
        active = np.argpartition(conv, -max_active)[-max_active:]
    else:
        active = np.arange(conv.size)
    vals = conv[active]
    w_vals = np.exp(beta * (vals - cur_max))
    w_sum = w_vals.sum()
    if w_sum == 0.0:
        w_vals = np.ones_like(w_vals) / w_vals.size
    else:
        w_vals /= w_sum

    # ----- build full weight vector w (length 2*n‑1) ----------------------
    w = np.zeros_like(conv)
    w[active] = w_vals

    # ----- correlation a ⋆ w (gives gradient) ---------------------------
    fw = np.fft.rfft(w, n=L)
    corr_full = np.fft.irfft(fa * np.conj(fw), n=L)
    start = conv.size - 1                # index where shift = 0 appears
    grad = 2.0 * corr_full[start : start + n]

    return conv, grad, cur_max


def _score_from_max(n: int, max_val: float) -> float:
    """Exact objective value when the sequence is already normalised."""
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------
def _analytic_root(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _power_diff(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^p – i^p, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** p - i ** p
    a /= a.sum()
    return a


def _exponential_decay(n: int, c: float) -> np.ndarray:
    """a_i ∝ exp(-c·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-c * i)
    a /= a.sum()
    return a


def _maybe_use_global(n: int) -> np.ndarray | None:
    """Return a normalised copy of the global variable ``height_sequence_1`` if it exists."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    if cand.size != n:
                        # resize by simple interpolation if lengths differ
                        cand = np.interp(
                            np.linspace(0, cand.size - 1, n),
                            np.arange(cand.size),
                            cand,
                        )
                    cand /= cand.sum()
                    return cand
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    # FFT length that works for both convolution and correlation
    L = _fft_len_for_n_extended(n)

    # ------------------------------------------------------------------
    # Build a few deterministic start points and keep the best one
    # ------------------------------------------------------------------
    starts = []

    # 1) analytic √‑difference
    starts.append(_analytic_root(n))

    # 2) a few power‑difference shapes
    for p in np.linspace(0.30, 0.80, 11):
        starts.append(_power_diff(n, p))

    # 3) very mild exponential decays (c near 0 gives almost uniform)
    for c in np.linspace(0.0, 0.001, 5):
        starts.append(_exponential_decay(n, c))

    # 4) previously saved global construction, if any
    glob = _maybe_use_global(n)
    if glob is not None:
        starts.append(glob)

    # evaluate all starters (fast FFT based)
    best_score = np.inf
    best_a = None
    for a in starts:
        conv = _conv_self_fft(a, L, n)
        cur_max = float(conv.max())
        cur_score = _score_from_max(n, cur_max)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

    print(f"Initial best score after cheap search: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Gradient‑descent optimisation on the smoothed max objective
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = best_score
    best = cur_a.copy()
    best_score = cur_score

    # hyper‑parameters
    max_active = 30                      # size of the active set
    beta_start = 5.0
    beta_end = 3000.0                    # final temperature (very sharp soft‑max)
    eta_start = 0.25                     # initial step size
    eta_end = 0.001                      # final step size
    momentum = 0.9
    grad_prev = np.zeros_like(cur_a)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # a tiny safety margin

    iteration = 0
    next_print = start_time + 5.0

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total

        # linear schedules for β and η
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # ----- compute convolution, gradient and max value -----------------
        conv, grad, cur_max = _conv_and_grad(cur_a, n, L, beta, max_active)
        cur_score = _score_from_max(n, cur_max)

        # keep global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur_a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update (projection stays on the simplex)
        factor = np.exp(-eta * grad)
        # protect against overflow/underflow of the factor
        factor = np.clip(factor, 1e-12, 1e12)
        cur_a = cur_a * factor

        # occasional tiny random jitter to help escape flat regions
        if rng.random() < 0.001:
            jitter = np.exp(rng.normal(0.0, 5e-4, size=n))
            cur_a *= jitter

        # renormalise onto the probability simplex
        cur_a = np.clip(cur_a, 0.0, None)
        s = cur_a.sum()
        if s == 0.0:
            cur_a = np.full_like(cur_a, 1.0 / n)
        else:
            cur_a /= s

        iteration += 1

        # periodic progress output (every ~5 s)
        if now >= next_print:
            print(
                f"[{elapsed:6.2f}s] iter {iteration:6d}, cur={cur_score:.6f}, "
                f"best={best_score:.6f}, β={beta:.1f}, η={eta:.5f}"
            )
            next_print = now + 5.0

    # ------------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
