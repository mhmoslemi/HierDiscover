def project_onto_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all values zero or negative, set uniform distribution
        return np.ones(n) / n
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=42, budget_s=1000., n=400, ...):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 1.0

    best_seq = None
    best_score = float('inf')
    # maybe include height_sequence_1.
    ...
    restarts = 0
    while time.monotonic() < deadline:
        restarts += 1
        # random initialization
        a = rng.random(n)
        a = a / a.sum()
        # mixing with best
        if best_seq is not None and rng.random() < 0.3:
            a = (a + best_seq) / 2
            a = a / a.sum()
        # algorithm parameters
        max_iters = 2000
        for it in range(max_iters):
            cur_time = time.monotonic()
            if cur_time >= deadline:
                break
            conv = np.convolve(a, a)
            max_b = conv.max()
            score = 2 * n * max_b
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a.copy()
                elapsed = cur_time - start
                print(f"[{elapsed:.2f}s] new best {best_score:.6f} (restart {restarts})")
            # gradient
            k = np.argmax(conv)
            # compute gradient shift
            grad = np.zeros_like(a)
            # compute overlapping region indices
            # j indices such that i = k - j in [0,n-1] => j from max(0,k-(n-1)) to min(n-1, k)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2 * a[k - idx]  # from max_b gradient, factor 2
            # derivative of objective (2n * max_b) is scaled by 2n:
            grad_obj = grad * (2 * n)
            # step size schedule: schedule decreasing from eta0 to eta_end
            # used formula:
            eta0 = 0.1
            eta_end = 0.001
            eta = eta0 * (1 - it / max_iters) + eta_end * (it / max_iters)
            # update
            a_new = a - eta * grad_obj
            # project onto simplex
            a = project_onto_simplex(a_new)
            # optional jitter
            if rng.random() < 0.001:
                a += rng.normal(scale=0.01, size=n)
                a = project_onto_simplex(a)
        # end inner loop

    # final best
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]

"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Algorithm
---------
*   The objective is   2·n·max_k (a∗a)_k / (∑a)^2 .
    Since it is homogeneous we may normalise the vector to sum to 1.
*   For a fixed bound   M = max_k (a∗a)_k  the feasible set
        { b ≥ 0 | (a∗b)_k ≤ M for all k }
    is a convex polyhedron.  Solving the linear programme

        maximise   ∑ b_j
        subject to b_j ≥ 0,
                   (a∗b)_k ≤ M  (k = 0,…,2n‑2)

    gives a “larger” vector b that still respects the same
    convolution bound.  After renormalising b to sum‑one we blend
    the current iterate a with b :

        a ← (1‑t)·a + t·b ,    0 < t ≤ 1 .

    The new a has the same ℓ∞‑convolution bound (or a smaller one);
    by repeating the LP–blend step the bound M is monotonically
    reduced.  This is exactly the scheme described in the problem
    statement, but here it is implemented directly with SciPy’s
    high‑performance LP solver.
*   The whole procedure is embedded into a multi‑restart loop:
    many random starts are tried (or the previously found best
    construction is reused).  Each run performs a bounded number of
    LP‑updates; the best sequence over the whole budget is returned.
*   All heavy work (convolution, LP matrix assembly) is performed
    with NumPy/SciPy; the code respects the 1000 s time budget and
    prints progress stamps.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Convolution helpers ----------------------------------------------------
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length for a full linear self‑convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using FFT (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def conv_self(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper – decides whether to use FFT or direct np.convolve."""
    n = a.size
    # direct convolution is faster for n ≤ ~300
    if n <= 300:
        return np.convolve(a, a)
    else:
        return _conv_self(a, _fft_len_for_n(n))

# ----------------------------------------------------------------------
# Linear‑program sub‑routine --------------------------------------------
# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (size (2n‑1) × n) such that
        (A @ b)[k] = (f ∗ b)[k]   for all k.
    Row `k` contains the vector f shifted by `k` positions.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=f.dtype)
    # each column j receives a copy of f starting at row j
    for j in range(n):
        A[j : j + n, j] = f
    return A

def _lp_max_sum_under_bound(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max ∑ b_j   subject to  b ≥ 0  and  (f∗b)_k ≤ M.
    Returns the optimal b (as a NumPy array) or None if the LP fails.
    """
    n = f.size
    A = _build_conv_matrix(f)        # (2n‑1) × n
    b_ub = np.full(A.shape[0], M)    # right‑hand side of the inequality
    c = -np.ones(n)                  # maximise ∑b  ⇔ minimise –∑b
    # SciPy’s ‘highs’ method works well on dense problems of this size
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
    if not res.success:
        return None
    return res.x

# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex ---------------
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x | x_i ≥ 0, ∑x_i = 1}.
    Implementation follows Duchi et al. (2008).
    """
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Returns a list of non‑negative heights that (approximately) minimise
    evaluate_sequence() within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size (the algorithm works fairly well for n ≈ 200–400)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 300))
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # keep the best construction seen so far
    # ------------------------------------------------------------------
    best_a = None
    best_score = float("inf")
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # safety margin

    # ------------------------------------------------------------------
    # optionally start from a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s                         # normalise to sum = 1
                    # evaluate the candidate (fast without the costly security checks)
                    cur_score = 2.0 * cand.size * np.max(conv_self(cand))
                    if cur_score < best_score:
                        best_score = cur_score
                        best_a = cand.copy()
        except Exception:
            pass

    # ------------------------------------------------------------------
    # parameters for the inner LP‑based optimisation loop
    # ------------------------------------------------------------------
    max_outer_iters = 30               # how many LP‑blend steps per restart
    t_blend = 0.2                      # initial blending factor
    min_t = 1e-4                       # stop when the blend factor gets too tiny
    # ------------------------------------------------------------------
    restart_counter = 0
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        restart_counter += 1

        # ----- initialise a fresh starting point -----------------------
        if best_a is not None and rng.random() < 0.25:
            # warm‑start from the current best but add a tiny perturbation
            a = best_a + rng.normal(scale=0.02, size=n)
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
        else:
            # completely random start (uniform + jitter)
            a = rng.random(n)
            a = a / a.sum()

        # ----- inner optimisation loop (LP + blending) -----------------
        for outer in range(max_outer_iters):
            now = time.monotonic()
            if now >= deadline:
                break

            # current convolution bound M
            conv = conv_self(a)
            M = conv.max()
            cur_score = 2.0 * n * M
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f} "
                      f"(restart {restart_counter}, outer {outer})")

            # solve the LP: maximise sum(b) s.t. (a ∗ b) ≤ M
            b = _lp_max_sum_under_bound(a, M)
            if b is None:
                # LP failed – give up this restart
                break

            sum_b = b.sum()
            if sum_b <= 0.0:
                break

            # normalise the LP solution to a probability vector
            g = b / sum_b

            # blend the current iterate with g
            a_new = (1.0 - t_blend) * a + t_blend * g

            # re‑evaluate the bound after blending
            new_M = conv_self(a_new).max()
            if new_M < M - 1e-12:
                # successful improvement – keep the blended vector
                a = a_new
                # (optional) slowly increase the step size to accelerate later stages
                t_blend = min(1.0, t_blend * 1.05)
            else:
                # no improvement – shrink the blending factor
                t_blend *= 0.5
                if t_blend < min_t:
                    # the blend is too small, abort this restart
                    break

        # ----------------------------------------------------------------
        # occasional global jitter to escape stagnation
        # ----------------------------------------------------------------
        if time.monotonic() < deadline and rng.random() < 0.05:
            if best_a is not None:
                a = best_a + rng.normal(scale=0.03, size=n)
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
    # --------------------------------------------------------------------
    # Return the best sequence discovered (evaluation routine will clip)
    # --------------------------------------------------------------------
    if best_a is None:
        # fall‑back – uniform distribution (guaranteed to be feasible)
        best_a = np.full(n, 1.0 / n)

    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
