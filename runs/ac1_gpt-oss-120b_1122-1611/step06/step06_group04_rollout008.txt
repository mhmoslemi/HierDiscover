Now define refine:

```
def refine(a, max_time):
    if max_time <= 0:
        return a
    start = time.monotonic()
    n = len(a)
    fft_len = _fft_len_for_n(n)
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # hyperparameters
    lr_start = 0.3
    lr_end = 0.001
    beta_start = 0.5
    beta_end = 2500.0
    lambda_var = 0.02  # weight for variance term

    while time.monotonic() - start < max_time:
        elapsed = time.monotonic() - start
        total = max_time
        ratio = elapsed / total
        lr = lr_start * (1 - ratio) + lr_end * ratio
        beta = beta_start * (1 - ratio) + beta_end * ratio

        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        # softmax weights
        diff = conv - cur_max
        w_soft = np.exp(beta * diff)
        w_sum = w_soft.sum()
        if w_sum == 0.0:
            mask = conv == cur_max
            w_soft = mask.astype(float)
            w_sum = w_soft.sum()
        w_soft = w_soft / w_sum

        # variance term weight
        avg = conv.mean()
        w_var = (conv - avg) * lambda_var
        # combine
        w = w_soft + w_var

        grad = _gradient_correlation(a, w)

        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = a - step
        a = _proj_simplex(a)

        # occasional random jitter to escape
        if rng.random() < 0.01:
            factor = np.exp(rng.normal(0.0, 0.1, size=n))
            a = a * factor
            a = _proj_simplex(a)

    return a

def refine_sequence(a, time_limit, rng):
    if time_limit <= 0:
        return a
    n = a.size
    fft_len = _fft_len_for_n(n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    # hyperparameters
    lr_start = 0.25
    lr_end = 0.001
    beta_start = 0.5
    beta_end = 2500.0
    lambda_var = 0.01
    # schedule
    start = time.monotonic()
    while True:
        now = time.monotonic()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        ratio = elapsed / time_limit
        lr = lr_start * (1 - ratio) + lr_end * ratio
        beta = beta_start * (1 - ratio) + beta_end * ratio

        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        # softmax weights
        diff = conv - cur_max  # <=0
        w_soft = np.exp(beta * diff)
        w_sum = w_soft.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback to uniform weight on max indices
            mask = conv == cur_max
            w_soft = mask.astype(float)
            w_sum = w_soft.sum()
        w_soft = w_soft / w_sum

        # variance term
        avg = conv.mean()
        w_var = (conv - avg) * lambda_var

        w = w_soft + w_var

        grad = _gradient_correlation(a, w)

        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1 - 0.9**t)
        v_hat = v / (1 - 0.999**t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = a - step

        a = _proj_simplex(a)

        # occasional jitter to avoid stagnation
        if rng.random() < 0.01:
            factor = np.exp(rng.normal(0.0, 0.1, size=n))
            a = a * factor
            a = _proj_simplex(a)

    return a

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm summary
-----------------
* Generate many candidate shapes parameterised by a few continuous variables:
    - Power‑law:      a_i ∝ (i + α)^{-p}
    - Sqrt‑shift:     a_i ∝ √(i + c) – √i
* Normalise each candidate to the simplex (∑a_i = 1) and evaluate the
  cheap surrogate score  S(a) = 2·n·max_k (a∗a)_k .
* If a candidate looks promising, spend a fraction of the remaining budget
  refining it with a projected‑Adam optimiser.
  The optimiser minimises a *smooth* surrogate of the L∞‑objective:
      f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)          (soft‑max of the convolution)
  plus a small variance‑penalty  λ·‖a∗a – mean‖² to keep the whole
  convolution flat.
  The gradient of the surrogate is obtained analytically via
  correlation (⋆) and the usual Adam update is performed.
* Occasionally jitter the best solution (log‑normal multiplicative noise)
  and run a short refinement, to escape possible shallow minima.
* All operations respect the simplex constraints (non‑negative, unit sum)
  via a Euclidean projection.
* The routine stops when the time budget (≈1000 s) is exhausted and returns
  the best vector found, clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length for a linear convolution of size n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for use with FFT."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return linear convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of a weighted correlation: ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w .
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of Σ_k w_k (a∗a)_k.
    ``a`` is length n, ``w`` is length 2·n‑1.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # Length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reversed a for correlation
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto {x ≥ 0, Σ x = 1}."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero -> uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Fast evaluation of the objective used by the optimiser (no clipping).
# ----------------------------------------------------------------------
def compute_score(a: np.ndarray) -> float:
    n = a.size
    conv = _conv_self(a, _fft_len_for_n(n))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Parameterised shape generators.
# ----------------------------------------------------------------------
def generate_power_law(n: int, p: float, alpha: float) -> np.ndarray:
    """a_i ∝ (i + α)^{-p}, normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    a = (idx + alpha) ** (-p)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def generate_sqrt_shift(n: int, c: float) -> np.ndarray:
    """a_i ∝ √(i + c) – √i, normalised to sum 1."""
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + c) - np.sqrt(idx)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


# ----------------------------------------------------------------------
# Refinement by projected Adam on a smooth surrogate (soft‑max + variance).
# ----------------------------------------------------------------------
def refine_sequence(a0: np.ndarray, time_limit: float,
                    rng: np.random.Generator) -> np.ndarray:
    """
    Refine a candidate vector using Adam on a surrogate of the max‑convolution.
    The surrogate is  soft‑max(β·conv)  +  λ·‖conv – mean‖² .
    """
    if time_limit <= 0.0:
        return a0
    a = a0.copy()
    n = a.size
    fft_len = _fft_len_for_n(n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # Schedules (linear in the allotted refinement time)
    lr_start, lr_end = 0.25, 0.001
    beta_start, beta_end = 0.5, 5000.0   # sharper than the baseline
    lambda_start, lambda_end = 0.10, 0.0   # decay variance weight

    start = time.monotonic()
    while True:
        now = time.monotonic()
        elapsed = now - start
        if elapsed >= time_limit:
            break
        ratio = elapsed / time_limit

        # learning rate & temperature
        lr = lr_start * (1.0 - ratio) + lr_end * ratio
        beta = beta_start * (1.0 - ratio) + beta_end * ratio
        lam = lambda_start * (1.0 - ratio) + lambda_end * ratio

        # convolution and soft‑max weights
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        diff = conv - cur_max                         # ≤ 0
        w_soft = np.exp(beta * diff)
        w_sum = w_soft.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            mask = conv == cur_max
            w_soft = mask.astype(float)
            w_sum = w_soft.sum()
        w_soft /= w_sum

        # variance‑penalty weights (can be negative)
        avg = conv.mean()
        w_var = (conv - avg) * lam

        # total weight vector
        w = w_soft + w_var

        # gradient
        grad = _gradient_correlation(a, w)

        # Adam update
        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = a - step
        a = _proj_simplex(a)

        # occasional jitter to avoid stagnation
        if rng.random() < 0.01:
            factor = np.exp(rng.normal(0.0, 0.1, size=n))
            a = a * factor
            a = _proj_simplex(a)

    return a

    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Initialise best solution (use previous global if available).
    # ------------------------------------------------------------------
    best = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                    sc = compute_score(cand)
                    best = cand.copy()
                    best_score = sc
                    print(f"Initial best taken from height_sequence_1: score={best_score:.6f}")
        except Exception:
            pass

    # fallback initial shape (sqrt‑difference)
    if best is None:
        n0 = int(kwargs.get("n", 3800))
        idx = np.arange(n0, dtype=np.float64)
        a0 = np.sqrt(idx + 1.0) - np.sqrt(idx)
        a0 = a0 / a0.sum()
        best = a0.copy()
        best_score = compute_score(best)
        print(f"Initial sqrt‑diff shape: n={n0}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Random‑search parameters.
    # ------------------------------------------------------------------
    n_options = [1800, 2400, 3000, 3800, 5000, 6400, 8000]
    p_bounds = (0.30, 0.80)         # for power‑law exponent
    alpha_bounds = (0.05, 10.0)     # shift for power‑law
    c_bounds = (0.05, 5.0)          # c for sqrt‑shift

    # ------------------------------------------------------------------
    # Main loop: generate candidates, keep the promising ones, refine.
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 1.0:
            break

        # Randomly pick a length and a shape type.
        n = int(rng.choice(n_options))
        shape_type = rng.integers(0, 2)          # 0 → power‑law, 1 → sqrt‑shift

        if shape_type == 0:
            p = rng.uniform(*p_bounds)
            alpha = rng.uniform(*alpha_bounds)
            a = generate_power_law(n, p, alpha)
        else:
            c = rng.uniform(*c_bounds)
            a = generate_sqrt_shift(n, c)

        cur_score = compute_score(a)

        # If the candidate looks *significantly* better, spend time refining it.
        # The factor 0.995 corresponds to a 0.5 % improvement over the current best.
        if cur_score < best_score * 0.995:
            # allocate at most 30 s or the remaining time minus a tiny safety buffer.
            time_for_refine = min(30.0, max(0.0, remaining - 5.0))
            a_refined = refine_sequence(a, time_for_refine, rng)
            refined_score = compute_score(a_refined)
            if refined_score < best_score:
                best = a_refined.copy()
                best_score = refined_score
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best after refine: n={n}, score={best_score:.6f}")

        # Occasionally try a small jitter of the *current* best solution.
        if rng.random() < 0.001:
            jitter = np.exp(rng.normal(0.0, 0.08, size=best.size))
            a_try = best * jitter
            a_try = _proj_simplex(a_try)
            # a brief refinement (max 5 s or whatever fits)
            time_for_refine = min(5.0, max(0.0, remaining - 5.0))
            a_refined = refine_sequence(a_try, time_for_refine, rng)
            refined_score = compute_score(a_refined)
            if refined_score < best_score:
                best = a_refined.copy()
                best_score = refined_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] improved by jitter: score={best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Final post‑processing: clip to the limits of evaluate_sequence and
    # return as plain Python list.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    result = [float(x) for x in best]

    # One final call to evaluate_sequence gives the official score (and
    # guarantees that we haven't produced an invalid vector).
    final_score = evaluate_sequence(result)   # type: ignore  # assumed to be imported
    total_elapsed = time.monotonic() - start_time
    print(f"Finished after {total_elapsed:.2f}s – best reported "
          f"score = {final_score:.6f} (internal approx {best_score:.6f})")
    return result
