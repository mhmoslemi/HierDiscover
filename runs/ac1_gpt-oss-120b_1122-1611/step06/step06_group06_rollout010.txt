def propose_candidate(seed=0, budget_s=1000.0, **kwargs):
   rng = np.random.default_rng(seed)
   n = int(kwargs.get("n", 3800))
   n = max(2,n)
   fft_len = _fft_len_for_n(n)

   # helper _normalise

   # candidate generation:
        pool = []

        # sqrt diff
        idx = np.arange(n, dtype=np.float64)
        a_sqrt = np.sqrt(idx+1)-np.sqrt(idx)
        pool.append(_normalise(a_sqrt))

        # previous best
        if 'height_sequence_1' in globals():
            try:
                prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
                if prev.ndim==1 and prev.size>0:
                    pool.append(_normalise(prev))
            except:
                pass

        # power diff families
        for p in np.linspace(0.2, 2.0, 15):
            a = (np.arange(n, dtype=np.float64)+1)**(-p) - (np.arange(n, dtype=np.float64)+2)**(-p)
            pool.append(_normalise(a))

        # exponential families
        for lam in np.linspace(0.0005, 0.04, 15):
            a = (1.0 - np.exp(-lam))* np.exp(-lam * np.arange(n, dtype=np.float64))
            pool.append(_normalise(a))

        # beta family
        for alpha in np.linspace(0.2, 2.5, 15):
            x = (np.arange(n, dtype=np.float64)+0.5)/n
            a = np.power(x, alpha-1) * np.power(1-x, alpha-1)
            pool.append(_normalise(a))

        # random dirichlet
        for _ in range(40):
            alpha = rng.uniform(0.1, 2.0)
            a = rng.gamma(alpha, 1.0, size=n)
            pool.append(_normalise(a))

   # evaluate each candidate using _score_of
   scores = [_score_of(v, n, fft_len) for v in pool]
   # select top K candidates
   K = 5
   # sort indices
   top_idx = np.argsort(scores)[:K]  # lower scores better
   top_candidates = [pool[i] for i in top_idx]
   best_score = float('inf')
   best = None

   # run mirror descent for each candidate
   start_time = time.monotonic()
   total_deadline = start_time + float(budget_s)
   safety = 5.0
   per_candidate_time = (budget_s - safety) / max(1, len(top_candidates))
   for run_i, init_vec in enumerate(top_candidates):
        run_start = time.monotonic()
        run_deadline = min(start_time + (run_i+1)*per_candidate_time, total_deadline - safety) # ensure not exceed
        cur = init_vec.copy()
        grad_prev = np.zeros_like(cur)
        momentum = 0.8

        # randomize schedule per run
        eta_start = rng.uniform(1.2, 2.0)
        eta_end = rng.uniform(1e-4, 1e-2)
        beta_start = rng.uniform(0.1, 0.5)
        beta_end = rng.uniform(4000, 8000)
        rel_tol = rng.uniform(0.994, 0.9995)  # 0.5% to 0.001% tolerance
        max_active = rng.integers(10, 25) # 10-24

        prob_block = 0.04
        prob_pair = 0.1

        while time.monotonic() < run_deadline:
            conv = _conv_self(cur, fft_len)
            cur_max = float(conv.max())
            cur_score = 2.0 * n * cur_max

            # update global best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = cur.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] new global best = {best_score:.9f}")

            # schedule fraction within this run
            now = time.monotonic()
            elapsed = now - run_start
            total_run = run_deadline - run_start
            frac = elapsed / total_run if total_run > 0 else 0.0
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # active set detection
            threshold = cur_max * rel_tol
            active = np.where(conv >= threshold)[0]
            if active.size > max_active:
                # keep largest values only
                idx_top = np.argpartition(conv, -max_active)[-max_active:]
                active = idx_top

            # compute gradient via softmax over active set
            if active.size > 0:
                # compute softmax weights
                w_raw = np.exp(beta * (conv[active] - cur_max))
                w_sum = w_raw.sum()
                w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0/ w_raw.size)
                # build weight vector of length 2n-1 (zero except active)
                w_vec = np.zeros(2*n-1, dtype=np.float64)
                w_vec[active] = w
                grad = _gradient_correlation(cur, w_vec)
            else:
                # fall back to using full softmax over all entries (computationally heavier)
                w_raw = np.exp(beta * (conv - cur_max))
                w = w_raw / w_raw.sum()
                grad = _gradient_correlation(cur, w)

            # momentum smoothing
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # exponentiated gradient update
            factor = np.exp(-eta * grad)
            cur = cur * factor
            # enforce non-neg and renormalize
            cur = np.clip(cur, 0.0, None)
            s = cur.sum()
            if s > 0:
                cur /= s
            else:
                cur = np.full_like(cur, 1.0 / n)

            # occasional random block
            if rng.random() < prob_block:
                width = rng.integers(1, max(2, n // 20))
                start = rng.integers(0, n - width + 1)
                blk_factor = np.exp(rng.normal(0.0, 0.18))
                cur[start:start+width] *= blk_factor
                cur = np.clip(cur, 0.0, None)
                cur /= cur.sum()

            # occasional pairwise mass transfer
            if rng.random() < prob_pair:
                i, j = rng.integers(0, n, size=2)
                if i != j and cur[i] > 1e-14:
                    delta = rng.random() * cur[i] * 0.5
                    cur[i] -= delta
                    cur[j] += delta
                    cur = np.clip(cur, 0.0, None)
                    cur /= cur.sum()

   # after all runs, SA for final improvement
   sa_deadline = total_deadline
   cur_sa = best.copy()
   cur_sa_score = best_score
   # cooling schedule:
   T_start = 1e-2
   T_end = 1e-7
   it = 0
   while time.monotonic() < sa_deadline:
        now = time.monotonic()
        frac = (now - (total_deadline - safety)) / safety if safety > 0 else 0.0
        T = T_start * (1.0 - frac) + T_end * frac
        # propose pairwise move
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_sa[i] < 1e-14:
            continue
        delta = rng.random() * cur_sa[i] * 0.5
        new_a = cur_sa.copy()
        new_a[i] -= delta
        new_a[j] += delta
        new_score = _score_of(new_a, n, fft_len)
        # accept if better or prob
        if new_score < cur_sa_score or rng.random() < np.exp(-(new_score - cur_sa_score)/max(T, 1e-12)):
            cur_sa = new_a
            cur_sa_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = new_a.copy()
                print(f"[{now - start_time:.2f}s] SA improved to {best_score:.9f}")
        it += 1

        # occasional smoothing every 10000 iterations
        if it % 10000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            smoothed_score = _score_of(smoothed, n, fft_len)
            if smoothed_score < cur_sa_score:
                cur_sa = smoothed
                cur_sa_score = smoothed_score
                if smoothed_score < best_score:
                    best_score = smoothed_score
                    best = smoothed.copy()
                    print(f"[{time.monotonic() - start_time:.2f}s] SA smoothing improved to {best_score:.9f}")

   best = np.clip(best, 0.0, 1000.0)
   return [float(v) for v in best]

"""
propose_candidate
=================

Search for a non‑negative step‑function (a probability vector) that minimises
``evaluate_sequence``.  The objective after normalising the sum to 1 is

    score(a) = 2 * n * max (a * a)                     (self‑convolution)

which is exactly the value returned by ``evaluate_sequence`` for a vector with
sum(a)=1 and without clipping.  The algorithm combines several ideas:

* **Rich initial pool** – analytic √‑difference, power‑difference,
  exponential, symmetric Beta‑shaped families (including the arcsine /
  α‑parameterised shapes) and many random Dirichlet draws.  The best few
  candidates are kept as starting points.
* **Multiple mirrored‑descent runs** – each start runs an exponentiated‑gradient
  (mirror) descent on the *soft‑max* surrogate of the max‑convolution.
  Parameters (step‑size, temperature, active‑set tolerance, random‑move
  probabilities) are randomised per run to explore different regions.
* **Fast convolution & gradient** – all convolutions are performed with FFT;
  the gradient of the (soft‑max) surrogate is obtained by a single
  correlation FFT, i.e. ``_gradient_correlation``.
* **Light stochastic moves** – occasional block‑scale perturbations and
  pairwise mass transfers keep the iterates from stagnating.
* **Final simulated‑annealing polish** – the best vector found is refined
  with a short SA phase (pairwise mass moves with exponential cooling) and
  an occasional smoothing kernel.

The routine respects a user‑supplied time budget (default 1000 s), prints
progress stamps and finally returns the best sequence as a plain list of
floats (already respecting the 0 ≤ value ≤ 1000 limits of ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT, truncated to length 2·|a|‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of Σ_k w_k (a*a)_k = 2·a ⋆ w   (cross‑correlation).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of the weighted sum of convolution entries."""
    n = a.size
    L = w.size                 # = 2·n‑1
    # FFT length large enough for a_rev (size n) convolved with w (size L)
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Fast exact objective for a probability vector (sum = 1).
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """2·n·max(a*a) for a normalised non‑negative vector."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    n = max(2, n)
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Normalise helper (projects onto the probability simplex)
    # ------------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # ------------------------------------------------------------------
    # Build a diverse pool of starting vectors
    # ------------------------------------------------------------------
    pool = []

    # a) Analytic √‑difference
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    pool.append(_normalise(a_sqrt))

    # b) Previous construction, if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                pool.append(_normalise(prev))
        except Exception:
            pass

    # c) Power‑difference families  a_i = (i+1)^-p - (i+2)^-p
    for p in np.linspace(0.2, 2.0, 15):
        i = np.arange(n, dtype=np.float64)
        a = ( (i + 1.0) ** (-p) - (i + 2.0) ** (-p) )
        pool.append(_normalise(a))

    # d) Exponential families  a_i = (1‑e^{-λ})·e^{-λ i}
    for lam in np.linspace(0.0005, 0.04, 15):
        a = (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))
        pool.append(_normalise(a))

    # e) Symmetric Beta families  a_i ∝ x^{α‑1}(1‑x)^{α‑1}
    for alpha in np.linspace(0.2, 2.5, 15):
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        pool.append(_normalise(a))

    # f) Random Dirichlet samples
    for _ in range(40):
        alpha = rng.uniform(0.1, 2.0)
        a = rng.gamma(alpha, 1.0, size=n)
        pool.append(_normalise(a))

    # ------------------------------------------------------------------
    # Evaluate the pool quickly (FFT‑based exact score) and keep the best
    # ------------------------------------------------------------------
    scores = [_score_of(v, n, fft_len) for v in pool]
    # we will keep the top K distinct starts
    K = min(5, len(pool))
    top_idx = np.argsort(scores)[:K]
    starts = [pool[i].copy() for i in top_idx]

    best = None
    best_score = np.inf

    start_time = time.monotonic()
    total_deadline = start_time + float(budget_s)
    # reserve a few seconds for final SA stage
    safety = 5.0
    remaining_for_descent = max(0.0, budget_s - safety)
    per_run_budget = remaining_for_descent / max(1, len(starts))

    # ------------------------------------------------------------------
    # Mirror‑descent runs (one per selected start)
    # ------------------------------------------------------------------
    for run_id, init_vec in enumerate(starts):
        run_start = time.monotonic()
        run_deadline = run_start + per_run_budget
        cur = init_vec.copy()
        grad_prev = np.zeros_like(cur)

        # randomise a few hyper‑parameters per run
        eta_start = rng.uniform(1.2, 2.0)
        eta_end   = rng.uniform(1e-4, 1e-2)
        beta_start = rng.uniform(0.1, 0.5)
        beta_end   = rng.uniform(4000.0, 8000.0)
        rel_tol = rng.uniform(0.992, 0.9995)      # active‑set tolerance
        max_active = rng.integers(12, 30)        # cap on active indices
        prob_block = rng.uniform(0.025, 0.05)    # block‑scale perturb probability
        prob_pair  = rng.uniform(0.07, 0.12)     # pairwise mass‑transfer probability
        momentum = 0.85

        while time.monotonic() < run_deadline:
            # ----- self‑convolution, current score -----
            conv = _conv_self(cur, fft_len)
            cur_max = float(conv.max())
            cur_score = 2.0 * n * cur_max

            # ----- global best update -----
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = cur.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] new best = {best_score:.9f}")

            # ----- schedule (linear interpolation) -----
            now = time.monotonic()
            elapsed = now - run_start
            frac = elapsed / (run_deadline - run_start) if run_deadline > run_start else 0.0
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            # ----- active set (indices close to the current maximum) -----
            act = np.where(conv >= cur_max * rel_tol)[0]
            if act.size > max_active:
                # keep the largest entries only
                act = np.argpartition(conv, -max_active)[-max_active:]

            # ----- gradient of the soft‑max surrogate -----
            # build weight vector w of length 2·n‑1
            w_vec = np.zeros(2 * n - 1, dtype=np.float64)
            if act.size > 0:
                # softmax only over the active indices
                w_raw = np.exp(beta * (conv[act] - cur_max))
                w_sum = w_raw.sum()
                w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)
                w_vec[act] = w
            else:
                w_raw = np.exp(beta * (conv - cur_max))
                w = w_raw / w_raw.sum()
                w_vec[:] = w
            grad = _gradient_correlation(cur, w_vec)

            # ----- momentum smoothing -----
            grad = momentum * grad_prev + (1.0 - momentum) * grad
            grad_prev = grad

            # ----- exponentiated‑gradient (mirror) update -----
            factor = np.exp(-eta * grad)
            cur = cur * factor
            cur = np.clip(cur, 0.0, None)
            s = cur.sum()
            if s > 0.0:
                cur /= s
            else:
                cur = np.full_like(cur, 1.0 / n)

            # ----- occasional random block perturbation -----
            if rng.random() < prob_block:
                width = rng.integers(1, max(2, n // 20))
                start = rng.integers(0, n - width + 1)
                blk_factor = np.exp(rng.normal(0.0, 0.18))
                cur[start:start + width] *= blk_factor
                cur = np.clip(cur, 0.0, None)
                cur /= cur.sum()

            # ----- occasional pairwise mass transfer -----
            if rng.random() < prob_pair:
                i, j = rng.integers(0, n, size=2)
                if i != j and cur[i] > 1e-14:
                    delta = rng.random() * cur[i] * 0.5
                    cur[i] -= delta
                    cur[j] += delta
                    cur = np.clip(cur, 0.0, None)
                    cur /= cur.sum()

    # ------------------------------------------------------------------
    # Simulated‑annealing polishing (final ~safety seconds)
    # ------------------------------------------------------------------
    sa_deadline = total_deadline
    cur_sa = best.copy()
    cur_sa_score = best_score
    T_start = 5e-3
    T_end = 1e-7
    it = 0

    while time.monotonic() < sa_deadline:
        now = time.monotonic()
        # exponential cooling relative to the SA interval
        frac = (now - (total_deadline - safety)) / safety if safety > 0 else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        # propose a random pairwise transfer
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_sa[i] < 1e-14:
            continue
        delta = rng.random() * cur_sa[i] * 0.5
        new_a = cur_sa.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _score_of(new_a, n, fft_len)

        if new_score < cur_sa_score or rng.random() < np.exp(-(new_score - cur_sa_score) / max(T, 1e-12)):
            cur_sa = new_a
            cur_sa_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = new_a.copy()
                print(f"[{now - start_time:6.2f}s] SA improvement → {best_score:.9f}")

        it += 1

        # occasional smoothing (helps to flatten tiny peaks)
        if it % 5000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            smoothed_score = _score_of(smoothed, n, fft_len)
            if smoothed_score < cur_sa_score:
                cur_sa = smoothed
                cur_sa_score = smoothed_score
                if smoothed_score < best_score:
                    best_score = smoothed_score
                    best = smoothed.copy()
                    print(f"[{time.monotonic() - start_time:6.2f}s] SA smoothing improved to {best_score:.9f}")

    # ------------------------------------------------------------------
    # Return the best sequence (clip to the limits imposed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
