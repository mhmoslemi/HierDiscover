"""
Propose a step‐function that (approximately) minimizes
    evaluate_sequence(sequence) = 2·n·max(conv(a,a)) / (sum a)^2 .
The algorithm is a hybrid of three ideas:

1. **Good initial guess** – we scan two analytic one‑parameter families
   (power‑law differences and fractional‑power differences) and also
   reuse the best known construction (`height_sequence_1` if present).

2. **Simulated‑annealing on the exact objective** – a single move transfers a
   small amount of mass from one index `i` to another `j`.  The convolution
   is updated incrementally (O(n) per move) using the identity

        (a+Δ)*(a+Δ) = a*a + 2·a⋆Δ + Δ*Δ .

   Acceptance follows a temperature schedule; the temperature is lowered
   linearly to a very small value, allowing occasional uphill steps early
   on and forcing greedy descent later.

3. **Polishing phase** – after the SA loop we run a few exponentiated‑gradient
   (mirror‑descent) steps on the smooth `log‑sum‑exp` surrogate of the max,
   starting from the best solution found.  This often removes the last bits
   of excess peak.

The whole procedure respects the supplied time budget (≈1000 s) and returns
the best sequence discovered, already clipped to the limits used by
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a via FFT (real input)."""
    n = a.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    inds = np.arange(1, n + 1)
    cond = u - cssv / inds > 0
    if not np.any(cond):
        # all elements will be zero
        return np.zeros_like(v)
    rho = inds[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Analytic families (same as in the previous solution)
# ----------------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^r - i^r, 0<r<1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_shape_p(p: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^(-p) - (i+2)^(-p)."""
    k = np.arange(1, n + 2, dtype=np.float64)      # 1 .. n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s

    # 1) Analytic families – cheap global scan for a decent start

        cand = _gen_shape_p(p, n)
        if cand is not None:
            _evaluate_candidate(cand)

    if best_a is None:
        # fallback – uniform distribution (should never happen)
        best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 2) Simulated‑annealing with incremental convolution updates
    # ------------------------------------------------------------------
    # Normalise once for the SA loop.
    cur_a = best_a.copy()
    cur_a /= cur_a.sum()
    conv_len = 2 * n - 1
    # padded view: zeros before and after the true support
    pad_offset = n - 1
    a_pad = np.concatenate([np.zeros(pad_offset), cur_a, np.zeros(pad_offset)], dtype=np.float64)

    # exact convolution for the current state (used for scoring)
    cur_conv = _conv_fft(cur_a)
    cur_score = _score_from_conv(cur_conv)

    best_score = cur_score
    best_a = cur_a.copy()
    last_improve = start_time

    # SA hyper‑parameters
    T_start = 0.02                 # initial temperature
    T_end = 1e-6                   # final temperature
    max_move_frac = 0.45           # at hottest we may move up to 45 % of a[i]
    min_move_frac = 0.001          # near the end we move virtually nothing
    restart_gap = 180.0            # seconds without improvement → jitter restart
    recompute_interval = 5000      # full FFT recomputation to avoid drift

    iteration = 0
    while time.time() < deadline:
        now = time.time()
        elapsed = now - start_time
        total = deadline - start_time
        progress = elapsed / total

        # linear temperature schedule
        T = max(T_end, T_start * (1.0 - progress) + T_end * progress)

        # --------------------------------------------------------------
        # Propose a pairwise mass transfer (i -> j)
        # --------------------------------------------------------------
        i = rng.integers(0, n)
        # ensure we move some positive mass
        if cur_a[i] <= 1e-14:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # magnitude of the move – larger when temperature is high
        move_frac = max_move_frac * (T / T_start) + min_move_frac
        delta = cur_a[i] * move_frac * rng.random()
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # Incremental update of the convolution
        # ------------------------------------------------------------------
        # slices of the padded vector (views – no copy)
        slice_i = a_pad[i : i + conv_len]
        slice_j = a_pad[j : j + conv_len]

        delta_conv = 2.0 * delta * (slice_j - slice_i)    # shape (conv_len,)
        # apply the update
        cur_conv += delta_conv
        idx_2i = 2 * i
        idx_2j = 2 * j
        idx_ij = i + j
        cur_conv[idx_2i] += delta * delta
        cur_conv[idx_2j] += delta * delta
        cur_conv[idx_ij] -= 2.0 * delta * delta

        # update the probability vector and its padded representation
        cur_a[i] -= delta
        cur_a[j] += delta
        a_pad[pad_offset + i] = cur_a[i]
        a_pad[pad_offset + j] = cur_a[j]

        new_score = _score_from_conv(cur_conv)
        diff = new_score - cur_score

        # --------------------------------------------------------------
        # Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if diff < 0.0:
            accept = True
        else:
            # probabilistic uphill move
            if T > 0.0 and rng.random() < math.exp(-diff / T):
                accept = True

        if accept:
            cur_score = new_score
            # record best solution
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = cur_a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best (SA) score = {best_score:.6f}")
        else:
            # revert everything (inverse of the update above)
            cur_conv -= delta_conv
            cur_conv[idx_2i] -= delta * delta
            cur_conv[idx_2j] -= delta * delta
            cur_conv[idx_ij] += 2.0 * delta * delta
            cur_a[i] += delta
            cur_a[j] -= delta
            a_pad[pad_offset + i] = cur_a[i]
            a_pad[pad_offset + j] = cur_a[j]

        iteration += 1

        # ------------------------------------------------------------------
        # Periodic full recomputation (prevents accumulated rounding errors)
        # ------------------------------------------------------------------
        if iteration % recompute_interval == 0:
            cur_conv = _conv_fft(cur_a)

        # ------------------------------------------------------------------
        # Stagnation handling – jittered restart
        # ------------------------------------------------------------------
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            cur_a = best_a * jitter
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a /= cur_a.sum()
            a_pad[pad_offset:pad_offset + n] = cur_a
            cur_conv = _conv_fft(cur_a)
            cur_score = _score_from_conv(cur_conv)
            last_improve = now
            print(
                f"[{elapsed:6.2f}s] restart (jitter) – current best {best_score:.6f}"
            )

    # ----------------------------------------------------------------------
    # 3) Optional polishing with exponentiated‑gradient on a smooth surrogate
    # ----------------------------------------------------------------------
    # Use a short warm‑up of the soft‑max surrogate to squeeze the remaining
    # slack out of the solution.
    polish_deadline = deadline - 0.2
    cur_a = best_a.copy()
    cur_a = cur_a / cur_a.sum()
    fft_len = _next_pow2(2 * n - 1)
    beta_start, beta_end = 0.5, 8000.0
    eta_start, eta_end = 1.2, 0.02
    t_start = time.time()
    while time.time() < polish_deadline:
        # compute convolution via FFT (exact)
        fa = np.fft.rfft(cur_a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        # soft‑max over the convolution
        beta = beta_start + (beta_end - beta_start) * ((time.time() - t_start) /
                                                       (polish_deadline - t_start))
        mx = conv.max()
        exp_term = np.exp(beta * (conv - mx))
        w = exp_term / exp_term.sum()                  # soft‑max weights
        # gradient of the surrogate: 2·a ⋆ w   (correlation)
        # correlation via FFT: reverse a, convolve with w
        a_rev = cur_a[::-1]
        L = w.size + n - 1
        fft_len2 = 1 << ((L - 1).bit_length())
        fw = np.fft.rfft(w, n=fft_len2)
        fa_rev = np.fft.rfft(a_rev, n=fft_len2)
        corr = np.fft.irfft(fw * fa_rev, n=fft_len2)[: L]
        grad = 2.0 * corr[n - 1 : n - 1 + n]

        # mirror‑descent (exponentiated) update
        eta = eta_start * (1.0 - (time.time() - t_start) /
                           (polish_deadline - t_start)) + eta_end * (
            (time.time() - t_start) / (polish_deadline - t_start))
        factor = np.exp(-eta * grad)
        cur_a = cur_a * factor
        cur_a = np.clip(cur_a, 0.0, None)
        cur_a /= cur_a.sum()

        # keep the best
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            print(f"[polish] improved to {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best vector (already normalised, safe for evaluate_sequence)
    # ----------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
