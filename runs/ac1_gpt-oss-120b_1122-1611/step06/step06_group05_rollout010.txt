"""
propose_candidate
=================

Search for a low‑value step‑function using a tiny parametric family.
Two families are tried:

1. **Power‑law**    a_i(p) = (i+1)^(-p)                (p∈(0,1))
2. **Shifted √‑dif** a_i(α) = √(i+α+1) – √(i+α)        (α≥0)

Both families are normalised to unit ℓ₁‑norm (so that the denominator of
``evaluate_sequence`` equals 1).  For every candidate the convolution
 a∗a  is obtained with an FFT (O(n log n)).  The `score` we minimise is

        score = 2·n·max(a∗a)                      ( = evaluate_sequence )

A simple grid search over (p,α) is performed, followed by a light
hill‑climbing phase that moves a tiny amount of mass between two
random positions while keeping the sum unchanged.  The whole routine
stops as soon as the time budget (≈1000 s) is exhausted and returns the
best sequence found.

The method is deterministic apart from the controlled random jitter and
fits the required signature.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power‑of‑two length that can hold a full self‑convolution.
# ----------------------------------------------------------------------
def _fft_len(n: int) -> int:
    """Length for an FFT that can hold the convolution of two vectors of length n."""
    size = 2 * n - 1
    # smallest power of two ≥ size
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Compute the evaluation score (2·n·max(a∗a)) for a vector that already sums to 1.
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(conv) assuming sum(a)=1."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return 2.0 * a.size * float(conv.max())


# ----------------------------------------------------------------------
def _make_powerlaw(p: float, n: int) -> np.ndarray:
    """Unnormalised power‑law vector a_i = (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()                     # normalise to ℓ₁‑norm = 1
    return a


# ----------------------------------------------------------------------
def _make_sqrt_shift(alpha: float, n: int) -> np.ndarray:
    """Unnormalised √‑difference vector a_i = √(i+α+1) – √(i+α)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + alpha + 1.0) - np.sqrt(i + alpha)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The function performs a cheap parametric search
    over power‑law and shifted‑√ families, followed by a short hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size ----------------------------------------------------
    n = int(kwargs.get("n", 3000))
    if n < 2:
        n = 2
    fft_len = _fft_len(n)

    # ------------------------------------------------------------------
    # bookkeeping ------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)   # tiny safety margin
    best_a = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) systematic grid over the two families -------------------------
    # ------------------------------------------------------------------
    # power‑law grid (p from 0.45 to 0.58 – the region where theory
    # predicts the optimum lives)
    p_grid = np.linspace(0.45, 0.58, 28)          # 28 ≈ 1 s on a modern CPU
    # shifted‑√ grid (α from 0 to 200 in steps of 20)
    alpha_grid = np.arange(0.0, 201.0, 20.0)

    # ---- power‑law ----------------------------------------------------
    for p in p_grid:
        if time.monotonic() >= deadline:
            break
        a = _make_powerlaw(p, n)
        cur_score = _score_of(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.monotonic()-start_time:6.2f}s] "
                  f"new best (power p={p:.4f})  score={best_score:.6f}")

    # ---- shifted √ ----------------------------------------------------
    for alpha in alpha_grid:
        if time.monotonic() >= deadline:
            break
        a = _make_sqrt_shift(alpha, n)
        cur_score = _score_of(a, fft_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.monotonic()-start_time:6.2f}s] "
                  f"new best (sqrt α={alpha:.0f})  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) brief hill‑climbing starting from the best candidate found -----
    # ------------------------------------------------------------------
    # Work only if we still have a reasonable amount of time left.
    # The hill‑climb moves a tiny ε‑fraction of mass from one index to another.
    if best_a is not None:
        # pre‑compute the FFT of the current best vector
        a = best_a.copy()
        fa = np.fft.rfft(a, n=fft_len)

        # we keep the current best score as a reference
        cur_score = best_score

        # parameters
        max_iters = 5000                       # safety cap
        eps_start = 0.02                       # initial perturbation magnitude
        eps_end = 0.0005                       # final magnitude (annealed)
        it = 0
        while time.monotonic() < deadline and it < max_iters:
            it += 1
            # linearly anneal the step size
            frac = it / max_iters
            eps = eps_start * (1.0 - frac) + eps_end * frac

            # pick two distinct positions
            i, j = rng.choice(n, size=2, replace=False)

            # amount of mass to move (proportional to the source entry)
            delta = eps * a[i]

            # enforce positivity
            if delta > a[i]:
                delta = a[i]

            # propose new vector
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # recompute convolution via rank‑1 update (fast enough for these sizes)
            # we simply recompute the FFT – cheap for n~few thousand
            fa_new = np.fft.rfft(a_new, n=fft_len)
            conv = np.fft.irfft(fa_new * fa_new, n=fft_len)[: 2 * n - 1]
            new_score = 2.0 * n * float(conv.max())

            if new_score < cur_score - 1e-12:        # accept improvement
                a = a_new
                fa = fa_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
                    print(f"[{time.monotonic()-start_time:6.2f}s] "
                          f"hill‑climb iter {it:4d}  score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – in the extremely unlikely event that nothing was found
    # ------------------------------------------------------------------
    if best_a is None:
        # uniform distribution (score = 2) is always valid
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0

    # ------------------------------------------------------------------
    # Clip to the limits enforced by `evaluate_sequence` and return
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    # convert to plain Python list of floats (the evaluator expects a list)
    return [float(v) for v in best_a]
