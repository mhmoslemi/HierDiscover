            max_c = c.max()

            K = np.where(c >= threshold)[0]
            g = np.zeros_like(a)
            for k in K:
                lo = max(0, k - (n - 1))
                hi = min(k, n - 1)
                # a indices for part: a[lo:hi+1]
                g[lo:hi+1] += a[k - np.arange(lo, hi+1)]

"""
Propose a step‑function that lowers the value of
    score = 2·n·max(a*a) / (sum(a)**2)

The algorithm works in three layers:

1️⃣  **Rich initial portfolio** – several analytic shapes (uniform,
    exponential, linear, an arcsine‑type edge‑heavy shape and an
    inverse‑sqrt shape) together with a handful of Dirichlet draws.
    The best of them (according to the exact evaluation) is taken as
    the starting point.  If a previous solution lives in the global
    ``height_sequence_1`` variable it is also considered.

2️⃣  **Multi‑peak mirror descent** (≈20 % of the time budget).  
    The exact sub‑gradient of the objective ``max(conv)`` is not taken
    from a single index but from *all* convolution entries that are
    within a small tolerance of the current maximum.  The aggregated
    sub‑gradient is fed into an exponentiated‑gradient (entropy‑mirror)
    update with a linearly decreasing step‑size.

3️⃣  **Fast stochastic refinement** (the remaining ~80 %).  
    A donor index is sampled proportionally to its contribution to the
    current peak, a recipient is chosen favouring small masses and a
    random amount of mass is transferred.  The self‑convolution is
    updated incrementally (O(n)) and the move is accepted by a
    simulated‑annealing rule with a temperature that cools linearly to
    10⁻⁶.  Every few successful moves a full FFT convolution is
    recomputed to avoid drift; occasional block‑wise log‑normal jitter
    helps to escape shallow plateaux.  Stagnation for >35 s triggers a
    soft restart from the best solution found so far.

All steps respect the 5 s safety margin required by the harness.
The function finally returns the best sequence seen (clipped to the
range required by ``evaluate_sequence``).  The whole routine runs in
≤ 1000 s and needs only NumPy (no external solvers).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns the first ``2·len(a)-1`` entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` (donor) to ``j`` (recipient)
    and update the convolution ``c`` incrementally.
    Returns the new a and c.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # shift of donor i (subtract) and recipient j (add)
    c_new = c.copy()
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ part
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _candidate_arcsine(n: int) -> np.ndarray:
    """
    Edge‑heavy “arcsine” shape:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    """
    i = np.arange(n, dtype=np.float64)
    denom = np.sqrt((i + 0.5) * (n - i - 0.5))
    a = 1.0 / denom
    a /= a.sum()
    return a


def _candidate_inverse_sqrt(n: int) -> np.ndarray:
    """
    Symmetric inverse‑sqrt shape:
        a_i ∝ 1/√(i+1) + 1/√(n‑i)
    """
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0) + 1.0 / np.sqrt(n - i)
    a /= a.sum()
    return a


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A diversified list of start vectors."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) exponential decays
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # 3) linear decreasing
    v = (np.arange(n, 0, -1, dtype=np.float64))
    v /= v.sum()
    cand.append(v)

    # 4) arcsine‑type edge heavy
    cand.append(_candidate_arcsine(n))

    # 5) inverse‑sqrt edge heavy
    cand.append(_candidate_inverse_sqrt(n))

    # 6) a few Dirichlet draws (moderately concentrated)
    for _ in range(6):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand

    eta_start = 6.0

    temp_start = 0.12

    move_counter = 0
    RECOMPUTE_EVERY = 400   # full FFT recompute every 400 accepted moves

    # --------------------------------------------------------------

        # Phase 1 – multi‑peak mirror descent
        # ----------------------------------------------------------
        if now < phase1_end:
            # linear schedule for the step size
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # recompute full convolution after the update
            c = _self_conv(a, fft_len)
            cur_score = 2.0 * n * c.max()

        # ----------------------------------------------------------
        # Phase 2 – fast stochastic refinement (simulated annealing)
        # ----------------------------------------------------------
        else:
            # linear temperature cooling
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # current peak location
            k_max = int(c.argmax())

            # contribution of each index to the peak (used for donor choice)
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            if total_contrib > 0.0:
                donor = rng.choice(n, p=contrib / total_contrib)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                continue  # nothing to move from this donor

            # choose a recipient – bias toward low mass locations
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            wsum = recv_weights.sum()
            if wsum > 0.0:
                recv_weights /= wsum
                recipient = rng.choice(n, p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # decide how much mass to move (≤50 % of donor's mass)
            max_delta = 0.5 * a[donor]
            delta = rng.random() * max_delta

            # apply the move incrementally
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # periodic full recomputation to keep numerical noise in check
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len)
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter (≈3 % of iterations)
            if rng.random() < 0.03:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len)
                cur_score = 2.0 * n * c.max()

        # ----------------------------------------------------------
        # Keep the best solution seen so far
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Soft restart after prolonged stagnation
        # ----------------------------------------------------------
        if now - last_improve > 35.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.11, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
