"""
Optimization of a step-function sequence to minimize evaluate_sequence.
Uses an active‑set linear‑programming step inspired by the reference
algorithm, together with a simple Adam‑style gradient fallback.
The LP maximises the total mass while keeping the convolution of the
current sequence with the new candidate below the current maximum.
Repeated convex combinations shrink the L∞‑norm of the convolution.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(x: int) -> int:
    """Next power of two ≥ x (used for FFT length)."""
    return 1 << ((x - 1).bit_length())

def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for use with FFT."""
    size = 2 * n - 1
    return _next_pow2(size)

def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("input must be 1‑D")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w

def _build_active_constraints(f: np.ndarray, active_ks: np.ndarray) -> csr_matrix:
    """
    Build a sparse matrix A where each row corresponds to an active
    convolution index k and each column j contains the coefficient f[i]
    with i = k-j whenever 0 ≤ i, j < n.
    """
    n = f.size
    rows = active_ks.size
    data = []
    row_ind = []
    col_ind = []
    for r, k in enumerate(active_ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # i iterates over f indices that appear in the sum for this k
        for i in range(i_start, i_end + 1):
            j = k - i
            # coefficient f[i] for variable b_j
            data.append(f[i])
            row_ind.append(r)
            col_ind.append(j)
    return csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    The algorithm alternates a linear‑programming step (maximise mass under
    the current convolution bound) with a short gradient‑descent fallback.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Problem size – allow a modest length; larger sizes give
    #    diminishing returns for the LP approach.
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # 2) Initial sequence – analytic sqrt‑difference (good start)
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()                       # normalise to unit sum (probability simplex)

    # If a previously discovered construction exists, prefer it
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand /= s
                    # quick evaluation (same normalisation as a)
                    fft_len_cand = _fft_len_for_n(cand.size)
                    score_cand = 2.0 * cand.size * np.max(_conv_self_fft(cand, fft_len_cand))
                    score_cur = 2.0 * n * np.max(_conv_self_fft(a, _fft_len_for_n(n)))
                    if score_cand < score_cur:
                        a = cand
                        n = cand.size
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT length (used for all convolutions)
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 4) Optimisation state
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * np.max(_conv_self_fft(best, fft_len))
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin
    last_improve = start_time

    # Parameters for the outer loop
    max_no_improve = 120.0               # seconds without improvement before a random restart
    lp_t = 0.25                          # convex‑combination factor for the LP step
    grad_t = 0.05                        # step size for the gradient fallback
    tol_rel = 1e-6                       # relative tolerance for selecting the active set

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ------------------------------------------------------------------
        # 4a) Compute the current convolution and score
        # ------------------------------------------------------------------
        conv = _conv_self_fft(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ------------------------------------------------------------------
        # 4b) Keep the best solution seen so far
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 4c) Build the active set (indices where the convolution is almost maximal)
        # ------------------------------------------------------------------
        # We use a relative tolerance so that the set stays small.
        active_mask = conv >= cur_max - max(1e-12, tol_rel * cur_max)
        active_ks = np.nonzero(active_mask)[0]
        # If the active set becomes too large we randomly prune it.
        max_active = 200
        if active_ks.size > max_active:
            active_ks = rng.choice(active_ks, size=max_active, replace=False)

        # ------------------------------------------------------------------
        # 4d) Linear‑programming step
        # ------------------------------------------------------------------
        # maximise Σ b_j   s.t.   (a * b)_k ≤ cur_max  for every active k,
        #      b_j ≥ 0.
        A = _build_active_constraints(a, active_ks)
        b_ub = np.full(A.shape[0], cur_max, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # minimise –Σ b_j  ⇔ maximise Σ b_j
        bounds = [(0, None)] * n
        lp_res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})
        if lp_res.success:
            b = lp_res.x
            # Normalise to unit sum (simplex) – the LP already pushes the sum up,
            # but the normalisation keeps the iterates stable.
            total = b.sum()
            if total > 0:
                b /= total
                # Convex combination with the current iterate.
                a = (1.0 - lp_t) * a + lp_t * b
                a = _proj_simplex(a)            # keep exact unit sum, non‑negative
                continue   # go to next outer iteration
        # ------------------------------------------------------------------
        # 4e) Fallback – a single Adam‑style gradient step (the code from the
        #     previous solution) if the LP fails or is too cheap.
        # ------------------------------------------------------------------
        # Compute soft‑max weights (β = 1) – essentially a uniform weight over the
        # active set, but we keep the same machinery as before.
        beta = 1.0
        diff = conv - cur_max
        w_raw = np.exp(beta * diff)               # only the max element is ≈1, others ≈0
        w = w_raw / w_raw.sum()
        # Gradient of the smooth surrogate with these weights.
        # This reduces to the correlation of a with the weight vector.
        # We reuse the FFT‑based correlation from the previous code.
        # --------------------------------------------------------------
        # Correlation a ⋆ w (Eq. 9 in the description)
        # --------------------------------------------------------------
        L = w.size                     # = 2·n‑1
        fft_len_corr = 1 << ((L + n - 2 - 1).bit_length())
        fw = np.fft.rfft(w, n=fft_len_corr)
        fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
        corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: L + n - 1]
        start = n - 1
        grad = 2.0 * corr_full[start:start + n]

        # Simple SGD step with projection – no Adam bookkeeping needed.
        a = a - grad_t * grad
        a = _proj_simplex(a)

        # ------------------------------------------------------------------
        # 4f) Random restart if we are stuck for a while
        # ------------------------------------------------------------------
        if now - last_improve > max_no_improve:
            # re‑initialise from a random Dirichlet draw (adds diversity)
            a = rng.random(n)
            a = _proj_simplex(a.astype(np.float64))
            last_improve = now
            print(f"[{elapsed:6.2f}s] random restart – best so far {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 5) Return the best vector (apply the clipping imposed by evaluate_sequence)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Optimisation of a step‑function to minimise ``evaluate_sequence``.

The method alternates a small linear‑programming (LP) step with a
gradient fallback.  For the current vector ``a`` we compute the maximal
convolution value  M = max_k (a*a)_k .  The LP maximises the total mass
∑b_j under the linear constraints (a * b)_k ≤ M for every index *k*
that is (almost) active.  By convex‑combining the LP solution with the
current iterate we obtain a new vector whose maximal convolution is not
larger than before, while its ℓ₁‑norm is larger – after renormalising to
the probability simplex the objective 2·n·max_k(a*a)_k strictly
decreases.  If the LP fails we take a single projected‑gradient step.
Random restarts prevent stagnation.

The routine respects the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper: next power‑of‑two (used for FFT lengths)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for FFT‑based convolution."""
    size = 2 * n - 1
    return _next_pow2(size)


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex {x≥0, Σx = 1}
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex."""
    if v.ndim != 1:
        raise ValueError("input must be 1‑D")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Build the (very sparse) constraint matrix for the LP.
# Each row corresponds to an “active’’ convolution index k and each
# column j carries the coefficient f[i] with i = k‑j whenever the
# product f[i]·b[j] participates in (f*b)_k.
# ----------------------------------------------------------------------
def _build_active_constraints(f: np.ndarray, active_ks: np.ndarray) -> csr_matrix:
    n = f.size
    rows = active_ks.size
    data = []
    row_ind = []
    col_ind = []
    for r, k in enumerate(active_ks):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            j = k - i
            data.append(f[i])
            row_ind.append(r)
            col_ind.append(j)
    return csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Length of the step sequence.  The default (3800) matches the
        previously successful construction.

    Returns
    -------
    list[float]
        The best sequence found (values clipped to the limits imposed by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    # --------------------------------------------------------------
    # 2) Initial vector – analytic “sqrt‑difference’’ start.
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()                       # normalise to unit sum (probability simplex)

    # (Optionally) take a previously discovered construction if it is better.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand /= s
                    fft_len_cand = _fft_len_for_n(cand.size)
                    score_cand = 2.0 * cand.size * np.max(_conv_self_fft(cand, fft_len_cand))
                    score_cur = 2.0 * n * np.max(_conv_self_fft(a, _fft_len_for_n(n)))
                    if score_cand < score_cur:
                        a = cand
                        n = cand.size
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3) Pre‑compute FFT length (used for all convolutions)
    # --------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 4) Optimisation state
    # --------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * np.max(_conv_self_fft(best, fft_len))

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin
    last_improve = start_time

    # Parameters
    max_no_improve = 120.0               # seconds without improvement → random restart
    lp_step = 0.25                        # convex‑combination factor for the LP step
    grad_step = 0.05                      # step size for the gradient fallback
    tol_rel = 1e-6                        # relative tolerance for the active set

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- compute self‑convolution and current score ----
        conv = _conv_self_fft(a, fft_len)                 # length 2·n‑1
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- record the best solution so far ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ---- determine the active set (indices where the convolution is almost maximal) ----
        active_mask = conv >= cur_max - max(1e-12, tol_rel * cur_max)
        active_ks = np.nonzero(active_mask)[0]

        # prune if the set gets too large (keeps the LP cheap)
        max_active = 200
        if active_ks.size > max_active:
            active_ks = rng.choice(active_ks, size=max_active, replace=False)

        # ---- linear‑programming step (maximise total mass under the current bound) ----
        A = _build_active_constraints(a, active_ks)
        b_ub = np.full(A.shape[0], cur_max, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # minimise –∑b_j  ⇔ maximise ∑b_j
        bounds = [(0, None)] * n
        lp_res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                         method='highs', options={'presolve': True})

        if lp_res.success:
            b = lp_res.x
            s = b.sum()
            if s > 0:
                b /= s                               # project onto the simplex
                # convex combination with the current iterate
                a = (1.0 - lp_step) * a + lp_step * b
                a = _proj_simplex(a)
                continue   # go to next outer iteration

        # ---- fallback: a single projected‑gradient step (soft‑max with β=1) ----
        beta = 1.0
        diff = conv - cur_max                # ≤ 0, zero at the maximum entry
        w_raw = np.exp(beta * diff)
        w_sum = w_raw.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.zeros_like(w_raw)
            w[np.argmax(conv)] = 1.0
        else:
            w = w_raw / w_sum

        # gradient of Σ_k w_k (a*a)_k = 2·a ⋆ w  (correlation)
        L = w.size                           # = 2·n‑1
        fft_len_corr = 1 << ((L + n - 2 - 1).bit_length())
        fw = np.fft.rfft(w, n=fft_len_corr)
        fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
        corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: L + n - 1]
        start = n - 1
        grad = 2.0 * corr_full[start:start + n]

        a = a - grad_step * grad
        a = _proj_simplex(a)

        # ---- random restart if we have been stuck for a while ----
        if now - last_improve > max_no_improve:
            a = rng.random(n)
            a = _proj_simplex(a.astype(np.float64))
            last_improve = now
            print(f"[{elapsed:6.2f}s] random restart – best so far {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (apply the clipping imposed by
    #    evaluate_sequence).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
