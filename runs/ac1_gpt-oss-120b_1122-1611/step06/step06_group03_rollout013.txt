"""
Algorithm: hybrid simulated‑annealing + full‑softmax exponentiated‑gradient polishing.

* Initialise the step‑function with the analytically optimal “√‑difference’’
  shape a_i = √(i+1)‑√i (normalised to sum = 1).  If a previous construction
  `height_sequence_1` exists with a better internal score, use it instead.

* Stage 1 – Simulated annealing (≈ 70 % of the time budget).  Random
  “mass‑transfer”, block‑scaling and jitter moves are proposed; a Metropolis
  criterion with an exponentially decaying temperature accepts improvements
  and occasional worsening steps.  The objective used during the search is
  the fast internal score  2·n·max(a∗a)  (the denominator is constant = 1).

* Stage 2 – Deterministic polishing (remaining time).  The L∞‑objective
  max(a∗a) is approximated by a smooth log‑sum‑exp (soft‑max) with an
  increasing sharpness parameter β.  In each iteration a weighted gradient
  (the correlation of the soft‑max weights with a) is computed via FFT, and
  an exponentiated‑gradient (mirror‑descent on the simplex) update is applied.
  Step‑size η decays while β grows, driving the distribution toward a shape
  that flattens the self‑convolution.  The best vector seen during polishing
  is kept.

The routine respects the 1000 s budget, prints progress stamps and finally
returns a list of non‑negative heights (clipped to the limits imposed by
`evaluate_sequence`).  It works for any length `n` (default 3800) and uses
only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two vectors of length `n`."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _fft_len_for_lengths(l1: int, l2: int) -> int:
    """Length for convolution of lengths `l1` and `l2`."""
    size = l1 + l2 - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real‑to‑real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of 2·n·max(a∗a) for a normalised vector (∑a=1)."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Initialisations
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _maybe_use_global() -> np.ndarray | None:
    """If a global construction `height_sequence_1` exists, return it
    normalised to sum = 1."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Polishing – full‑softmax exponentiated‑gradient
# ----------------------------------------------------------------------
def _polish_full(
    a: np.ndarray,
    n: int,
    fft_len_self: int,
    fft_len_corr: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Refine `a` by minimizing max(a∗a) with a soft‑max approximation.
    Returns the best vector seen and its internal score (2·n·max).
    """
    start_time = time.monotonic()
    best = a.copy()
    best_score = _compute_score(best, fft_len_self, n)

    # schedule parameters
    eta_start = 0.18
    eta_end = 0.001
    beta_start = 2.0
    beta_end = 20000.0
    momentum = 0.9
    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:
        # current convolution & score
        conv = _conv_self(a, fft_len_self)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # keep the best one
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best = a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] polish: new best = {best_score:.6f}")

        # schedule (based on elapsed fraction)
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start * (beta_end / beta_start) ** frac   # exponential growth

        # soft‑max weights (stable formulation)
        w = np.exp(beta * (conv - cur_max))
        w_sum = w.sum()
        if w_sum == 0.0:                      # under‑flow – fall back to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient = 2 · correlation(w, a)  (computed via FFT)
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len_corr)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_corr)
        corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: (len(w) + n - 1)]
        grad = 2.0 * corr_full[n - 1 : n - 1 + n]

        # momentum smoothing
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update inside the simplex
        # clip the exponent to avoid overflow
        eta_grad = eta * grad
        eta_grad = np.clip(eta_grad, -30, 30)      # safety
        factor = np.exp(-eta_grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # early exit guard
        if time.monotonic() > deadline - 0.05:
            break

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    fft_len_self = _fft_len_for_n(n)                # for a∗a
    fft_len_corr = _fft_len_for_lengths(2 * n - 1, n)   # for gradient (w * a_rev)

    # --------------------------------------------------------------
    # Starting vector
    # --------------------------------------------------------------
    a = _analytic_start(n)

    cand_global = _maybe_use_global()
    if cand_global is not None and cand_global.size == n:
        # choose the better of analytic or global start (fast internal score)
        if _compute_score(cand_global, _fft_len_for_n(n), n) < _compute_score(a, fft_len_self, n):
            a = cand_global

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    best_score = _compute_score(best, fft_len_self, n)

    cur_a = a.copy()
    cur_score = best_score

    start_time = time.monotonic()
    # reserve a few seconds for polishing
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Simulated‑annealing parameters
    # --------------------------------------------------------------
    T0 = best_score * 0.5               # relatively hot start
    Tmin = 1e-9
    sigma_block = 0.40
    max_pair_frac = 0.60

    # --------------------------------------------------------------
    # Main SA loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        T = T0 * (Tmin / T0) ** (elapsed / total)

        # ----- propose a move -----
        a_new = cur_a.copy()
        r = rng.random()
        if r < 0.4:                     # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 1e-12:
                delta = rng.random() * a_new[i] * max_pair_frac
                a_new[i] -= delta
                a_new[j] += delta
        elif r < 0.7:                   # block scaling
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, sigma_block))
            a_new[start:start + width] *= factor
        elif r < 0.85:                  # local jitter on a few indices
            k = max(1, n // 1000)
            idxs = rng.integers(0, n, size=k)
            jitter = np.exp(rng.normal(0.0, sigma_block / 2.0, size=k))
            a_new[idxs] *= jitter
        else:                           # global jitter
            jitter = np.exp(rng.normal(0.0, sigma_block / 3.0, size=n))
            a_new *= jitter

        # ----- normalise onto the simplex -----
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s

        # ----- evaluate -----
        new_score = _compute_score(a_new, fft_len_self, n)

        # ----- Metropolis acceptance -----
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp((cur_score - new_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = cur_a.copy()

        )

    # --------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
