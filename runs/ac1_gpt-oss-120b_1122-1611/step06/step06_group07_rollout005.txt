import numpy as np
import time

def _fft_len_for_n(n):
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[:2*a.size - 1]

def _softmax(x, beta):
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s

def _gradient_via_correlation(a, w):
    # a rev convolution to compute correlation
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    return 2.0 * corr_full[n-1:n-1+n]

def _score(a, fft_len):
    conv = _conv_self(a, fft_len)
    maxb = conv.max()
    n = a.size
    # sum is normalized to 1 in our usage, but we compute for completeness
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * maxb / (sum_a ** 2))

def _gen_shape_r(r, n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

def _gen_shape_p(p, n):
    k = np.arange(1, n + 2, dtype=np.float64)
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Evolutionary‐search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The algorithm builds a small population of
    candidates (analytic families, random Dirichlet draws and the historic
    ``height_sequence_1``) and then repeatedly mutates the best individuals.
    Mutations consist of mass transfers, block scalings and log‑space jitter.
    Occasionally a short exponentiated‑gradient refinement is applied.
    Simulated‑annealing‐style acceptance of worse offspring keeps the search
    from stagnating.  The search stops when the time budget is exhausted and the
    best sequence found is returned.
    """
    rng = np.random.default_rng(seed)

    # problem size (default, may be overridden)
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # -----------------------------------------------------------------
    # Helper to evaluate and store a candidate (returns (score, a))
    # -----------------------------------------------------------------
    def evaluate_candidate(a):
        sc = _score(a, fft_len)
        return sc, a

    # -----------------------------------------------------------------
    # Initialise population
    # -----------------------------------------------------------------
    population = []  # list of (score, array)
    best_seq = None
    best_score = np.inf

    def try_add(a):
        nonlocal best_seq, best_score
        if a is None:
            return
        sc = _score(a, fft_len)
        population.append((sc, a))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[init] candidate score = {sc:.6f}")

    # 1. historic sequence
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                    if cand.size != n:
                        # resize by truncation/padding
                        if cand.size > n:
                            cand = cand[:n]
                        else:
                            cand = np.pad(cand, (0, n - cand.size))
                    try_add(cand)
        except Exception:
            pass

    # 2. analytic families (r and p)
    for r in np.linspace(0.5, 0.96, 25):
        a = _gen_shape_r(r, n)
        try_add(a)
    for p in np.linspace(0.1, 2.5, 30):
        a = _gen_shape_p(p, n)
        try_add(a)

    # 3. random Dirichlet draws (sparse and dense)
    for _ in range(20):
        # sparse draw (shape < 1)
        a = rng.gamma(shape=0.3, scale=1.0, size=n)
        a /= a.sum()
        try_add(a)

    for _ in range(10):
        a = rng.gamma(shape=2.0, scale=1.0, size=n)
        a /= a.sum()
        try_add(a)

    # keep only top K
    max_pop = 30
    population.sort(key=lambda x: x[0])
    population = population[:max_pop]

    # -----------------------------------------------------------------
    # Core evolutionary loop
    # -----------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # small safety margin

    # annealing schedule
    T_start = 0.1
    T_end = 1e-5

    # parameters
    block_prob = 0.07
    pair_prob = 0.15
    jitter_prob = 0.10
    grad_refine_prob = 0.20   # apply short gradient refinement

    iter_count = 0
    while time.monotonic() < deadline:
        iter_count += 1
        # pick a parent preferentially among the top half
        parent_idx = rng.integers(0, min(len(population), max_pop // 2))
        parent_score, parent = population[parent_idx]

        # clone parent
        child = parent.copy()

        # --- mutations ---
        # block scaling
        if rng.random() < block_prob:
            width = rng.integers(1, max(2, n // 25))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.3))
            child[start:start+width] *= factor

        # pairwise mass transfer
        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and child[i] > 1e-16:
                delta = rng.random() * child[i] * 0.5
                child[i] -= delta
                child[j] += delta

        # log‑space jitter (multiplicative noise)
        if rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            child *= jitter

        # ensure non‑negative and renormalise
        child = np.clip(child, 0.0, None)
        s = child.sum()
        if s == 0.0:
            child = np.full_like(child, 1.0 / n)
        else:
            child /= s

        # optional short gradient refinement
        if rng.random() < grad_refine_prob:
            # run a few exponentiated‑gradient steps with moderate beta
            beta = 1.0
            eta = 0.5
            for _ in range(8):
                conv = _conv_self(child, fft_len)
                w = _softmax(conv, beta)
                grad = _gradient_via_correlation(child, w)
                factor = np.exp(-eta * grad)
                child = child * factor
                child = np.clip(child, 0.0, None)
                s = child.sum()
                if s == 0.0:
                    child = np.full_like(child, 1.0 / n)
                else:
                    child /= s
                # slowly increase beta, decrease eta
                beta = min(500.0, beta * 1.5)
                eta = max(0.02, eta * 0.9)

        # evaluate child
        child_score = _score(child, fft_len)

        # accept according to simulated‑annealing rule
        now = time.monotonic()
        elapsed = now - start_time
        fraction = elapsed / (deadline - start_time)
        temperature = T_start * (1 - fraction) + T_end * fraction

        accept = False
        if child_score < parent_score:
            accept = True
        else:
            # probability to replace the parent (allows exploration)
            prob = np.exp(-(child_score - parent_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # replace parent in population
            population[parent_idx] = (child_score, child)

        # maintain sorted population (keep best up to max_pop)
        population.sort(key=lambda x: x[0])
        if len(population) > max_pop:
            population = population[:max_pop]

        # update global best
        if child_score < best_score - 1e-12:
            best_score = child_score
            best_seq = child.copy()
            print(f"[{elapsed:6.2f}s] new global best = {best_score:.6f}")

    # -----------------------------------------------------------------
    # Return best found (clipped to limits required by evaluate_sequence)
    # -----------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that accommodates a linear convolution
    of two vectors of length `n` (full length = 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max with temperature `beta`."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth surrogate  Σ_k w_k (a∗a)_k.
    It is 2·a⋆w (correlation).  Computed with a single FFT‑based convolution.
    """
    n = a.size
    a_rev = a[::-1]                     # reversed a
    L = w.size + n - 1                  # length of the full convolution
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    # the centre slice gives the correlation of length n
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective used by `evaluate_sequence` (sum is assumed ≠0)."""
    conv = _conv_self(a, fft_len)
    max_b = conv.max()
    n = a.size
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


# ----------------------------------------------------------------------
# Analytic families (used for the initial population)
# ----------------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^r – i^r  (0<r<1 gives a decreasing shape)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_shape_p(p: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^{-p} – (i+2)^{-p}  (p>0)."""
    k = np.arange(1, n + 2, dtype=np.float64)    # 1 … n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Evolutionary‑search for a non‑negative step function that minimizes the
    security‑checked `evaluate_sequence` objective.

    The algorithm builds a modest population (analytic families, historic
    solution, random Dirichlet draws).  In each iteration it mutates a parent
    (mass transfer, block scaling, log‑space jitter) and optionally refines the
    offspring with a few exponentiated‑gradient steps on a smooth soft‑max
    surrogate.  A simulated‑annealing acceptance rule lets occasional worse
    offspring survive, aiding exploration.  The best vector seen before the
    time budget expires is returned (already clipped to the limits imposed by
    `evaluate_sequence`).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (may be overridden by the harness)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Helper to keep the best solution seen so far
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # Initialise a small population (sorted by score, low → high)
    # ------------------------------------------------------------------
    pop = []                     # list of (score, array)

    def add_to_pop(a: np.ndarray):
        nonlocal best_seq, best_score
        if a is None:
            return
        sc = _score(a, fft_len)
        pop.append((sc, a))
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[init] add candidate score = {sc:.6f}")

    # historic sequence (if available)
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                    if cand.size != n:
                        if cand.size > n:
                            cand = cand[:n]
                        else:
                            cand = np.pad(cand, (0, n - cand.size))
                    add_to_pop(cand)
        except Exception:
            pass

    # analytic families
    for r in np.linspace(0.5, 0.96, 25):
        add_to_pop(_gen_shape_r(r, n))
    for p in np.linspace(0.1, 2.5, 30):
        add_to_pop(_gen_shape_p(p, n))

    # random Dirichlet draws – sparse & dense
    for _ in range(20):
        a = rng.gamma(shape=0.3, scale=1.0, size=n)   # sparse
        a /= a.sum()
        add_to_pop(a)
    for _ in range(10):
        a = rng.gamma(shape=2.0, scale=1.0, size=n)   # dense
        a /= a.sum()
        add_to_pop(a)

    # keep only the best few
    pop.sort(key=lambda x: x[0])
    max_pop = 30
    pop = pop[:max_pop]

    # ------------------------------------------------------------------
    # Evolutionary loop (time‑bounded)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)   # a couple of seconds safety

    # annealing schedule (temperature decreases linearly)
    T_start, T_end = 0.12, 1e-5

    # mutation probabilities
    prob_block = 0.07
    prob_pair = 0.15
    prob_jitter = 0.10
    prob_refine = 0.20

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        fraction = elapsed / (deadline - start)
        temperature = T_start * (1.0 - fraction) + T_end * fraction

        # ------------------------------------------------------------------
        # 1) pick a parent (biased to the best half of the pool)
        # ------------------------------------------------------------------
        parent_idx = rng.integers(0, max(1, len(pop) // 2))
        parent_score, parent = pop[parent_idx]

        child = parent.copy()

        # ------------------------------------------------------------------
        # 2) apply stochastic mutations
        # ------------------------------------------------------------------
        # block scaling
        if rng.random() < prob_block:
            width = rng.integers(1, max(2, n // 25))
            start_b = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.3))
            child[start_b:start_b + width] *= factor

        # pairwise mass transfer
        if rng.random() < prob_pair:
            i, j = rng.integers(0, n, size=2)
            if i != j and child[i] > 1e-16:
                delta = rng.random() * child[i] * 0.5
                child[i] -= delta
                child[j] += delta

        # log‑space jitter (multiplicative noise)
        if rng.random() < prob_jitter:
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            child *= jitter

        # re‑normalise onto the simplex
        child = np.clip(child, 0.0, None)
        s = child.sum()
        if s == 0.0:
            child = np.full_like(child, 1.0 / n)
        else:
            child /= s

        # ------------------------------------------------------------------
        # 3) optional short exponentiated‑gradient refinement
        # ------------------------------------------------------------------
        if rng.random() < prob_refine:
            beta = 1.0
            eta = 0.6
            for _ in range(8):
                conv = _conv_self(child, fft_len)
                w = _softmax(conv, beta)
                grad = _gradient_via_correlation(child, w)
                factor = np.exp(-eta * grad)
                child = child * factor
                child = np.clip(child, 0.0, None)
                s = child.sum()
                if s == 0.0:
                    child = np.full_like(child, 1.0 / n)
                else:
                    child /= s
                beta = min(500.0, beta * 1.5)
                eta = max(0.02, eta * 0.9)

        # ------------------------------------------------------------------
        # 4) evaluate the offspring
        # ------------------------------------------------------------------
        child_score = _score(child, fft_len)

        # ------------------------------------------------------------------
        # 5) simulated‑annealing acceptance for the parent slot
        # ------------------------------------------------------------------
        accept = False
        if child_score < parent_score:
            accept = True
        else:
            prob = np.exp(-(child_score - parent_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            pop[parent_idx] = (child_score, child)

        # ------------------------------------------------------------------
        # 6) keep the population trimmed and sorted
        # ------------------------------------------------------------------
        pop.sort(key=lambda x: x[0])
        if len(pop) > max_pop:
            pop = pop[:max_pop]

        # ------------------------------------------------------------------
        # 7) update global best and log
        # ------------------------------------------------------------------
        if child_score < best_score - 1e-12:
            best_score = child_score
            best_seq = child.copy()
            print(f"[{elapsed:6.2f}s] new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (already clipped to the bounds used by
    # `evaluate_sequence`).  The list is plain Python floats as required.
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
