def solve_lp_max_sum(f, M):
    n = len(f)
    m = 2 * n - 1
    # Build constraints A @ g <= M
    # Use scipy.sparse.diags to create matrix
    import scipy.sparse as sp
    data = [np.full(n, fi) for fi in f]  # list of arrays length n each
    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(m, n), format='csc')
    b = np.full(m, M)
    c = -np.ones(n)   # because we minimize -sum(g)
    # bounds: (0, None)
    res = sp.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return zeros
        return np.zeros_like(f)
    return res.x
```

def _softmax_step(a, eta, beta, fft_len_self, fft_len_grad):
    conv = _self_conv(a, fft_len_self)
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum
    rev_a = a[::-1]
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
    grad = 2.0 * grad_full[n - 1 : n - 1 + n]
    a = a - eta * grad
    a = _project_simplex(a)
    return a

"""
Hybrid optimizer for the step‑function bound.

The algorithm works on a probability vector a (a_i ≥ 0, Σ a_i = 1) and
minimises

    f(a) = 2·n·max(a * a),

which is exactly the value returned by ``evaluate_sequence`` (the sum of a
does not matter because the objective is homogeneous).

Two complementary phases are used for each length n:

1.  **Smoothed projected gradient descent** (≈30 % of the time):
    a soft‑max with temperature β approximates the max‐operator.
    The exact gradient of the soft‑max objective is
    g = 2·(p * rev(a)) and an additive Euclidean step is performed,
    followed by Euclidean projection onto the simplex.

2.  **LP‑based Frank‑Wolfe mixing** (the remaining time):
    For the current vector a we compute M = max(a*a).  The linear program

        max   Σ g_i
        s.t.  a * g ≤ M      (convolution, elementwise)
              g_i ≥ 0

    yields a feasible direction g that tries to increase the total mass
    while keeping the convolution below the current peak M.
    After normalising g we take a convex combination
        a ← (1−η)·a + η·ĝ
    with a small step size η (chosen from a short line‑search).  The
    iterate is again projected onto the simplex.

Both phases are surrounded by stagnation handling: after 30 s without
improvement a multiplicative log‑normal jitter around the best point is
applied.  The optimiser is run for several candidate lengths (including
the previous construction if available) while respecting the 1000 s
budget.  The best sequence found is returned as a plain list of floats.

The implementation uses only NumPy, SciPy (sparse matrices and ``linprog``)
and the provided ``evaluate_sequence`` for the final check.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of prescribed length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_conv(a: np.ndarray, fft_len_self: int) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return _self_conv(a, fft_len_self).max()


# ----------------------------------------------------------------------
# Linear‑programming sub‑routine (direction g)
# ----------------------------------------------------------------------


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP
        maximise   Σ g_i
        subject to a * g ≤ M  (convolution)
                 g_i ≥ 0
    The matrix A (shape (2n‑1, n)) has A[k,i] = f[i] if i ≤ k ≤ i+n‑1,
    otherwise 0.  A is built as a sparse diagonal‑stack.
    """
    n = f.shape[0]
    m = 2 * n - 1

    # Build A via a stack of constant diagonals.
    # Each diagonal i (offset i) consists of f[i] repeated n times.
    diags_data = [np.full(n, val, dtype=np.float64) for val in f]
    offsets = np.arange(n, dtype=int)
    A = sp.diags(diags_data, offsets, shape=(m, n), format="csc")

    b = np.full(m, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -Σg  → maximise Σg
    bounds = [(0, None)] * n

    try:
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if res.success:
            return res.x
    except Exception as e:
        print("LP failure:", e)

    # fallback – return zeros (will trigger jitter)
    return np.zeros_like(f)


# ----------------------------------------------------------------------
# Hybrid optimisation for a fixed length
# ----------------------------------------------------------------------


def _optimise_one_length_hybrid(a_init: np.ndarray,
                                deadline: float,
                                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hybrid optimisation for a given length n.
    Returns the best vector and its objective value (2·n·max_conv).
    """
    a = a_init.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # initial evaluation
    cur_max = _max_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * cur_max

    start = time.monotonic()
    total = deadline - start
    phase1_end = start + 0.30 * total   # smoothed gradient
    phase2_end = start + 0.60 * total   # exact sub‑gradient
    # parameters for phase‑1
    eta_start, eta_end = 1.0, 0.01
    beta_start, beta_end = 0.5, 3500.0
    # parameters for phase‑2
    sub_eta0 = 0.20
    # LP‑mixing step‑size candidates
    eta_cands = (0.02, 0.05, 0.10, 0.20)

    last_improve = start
    it = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        # ------------------------------------------------------------------
        # Phase 1 – smoothed gradient (soft‑max)
        # ------------------------------------------------------------------
        if now < phase1_end:
            frac = (now - start) / (phase1_end - start)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start + (beta_end - beta_start) * frac

            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()
            # stable soft‑max
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            rev_a = a[::-1]
            fp = np.fft.rfft(p, n=fft_len_grad)
            fra = np.fft.rfft(rev_a, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            a = a - eta * grad
            a = _project_simplex(a)

        # ------------------------------------------------------------------
        # Phase 2 – exact sub‑gradient of the current peak
        # ------------------------------------------------------------------
        elif now < phase2_end:
            conv = _self_conv(a, fft_len_self)
            k_max = int(conv.argmax())

            g = np.zeros_like(a)
            i_min = max(0, k_max - (n - 1))
            i_max = min(n - 1, k_max)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k_max - idx_i
            g[idx_i] = 2.0 * a[idx_j]

            eta = sub_eta0 / np.sqrt(it)
            a = a - eta * g
            a = _project_simplex(a)

        # ------------------------------------------------------------------
        # Phase 3 – LP‑based Frank‑Wolfe mixing
        # ------------------------------------------------------------------
        else:
            conv = _self_conv(a, fft_len_self)
            M = conv.max()
            g = _solve_lp_max_sum(a, M)

            if g.sum() <= 1e-12:                     # LP failed → jitter
                a = best.copy()
                noise = np.exp(rng.normal(0.0, 0.10, size=n))
                a = a * noise
                a = _project_simplex(a)
                last_improve = time.monotonic()
                continue

            g_norm = g / g.sum()

            # simple line‑search over a handful of step sizes
            cur_score = 2.0 * n * M
            best_local_a = a
            best_local_score = cur_score

            for eta in eta_cands:
                a_tmp = (1.0 - eta) * a + eta * g_norm
                a_tmp = _project_simplex(a_tmp)
                score_tmp = 2.0 * n * _max_conv(a_tmp, fft_len_self)
                if score_tmp < best_local_score:
                    best_local_score = score_tmp
                    best_local_a = a_tmp

            a = best_local_a
            cur_score = best_local_score

        # ------------------------------------------------------------------
        # Evaluate current candidate
        # ------------------------------------------------------------------
        cur_score = 2.0 * n * _max_conv(a, fft_len_self)

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # Stagnation handling
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * noise
            a = _project_simplex(a)
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    return best, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose candidate lengths.
    # ------------------------------------------------------------------
    lengths = []

    # If a previous construction exists, start from it.
    prev_seq = globals().get("height_sequence_1")
    if isinstance(prev_seq, (list, tuple, np.ndarray)):
        arr = np.asarray(prev_seq, dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            lengths.append(arr.size)

    # Additional lengths (powers of two or near‑optimal sizes)
    for n in (1024, 2048, 3000, 3809, 4096, 5000):
        if n not in lengths:
            lengths.append(n)

    best_overall = None
    best_score_overall = float("inf")

    # ------------------------------------------------------------------
    # 2) Optimise each length while time permits.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # ---- initialise a probability vector on the simplex ----------------
        if prev_seq is not None and isinstance(prev_seq, (list, tuple, np.ndarray)):
            seq = np.asarray(prev_seq, dtype=np.float64)
            if seq.size == n:
                a0 = seq.copy()
            else:
                # linear interpolation to required size
                x_old = np.linspace(0.0, 1.0, seq.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(x_new, x_old, seq)
        else:
            a0 = rng.dirichlet(np.ones(n))

        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / n)

        # ---- run hybrid optimiser for this length ------------------------
        vec, score = _optimise_one_length_hybrid(a0, deadline, rng)

        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - (deadline - (budget_s - 5.0))
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback (should never happen)
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    # 4) Clip to the limits enforced by ``evaluate_sequence`` and return.
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    return [float(x) for x in best_overall]
