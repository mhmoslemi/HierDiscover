"""
Step‑function optimizer.

The routine searches for a non‑negative sequence `a` (treated as a
probability vector) that minimises

    score = 2·n·max_k (a ∗ a)_k   (where “∗” is linear convolution)

subject to a_i ≥ 0 and ∑ a_i = 1 (the evaluator normalises anyway).

The search proceeds in three stages:

1. **Low‑dimensional families** – exhaustive scan of two‑parameter
   families:
       * a_i = (i + c + 1)^r – (i + c)^r       (0.5 ≤ r ≤ 0.95, 0 ≤ c ≤ 4)
       * a_i = (i + b + 1)^{-p} – (i + b + 2)^{-p}
         (0.1 ≤ p ≤ 2.0,   0 ≤ b ≤ 10)
   Each candidate is normalised to sum 1 and evaluated exactly.
   The best of these seeds the next stage.

2. **Exponential‑gradient refinement** – minimise the true
   max‑convolution via a smooth surrogate.
   For a temperature β we form soft‑max weights
        w_i ∝ exp(β·conv_i)
   and take the gradient 2·a⋆w (correlation).  A multiplicative
   update `a ← a·exp(−η·grad)` is performed and the vector is
   re‑normalised onto the simplex.  β is increased exponentially
   from 0.1 to 1e5, while η decays linearly from 3.0 to 0.01.
   Random block perturbations and pairwise mass transfers keep the
   search from stagnating.  If no improvement is seen for a while,
   a jittered restart around the current best solution is performed.

3. **Finalisation** – the best vector found (clipped to the limits
   imposed by `evaluate_sequence`) is returned as a plain list of
   floats.

The whole procedure respects the supplied time budget (≈ 1000 s) and
always returns a valid candidate before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that accommodates a linear convolution
    of two length‑n arrays (full length = 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable softmax with temperature `beta`."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  sum_i w_i·(a∗a)_i  with respect to a.
    It equals 2·a ⋆ w (correlation).  Both a and w are real vectors;
    w must have length 2·len(a)‑1.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _exact_score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value 2·n·max(conv) for a normalised vector."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Analytic families (two parameters each)
# ----------------------------------------------------------------------
def _shape_r_shift(r: float, c: float, n: int) -> np.ndarray | None:
    """
    a_i = (i + c + 1)^r – (i + c)^r , 0 < r < 1, c ≥ 0.
    Returns a normalised probability vector or None if degenerate.
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + c + 1.0) ** r - (i + c) ** r
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _shape_p_shift(p: float, b: float, n: int) -> np.ndarray | None:
    """
    a_i = (i + b + 1)^{-p} – (i + b + 2)^{-p} , p > 0, b ≥ 0.
    Returns a normalised probability vector or None if degenerate.
    """
    k = np.arange(1, n + 2, dtype=np.float64) + b    # 1 … n+1 shifted by b
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    `evaluate_sequence`.  The method combines a grid search over
    low‑dimensional families with an exponentiated‑gradient refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    best_a = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 0)  Existing known construction, if any
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.01:
                    cand = cand / s
                    sc = _exact_score(cand, _fft_len_for_n(cand.size))
                    if sc < best_score:
                        best_a, best_score = cand.copy(), sc
                        n = cand.size
                        fft_len = _fft_len_for_n(n)
                        print(f"[init] height_sequence_1 score = {sc:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1)  Exhaustive scan of two‑parameter families
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # 2 s safety

    # r‑family (0.5 … 0.95) with shift c (0 … 4)
    r_vals = np.linspace(0.5, 0.95, 91)          # step ≈0.005
    c_vals = np.linspace(0.0, 4.0, 9)           # step ≈0.5
    for r in r_vals:
        for c in c_vals:
            a = _shape_r_shift(r, c, n)
            if a is None:
                continue
            sc = _exact_score(a, fft_len)
            if sc < best_score:
                best_a, best_score = a, sc
                print(f"[grid‑r] r={r:.3f} c={c:.2f} score={sc:.6f}")

    # p‑family (0.1 … 2.0) with shift b (0 … 10)
    p_vals = np.linspace(0.1, 2.0, 96)          # step ≈0.02
    b_vals = np.linspace(0.0, 10.0, 11)        # step ≈1.0
    for p in p_vals:
        for b in b_vals:
            a = _shape_p_shift(p, b, n)
            if a is None:
                continue
            sc = _exact_score(a, fft_len)
            if sc < best_score:
                best_a, best_score = a, sc
                print(f"[grid‑p] p={p:.3f} b={b:.1f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 2)  Exponentiated‑gradient refinement (soft‑max → hard‑max)
    # --------------------------------------------------------------
    if best_a is None:
        # fallback to uniform distribution if everything failed
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _exact_score(best_a, fft_len)

    cur_a = best_a.copy()
    cur_score = best_score
    last_improve = time.monotonic()

    # Hyper‑parameters (tuned empirically)
    eta_start, eta_end = 3.0, 0.01                     # learning‑rate schedule
    beta_start, beta_end = 0.1, 1e5                  # soft‑max temperature schedule
    block_prob, pair_prob = 0.06, 0.12              # exploration probabilities
    restart_gap = 120.0                              # seconds without improvement

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        progress = elapsed / total

        # schedules
        eta = eta_start * (1.0 - progress) + eta_end * progress
        # exponential increase for beta (more aggressive than linear)
        beta = beta_start * (beta_end / beta_start) ** progress

        # full convolution for bookkeeping
        conv = _conv_self(cur_a, fft_len)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        # keep the best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        #   Soft‑max weights (smooth approximation of the max)
        # ----------------------------------------------------------
        use_hard = progress > 0.92  # in the last ~8 % of the run switch to hard max
        if use_hard:
            w = np.zeros_like(conv)
            w[int(conv.argmax())] = 1.0
        else:
            w = _softmax(conv, beta)

        # gradient w.r.t. a
        grad = _gradient_via_correlation(cur_a, w)

        # exponentiated‑gradient update
        factor = np.exp(-eta * grad)
        cur_a = cur_a * factor
        cur_a = np.clip(cur_a, 0.0, None)
        s = cur_a.sum()
        if s == 0.0:
            cur_a = np.full_like(cur_a, 1.0 / n)
        else:
            cur_a = cur_a / s

        # ----------------------------------------------------------
        # Random exploration moves
        # ----------------------------------------------------------
        if rng.random() < block_prob:
            # block perturbation – multiply a random contiguous block
            max_width = max(2, n // 30)
            width = rng.integers(1, max_width + 1)
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.30))
            cur_a[start:start + width] *= mult
            cur_a = cur_a / cur_a.sum()

        if rng.random() < pair_prob:
            # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > 1e-14:
                delta = rng.random() * cur_a[i] * 0.45
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = cur_a / cur_a.sum()

        # ----------------------------------------------------------
        # Stagnation detection → soft restart
        # ----------------------------------------------------------
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.18, size=n))
            cur_a = best_a * jitter
            cur_a = cur_a / cur_a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 3)  Return best vector (clipped to admissible range)
    # --------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    # Safety normalisation (the evaluator will re‑normalise if needed)
    total = best.sum()
    if total < 0.01:
        # fall back to uniform if something went terribly wrong
        best = np.full_like(best, 1.0 / best.size)
    else:
        best = best / total
    return [float(v) for v in best]
