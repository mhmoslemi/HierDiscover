"""
Optimiser for the self‑convolution bound.

Algorithm
---------
1️⃣  Initialise a non‑negative vector `a` on the simplex
    (sum = √(2 n)).  If a previous solution is available (`height_sequence_1`)
    it is linearly interpolated to the current length `n`.

2️⃣  Parameterise the simplex by latent variables `y` (log‑space):
        a = exp(y) / sum(exp(y)) * √(2 n)

3️⃣  Optimise `y` with Adam (first‑order momentum) on a *smooth* surrogate of
    the objective:
        f_β(a) = softmax_β( a * a )
    where the softmax is taken over the self‑convolution `c = a*a` with a
    temperature β that is annealed from a small value (β≈0.5) to a larger one
    (β≈8).  The gradient of the softmax surrogate is obtained analytically
    via FFT‑based correlation:
        ∇_a f_β = 2 * corr(w, a) ,   w = softmax_β(c)

4️⃣  The gradient w.r.t. `y` follows from the chain rule on the simplex:
        ∇_y = a * ( ∇_a – (a·∇_a)/√(2 n) )

5️⃣  Adam updates `y`.  The current score `2·n·max(a*a)` is evaluated each step
    and the best vector is stored.

6️⃣  If no improvement occurs for 30 s the optimiser is *restarted* from the
    best solution with a small log‑normal perturbation (helps escape plateaus).

The routine respects the 1000 s budget (with a 5 s safety margin) and returns
the best step‑function found as a plain list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a * a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max over the convolution with temperature `beta`."""
    # numerically stable: subtract the maximum exponent
    max_exp = np.max(beta * conv)
    w = np.exp(beta * conv - max_exp)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _grad_wrt_a(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate w.r.t. ``a``.
    Returns 2·corr(w, a) for the first ``len(a)`` entries.
    """
    # correlation = conv(w, rev(a)) shifted by n‑1
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    conv_full = np.fft.irfft(fw * fa_rev, n=fft_len)
    # the true linear convolution length is len(w)+len(a_rev)-1 = 3·n‑2
    conv_len = w.size + a_rev.size - 1
    conv_full = conv_full[:conv_len]
    start = a.size - 1               # = n‑1
    return 2.0 * conv_full[start:start + a.size]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Choose a working length – a power of two is convenient.
    # --------------------------------------------------------------
    n = 8192                       # ≈8 k entries give a good trade‑off

    # --------------------------------------------------------------
    # 1) Initialise a vector on the simplex (sum = √(2 n)).
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if init.ndim == 1 and init.size > 0 and np.isfinite(init).all():
            if init.size != n:
                # linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, init.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, init)
            else:
                a = init.copy()
            a = np.clip(a, 0.0, None)
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    target_sum = math.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s * target_sum

    # --------------------------------------------------------------
    # 2) Log‑space parameterisation of the simplex.
    # --------------------------------------------------------------
    eps0 = 1e-30
    y = np.log(np.clip(a, eps0, None))

    # --------------------------------------------------------------
    # 3) Adam bookkeeping.
    # --------------------------------------------------------------
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # --------------------------------------------------------------
    # 4) Pre‑compute FFT lengths.
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for correlation w ⋆ a

    # --------------------------------------------------------------
    # 5) Initial evaluation.
    # --------------------------------------------------------------
    conv_self = _self_conv_fft(a, fft_len_self)
    cur_score = 2.0 * n * conv_self.max()
    best_score = cur_score
    best_a = a.copy()
    print(f"[0.00s] initial score {best_score:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    last_improve = start_time
    it = 0

    # --------------------------------------------------------------
    # 6) Annealing / schedule parameters.
    # --------------------------------------------------------------
    beta_start = 0.5
    beta_end = 8.0
    step_start = 0.5
    step_end = 0.05

    # --------------------------------------------------------------
    # 7) Main optimisation loop.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        frac_total = (now - start_time) / (deadline - start_time)

        # ---- annealed temperature and step size -------------------
        beta = beta_start + frac_total * (beta_end - beta_start)
        step = step_start * (1.0 - frac_total) + step_end * frac_total

        # ---- recover a from log‑space -------------------------------
        a_raw = np.exp(y)
        a = a_raw / a_raw.sum() * target_sum

        # ---- self‑convolution & current bound ----------------------
        conv_self = _self_conv_fft(a, fft_len_self)
        cur_score = 2.0 * n * conv_self.max()

        # ---- soft‑max weights over the convolution -----------------
        w = _softmax_weights(conv_self, beta)

        # ---- gradient w.r.t. a (correlation) -----------------------
        g_a = _grad_wrt_a(w, a, fft_len_grad)

        # ---- gradient w.r.t. y (simplex chain rule) ----------------
        dot_ga = np.dot(g_a, a)                     # ⟨g, a⟩
        grad_y = a * (g_a - dot_ga / target_sum)

        # ---- Adam update -------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y * grad_y)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y -= step * m_hat / (np.sqrt(v_hat) + eps)

        # ---- keep the best solution seen so far --------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] iter {it:5d} new best {best_score:.6f}")

        # ---- restart if we have stagnated for >30 s ---------------
        if now - last_improve > 30.0:
            # restart from the current best with a modest log‑normal jitter
            y = np.log(np.clip(best_a, eps0, None))
            y += rng.normal(0.0, 0.09, size=y.shape)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation")

    # --------------------------------------------------------------
    # 8) Return the best vector (clipped to safe limits).
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
