"""
Propose a non‑negative step sequence that (approximately) minimizes
``evaluate_sequence``.  The algorithm works in three stages:

1. **Parameter sweep** – try cheap parametric families
   a_i ∝ (i+shift+1)^(-p)·((i+1)/n)^s   (p≈0.5, shift≈0…30, s≈0…1.5)
   plus a few random Dirichlet starts.  The best candidate from this
   sweep is kept.

2. **Sub‑gradient descent on the true max‑convolution** – at each
   iteration we locate the index k where (a∗a)_k is maximal,
   compute the exact gradient of that single term
   (which is 2·a reversed and shifted), take a small step and
   project back onto the probability simplex.  The step size is
   linearly annealed from 0.5 down to 0.001.

3. **Occasional stochastic jitter** (multiplicative log‑normal noise)
   and a simple restart mechanism when the score does not improve for
   a while.  The best vector seen throughout the whole budget is
   returned (clipped to the limits of ``evaluate_sequence``).

The whole search respects the time budget (default 1000 s) and prints
progress so that the optimiser can be monitored.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear convolution utilities (FFT based)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 (length for linear convolution)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (real FFT, length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may produce tiny imag parts – discard them.
    return np.real_if_close(conv)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution.
        return np.full_like(v, 1.0 / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical cleanup – enforce exact sum 1.
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / v.size)
    return w


# ----------------------------------------------------------------------
# Scoring helper – exact objective used by evaluate_sequence
# ----------------------------------------------------------------------
def _score(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(a∗a) (the numerator of evaluate_sequence for sum(a)=1)."""
    n = a.size
    conv = _conv_self(a, fft_len)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that (approximately) minimizes
    ``evaluate_sequence`` within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Length of the step sequence (default: 3800).

    Returns
    -------
    list[float]
        The best sequence found (values clipped to the limits imposed by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    n = max(2, n)

    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 2) Initialise best candidate – try to reuse a previous winner.
    # --------------------------------------------------------------
    best = None
    best_score = np.inf

    # If a globally stored solution exists, use it as a starting point.
    if "height_sequence_1" in globals():

            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():

                if cand.sum() > 0.0:

                    sc = _score(cand, fft_len)
                    best, best_score = cand, sc
        except Exception:

    # Uniform start (fallback) – convenient for the simplex projection.
    if best is None:
        a0 = np.full(n, 1.0 / n)
        best, best_score = a0.copy(), _score(a0, fft_len)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin

    # --------------------------------------------------------------
    # Parameter ranges
    p_vals = np.linspace(0.45, 0.60, 7)          # around the sqrt‑diff exponent 0.5
    shift_vals = np.arange(0, 31, 5)            # integer shift of the index
    s_vals = np.linspace(0.0, 1.5, 4)           # extra polynomial weighting

    # Helper to evaluate a candidate
    def _try_candidate(seq):
        nonlocal best, best_score
        sc = _score(seq, fft_len)
        if sc < best_score - 1e-12:

            print(f"[{elapsed:6.2f}s] new best from sweep = {best_score:.6f}")

                if seq_sum == 0.0 or not np.isfinite(seq_sum):

    # --------------------------------------------------------------
    a = best.copy()          # current iterate
    last_improve = time.monotonic()

    # Hyper‑parameters for the descent

    lr_end = 0.001           # final learning rate

    # ------------------------------------------------------------------
