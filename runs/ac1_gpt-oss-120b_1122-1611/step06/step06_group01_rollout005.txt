"""Optimization for minimizing the L_inf norm of self‑convolution.

The algorithm works in three stages:

1. **Parametric warm‑start** – a small family of simple shapes (Gaussian,
   power‑law, triangular) is sampled for each candidate length.  The best
   shape (according to ``evaluate_sequence``) becomes the starting point.

2. **Projected gradient descent with a *hard* max‑gradient** – we minimise
   the exact maximum of the convolution (instead of a soft‑max).  The
   gradient of the max is obtained analytically as
   ``g = 2·(p ⋆ rev(a))`` where ``p`` is the uniform distribution over the
   active peak indices.  The update is performed with Nesterov momentum
   and a step‑size that decays like ``1/√t``.  After each step the vector
   is projected onto the probability simplex.

   while, a multiplicative log‑normal jitter is applied to the current

list of floats.
"""

def _gradient_max_convolution(a: np.ndarray,
                              fft_len_self: int,
                              fft_len_grad: int) -> Tuple[np.ndarray, float]:
    """
    Returns the gradient of ``max(a*a)`` w.r.t. ``a`` and the current maximum.
    """
    conv = _self_conv(a, fft_len_self)
    max_val = conv.max()
    # active set
    active = np.where(np.abs(conv - max_val) < 1e-12)[0]
    if active.size == 0:
        # numerical safety – fall back to soft‑max with large β
        active = np.array([conv.argmax()], dtype=int)
    p = np.zeros_like(conv)
    p[active] = 1.0 / active.size
    rev_a = a[::-1]
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    g_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * a.size - 2]
    grad = 2.0 * g_full[a.size - 1 : a.size - 1 + a.size]
    return grad, max_val

def _optimise_length(a0: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator) -> Tuple[np.ndarray, float]:
    """
    Optimises a single length using momentum projected gradient descent.
    Returns the best vector and its evaluation score.
    """
    a = a0.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # keep track of the best solution seen
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    # parameters
    eta0 = 0.8          # initial step size multiplier
    mu = 0.9            # momentum factor
    stagnation_limit = 30.0
    last_improve = time.monotonic()
    v = np.zeros_like(a)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # compute gradient of the exact max
        grad, cur_max = _gradient_max_convolution(a, fft_len_self, fft_len_grad)

        eta = eta0 / np.sqrt(it)
        v = mu * v - eta * grad
        a = a + v
        a = _project_simplex(a)

        cur_score = evaluate_sequence(a.tolist())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - (deadline - (budget_s - 5.0))
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:6d} new best {best_score:.6f}")

        # stagnation handling
        if time.monotonic() - last_improve > stagnation_limit:
            # apply a modest log‑normal jitter and reset momentum
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = best * noise
            a = _project_simplex(a)
            v = np.zeros_like(a)
            last_improve = time.monotonic()
            elapsed = last_improve - (deadline - (budget_s - 5.0))
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    return best, best_score

def _generate_parametric_candidates(n: int) -> List[np.ndarray]:
    """
    Returns a small list of candidate vectors generated from simple
    parametric families (Gaussian, power‑law, triangular).
    """
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    candidates = []

    # Gaussian – various std‑dev
    for sigma in (n * 0.12, n * 0.15, n * 0.20, n * 0.25):
        g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        candidates.append(g)

    # Power‑law – a_i ∝ (i+1)^{-p}
    for p in (0.5, 0.8, 1.0, 1.2):
        pl = 1.0 / (x + 1) ** p
        candidates.append(pl)

    # Triangular
    tri = np.minimum(x, n - 1 - x) + 1.0
    candidates.append(tri)

    # Uniform (for completeness)
    candidates.append(np.ones_like(x))

    # Normalise each candidate to sum = 1
    normed = []
    for v in candidates:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0:
            normed.append(v / s)
    return normed

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Entry point used by the evaluation harness.
    Searches for a step‑function with a low value of
    ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    # give a small safety buffer
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # pick a few lengths (including the previous construction if available)
    lengths = []
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
            lengths.append(prev.size)
    # add a range of lengths, powers of two up to about 8000
    for n in (1024, 2048, 3000, 3809, 4096, 6000, 8192):
        if n not in lengths:
            lengths.append(n)

    best_overall = None
    best_score_overall = float("inf")
    start_global = time.monotonic()

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # initialise a0 – either previous best of same size, interpolation,
        # or a parametric candidate that looks promising.
        if "height_sequence_1" in globals():
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq.size == n:
                a0 = seq.copy()
            else:
                x_old = np.linspace(0.0, 1.0, seq.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(x_new, x_old, seq)
        else:
            a0 = rng.dirichlet(np.ones(n)).astype(np.float64)

        # try a small set of parametric seeds and keep the best one
        parametric = _generate_parametric_candidates(n)
        best_param = a0
        best_param_score = evaluate_sequence(best_param.tolist())
        for cand in parametric:
            sc = evaluate_sequence(cand.tolist())
            if sc < best_param_score:
                best_param_score = sc
                best_param = cand

        # blend the Dirichlet / interpolated start with the best parametric
        a0 = 0.5 * a0 + 0.5 * best_param
        a0 = _project_simplex(a0)

        # run optimiser for this length
        vec, score = _optimise_length(a0, deadline, rng)

        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - start_global
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

    # fallback
    if best_overall is None:
        best_overall = np.array([1.0])

    # Clip to limits that evaluate_sequence enforces (0..1000)
    best_overall = np.clip(best_overall, 0.0, 1000.0)

    return [float(x) for x in best_overall]

"""
Optimisation of a non‑negative step function.

The goal is to minimise

    E(a) = 2·n·max(a*a) / (sum(a)²)

for a vector a≥0.  After projecting a onto the probability simplex
(∑a_i = 1) the denominator becomes 1 and the problem reduces to
minimising the L∞‑norm of the self‑convolution.

The algorithm proceeds in three stages for each candidate length n.

1. **Parametric warm‑start** – a handful of simple families (Gaussian,
   power‑law, triangular, uniform) are generated, normalised to sum = 1,
   and the best one (according to the exact objective) seeds the optimiser.

2. **Projected gradient descent with exact max‑gradient** – the gradient of
   `max(a*a)` is obtained analytically as
        g = 2·(p ⋆ rev(a))
   where `p` is the uniform distribution over the active peak indices
   (indices where the convolution attains its maximum).  A Nesterov‑style
   momentum update with a step size ηₜ = η₀/√t is applied, followed by an
   Euclidean projection onto the simplex.

3. **Stagnation handling** – if no improvement is observed for a while
   (default 30 s) a modest log‑normal jitter is applied to the current
   best vector and the optimiser is restarted.

All lengths are tried sequentially within the global time budget
(~1000 s).  The best sequence found over all lengths is returned as a
plain list of floats (already clipped to the limits enforced by
`evaluate_sequence`).

This approach uses the exact sub‑gradient of the true maximum (instead of
a soft‑max approximation) and a momentum scheme, which empirically
produces a finer descent near the optimum.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Result length is 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_max_convolution(
    a: np.ndarray, fft_len_self: int, fft_len_grad: int
) -> Tuple[np.ndarray, float]:
    """
    Compute the gradient of ``max(a*a)`` w.r.t. ``a`` and the current maximum.
    Returns (gradient, max_value).
    """
    conv = _self_conv(a, fft_len_self)
    max_val = conv.max()
    # indices where the maximum is attained (within a tiny tolerance)
    tol = max(1e-14, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]
    # probability vector supported on the active set
    p = np.zeros_like(conv)
    p[active] = 1.0 / active.size
    rev_a = a[::-1]
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * a.size - 2]
    grad = 2.0 * grad_full[a.size - 1 : a.size - 1 + a.size]
    return grad, max_val


def _score_from_vector(v: np.ndarray, n: int) -> float:
    """
    Exact objective value for a vector whose sum is (approximately) 1:
        score = 2·n·max(v*v)
    """
    fft_len = _next_pow2(2 * n - 1)
    conv = _self_conv(v, fft_len)
    return 2.0 * n * conv.max()


def _generate_parametric_candidates(n: int) -> List[np.ndarray]:
    """
    Produce a small suite of simple, analytically‑defined vectors.
    Each candidate is normalised to sum = 1.
    """
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    cand = []

    # Gaussian shapes
    for sigma in (n * 0.13, n * 0.16, n * 0.20, n * 0.24):
        g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        cand.append(g)

    # Power‑law decays a_i ∝ (i+1)^{-p}
    for p in (0.5, 0.8, 1.0, 1.2):
        pl = 1.0 / (x + 1) ** p
        cand.append(pl)

    # Symmetric triangular bump
    tri = np.minimum(x, n - 1 - x) + 1.0
    cand.append(tri)

    # Uniform (useful as a sanity check)
    cand.append(np.ones_like(x))

    # Normalise each candidate to a probability vector
    out = []
    for v in cand:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0:
            out.append(v / s)
    return out


# ----------------------------------------------------------------------
# Core optimiser for a single length
# ----------------------------------------------------------------------


def _optimise_one_length(
    a0: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
    start_time: float,
) -> Tuple[np.ndarray, float]:
    """
    Projected gradient descent with exact max‑gradient and momentum.
    Returns the best vector seen and its objective value.
    """
    a = a0.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # initial best
    best = a.copy()
    best_score = _score_from_vector(best, n)

    # optimiser state
    v = np.zeros_like(a)          # momentum term
    eta0 = 0.8                     # initial step‑size factor
    mu = 0.9                       # momentum coefficient
    stagnation_limit = 30.0       # seconds without improvement before a restart
    last_improve = time.monotonic()

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # gradient of the exact maximum
        grad, cur_max = _gradient_max_convolution(a, fft_len_self, fft_len_grad)

        # step‑size decays like 1/√t
        eta = eta0 / math.sqrt(it)

        # Nesterov‑style momentum update
        v = mu * v - eta * grad
        a = a + v

        # projection back onto the simplex (keeps the sum exactly 1)
        a = _project_simplex(a)

        # current objective value (fast – no extra FFT)
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(
                f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}"
            )

        # --------------------------------------------------------------
        # Stagnation handling – random jitter around the current best
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > stagnation_limit:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = best * noise
            a = _project_simplex(a)
            v = np.zeros_like(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point required by the challenge harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345, budget_s: float = 1000.0, **kwargs
) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function respects the given time budget and returns the best sequence
    found as a plain list of floats.
    """
    rng = np.random.default_rng(seed)

    # give a small safety margin for housekeeping
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # Lengths to explore: keep the previous construction (if any) and a few
    # useful sizes.  The list is ordered from small to large so that we spend
    # a little time on cheap candidates first.
    # ------------------------------------------------------------------
    lengths = []

    if "height_sequence_1" in globals():
        hs = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if hs.ndim == 1 and hs.size > 0 and np.isfinite(hs).all():
            lengths.append(hs.size)

    for n in (1024, 2048, 3000, 3809, 4096, 6000, 8192):
        if n not in lengths:
            lengths.append(n)

    best_overall = None
    best_score_overall = float("inf")

    # ------------------------------------------------------------------
    # Optimise each length in turn (sequentially; the loop respects the
    # global deadline).
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # ------------------------------------------------------------------
        # Initialisation – combine an interpolated previous solution (if any)
        # with a parametric warm‑start.
        # ------------------------------------------------------------------
        if "height_sequence_1" in globals():
            prev = np.asarray(
                globals()["height_sequence_1"], dtype=np.float64
            )
            if prev.size == n:
                a0 = prev.copy()
            else:
                # linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, prev.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(x_new, x_old, prev)
        else:
            a0 = rng.dirichlet(np.ones(n)).astype(np.float64)

        # ------------------------------------------------------------------
        # Explore a few hand‑crafted families and keep the best one as a seed.
        # ------------------------------------------------------------------
        parametric = _generate_parametric_candidates(n)
        best_param = a0
        best_param_score = _score_from_vector(best_param, n)

        for cand in parametric:
            sc = _score_from_vector(cand, n)
            if sc < best_param_score:
                best_param_score = sc
                best_param = cand

        # Blend the Dirichlet / interpolated start with the best parametric
        # candidate (helps escape a bad corner of the simplex).
        a0 = 0.5 * a0 + 0.5 * best_param
        a0 = _project_simplex(a0)

        # ------------------------------------------------------------------
        # Run the optimiser for the current length.
        # ------------------------------------------------------------------
        vec, score = _optimise_one_length(a0, deadline, rng, start_time)

        # ------------------------------------------------------------------
        # Keep the global best over all lengths.
        # ------------------------------------------------------------------
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}"
            )

    # ------------------------------------------------------------------
    # Fallback – should never happen but guards against an empty result.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # The evaluation harness caps values to [0,1000]; enforce the same.
    best_overall = np.clip(best_overall, 0.0, 1000.0)

    return [float(x) for x in best_overall]
