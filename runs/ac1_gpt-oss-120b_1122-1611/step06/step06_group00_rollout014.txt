def propose_candidate(...):
    rng = np.random.default_rng(seed)
    max_time = budget_s - 5
    deadline = time.monotonic() + max_time
    n = 6000
    a = np.full(n, 1.0/n)  # start uniform
    best_score = np.inf
    best = a.copy()
    step0 = 1.0
    step_end = 1e-4
    it = 0
    while time.monotonic() < deadline:
        # schedule step
        frac = (time.monotonic() - start) / (deadline - start)
        eta = step0 * (1-frac) + step_end * frac
        # compute conv via FFT
        ft_len = _next_pow2(2*n-1)
        a_fft = np.fft.rfft(a, ft_len)
        c = np.fft.irfft(a_fft * a_fft, ft_len)[:2*n-1]
        max_c = c.max()
        score = 2.0 * n * max_c
        if score < best_score:
            best_score = score
            best = a.copy()
            print("new best", score)
        # compute gradient of L2 error
        M = 1.0/(2*n-1) * (a.sum()**2)  # sum a=1, but keep general
        residual = c - M
        # correlation with a reversed
        ft_len2 = _next_pow2(3*n - 2)
        r_fft = np.fft.rfft(residual, ft_len2)
        a_rev_fft = np.fft.rfft(a[::-1], ft_len2)
        conv_full = np.fft.irfft(r_fft * a_rev_fft, ft_len2)
        grad = 4.0 * conv_full[n-1:n-1+n]  # length n
        # exponentiated gradient step
        factor = np.exp(-eta * grad)
        a = a * factor
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a = a / a_sum
        it += 1
        if it % 5000 == 0:
            # periodic full recomputation (just done)
            pass
        # occasional random perturb
        if rng.random() < 0.001:
            a = _random_block_perturb(a, rng)
    # after loop, return best
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # Determine n
    n = 5000  # moderate size
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full_like(a, 1.0 / a.size)
            n = a.size
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Precompute lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # Initialize tracking
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.65 * total_time  # more time for gradient

    # Hyperparameters
    eta_start = 0.2
    eta_end = 1e-5
    temp_start = 0.05
    temp_end = 1e-7
    move_counter = 0
    RECOMPUTE_EVERY = 1000

    # Main loop
    while time.monotonic() < deadline:
        now = time.monotonic()
        # Phase decisions
        if now < phase1_end:
            # Gradient descent pass
            # schedule eta
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            # compute convolution and gradient
            c, grad = _gradient_l2_error(a, fft_len_self, fft_len_grad)

            a_tmp = a - eta * grad
            a = _project_simplex(a_tmp)
        else:

            # schedule temperature
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)
            # locate current peak
            k_max = int(c.argmax())
            # contributions
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

def _gradient_l2_error(a, fft_len_self, fft_len_grad):
    n = a.size
    # self convolution
    a_fft = np.fft.rfft(a, n=fft_len_self)
    c = np.fft.irfft(a_fft * a_fft, n=fft_len_self)[:2*n-1]
    # target mean
    sum_a = a.sum()
    M = (sum_a * sum_a) / (2 * n - 1)
    r = c - M
    # reverse a
    a_rev = a[::-1]
    # convolution of residual with reversed a
    r_fft = np.fft.rfft(r, n=fft_len_grad)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len_grad)
    full = np.fft.irfft(r_fft * a_rev_fft, n=fft_len_grad)
    grad = 4.0 * full[n-1:n-1+n]  # indices to match a
    return c, grad

def _project_simplex(v):
    # Projects v onto probability simplex (nonnegative components that sum to 1)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from a known solution if available,
   otherwise a uniform distribution on the simplex (sum = 1).

2. **Phase 1 – global L2‑error descent** (≈60 % of the budget)  
   The objective to be minimised is the squared L2 distance between the
   self‑convolution and its average value (which is the optimal lower
   bound).  Using the identity  

        ∂/∂a_i ∑_k (c_k−M)² = 4·∑_k (c_k−M)·a_{k−i},

   the gradient is obtained by a single extra FFT (convolution of the
   residual with the reversed vector).  An explicit gradient step
   `a ← a – η·grad` is performed and the result is projected back onto
   the probability simplex (non‑negative, sum = 1).

3. **Phase 2 – stochastic fine‑tuning** (≈40 % of the budget)  
   The current peak `k_max = argmax(c)` is located.
   A donor index is chosen proportionally to the contribution
   `a_i·a_{k_max‑i}` to this peak; a recipient is chosen favouring small
   entries.  A random fraction `δ ≤ 0.5·a_donor` is transferred and the
   convolution is updated **incrementally** in O(n).  Moves are accepted
   using a Metropolis rule with a linearly cooling temperature.

4. **Restart on stagnation** – if no improvement is seen for 25 s the
   current best solution is perturbed by a mild log‑normal jitter and
   optimisation continues.

All helper functions are top‑level (no closures) and the routine respects
the 1000 s budget (5 s safety margin).  The returned list is clipped to
the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_l2_error(
    a: np.ndarray, fft_len_self: int, fft_len_grad: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns the self‑convolution ``c`` and the gradient of the L2‑error
    (∑_k (c_k−M)²) with respect to ``a``.
    """
    n = a.size

    # ----- self‑convolution -------------------------------------------------
    a_fft = np.fft.rfft(a, n=fft_len_self)
    c = np.fft.irfft(a_fft * a_fft, n=fft_len_self)[: 2 * n - 1]

    # ----- target (average) value -----------------------------------------
    M = (a.sum() ** 2) / (2 * n - 1)  # same as np.mean(c) for any a
    resid = c - M

    # ----- gradient = 4·(resid * a_rev) ------------------------------------
    a_rev = a[::-1]
    r_fft = np.fft.rfft(resid, n=fft_len_grad)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len_grad)
    full = np.fft.irfft(r_fft * a_rev_fft, n=fft_len_grad)

    # The slice [n‑1 : n‑1+n] corresponds to indices i of the original a.
    grad = 4.0 * full[n - 1 : n - 1 + n]
    return c, grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # subtract donor contribution, add recipient contribution
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a simplex projection.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise – use a known construction if it exists, otherwise uniform
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            a = _project_simplex(a)
        else:
            n = 5000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 5000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    # pre‑computed FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the current state
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time
    # allocate 60 % of the time to the L2‑error descent
    phase1_end = start_time + 0.60 * total_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Phase‑1 (gradient) schedule
    eta_start = 0.2
    eta_end = 1e-5

    # Phase‑2 (stochastic) schedule
    temp_start = 0.05
    temp_end = 1e-7

    move_counter = 0
    RECOMPUTE_EVERY = 1000  # full FFT recomputation interval

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase 1 – global L2‑error gradient descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedule for step size
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # compute convolution and gradient in one go
            c, grad = _gradient_l2_error(a, fft_len_self, fft_len_grad)

            # projected gradient step (maintain sum = 1, non‑negative)
            a_tmp = a - eta * grad
            a = _project_simplex(a_tmp)

        else:
            # --------------------------------------------------------------
            # Phase 2 – stochastic refinement (simulated annealing)
            # --------------------------------------------------------------
            # linear temperature cooling
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # locate the current peak of the convolution
            k_max = int(c.argmax())

            # contribution of each index to the peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # choose donor proportional to its contribution
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                continue  # nothing to move

            # choose recipient – bias towards small entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            rw_sum = recv_weights.sum()
            if rw_sum > 0.0:
                recv_weights /= rw_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # amount of mass to move
            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            # apply the move (incrementally update the convolution)
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # full recomputation every now and then (drift correction)
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter (to escape plateaus)
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 25.0:
            # re‑initialise from the best with moderate log‑normal noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart with jitter (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
