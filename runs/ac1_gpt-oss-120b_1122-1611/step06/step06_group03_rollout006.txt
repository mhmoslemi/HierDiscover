"""
Propose a step‑function minimizing

    score = 2·n·max(a∗a) / (∑a)²

The algorithm works in three stages:

1.  **Initialisation** – start from the analytically optimal “√‑difference’’
    sequence (a_i = √(i+1)‑√i) or a previously stored construction.
2.  **Exploratory simulated‑annealing (SA)** – a fast Metropolis search with
    several cheap moves (pair‑transfer, block‑scaling, swap, jitter, shift).
    Only ~30 % of the time‑budget is spent here; the aim is to escape bad
    basins and find a promising region.
3.  **Deterministic polishing** – exponentiated‑gradient (mirror‑descent)
    on a smooth approximation of the L∞‑objective.  The soft‑max weight
    concentration (parameter ``beta``) is increased while the step‑size
    (parameter ``eta``) is decreased, together with momentum, to drive the
    maximal convolution value down.  The remaining ~70 % of the budget is
    devoted to this phase, which usually yields the strongest improvement.

All vectors are kept on the probability simplex (∑a = 1) and are finally
clipped to the limits imposed by ``evaluate_sequence`` before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length sufficient for the linear convolution of two length‑n vectors."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real‑to‑real FFT (returns the full 2·n‑1 array)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """
    Fast evaluation of the objective for a vector already normalised
    (∑a = 1).  Returns the value that ``evaluate_sequence`` would produce
    (ignoring the final clipping of a).
    """
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic start
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _maybe_use_global() -> np.ndarray | None:
    """If a previous construction exists in the global namespace, return it normalised."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:

    return None


# ----------------------------------------------------------------------
# Gradient of a weighted sum of convolution entries
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k with respect to a.
    Implements the correlation a ⋆ w (see the original code).
    """
    n = a.size
    L = w.size
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Polishing / deterministic descent
# ----------------------------------------------------------------------
def _polish(
    a: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    start_score: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on a smoothed max of the convolution.
    The schedule slowly raises the soft‑max concentration (beta) while
    lowering the learning rate (eta).  Momentum is used to stabilise updates.
    """
    best = a.copy()
    best_score = start_score
    grad_ma = np.zeros_like(a)          # momentum accumulator

    # schedule parameters (tuned empirically)
    eta_start = 0.30
    eta_end = 0.003
    beta_start = 20.0
    beta_end = 5000.0
    momentum = 0.85
    max_active = 30                     # keep the active set reasonably small
    grad_clip = 10.0                    # prevent overflow in exp(-eta*grad)

    start_time = time.monotonic()
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1) compute convolution and current score
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # keep global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            now = time.monotonic()
            print(f"[{now - start_time:6.2f}s] polish: new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2) schedule for eta and beta
        # --------------------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # --------------------------------------------------------------
        # 3) build the soft‑max weight vector on the active indices
        # --------------------------------------------------------------
        # indices whose convolution value is within 0.1 % of the current max
        tol = cur_max * 0.999
        active = np.where(conv >= tol)[0]
        if active.size > max_active:
            # keep the largest ones only
            active = np.argsort(conv)[-max_active:]

        # soft‑max on the active set
        if active.size == 0:          # fallback – should never happen
            w = np.exp(beta * (conv - cur_max))
            w /= w.sum()
        else:
            w_raw = np.exp(beta * (conv[active] - cur_max))
            w_active = w_raw / w_raw.sum()
            w = np.zeros_like(conv)
            w[active] = w_active

        # --------------------------------------------------------------
        # 4) gradient of Σ w_k·(a∗a)_k
        # --------------------------------------------------------------
        grad = _gradient_correlation(a, w)

        # momentum smoothing
        grad_ma = momentum * grad_ma + (1.0 - momentum) * grad

        # --------------------------------------------------------------
        # 5) exponentiated‑gradient update (project onto the simplex)
        # --------------------------------------------------------------
        # clip the gradient to avoid numerical overflow of exp()
        grad_clip_arr = np.clip(grad_ma, -grad_clip, grad_clip)
        factor = np.exp(-eta * grad_clip_arr)
        a = a * factor
        # safeguard against accidental zeros
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # quick early out when we are within the final 0.1 s
        if time.monotonic() > deadline - 0.1:
            break

    return best, best_score

# Simulated annealing utilities

def _move_pair_transfer(a: np.ndarray, rng: np.random.Generator, max_frac: float) -> np.ndarray:
    """Transfer a random fraction of mass from i to j."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] < 1e-15:
        return a
    delta = rng.random() * a[i] * max_frac
    a[i] -= delta
    a[j] += delta
    return a


def _move_block_scale(a: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    a[start:start + width] *= factor
    return a


def _move_global_jitter(a: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    """Multiply every entry by an independent log‑normal factor."""
    jitter = np.exp(rng.normal(0.0, sigma, size=a.shape))
    a *= jitter
    return a


def _move_swap(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap the values of two random positions."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i != j:
        a[i], a[j] = a[j], a[i]
    return a


def _move_shift(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cyclically shift the whole vector (non‑circular convolution)."""
    n = a.size
    shift = rng.integers(1, n)
    return np.roll(a, shift)


def _normalise(a: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.size)
    return a / s

    best_score = cur_score

    start_time = time.monotonic()
    total_budget = float(budget_s)

    # --------------------------------------------------------------
    # Simulated annealing – use about 30 % of the time budget
    # --------------------------------------------------------------
    sa_fraction = 0.30
    sa_deadline = start_time + total_budget * sa_fraction

    # SA hyper‑parameters (tuned empirically)
    T0 = best_score * 0.5          # initial temperature
    Tmin = 1e-9
    sigma_block = 0.35
    sigma_jitter = 0.20
    max_pair_frac = 0.6

    move_fns = [
        (_move_pair_transfer, 0.45),
        (_move_block_scale,     0.30),
        (_move_swap,            0.10),
        (_move_global_jitter,   0.10),
        (_move_shift,           0.05),
    ]
    moves, probs = zip(*move_fns)
    probs = np.array(probs, dtype=np.float64)
    probs /= probs.sum()

    while time.monotonic() < sa_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = sa_deadline - start_time
        # exponential cooling
        T = T0 * (Tmin / T0) ** (elapsed / total)

        # ----------------------------------------------------------
        # Propose a random move
        # ----------------------------------------------------------
        a_new = cur_a.copy()
        move_idx = rng.choice(len(moves), p=probs)
        move_fn = moves[move_idx]

        if move_fn is _move_pair_transfer:
            a_new = _move_pair_transfer(a_new, rng, max_pair_frac)
        elif move_fn is _move_block_scale:
            a_new = _move_block_scale(a_new, rng, sigma_block)
        elif move_fn is _move_global_jitter:
            a_new = _move_global_jitter(a_new, rng, sigma_jitter)
        elif move_fn is _move_swap:
            a_new = _move_swap(a_new, rng)
        elif move_fn is _move_shift:
            a_new = _move_shift(a_new, rng)

        a_new = _normalise(a_new)

        # ----------------------------------------------------------
        # Evaluate the move (fast internal score)
        # ----------------------------------------------------------
        new_score = _compute_score(a_new, fft_len, n)

        # ----------------------------------------------------------
        # Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp((cur_score - new_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = cur_a.copy()
                now = time.monotonic()
                print(f"[{now - start_time:6.2f}s] SA new best = {best_score:.6f}")

        # safety – leave a tiny margin for polishing
        if time.monotonic() > sa_deadline - 0.05:
            break

    # --------------------------------------------------------------
    # Polishing – deterministic mirror‑descent (the bulk of the budget)
    # --------------------------------------------------------------
    remaining = start_time + total_budget - time.monotonic()
    polish_deadline = time.monotonic() + max(0.0, remaining - 0.5)  # keep 0.5 s safety
    if polish_deadline > time.monotonic():
        best, best_score = _polish(best, n, fft_len, polish_deadline, best_score)

    # --------------------------------------------------------------
    # Final greedy hill‑climb (accept only improvements) – tiny budget
    # --------------------------------------------------------------
    final_deadline = start_time + total_budget - 0.1
    a = best.copy()
    a_score = best_score
    while time.monotonic() < final_deadline:
        cand = a.copy()
        # simple pairwise transfer only (deterministic descent)
        i, j = rng.integers(0, n, size=2)
        if i != j and cand[i] > 1e-15:
            delta = rng.random() * cand[i] * 0.2
            cand[i] -= delta
            cand[j] += delta
            cand = _normalise(cand)
            cand_score = _compute_score(cand, fft_len, n)
            if cand_score < a_score - 1e-12:
                a = cand
                a_score = cand_score
                now = time.monotonic()
                print(f"[{now - start_time:6.2f}s] final climb new best = {a_score:.6f}")

    best = a
    best_score = a_score

    # --------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)

    # optional logging of the true score (if evaluate_sequence is available)
    if 'evaluate_sequence' in globals():
        true_score = evaluate_sequence(best.tolist())
        print(f"True evaluate_sequence score: {true_score:.6f}")

    return [float(v) for v in best]
