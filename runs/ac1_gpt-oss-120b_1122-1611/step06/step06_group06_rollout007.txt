def propose_candidate(seed=..., budget_s=1000.0, n=3800):
    rng = np.random.default_rng(seed)
    # initial candidate from families
    # pick best
    best = cur = best_initial
    best_score = evaluate_score(cur)

    deadline = start + budget_s
    # parameters
    beta = 0.1
    beta_end = 10000.0
    eta = 0.5
    eta_end = 0.001
    momentum = 0.9
    grad_prev = np.zeros_like(cur)
    
    while time.monotonic() < deadline:
        # compute conv, max
        conv = _conv_self(cur, fft_len)
        max_val = conv.max()
        score = 2 * n * max_val
        if score < best_score - 1e-12:
            best_score = score
            best = cur.copy()
            print...
        # schedule
        elapsed = time.monotonic() - start
        frac = elapsed / budget_s
        beta = beta_start + (beta_end - beta_start) * frac
        eta = eta_start * (1-frac) + eta_end * frac
        
        # compute subgradient via active set: indices where conv >= max*(1 - eps)
        eps = 0.001 # 0.1% maybe smaller
        active = np.where(conv >= max_val * (1 - eps))[0]
        # compute weights via softmax
        w_raw = np.exp(beta * (conv[active] - max_val))
        w = w_raw / w_raw.sum()
        # gradient: for each active index k, for each i in overlap
        grad = np.zeros_like(cur)
        for k, weight in zip(active, w):
            lo = max(0, k - (n - 1))
            hi = min(k, n-1)
            i_range = np.arange(lo, hi+1, dtype=int)
            grad[i_range] += weight * 2.0 * cur[k - i_range]
        # (Alternatively use convolution method _gradient_correlation) but it's fine.
        
        # momentum
        grad = momentum * grad_prev + (1-momentum) * grad
        grad_prev = grad
        
        # update using exponentiated gradient (mirror)
        cur = cur * np.exp(-eta * grad)
        cur = np.maximum(cur, 0.0)
        s = cur.sum()
        cur = cur / s
        
        # occasional guided transfer: every 1000 iterations
        if iteration % 500 == 0:
            # compute gradient directions: pick i where grad large positive, j where small.
            i = np.argmax(grad)
            j = np.argmin(grad)
            if cur[i] > 0:
                delta = min(cur[i] * 0.01, cur[i])
                cur[i] -= delta
                cur[j] += delta
                cur = cur / cur.sum()
        iteration += 1

"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Rich initial pool** – generate many analytic families:
   * classic √‑difference,
   * √‑difference with an offset (shifted square‑root),
   * power‑difference with exponent p∈[0.2,2] and optional shift,
   * exponential families,
   * random Dirichlet draws,
   * the previous best sequence (if present).
   All candidates are normalised to sum = 1 and scored with a fast FFT
   convolution; the best one is kept as the starting point.

2. **Mirror‑descent on the simplex** – minimises the smooth
   log‑sum‑exp surrogate of the ‑∞‑norm of the self‑convolution.
   * The convolution and its maximum are obtained by an FFT.
   * The active set consists of the indices whose value is within
     ``rel_tol`` of the current maximum (default 0.9995).  A soft‑max
     over this set yields a weight vector w, and the (exact) sub‑gradient
     of the surrogate is obtained by a single FFT‑based correlation.
   * An exponentiated‑gradient (mirror‑descent) update is performed,
     with a linear schedule for the step‑size (η) and the soft‑max
     temperature (β).  Momentum stabilises the iterates.
   * Very occasional random block perturbations and pairwise mass
     transfers keep the search from stagnating.

3. **Gradient‑guided simulated annealing** – the remaining time is spent
   on a short SA phase that uses directed mass‑transfer moves
   (from heavy to light positions) together with a classic Metropolis
   acceptance rule.  A gentle smoothing move is performed every few
   thousand iterations.

4. The best sequence seen during the whole run is clipped to the limits
   required by ``evaluate_sequence`` and returned.

The routine respects the 1000 s budget, prints intermediate progress,
and works for any length ``n`` (default 3800).  It is deliberately
stateless – all helpers are top‑level functions.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length n (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution via FFT (exact, length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """Fast objective for a probability vector (sum = 1):
    2·n·max(a∗a)."""
    return 2.0 * n * _conv_self(a, fft_len).max()


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_k w_k·(a∗a)_k = 2·a⋆w (correlation) via FFT."""
    n = a.size
    L = w.size                     # = 2·n‑1
    # length for the correlation FFT (next power of two)
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _sqrt_diff(n: int, shift: float = 0.0) -> np.ndarray:
    """a_i = sqrt(i+shift+1) - sqrt(i+shift)  (difference of sqrt)."""
    i = np.arange(n, dtype=np.float64)
    return np.sqrt(i + shift + 1.0) - np.sqrt(i + shift)


def _power_diff(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a_i = (i+shift+1)^(-p) - (i+shift+2)^(-p)."""
    i = np.arange(n, dtype=np.float64)
    return (i + shift + 1.0) ** (-p) - (i + shift + 2.0) ** (-p)


def _exponential(n: int, lam: float, shift: float = 0.0) -> np.ndarray:
    """a_i = (1‑exp(-lam))·exp(-lam·(i+shift))."""
    i = np.arange(n, dtype=np.float64)
    return (1.0 - np.exp(-lam)) * np.exp(-lam * (i + shift))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence().  Returns the best sequence found before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of starting points
    # ------------------------------------------------------------------
    candidates = []

    # a) classic √‑difference
    candidates.append(_normalise(_sqrt_diff(n)))

    # b) √‑difference with a few shifts
    for sh in (0.5, 1.0, 2.0, 5.0):
        candidates.append(_normalise(_sqrt_diff(n, shift=sh)))

    # c) Power‑difference families (grid over p and shift)
    for p in np.linspace(0.2, 2.0, 12):
        for sh in (0.0, 0.5, 1.0, 2.0):
            candidates.append(_normalise(_power_diff(n, p, shift=sh)))

    # d) Exponential families (grid over λ and shift)
    for lam in np.linspace(0.0005, 0.02, 12):
        for sh in (0.0, 1.0, 3.0):
            candidates.append(_normalise(_exponential(n, lam, shift=sh)))

    # e) Random Dirichlet draws
    for _ in range(80):
        alpha = rng.uniform(0.1, 2.0)
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(_normalise(a))

    # f) Previous construction, if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                candidates.append(_normalise(prev))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) Pick the best candidate from the pool (fast scoring)
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    for a in candidates:
        sc = _score_of(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    print(f"[init] best score from pool = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 3) Mirror‑descent refinement (exponentiated‑gradient)
    # ------------------------------------------------------------------
    cur = best.copy()
    grad_prev = np.zeros_like(cur)

    start_time = time.monotonic()
    total_budget = float(budget_s)

    # Reserve a small fraction (≈5 %) for the final SA phase
    md_end = start_time + 0.95 * total_budget

    # schedules
    eta_start, eta_end = 0.8, 0.001          # step‑size
    beta_start, beta_end = 0.1, 20000.0     # soft‑max temperature
    momentum = 0.9

    rel_tol = 0.9995          # active set tolerance
    max_active = 30
    prob_block = 0.025        # occasional block perturbation
    prob_pair = 0.07          # occasional pairwise transfer

    iteration = 0
    while time.monotonic() < md_end:
        # self‑convolution and current score
        conv = _conv_self(cur, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # keep global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.9f}")

        # schedule parameters (linear in elapsed fraction)
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_budget
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # active set (indices within rel_tol of the maximum)
        active = np.where(conv >= cur_max * rel_tol)[0]
        if active.size > max_active:
            # keep the largest values only
            idx_sorted = np.argsort(conv)[-max_active:]
            active = idx_sorted

        # build weight vector w (length 2·n‑1)
        if active.size > 0:
            w_raw = np.exp(beta * (conv[active] - cur_max))
            w_sum = w_raw.sum()
            w_vals = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)
            w_vec = np.zeros_like(conv)
            w_vec[active] = w_vals
            grad = _gradient_correlation(cur, w_vec)
        else:
            # fallback: full soft‑max
            w_raw = np.exp(beta * (conv - cur_max))
            w_vec = w_raw / w_raw.sum()
            grad = _gradient_correlation(cur, w_vec)

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient (mirror) update
        factor = np.exp(-eta * grad)
        cur = cur * factor
        cur = np.clip(cur, 0.0, None)
        cur /= cur.sum()

        # occasional random block perturbation
        if rng.random() < prob_block:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            blk_factor = np.exp(rng.normal(0.0, 0.2))
            cur[start:start + width] *= blk_factor
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()

        # occasional pairwise mass transfer
        if rng.random() < prob_pair:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur[i] > 1e-14:
                delta = rng.random() * cur[i] * 0.5
                cur[i] -= delta
                cur[j] += delta
                cur = np.clip(cur, 0.0, None)
                cur /= cur.sum()

        iteration += 1

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning (remaining budget)
    # ------------------------------------------------------------------
    sa_deadline = start_time + total_budget
    cur_sa = best.copy()
    cur_sa_score = best_score
    iteration = 0

    # temperature schedule (exponential cooling)
    T_start = 5e-3
    T_end = 1e-7

    # pre‑compute sorted index groups for directed moves
    heavy_frac = 0.05
    light_frac = 0.05
    n_heavy = max(2, int(heavy_frac * n))
    n_light = max(2, int(light_frac * n))

    while time.monotonic() < sa_deadline:
        now = time.monotonic()
        elapsed_sa = now - md_end
        total_sa = sa_deadline - md_end
        frac_sa = elapsed_sa / total_sa if total_sa > 0 else 0.0
        T = T_start * (1.0 - frac_sa) + T_end * frac_sa

        # directed pair move: from heavy to light
        idx_sorted = np.argsort(cur_sa)
        low_idx = idx_sorted[:n_light]
        high_idx = idx_sorted[-n_light:]
        i = rng.choice(high_idx)
        j = rng.choice(low_idx)

        # propose a move (transfer a fraction of mass from i to j)
        delta = rng.random() * cur_sa[i] * 0.4   # at most 40 % of i
        if delta <= 0.0:
            continue
        new_a = cur_sa.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _score_of(new_a, n, fft_len)

        improve = new_score < cur_sa_score
        if improve or rng.random() < np.exp(-(new_score - cur_sa_score) / max(T, 1e-12)):
            cur_sa = new_a
            cur_sa_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = new_a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        # occasional gentle smoothing (helps to flatten tiny spikes)
        if iteration % 4000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = _normalise(smoothed)
            cur_sa = smoothed
            cur_sa_score = _score_of(cur_sa, n, fft_len)

        iteration += 1

    # ------------------------------------------------------------------
    # 5) Return the best sequence (respect clipping constraints)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    print(f"[final] best score = {best_score:.9f}  (length={n})")
    return [float(v) for v in best]
