"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
* Use several independent starts (uniform, √‑difference, Dirichlet samples,
  optional previously found solution).
* For each start run a pure sub‑gradient descent on the true objective
  ``score = 2·n·max(conv)``.
  – The sub‑gradient of the max‑convolution is obtained by a weighted
    correlation of the current vector with a mask that is 1 on the
    (almost) maximal convolution indices.
  – Updates are performed with an *exponentiated‑gradient* (multiplicative)
    rule which automatically stays on the probability simplex.
  – Step size η decays as η₀/√t.
* Occasionally add a mild log‑normal jitter to escape shallow plateaus.
* Keep the best vector (lowest score) seen over the whole budget and
  return it after clipping to the limits of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length for a linear convolution of size n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return next power‑of‑two ≥ 2·n‑1 for FFT‑based convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of  Σ_k w_k (a∗a)_k  (correlation).
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w  (correlation).
    ``a`` is length n, ``w`` length 2·n‑1.
    """
    n = a.size
    L = w.size  # = 2·n‑1
    # Length required for full linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reversed a for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Simple exponentiated‑gradient (multiplicative) update on the simplex.
# ----------------------------------------------------------------------
def _exp_gradient_step(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """a ← a * exp(‑η·grad)  then re‑normalise to sum 1 (stays non‑negative)."""
    # avoid under‑flow / overflow when grad is large
    factor = np.exp(-eta * grad)
    a = a * factor
    s = a.sum()
    if s <= 0.0 or not np.isfinite(s):
        # fallback – uniform distribution
        a = np.full_like(a, 1.0 / a.size)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Score (the quantity returned by evaluate_sequence, up to the clipping).
# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the task.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget (seconds).  The routine stops a little before the
        deadline to guarantee a return.
    n : int, optional
        Length of the sequence.  Default 3800 (a size that has proven
        successful in earlier experiments).

    Returns
    -------
    list[float]
        The best sequence found (values already clipped to [0, 1000]).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Build a small pool of seeds (deterministic + random).
    # --------------------------------------------------------------
    init_vectors = []

    # 1) Analytic √‑difference start (classic good baseline)
    i = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(i + 1.0) - np.sqrt(i)
    a_sqrt /= a_sqrt.sum()
    init_vectors.append(a_sqrt)

    # 2) Uniform distribution (worst baseline, but cheap)
    init_vectors.append(np.full(n, 1.0 / n))

    # 3) Random Dirichlet draws with a few concentration parameters.
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a_rand = rng.dirichlet(np.full(n, alpha))
        init_vectors.append(a_rand)

    # 4) If a previously discovered sequence exists, try it.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                    init_vectors.append(cand)
        except Exception:
            pass  # ignore malformed globals

    # --------------------------------------------------------------
    # Optimisation state
    # --------------------------------------------------------------
    best = None
    best_score = np.inf

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # Configuration of the optimisation loop
    eta0 = 0.5                 # initial step‑size for EG
    jitter_prob = 0.018        # probability of random multiplicative jitter
    jitter_sigma = 0.12        # log‑normal std. dev. for jitter
    max_iter_per_start = 20000

    # --------------------------------------------------------------
    # Run a short optimisation from each initial vector.
    # --------------------------------------------------------------
    for start_id, a in enumerate(init_vectors):
        # Make a fresh copy, guarantee exact sum 1.
        a = np.copy(a)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        t = 0
        while True:
            now = time.monotonic()
            if now >= deadline or t >= max_iter_per_start:
                break
            t += 1

            # ------------------------------------------------------
            # 1) Compute convolution and current score.
            # ------------------------------------------------------
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            # ------------------------------------------------------
            # 2) Record best solution seen so far.
            # ------------------------------------------------------
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

            # ------------------------------------------------------
            # 3) Sub‑gradient of the max‑convolution.
            # ------------------------------------------------------
            # Identify all indices that attain (or are extremely close to) the maximum.
            # A tiny tolerance avoids missing ties due to floating‑point noise.
            tol = 1e-14
            mask = conv >= cur_max - tol
            w = mask.astype(np.float64)
            w_sum = w.sum()
            if w_sum == 0.0:                     # fallback – single maximal index
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
                w_sum = 1.0
            w /= w_sum

            grad = _gradient_correlation(a, w)    # shape (n,)

            # ------------------------------------------------------
            # 4) Exponentiated‑gradient step with a √‑decaying stepsize.
            # ------------------------------------------------------
            eta = eta0 / np.sqrt(t)
            a = _exp_gradient_step(a, grad, eta)

            # ------------------------------------------------------
            # 5) Random jitter occasionally (helps escaping flat regions).
            # ------------------------------------------------------
            if rng.random() < jitter_prob:
                factor = np.exp(rng.normal(0.0, jitter_sigma, size=n))
                a = a * factor
                s = a.sum()
                if s > 0.0:
                    a /= s
                else:
                    a = np.full_like(a, 1.0 / n)

    # --------------------------------------------------------------
    # If nothing succeeded (should never happen), fall back to uniform.
    # --------------------------------------------------------------
    if best is None:
        best = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # Apply the clipping required by evaluate_sequence and return.
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
