def _adam_optimize(a_init, deadline, rng):
    a = a_init.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    # Check for both for gradient
    # Adam parameters
    lr = 0.1
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best = a.copy()
    best_score = float('inf')
    t = 0
    while time.monotonic() < deadline:
        t += 1
        conv = _self_conv(a, fft_len_self)
        max_val = conv.max()
        cur_score = 2 * n * max_val
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
        # gradient
        k = int(conv.argmax())
        g = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx_i = np.arange(i_min, i_max + 1)
        idx_j = k - idx_i
        g[idx_i] = 2.0 * a[idx_j]
        # Adam update (additive)
        m = beta1 * m + (1 - beta1) * g
        v = beta2 * v + (1 - beta2) * (g * g)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # step decay maybe
        if t % 5000 == 0:
            lr *= 0.9
        # project onto simplex
        a = _project_simplex(a)
        # occasional noise for stagnation
        # track improvements maybe
    return best, best_score

"""
Propose a step‑function with a lower evaluation score.

Algorithm
---------
1. *Coarse parametric search* – for each candidate length n we
   generate a modest set of structured sequences:
   – uniform,
   – Dirichlet draws,
   – geometric (a_i ∝ r^i),
   – power‑law (a_i ∝ (i+1)^‑p),
   – linearly decreasing with variable support length.
   Every sequence is projected onto the probability simplex (sum = 1)
   and its score 2·n·max(conv) is evaluated (fast FFT convolution).

2. *Local refinement* – the best candidate for the current length is
   improved by a projected‐Adam optimisation that works on the exact
   sub‑gradient of the max‑convolution.  Adam’s adaptive step size and
   momentum accelerate convergence; a simple stagnation‑restart
   (multiplicative log‑normal jitter) helps escape flat regions.

3. *Global bookkeeping* – the best sequence over all lengths is kept.
   The routine respects the 1000 s budget (5 s safety margin) and
   always returns a feasible non‑negative vector.

The implementation uses only NumPy and SciPy‑style code, no
closures or external files, and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parameterised families of step‑functions
# ----------------------------------------------------------------------


def _candidate_sequences(n: int, rng: np.random.Generator):
    """Yield a modest collection of candidate vectors for length n."""
    # 1) uniform
    yield np.full(n, 1.0 / n)

    # 2) a few Dirichlet draws (exploratory)
    for _ in range(5):
        yield rng.dirichlet(np.ones(n))

    # 3) geometric (a_i ∝ r^i)
    # choose eps = 1 - r in a range that gives r close to 1
    # (eps ≈ 2/n is the uniform bound)
    eps_vals = np.linspace(2.0 / n, 0.01, 80)
    for eps in eps_vals:
        r = 1.0 - eps
        if r <= 0.0:
            continue
        denom = 1.0 - r ** n
        if denom == 0.0:
            continue
        C = (1.0 - r) / denom
        seq = C * (r ** np.arange(n))
        yield seq

    # 4) power‑law (a_i ∝ (i+1)^‑p)
    p_vals = np.linspace(0.0, 5.0, 80)
    for p in p_vals:
        seq = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        s = seq.sum()
        if s > 0.0:
            yield seq / s

    # 5) linearly decreasing with variable support length L
    L_vals = np.arange(max(int(0.5 * n), 1), n + 1, max(int(0.05 * n), 1))
    for L in L_vals:
        c = 2.0 / (L + 1)
        seq = np.zeros(n, dtype=np.float64)
        idx = np.arange(L, dtype=np.float64)
        seq[:L] = c * (1.0 - idx / L)
        yield seq


# ----------------------------------------------------------------------
# Adam optimisation on the exact sub‑gradient of max(conv)
# ----------------------------------------------------------------------


def _adam_optimize(a_init: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Refine a vector by projected Adam on the exact max‑convolution."""
    a = a_init.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)

    # Adam hyper‑parameters
    lr = 0.12
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # bookkeeping
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * conv.max()
    last_improve = time.monotonic()
    start = last_improve
    it = 0

    while time.monotonic() < deadline:
        it += 1
        conv = _self_conv(a, fft_len_self)
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}")

        # exact sub‑gradient of the peak entry
        k = int(conv.argmax())
        g = np.zeros_like(a)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx_i = np.arange(i_min, i_max + 1)
        idx_j = k - idx_i
        g[idx_i] = 2.0 * a[idx_j]

        # Adam update (additive)
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # step‑size decay
        if it % 10000 == 0:
            lr *= 0.85

        # projection onto the simplex
        a = _project_simplex(a)

        # stagnation handling – jitter around the best point
        now = time.monotonic()
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = _project_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    return best, best_score

# Search a single length (coarse + refinement)

def _search_one_length(n: int,
                       deadline: float,
                       rng: np.random.Generator,
                       start_vec: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """Explore a few parametric families and then refine with Adam."""
    fft_len_self = _next_pow2(2 * n - 1)

    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Coarse parametric sweep
    # ------------------------------------------------------------------
    for cand in _candidate_sequences(n, rng):
        if time.monotonic() >= deadline:
            break
        # enforce feasibility
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s <= 0.0:

        cand = cand / s
        conv = _self_conv(cand, fft_len_self)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 2) Optional: use the provided start vector (e.g. previous best)
    # ------------------------------------------------------------------
    if start_vec is not None:
        seq = np.asarray(start_vec, dtype=np.float64)
        seq = np.clip(seq, 0.0, None)
        s = seq.sum()
        if s > 0.0:
            seq = seq / s
            conv = _self_conv(seq, fft_len_self)
            score = 2.0 * n * conv.max()
            if score < best_score:

                best_vec = seq.copy()

    # ------------------------------------------------------------------
    # 3) Local refinement with Adam (if time permits)
    # ------------------------------------------------------------------
    if best_vec is not None and time.monotonic() < deadline:

        if score_ref < best_score:
