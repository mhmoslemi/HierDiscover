def _proj_simplex(v):
    """Project onto probability simplex {x>=0, sum=1}."""
    # https://arxiv.org/abs/1101.6081
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # return uniform distribution
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = v - theta
    w[w < 0] = 0
    return w

"""
Propose a non‑negative step‑function that (approximately) minimises
evaluate_sequence().

Algorithm
---------
1.  Use the best known construction (``height_sequence_1`` if available)
    or a simple sqrt‑difference shape as a start.
2.  Perform a cheap parametric sweep over two one‑parameter families:
        a_i = (i+1)^γ – i^γ                (0<γ<1)
        a_i = (i+δ)^−p                     (p>0, δ≥0)
    The family member with the smallest objective (2·n·max(a∗a))
    becomes the new starting point.
3.  Refine the start with a projected sub‑gradient descent on the exact
    ℓ∞‑objective.  At each iteration:
        * compute the self‑convolution via an FFT,
        * find the set of indices where the convolution attains its
          maximum (active set),
        * form a weight vector that is uniform on the active set,
        * obtain a sub‑gradient of the max‑convolution using the same
          FFT tricks that the previous code used for the smooth surrogate,
        * take a diminishing step in the negative gradient direction,
        * project back onto the probability simplex (non‑negative,
          sum‑to‑one) by simple clamping and renormalisation.
    Simple‑projected gradient descent is fast, requires only a few
    FFTs per iteration and converges for convex objectives.
4.  To avoid stagnation we inject occasional Gaussian jitter and,
    if no improvement occurs for a configurable period, we restart the
    optimisation from the current best solution (plus a small random
    perturbation).
5.  The loop runs until the time budget (≈1000 s) is exhausted.  The
    best sequence found is returned (clipped to the limits enforced by
    ``evaluate_sequence``).

The implementation is completely self‑contained, uses only NumPy and
the two helper FFT routines defined below, and respects the 2‑CPU
limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation a ⋆ w (used for the sub‑gradient of Σ w_k·(a∗a)_k).
# Returns a vector of length len(a); the correlation is computed via
# FFT of a reversed and w (which has length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _grad_corr(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    # convolution of a_rev and w gives the correlation a ⋆ w
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: 3 * a.size - 2]
    start = a.size - 1
    return 2.0 * corr_full[start:start + a.size]


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex:  x_i ≥ 0, Σ x_i = 1.
# This is **not** the Euclidean projection, but it is cheap and works
# well together with the diminishing step‑size schedule used below.
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Score for a normalised sequence (sum == 1) – identical to what
# evaluate_sequence would return after the internal clipping.
# ----------------------------------------------------------------------
def _score(a: np.ndarray, fft_len_conv: int) -> float:
    conv = _conv_self(a, fft_len_conv)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Create a candidate from the “power‑difference’’ family:
#      a_i = (i+1)^γ – i^γ ,   0 < γ < 1.
# ----------------------------------------------------------------------
def _candidate_gamma(gamma: float, n: int) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + 1.0, gamma) - np.power(i, gamma)
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


# ----------------------------------------------------------------------
# Create a candidate from the “inverse‑power’’ family:
#      a_i = (i + δ)^(-p) ,   p > 0, δ ≥ 0.
# ----------------------------------------------------------------------
def _candidate_power(p: float, delta: float, n: int) -> np.ndarray | None:
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + delta + 1.0, -p)          # +1 to avoid div‑by‑0 at i=0 if delta=0
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s
