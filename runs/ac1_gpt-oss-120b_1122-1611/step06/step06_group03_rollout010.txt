"""
Simple pairwise‐transfer simulated annealing with fast incremental convolution.

The objective after normalising ∑aᵢ=1 is   score = 2·n·maxₖ (a∗a)ₖ.
Changing a single entry a[i] by –Δ and a[j] by +Δ changes the full
convolution only by the two shifted copies of the current vector:

    (a∗a)ₖ → (a∗a)ₖ – 2·Δ·aₖ₋ᵢ + 2·Δ·aₖ₋ⱼ  (linear part)
    + Δ² at k=2i, + Δ² at k=2j, –2·Δ² at k=i+j (quadratic correction).

Using this O(n) update we can evaluate millions of candidate moves per
second, allowing a long simulated‑annealing run (≈ 100 000 – 200 000
iterations) within the 1000 s budget.  The algorithm starts from the
analytically known √‑difference shape (or a few simple power‑law
alternatives) and performs a temperature‑controlled Metropolis walk.
After the SA phase a tiny greedy “polish’’ that always accepts
improvements on the current maximum convolution is applied.

The routine returns the best vector found (already on the probability
simplex, i.e. ∑aᵢ=1) – it satisfies all checks in ``evaluate_sequence``.
"""

import time
import numpy as np
from typing import List


def _initial_candidates(n: int) -> List[np.ndarray]:
    """Generate a handful of analytically motivated starting vectors."""
    i = np.arange(n, dtype=np.float64)

    # 1. √‑difference (the classic start)
    a_sqrt = np.sqrt(i + 1.0) - np.sqrt(i)
    a_sqrt /= a_sqrt.sum()
    cand = [a_sqrt]

    # 2. Power‑law families aᵢ ∝ (i+1)^{-p}
    for p in np.arange(0.35, 0.71, 0.05):
        a = (i + 1.0) ** (-p)
        a /= a.sum()
        cand.append(a)

    # 3. If a previous construction is present, reuse it (normalised)
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if old.ndim == 1 and old.size >= 2 and np.isfinite(old).all():
                old = np.clip(old, 0.0, None)
                s = old.sum()
                if s > 0.0:
                    cand.append(old / s)
        except Exception:
            pass

    return cand


def _best_initial_vector(candidates: List[np.ndarray]) -> np.ndarray:
    """Pick the candidate with the smallest max‑convolution (fast evaluation)."""
    best_vec = None
    best_max = np.inf
    for a in candidates:
        if a.size == 0:
            continue
        conv = np.convolve(a, a)
        cur_max = conv.max()
        if cur_max < best_max:
            best_max = cur_max
            best_vec = a.copy()
    # Fallback – should never happen
    if best_vec is None:
        best_vec = candidates[0].copy()
    return best_vec


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The algorithm consists of:

    * a fast initialisation (√‑difference or simple power‑law);
    * a long simulated‑annealing phase that performs cheap pairwise
      mass‑transfer moves, using an O(n) incremental update of the
      self‑convolution;
    * a short deterministic polishing loop that only accepts moves that
      strictly lower the current maximum convolution.

    The function respects the given time budget and returns the best
    vector (as a plain Python list) found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    # --------------------------------------------------------------
    # Initial vector
    # --------------------------------------------------------------
    candidates = _initial_candidates(n)
    a = _best_initial_vector(candidates)          # rows sum to 1
    conv = np.convolve(a, a)                      # full linear convolution (length 2n‑1)
    cur_max = conv.max()

    best_a = a.copy()
    best_max = cur_max

    # --------------------------------------------------------------
    # Simulated‑annealing configuration
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 0.5)   # keep a tiny safety margin

    total_time = deadline - start_time

    # Temperature starts relatively hot (≈ 50 % of current max) and decays
    T0 = cur_max * 0.5
    Tmin = cur_max * 1e-8

    # Maximal fraction of source mass that may be moved – shrinks over time
    max_frac_start = 0.5
    max_frac_end = 0.01

    # --------------------------------------------------------------
    # Main SA loop – cheap pairwise mass transfers
    # --------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        elapsed = time.time() - start_time
        frac = elapsed / total_time

        # exponential temperature decay
        T = T0 * (Tmin / T0) ** frac

        # linearly decreasing maximal transfer fraction
        max_frac = max_frac_start * (1.0 - frac) + max_frac_end * frac

        # choose a source index with non‑zero mass
        i = rng.integers(0, n)
        if a[i] < 1e-15:
            continue

        # choose a distinct destination index
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # propose moving a random fraction of a[i]
        delta = rng.random() * a[i] * max_frac
        if delta <= 0.0:
            continue

        # ----- incremental convolution update --------------------------------
        conv_new = conv.copy()
        # linear part – two shifted copies of the current vector
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a

        # quadratic correction (only three indices are affected)
        delta2 = delta * delta
        idx2i = 2 * i
        if idx2i < conv_new.size:
            conv_new[idx2i] += delta2
        idx2j = 2 * j
        if idx2j < conv_new.size:
            conv_new[idx2j] += delta2
        idxij = i + j
        if idxij < conv_new.size:
            conv_new[idxij] -= 2.0 * delta2

        new_max = conv_new.max()

        # ----- Metropolis acceptance -----------------------------------------
        accept = False
        if new_max < cur_max:
            accept = True
        else:
            # accept with probability exp(-(new‑old)/T)
            if T > 0.0:
                prob = np.exp(-(new_max - cur_max) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            # commit the move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_max = new_max

            # record the best ever seen
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                now = time.time()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best  max={best_max:.10f}  score={2.0 * n * best_max:.6f}")

        iteration += 1

        # occasional renormalisation to combat floating‑point drift (cheap)
        if iteration % 20000 == 0:
            s = a.sum()
            if s != 0.0:
                a /= s
                conv = np.convolve(a, a)
                cur_max = conv.max()

    # --------------------------------------------------------------
    # Deterministic polishing – only accept strict improvements
    # --------------------------------------------------------------
    polish_deadline = deadline
    while time.time() < polish_deadline:
        # index where the current convolution attains its maximum
        k = int(conv.argmax())
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)

        # contributions a[i]·a[k‑i] for all i that participate in convₖ
        idxs = np.arange(lo, hi + 1, dtype=int)
        contrib = a[idxs] * a[k - idxs]

        # source = index with largest contribution to the offending max
        i = idxs[contrib.argmax()]

        # pick a destination far from i (uniform random, avoiding i)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # a small transfer – the polishing phase works with tiny steps
        delta = min(a[i] * 0.1, a[i] * 0.02)
        if delta <= 0.0:
            break

        # incremental update (same formula as in SA)
        conv_new = conv.copy()
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        delta2 = delta * delta
        idx2i = 2 * i
        if idx2i < conv_new.size:
            conv_new[idx2i] += delta2
        idx2j = 2 * j
        if idx2j < conv_new.size:
            conv_new[idx2j] += delta2
        idxij = i + j
        if idxij < conv_new.size:
            conv_new[idxij] -= 2.0 * delta2

        new_max = conv_new.max()
        if new_max < cur_max - 1e-14:               # strict improvement
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_max = new_max
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_a = a.copy()
                now = time.time()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] polish improve max={best_max:.10f}  score={2.0 * n * best_max:.6f}")
        else:
            # no improvement – break if we cannot find any improving move
            # (a very small heuristic, we simply stop after a few failed attempts)
            break

    # --------------------------------------------------------------
    # Final post‑processing
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)   # required by evaluate_sequence
    return [float(v) for v in best_a]
