def _optimise_one_length(p0, deadline, rng):
    p = p0.copy()
    n = len(p)
    # Precompute FFT lengths
    fft_len = _next_pow2(2*n-1)
    fft_len_grad = _next_pow2(3*n - 2)
    # Adam parameters for phase 1
    beta1=0.9, beta2=0.999, eps=1e-8
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    t = 0
    # schedule: temperature beta from beta_start to beta_end, lr from lr_start to lr_end (or fixed small constant)
    # Use linear schedule for learning rate (lr_start ~ 1.0, lr_end ~0.001).
    # Phase1 fraction = 0.5 of time; rest for phase2.
    start = time.monotonic()
    total = deadline - start
    phase1_end = start + 0.5 * total

    best = p.copy()
    best_score = 2*n * np.max(np.convolve(p,p))
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        t+=1
        if now < phase1_end:
            # compute schedule
            frac = (now - start) / (phase1_end - start)
            lr = lr_start * (1-frac) + lr_end * frac
            beta_temp = beta_start + (beta_end - beta_start)*frac

            # compute conv (self convolution)
            conv = _self_conv(p, fft_len)
            max_c = np.max(conv)
            # compute softmax weights
            w = np.exp(beta_temp * (conv - max_c))
            w_sum = np.sum(w)
            w = w / w_sum
            # compute gradient = 2*(w * rev(p)) via FFT
            rev_p = p[::-1]
            fp = np.fft.rfft(w, n=fft_len_grad)
            fra = np.fft.rfft(rev_p, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[:3*n-2]
            grad = 2.0 * grad_full[n-1:n-1+n]
            # Adam update (descent)
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            p = p - step
        else:
            # phase 2: subgradient of max conv approximated
            conv = _self_conv(p, fft_len)
            kmax = int(conv.argmax())
            # subgradient g_i = 2*p_{kmax-i}
            g = np.zeros_like(p)
            i_min = max(0, kmax - (n-1))
            i_max = min(n-1, kmax)
            idx_i = np.arange(i_min, i_max+1)
            idx_j = kmax - idx_i
            g[idx_i] = 2.0 * p[idx_j]
            # Adam update
            lr = sub_lr_start / np.sqrt(t)  # decreasing
            m = beta1*m + (1-beta1)*g
            v = beta2*v + (1-beta2)*(g**2)
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            p = p - step

        # project onto simplex
        p = _project_simplex(p)
        # evaluate score
        conv = _self_conv(p, fft_len)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = p.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:.2f}s] n={n} it={t} new best {best_score:.6f}")

        # stagnation: if no improvement for X seconds, jitter
        if now - last_improve > 30.0:
            # jitter around best
            p = best.copy()
            # lognormal noise
            noise = np.exp(rng.normal(0.0, 0.15, size=n))
            p = p * noise
            p = _project_simplex(p)
            # reset Adam moments
            m = np.zeros_like(p)
            v = np.zeros_like(p)
            t = 0
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:.2f}s] n={n} restart (stagnation)")

    return best, best_score

"""
Propose a step‑function (probability vector) that minimizes
``evaluate_sequence``.
Algorithm
---------
* Initialise from the previous best construction if available,
  otherwise draw a Dirichlet sample.
* Optimise a *single* length (the length of the previous best,
  default = 3809) with a two‑phase Adam‑based projected gradient
  descent:
    – Phase 1 (≈ 50 % of the allotted time) – minimise a
      soft‑max approximation of the peak of the self‑convolution.
      Temperature β grows linearly from 0.5 to 5000 while a
      linearly decaying learning‑rate schedule is used.
    – Phase 2 – exact sub‑gradient descent on the true peak.
      Step‑size follows η₀/√t.
* After each update the iterate is projected onto the probability
  simplex (Euclidean projection).  Stagnation (>30 s without
  improvement) triggers a multiplicative log‑normal jitter around
  the best solution and resets the Adam moments.
* If time remains a second length (5000) is processed in the same
  way; the best overall vector is returned.
* The returned list is clipped to the limits enforced by
  ``evaluate_sequence`` (0 ≤ x ≤ 1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    O(n log n) algorithm (Wang & Carreira‑Perpinán, 2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Optimiser for a single length – Adam + simplex projection
# ----------------------------------------------------------------------


def _optimise_one_length_adam(p0: np.ndarray,
                              deadline: float,
                              rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Adam‑based projected gradient descent for a fixed length.
    Returns the best probability vector found and its evaluation score
    (which equals ``2 * n * max(conv(p,p))`` for a normalized vector).
    """
    # Normalise to the probability simplex
    p = p0.copy()
    p = np.maximum(p, 0.0)
    s = p.sum()
    p = p / s if s > 0 else np.full_like(p, 1.0 / p.size)

    n = p.size
    # FFT lengths (powers of two)
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(p)
    v = np.zeros_like(p)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # scheduling parameters
    start = time.monotonic()
    total = deadline - start
    phase1_end = start + 0.5 * total

    # learning‑rate schedule (phase 1)
    lr_start, lr_end = 1.0, 0.01
    beta_start, beta_end = 0.5, 5000.0

    # phase 2 sub‑gradient base step
    sub_lr0 = 0.25

    # bookkeeping
    best = p.copy()
    # exact evaluation (sum(p)=1) => objective = 2 * n * max(conv)
    conv = _self_conv(p, fft_len_self)
    best_score = 2.0 * n * conv.max()
    last_improve = start
    it = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        if now < phase1_end:
            # ------- Phase 1 – smoothed soft‑max gradient -------
            frac = (now - start) / (phase1_end - start)
            lr = lr_start * (1.0 - frac) + lr_end * frac
            beta_tmp = beta_start + (beta_end - beta_start) * frac

            conv = _self_conv(p, fft_len_self)
            max_c = conv.max()

            # soft‑max weights (stable)
            w = np.exp(beta_tmp * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w = w / w_sum

            # gradient = 2·(w * rev(p))  (convolution via FFT)
            rev_p = p[::-1]
            fp = np.fft.rfft(w, n=fft_len_grad)
            fra = np.fft.rfft(rev_p, n=fft_len_grad)
            grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]
        else:
            # ------- Phase 2 – exact sub‑gradient of the peak -------
            conv = _self_conv(p, fft_len_self)
            kmax = int(conv.argmax())

            # sub‑gradient g_i = 2·p_{kmax‑i} when index exists
            g = np.zeros_like(p)
            i_min = max(0, kmax - (n - 1))
            i_max = min(n - 1, kmax)
            idx_i = np.arange(i_min, i_max + 1, dtype=int)
            idx_j = kmax - idx_i
            g[idx_i] = 2.0 * p[idx_j]

            # decreasing step size
            lr = sub_lr0 / np.sqrt(it)
            grad = g

        # ----- Adam update (descent) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        p = p - step

        # ----- Projection onto the simplex -----
        p = _project_simplex(p)

        # ----- Evaluation -----
        conv = _self_conv(p, fft_len_self)
        cur_score = 2.0 * n * conv.max()

        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = p.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}")

        # ----- Stagnation handling -----
        if now - last_improve > 30.0:
            # jitter around the best solution
            p = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            p = p * noise
            p = _project_simplex(p)
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n} restart (stagnation)")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence (list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budget (leave a small safety margin)
    # ------------------------------------------------------------------
    safety = 5.0
    deadline = time.monotonic() + max(0.0, float(budget_s) - safety)

    # ------------------------------------------------------------------
    # Choose primary length (use previous construction if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        primary_len = seq0.size
    else:
        primary_len = 3809

    # ------------------------------------------------------------------
    # Helper to build an initial probability vector for a given length
    # ------------------------------------------------------------------
    def _make_initial(length: int) -> np.ndarray:
        if "height_sequence_1" in globals():
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq.size == length:
                a0 = seq.copy()
            else:
                # Linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, seq.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, length, endpoint=False)
                a0 = np.interp(x_new, x_old, seq)
        else:
            a0 = rng.dirichlet(np.ones(length)).astype(np.float64)

        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / length)
        return a0

    best_overall = None
    best_score_overall = float("inf")

    # ------------------------------------------------------------------
    # Phase A – optimise the primary length (most time budget)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        a0 = _make_initial(primary_len)
        vec, score = _optimise_one_length_adam(a0, deadline, rng)
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - (deadline - (budget_s - safety))
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={primary_len} score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # Phase B – if time permits, try a different length (5000)
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        secondary_len = 5000
        a0 = _make_initial(secondary_len)
        vec, score = _optimise_one_length_adam(a0, deadline, rng)
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - (deadline - (budget_s - safety))
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={secondary_len} score={best_score_overall:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    # Clip to the limits enforced by ``evaluate_sequence`` and return
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)

    return [float(x) for x in best_overall]
