"""
Propose a non‑negative step function that (approximately) minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. **Problem size** – use the length of the previously found best construction
   (`height_sequence_1`) if available, otherwise fall back to ``n = 3800``.
   All candidates are normalised to unit sum, because the objective
   ``2·n·max(a∗a) / (∑a)²`` is invariant under scaling.

2. **Fast helpers** – FFT based linear convolution, correlation (the
   sub‑gradient of a weighted sum of convolution entries) and a Euclidean
   projection onto the probability simplex.

3. **Exploratory scan** – a cheap exhaustive sweep over a few
   parametrised families:
   * Power‑law shapes ``a_i ∝ (i+shift)^{‑p}``  (p≈0.4…0.6, shift≈0.1…6).
   * Exponential decay ``a_i ∝ exp(‑λ i)`` (λ small).
   * The classic ``sqrt(i+1)-sqrt(i)`` construction.
   The best sequence from this scan is taken as the starting point.

4. **Simulated annealing / sub‑gradient moves** – the main search loop.
   * **Random mass‑transfer move** – pick two indices, move a small
     fraction of the source mass to the target, keep the sum constant,
     project onto the simplex and accept the move with a Metropolis rule.
   * **Sub‑gradient move** – identify the currently maximal convolution
     entries, build a uniform weight vector on the active set, obtain the
     sub‑gradient ``g = 2·a⋆w`` (via correlation) and take a short step
     in the direction ``‑g`` followed by simplex projection.
   * Temperature and step‑size decay linearly with elapsed time.
   * If no improvement has been seen for ``restart_gap`` seconds the
     current state is re‑initialised from the best solution with a small
     log‑normal jitter (helps escaping plateaus).

5. **Termination** – the loop runs until the 1000 s budget (minus a small
   safety margin) is exhausted; the best sequence found is returned
   (clipped to the limits imposed by ``evaluate_sequence``).

The routine is deliberately deterministic after the seed is set, uses only
NumPy (no external binaries) and respects the 2‑CPU limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper 1: smallest power‑of‑two length for a linear convolution of size n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for use with FFT."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Helper 2: linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the linear convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Helper 3: correlation needed for the sub‑gradient of Σ w_k (a∗a)_k.
# grad = 2·a ⋆ w   (⋆ = correlation, not convolution)
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Compute ∇_a Σ_k w_k·(a∗a)_k = 2·a⋆w."""
    n = a.size
    L = w.size                     # = 2·n‑1
    # correlation length = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reversed a for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Helper 4: Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a 1‑D array onto {x≥0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical clean‑up
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Utility: evaluate the objective for a unit‑sum vector (no clipping needed).
# ----------------------------------------------------------------------
def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a∗a) for a vector that already sums to 1."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Main entry point – called by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Length of the step sequence.  If omitted, the length of the
        previously discovered best construction is used, otherwise 3800.

    Returns
    -------
    list[float]
        The best sequence found (values clipped to the limits imposed by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # 2) Helper array of indices (used many times)
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    # --------------------------------------------------------------
    # 3) Initialise best candidate with anything reasonable
    # --------------------------------------------------------------
    best = None
    best_score = np.inf

    # 3a – use a previously found construction if it exists
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                cand_sum = cand.sum()
                if cand_sum > 0:
                    cand /= cand_sum
                    sc = _score(cand, n, fft_len)
                    if sc < best_score:
                        best = cand.copy()
                        best_score = sc
        except Exception:
            pass

    # 3b – analytic “sqrt‑difference’’ start
    sqrt_seq = np.sqrt(idx + 1.0) - np.sqrt(idx)
    sqrt_seq /= sqrt_seq.sum()
    sc = _score(sqrt_seq, n, fft_len)
    if sc < best_score:
        best = sqrt_seq.copy()
        best_score = sc

    # --------------------------------------------------------------
    # 4) Cheap parametric scan (power‑law & exponential families)
    # --------------------------------------------------------------
    # Power‑law a_i ∝ (i + shift)^{‑p}
    p_vals = np.arange(0.35, 0.66, 0.01)
    shift_vals = np.arange(0.1, 6.1, 0.5)

    for p in p_vals:
        for shift in shift_vals:
            a = (idx + shift) ** (-p)
            a_sum = a.sum()
            if a_sum == 0:
                continue
            a /= a_sum
            sc = _score(a, n, fft_len)
            if sc < best_score:
                best = a.copy()
                best_score = sc
                print(f"[scan] p={p:.3f} shift={shift:.2f} score={sc:.6f}")

    # Exponential decay a_i ∝ exp(‑λ i)
    lam_vals = np.linspace(0.0005, 0.015, 20)
    for lam in lam_vals:
        a = np.exp(-lam * idx)
        a /= a.sum()
        sc = _score(a, n, fft_len)
        if sc < best_score:
            best = a.copy()
            best_score = sc
            print(f"[scan] λ={lam:.5f} score={sc:.6f}")

    # --------------------------------------------------------------
    # 5) Main search – simulated annealing + occasional sub‑gradient step
    # --------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    start_time = time.monotonic()
    # keep a safety margin of a couple of seconds
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # scheduling parameters
    restart_gap = 140.0                     # seconds without improvement → restart
    last_improve = start_time

    # temperature schedule (linear interpolation)
    T_start = 0.05 * cur_score              # a few percent of the current value
    T_end = 1e-5

    # step‑size for random mass transfers
    move_scale_start = 0.4
    move_scale_end = 0.02

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        ratio = min(1.0, elapsed / total)

        # temperature and move‑scale decay
        T = T_start * (1.0 - ratio) + T_end * ratio
        move_scale = move_scale_start * (1.0 - ratio) + move_scale_end * ratio

        # ------------------------------------------------------------------
        # Choose a proposal: 90 % random mass‑transfer, 10 % sub‑gradient step
        # ------------------------------------------------------------------
        if rng.random() < 0.90:
            # ---- random mass‑transfer between two positions ----------------
            src = rng.integers(0, n)
            if a[src] <= 1e-12:
                # nothing to move – skip this iteration
                continue
            dst = rng.integers(0, n - 1)
            if dst >= src:
                dst += 1
            delta = rng.random() * a[src] * move_scale
            a_new = a.copy()
            a_new[src] -= delta
            a_new[dst] += delta
        else:
            # ---- sub‑gradient step based on the current active set -------
            conv = _conv_self(a, fft_len)
            max_val = conv.max()
            active = np.isclose(conv, max_val, atol=1e-12)
            w = np.zeros_like(conv)
            if np.any(active):
                w[active] = 1.0 / active.sum()
            else:
                # fallback – single maximum
                w[np.argmax(conv)] = 1.0
            grad = _gradient_correlation(a, w)
            # learning‑rate schedule (mirrors the original Adam schedule)
            lr = 0.25 * (1.0 - ratio) + 0.001 * ratio
            a_new = a - lr * grad
        # ------------------------------------------------------------------
        # Project back onto the simplex (keeps sum = 1 and non‑negativity)
        # ------------------------------------------------------------------
        a_new = np.clip(a_new, 0.0, None)
        a_new = _proj_simplex(a_new)

        # Evaluate the proposal
        new_score = _score(a_new, n, fft_len)
        delta_score = new_score - cur_score

        # Metropolis acceptance criterion
        if delta_score < 0 or rng.random() < np.exp(-delta_score / max(T, 1e-12)):
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Restart if we have been stuck for a while
        # ------------------------------------------------------------------
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.10, size=n))
            a = best * jitter
            a = _proj_simplex(a)
            cur_score = _score(a, n, fft_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – jittered current score {cur_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 6) Return the best sequence (apply the clipping required by
    #    ``evaluate_sequence`` – values >1000 are forced down, negatives
    #    are forced up).
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
