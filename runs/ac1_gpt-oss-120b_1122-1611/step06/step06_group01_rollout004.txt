def optimise_one_length(a_init, deadline, rng):
    a = a_init.copy()
    n = len(a)
    fft_len_self = next_pow2(2*n-1)
    fft_len_grad = next_pow2(3*n-2)
    best = a.copy()
    best_score = 2*n* self_conv(a, fft_len_self).max()
    # Params
    start_time = time.monotonic()
    total_time = deadline - start_time
    # We'll allocate 70% for softmax + Adam, then 30% for subgradient.
    phase1_end = start_time + 0.70 * total_time

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    t=0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        t += 1
        # compute schedule param
        if now < phase1_end:
            frac = (now - start_time) / (phase1_end - start_time)
            eta = 0.5 * (1-frac) + 0.02*frac # e.g., decaying from .5 to .02
            beta = 0.5 + (2000-0.5) * frac  # increase
            # compute gradient via softmax
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()
            # softmax
            w = np.exp(beta * (conv - max_c))
            # Might overflow; we can clip
            w = np.clip(w, 0, np.finfo(np.float64).max)
            w_sum = w.sum()
            if w_sum ==0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum
            # gradient
            rev_a = a[::-1]
            grad_full = _conv_fft(p, rev_a, fft_len_grad) # compute conv via FFT
            # get gradient slice length n
            grad = 2.0 * grad_full[n-1:n-1+n]
        else:
            # exact subgradient at max index
            conv = _self_conv(a, fft_len_self)
            k = int(conv.argmax())
            g = np.zeros_like(a)
            i_min = max(0, k-(n-1))
            i_max = min(n-1, k)
            idx_i = np.arange(i_min, i_max+1)
            idx_j = k - idx_i
            g[idx_i] = 2.0 * a[idx_j]
            grad = g

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # projection onto simplex
        a = _project_simplex(a)
        # Evaluate score
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now-start_time:6.2f}s] n={n} it={t} new best {best_score:.6f}")

        # stagnation: if no improve for some time, restart jitter around best
        if now - last_improve > 20.0:  # earlier restart
            # jitter
            a = best.copy()
            # multiplicative noise lognormal, std ~5%
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * noise
            a = _project_simplex(a)
            # reset adam moments
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            last_improve = now
            print(f"[{now-start_time:6.2f}s] n={n} restart due to stagnation")

    return best, best_score

def _conv_fft(x, y, fft_len):
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    return np.fft.irfft(fx * fy, n=fft_len)[: len(x) + len(y) - 1]

    fp = np.fft.rfft(p, n=fft_len)

    fft_len_grad = _next_pow2(3*n-2)
    best = a.copy()

    phase1_end = start + 0.70*(deadline-start)
    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    it = 0
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1
        if now < phase1_end:
            # softmax gradient
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()
            # schedule
            frac = (now - start) / (phase1_end - start)
            eta = 0.5 * (1-frac) + 0.01 * frac  # decays
            beta_temp = 0.5 * ((5000.0/0.5) ** frac)  # exponential

            w = np.exp(beta_temp * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum
            # gradient
            rev_a = a[::-1]
            grad_full = _conv_fft(p, rev_a, fft_len_grad)
            grad = 2.0 * grad_full[n-1:n-1+n]

            conv = _self_conv(a, fft_len_self)
            k = int(conv.argmax())
            g = np.zeros_like(a)

def _lp_refine(f, M):
    n = f.size
    # Convert f to numpy array of float64
    from scipy.sparse import lil_matrix
    rows = 2*n - 1
    G = lil_matrix((rows, n))
    for i in range(n):
        G[i:i+n, i] = f[i]
    # Use scipy.optimize.linprog
    from scipy.optimize import linprog
    c = -np.ones(n)  # maximize sum g
    bounds = [(0, None)] * n
    # A_ub = G, b_ub = M
    res = linprog(c, A_ub=G, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if res.success:
        g = res.x
        # scale to sum=1
        s = g.sum()
        if s > 0:
            g = g / s
        return g, res.fun
    else:
        return None, None

"""
Search for step-function sequence minimizing evaluate_sequence.

Algorithm:
- Enumerate a list of candidate lengths (including the previously known length).
- For each length, allocate a fraction of remaining time.
- Initialise several starting vectors (interpolated best from previous length,
  random Dirichlet, uniform) and run a two‑phase projected Adam optimizer:
    * Phase 1: smoothed soft‑max approximation of max(convolution) with a
      temperature that grows exponentially; gradient obtained by FFT.
    * Phase 2: exact sub‑gradient at the peak.
- Periodically restart with log‑normal jitter if no improvement.
- For moderate lengths (≤2000) perform a linear‑programming refinement:
  maximize the ℓ₁‑norm of a vector g under the constraint f * g ≤ current
  peak, then replace the best candidate if it improves the bound.
- Keep the best vector seen across all lengths and return it.

All vectors are kept on the probability simplex (sum = 1) so that the
objective coincides with ``evaluate_sequence`` up to a constant factor.
"""

import time
import numpy as np
from scipy.sparse import lil_matrix

"""
Search for a non‑negative step‑function that minimises
``evaluate_sequence``.

Algorithm
---------

* Choose a handful of candidate lengths (including the length of the
  previously found construction, if any).

* For each length allocate a modest share of the remaining time budget,
  initialise several starting points (interpolated best, Dirichlet random,
  uniform) and run a **projected Adam optimiser**.

  – *Phase 1* (≈ 70 % of the allotted time for the length) uses a
    soft‑max approximation of the max‑convolution.  The temperature β grows
    exponentially from 0.5 to 5 000 and the gradient
    `∇ = 2·(p * rev(a))` is obtained by FFT.

  – *Phase 2* (the rest) uses the exact sub‑gradient at the current peak
    of the self‑convolution.

  – After each step the iterate is projected onto the probability simplex
    (sum = 1, aᵢ ≥ 0).  Stagnation for >30 s triggers a log‑normal jitter
    restart.

* For moderate lengths (n ≤ 2000) a **LP refinement** is performed:
  maximise ∑ gⱼ subject to gⱼ ≥ 0 and `f * g ≤ M` (pointwise) where
  `M = max(f * f)`.  If the refined vector improves the bound it replaces the
  current best.

* The best vector over all lengths is returned (clipped to the limits
  enforced by ``evaluate_sequence``).  The vector is kept on the simplex,
  so its objective equals ``evaluate_sequence`` up to a constant factor.
"""

# ----------------------------------------------------------------------
# Imports
# ----------------------------------------------------------------------
import time
import numpy as np
from scipy.sparse import lil_matrix
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a * a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Convolution of 1‑D arrays x and y via FFT (full, length = len(x)+len(y)-1)."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv_full = np.fft.irfft(fx * fy, n=fft_len)
    return conv_full[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# LP refinement (used for modest lengths)
# ----------------------------------------------------------------------


def _lp_refine(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Given a vector f (sum = 1) and its peak M = max(f * f),
    solve   max Σ g_j   s.t.  g ≥ 0  and  f * g ≤ M pointwise.
    Returns the optimiser g (renormalised to sum = 1) or None on failure.
    """
    n = f.size
    rows = 2 * n - 1

    # Build the Toeplitz‑like convolution matrix G where (G g)_k = Σ_i f_i g_{k-i}
    G = lil_matrix((rows, n), dtype=np.float64)
    for i in range(n):
        G[i:i + n, i] = f[i]
    G = G.tocsr()

    c = -np.ones(n)                     # maximise Σ g  ↔  minimise −Σ g
    b = np.full(rows, M)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(
        c,
        A_ub=G,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if res.success:
        g = res.x
        s = g.sum()
        if s > 0:
            g = g / s
        return g
    return None

# Core optimiser for a single length
# ----------------------------------------------------------------------


def _optimise_one_length(a_init: np.ndarray, deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Two‑phase projected Adam optimiser for a fixed length.
    Returns the best vector (sum = 1) found and its objective
    2 · n · max(conv).
    """
    a = a_init.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # initial score
    conv = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * conv.max()

    start = time.monotonic()
    phase1_end = start + 0.70 * (deadline - start)

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    it = 0
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        # --------------------------------------------------------------
        # Phase selection
        # --------------------------------------------------------------
        if now < phase1_end:
            # soft‑max (smoothed) gradient
            conv = _self_conv(a, fft_len_self)
            max_c = conv.max()

            # schedules
            frac = (now - start) / (phase1_end - start)
            eta = 0.5 * (1.0 - frac) + 0.01 * frac          # linear decay
            beta = 0.5 * ((5000.0 / 0.5) ** frac)           # exponential growth

            # soft‑max weights (stable)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                p = np.full_like(w, 1.0 / w.size)
            else:
                p = w / w_sum

            # gradient = 2·(p * rev(a))
            rev_a = a[::-1]
            grad_full = _conv_fft(p, rev_a, fft_len_grad)
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]
        else:
            # exact sub‑gradient at the current peak
            conv = _self_conv(a, fft_len_self)
            k = int(conv.argmax())
            g = np.zeros_like(a)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k - idx_i
            g[idx_i] = 2.0 * a[idx_j]
            grad = g
            eta = 0.2 / np.sqrt(it)                        # decreasing step

        # ------------------------------------------------------------------
        # Adam update + simplex projection
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # ------------------------------------------------------------------
        # Evaluation
        # ------------------------------------------------------------------
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}")

        # ------------------------------------------------------------------
        # Stagnation handling
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * noise
            a = _project_simplex(a)
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found (list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Candidate lengths (ascending, includes previous construction if any)
    # ------------------------------------------------------------------
    lengths = []
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
            lengths.append(seq0.size)

    # additional lengths – a mix of powers of two and a few odd numbers
    for n in (1024, 2048, 3000, 3809, 5000, 8000):
        if n not in lengths:
            lengths.append(n)

    lengths = sorted(set(lengths))

    best_overall = None
    best_score_overall = np.inf
    # store the best vector found so far (for interpolation to larger n)
    current_best = None

    # ------------------------------------------------------------------
    # Iterate over lengths, giving each a modest slice of the remaining budget
    # ------------------------------------------------------------------
    for idx, n in enumerate(lengths):
        now = time.monotonic()
        if now >= deadline:
            break

        remaining = deadline - now
        # Give each length at most 150 s, but keep some budget for later sizes
        portion = min(remaining * 0.20, 150.0)
        length_deadline = now + portion

        # ------------------------------------------------------------------
        # Build a small pool of initial vectors for this length
        # ------------------------------------------------------------------
        init_vectors = []

        # 1) interpolate the current best (if any) to the new size
        if current_best is not None:
            x_old = np.linspace(0.0, 1.0, current_best.size, endpoint=False)
            x_new = np.linspace(0.0, 1.0, n, endpoint=False)
            a_interp = np.interp(x_new, x_old, current_best)
            a_interp = np.clip(a_interp, 0.0, None)
            s = a_interp.sum()
            if s > 0.0:
                a_interp = a_interp / s
            else:
                a_interp = np.full(n, 1.0 / n)
            init_vectors.append(a_interp)

        # 2) Dirichlet random start
        init_vectors.append(rng.dirichlet(np.ones(n)).astype(np.float64))

        # 3) Uniform start
        init_vectors.append(np.full(n, 1.0 / n))

        # ------------------------------------------------------------------
        # Optimise each start while we still have time for this length
        # ------------------------------------------------------------------
        for a0 in init_vectors:
            if time.monotonic() >= length_deadline:
                break
            # give each start roughly half of the remaining slice
            sub_deadline = time.monotonic() + min(30.0, (length_deadline - time.monotonic()) / 2.0)
            sub_deadline = min(sub_deadline, length_deadline)

            vec, score = _optimise_one_length(a0, sub_deadline, rng)

            # update global best
            if score < best_score_overall - 1e-12:
                best_score_overall = score
                best_overall = vec.copy()
                elapsed_total = time.monotonic() - start_time
                print(f"[{elapsed_total:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

            # keep a good candidate for the next length
            if current_best is None or score < best_score_overall + 1e-8:
                current_best = vec.copy()

            # ------------------------------------------------------------------
            # LP refinement for moderate sizes (≤2000)
            # ------------------------------------------------------------------
            if n <= 2000:
                conv = _self_conv(vec, _next_pow2(2 * n - 1))
                M = conv.max()
                g = _lp_refine(vec, M)
                if g is not None:
                    conv_g = _self_conv(g, _next_pow2(2 * n - 1))
                    score_g = 2.0 * n * conv_g.max()
                    if score_g < best_score_overall - 1e-12:
                        best_score_overall = score_g
                        best_overall = g.copy()
                        elapsed_total = time.monotonic() - start_time
                        print(f"[{elapsed_total:6.2f}s] *** GLOBAL BEST (LP) n={n} score={best_score_overall:.6f}")
                        current_best = g.copy()

    # ------------------------------------------------------------------
    # Fallback – should never happen but guards against empty result.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    # Clip to the limits enforced by ``evaluate_sequence`` and return.
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    return [float(x) for x in best_overall]
