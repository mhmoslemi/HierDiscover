    fft_len = 1 << ((2*n -1 -1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)

Implementation:

```
def gradient_correlation(a, w):
    n = a.shape[0]
    L = w.shape[0]
    conv_len = L + n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reversed a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[:conv_len]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad
```

"""
Propose a non‑negative step‑function vector that (approximately) minimizes
``evaluate_sequence``.  The algorithm combines:

*   Several families of initialisations (random simplex, sparse Dirichlet,
    sqrt‑difference, sine, exponential, and the previously discovered vector).
*   A soft‑max – Adam optimiser that minimises a smooth surrogate of the
    max‑convolution objective while keeping the vector on the probability simplex.
*   Random perturbations and a short hill‑climbing phase for local refinement.
*   Multiple lengths (n) are tried; the best vector over the whole budget is
    returned.

The implementation is fully self‑contained, uses only NumPy, and respects the
1000 s time budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility: projection onto the probability simplex (Σx = 1, x ≥ 0)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a 1‑D array onto the simplex."""
    if v.ndim != 1:
        raise ValueError("Input must be 1‑D")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Convolution and gradient helpers (FFT‑based, real‑valued)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w.
    ``a`` length n, ``w`` length 2·n‑1.
    """
    n = a.size
    L = w.size  # = 2·n‑1
    conv_len = L + n - 1
    fft_len = _next_pow2(conv_len)
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[:conv_len]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _score(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective used by ``evaluate_sequence`` (without clipping)."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Core optimiser for a single length & initial vector
# ----------------------------------------------------------------------
def _optimize_one(
    a0: np.ndarray,
    n: int,
    rng: np.random.Generator,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation with a soft‑max surrogate of the max‑convolution.
    Returns the best vector found for this run and its score.
    """
    a = a0.copy()
    fft_len_conv = _next_pow2(2 * n - 1)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # schedules
    max_iters = 2000
    lr_start = 0.30
    lr_end = 0.001
    beta_start = 0.5
    beta_end = 800.0

    best_local = a.copy()
    best_score = _score(a, n, fft_len_conv)

    while t < max_iters and time.monotonic() < deadline - 0.5:
        t += 1
        ratio = t / max_iters
        lr = lr_start * (1.0 - ratio) + lr_end * ratio
        beta = beta_start + (beta_end - beta_start) * ratio

        # self‑convolution
        conv = _conv_self(a, fft_len_conv)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_local = a.copy()

        # soft‑max weights (numerically stable)
        diff = conv - cur_max          # ≤ 0
        w_raw = np.exp(beta * diff)
        w_sum = w_raw.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            mask = conv == cur_max
            w_raw = np.where(mask, 1.0, 0.0)
            w_sum = w_raw.sum()
        w = w_raw / w_sum

        # gradient of the surrogate
        grad = _gradient_correlation(a, w)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _proj_simplex(a)

        # occasional random jitter to escape shallow plateaus
        if rng.random() < 0.01:
            factor = np.exp(rng.normal(0.0, 0.2, size=n))
            a = a * factor
            a = _proj_simplex(a)

        if rng.random() < 0.01:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-15:
                delta = rng.random() * a[i] * 0.3
                a[i] -= delta
                a[j] += delta
                a = _proj_simplex(a)

    # ------------------------------------------------------------------
    # Light‑weight hill‑climbing using cheap mass‑transfer moves
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 0.1:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] < 1e-12:
            continue
        delta = rng.random() * a[i] * 0.2
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        if a_new[i] < 0:
            continue
        conv_new = _conv_self(a_new, fft_len_conv)
        score_new = 2.0 * n * float(conv_new.max())
        if score_new < best_score - 1e-12:
            a = a_new
            best_score = score_new
            best_local = a.copy()

    return best_local, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine explores several sequence lengths, diverse initialisations
    and runs a soft‑max/Adam optimiser with occasional random perturbations.
    The best vector found within the time budget is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)   # tiny safety margin

    # --------------------------------------------------------------
    # Candidate lengths to try (more elements → finer spreading, but slower)
    # --------------------------------------------------------------
    lengths = [1500, 2500, 3500, 4500, 5500, 7000, 8500, 10000]
    rng.shuffle(lengths)

    best_global = None
    best_score_global = float("inf")
    length_index = 0

    while time.monotonic() < deadline - 5.0:    # keep a few seconds spare
        # pick next length (cycle through the shuffled list)
        n = lengths[length_index % len(lengths)]
        length_index += 1

        # reserve a modest time slice for this trial (≈6 s)
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 6.0:
            break
        trial_deadline = now + min(6.0, remaining * 0.6)

        # ----------------------------------------------------------
        # Pick an initial vector from a diverse family
        # ----------------------------------------------------------
        init_type = rng.choice(
            ["random", "sparse", "sqrt", "sine", "exp", "prev"]
        )
        if init_type == "random":
            a0 = rng.random(n)
            a0 = a0 / a0.sum()
        elif init_type == "sparse":
            a0 = rng.dirichlet(np.full(n, 0.5))
        elif init_type == "sqrt":
            i = np.arange(n, dtype=np.float64)
            a_raw = np.sqrt(i + 1.0) - np.sqrt(i)
            a0 = a_raw / a_raw.sum()
        elif init_type == "sine":
            i = np.arange(n, dtype=np.float64)
            a0 = np.sin(np.pi * (i + 0.5) / n)      # positive sine wave
            a0 = a0 / a0.sum()
        elif init_type == "exp":
            i = np.arange(n, dtype=np.float64)
            a0 = np.exp(-i / (n / 5.0))
            a0 = a0 / a0.sum()
        else:  # "prev"
            if "height_sequence_1" in globals():
                prev = np.asarray(
                    globals()["height_sequence_1"], dtype=np.float64
                )
                if prev.ndim == 1 and prev.size > 0:
                    # Resample to required length by linear interpolation
                    if prev.size != n:
                        x_old = np.linspace(0.0, 1.0, prev.size)
                        x_new = np.linspace(0.0, 1.0, n)
                        a0 = np.interp(x_new, x_old, prev)
                    else:
                        a0 = prev.copy()
                    a0 = np.clip(a0, 0.0, None)
                    s = a0.sum()
                    if s > 0:
                        a0 = a0 / s
                    else:
                        a0 = rng.random(n); a0 /= a0.sum()
                else:
                    a0 = rng.random(n); a0 /= a0.sum()
            else:
                a0 = rng.random(n); a0 /= a0.sum()

        # ----------------------------------------------------------
        # Run the optimiser for this trial
        # ----------------------------------------------------------
        candidate, cand_score = _optimize_one(a0, n, rng, trial_deadline)

        # ----------------------------------------------------------
        # Keep the overall best
        # ----------------------------------------------------------
        if cand_score < best_score_global - 1e-12:
            best_score_global = cand_score
            best_global = candidate.copy()
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] new global best – score={best_score_global:.6f}, n={n}"
            )

    # Fallback in the unlikely event that no candidate was produced
    # ------------------------------------------------------------------
    if best_global is None:

    # Apply the clipping that ``evaluate_sequence`` would apply
    best_global = np.clip(best_global, 0.0, 1000.0)

    # Return as a plain Python list of floats
    return [float(v) for v in best_global]
