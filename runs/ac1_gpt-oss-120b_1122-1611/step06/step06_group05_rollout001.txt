"""
Propose a (non‑negative) step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
*  Use a soft‑max parametrisation  a = softmax(x)  (guarantees a_i ≥ 0,
   ∑a_i = 1).  The objective 2·n·max_k(a∗a)_k  is approximated by the
   smooth surrogate  f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k).
*  Optimise f_β with Adam while
      – slowly raising the temperature β (so the surrogate becomes the
        true max),
      – applying a small entropy regularisation that encourages a
        spread‑out shape in early iterations,
      – occasionally smoothing the logits to stay in the low‑frequency
        sub‑space (the optimum appears to be a smooth curve).
*  Run several short‑duration restarts from diverse initialisations
   (uniform, √‑difference, exponential, arcsine‑like, random Dirichlet
   and, if present, the previous best construction).  The best sequence
   seen across all runs is returned.
*  All heavy work (convolution, correlation, FFT lengths) is pre‑computed
   once per run; the inner loop therefore costs only a few FFTs per
   iteration and easily fits the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """smallest power‑of‑two length that can hold a linear convolution of two
    vectors of length n (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """self‑convolution of a (real‑valued) using an FFT of length `fft_len`.
    Returns the full linear convolution of length 2·len(a)‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_corr(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """gradient of Σ_k w_k·(a∗a)_k w.r.t. a.
    Implements 2·(a ⋆ w) where ⋆ denotes discrete correlation.
    The result has length ``len(a)``."""
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: 3 * a.size - 2]
    start = a.size - 1
    return 2.0 * corr_full[start:start + a.size]


def _smooth_array(arr: np.ndarray, sigma: float) -> np.ndarray:
    """simple Gaussian smoothing (non‑periodic, mode='same') using a small
    kernel.  ``sigma`` is the standard deviation in index units."""
    if sigma <= 0.5:
        return arr
    # Build a truncated Gaussian kernel
    radius = int(3 * sigma)
    x = np.arange(-radius, radius + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    # Convolve – the kernel is small, direct convolution is fine
    smoothed = np.convolve(arr, kernel, mode='same')
    return smoothed


# ----------------------------------------------------------------------
# Core optimiser (one start → one optimisation run)
# ----------------------------------------------------------------------
def _run_optimizer(init_a: np.ndarray,
                   n: int,
                   time_budget: float,
                   rng: np.random.Generator,
                   verbose_tag: str = ""):
    """
    Optimises a starting vector ``init_a`` (already normalised to sum‑1)
    for at most ``time_budget`` seconds.
    Returns the best ``a`` seen and its objective value.
    """
    # FFT lengths (stay constant during the run)
    fft_len_conv = _fft_len_for_n(n)
    fft_len_corr = 1 << ((3 * n - 2 - 1).bit_length())

    # Adam state
    x = np.log(init_a + 1e-30)          # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # Schedules
    beta_start = 5.0
    beta_end = 8000.0          # very aggressive – pushes towards true max
    lr_start = 2.0
    lr_end = 0.05
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # Entropy regularisation (decays to zero)
    lambda_start = 0.20
    lambda_end = 0.0

    # Smoothing parameters (mild – keep shape smooth)
    smooth_sigma = 12.0       # kernel width
    smooth_alpha = 0.15      # how much of the smoothed version to blend in

    # Book‑keeping
    start_time = time.monotonic()
    deadline = start_time + time_budget
    best_a = init_a.copy()
    best_score = 2.0 * n * np.max(_conv_self(best_a, fft_len_conv))
    it = 0

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ---- schedules -------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)
        lr = lr_start * (1.0 - elapsed / total) + lr_end * (elapsed / total)
        lam = lambda_start + (lambda_end - lambda_start) * (elapsed / total)

        # ---- current probabilities ------------------------------------
        # soft‑max (logits → probabilities)
        max_x = np.max(x)
        a = np.exp(x - max_x)
        a /= a.sum()

        # ---- objective (exact) ----------------------------------------
        conv = _conv_self(a, fft_len_conv)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- bookkeeping ------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{verbose_tag}{elapsed:6.2f}s] new best {best_score:.6f}")

        # ---- smooth‑max weights (soft surrogate) ----------------------
        # w_k ∝ exp(β·(conv_k - cur_max)) – subtraction keeps numbers ≤ 0
        exp_arg = beta * (conv - cur_max)
        exp_arg = np.clip(exp_arg, -700, 0)      # safety for overflow
        w_raw = np.exp(exp_arg)
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)

        # ---- gradient w.r.t. a (soft‑max surrogate) --------------------
        grad_a = _grad_corr(a, w, fft_len_corr)

        # ---- entropy regularisation (encourage spread) ---------------
        if lam > 0:
            # ∇(−λ·H) = λ·(log a + 1)
            grad_a += lam * (np.log(a + 1e-30) + 1.0)

        # ---- chain rule → gradient w.r.t. logits x --------------------
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # ---- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ---- occasional smoothing of logits (helps avoid high‑frequency
        #      noise and encourages a globally smooth shape) ------------
        if rng.random() < 0.12:                # apply smoothing with 12 % prob.
            x_smooth = _smooth_array(x, smooth_sigma)
            x = (1.0 - smooth_alpha) * x + smooth_alpha * x_smooth

        # ---- jitter – simple exploration --------------------------------
        if rng.random() < 0.025:
            x += rng.normal(0.0, 0.06, size=n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point – orchestrates several restarts and picks the best.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the supplied time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size – allow the user to override, otherwise use the
    # length of the previously discovered construction if it exists,
    # falling back to a default of 3809.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
                n = int(prev.size)
                # Normalise to unit sum – the optimiser works in that scale.
                prev = np.clip(prev, 0.0, None)
                prev_sum = prev.sum()
                if prev_sum > 0:
                    prev = prev / prev_sum
                has_prev = True
            else:
                raise ValueError
        except Exception:
            prev = None
            n = int(kwargs.get("n", 3809))
            has_prev = False
    else:
        prev = None
        n = int(kwargs.get("n", 3809))
        has_prev = False

    if n < 2:
        n = 2

    # --------------------------------------------------------------
    # Build a list of diverse initial vectors (all already normalised to sum‑1)
    # --------------------------------------------------------------
    inits = []

    # 1. Uniform
    inits.append(np.full(n, 1.0 / n))

    # 2. √‑difference shape (the classic start used previously)
    i = np.arange(n, dtype=np.float64)
    sqrt_shape = np.sqrt(i + 1.0) - np.sqrt(i)
    sqrt_shape /= sqrt_shape.sum()
    inits.append(sqrt_shape)

    # 3. Exponential decay (scale ≈ n/12)
    scale = n / 12.0
    exp_shape = np.exp(-i / scale)
    exp_shape /= exp_shape.sum()
    inits.append(exp_shape)

    # 4. Arcsine‑like (singular at the ends – we truncate to avoid infinities)
    eps = 1e-6
    x = (i + 0.5) / n
    arcsine_shape = 1.0 / np.sqrt(np.clip(x * (1.0 - x), eps, None))
    arcsine_shape /= arcsine_shape.sum()
    inits.append(arcsine_shape)

    # 5. Random Dirichlet (α = 0.5 gives a fairly spiky start)
    dirichlet_shape = rng.dirichlet(np.full(n, 0.5))
    inits.append(dirichlet_shape)

    # 6. Previously discovered best (if available)
    if has_prev:
        inits.append(prev)

    # --------------------------------------------------------------
    # Main loop – allocate time among the initialisers and keep the global best.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # 1 s safety margin

    global_best_a = None
    global_best_score = np.inf
    attempts_done = 0
    total_inits = len(inits)

    while time.monotonic() < deadline and attempts_done < total_inits:
        # Remaining time and how many attempts are left → allocate proportionally
        now = time.monotonic()
        remaining = deadline - now
        attempts_left = total_inits - attempts_done
        # Give each remaining attempt at least 30 s, but honour the overall budget.
        slice_budget = max(20.0, remaining / attempts_left)

        init_a = inits[attempts_done]
        tag = f"run{attempts_done + 1}/"
        print(f"[{now - start_time:6.2f}s] start {tag} (budget {slice_budget:.1f}s)")

        best_a, best_score = _run_optimizer(init_a, n, slice_budget,
                                            rng, verbose_tag=tag)

        if best_score < global_best_score:
            global_best_score = best_score
            global_best_a = best_a.copy()
            print(f"[{now - start_time:6.2f}s] *** GLOBAL BEST UPDATED: {global_best_score:.6f}")

        attempts_done += 1

    # --------------------------------------------------------------
    # Post‑processing – a tiny blend with the uniform distribution can
    # sometimes shave a fraction off the peak (explore a few α values).
    # --------------------------------------------------------------
    if global_best_a is None:
        # Should never happen, but fall back to a uniform vector.
        global_best_a = np.full(n, 1.0 / n)

    uniform = np.full(n, 1.0 / n)
    best_score = global_best_score
    best_a = global_best_a

    # Simple line‑search in the direction of uniform
    for alpha in np.linspace(0.0, 0.15, 16):
        cand = (1.0 - alpha) * global_best_a + alpha * uniform
        cand /= cand.sum()                     # renormalise (tiny numeric drift)
        # Evaluate quickly – the exact objective is 2·n·max(conv)
        conv = _conv_self(cand, _fft_len_for_n(n))
        cand_score = 2.0 * n * float(conv.max())
        if cand_score < best_score:
            best_score = cand_score
            best_a = cand.copy()
            print(f"[blend] alpha={alpha:.3f} improved score to {best_score:.6f}")

    # --------------------------------------------------------------
    # Clip to the limits enforced by evaluate_sequence and return.
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
