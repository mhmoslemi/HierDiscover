Simplex projection function:

def proj_simplex(v):
    # v is 1D array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Projected gradient descent on a smoothed ℓ∞‑objective.

The objective of ``evaluate_sequence`` for a non‑negative vector `a`
with sum 1 is `score = 2·n·max(a∗a)`.  The algorithm keeps the
distribution on the simplex and repeatedly:

* computes the self‑convolution,
* builds a soft‑max weight `w_k ∝ exp(β·(conv_k‑max))`,
* obtains the gradient of the smoothed max
  (`∂/∂a  (1/β)·log Σ exp(β·conv_k) = 2·(a⋆w)`),
* takes a projected gradient step (`a ← a − η·grad`),
* projects back onto the probability simplex.

The temperature `β` and step size `η` are annealed, and occasional
random jitter prevents stagnation.  If the best score does not improve
for a while the algorithm restarts from the current best solution.
The routine respects the 1000 s budget and returns the best sequence
found (clipped to the limits of ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power‑of‑two length for a FFT that can hold a convolution
# of length 2·n‑1.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Self‑convolution via FFT (returns array of length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation a ⋆ w (used for the gradient of Σ w_k·(a∗a)_k).
# Returns a vector of length len(a).
# ----------------------------------------------------------------------
def _grad_corr(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    # Correlation a ⋆ w = (a_rev * w) where * is convolution.
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: 3 * a.size - 2]
    # central part corresponds to shifts 0 … n‑1
    start = a.size - 1
    return 2.0 * corr_full[start:start + a.size]


# ----------------------------------------------------------------------
# Euclidean projection of a vector onto the probability simplex.
# Guarantees non‑negativity and sum == 1.
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex {x≥0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("simplex projection only defined for 1‑D arrays")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    approximately minimises ``evaluate_sequence`` within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------ problem size ------------------------------
    # Use a larger support than the previous run – larger n gives a finer
    # chance to flatten the convolution.
    n = int(kwargs.get("n", 8000))
    if n < 2:
        n = 2

    # ---------------------- initial distribution ------------------------
    # Start from the classic √‑difference shape, normalised to sum = 1.
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a = a / a.sum()

    # If a previously discovered construction is available and better,
    # adopt it as the start point (the global variable may exist).
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    cand = cand / cand.sum()
                    # compute scores for both candidates
                    fft_len_cand = _fft_len_for_n(cand.size)
                    score_cand = 2.0 * cand.size * np.max(_conv_self(cand, fft_len_cand))
                    fft_len_a = _fft_len_for_n(a.size)
                    score_a = 2.0 * a.size * np.max(_conv_self(a, fft_len_a))
                    if score_cand < score_a:
                        a = cand
                        n = cand.size
        except Exception:
            pass

    # -------------------------- pre‑computations -----------------------
    fft_len_conv = _fft_len_for_n(n)                    # for self‑convolution
    fft_len_corr = 1 << ((3 * n - 2 - 1).bit_length())  # for correlation

    # -------------------------- schedules -----------------------------
    # Temperature for the soft‑max surrogate of the ℓ∞‑norm.
    beta_start = 1.0
    beta_end = 600.0
    # Step size for the projected gradient descent.
    eta_start = 0.35
    eta_end = 0.003

    # Restart configuration.
    restart_gap = 80.0          # seconds without improvement → restart
    jitter_prob = 0.04
    jitter_scale = 0.02

    # -------------------------- bookkeeping ---------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety margin
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_conv_self(best_a, fft_len_conv))
    last_improve = start_time
    it = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        prog = elapsed / total

        # ----- schedules ------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * prog
        eta = eta_start * (1.0 - prog) + eta_end * prog

        # ----- compute convolution and soft‑max weights ----------------
        conv = _conv_self(a, fft_len_conv)          # length 2·n‑1
        max_conv = float(conv.max())
        # soft‑max over the convolution entries (more stable than hard mask)
        exp_arg = beta * (conv - max_conv)          # ≤ 0
        exp_arg = np.clip(exp_arg, -700.0, 0.0)     # avoid overflow
        w_raw = np.exp(exp_arg)
        w_sum = w_raw.sum()
        if w_sum > 0.0:
            w = w_raw / w_sum
        else:                                       # degenerate – uniform weight
            w = np.full_like(w_raw, 1.0 / w_raw.size)

        # ----- gradient of the smoothed max ---------------------------
        grad = _grad_corr(a, w, fft_len_corr)       # shape (n,)

        # ----- projected gradient step ---------------------------------
        a_new = a - eta * grad
        a = _project_simplex(a_new)

        # ----- occasional stochastic perturbations --------------------
        if rng.random() < jitter_prob:
            a += rng.normal(0.0, jitter_scale, size=n)
            a = _project_simplex(a)

        # ----- evaluate current score ----------------------------------
        cur_score = 2.0 * n * np.max(_conv_self(a, fft_len_conv))
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- restart if stuck ----------------------------------------
        if now - last_improve > restart_gap:
            # Re‑initialise from the current best plus a small jitter.
            a = best_a.copy()
            a += rng.normal(0.0, 0.03, size=n)
            a = _project_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best construction, respecting the clipping performed by
    # ``evaluate_sequence`` (non‑negative, ≤ 1000).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
