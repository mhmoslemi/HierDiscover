    i = np.arange(n, dtype=np.float64)

    a = np.maximum(a, 0.0)

        cur_max = conv.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur_a.copy()

            print(f"[{elapsed:.2f}s] polish: new best score = {best_score:.6f}")

        # schedule
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        # compute fraction
        frac = elapsed / total
        eta = eta0 * (1 - frac) + eta1 * frac
        beta = beta0 + (beta1 - beta0) * frac

        # choose active indices (top K)
        K = max_active
        if K >= conv.size:

        else:

        # momentum
        grad = momentum * grad_prev + (1 - momentum) * grad

"""
Propose a non‑negative step function that minimizes

    score = 2·n·max_k (a∗a)_k / (∑a_i)²

The algorithm works in three phases:

1. **Analytic & global start** – the classic “√‑difference’’ sequence and,
   if present, a previously saved construction (`height_sequence_1`).

2. **Randomised parametric search** – we draw a large number of simple
   analytic shapes (power‑law or difference‑of‑power) defined by only one or
   two continuous parameters.  All candidates are normalised to a unit sum
   and scored with a fast FFT‑based convolution; the best shape found
   becomes the new starting point.

3. **Deterministic polishing** – a mirror‑descent / exponentiated‑gradient
   routine that directly minimises the max‑convolution.  It uses a
   soft‑max approximation with an annealing parameter β and an adaptive
   learning‑rate η, the active set being the top‑K convolution entries
   (K≈80).  The polishing runs until the remaining time budget is exhausted.

The whole routine respects the 1000 s budget, prints intermediate progress
and finally returns a list of non‑negative heights (clipped to the limits
required by ``evaluate_sequence``).  The implementation uses only NumPy
and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution helpers (FFT based)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of two
    vectors of length *n*."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via real‑to‑real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast internal objective for a *unit‑sum* vector a:
    score = 2·n·max(conv)."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Analytic starting families
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _shape_power_law(n: int, p: float, offset: float) -> np.ndarray:
    """a_i ∝ (i + offset)^(-p)  (offset>0 to avoid division by zero)."""
    i = np.arange(n, dtype=np.float64)
    a = np.power(i + offset, -p)
    a /= a.sum()
    return a


def _shape_diff_power(n: int, q: float, offset: float) -> np.ndarray:
    """
    a_i = (i+offset)^q - (i+offset-1)^q   (offset≥1 ensures positivity).
    This is the discrete derivative of x↦x^q.
    """
    i = np.arange(n, dtype=np.float64)
    f = np.power(i + offset, q)
    f_prev = np.power(i + offset - 1.0, q)
    a = f - f_prev
    a = np.maximum(a, 0.0)      # numerical safety
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of a soft‑max weighted sum of convolution entries
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k  (w is a vector of length 2·n‑1).
    Implements the correlation a ⋆ w in O(n log n) using FFTs.
    """
    n = a.size
    L = w.size                     # should be 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

# Optional use of a previously saved construction
# ----------------------------------------------------------------------
def _maybe_use_global() -> np.ndarray | None:
    """If a global variable ``height_sequence_1`` exists, return it normalised."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Deterministic polishing (mirror‑descent on a soft‑max surrogate)
# ----------------------------------------------------------------------
def _polish_extended(a: np.ndarray,
                     n: int,
                     fft_len: int,
                     deadline: float,
                     start_score: float) -> tuple[np.ndarray, float]:
    """
    Refine *a* by exponentiated‑gradient steps on the smooth surrogate

        Φ(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

    The surrogate approaches max(a∗a) as β→∞.  A schedule for η (step‑size)
    and β (sharpness) is used, together with a momentum term.
    """
    best = a.copy()
    best_score = start_score
    cur_a = a.copy()
    start_time = time.monotonic()

    # hyper‑parameters (tuned empirically)
    eta_start, eta_end = 0.15, 0.001          # learning‑rate schedule
    beta_start, beta_end = 5.0, 5000.0        # soft‑max sharpness schedule
    momentum = 0.9
    max_active = 80                           # size of the active set

    grad_prev = np.zeros_like(a)

    while time.monotonic() < deadline:
        # ----- evaluate current point -----
        conv = _conv_self(cur_a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur_a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] polish: new best = {best_score:.6f}")

        # ----- schedule η and β -----
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = elapsed / total
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start + (beta_end - beta_start) * frac

        # ----- active set (top‑K entries of the convolution) -----
        K = max_active
        if K >= conv.size:
            active = np.arange(conv.size)
        else:
            # argpartition is O(n) and gives the K largest indices
            active = np.argpartition(conv, -K)[-K:]

        # ----- soft‑max weights on the active set -----
        # (conv ≤ cur_max, so exponent ≤ 0 → numerically stable)
        w_raw = np.exp(beta * (conv[active] - cur_max))
        w_sum = w_raw.sum()
        if w_sum == 0.0:                     # extreme under‑flow guard
            w = np.zeros_like(w_raw)
            w[np.argmax(conv[active])] = 1.0
        else:
            w = w_raw / w_sum

        # ----- gradient of the surrogate -----
        w_full = np.zeros_like(conv)
        w_full[active] = w
        grad = _gradient_correlation(cur_a, w_full)

        # ----- momentum -----
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient update (projection onto simplex) -----
        factor = np.exp(-eta * grad)
        cur_a = cur_a * factor
        cur_a = np.clip(cur_a, 0.0, None)
        s = cur_a.sum()
        if s == 0.0:
            cur_a = np.full_like(cur_a, 1.0 / n)
        else:
            cur_a /= s

        # ----- early exit check -----
        if time.monotonic() > deadline - 0.1:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.

    The search proceeds as:
      * analytic start or previously saved construction,
      * a fast randomised exploration over simple analytic families,
      * a deterministic polishing step that directly attacks the max‑convolution.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    n = max(2, n)                     # enforce a minimum size
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Initial candidates
    # --------------------------------------------------------------
    a_analytic = _analytic_start(n)

    cand_global = _maybe_use_global()
    if cand_global is not None and cand_global.size == n:
        # compare analytic vs global using the cheap internal score
        score_analytic = _compute_score(a_analytic, fft_len, n)
        score_global = _compute_score(cand_global, _fft_len_for_n(n), n)
        if score_global < score_analytic:
            a_best = cand_global.copy()
            best_score = score_global
        else:
            a_best = a_analytic.copy()
            best_score = score_analytic
    else:
        a_best = a_analytic.copy()
        best_score = _compute_score(a_best, fft_len, n)

    print(f"initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # Randomised parametric search (budget ≈ 30 % of total time)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    param_deadline = start_time + min(0.30 * budget_s, 60.0)   # ≤ 60 s

    while time.monotonic() < param_deadline:
        # draw a random shape
        shape_type = rng.choice(["power_law", "diff_power"])
        if shape_type == "power_law":
            p = rng.uniform(0.25, 1.00)          # exponent
            offset = rng.uniform(0.05, 5.0)      # shift to avoid singularity
            a_candidate = _shape_power_law(n, p, offset)
        else:
            q = rng.uniform(0.20, 2.00)          # power for the primitive
            offset = rng.uniform(1.0, 10.0)      # must be ≥ 1 for positivity
            a_candidate = _shape_diff_power(n, q, offset)

        # fast internal score (unit‑sum vectors)
        score = _compute_score(a_candidate, fft_len, n)

        if score < best_score - 1e-12:
            best_score = score
            a_best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] param‑search new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Deterministic polishing (use the remaining time)
    # --------------------------------------------------------------
    now = time.monotonic()
    remaining = (start_time + budget_s) - now - 0.5   # 0.5 s safety margin
    if remaining > 5.0:
        deadline = now + remaining
        a_best, best_score = _polish_extended(a_best, n, fft_len,
                                              deadline, best_score)
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] polishing finished, best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # --------------------------------------------------------------
    a_best = np.clip(a_best, 0.0, 1000.0)

    # Convert to a plain Python list of floats
    return [float(v) for v in a_best]
