def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Exponentiated‑gradient minimisation of max(self‑convolution) on the simplex.
    Uses a sub‑gradient of the ℓ∞‑norm and a projection onto the probability simplex.
    Multiple restarts and simple annealing are performed within the given time budget.
    """
    import time, numpy as np
    rng = np.random.default_rng(seed)
    # Choose problem size
    n = int(kwargs.get("n", 2000))
    # maybe clamp to <= 5000
    if n < 2: n = 2
    # initial sequence
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0:
                    cand = cand / cand.sum()
                    if cand.size == n:
                        a = cand.copy()
                    else:
                        # adjust size by truncation or padding with zeros
                        if cand.size > n:
                            a = cand[:n]
                        else:
                            pad = np.zeros(n)
                            pad[:cand.size] = cand
                            a = pad
                else:
                    a = None
            else:
                a = None
        except Exception:
            a = None
    else:
        a = None
    if a is None:
        # use sqrt-difference shape
        i = np.arange(n, dtype=np.float64)
        a = np.sqrt(i+1) - np.sqrt(i)
        a = a / a.sum()
    # keep best
    best_a = a.copy()
    best_score = 2.0 * n * np.max(_conv_self(a, _fft_len_for_n(n)))
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 0.5)   # safety margin
    # Pre compute FFT length
    fft_len = _fft_len_for_n(n)
    # Hyperparameters
    eta0 = 1.0   # initial step size for exponentiated gradient
    # tolerance for active set
    tol_rel = 1e-12
    # Loop
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Compute conv via FFT
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        # Compute score
        cur_score = 2.0 * n * cur_max
        # Record best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best {best_score:.6f}")
        # Determine active indices
        eps = max(tol_rel * cur_max, 1e-15)
        active_idx = np.where(conv >= cur_max - eps)[0]
        # Compute subgradient (average)
        g = np.zeros_like(a)
        for k in active_idx:
            # gradient contribution for shift k
            # shift forward: a[i+k]; shift backward: a[i-k]
            if k == 0:
                g += 2.0 * a
                continue
            # forward
            g[:n - k] += a[k:]
            # backward
            g[k:] += a[:n - k]
        # average
        g /= len(active_idx)
        # Exponentiated gradient update
        step = eta0 / np.sqrt(it)
        # Avoid overflow: clip step*g
        exp_factor = np.exp(-step * (g - g.mean()))  # shift to improve numerical stability
        a = a * exp_factor
        # Renormalise to sum 1
        sum_a = a.sum()
        if sum_a <= 0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / sum_a
        # occasional random jitter to escape local minima
        if rng.random() < 0.01:
            # random multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * noise
            a = a / a.sum()

        # not implementing restart for now.
    # After loop return best
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]

"""
Propose a step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.

Algorithm
---------
*   The vector is kept on the probability simplex (∑a_i = 1, a_i ≥ 0).
    The objective 2·n·max(a∗a) is scale‑invariant, therefore we may
    normalise the sum to one and only minimise the infinity‑norm of the
    self‑convolution.
*   A *mirror‑descent* (exponentiated‑gradient) scheme is used:
        a ← a ∘ exp(‑η·g)   →   renormalise.
    This update respects the simplex and avoids the need for an explicit
    projection.
*   g is a sub‑gradient of the maximal convolution entry.
    If K = {k | (a∗a)_k is maximal} then
        g_i = 1/|K| · Σ_{k∈K} ( a_{i+k} + a_{i‑k} ),
    where out‑of‑range accesses are treated as zero.
*   The step size η decays as η₀/√t (t = iteration counter).
*   To escape local plateaus a small amount of multiplicative Gaussian
    jitter is injected from time to time and a full restart (from the
    current best) is performed if no improvement is seen for a few
    minutes.
*   The routine works for any length; a moderate size (≈500–1500) is
    chosen automatically (the user can override it with ``n=…``).  The
    best vector found before the time budget expires is returned as a
    plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Smallest power‑of‑two length that can hold a full linear convolution
    of two vectors of length n (i.e. length 2·n‑1).
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution using FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a sub‑gradient of  max_k (a∗a)_k  with respect to a.
    The gradient is the average of the gradients of all active (maximal)
    convolution entries.
    """
    n = a.size
    cur_max = conv.max()
    # indices where the convolution attains (almost) the maximum
    eps = max(1e-12, cur_max * 1e-12)
    active = np.where(conv >= cur_max - eps)[0]

    g = np.zeros_like(a)
    for k in active:
        if k == 0:
            # (a∗a)_0 = a_0^2 , gradient = 2·a_0
            g[0] += 2.0 * a[0]
            continue
        # forward shift part: a_{i+k}
        end = n - k
        if end > 0:
            g[:end] += a[k:]
            # backward shift part: a_{i‑k}
            g[k:] += a[:end]
        # if k >= n the slices are empty – nothing to add
    if active.size > 0:
        g /= float(active.size)
    return g

        2. a decreasing √‑difference shape (a_i = √(i+1)‑√i);
        3. a random Dirichlet vector.
    The result is normalised to unit sum.
    """
    # 1. try the global construction
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    if cand.size != n:
                        # truncate / pad with zeros to the required length
                        if cand.size > n:
                            cand = cand[:n]
                        else:
                            pad = np.zeros(n, dtype=np.float64)
                            pad[:cand.size] = cand
                            cand = pad
                    cand = cand / cand.sum()
                    return cand
        except Exception:
            pass

    # 2. √‑difference decreasing shape (approximately 1/√i)
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 42,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size ------------------------------------------------------
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 800))          # a moderate size – fast iterations
    if n < 2:
        n = 2

    # ------------------------------------------------------------------
    # initialise a probability vector (∑a = 1) -------------------------
    # ------------------------------------------------------------------
    a = _initial_shape(n, rng)

    # ------------------------------------------------------------------
    # bookkeeping -------------------------------------------------------
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)
    best_a = a.copy()
    conv = _conv_self(a, fft_len)
    best_score = 2.0 * n * conv.max()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # safety margin
    last_improve = start_time

    # hyper‑parameters
    eta0 = 0.9                      # initial step for exponentiated gradient
    jitter_prob = 0.015             # probability of a stochastic jitter per iter
    restart_gap = 120.0             # seconds without improvement before a restart

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # --------------------------------------------------------------
        # evaluate current objective
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # keep the best solution seen so far
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # sub‑gradient of the max‑convolution entry
        # --------------------------------------------------------------
        g = _subgradient_max_conv(a, conv)

        # --------------------------------------------------------------
        # exponentiated‑gradient (mirror descent) update
        # --------------------------------------------------------------
        step = eta0 / np.sqrt(it)                # diminishing step size
        # work in log‑space for numerical stability
        log_a = np.log(a + 1e-30) - step * g
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum <= 0.0:
            # fallback – uniform vector (should never happen)
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # --------------------------------------------------------------
        # occasional stochastic perturbation – helps escaping plateaus
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            # multiplicative log‑normal noise
            noise = np.exp(rng.normal(0.0, 0.03, size=n))
            a *= noise
            a /= a.sum()

        # --------------------------------------------------------------
        # soft restart if we have been stuck for a while
        # --------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > restart_gap:
            # re‑initialise from the current best with a little jitter
            a = best_a.copy()
            log_a = np.log(a + 1e-30) + rng.normal(0.0, 0.12, size=n)
            a = np.exp(log_a)
            a /= a.sum()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector (values are already non‑negative and well
    # within the magnitude limits imposed by ``evaluate_sequence``)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
