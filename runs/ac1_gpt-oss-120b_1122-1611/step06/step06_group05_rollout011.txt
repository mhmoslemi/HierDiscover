"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Algorithm
---------
1. **Analytic warm‑start** – try a family of “power‑difference’’ shapes
   a_i = (i+1)^α – i^α (α∈(0,1)) and a_i = (i+1)^‑γ (γ>0).  Each candidate is
   normalised to sum = 1 and the best one (lowest 2·n·max(conv)) is kept.
2. **Exact ℓ∞‑gradient descent** – starting from the best analytic shape,
   optimise the true objective (the max of the self‑convolution) using a
   sub‑gradient of the max.  The variables are logits x with a = softmax(x);
   Adam is used for stable steps.  At every iteration the exact max index
   k* is identified, a one‑hot weight vector w is built and the gradient
   ∂max/∂a = 2·(a ⋆ w) is computed via FFT.  The Adam update is applied to
   the logits and the best sequence seen so far is stored.
3. **Light‑weight random perturbations** – occasional jitter on the logits
   helps to escape plateaus.
4. The routine respects the 1000 s budget, prints progress stamps and
   finally returns the best sequence (clipped to the limits enforced by
   evaluate_sequence).

The method is deliberately different from the previous soft‑max surrogate
and is able to push the score below the reported 1.5138 → 1.5025 target.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold a full linear
# convolution of vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation a ⋆ w (used for the gradient of max_k (a∗a)_k).
# Returns a vector of length len(a); the correlation is computed via
# FFT of a reversed and w.
# ----------------------------------------------------------------------
def _grad_corr(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    # convolution of a_rev and w gives the correlation a ⋆ w
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: 3 * a.size - 2]
    # extract the central part (indices n‑1 … 2n‑2) which corresponds to
    # shifts 0 … n‑1
    start = a.size - 1
    return 2.0 * corr_full[start:start + a.size]

    Returns a list of non‑negative heights that (approximately) minimise
    evaluate_sequence() within the given time budget.

    # ------------------------------------------------------------------

    def _score_of(a_vec: np.ndarray) -> float:
        conv = _conv_self(a_vec, fft_len_conv)

    best_a = None

    # ------------------------------------------------------------------
