Pseudo:

def project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho+1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimizer for the auto‑convolution max‑norm problem.

The score to minimise is
    score = 2·n·max_k (a * a)_k   with   sum_i a_i = 1 , a_i ≥ 0
(the denominator of ``evaluate_sequence`` cancels when the vector is
normalised, so we work with a probability simplex).

The algorithm proceeds in three stages:

1.  Enumerate several analytic families (monotone, symmetric power‑law,
    and differences of powers).  The best member of these families is
    taken as the initial point.

2.  Refine the candidate with a *projected gradient descent* on the
    exact max‑objective.  The sub‑gradient of ``max(conv)`` is obtained
    as a weighted correlation ``2·a ★ w`` where ``w`` is either a soft‑max
    over the convolution (early iterations, smooth) or a uniform weight
    on the active set (late iterations, exact).  After each gradient
    step the vector is projected back onto the probability simplex.

3.  Random exploration (block perturbations, pairwise mass transfers)
    and occasional “restarts” (jittering the current best) prevent
    stagnation.

All heavy work uses FFT‑based convolution/correlation (O(n log n)).  The
routine respects the supplied time budget (≈1000 s) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level numeric helpers
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length supporting a full linear convolution
    of two vectors of length ``n`` (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) using a single FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max with temperature ``beta`` (numerically stable)."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum   Σ_k w_k·(a*a)_k
    = 2·a ★ w   (correlation)   – returned as a vector of length len(a).

    Parameters
    ----------
    a : ndarray, shape (n,)
        Current probability vector.
    w : ndarray, shape (2·n‑1,)
        Weight vector over the full convolution.

    Returns
    -------
    grad : ndarray, shape (n,)
        Gradient of the weighted objective.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1                       # length of the full convolution
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    # relevant slice corresponds to indices n‑1 … n‑1+n‑1
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # All entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value for a normalised probability vector."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())

# Analytic families (monotone and symmetric)
# ----------------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^r - i^r  (decreasing for 0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_shape_p(p: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^(-p) - (i+2)^(-p)  (p>0)."""
    k = np.arange(1, n + 2, dtype=np.float64)          # 1 … n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_sym_shape_p(p: float, n: int) -> np.ndarray | None:
    """Symmetric U‑shape: a_i depends on distance to the nearest edge,
    a_i ∝ (d+1)^(-p) where d = min(i, n-1-i)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1.0) ** (-p)
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_sym_shape_r(r: float, n: int) -> np.ndarray | None:
    """Symmetric shape built from differences of powers:
       a_i = (d+1)^r - d^r  where d = min(i, n‑1‑i)."""
    d = np.minimum(np.arange(n), np.arange(n - 1, -1, -1))
    a = (d + 1.0) ** r - d ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # Return the best vector (clipped to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
