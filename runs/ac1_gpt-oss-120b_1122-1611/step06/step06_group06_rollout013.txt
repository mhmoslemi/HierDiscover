def _project_simplex(v):
    # v: numpy array
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # If all these are false, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Algorithm (high‑level)
----------------------
1. Build a diverse pool of analytical shapes (sqrt‑difference, power‑difference,
   exponential, concave‑power) together with many random Dirichlet samples.
   All vectors are normalised to sum = 1 and scored with a fast FFT‑based
   convolution.  The best vector from the pool is taken as a start point.

2. Refine the start point with a **projected additive gradient descent**
   on a smooth approximation of the L∞‑objective.
   * At each iteration the self‑convolution is computed.
   * The K largest convolution entries are taken as the active set and an
     (optional) soft‑max weighting is built.
   * The sub‑gradient of the weighted sum of convolutions is obtained by a
     correlation computed via FFT.
   * An Adam‑style update (first‑ and second‑moment estimates) is applied
     to the probability vector, after which the vector is projected back onto
     the probability simplex.
   * Occasionally a random block scaling or a pairwise mass‑transfer move is
     performed to avoid stagnation.

3. When only ~20 s remain a short simulated‑annealing phase performs
   additional random mass‑transfer proposals and accepts worsening moves with a
   temperature that decays exponentially.

4. The best vector seen during the whole run is clipped to the limits imposed
   by ``evaluate_sequence`` and returned as a plain ``list[float]``.

All heavy work is done with NumPy FFTs; the routine respects a 1000 s budget,
prints intermediate scores and always returns a feasible sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT length, convolution, gradient, simplex projection)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Exact self‑convolution a*a using real FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k.
    Implements 2·a ⋆ w where ⋆ denotes discrete correlation.
    """
    n = a.size
    L = w.size                 # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex {x ≥ 0, Σx = 1}.
    Implements the algorithm of [Wang & Carreira‑Perpiñán 2013].
    """
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_fast(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Fast evaluation of 2·n·max(a∗a) for a vector that is already normalised
    (sum = 1).  This equals the value returned by evaluate_sequence().
    """
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Shape generators (all return a non‑negative vector, not normalised)
# ----------------------------------------------------------------------
def _shape_sqrt_diff(n: int) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    return np.sqrt(idx + 1.0) - np.sqrt(idx)


def _shape_power_diff(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return (i + 1.0) ** (-p) - (i + 2.0) ** (-p)


def _shape_exponential(n: int, lam: float) -> np.ndarray:
    return (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))


def _shape_concave_power(n: int, alpha: float, delta: float) -> np.ndarray:
    """
    a_i = (i+δ+1)^α - (i+δ)^α   with 0 < α < 1  (concave increasing f → decreasing a).
    """
    i = np.arange(n, dtype=np.float64)
    return (i + delta + 1.0) ** alpha - (i + delta) ** alpha


def _shape_log_diff(n: int, delta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return np.log(i + delta + 2.0) - np.log(i + delta + 1.0)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Normalisation helper
    # --------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # --------------------------------------------------------------
    # 1) Build a rich pool of starting points
    # --------------------------------------------------------------
    pool = []

    # Analytic sqrt‑difference (the classic start)
    pool.append(_normalise(_shape_sqrt_diff(n)))

    # Power‑difference families
    for p in np.linspace(0.2, 2.0, 24):
        pool.append(_normalise(_shape_power_diff(n, p)))

    # Exponential families
    for lam in np.linspace(0.0005, 0.05, 24):
        pool.append(_normalise(_shape_exponential(n, lam)))

    # Concave‑power families (α in (0,1), shift δ)
    alphas = np.linspace(0.25, 0.95, 12)
    deltas = np.linspace(0.0, n / 10.0, 5)
    for α in alphas:
        for δ in deltas:
            pool.append(_normalise(_shape_concave_power(n, α, δ)))

    # Log‑difference families (another slowly varying shape)
    for δ in np.linspace(0.0, n / 10.0, 5):
        pool.append(_normalise(_shape_log_diff(n, δ)))

    # Random Dirichlet samples (diverse)
    for _ in range(200):
        alpha = rng.uniform(0.2, 2.0)
        sample = rng.gamma(alpha, 1.0, size=n)
        pool.append(_normalise(sample))

    # If a previous construction exists, use it as a starter as well
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                pool.append(_normalise(prev))
        except Exception:
            pass

    # --------------------------------------------------------------
    # 2) Pick the best candidate from the pool (fast scoring)
    # --------------------------------------------------------------
    best = None
    best_score = np.inf
    for vec in pool:
        sc = _score_fast(vec, n, fft_len)
        if sc < best_score:
            best_score = sc
            best = vec.copy()
    print(f"[init] best pool score = {best_score:.9f}")

    # --------------------------------------------------------------
    # 3) Projected additive (Adam‑style) gradient descent
    # --------------------------------------------------------------
    cur = best.copy()
    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    it = 0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 20.0)   # reserve last ≈20 s for SA

    # hyper‑parameters for the descent
    eta_start, eta_end = 0.8, 0.001   # learning‑rate schedule (linear)
    num_active = 12                  # size of the active set for the soft‑max
    prob_block = 0.037               # random block scaling probability
    prob_pair = 0.09                 # random pairwise transfer probability

    while time.monotonic() < deadline:
        it += 1
        # ---- convolution & current score ----
        conv = _conv_self(cur, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ---- keep global best ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.9f}")

        # ---- schedule for learning rate ----
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # ---- build active set (largest convolution entries) ----
        if conv.size <= num_active:
            act_idx = np.arange(conv.size)
        else:
            act_idx = np.argpartition(-conv, num_active)[:num_active]

        # soft‑max weights over the active set (β grows with time)
        beta = 0.1 + 4999.9 * frac               # linearly from 0.1 → 5000
        rel = conv[act_idx] - cur_max
        w_raw = np.exp(beta * rel)
        w_raw_sum = w_raw.sum()
        w = np.zeros_like(conv)
        if w_raw_sum > 0.0:
            w[act_idx] = w_raw / w_raw_sum
        else:
            w[act_idx] = 1.0 / act_idx.size

        # ---- gradient of the soft‑max surrogate ----
        grad = _gradient_correlation(cur, w)

        # ---- Adam update (on the probability vector) ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)

        # gradient descent step (move opposite to the gradient)
        cur = cur - step
        # project back onto the simplex
        cur = _project_simplex(cur)

        # ---- occasional random block perturbation ----
        if rng.random() < prob_block:
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.22))
            cur[start:start + width] *= factor
            cur = _project_simplex(cur)

        # ---- occasional pairwise mass transfer ----
        if rng.random() < prob_pair:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur[i] > 1e-14:
                delta = rng.random() * cur[i] * 0.45
                cur[i] -= delta
                cur[j] += delta
                cur = _project_simplex(cur)

    # --------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning (use the remaining budget)
    # --------------------------------------------------------------
    sa_deadline = start_time + float(budget_s)
    cur_sa = best.copy()
    cur_sa_score = best_score
    T_start, T_end = 5e-3, 1e-6
    it_sa = 0

    while time.monotonic() < sa_deadline:
        it_sa += 1
        now = time.monotonic()
        frac = (now - deadline) / (sa_deadline - deadline) if sa_deadline > deadline else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        # propose a random mass‑transfer move
        i, j = rng.integers(0, n, size=2)
        if i != j and cur_sa[i] > 1e-14:
            delta = rng.random() * cur_sa[i] * 0.5
            new_vec = cur_sa.copy()
            new_vec[i] -= delta
            new_vec[j] += delta

            new_score = _score_fast(new_vec, n, fft_len)
            improve = new_score < cur_sa_score
            if improve or rng.random() < np.exp(-(new_score - cur_sa_score) / max(T, 1e-12)):
                cur_sa = new_vec
                cur_sa_score = new_score
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best = new_vec.copy()
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        # occasional gentle smoothing (helps to flatten tiny spikes)
        if it_sa % 6000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            cur_sa = smoothed
            cur_sa_score = _score_fast(cur_sa, n, fft_len)

    # --------------------------------------------------------------
    # 5) Return the best sequence respecting evaluate_sequence's caps
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
