def propose_candidate(seed=42, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    n = int(kwargs.get('n', 3800))
    if n < 2: n=2
    fft_len = _fft_len_for_n(n)
    # helper functions defined above.

    # initial best: analytic shape
    best_a = _analytic_start(n)
    best_score = _compute_score(best_a, fft_len, n)

    start_time = time.time()
    deadline = start_time + max(0.0, budget_s - 5.0)  # reserve 5 sec for polishing and final clamping.

    # Phase 1: random param search
    # allocate maybe 60% of time to param search
    param_deadline = start_time + 0.6 * (deadline - start_time)
    while time.time() < param_deadline:
        # sample p, shift, maybe exponentials
        p = rng.uniform(0.3, 1.2)
        shift = rng.uniform(0.0, 20.0) # maybe integer or float
        # produce vector
        i = np.arange(n, dtype=np.float64)
        a = (i + shift + 1.0) ** (-p)
        # ensure non-negative
        a_sum = a.sum()
        if a_sum == 0: continue
        a /= a_sum
        # compute score (fast)
        score = _compute_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            elapsed = time.time() - start_time
            print(f"[{elapsed:.2f}s] new best param score={best_score:.6f}, p={p:.4f}, shift={shift:.2f}")

    # Phase 2: polishing via exponentiated gradient for best candidate
    # Let's allocate remaining time minus 0.5sec for final clipping
    polish_deadline = deadline
    best_a, best_score = _polish(best_a, n, fft_len, polish_deadline, best_score)

    # final clipping
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a.tolist()]

import numpy as np

def _fft_len_for_n(n):
    size = 2*n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2*a.size - 1]
    return conv

def _score_from_conv(conv, n):
    return 2.0 * n * float(conv.max())

def _compute_score(a, fft_len, n):
    conv = _conv_self(a, fft_len)
    return _score_from_conv(conv, n)

def _analytic_start(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a

            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None

def _gradient_correlation(a, w):
    """
    Gradient of Σ_k w_k (a*a)_k = 2 * (a ⋆ w) where ⋆ denotes
    correlation of a with w.  Both a and w are real.
    """
    n = a.size
    L = w.size               # should be 2*n-1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

def _polish_exponential(a, n, fft_len, deadline, start_time, init_score):
    """
    Mirror‑descent (exponentiated‑gradient) polishing.
    The soft‑max temperature β grows from β0 to β1 while the step
    size η shrinks from η0 to η1.
    """
    best = a.copy()
    best_score = init_score
    # hyper‑parameters (tuned heuristically)
    eta0 = 0.3
    eta1 = 0.001
    beta0 = 5.0
    beta1 = 5000.0
    max_active = 40   # cap of active indices used for the soft‑max

    while time.time() < deadline:
        # progress dependent schedules
        now = time.time()
        elapsed = now - start_time
        total = deadline - start_time
        if total <= 0:
            break
        eta = eta0 * (1.0 - elapsed / total) + eta1 * (elapsed / total)
        beta = beta0 + (beta1 - beta0) * (elapsed / total)

        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())

        # active set: indices whose value is within 0.2 % of the current maximum
        tol = cur_max * 0.998
        active = np.where(conv >= tol)[0]
        if active.size > max_active:
            # keep the largest values only
            idx = np.argpartition(conv, -max_active)[-max_active:]
            active = idx

        # soft‑max weights on the active set
        w = np.zeros_like(conv)
        if active.size:
            # use shifted values to improve numerical stability
            vals = conv[active] - cur_max
            exp_vals = np.exp(beta * vals)
            probs = exp_vals / exp_vals.sum()
            w[active] = probs
        else:
            # fallback: use all indices
            vals = conv - cur_max
            exp_vals = np.exp(beta * vals)
            probs = exp_vals / exp_vals.sum()
            w = probs

        grad = _gradient_correlation(a, w)

        a = a * np.exp(-eta * grad)
        a = np.clip(a, 0.0, None)
        s = a.sum()

        else:

            best = a.copy()
            print(f"[{now - start_time:6.2f}s] polish new best = {best_score:.6f}")

    return best, best_score

        if s == 0.0:

        if score < best_score:

    # ----------------------------------------------------------------------
    # Final clipping – required by evaluate_sequence()
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # Convert to plain Python list of floats
    return [float(v) for v in best.tolist()]

    entries.  The gradient of this weighted sum is obtained by a single

    sharper, driving the vector towards a true local minimum of the

        # indices whose value is close to the current maximum
        tol = cur_max * 0.998
        active = np.where(conv >= tol)[0]
        if active.size > max_active:
            # keep only the largest values
            active = np.argpartition(conv, -max_active)[-max_active:]

        # soft‑max weights on the active set
        w = np.zeros_like(conv)
        if active.size:
            vals = conv[active] - cur_max
            exp_vals = np.exp(beta * vals)
            probs = exp_vals / exp_vals.sum()
            w[active] = probs
        else:
            # fallback (should never happen)
            vals = conv - cur_max
            exp_vals = np.exp(beta * vals)
            probs = exp_vals / exp_vals.sum()
            w = probs

        # gradient of the weighted sum
        grad = _gradient_correlation(a, w)

        # exponentiated‑gradient update and projection onto the simplex
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s

        # evaluate the true objective after the update
        cur_score = _score_from_conv(_conv_self(a, fft_len), n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{now - polish_start:6.2f}s] polish new best = {best_score:.6f}")

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget elapses.
    The routine combines a cheap random search over a shifted power‑law
    family with a mirror‑descent polishing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Initial vector (analytic start or a previously stored construction)
    # --------------------------------------------------------------
    a = _analytic_start(n)
    cand_global = _maybe_use_global()
    if cand_global is not None:
        # pick the better start (fast internal score)
        score_analytic = _compute_score(a, fft_len, n)
        score_global = _compute_score(cand_global, _fft_len_for_n(cand_global.size), cand_global.size)
        if score_global < score_analytic:
            a = cand_global
            n = a.size
            fft_len = _fft_len_for_n(n)

    best = a.copy()
    best_score = _compute_score(best, fft_len, n)

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # small safety margin

    # --------------------------------------------------------------
    # Phase 1 – random search in the shifted power‑law family
    # --------------------------------------------------------------
    # Use roughly 60 % of the total budget for this cheap exploration
    phase1_deadline = start_time + 0.60 * (deadline - start_time)

    while time.time() < phase1_deadline:
        # sample parameters
        p = rng.uniform(0.30, 1.30)          # exponent (controls decay)
        shift = rng.uniform(0.0, 30.0)       # shift of the index (avoids a huge first entry)

        i = np.arange(n, dtype=np.float64)
        a_candidate = (i + shift + 1.0) ** (-p)
        s = a_candidate.sum()
        if s == 0.0:
            continue
        a_candidate /= s

        score = _compute_score(a_candidate, fft_len, n)
        if score < best_score:
            best_score = score
            best = a_candidate.copy()
            elapsed = time.time() - start_time
            print(
                f"[{elapsed:6.2f}s] new best (param) score = {best_score:.6f}  p={p:.4f}  shift={shift:.2f}"
            )

    # --------------------------------------------------------------
    # Phase 2 – deterministic polishing (mirror‑descent)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        polish_start = time.time()
        polish_deadline = polish_start + remaining
        best, best_score = _polish_exponential(
            best, n, fft_len, polish_deadline, polish_start, best_score
        )

    # --------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)

    # Return a plain Python list of floats
    return [float(v) for v in best.tolist()]
