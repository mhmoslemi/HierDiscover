import time
import numpy as np
from scipy.special import logsumexp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable softmax of logits."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method optimises a smooth surrogate of the
    infinity‑norm of the self‑convolution using Adam on the log‑softmax
    parametrisation of the sequence.  A linear schedule drives the
    surrogate towards the true max, while periodic restarts from the
    best‑found point avoid stagnation.
    """
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # Choose a length n.  Prefer the length of a previously found solution
    # if available, otherwise fall back to a moderate size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        base = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        else:
            base /= base.sum()
        n = max(2000, base.size)          # never go below 2000
        # resample to exactly n (might be the same)
        if base.size != n:
            # linear interpolation to the new size
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(new_x, old_x, base)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        else:
            a = base.copy()
    else:
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Parameterisation: x are unconstrained logits, a = softmax(x)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(a + eps)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # bookkeeping for the best solution seen so far
    best_a = a.copy()
    best_score = 2.0 * n * np.fft.irfft(np.fft.rfft(a, n=_next_pow2(2 * n - 1)) *
                                       np.fft.rfft(a, n=_next_pow2(2 * n - 1)),
                                       n=_next_pow2(2 * n - 1))[:2 * n - 1].max()
    last_improve = start

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    # Adam
    lr = 0.6
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # Soft‑max temperature (beta) schedule – moves from smooth to sharp
    beta_start = 5.0
    beta_end = 200.0
    beta_ramp = 2000          # iterations over which beta is increased

    # Restart control
    RESTART_WAIT = 30.0       # seconds without improvement before a restart

    # FFT lengths (pre‑computed to avoid repeated power‑of‑two work)
    fft_len_conv = _next_pow2(2 * n - 1)          # for a*a
    fft_len_corr = _next_pow2(3 * n - 2)          # for w * a (correlation)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Convert logits to a probability vector (sum = 1)
        # --------------------------------------------------------------
        a = _softmax_logits(x)          # shape (n,)

        # --------------------------------------------------------------
        # Compute the linear self‑convolution a*a using FFT
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[:2 * n - 1]   # length 2n‑1

        # --------------------------------------------------------------
        # Current score (exact evaluation without the extra checks)
        # --------------------------------------------------------------
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max

        # Keep the best record
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft‑max over the convolution – a smooth surrogate for max(c)
        # --------------------------------------------------------------
        beta = (beta_end if it > beta_ramp
                else beta_start + (beta_end - beta_start) * it / beta_ramp)

        # Robust log‑sum‑exp
        scaled = beta * conv_full
        lse = logsumexp(scaled)
        w = np.exp(scaled - lse)          # shape (2n‑1,) , sums to 1

        # --------------------------------------------------------------
        # Gradient of the surrogate w.r.t. a (correlation of w and a)
        # --------------------------------------------------------------
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)
        # Valid part of the correlation has length n and starts at index n‑1
        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]   # shape (n,)

        # --------------------------------------------------------------
        # Chain rule to obtain gradient w.r.t. logits x
        # --------------------------------------------------------------
        dot = np.dot(grad_a, a)           # scalar
        grad_x = a * (grad_a - dot)       # shape (n,)

        # --------------------------------------------------------------
        # Adam update on the logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # --------------------------------------------------------------
        # Occasionally add a tiny random jitter to escape plateaus
        # --------------------------------------------------------------
        if it % 500 == 0:
            x += rng.normal(0.0, 0.02, size=n)

        # --------------------------------------------------------------
        # Hard restart if no improvement for a while
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > RESTART_WAIT:
            # Re‑initialise logits around the current best solution
            x = np.log(best_a + eps) + rng.normal(0.0, 0.09, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector we have found (already normalised, safe‑clipped)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(val) for val in best_a]

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s
