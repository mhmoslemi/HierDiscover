def _gradient_correlation(a, w):
    n = a.size
    L = w.size  # 2*n-1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

Implementation of project onto simplex:
def _project_simplex(v):
    """Projects v onto the probability simplex {x >=0, sum=1}."""
    # Sorting descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / (np.arange(len(v)) + 1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):

def _gradient_correlation(a, w):
    """Compute gradient of softmax-weighted convolution: 2 * correlation of a and w."""
    n = a.size
    L = w.size  # should be 2*n-1
    # convolution length needed: L + n - 1 (since correlating a_rev length n with w length L)
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reverse a
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    # start index for correlation aligning
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

"""
Propose a non‑negative step‑function that (approximately) minimises

    score = 2·n·max_k (a∗a)_k / (∑a_i)^2

The algorithm consists of three stages:

1. **Deterministic warm‑start** – try several analytic families
   a_i = (i+α+1)^q − (i+α)^q (including the classic √‑difference
   case q=½) and a previously stored construction (if any).  The best
   of these is taken as the initial vector.

2. **Large‑scale smooth optimisation** – minimise a smooth
   approximation of the L∞‑objective,
        L_β(a) = (1/β)·log∑_k exp(β·(a∗a)_k) .
   The gradient of L_β is a weighted correlation
        ∇L_β = 2·a⋆w ,   w_k ∝ exp(β·(a∗a)_k) .
   We perform a **projected exponentiated‑gradient** (multiplicative)
   update on the simplex, slowly raising β (≈10 → ≈8000) and
   decreasing the step‑size η (≈0.5 → ≈0.001).  The whole time budget
   (≈1000 s) is spent in this phase; occasional tiny log‑normal jitter
   helps escape plateaus.

3. **Return the best vector seen** – clipped to the limits required by
   ``evaluate_sequence``.  The vector is kept normalised (∑a_i = 1),
   which is scale‑invariant for the objective.

The implementation uses only NumPy and the standard library, relies on
FFT for fast convolution/correlation, and respects the 1000 s time
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via real FFT (return length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a vector normalised to sum = 1."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of the smooth max (soft‑max) objective
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k  with respect to a.
    Implements 2·a⋆w via FFT (a⋆w is the correlation of a and w).
    """
    n = a.size
    L = w.size               # should be 2·n‑1
    # required length for a_rev * w convolution
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Simple analytic start (√‑difference)
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """Classic start a_i = √(i+1)−√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optional previous construction from the global namespace
# ----------------------------------------------------------------------
def _maybe_use_global() -> np.ndarray | None:
    """If a global variable `height_sequence_1` exists, return it normalised to sum = 1."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 1. Deterministic warm‑start (analytic families + possible global)
    # ------------------------------------------------------------------
    best_a = _analytic_start(n)
    best_score = _compute_score(best_a, fft_len, n)

    # try a previously stored vector (if any)
    cand_global = _maybe_use_global()
    if cand_global is not None and cand_global.size == n:
        score_global = _compute_score(cand_global, _fft_len_for_n(cand_global.size), n)
        if score_global < best_score:
            best_a = cand_global.copy()
            best_score = score_global

    # sweep a small family a_i = (i+α+1)^q – (i+α)^q
    i = np.arange(n, dtype=np.float64)
    q_vals = np.linspace(0.20, 0.80, 13)          # 13 values between 0.2 and 0.8
    alpha_vals = np.linspace(0.0, 5.0, 6)         # 6 values between 0 and 5
    for q in q_vals:
        for alpha in alpha_vals:
            a = (i + 1.0 + alpha) ** q - (i + alpha) ** q
            s = a.sum()
            if s <= 0.0:
                continue
            a /= s
            score = _compute_score(a, fft_len, n)
            if score < best_score:
                best_a = a.copy()
                best_score = score

    # ------------------------------------------------------------------
    # 2. Large‑scale smooth optimisation (exponentiated‑gradient)
    # ------------------------------------------------------------------
    a = best_a.copy()
    best = a.copy()
    best_score = _compute_score(best, fft_len, n)

    start_time = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # hyper‑parameters for the schedule
    eta_start = 0.5          # initial multiplicative step size
    eta_end   = 0.001        # final step size
    beta_start = 10.0        # soft‑max temperature at start
    beta_end   = 8000.0      # final temperature (strongly focuses on the max)

    iteration = 0
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # Current objective
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # Schedule for η and β
        # --------------------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0
        eta = eta_start * (1.0 - frac) + eta_end * frac
        # quadratic increase for β gives a gentle early phase
        beta = beta_start + (beta_end - beta_start) * (frac ** 2)

        # --------------------------------------------------------------
        # Soft‑max weights w_k ∝ exp(β·(conv_k − cur_max))
        # --------------------------------------------------------------
        exp_args = beta * (conv - cur_max)          # ≤ 0, numerically stable
        w = np.exp(exp_args)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # Gradient of the smoothed objective
        # --------------------------------------------------------------
        grad = _gradient_correlation(a, w)

        # --------------------------------------------------------------
        # Exponentiated‑gradient multiplicative update
        # --------------------------------------------------------------
        # Clip the exponent to avoid overflow / underflow
        exp_update = -eta * grad
        exp_update = np.clip(exp_update, -50.0, 50.0)
        factor = np.exp(exp_update)
        a = a * factor

        # --------------------------------------------------------------
        # Projection onto the probability simplex (simple normalisation)
        # --------------------------------------------------------------
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --------------------------------------------------------------
        # Occasional tiny jitter to escape tiny plateaus
        # --------------------------------------------------------------
        iteration += 1
        if iteration % 5000 == 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * jitter
            a = a / a.sum()

        # --------------------------------------------------------------
        # Early exit guard
        # --------------------------------------------------------------
        if now > deadline - 0.1:
            break

    # ------------------------------------------------------------------
    # 3. Return the best candidate seen (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
