Propose a step‑function that improves the bound.
--------------------------------------------------------------------
Algorithm
---------

* Phase 1 (≈30 % of the time) – **weighted mirror descent**.  
  The current self‑convolution `c = a*a` is smoothed by a soft‑max
  with a temperature `eps`.  The resulting weights `w` give a
  differentiable surrogate of `max(c)`.  The gradient of the
  surrogate w.r.t. the distribution `a` is computed analytically
  (`∂c_k/∂a_i = 2·a_{k‑i}`) and applied with an exponentiated‑gradient
  (entropy mirror) update.  Both the temperature `eps` and the step
  size `eta` decay linearly so the method starts with a strong
  smoothing (global exploration) and ends with a near‑exact sub‑gradient
  step (fine tuning).

* Phase 2 (≈70 % of the time) – **fast stochastic refinement**.  
  The same donor/recipient mass‑transfer moves as in the reference
  implementation are used, but the temperature schedule is lighter
  and the move is accepted by a Metropolis rule.  Periodic full
  recomputation of the convolution eliminates numerical drift and a
  light block‑jitter is applied occasionally.

* Restarts – if no improvement is seen for 25 s the current best
  solution is perturbed with a modest log‑normal noise and optimisation
  continues.

All updates keep the vector non‑negative and on the simplex, which
matches the expectations of `evaluate_sequence`.  The routine stops a
few seconds before the 1000 s budget and returns the best sequence it
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new a and c."""
    n = a.size
    # new a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # incremental update of convolution
    c_new = c.copy()
    # shift of donor i (subtract) and recipient j (add)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ part
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise the probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            n = 3840
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # default length – large enough to give a fine resolution
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the current state
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)                # current self‑convolution
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time
    # allocate 30 % of the time to weighted mirror descent
    phase1_end = start_time + 0.30 * total_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Phase‑1 (mirror descent) schedule
    eta_start = 2.0            # initial step‑size for exp‑gradient
    eta_end = 1e-3             # final step‑size
    eps_min = 1e-12            # lower bound for the soft‑max temperature

    # Phase‑2 (stochastic) schedule
    temp_start = 0.09
    temp_end = 1e-6

    move_counter = 0
    RECOMPUTE_EVERY = 300

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – weighted mirror descent (entropy‑mirror with soft‑max)
        # --------------------------------------------------------------
        if now < phase1_end:
            # Linear schedules for step size and soft‑max temperature
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # soft‑max temperature proportional to current dynamic range
            range_c = c.max() - c.min()
            eps = max(range_c * max(0.2 * (1.0 - frac), 0.01), eps_min)

            # soft‑max (stable)
            max_c = c.max()
            w = np.exp((c - max_c) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:          # pathological – fall back to hard max
                w = np.zeros_like(c)
                w[int(c.argmax())] = 1.0
                w_sum = 1.0
            w = w / w_sum               # shape (2n‑1,)

            # gradient of the smooth max w.r.t. a
            a_rev = a[::-1]             # reversed a, length n
            fw = np.fft.rfft(w, n=fft_len_grad)
            fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: 3 * n - 1]
            grad = 2.0 * conv_full[n - 1:n - 1 + n]   # length n

            # exponentiated‑gradient update on the simplex
            factor = np.exp(-eta * grad)
            # avoid extreme underflow / overflow
            factor = np.clip(factor, 1e-40, 1e40)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

            # recompute convolution and current score
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        else:
            # --------------------------------------------------------------
            # Phase 2 – fast stochastic refinement (simulated annealing)
            # --------------------------------------------------------------
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # locate current peak of the convolution
            k_max = int(c.argmax())

            # contributions of each index to that peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # choose donor proportional to its contribution (or uniformly)
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                continue

            # recipient favours small entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            # apply move (incremental update)
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # Periodic full recomputation to avoid drift
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter (≈2 % of the iterations)
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 25.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
