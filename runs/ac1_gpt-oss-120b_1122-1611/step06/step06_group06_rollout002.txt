"""
Propose a non‑negative step‑function (vector of heights) that minimises
``evaluate_sequence``.  

The algorithm consists of three stages:

1. **Diverse initial pool** – analytic √‑difference, power‑difference,
   exponential families and random Dirichlet draws.  The best among them
   (according to the exact L∞ objective) seeds the optimisation.

2. **Smooth‑max Adam optimisation** – the discrete max of the self‑convolution
   is approximated by a temperature‑controlled soft‑max
   `L = (1/α)·log Σ exp(α·conv)`.  The gradient of `L` w.r.t. the
   coefficients is computed by a single FFT‑based correlation.
   Adam with a linearly decaying learning rate and a linearly increasing
   temperature (α) performs projected gradient descent onto the probability
   simplex.

3. **Simulated‑annealing fine‑tuning** – with the remaining time a simple
   Metropolis–Hastings walk (mass transfers between two coordinates) is run.
   Accepted moves are saved if they improve the true objective
   `2·n·max(conv)`.  Occasional smoothing moves help to escape shallow traps.

The routine respects a total time budget (default 1000 s), prints progress
messages and finally returns the best sequence found as a plain list of
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power‑of‑two length sufficient for a convolution of length m
# ----------------------------------------------------------------------
def _next_pow2(m: int) -> int:
    """Return the smallest power of two ≥ m."""
    return 1 << ((m - 1).bit_length())

# ----------------------------------------------------------------------
# Self‑convolution via FFT (exact, length 2·n‑1)
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray) -> np.ndarray:
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return conv

# ----------------------------------------------------------------------
# Gradient of Σ_k w_k·(a∗a)_k  = 2·a ⋆ w   (correlation with weight vector w)
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Return ∇_a Σ_k w_k·(a∗a)_k ."""
    n = a.shape[0]
    L = w.shape[0]                       # = 2·n‑1
    fft_len = _next_pow2(L + n - 1)       # length needed for full correlation
    w_fft = np.fft.rfft(w, n=fft_len)
    a_rev_fft = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σ x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries are zero or negative
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Main optimisation routine: propose_candidate()
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising `evaluate_sequence`.
    Returns the best sequence discovered before the time budget expires.
    """

    start_time = time.monotonic()

    # --------------------------------------------------------------
    # Problem size (use previous best length if available)
    # --------------------------------------------------------------
    prev_seq = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                prev_seq = arr
        except Exception:
            prev_seq = None

    n = int(kwargs.get("n", prev_seq.size if prev_seq is not None else 3800))
    if n < 2:
        n = 2

    # --------------------------------------------------------------
    # Build a diverse initial pool of candidate vectors (sum = 1)
    # --------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s <= 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    candidates = []

    # a) Analytic √‑difference shape
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    candidates.append(_normalise(a_sqrt))

    # b) Previous construction, if any
    if prev_seq is not None:
        candidates.append(_normalise(prev_seq[:n]))

    # c) Power‑difference families a_i = (i+1)^-p - (i+2)^-p
    for p in np.linspace(0.2, 2.0, 12):
        a = ((np.arange(n, dtype=np.float64) + 1) ** (-p)
             - (np.arange(n, dtype=np.float64) + 2) ** (-p))
        candidates.append(_normalise(a))

    # d) Exponential families a_i = (1−e^{−λ})·e^{−λ i}
    for lam in np.linspace(0.0005, 0.02, 12):
        a = (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))
        candidates.append(_normalise(a))

    # e) Random Dirichlet draws
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0)
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(_normalise(a))

    # --------------------------------------------------------------
    # Pick the best candidate from the pool (exact objective)
    # --------------------------------------------------------------
    best = None
    best_score = np.inf
    for a in candidates:
        conv = _conv_self(a)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best = a.copy()
    print(f"[init] best score = {best_score:.9f}")

    # --------------------------------------------------------------
    # Stage 1 – Smooth‑max Adam optimisation (≈75 % of the time budget)
    # --------------------------------------------------------------
    cur = best.copy()
    m = np.zeros_like(cur)          # first moment
    v = np.zeros_like(cur)          # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    it = 0

    total_grad_time = budget_s * 0.75
    grad_deadline = start_time + total_grad_time
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)   # needed for correlation

    while time.monotonic() < grad_deadline:
        now = time.monotonic()
        # linear schedules
        frac = (now - start_time) / total_grad_time
        lr = 0.5 * (1.0 - frac) + 0.02 * frac           # learning rate
        alpha = 0.5 + (200.0 - 0.5) * frac              # soft‑max temperature

        # ---- current self‑convolution ---------------------------------
        af = np.fft.rfft(cur, n=fft_len_self)
        conv = np.fft.irfft(af * af, n=fft_len_self)[: 2 * n - 1]
        cur_max = conv.max()

        # ---- soft‑max weights over the convolution entries -------------
        w_raw = np.exp(alpha * (conv - cur_max))       # numerical stability
        w = w_raw / w_raw.sum()

        # ---- gradient via correlation ----------------------------------
        w_fft = np.fft.rfft(w, n=fft_len_grad)
        a_rev_fft = np.fft.rfft(cur[::-1], n=fft_len_grad)
        corr_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len_grad)[: (2 * n - 1) + n - 1]
        start = n - 1
        grad = 2.0 * corr_full[start:start + n]

        # ---- Adam update -----------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        cur = cur - step

        # occasional random perturbation to avoid stagnation
        if rng.random() < 0.01:
            cur += 0.05 * rng.standard_normal(n)

        # ---- projection onto the simplex --------------------------------
        cur = _project_simplex(cur)

        # ---- evaluate true objective for the new point ------------------
        af_new = np.fft.rfft(cur, n=fft_len_self)
        conv_new = np.fft.irfft(af_new * af_new, n=fft_len_self)[: 2 * n - 1]
        cur_score = 2.0 * n * conv_new.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur.copy()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.9f}")

        it += 1

    # --------------------------------------------------------------
    # Stage 2 – Simulated‑annealing fine‑tuning (remaining time)
    # --------------------------------------------------------------
    sa_deadline = start_time + budget_s
    cur_sa = best.copy()
    cur_sa_score = best_score
    T_start, T_end = 5e-3, 1e-6
    while time.monotonic() < sa_deadline:
        now = time.monotonic()
        # exponential cooling based on the fraction of SA time used
        frac = (now - grad_deadline) / (sa_deadline - grad_deadline) if sa_deadline > grad_deadline else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        # propose a random mass‑transfer move
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_sa[i] < 1e-14:
            continue
        delta = rng.random() * cur_sa[i] * 0.5   # move at most half of a[i]
        new_a = cur_sa.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # evaluate candidate
        af = np.fft.rfft(new_a, n=fft_len_self)
        conv = np.fft.irfft(af * af, n=fft_len_self)[: 2 * n - 1]
        new_score = 2.0 * n * conv.max()

        # Metropolis acceptance
        if new_score < cur_sa_score or rng.random() < np.exp(-(new_score - cur_sa_score) / max(T, 1e-12)):
            cur_sa = new_a
            cur_sa_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = new_a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        # occasional gentle smoothing (helps to flatten tiny peaks)
        if rng.random() < 0.001:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode="same")
            smoothed = smoothed / smoothed.sum()
            cur_sa = smoothed
            cur_sa_score = 2.0 * n * (_conv_self(cur_sa).max())

    # --------------------------------------------------------------
    # Return the best sequence (clipped to evaluate_sequence limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
