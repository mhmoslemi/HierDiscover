"""
Propose a step‑function using smooth‑max mirror descent.

Algorithm
---------
1. **Initialisation** – start from the best known sequence (if present)
   or from a random Dirichlet vector.  Normalise to sum = 1.

2. **Smooth‑max mirror descent** – repeatedly
   * compute the self‑convolution `c = a*a`,
   * build a soft‑max weight `w_k ∝ exp(α·c_k)` (α grows from 0.5 to 18),
   * compute the gradient of the smoothed peak  
         g_i = 2·∑_k w_k·a_{k‑i}  (via a single FFT‑based convolution),
   * apply an exponentiated‑gradient update  
         a_i ← a_i·exp(‑η·g_i)   (η decays from 2 → 0.001),
   * renormalise onto the simplex.

   The update shrinks entries that contribute most to large convolution
   values, thus flattening the autocorrelation.

3. **Exploration & restarts** – every few hundred successful updates a
   full convolution is recomputed to avoid drift; with a small
   probability a random log‑normal block perturbation is applied; if no
   improvement is seen for 30 s a gentle random jitter is added and the
   process continues.

4. **Budget handling** – the loop runs until the time budget (1000 s
   minus a 5 s safety margin) expires.  The best vector observed is
   returned.

The method is pure NumPy, uses only FFTs (O(n log n) per iteration) and
fits easily into the allowed time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT length ``fft_len``."""
    return _fft_convolve(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initialise a on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            n = 3840
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # No prior candidate – start from a Dirichlet draw
        n = 3840
        a = rng.dirichlet(np.ones(n) * 0.3).astype(np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)            # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)            # for a * w (len 2n‑1)

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)                # current self‑convolution
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters (linear schedules)
    # ------------------------------------------------------------------
    # smooth‑max temperature (α) – from low to high
    alpha_start = 0.5
    alpha_end = 18.0

    # exponentiated‑gradient step size (η) – from aggressive to tiny
    eta_start = 2.0
    eta_end = 0.001

    # simulated‑annealing temperature for occasional Metropolis moves
    temp_start = 0.08
    temp_end = 1e-6

    # counters
    accepted = 0
    RECOMPUTE_EVERY = 400        # full self‑conv to clean drift

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_total = elapsed / total_time

        # --------------------------------------------------------------
        # 5.1  Smooth‑max schedule
        # --------------------------------------------------------------
        alpha = alpha_start + (alpha_end - alpha_start) * frac_total
        eta = eta_start + (eta_end - eta_start) * (1.0 - frac_total)  # decay

        # --------------------------------------------------------------
        # 5.2  Compute current convolution and soft‑max weights
        # --------------------------------------------------------------
        # c = a * a  (already computed from previous iteration, but we need it
        # for the weight; recompute only when we have performed a move that
        # might have corrupted it)
        c = _self_conv(a, fft_len_self)
        c_max = c.max()
        # soft‑max (stable)
        w = np.exp(alpha * (c - c_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 5.3  Gradient of the smooth‑max (g_i = 2 * Σ_k w_k * a_{k-i})
        # --------------------------------------------------------------
        conv_full = _fft_convolve(a, w, fft_len_grad)          # length = 3n‑2
        g = 2.0 * conv_full[n - 1: 2 * n - 1]                  # slice length n

        # --------------------------------------------------------------
        # 5.4  Exponentiated‑gradient step on the simplex
        # --------------------------------------------------------------
        log_factor = -eta * g
        # avoid overflow/underflow
        log_factor = np.clip(log_factor, -700, 700)            # np.exp safe range
        factor = np.exp(log_factor)

        a_new = a * factor
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s

        # --------------------------------------------------------------
        # 5.5  Evaluate new objective (exact self‑convolution)
        # --------------------------------------------------------------
        c_new = _self_conv(a_new, fft_len_self)
        new_score = 2.0 * n * c_new.max()

        # --------------------------------------------------------------
        # 5.6  Metropolis acceptance (helps to escape flat regions)
        # --------------------------------------------------------------
        # linear temperature schedule for simulated‑annealing flavour
        temp_frac = (now - start_time) / total_time
        temperature = temp_start + (temp_end - temp_start) * (1.0 - temp_frac)
        temperature = max(temperature, 1e-12)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                accept = True

        if accept:
            a = a_new
            c = c_new
            cur_score = new_score
            accepted += 1

            # periodic exact recomputation to wipe numerical drift
            if accepted % RECOMPUTE_EVERY == 0:
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # 5.7  Occasional global jitter (≈3 % of iterations)
        # --------------------------------------------------------------
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # 5.8  Keep the best solution so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.9  Soft restart after stagnation (≈30 s without improvement)
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # add a gentle log‑normal perturbation around the current best
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            accepted = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
