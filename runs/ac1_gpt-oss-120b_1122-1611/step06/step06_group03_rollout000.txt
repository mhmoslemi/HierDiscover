def project_simplex(v):
    # returns projection onto simplex {x>=0, sum=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w
```

import numpy as np
import time

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _fft_len_for_n(n):
    size = 2 * n - 1
    return _next_pow2(size)

def _conv_self(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[:2 * a.size - 1]
    return conv

def _gradient_correlation(a, w):
    n = a.size
    L = w.size
    fft_len = _next_pow2(n + L - 1)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(fa_rev * fw, n=fft_len)[:n+L-1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

def _score_from_conv(conv, n):
    return 2.0 * n * float(conv.max())

def _compute_score(a, fft_len, n):
    conv = _conv_self(a, fft_len)
    return _score_from_conv(conv, n)

def _analytic_start(n):
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i+1.0) - np.sqrt(i)
    a /= a.sum()
    return a

def _maybe_use_global():
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None

def _project_onto_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    # see https://doi.org/10.1109/TPAMI.2008.56
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero case
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid continuation gradient descent on a smooth approximation of the
    L∞ objective. Starts from the analytically known √‑difference shape (or a
    stored best construction) and performs Adam‑style updates on the
    log‑sum‑exp of the auto‑convolution, gradually increasing the
    smoothing parameter β. The iterates are projected onto the probability
    simplex after each step. The best vector seen during the run is returned.
    """
    rng = np.random.default_rng(seed)

    n = int(kwargs.get("n", 3809))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # initial vector
    a = _analytic_start(n)
    global height_sequence_1  # maybe defined elsewhere
    cand_global = _maybe_use_global()
    if cand_global is not None:
        # compare internal scores
        cand_fft_len = _fft_len_for_n(cand_global.size)
        if _compute_score(cand_global, cand_fft_len, cand_global.size) < _compute_score(a, fft_len, n):
            a = cand_global
            n = a.size
            fft_len = _fft_len_for_n(n)

    # ensure a is simplex
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    best_a = a.copy()
    best_score = _compute_score(best_a, fft_len, n)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # safety margin

    # Adam variables
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # parameters for continuation
    beta_start = 0.5
    beta_end = 64.0

    # iteration counters
    it = 0

    # main loop
    while time.monotonic() < deadline:
        # compute current beta based on elapsed fraction
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        beta = beta_start + (beta_end - beta_start) * frac

        # convolution
        conv = _conv_self(a, fft_len)
        max_c = conv.max()
        # softmax weights
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            # fall back to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient
        grad = _gradient_correlation(a, w)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = 0.05 / (np.sqrt(v_hat) + eps)  # base learning rate
        a = a - step * m_hat

        # projection onto simplex
        a = _project_onto_simplex(a)

        # occasional evaluation
        if it % 30 == 0:
            cur_score = _compute_score(a, fft_len, n)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] it {it} new best {best_score:.6f}")

        # occasional random perturbation to escape shallow minima
        if it % 200 == 0 and rng.random() < 0.05:
            # random block scaling
            width = rng.integers(1, max(2, n // 25))
            pos = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.25))
            a[pos:pos+width] *= factor
            a = _project_onto_simplex(a)

    # final polishing with larger beta and a few Adam steps (no time check)
    remain = deadline - time.monotonic()
    if remain > 2.0:
        # run a short high‑beta polish
        high_beta = 200.0
        for _ in range(200):
            conv = _conv_self(best_a, fft_len)
            max_c = conv.max()
            w = np.exp(high_beta * (conv - max_c))
            w = w / w.sum()
            grad = _gradient_correlation(best_a, w)

            best_a = _project_onto_simplex(best_a)

    # final clipping for evaluate_sequence
    best_a = np.clip(best_a, 0.0, 1000.0)

    # Return as list of floats
    return [float(v) for v in best_a]

"""
Hybrid continuation gradient descent optimizer for the step‑function problem.

The objective is 2·n·max_k (a∗a)_k with the constraint Σ a_i = 1, a_i ≥ 0.
We minimise a smooth approximation of the L∞‑norm:
    f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

For a given β the gradient is a weighted sum of the auto‑convolution
coefficients.  We use an Adam‑style update and project each iterate onto
the probability simplex.  The smoothing parameter β is increased
continuously (continuation) from a small value (≈0.5) to a large one
(≈64) so the optimiser first sees a smooth landscape and later focuses
on the true peak.

The algorithm starts from the analytically known √‑difference shape
(or a previously stored best construction) and performs many inexpensive
FFT‑based updates within the allotted time budget.  Random jitter and
restart heuristics are added to escape shallow minima.  A short high‑β
polishing phase finishes the run.

The function `propose_candidate` returns the best coefficient vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n(n: int) -> int:
    """Length of the FFT that can hold the linear convolution of two length‑n vectors."""
    size = 2 * n - 1
    return _next_pow2(size)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k.
    Implements g = 2·(a ⋆ w) where ⋆ denotes correlation.
    """
    n = a.size
    L = w.size
    fft_len = _next_pow2(n + L - 1)
    # a reversed for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    fw = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(fa_rev * fw, n=fft_len)[: n + L - 1]
    start = n - 1
    grad = 2.0 * corr_full[start : start + n]
    return grad


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective for a simplex‑normalised vector."""
    return 2.0 * n * float(conv.max())


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast objective evaluator for a normalised vector."""
    conv = _conv_self(a, fft_len)
    return _score_from_conv(conv, n)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _maybe_use_global() -> np.ndarray | None:
    """If a previous construction exists as ``height_sequence_1`` return it normalised."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Projection onto the probability simplex
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x ≥ 0, Σx = 1}.
    Reference: Wang & Carreira‑Perpiñán (2013).
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget elapses.

    The optimiser performs continuation on a smoothed L∞ objective,
    using Adam updates and simplex projection, with occasional random
    perturbations and a final high‑β polishing phase.
    """
    rng = np.random.default_rng(seed)

    # problem size -------------------------------------------------------
    n = int(kwargs.get("n", 3809))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # initial vector ------------------------------------------------------
    a = _analytic_start(n)
    cand_global = _maybe_use_global()
    if cand_global is not None:
        # pick whichever start has the smaller internal score
        score_global = _compute_score(cand_global, _fft_len_for_n(cand_global.size), cand_global.size)
        score_analytic = _compute_score(a, fft_len, n)
        if score_global < score_analytic:
            a = cand_global
            n = a.size
            fft_len = _fft_len_for_n(n)

    a = np.clip(a, 0.0, None)
    a /= a.sum()                     # enforce simplex

    best_a = a.copy()
    best_score = _compute_score(best_a, fft_len, n)

    # bookkeeping ---------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a safety margin

    # Adam state -----------------------------------------------------------
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # continuation parameters ---------------------------------------------
    beta_start = 0.5
    beta_end = 64.0

    it = 0
    last_improve_it = 0

    # ---------------------------------------------------------------------
    # Main optimisation loop
    # ---------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --- continuation schedule for β ---------------------------------
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        beta = beta_start + (beta_end - beta_start) * frac

        # --- convolution and soft‑max weights ----------------------------
        conv = _conv_self(a, fft_len)
        max_c = conv.max()
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient ----------------------------------------------------
        grad = _gradient_correlation(a, w)

        # --- Adam update -------------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # base learning rate (tuned empirically)
        step = 0.05 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # --- projection onto the simplex ---------------------------------
        a = _project_onto_simplex(a)

        # --- occasional evaluation ----------------------------------------
        if it % 200 == 0:
            cur_score = _compute_score(a, fft_len, n)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve_it = it
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] it {it:6d} new best = {best_score:.6f}")

        # --- random jitter to escape plateaus ----------------------------
        if it % 500 == 0 and rng.random() < 0.07:
            width = rng.integers(1, max(2, n // 30))
            pos = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.25))
            a[pos : pos + width] *= factor
            a = _project_onto_simplex(a)

        # --- restart if no improvement for a long while -----------------
        if it - last_improve_it > 3000:
            # re‑initialise around the current best with a stronger jitter
            a = best_a.copy()
            jitter = np.exp(rng.normal(0.0, 0.5, size=n))
            a *= jitter
            a = _project_onto_simplex(a)
            last_improve_it = it
            # reset Adam statistics to avoid stale momentum
            m.fill(0.0)
            v.fill(0.0)

    # ---------------------------------------------------------------------
    # Final polishing with a very large β (focus on the true max)
    # ---------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 3.0:
        high_beta = 200.0
        # a few deterministic gradient steps without Adam (simple step size)
        for _ in range(250):
            conv = _conv_self(best_a, fft_len)
            max_c = conv.max()
            w = np.exp(high_beta * (conv - max_c))
            w /= w.sum()
            grad = _gradient_correlation(best_a, w)
            # modest step
            best_a = best_a - 0.01 * grad
            best_a = _project_onto_simplex(best_a)

    # ---------------------------------------------------------------------
    # Return the final candidate (clipped as required by evaluate_sequence)
    # ---------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
