"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Algorithm
---------
*   Work in the probability simplex (sum = 1).  The objective
    `E(a)=2·n·max_k (a*a)_k` is invariant under scaling, so the sum
    constraint does not affect the optimum.
*   The objective is the ℓ∞‑norm of the discrete self‑convolution.
    Its sub‑gradient at a point `a` is obtained from the (possibly
    several) indices where the convolution attains its maximum:
        ∂E/∂a = 2·∑_{k∈A} a shifted by k,
    where `A` is the active set (indices with value within a tiny
    tolerance of the maximum).  The shift is the correlation
    `a ⋆ e_k`.  This is computed efficiently with a single FFT.
*   We keep an explicit log‑parameter `x` (soft‑max logits) and apply
    Adam updates to the sub‑gradient expressed in the log‑space:
        grad_x = a * (grad_a - ⟨grad_a,a⟩) .
*   A simple schedule linearly ramps the learning‑rate from `lr_start`
    to `lr_end` during the allotted time.  Small random jitter on the
    logits and occasional block‑wise shifts are injected to escape
    flat plateaux.
*   Whenever a better objective value is found it is stored; if no
    improvement occurs for `restart_gap` seconds the optimisation is
    warm‑restarted from the current best solution (with a fresh jitter).
*   The routine respects the 1000 s budget, prints progress stamps and
    finally returns the best sequence (clipped to the limits enforced
    by `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Correlation of a with a weight vector w (used for the sub‑gradient).
# Returns a vector of length len(a); the correlation is computed via FFT.
# ----------------------------------------------------------------------
def _grad_corr(a: np.ndarray, w: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """
    Compute 2·(a ⋆ w) where w has length 2·len(a)‑1.
    The result has length len(a).
    """
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len_corr)[: 3 * a.size - 2]
    start = a.size - 1
    return 2.0 * corr_full[start:start + a.size]

            m_hat = m / bias_correction1
            v_hat = v / bias_correction2
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            x -= step

            # ----- occasional stochastic perturbations --------------
            if rng.random() < jitter_prob:
                # small Gaussian jitter on all logits
                x += rng.normal(0.0, 0.05, size=n)

            if rng.random() < block_prob:
                # pick a contiguous block and shift its logits together
                width = rng.integers(1, max(2, n // 30))
                start = rng.integers(0, n - width + 1)
                delta = rng.normal(0.0, 0.20)
                x[start:start + width] += delta

            # ----- restart if we are stuck for a while -------------
            if now - inner_last_improve > restart_gap:
                # Return to the best known solution (with a little jitter)
                x = np.log(best_a + 1e-30)
                x += rng.normal(0.0, 0.12, size=n)
                m.fill(0.0)
                v.fill(0.0)
                inner_last_improve = now
                print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")
                break   # break inner loop and start a fresh run

    # ------------------------------------------------------------------
    # Return the best sequence found – clip to limits required by
    # evaluate_sequence (negative → 0, >1000 → 1000)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
