"""
Propose a non‑negative step function that minimizes

    2·n·max(a∗a) / (sum a)²

using a smooth‑max gradient descent with an exponential‑gradient update.
The algorithm works on a moderately large vector (default ≈ 8000 entries),
starts from a portfolio of hand‑crafted shapes, and then repeatedly:

1. Computes the self‑convolution a∗a (FFT, O(n log n)).
2. Approximates the peak with a soft‑max (log‑sum‑exp) whose temperature λ
   is increased from 5 to 200 during the run, making the surrogate ever
   closer to the true maximum.
3. Evaluates the gradient of the surrogate objective and applies an
   exponentiated‑gradient (multiplicative) update aᵢ ← aᵢ·exp(‑η·∂obj/∂aᵢ),
   with a step‑size η that decays linearly.
4. Projects back onto the probability simplex (clip negatives, renormalise).
5. Keeps the best sequence seen (exact objective computed via evaluate_sequence)
   and injects occasional log‑normal jitter or restarts after stagnation.

The whole routine respects a 1000 s budget (with a 5 s safety margin) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a∗a using an FFT of length ``fft_len``.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution x∗y using an FFT of length ``fft_len``.
    Returns the full (len(x)+len(y)‑1) result.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: x.size + y.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator,
                         max_width_frac: float = 0.05,
                         sigma: float = 0.12) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    Afterwards the vector is projected back onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, int(max_width_frac * n)))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a small diverse portfolio of hand‑crafted probability vectors.
    All candidates are already normalised to sum = 1.
    """
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Exponential decays
    for base in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = base ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # Linear decreasing
    v = (np.arange(n, 0, -1, dtype=np.float64))
    v /= v.sum()
    cand.append(v)

    # Two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # Piecewise constant (heavy first half)
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    v /= v.sum()
    cand.append(v)

    # Arcsine‑like shape (heavy tails)
    idx = np.arange(n, dtype=np.float64) + 0.5
    v = 1.0 / np.sqrt(idx * (n + 0.5 - idx))
    v /= v.sum()
    cand.append(v)

    # Dirichlet draws (moderately concentrated)
    for _ in range(6):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose problem size and initialise a from a small portfolio
    # ------------------------------------------------------------------
    # Desired length – a value that gives enough degrees of freedom
    # yet keeps each iteration cheap. 8000 works well for the time budget.
    target_n = 8000

    # If a previous solution exists, up‑sample it to the target length
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if old.ndim == 1 and old.size > 0 and np.isfinite(old).all():
            old_n = old.size
            # Linear interpolation onto the new grid
            x_old = np.linspace(0.0, 1.0, old_n)
            x_new = np.linspace(0.0, 1.0, target_n)
            a0 = np.interp(x_new, x_old, old)
        else:
            a0 = None
    else:
        a0 = None

    # Build the candidate pool (including the up‑sampled old solution if any)
    candidates = _initial_candidates(target_n, rng)
    if a0 is not None:
        # Normalise the interpolated vector
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 == 0.0:
            a0 = np.full(target_n, 1.0 / target_n, dtype=np.float64)
        else:
            a0 /= s0
        candidates.append(a0)

    # Quick cheap evaluation (exact convolution) to pick the best starter
    fft_len_self = _next_pow2(2 * target_n - 1)
    best_start_score = np.inf
    a = None
    for v in candidates:
        c = _self_conv_fft(v, fft_len_self)
        score = 2.0 * target_n * c.max()               # sum(v)==1 ⇒ denominator =1
        if score < best_start_score:
            best_start_score = score
            a = v.copy()

    # Guarantee we have a valid starting vector
    if a is None:
        a = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    n = a.size
    # ------------------------------------------------------------------
    # 2) Book‑keeping for the current state
    # ------------------------------------------------------------------
    c = _self_conv_fft(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (all schedules are linear in elapsed time)
    # ------------------------------------------------------------------
    # λ – temperature for the smooth‑max (log‑sum‑exp)
    lam_start = 5.0
    lam_end = 200.0

    # η – step‑size for the exponentiated‑gradient update
    eta_start = 0.5
    eta_end = 0.001

    # Stagnation handling
    stagnation_threshold = 30.0          # seconds without improvement → restart

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = min(1.0, elapsed / total_time)

        # --------------------------------------------------------------
        # 4.1  Update scheduled hyper‑parameters
        # --------------------------------------------------------------
        lam = lam_start + (lam_end - lam_start) * frac
        eta = eta_start + (eta_end - eta_start) * (1.0 - frac)   # shrink η

        # --------------------------------------------------------------
        # 4.2  Compute the current self‑convolution
        # --------------------------------------------------------------
        c = _self_conv_fft(a, fft_len_self)   # length 2n‑1
        max_c = c.max()

        # --------------------------------------------------------------
        # 4.3  Smooth‑max (log‑sum‑exp) and its softmax weights
        # --------------------------------------------------------------
        # stable log‑sum‑exp
        shifted = lam * (c - max_c)          # all ≤ 0
        w_raw = np.exp(shifted)
        w_sum = w_raw.sum()
        if w_sum == 0.0:                     # numerical safety
            w = np.full_like(c, 1.0 / c.size)
            m = max_c
        else:
            w = w_raw / w_sum                # softmax probabilities
            # smooth max = max_c + log(w_sum)/lam
            m = max_c + np.log(w_sum) / lam

        # --------------------------------------------------------------
        # 4.4  Gradient of the surrogate objective
        # --------------------------------------------------------------
        # g_i = 2 * Σ_j a_j w_{i+j}  (i = 0..n‑1)
        a_rev = a[::-1]
        conv_len = _next_pow2(a_rev.size + w.size - 1)
        conv_full = _conv_fft(a_rev, w, conv_len)
        # extract the slice that corresponds to i = 0..n‑1
        start_idx = n - 1
        g_vec = 2.0 * conv_full[start_idx:start_idx + n]   # shape (n,)

        # Since we keep a normalised (sum ≈ 1), denominator = 1 ².
        # Full gradient of the smooth objective:
        #   ∂obj = (2 n)·(g_i – 2·m)      (because S≈1)
        grad = (2.0 * n) * (g_vec - 2.0 * m)

        # --------------------------------------------------------------
        # 4.5  Exponentiated‑gradient update (multiplicative)
        # --------------------------------------------------------------
        delta = -eta * grad
        # Clamp the exponent to avoid overflow/underflow
        np.clip(delta, -10.0, 10.0, out=delta)
        a = a * np.exp(delta)

        # Project back onto the simplex (clip negatives, renormalise)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # --------------------------------------------------------------
        # 4.6  Evaluate (fast) current score and keep the best
        # --------------------------------------------------------------
        c = _self_conv_fft(a, fft_len_self)
        cur_score = 2.0 * n * c.max()          # sum(a)==1 ⇒ denominator =1
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best ≈ {best_score:.6f}")

        # --------------------------------------------------------------
        # 4.7  Occasional global jitter (helps escaping shallow basins)
        # --------------------------------------------------------------
        if rng.random() < 0.02:                # ~2 % of iterations
            a = _random_block_perturb(a, rng,
                                      max_width_frac=0.03,
                                      sigma=0.15)
            c = _self_conv_fft(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # 4.8  Restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > stagnation_threshold:
            # Re‑initialise from the best seen so far with moderate noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv_fft(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – cur_score {cur_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
