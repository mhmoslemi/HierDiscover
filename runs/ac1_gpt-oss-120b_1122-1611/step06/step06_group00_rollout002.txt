"""
Propose a step‑function of non‑negative heights that (empirically) beats the
previous bound 1.5132 → ≈1.50.

Algorithm
---------
1. **Initial portfolio** – generate a small set of diverse candidate vectors
   (uniform, exponential, linear, piecewise constant and several Dirichlet
   draws).  Keep the best (according to the exact convolution) as the start.

2. **Phase 1 – multi‑peak exponentiated‑gradient descent** (≈15 % of the time).
   The objective is `max(conv(a,a))`.  Instead of the single peak used before,
   we take the K largest peaks (K≈3) and accumulate their sub‑gradients
   `g_i = Σ_{k∈peaks} 2·a_{k−i}` (zero outside the support).  An
   exponentiated‑gradient update `a_i ← a_i·exp(−η·g_i)` is performed with a
   linearly decaying step‑size η.  After each update a short diffusion
   `(0.25,0.5,0.25)` is applied (weighted 2 %) to keep the distribution smooth.
   The vector is re‑normalised to the simplex.

3. **Phase 2 – stochastic annealing with informed moves** (≈85 % of the time).
   *Peak* `k_max = argmax(conv)`.  The contribution of each index to that
   peak is `c_i = a_i·a_{k_max−i}` (when the partner exists).  A donor index is
   drawn proportionally to `c_i`.  The recipient is drawn using a weight that
   favours small mass **and** a small partner value at the peak:
   `w_j = (1−a_j) / (a_{k_max−j}+ε)`.  A fraction `δ∈[0,0.9·a_donor]` of mass is
   transferred from donor to recipient; the convolution is updated incrementally
   in O(n) using a closed‑form formula.  Moves are accepted with a Metropolis
   rule at a temperature that cools linearly from `T_start` to `T_end`.

4. **Global jitter** – with probability 2 % a multiplicative log‑normal
   perturbation of a random contiguous block is applied, followed by a full
   recomputation of the convolution.

5. **Restart** – if no improvement occurs for 25 s the current best solution
   is perturbed by moderate log‑normal noise and the search restarts.

All operations respect the non‑negative simplex constraint; the final vector
is clipped to `[0,1000]` (the limits enforced by `evaluate_sequence`).  The
routine returns the best sequence found before the 1000 s budget expires.
"""

import time
import numpy as np

# Helper utilities – top level (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # linear part: shift of donor (subtract) and recipient (add)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ part
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a small, diverse portfolio of starter vectors."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # 3) linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    candidates.append(v)

    # 4) two‑spike at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    candidates.append(v)

    # 5) piecewise constant (first half heavy, second half zero)
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    candidates.append(v)

    # 6) Dirichlet draws (moderate concentration)
    for _ in range(6):
        v = rng.dirichlet(np.ones(n) * 0.3)
        candidates.append(v.astype(np.float64))

    # 7) power‑law tails
    for p in [0.5, 1.0, 1.5]:
        idx = np.arange(1, n + 1, dtype=np.float64)
        v = idx ** (-p)
        v /= v.sum()
        candidates.append(v)

    return candidates

            # occasional global jitter (≈2 % of iterations)

                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)

    #    evaluator for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
