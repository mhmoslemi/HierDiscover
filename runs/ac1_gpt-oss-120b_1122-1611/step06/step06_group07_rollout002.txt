def propose_candidate(seed=42, budget_s=1000.0, **kwargs):

    rng = np.random.default_rng(seed)

    n = int(kwargs.get("n", 3800))
    if n < 2: n = 2

    # Precompute FFT length L
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len - 1).bit_length())   (already in _fft_len_for_n)

    # Define starting set of initial distributions: r values, p values, Dirichlet, maybe current height_sequence_1.

    init_candidates = []

    # Add height_sequence_1 normalized
    if 'height_sequence_1' in globals():
        cand = np.asarray(height_sequence_1, dtype=np.float64)
        # Clip length to n (or if length mismatches, we can crop or pad)
        if cand.size == n:
            a_norm = cand.clip(0.0, None)
            if a_norm.sum() > 0:
                a_norm = a_norm / a_norm.sum()
                init_candidates.append(a_norm)

    # r family
    for r in np.linspace(0.5, 0.97, 10):
        a = _gen_shape_r(r, n)
        if a is not None:
            init_candidates.append(a)

    # p family
    for p in np.linspace(0.2, 2.0, 10):
        a = _gen_shape_p(p, n)
        if a is not None:
            init_candidates.append(a)

    # Dirichlet random seeds
    for _ in range(5):
        a = rng.dirichlet(np.ones(n))
        init_candidates.append(a)

    # Optionally add a small uniform cap: e.g., combination of uniform and previous best...
    a_uniform = np.full(n, 1.0/n)
    init_candidates.append(a_uniform)

    best_score = np.inf
    best_a = None

    start_time = time.monotonic()
    deadline = start_time + budget_s - 0.5  # margin

    # each optimization run limited by remaining time and by iteration count.
    for idx, a0 in enumerate(init_candidates):
        # compute per-run time limit: allocate proportionally: e.g., remaining_time / (remaining_cands)
        remaining_cands = len(init_candidates) - idx
        now = time.monotonic()
        remaining_time = deadline - now
        if remaining_time <= 0:
            break
        per_run_deadline = now + remaining_time / remaining_cands * 0.9  # use 90% to leave some for later.

        a_opt, sc = _optim_adam(a0, fft_len, per_run_deadline, rng)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"[{now - start_time:.2f}s] candidate {idx} improved to {best_score:.6f}")

    # final output
    if best_a is None:
        # fallback: uniform
        best_a = np.full(n, 1.0/n)

    # clip and ensure sum>0.01
    best_a = np.clip(best_a, 0.0, 1000.0)
    # Normalize just in case
    s = best_a.sum()
    if s == 0:

        best_a = best_a / s

    result = [float(v) for v in best_a.tolist()]
    print(f"Best score: {best_score:.6f} for n={n}")

def _optim_adam(a0, fft_len, deadline, rng):
    """Optimize using Adam on softmax surrogate.
    Returns best vector (normalized) and best score observed.
    """
    n = a0.size
    # Initialize log-parameters u such that softmax(u) = a0 (or use log+epsilon)
    eps = 1e-12
    u = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    t = 0

    # track best
    best_a = np.exp(u)
    best_a /= best_a.sum()
    conv = _conv_self(best_a, fft_len)
    best_score = float(2.0 * n * conv.max())
    # schedule parameters
    # We'll linearly increase beta and decrease lr.
    start_time = time.monotonic()
    total_time = deadline - start_time

    # We'll keep a heartbeat: after each iteration check time.
    # Use max iteration caps.
    while True:
        now = time.monotonic()

"""
Propose a candidate step‑function (probability vector) that minimises
evaluate_sequence.  The algorithm proceeds as follows:

1.  Generate a small portfolio of initial vectors:
    * Analytic families a_i = (i+1)^r – i^r   (r∈[0.5,0.95])
    * Analytic families a_i = (i+1)^(-p) – (i+2)^(-p)   (p∈[0.1,2])
    * Dirichlet random draws (uniform sum = 1)
    * If a previous best (`height_sequence_1`) is available, use it.
2.  For each start run an Adam‑accelerated mirror‑descent on the
    simplex.  The objective is replaced by a smooth surrogate:
        L_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)
    whose gradient is 2·a⋆w with w being a soft‑max over the
    convolution.  β is increased linearly from 0.5 to 5000,
    while the Adam learning rate decays linearly.
3.  The parameters are stored as logits u (so a = softmax(u));
    this automatically enforces non‑negativity and Σa = 1.
    Occasionally a tiny random jitter is added to escape plateaus.
4.  The best vector seen over all runs (according to the exact
    objective 2·n·max(a∗a)) is returned, clipped to the limits imposed
    by `evaluate_sequence`.

The whole search respects the supplied time budget (≈1000 s) and
prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of
    two length‑n vectors (full length = 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max with temperature β (numerically stable)."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the gradient ∇_a Σ_k w_k (a∗a)_k  = 2·a ⋆ w  (correlation).
    `w` must have length 2·len(a)‑1.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _gen_shape_r(r: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^r – i^r  (decreasing for 0<r<1).  Returns a normalised
    probability vector or None if something goes wrong."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_shape_p(p: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^(-p) – (i+2)^(-p)  (p>0)."""
    k = np.arange(1, n + 2, dtype=np.float64)              # 1 … n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]                                      # length n
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Optimisation core (Adam on the simplex via logits)
# ----------------------------------------------------------------------
def _adam_optimize(
    a_start: np.ndarray,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Run Adam‑mirrored descent on the soft‑max surrogate.
    Returns the best vector (normalised) and its exact score.
    """
    n = a_start.size
    eps = 1e-12
    # initialise logits u so that softmax(u) = a_start
    u = np.log(a_start + eps)

    # Adam state
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    beta1, beta2 = 0.9, 0.999
    t = 0

    # bookkeeping
    best_a = np.exp(u - np.max(u))
    best_a /= best_a.sum()
    conv = _conv_self(best_a, fft_len)
    best_score = float(2.0 * n * conv.max())

    start_time = time.monotonic()
    total_time = deadline - start_time
    # main loop
    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # ----- current probability vector (softmax of logits) -----
        shift = np.max(u)
        a = np.exp(u - shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----- exact score (for bookkeeping) -----
        conv = _conv_self(a, fft_len)
        cur_score = float(2.0 * n * conv.max())
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- surrogate gradient (soft‑max over convolution) -----
        elapsed = now - start_time
        # temperature schedule (slowly increase)
        beta = 0.5 + (5000.0 - 0.5) * (elapsed / total_time)
        w = _softmax(conv, beta)                   # length 2·n‑1, sums to 1

        # gradient w.r.t. a of the surrogate objective
        grad_a = _gradient_via_correlation(a, w)    # shape (n,)

        # gradient w.r.t. logits (mirror‑descent on simplex)
        avg = np.dot(grad_a, a)                    # ⟨grad_a , a⟩
        grad_u = grad_a - avg * a                  # ∇_u L = (I – a·1ᵀ) grad_a

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (linearly decaying)
        lr = 0.08 * (1.0 - elapsed / total_time) + 0.003 * (elapsed / total_time)

        u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # keep logits in a reasonable numeric range
        u = np.clip(u, -50.0, 50.0)

        # occasional tiny jitter to escape shallow plateaus
        if rng.random() < 0.001:
            jitter = rng.normal(0.0, 0.3, size=n)
            u += jitter

    return best_a, best_score

            best_score = sc
            best_a = a_opt.copy()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] candidate {idx} improved global best to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but be safe
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = float(2.0 * n * _conv_self(best_a, fft_len).max())

    # ------------------------------------------------------------------
    # Final sanitisation (exactly what evaluate_sequence does)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / total

    print(f"Final best score (approximate) = {best_score:.6f} for n={n}")

    # Convert to plain Python floats for the harness
    return [float(v) for v in best_a.tolist()]
