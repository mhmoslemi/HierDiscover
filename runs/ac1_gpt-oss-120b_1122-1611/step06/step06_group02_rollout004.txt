"""
Search for a non‑negative step function minimizing the max self‑convolution.

Algorithm
---------
* Multiple stochastic restarts on the simplex (∑a = 1, a_i ≥ 0).
* Each restart is refined by an Adam‑style exponentiated‑gradient descent.
  – The gradient is chosen either as the exact sub‑gradient of the worst
    convolution entry or as the gradient of a soft‑max (log‑sum‑exp) smooth
    approximation.  The soft‑max temperature β grows exponentially.
  – The update is multiplicative (mirrored‐descent on the simplex) with
    Adam momentum and a 1/√t learning‑rate schedule, followed by renormalisation.
* Four families of initial vectors are used:
    – Random Dirichlet (exponential draws).
    – Sine‑power shape (≈ “mountain” curves).
    – Inverse‑distance‑to‑edge shape (U‑shaped, mass near the ends).
    – Linear interpolation from the current global best.
* Light random moves (contiguous block perturbation, pairwise mass transfer)
  are applied occasionally to jump out of plateaus.
* The best candidate seen so far is repeatedly polished with a longer
  optimisation run.
* The whole search respects the supplied time budget (default 1000 s) and
  returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. ``a``."""
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    # shift for numerical stability
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)
    start = a.size - 1
    grad_part = conv_w_rev[start:start + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    # width ~ 1…n/30 (at least one)
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out = np.full_like(out, 1.0 / n)
    else:
        out /= total
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a random fraction of mass from i to j, accept only if improving."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    # cheap check
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a


# ----------------------------------------------------------------------
# Score helper (fast, avoids evaluate_sequence overhead)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponential variables."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """a_i ∝ sin^p(π·(i+0.5)/n)  with p∈[0.5,4]."""
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_inverse_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """U‑shaped: a_i ∝ (dist_to_nearest_edge)^{-p}  with p∈[0.5,5]."""
    p = rng.uniform(0.5, 5.0)
    idx = np.arange(n)
    # distance to the closest endpoint (1‑based to avoid zero)
    dist = np.minimum(idx + 1, n - idx)
    a = dist.astype(np.float64) ** (-p)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linearly interpolate an existing distribution to a new length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Local optimiser (Adam‑style exponentiated gradient)
# ----------------------------------------------------------------------


def _local_optim_adam(initial: np.ndarray,
                      rng: np.random.Generator,
                      deadline: float,
                      max_iters: int = 1500) -> tuple[np.ndarray, float]:
    """
    Refine ``initial`` (already on the simplex) using an Adam‑style
    exponentiated‑gradient scheme.  Returns the best vector seen and its
    (fast) score.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # soft‑max temperature schedule
    beta = 0.5
    beta_mul = 1.08

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # Choose a gradient (sub‑gradient vs. soft‑max)
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = min(1e5, beta * beta_mul)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # Adam‑style multiplicative update
        # --------------------------------------------------------------
        lr = 0.7 / np.sqrt(t)          # decreasing learning‑rate
        m = beta1 * m + (1.0 - beta1) * raw_grad
        v = beta2 * v + (1.0 - beta2) * (raw_grad * raw_grad)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # exponentiated‑gradient step
        a = a * np.exp(-lr * m_hat / (np.sqrt(v_hat) + eps))

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0 or np.isnan(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --------------------------------------------------------------
        # Random exploratory moves (occasionally)
        # --------------------------------------------------------------
        if rng.random() < 0.025:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.025:
            a = _pairwise_move(a, rng, fft_len_conv)

    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    # ----- Global best container -----
    best_vec = None
    best_score = float("inf")

    # ----- Load a previously known construction if present -----
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                total = cand.sum()
                if total == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand /= total
                cand_score = _score(cand, _fft_len_for_len(2 * cand.size - 1))
                best_vec = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start:6.2f}s] Loaded previous best: {best_score:.6f}")
        except Exception:
            pass

    # ----- Lengths and initialisation families -----
    lengths = [1500, 2000, 2500, 3000, 3500, 3809,
               4000, 4500, 5000, 6000, 8000, 10000]

    # order: Dirichlet, sine‑power, inverse‑power, interpolate
    init_weights = np.array([0.40, 0.20, 0.30, 0.10])

    # ----- Main stochastic search loop -----
    while time.time() < deadline:
        # Choose a length (uniformly for simplicity)
        n = int(rng.choice(lengths))

        # Choose an initialisation method
        which = rng.choice(len(init_weights), p=init_weights)

        if which == 0:          # random Dirichlet
            a0 = _init_random_dirichlet(rng, n)
        elif which == 1:        # sine‑power
            a0 = _init_sine_power(rng, n)
        elif which == 2:        # inverse‑distance‑to‑edge
            a0 = _init_inverse_power(rng, n)
        else:                   # interpolate from current best (if any)
            if best_vec is not None:
                a0 = _init_interpolate(rng, best_vec, n)
            else:
                a0 = _init_random_dirichlet(rng, n)

        # Number of iterations – more for smaller n, fewer for very large n
        iters = int(2000 * (5000.0 / max(n, 500)))   # ≈2000 when n≈5000
        iters = max(300, iters)                     # safety lower bound

        a_opt, score_opt = _local_optim_adam(a0, rng, deadline, max_iters=iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:6.2f}s] New global best: {best_score:.6f} (n={n})")

        # --------------------------------------------------------------
        # Polishing phase (if time permits)
        # --------------------------------------------------------------
        if best_vec is not None and time.time() < deadline - 1.0:
            a_pol, score_pol = _local_optim_adam(
                best_vec.copy(), rng, deadline, max_iters=3000)
            if score_pol < best_score:
                best_score = score_pol
                best_vec = a_pol.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:6.2f}s] Polished best: {best_score:.6f}")

    # ----- Fallback (should never happen) -----
    if best_vec is None:
        n = 3000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ----- Final clipping (evaluate_sequence does the same) -----
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Return plain Python floats for the harness
    return [float(x) for x in best_vec]
