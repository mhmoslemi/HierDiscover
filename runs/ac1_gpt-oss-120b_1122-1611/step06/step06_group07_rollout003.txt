import time
import numpy as np

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution
    of two vectors of length `n` (full length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    prod = fa * fa
    conv = np.fft.irfft(prod, n=fft_len)[: 2 * a.size - 1]
    return conv

def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max with temperature `beta`."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s

def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute gradient of the surrogate `w·(a*a)` w.r.t. `a`,
    i.e. 2·a ⋆ w (correlation).  `w` must have length 2·n‑1.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    return 2.0 * corr_full[n - 1 : n - 1 + n]

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum x = 1}."""
    # Implementation of the algorithm from Duchi et al. (2008)
    n = v.shape[0]
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find rho: last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # ensure sum is exactly 1 (numerical)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum

def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value for a probability vector `a`."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())

def _gen_shape_r(r: float, n: int) -> np.ndarray:
    """Shape a_i = (i+1)^r - i^r  (decreasing for 0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

def _gen_shape_p(p: float, n: int) -> np.ndarray:
    """Shape a_i = (i+1)^(-p) - (i+2)^(-p) (p>0)."""
    k = np.arange(1, n + 2, dtype=np.float64)   # 1..n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

                if s > 0.0:

        except Exception:
            pass

        a = _gen_shape_r(r, n0)
        if a is not None:
            init_candidates.append(a)
    for p in np.linspace(0.1, 2.0, 21):
        a = _gen_shape_p(p, n0)
        if a is not None:
            init_candidates.append(a)

    # c) random Dirichlet samples (moderately many)
    for _ in range(12):
        a = rng.dirichlet(alpha=np.ones(n0))
        init_candidates.append(a)

    # ------------------------------------------------------------------
    # 2) Pick the best of the initial pool.
    # ------------------------------------------------------------------
    best_a = None
    best_score = np.inf
    fft_len = _fft_len_for_n(n0)

    for cand in init_candidates:
        sc = _score(cand, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    n = best_a.size
    fft_len = _fft_len_for_n(n)
    print(f"[init] best initial score = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.1)  # safety margin

    cur_score = best_score
    last_improve = start_time

                    cur_a[i] -= delta
                    cur_a[j] += delta
            elif move == 2:

                jitter = np.exp(rng.normal(0.0, 0.12, size=n))
                cur_a = cur_a * jitter

                i, j = rng.integers(0, n, size=2)
                if i != j:
                    amount = rng.random() * min(cur_a[i], cur_a[j]) * 0.3
                    cur_a[i] += amount
                    cur_a[j] -= amount

            cur_a = cur_a / cur_a.sum()

        # Periodic evaluation of the exact objective.
        # ------------------------------------------------------------------

        # when a noticeable amount of time has passed since the last
        # check.

                best_a = cur_a.copy()
                last_improve = now

    # evaluate_sequence).
    # ------------------------------------------------------------------

    return [float(v) for v in best]

        elif progress < 0.90:
            # ---------- Phase B : projected sub‑gradient on true max ----------
            conv = _conv_self(cur_a, fft_len)
            max_val = conv.max()
            eps = max_val * 1e-12 + 1e-12
            mask = (conv >= max_val - eps).astype(np.float64)
            if mask.sum() == 0:
                w = np.zeros_like(conv)
            else:
                w = mask / mask.sum()
            grad = _gradient_via_correlation(cur_a, w)

            # Step size decays from ~0.2 to ~0.001 over this phase.
            eta = 0.2 * (1.0 - (progress - 0.60) / 0.30) + 0.001
            cur_a = cur_a - eta * grad
            cur_a = _proj_simplex(cur_a)

        else:
            # ---------- Phase C : stochastic local exploration ----------
            move = rng.integers(0, 4)
            if move == 0:
                # Block scaling
                width = rng.integers(1, max(2, n // 30))
                start_idx = rng.integers(0, n - width + 1)
                mult = np.exp(rng.normal(0.0, 0.25))
                cur_a[start_idx:start_idx + width] *= mult
            elif move == 1:
                # Pairwise mass transfer
                i, j = rng.integers(0, n, size=2)
                if i != j and cur_a[i] > 1e-14:
                    delta = rng.random() * cur_a[i] * 0.5
                    cur_a[i] -= delta
                    cur_a[j] += delta
            elif move == 2:
                # Global jitter (multiplicative noise)
                jitter = np.exp(rng.normal(0.0, 0.12, size=n))
                cur_a = cur_a * jitter
            else:
                # Small redistribution between two coordinates
                i, j = rng.integers(0, n, size=2)
                if i != j:
                    amount = rng.random() * min(cur_a[i], cur_a[j]) * 0.3
                    cur_a[i] += amount
                    cur_a[j] -= amount
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = cur_a / cur_a.sum()

        # --------------------------------------------------------------
        # Periodic exact evaluation (every ~0.5 s).
        # --------------------------------------------------------------
        if now - last_improve > 0.5:
            conv = _conv_self(cur_a, fft_len)
            cur_score = float(2.0 * n * conv.max())
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = cur_a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

            # Restart if stagnation persists.
            if now - last_improve > RESTART_TIMEOUT:
                jitter = np.exp(rng.normal(0.0, 0.18, size=n))
                cur_a = best_a * jitter
                cur_a = _proj_simplex(cur_a)
                last_improve = now
                print(f"[{elapsed:6.2f}s] restart – jitter around best")

    # ------------------------------------------------------------------
    # Return the best vector found (already normalised, safe for
    # evaluate_sequence).  Clip to the hard limits imposed by the evaluator.
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
