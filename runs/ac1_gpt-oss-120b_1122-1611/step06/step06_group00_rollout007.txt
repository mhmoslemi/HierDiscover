def _gradient_step(a, alpha, lr):
    # compute self convolution c
    conv_len = _next_pow2(2*a.size - 1)
    FA = np.fft.rfft(a, n=conv_len)
    c = np.fft.irfft(FA*FA, n=conv_len)[:2*a.size - 1]

    # compute softmax weights
    scaled = alpha * (c - np.max(c))
    w = np.exp(scaled)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # compute gradient via convolution a with reversed w
    w_rev = w[::-1]
    grad_len = _next_pow2(a.size + w_rev.size - 1)
    FA = np.fft.rfft(a, n=grad_len)
    FW = np.fft.rfft(w_rev, n=grad_len)
    grad_full = np.fft.irfft(FA * FW, n=grad_len)
    offset = w_rev.size - 1
    grad = 2.0 * grad_full[offset: offset + a.size]

    # update
    a_new = a - lr * grad
    # project onto simplex
    a_new = proj_simplex(a_new)
    return a_new, np.max(c)

def propose_candidate(seed=12345, budget_s=1000.0):

    deadline = time.monotonic() + max(0.0, budget_s-5)

    if "height_sequence_1" in globals():

        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            n = cand.size
            a = np.clip(cand,0, None)
            a = proj_simplex(a)
        else:
            n = 4000

    else:

        candidates = _initial_candidates(n, rng)  # using function defined earlier
        best_score = np.inf
        best = None
        for v in candidates:

            score = 2.0 * n * c.max()

"""
softmax‑gradient descent with simplex projection

We minimise the objective  

    J(a) = 2·n·max(c) / (sum(a)^2) ,  c = a * a  (linear self‑convolution)

The scaling of a never changes J, therefore we keep the L1‑norm equal to 1.
The non‑smooth max is approximated by a soft‑max

    w_k ∝ exp(α·c_k) ,   Σ w_k = 1

and we minimise the smooth surrogate  

    ϕ(a) = Σ_k w_k·c_k  = wᵀc

Its gradient is
    ∇ϕ(a)_j = 2· Σ_i a_i·w_{i+j}
which is a convolution of a with the reversed weight vector w.
Each iteration performs:

* FFT‑based self‑convolution to obtain c.
* Soft‑max with a linearly increasing temperature α (5 → 200).
* Gradient via a second FFT (a * w_rev).
* Multiplicative (exponentiated) update a ← a·exp(−η·∇ϕ) and projection onto the
  probability simplex.
* Occasionally we apply a random block perturbation and restart from the current
  best after long stagnation.

All steps are cheap (O(n log n)) and fit well into the 1000 s budget.
The best sequence seen so far is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit L1‑norm."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]  # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Diversified portfolio of starting vectors (all normalised to sum = 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Uses a soft‑max gradient descent on the
    self‑convolution with projection onto the simplex, plus occasional
    random perturbations and restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initialise a feasible a
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a.ndim == 1 and a.size > 0 and np.isfinite(a).all():
            a = np.clip(a, 0.0, None)
            a = proj_simplex(a)
        else:
            a = None
    else:
        a = None

    if a is None:
        n = 3840                      # default length (same order as previous runs)
        # try a few hand‑crafted starts and keep the best one (exact score)
        candidates = _initial_candidates(n, rng)
        best_score = np.inf
        best_vec = None
        fft_len = _next_pow2(2 * n - 1)
        for v in candidates:
            c = _self_conv(v, fft_len)
            score = 2.0 * n * c.max()
            if score < best_score:
                best_score = score
                best_vec = v.copy()
        a = best_vec

    n = a.size
    # pre‑computed FFT length for the self‑convolution (used many times)
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Book‑keeping of the current best solution
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - start_time) / total_time          # progress ∈[0,1]

        # ----- schedules -------------------------------------------------
        # learning rate (multiplicative / exponentiated‑gradient)
        eta_start, eta_end = 0.5, 0.001
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # soft‑max temperature (α): start smooth, finish sharp
        alpha_start, alpha_end = 5.0, 200.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ----- compute self‑convolution ---------------------------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]

        # ----- soft‑max weights -----------------------------------------
        c_max = conv.max()
        # shift to avoid overflow
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:                 # numerical safety – fallback to uniform
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient via convolution (a * w_rev) --------------------
        w_rev = w[::-1]                     # reverse for the cross‑correlation
        grad_len = _next_pow2(n + w_rev.size - 1)   # = next_pow2(3n‑2)
        fa = np.fft.rfft(a, n=grad_len)
        fw = np.fft.rfft(w_rev, n=grad_len)
        grad_full = np.fft.irfft(fa * fw, n=grad_len)
        offset = w_rev.size - 1               # = 2n‑2
        grad = 2.0 * grad_full[offset: offset + n]

        # ----- exponentiated‑gradient update & projection ---------------
        a = a * np.exp(-eta * grad)           # multiplicative update
        a = proj_simplex(a)                   # keep on the simplex

        # ----- evaluate current score ------------------------------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()

        # ----- keep the best solution ------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- occasional global jitter (≈2 % of iterations) -------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            # recompute score after jitter
            fa = np.fft.rfft(a, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{elapsed:6.2f}s] jitter improved best to {best_score:.6f}")

        # ----- soft restart after stagnation (no improvement >30 s) -------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score still {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence seen (clipped to the bounds enforced by
    #    evaluate_sequence for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
