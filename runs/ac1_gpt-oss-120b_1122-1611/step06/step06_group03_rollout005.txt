```
def _polish_expg(a, n, fft_len, deadline, start_score):
    best = a.copy()
    best_score = start_score
    start_time = time.monotonic()
    # hyperparameters
    eta_start=0.2
    eta_end=0.005
    beta_start=10.0
    beta_end=20000.0
    while time.monotonic() < deadline:
        # compute convolution and max
        conv = _conv_self(a, fft_len)  # O(N log N)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic()-start_time
            print(f"[{elapsed:.2f}s] polish: new best {best_score:.6f}")

        # schedule parameters
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        # adapt step size and beta
        eta = eta_start * (1 - elapsed/total) + eta_end * (elapsed/total)
        beta = beta_start * ((beta_end / beta_start) ** (elapsed/total))

        # compute softmax weights
        # shift to avoid underflow
        # w = exp(beta * (conv - cur_max))
        # careful with large beta; may produce zero weights for many
        # Use np.exp with dtype float64
        exp_term = np.exp(beta * (conv - cur_max))
        w_sum = exp_term.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # fallback: use uniform weights for stability
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = exp_term / w_sum

        # compute gradient = 2 * correlation of a with w
        grad = _gradient_correlation(a, w)  # returns length n vector

        # exponentiated gradient update
        factor = np.exp(-eta * grad)
        a = a * factor
        # clip nonnegative (should stay)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= s

        # early break
        if time.monotonic() > deadline - 0.1:
            break
    return best, best_score
```

"""
Step‑function optimisation for ``evaluate_sequence``.

The algorithm consists of three phases that all respect the 1000 s
budget.

1. **Fast parametric search** – several simple families of non‑negative
   vectors are generated (analytic √‑difference, power‑law,
   geometric, and a product of the two).  Each candidate is normalised
   to the probability simplex (∑a_i = 1) and its internal score
   ``2·n·max(conv)`` is evaluated with a single FFT.  The best few
   candidates are kept for the next phase.

2. **Optional short simulated‑annealing** – a cheap Metropolis walk
   performs mass transfers, block scalings and global jitter.  It is
   limited to at most 20 % of the total time and merely serves to
   escape very bad local minima.

3. **Deterministic polishing** – an exponentiated‑gradient (mirror‑descent)
   optimiser minimises a smooth approximation of the true
   ``max_k (a∗a)_k``.  The smoothing uses a soft‑max with a temperature
   β that is increased from 10 to 2·10⁴ while the step‑size η is
   annealed from 0.2 down to 0.005.  The gradient of the smoothed
   objective is obtained by a single correlation via FFT (``_gradient_correlation``).
   The update respects non‑negativity and the ℓ₁‑simplex.

The resulting vector is clipped to the limits required by
``evaluate_sequence`` and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via real FFT, returns length 2·len(a)-1 array."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the internal objective 2·n·max(a∗a) for a normalised vector."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Simple analytic start (√‑difference)
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """a_i = sqrt(i+1)-sqrt(i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Global fallback (previous best construction if present)
# ----------------------------------------------------------------------
def _maybe_use_global() -> np.ndarray | None:
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Parameterised families
# ----------------------------------------------------------------------
def _power_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^p - i^p , p>0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p - i ** p
    a /= a.sum()
    return a


def _geometric_shape(n: int, r: float) -> np.ndarray:
    """a_i ∝ r**i , 0<r<1."""
    a = r ** np.arange(n, dtype=np.float64)
    a /= a.sum()
    return a


def _product_shape(n: int, p: float, r: float) -> np.ndarray:
    """a_i ∝ ((i+1)^p - i^p) * r**i."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** p - i ** p) * (r ** i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Gradient of the soft‑max weighted convolution
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k with respect to a.
    Implements the correlation a ⋆ w using a single FFT pair.
    """
    n = a.size
    L = w.size                     # should be 2·n‑1
    # size large enough for linear convolution of length L and n
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Deterministic polishing (exponentiated‑gradient on soft‑max)
# ----------------------------------------------------------------------
def _polish_expg(a_start: np.ndarray,
                 n: int,
                 fft_len: int,
                 deadline: float,
                 start_score: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent polishing with a smooth max (soft‑max) surrogate.
    Returns the best vector found and its (internal) score.
    """
    a = a_start.copy()
    best = a.copy()
    best_score = start_score
    start_time = time.monotonic()

    # hyper‑parameters (may be tuned later)
    eta_start = 0.20
    eta_end = 0.005
    beta_start = 10.0
    beta_end = 20000.0

    while time.monotonic() < deadline:
        # current convolution and score
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # keep the best ever seen
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] polish: new best = {best_score:.6f}")

        # schedule η and β
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        if total <= 0:
            break
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start * ((beta_end / beta_start) ** (elapsed / total))

        # soft‑max weights (stable formulation)
        shift = cur_max
        exp_terms = np.exp(beta * (conv - shift))
        w_sum = exp_terms.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fall back to uniform weights (should almost never happen)
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_terms / w_sum

        # gradient of the smooth surrogate
        grad = _gradient_correlation(a, w)

        # exponentiated‑gradient update on the simplex
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # tiny safety margin before deadline
        if time.monotonic() > deadline - 0.05:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Short simulated‑annealing (optional, cheap exploration)
# ----------------------------------------------------------------------
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Very light SA – runs while time permits and returns the best found vector."""
    cur_a = a0.copy()
    cur_score = _compute_score(cur_a, fft_len, n)

    best_a = cur_a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    # Temperature schedule parameters
    T0 = cur_score * 0.30
    Tmin = 1e-7
    # move scales
    max_pair_frac = 0.5
    sigma_block = 0.35

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        # exponential cooling
        T = T0 * (Tmin / T0) ** (elapsed / total) if total > 0 else Tmin

        # ----- propose a move -----
        a_new = cur_a.copy()
        r = rng.random()
        if r < 0.5:                               # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a_new[i] > 1e-15:
                delta = rng.random() * a_new[i] * max_pair_frac
                a_new[i] -= delta
                a_new[j] += delta
        elif r < 0.8:                             # block scaling
            width = rng.integers(1, max(2, n // 15))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, sigma_block))
            a_new[start:start + width] *= factor
        else:                                      # global jitter
            jitter = np.exp(rng.normal(0.0, sigma_block / 2.0, size=n))
            a_new *= jitter

        # simplex projection
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= s

        new_score = _compute_score(a_new, fft_len, n)

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if T > 0.0:
                prob = np.exp((cur_score - new_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_a = a_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = cur_a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] SA new best = {best_score:.6f}")

        # early stop guard
        if time.monotonic() > deadline - 0.05:
            break

    return best_a, best_score

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget expires.
    The routine builds a small set of parametric candidates, optionally
    refines them with a brief simulated‑annealing phase and finally
    applies a long exponentiated‑gradient polishing.
    """
    rng = np.random.default_rng(seed)

    # -------------------- problem size --------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # -------------------- initial candidates --------------------
    candidates = []

    # analytic √‑difference
    candidates.append(_analytic_start(n))

    # maybe a previously stored construction
    cand_global = _maybe_use_global()
    if cand_global is not None:
        candidates.append(cand_global)

    # power‑law family
    for p in np.linspace(0.35, 0.85, 21):
        candidates.append(_power_shape(n, p))

    # geometric family
    for r in np.linspace(0.96, 0.9999, 15):
        candidates.append(_geometric_shape(n, r))

    # product family (coarse grid)
    for p in np.linspace(0.4, 0.8, 5):
        for r in np.linspace(0.97, 0.995, 5):
            candidates.append(_product_shape(n, p, r))

    # -------------------- evaluate candidates --------------------
    best_a = None
    best_score = float("inf")
    for a in candidates:
        score = _compute_score(a, fft_len, n)
        if score < best_score:
            best_score = score
            best_a = a.copy()

    # -------------------- time bookkeeping --------------------
    start_time = time.monotonic()
    overall_deadline = start_time + max(0.0, float(budget_s) - 0.5)   # safety margin
    elapsed = time.monotonic() - start_time
    remaining = overall_deadline - time.monotonic()

    # -------------------- short SA (use at most 20 % of the time) --------------------
    if remaining > 20.0:
        sa_deadline = time.monotonic() + remaining * 0.20
        best_a, best_score = _simulated_annealing(best_a, n, fft_len, sa_deadline, rng)
        remaining = overall_deadline - time.monotonic()

    # -------------------- polishing (majority of the budget) --------------------
    if remaining > 5.0:
        polish_deadline = overall_deadline
        best_a, best_score = _polish_expg(best_a, n, fft_len, polish_deadline, best_score)

    # -------------------- final sanity clipping --------------------
    best_a = np.clip(best_a, 0.0, 1000.0)

    # return as plain Python list for the harness
    return [float(v) for v in best_a]
