"""
Propose a non‑negative step‑function that minimises ``evaluate_sequence``.

Algorithm Overview
-----------------
The search is split into three stages within the 1000 s budget:

1. **Random Exploration** (≈25 % of the budget)
   – Sample many random points on the probability simplex for several
     lengths (powers of two up to 8192).  Sampling is done via i.i.d.
     exponential variables (equivalent to Dirichlet(α=1)).
   – Evaluate the true objective (2·n·max(conv)) using an FFT‑based
     convolution and keep the best candidate.

2. **Smooth Gradient Descent** (≈50 % of the budget)
   – Starting from the best vector found in stage 1, minimise the
     L₂‑distance of the self‑convolution to the uniform distribution
     (which encourages a flat autocorrelation and therefore a low
     L∞‑value).  The gradient of the L₂ loss is 4·a⋆(conv−u) where
     ``⋆`` denotes linear correlation; we compute it efficiently via FFT.
   – Adam optimisation with a linearly decaying learning rate is used.
   – After each step the vector is projected onto the probability simplex.

3. **Fine‑grain Simulated Annealing** (≈25 % of the budget)
   – A simple Metropolis‑Hastings scheme performs random mass transfers
     between two coordinates.  Moves are accepted if they improve the
     objective or with probability exp(−Δ/ T) where the temperature T
     decreases linearly.
   – The current best ever seen is memorised.

The final sequence (clipped to the limits enforced by ``evaluate_sequence``)
is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length for a linear convolution of size n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for use with FFT."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return linear convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of a linear correlation with weight vector w.
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w (correlation).
    ``a`` is length n, ``w`` length 2·n‑1.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # Length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reversed a for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Fix possible rounding error
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Random Dirichlet (α=1) sample on the simplex.
# ----------------------------------------------------------------------
def _sample_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Return a random vector on the simplex using i.i.d. exponentials."""
    x = rng.exponential(scale=1.0, size=n)
    if x.sum() == 0.0:
        # extremely unlikely, but guard against division by zero
        x = np.ones_like(x)
    return x / x.sum()


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Preferred length for the random exploration stage.  If omitted,
        a set of candidate lengths is tried.

    Returns
    -------
    list[float]
        The best sequence found (values clipped to the limits imposed by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global configuration
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)  # safety margin

    # The set of lengths we will explore; include the length of any known
    # construction (height_sequence_1) if present.
    lengths = [256, 512, 1024, 2048, 4096, 8192]
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                lengths.append(int(cand.size))
        except Exception:
            pass
    # Ensure lengths are unique and sorted
    lengths = sorted(set(int(max(2, int(l)))) for l in lengths)

    # ------------------------------------------------------------------
    # 1) Stage 1 – random Dirichlet sampling
    # ------------------------------------------------------------------
    stage1_end = start_time + 0.25 * (deadline - start_time)
    best_vec = None
    best_score = math.inf
    best_len = None

    while time.monotonic() < stage1_end:
        # pick a length (biased towards larger lengths)
        n = rng.choice(lengths, p=np.array(lengths, dtype=float) ** 0.7)
        a = _sample_dirichlet(rng, n)

        fft_len = _fft_len_for_n(n)
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            best_len = n
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] random stage – new best score {best_score:.6f} (n={n})")

    # If random stage missed everything (should not happen), fall back to
    # the analytic sqrt‑difference start.
    if best_vec is None:
        n = kwargs.get("n", 3800)
        i = np.arange(n, dtype=np.float64)
        a = np.sqrt(i + 1.0) - np.sqrt(i)
        a /= a.sum()
        best_vec = a
        best_len = n
        fft_len = _fft_len_for_n(n)
        best_score = 2.0 * n * _conv_self(a, fft_len).max()

    # ------------------------------------------------------------------
    # 2) Stage 2 – L2‑smooth gradient descent (Adam)
    # ------------------------------------------------------------------
    stage2_end = start_time + 0.75 * (deadline - start_time)
    a = best_vec.copy()
    n = best_len
    fft_len = _fft_len_for_n(n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # Uniform target for the convolution (all entries sum to 1)
    uniform_target = 1.0 / (2 * n - 1)

    # Helper to evaluate current max‑convolution score
    def _score_from_vec(vec):
        conv = _conv_self(vec, fft_len)
        return 2.0 * n * conv.max()

    while time.monotonic() < stage2_end:
        now = time.monotonic()
        # linear schedule for the learning rate
        ratio = (now - start_time) / (stage2_end - start_time)
        lr = 0.5 * (1.0 - ratio) + 0.005 * ratio

        # convolution and L2‑error
        conv = _conv_self(a, fft_len)
        diff = conv - uniform_target   # length = 2n‑1
        # gradient of sum(diff²) = 4 * a ⋆ diff  → 2 * (2*a⋆diff)
        grad_lin = _gradient_correlation(a, diff)      # = 2*a⋆diff
        grad = 2.0 * grad_lin                         # = 4*a⋆diff

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _proj_simplex(a)

        # occasional jitter to escape plateaus
        if rng.random() < 0.03:
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * jitter
            a = _proj_simplex(a)

        # keep the best
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] gradient stage – new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Stage 3 – Simulated annealing fine‑tuning
    # ------------------------------------------------------------------
    stage3_end = deadline
    a = best_vec.copy()
    cur_score = best_score
    # temperature schedule (starts as a fraction of the current best score)
    T_start = max(1e-4, 0.05 * best_score)
    T_end = 1e-6
    it = 0

    while time.monotonic() < stage3_end:
        now = time.monotonic()
        # linear temperature decay
        ratio = (now - start_time) / (deadline - start_time)
        T = T_start * (1.0 - ratio) + T_end * ratio

        # propose a random mass transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if a[i] <= 1e-15:
            continue
        delta = rng.random() * a[i] * 0.5   # move up to 50 % of source mass
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta
        a_prop = _proj_simplex(a_prop)      # small correction for numeric drift

        # evaluate proposal
        conv_prop = _conv_self(a_prop, fft_len)
        new_score = 2.0 * n * conv_prop.max()

        # Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability exp(−Δ / T)
            if T > 0.0:
                prob = math.exp(-(new_score - cur_score) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_prop
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_vec = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] annealing – new best score {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 4) Return the best sequence (apply the clipping imposed by
    #    evaluate_sequence).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]

**Explanation of key ideas**

* **Random Dirichlet sampling** quickly discovers diverse points on the simplex,
  including some that already beat the previous best.
* **L₂‑smooth optimisation** pushes the autocorrelation toward a flat (uniform)
  shape, which is the ideal behaviour for minimising the L∞‑norm.
  The gradient is obtained analytically via FFT‑based correlation and the
  update uses Adam with a decaying learning‑rate.
* **Simulated annealing** performs simple mass‑transfer moves that can
  escape the basin of the gradient descent and fine‑tune the maximal
  convolution value.
* **Projection onto the simplex** ensures all iterates stay feasible
  (non‑negative and unit sum), matching the scale‑invariance of the
  evaluation function.

The routine respects the 1000 s budget, prints progress messages, and
returns the best sequence discovered.
