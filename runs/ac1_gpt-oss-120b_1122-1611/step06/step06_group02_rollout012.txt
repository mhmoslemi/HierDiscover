"""
Exponential‑gradient with momentum + extended initialisation.

The optimiser works on the probability simplex by representing the
distribution directly (no log‑softmax) and performing multiplicative
updates a_i ← a_i·exp(–η·m_i), where m_i is a momentum‑averaged
sub‑gradient of the objective
    f(a) = 2·n·max_k (a*a)_k .
Both the exact sub‑gradient of the worst convolution entry and a
smooth “log‑sum‑exp’’ gradient are mixed.  Random block perturbations
and cheap pair‑wise mass‑transfer moves are applied to escape plateaus.
The search is performed for several lengths (up to 10000) with many
restarts; the current global best is repeatedly polished with longer
runs.  The whole routine respects the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def softmax_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     beta: float,
                     fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))          # safe, max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    # extract the n entries that correspond to derivative wrt a
    grad_part = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Simple random moves
# ----------------------------------------------------------------------


def random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out = np.full_like(out, 1.0 / n)
    else:
        out /= out_sum
    return out


def pairwise_move(a: np.ndarray,
                  rng: np.random.Generator,
                  fft_len_conv: int) -> np.ndarray:
    """Transfer a random fraction of mass from i to j (accept only if improving)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2_sum = a2.sum()
    if a2_sum == 0.0:
        a2 = np.full_like(a2, 1.0 / n)
    else:
        a2 /= a2_sum
    conv_old = conv_self(a, fft_len_conv)
    conv_new = conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a

# Score helper (fast)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()

def init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def init_gaussian(rng: np.random.Generator, n: int) -> np.ndarray:
    """Gaussian‑like bump centred in the middle (width sampled)."""
    # we deliberately centre the Gaussian – later optimisation will push mass to the edges.
    sigma = rng.uniform(n / 15.0, n / 5.0)
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def init_cosine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """Cosine‑based shape that is large near the ends and small in the centre."""
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (idx + 0.5) / n)
    vals = np.abs(vals)  # make it symmetric and non‑negative
    a = np.power(vals, p)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linearly interpolate an existing distribution to a target length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= total
    return a

# Local optimiser – exponential‑gradient + momentum
# ----------------------------------------------------------------------


def local_optimize(initial: np.ndarray,
                   rng: np.random.Generator,
                   deadline: float,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Refine ``initial`` using a momentum‑augmented exponentiated‑gradient
    scheme.  The routine returns the best vector discovered and its fast
    score (2·n·max(conv)).
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = fft_len_for_len(2 * n - 1)
    fft_len_grad = fft_len_for_len(3 * n - 2)

    # momentum vector (same shape as a)
    m = np.zeros_like(a)

    # hyper‑parameters
    base_lr = 0.9          # initial learning‑rate factor
    gamma = 0.9            # momentum decay
    beta = 0.5             # starting temperature for the smooth‑max gradient
    beta_mul = 1.06        # multiplicative growth per soft‑max step
    subgrad_prob = 0.55    # probability of using the exact sub‑gradient
    block_prob = 0.03
    pair_prob = 0.03

    best_vec = a.copy()
    best_score = fast_score(a, fft_len_conv)

    for t in range(1, max_iters + 1):
        now = time.time()
        if now >= deadline:
            break

        conv = conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ------------------------------------------------------------------
        # Gradient choice
        # ------------------------------------------------------------------
        if rng.random() < subgrad_prob:
            # Exact sub‑gradient of the worst entry (cheap)
            raw_grad = subgradient_step(a, conv)
        else:
            # Smooth max – β grows gradually
            beta = min(1e5, beta * beta_mul)
            raw_grad = softmax_gradient(a, conv, beta, fft_len_grad)

        # Scale to the true objective (2·n·max)
        grad_a = (2.0 * n) * raw_grad

        # Clip to keep numbers in a comfortable range
        grad_a = np.clip(grad_a, -10.0, 10.0)

        # Momentum update
        m = gamma * m + (1.0 - gamma) * grad_a

        # Exponentiated‑gradient multiplicative step
        lr = base_lr / np.sqrt(t)
        a = a * np.exp(-lr * m)

        # Re‑project onto the simplex
        a_sum = a.sum()
        if a_sum == 0.0 or not np.isfinite(a_sum):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ------------------------------------------------------------------
        # Random exploratory moves (occasionally)
        # ------------------------------------------------------------------
        if rng.random() < block_prob:
            a = random_block_perturb(a, rng)

        if rng.random() < pair_prob:
            a = pairwise_move(a, rng, fft_len_conv)

    # Final evaluation of best candidate
    final_score = fast_score(best_vec, fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score

# ----------------------------------------------------------------------


def fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. a.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def softmax_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     beta: float,
                     fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max
         f_β(a) = (1/β)·log Σ_i exp(β·conv_i)
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))          # safe: max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    # extract the n entries that correspond to derivative wrt a
    grad_part = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Random perturbation moves
# ----------------------------------------------------------------------


def random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out = np.full_like(out, 1.0 / n)
    else:
        out /= s
    return out


def pairwise_move(a: np.ndarray,
                  rng: np.random.Generator,
                  fft_len_conv: int) -> np.ndarray:
    """Transfer a random fraction of mass from i to j (accept only if improving)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    s = a2.sum()
    if s == 0.0:
        a2 = np.full_like(a2, 1.0 / n)
    else:
        a2 /= s
    conv_old = conv_self(a, fft_len_conv)
    conv_new = conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a


# ----------------------------------------------------------------------
# Fast score (avoids evaluate_sequence overhead)
# ----------------------------------------------------------------------


def fast_score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    a = rng.exponential(scale=1.0, size=n)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def init_gaussian(rng: np.random.Generator, n: int) -> np.ndarray:
    """Gaussian‑like bump centred in the middle (width sampled)."""
    sigma = rng.uniform(n / 15.0, n / 5.0)
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def init_cosine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """Cosine‑based shape large near the ends, small in the centre."""
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.cos(np.pi * (idx + 0.5) / n)
    vals = np.abs(vals)
    a = np.power(vals, p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linearly interpolate an existing distribution to a new length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Local optimiser – exponentiated gradient with momentum
# ----------------------------------------------------------------------


def local_optimize(initial: np.ndarray,
                   rng: np.random.Generator,
                   deadline: float,
                   max_iters: int = 3000) -> tuple[np.ndarray, float]:
    """
    Refine ``initial`` using a momentum‑augmented exponentiated‑gradient
    scheme.  Returns the best vector discovered and its fast score.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = fft_len_for_len(2 * n - 1)
    fft_len_grad = fft_len_for_len(3 * n - 2)

    # momentum on the gradient (same shape as a)
    m = np.zeros_like(a)

    # hyper‑parameters
    base_lr = 0.9
    gamma = 0.9                # momentum decay
    beta = 0.5                 # temperature for smooth‑max gradient
    beta_mul = 1.06
    subgrad_prob = 0.55
    block_prob = 0.03
    pair_prob = 0.03

    best_vec = a.copy()
    best_score = fast_score(a, fft_len_conv)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # Choose a gradient type
        # --------------------------------------------------------------
        if rng.random() < subgrad_prob:
            raw_grad = subgradient_step(a, conv)           # cheap exact sub‑gradient
        else:
            beta = min(1e5, beta * beta_mul)               # increase temperature
            raw_grad = softmax_gradient(a, conv, beta, fft_len_grad)

        # Scale to the true objective (2·n·max)
        grad_a = (2.0 * n) * raw_grad
        grad_a = np.clip(grad_a, -10.0, 10.0)

        # Momentum update
        m = gamma * m + (1.0 - gamma) * grad_a

        # Exponentiated‑gradient multiplicative step
        lr = base_lr / np.sqrt(t)
        a = a * np.exp(-lr * m)

        # Project back onto the simplex
        s = a.sum()
        if s == 0.0 or not np.isfinite(s):
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # --------------------------------------------------------------
        # Occasional random exploratory moves
        # --------------------------------------------------------------
        if rng.random() < block_prob:
            a = random_block_perturb(a, rng)

        if rng.random() < pair_prob:
            a = pairwise_move(a, rng, fft_len_conv)

    final_score = fast_score(best_vec, fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.4)   # tiny safety margin

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Seed with a previously known construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / s
                cand_score = fast_score(cand,
                                         fft_len_for_len(2 * cand.size - 1))
                best_vec = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start:5.2f}s] Seeded from previous best: "
                      f"{best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Length schedule (favours medium‑large vectors)
    # ------------------------------------------------------------------
    lengths = [2000, 3000, 3809, 4000, 5000, 6000, 8000, 10000, 12000]

    # ------------------------------------------------------------------
    # Initialisation families and probabilities
    # ------------------------------------------------------------------
    init_methods = ['dirichlet', 'sine', 'gaussian', 'cosine', 'interpolate']
    init_weights = np.array([0.30, 0.20, 0.20, 0.15, 0.15])
    init_weights = init_weights / init_weights.sum()

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # length selection – bias a little towards the larger sizes
        len_probs = np.array([0.08, 0.12, 0.12, 0.14, 0.20,
                              0.16, 0.10, 0.06, 0.02])
        len_probs = len_probs / len_probs.sum()
        n = int(rng.choice(lengths, p=len_probs))

        # pick an initialisation method
        which = rng.choice(len(init_methods), p=init_weights)
        method = init_methods[which]

        if method == 'dirichlet':
            a0 = init_random_dirichlet(rng, n)
        elif method == 'sine':
            a0 = init_sine_power(rng, n)
        elif method == 'gaussian':
            a0 = init_gaussian(rng, n)
        elif method == 'cosine':
            a0 = init_cosine_power(rng, n)
        elif method == 'interpolate' and best_vec is not None:
            a0 = init_interpolate(rng, best_vec, n)
        else:
            a0 = init_random_dirichlet(rng, n)   # fallback

        # iteration budget – more iterations for smaller n
        iters = int(3500 * (5000.0 / max(n, 2000)))

        a_opt, score_opt = local_optimize(a0, rng, deadline, max_iters=iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:5.2f}s] New global best: {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Polishing runs on the current global best when time permits
        # ------------------------------------------------------------------
        if best_vec is not None and time.time() < deadline - 2.0:
            a_pol, score_pol = local_optimize(best_vec.copy(),
                                              rng,
                                              deadline,
                                              max_iters=6000)
            if score_pol < best_score:
                best_score = score_pol
                best_vec = a_pol.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:5.2f}s] Polished best: {best_score:.6f}")

        # Stop early if the required bound has been reached
        if best_score <= 1.5025:
            print(f"[{time.time() - start:5.2f}s] Target reached: {best_score:.6f}")
            break

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 3000
        best_vec = np.full(n, 1.0 / n)
        best_score = fast_score(best_vec,
                                fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Return plain Python floats for the harness
    return [float(x) for x in best_vec]
