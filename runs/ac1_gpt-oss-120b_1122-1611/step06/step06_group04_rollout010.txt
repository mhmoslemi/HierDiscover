"""
Propose a non‑negative sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Search a small family of analytic windows**  
   – Power‑difference family   a_i = ((i+1)^β – i^β) / n^β   (β∈[0.35,0.65])  
   – Classical windows: Hann, Hamming, Blackman, Kaiser (β∈{0,2,5,8,12})  
   – Simple shapes: exponential decay, triangular and uniform.  
   For each length candidate (2000 … 5000) and each set of parameters we
   evaluate the cheap surrogate  

        score = 2·n·max_k (a∗a)_k                (sum(a)=1).

   The best window is kept as a starting point.

2. **Local refinement** – a few hundred Adam steps on the smooth
   surrogate ``soft‑max_β (a∗a)`` (β raised slowly from 0.5 to 5000).
   After each step the vector is projected onto the probability simplex
   (non‑negative, unit sum).  Occasional log‑normal perturbations and a
   simple mass‑transfer move help escape shallow minima.

3. **Return** the best sequence found (clipped to the limits enforced by
   ``evaluate_sequence``).  The whole routine respects the 1000 s time
   budget.
"""

import time
import numpy as np
from scipy.special import i0  # Kaiser window

# ----------------------------------------------------------------------
# Helper functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Next power‑of‑two ≥ 2·n‑1 for FFT‑based linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length = 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k  =  2·a ⋆ w .
    ``a`` – length n, ``w`` – length 2·n‑1.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reversed a for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


def _score_fast(a: np.ndarray, fft_len: int) -> float:
    """Fast surrogate score  =  2·n·max(a∗a)  (assumes Σa = 1)."""
    n = a.size
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Analytic families of candidate windows
# ----------------------------------------------------------------------


def _shape_power_diff(n: int, beta: float) -> np.ndarray:
    """a_i ∝ (i+1)^β – i^β   (telescopic sum = n^β)."""
    i = np.arange(n, dtype=np.float64)
    raw = (i + 1.0) ** beta - i ** beta
    # sum(raw) = n**beta by construction – use it for exact normalisation
    a = raw / (n ** beta)
    return a


def _shape_exponential(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i  (0<r<1)."""
    raw = r ** np.arange(n, dtype=np.float64)
    return raw / raw.sum()


def _shape_hann(n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    raw = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    return raw / raw.sum()


def _shape_hamming(n: int) -> np.ndarray:
    """Hamming window."""
    raw = 0.54 - 0.46 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    return raw / raw.sum()


def _shape_blackman(n: int) -> np.ndarray:
    """Blackman window."""
    raw = (0.42
           - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
           + 0.08 * np.cos(4.0 * np.pi * np.arange(n) / (n - 1)))
    return raw / raw.sum()


def _shape_kaiser(n: int, beta: float) -> np.ndarray:
    """Kaiser window with shape parameter beta."""
    k = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (k - alpha) / alpha
    raw = i0(beta * np.sqrt(1.0 - t * t))
    return raw / raw.sum()


def _shape_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape (zero at the ends)."""
    raw = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1))
    return raw / raw.sum()


def _shape_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n)

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed (reproducibility).
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Preferred length of the sequence; if omitted a small set of lengths

    Returns
    -------
    list[float]
        The best sequence found (clipped to the limits used in ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # tiny safety margin

    # 1) Choose a set of lengths to explore
    # ------------------------------------------------------------------
    n_fixed = int(kwargs.get("n", 0))
    if n_fixed >= 2:
        length_candidates = [n_fixed]
    else:

    # 2) Parameter grids for the analytic families
    # ------------------------------------------------------------------
    beta_grid = np.linspace(0.35, 0.65, 31)            # power‑difference
    r_grid = np.linspace(0.950, 0.9999, 30)           # exponential decay

    # ------------------------------------------------------------------
    # 3) Exhaustive (but cheap) evaluation of all candidates
    # ------------------------------------------------------------------
    best_score = np.inf
    best_seq = None

        for maker in (_shape_hann, _shape_hamming, _shape_blackman,

            a = maker(n)
            sc = _score_fast(a, fft_len)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a, n

        # ----- power‑difference family -----
        for beta in beta_grid:
            a = _shape_power_diff(n, beta)
            sc = _score_fast(a, fft_len)
            if sc < best_score:

        # ----- exponential decay -----
        for r in r_grid:
            a = _shape_exponential(n, r)
            sc = _score_fast(a, fft_len)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a, n

        # ----- Kaiser windows -----
        for kbeta in kaiser_grid:
            a = _shape_kaiser(n, kbeta)
            sc = _score_fast(a, fft_len)
            if sc < best_score:
                best_score, best_seq, best_n = sc, a, n

        # Occasionally print progress for long runs
        now = time.monotonic()
        if now - start_time > 30.0 and (now - start_time) % 30 < 0.01:
            print(f"[{now - start_time:6.2f}s] best so far = {best_score:.6f} (n={best_n})")

        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 4) Local refinement with Adam on a smooth surrogate
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = best_n
    fft_len = _fft_len_for_n(n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    # Schedules (linear in elapsed proportion)
    lr_start, lr_end = 0.40, 0.002
    beta_start, beta_end = 0.5, 5000.0

    # Random‑perturbation knobs
    noise_prob = 0.028
    noise_sigma = 0.12
    swap_prob = 0.06
    restart_gap = 120.0   # seconds without improvement → soft restart

    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        ratio = min(1.0, elapsed / total)

        # learning‑rate and soft‑max temperature
        lr = lr_start * (1.0 - ratio) + lr_end * ratio
        beta_temp = beta_start + (beta_end - beta_start) * ratio

        # current convolution and its maximum
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep the best solution seen so far
        if cur_score < best_score - 1e-12:
            best_score, best_seq, last_improve = cur_score, a.copy(), now
            print(f"[{elapsed:6.2f}s] new best → {best_score:.6f}")

        # soft‑max weights (stable computation)
        diff = conv - cur_max               # ≤ 0
        w_raw = np.exp(beta_temp * diff)
        w_sum = w_raw.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            mask = conv == cur_max
            w_raw = np.where(mask, 1.0, 0.0)
            w_sum = w_raw.sum()
        w = w_raw / w_sum

        # gradient of the surrogate
        grad = _gradient_correlation(a, w)

        # Adam update
        t += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** t)
        v_hat = v / (1.0 - 0.999 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        a = a - step
        a = _proj_simplex(a)

        # occasional log‑normal noise (helps to explore)
        if rng.random() < noise_prob:
            factor = np.exp(rng.normal(0.0, noise_sigma, size=n))
            a = a * factor
            a = _proj_simplex(a)

        # occasional pairwise mass transfer
        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-15:
                delta = rng.random() * a[i] * 0.5
                a[i] -= delta
                a[j] += delta
                a = _proj_simplex(a)

        # soft restart if we have been stuck for a while
        if now - last_improve > restart_gap:
            a = best_seq.copy()
            jitter = np.exp(rng.normal(0.0, 0.11, size=n))
            a = a * jitter
            a = _proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best vector (apply the clipping used by evaluate_sequence)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
