```
        a /= a.sum()

        cur_score = 2 * n * cur_max

            best_score = cur_score
            best_a = a.copy()

        kmax = int(np.argmax(conv))
```

```
    # Now we have a starting point best_a and best_score.
    # Use gradient descent with subgradient.
    x = np.log(best_a + 1e-30)   # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    lr_start = 0.8
    lr_end = 0.02
    jitter_prob = 0.03
    restart_gap = 120.0

    total = deadline - start
    it = 0
    last_improve = start

    fft_len = _fft_len_for_n(len(best_a))

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start

        # update lr
        lr = lr_start * (1 - elapsed/total) + lr_end * (elapsed/total)

        # compute a from logits
        a = np.exp(x - np.max(x))
        a /= a.sum()

        # compute convolution and score
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max / (np.sum(a) ** 2)   # sum=1 => 2n*cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (grad) = {best_score:.6f}")

        # compute subgradient from max index (or top ties)
        kmax = int(np.argmax(conv))
        # gradient a: g_i = 2*a[kmax-i] if within bounds else 0
        grad_a = np.zeros_like(a)
        lo = max(0, kmax - (n-1))
        hi = min(n-1, kmax)
        idx_i = np.arange(lo, hi+1)
        idx_j = kmax - idx_i
        grad_a[idx_i] = 2.0 * a[idx_j]

        # compute gradient wrt logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # jitter
        if rng.random() < jitter_prob:
            x += rng.normal(0.0, 0.05, size=n)

        # restart if stuck
        if now - last_improve > restart_gap:
            # reinitialize from best
            x = np.log(best_a + 1e-30)
            x += rng.normal(0.0, 0.2, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart - best score {best_score:.6f}")

    # final best
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
```

The score returned by `evaluate_sequence` is  

\[
\text{score}= \frac{2\,n\;\max_k (a * a)_k}{(\sum_i a_i)^2}
\]

where `*` is the linear convolution.  Because the score is invariant under scaling of the
vector we can **normalise the sum to 1** and minimise simply  

\[
\min_{a\ge 0,\;\sum a_i = 1}\; 2n \max_k (a*a)_k .
\]

A very good analytic construction is the “square‑root‑difference’’ shape  

\[
a_i \propto \sqrt{i+1}-\sqrt{i}\qquad (i=0,\dots ,n-1),
\]

which gives a score ≈ 1.514 for n≈3800.  
We improve on it by two ideas

1. **One‑parameter families** – try simple power‑law profiles  
   \(a_i \propto (i+1)^{-p}\) (and the same law with a small offset)  
   for many values of `p`.  Computing the score for a few dozen candidates
   is cheap (FFT of length ≈ 2 n) and gives a better starting point than the
   pure square‑root shape.

2. **Projected sub‑gradient descent on the simplex** –  
   after a good start we directly optimise the true objective (the
   max‑convolution) instead of a smooth surrogate.  
   For the active index `k* = argmax (a*a)_k` the sub‑gradient w.r.t. `a` is  

   \[
   (\nabla_a)_i = 
   \begin{cases}
   2\,a_{k^*-i} & 0\le k^*-i < n\\[2pt]
   0             &\text{otherwise}
   \end{cases}
   \]

   To keep the iterates on the probability simplex we re‑parameterise with
   logits `x` (`a = softmax(x)`) and use the usual soft‑max Jacobian.
   An Adam optimiser with a linearly annealed learning‑rate together with
   occasional random jitter and a restart when no improvement happens for
   120 s works well inside the 1000 s budget.

The algorithm consists of three phases:

* **Phase 0 – initialise** (previous best, square‑root shape or a power law).
* **Phase 1 – cheap grid search** over `p∈[0.35,0.80]` and a small offset `δ`
  to obtain the best‑scoring deterministic shape.
* **Phase 2 – sub‑gradient Adam optimisation** starting from that shape,
  until the time budget expires.

All helper functions are written at top level, no closures or lambdas are used,
and the code respects the 2‑CPU limit by staying in pure NumPy.

```python
"""
Propose a non‑negative step function that (approximately) minimises
evaluate_sequence().

Algorithm
---------
*   Load the best known construction (global `height_sequence_1`), otherwise
    start from the square‑root‑difference shape.
*   Perform a cheap grid search over power‑law exponents `p` and a small
    offset `δ` (`a_i ∝ (i+δ+1)^{-p}`); keep the best candidate.
*   Refine the best candidate with a projected sub‑gradient (Adam) method.
    The sub‑gradient is computed for the *active* convolution index
    (`k = argmax (a*a)_k`).  Logits `x` are updated and projected onto the
    simplex via the soft‑max Jacobian.
*   Random jitter and a hard‑restart after 120 s without improvement keep the
    search from stagnating.
*   Return the best sequence found (clipped to the limits enforced by
    evaluate_sequence).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities – length is the next power of two that can hold a full
# self‑convolution of a vector of length n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    # smallest power of two ≥ size
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·n‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Helper: evaluate the (scale‑invariant) score for a probability vector.
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """a is assumed to sum to 1."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point used by the judge.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence() within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ----------------- problem size ---------------------------------------
    n = int(kwargs.get("n", 3000))
    if n < 2:
        n = 2

    # ----------------- start from a known good construction -----------------
    best_a = None
    best_score = np.inf

    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    cand = cand / cand.sum()
                    fft_len_c = _fft_len_for_n(cand.size)
                    sc = _score_of(cand, cand.size, fft_len_c)
                    best_a = cand
                    best_score = sc
                    n = cand.size
        except Exception:
            pass

    # ----------------- default square‑root‑difference shape -----------------
    if best_a is None:
        i = np.arange(n, dtype=np.float64)
        a0 = np.sqrt(i + 1.0) - np.sqrt(i)          # √‑difference shape
        a0 = a0 / a0.sum()
        best_a = a0.copy()
        fft_len_tmp = _fft_len_for_n(n)
        best_score = _score_of(best_a, n, fft_len_tmp)

    # ----------------------------------------------------------------------
    # Phase 1 – cheap grid search over power‑law families.
    # ----------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)   # safety margin
    fft_len = _fft_len_for_n(n)

    # Grid over exponent p
    for p in np.linspace(0.35, 0.80, 37):                # 0.0125 step
        a = (np.arange(n, dtype=np.float64) + 1.0) ** (-p)
        a /= a.sum()
        sc = _score_of(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"[grid] p={p:.3f} → score={best_score:.6f}")
        if time.monotonic() >= deadline:
            break

    # Small offset δ (helps to reduce the very first coefficient)
    for delta in (0, 2, 5, 10):
        for p in np.linspace(0.45, 0.75, 31):
            idx = np.arange(n, dtype=np.float64) + delta + 1.0
            a = idx ** (-p)
            a /= a.sum()
            sc = _score_of(a, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"[grid] δ={delta}, p={p:.3f} → score={best_score:.6f}")
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # ----------------------------------------------------------------------
    # Phase 2 – sub‑gradient Adam optimisation on the simplex.
    # ----------------------------------------------------------------------
    # Parameterisation a = softmax(x) guarantees positivity & sum=1.
    x = np.log(best_a + 1e-30)          # logits
    m = np.zeros_like(x)               # Adam first moment
    v = np.zeros_like(x)               # Adam second moment

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr_start = 0.8
    lr_end = 0.02

    # Restart / jitter settings
    jitter_prob = 0.03
    restart_gap = 120.0                 # seconds without improvement → restart

    it = 0
    last_improve = start

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start

        # learning‑rate annealing (linear)
        lr = lr_start * (1.0 - elapsed / total) + lr_end * (elapsed / total)

        # current probability vector
        a = np.exp(x - np.max(x))
        a /= a.sum()

        # convolution & current score
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max               # sum(a)=1 ⇒ denominator =1
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ----- sub‑gradient of the active peak --------------------------------
        kmax = int(np.argmax(conv))

        # grad_a[i] = 2 * a[kmax-i] if the index is valid
        grad_a = np.zeros_like(a)
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        idx_i = np.arange(lo, hi + 1)
        idx_j = kmax - idx_i
        grad_a[idx_i] = 2.0 * a[idx_j]

        # transform gradient to the logit space (soft‑max Jacobian)
        dot = np.dot(grad_a, a)            # scalar <a,grad_a>
        grad_x = a * (grad_a - dot)        # shape (n,)

        # ----- Adam update ----------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- occasional stochastic jitter -----------------------------------
        if rng.random() < jitter_prob:
            x += rng.normal(0.0, 0.05, size=n)

        # ----- hard restart if stagnation ------------------------------------
        if now - last_improve > restart_gap:
            x = np.log(best_a + 1e-30)
            x += rng.normal(0.0, 0.2, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best sequence discovered (evaluate_sequence will clip it anyway)
    # ----------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
```
