"""
Optimization of a non‑negative step‑function (probability vector) that
minimises 2·n·max(a∗a) / (∑a)^2.

The search proceeds in three stages:

1. **Analytic families** – dense scans of the power‑difference families
   (a_i = (i+1)^r−i^r) and the shifted‑power‑law differences
   (a_i = (i+1+δ)^‑p − (i+2+δ)^‑p).  The best vectors are stored.

2. **Mixture refinement** – the two best shapes (one from each family) are
   linearly mixed with a weight w∈[0,1]; the convolution of the mixture is
   obtained analytically from the three constituent convolutions, and the
   best mixture is kept.

3. **Adam‑based soft‑max gradient descent** – the current best vector is
   re‑parameterised as a soft‑max of logits φ (so that the simplex and
   non‑negativity are satisfied automatically).  A smooth surrogate of the
   max‑convolution (log‑sum‑exp with a temperature β) provides a gradient
   that is back‑propagated to φ.  An Adam optimiser with a linear learning‑rate
   schedule refines the solution.  Random jitter, block perturbations and
   occasional restarts keep the search from stagnating.

The routine respects a time‑budget (default 1000 s) and finally returns the
best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Low‑level convolution / soft‑max utilities
# ------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two ≥ 2·n‑1 (full linear convolution length)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a∗a using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable softmax with temperature β."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ_k w_k (a∗a)_k w.r.t. a.
    It is 2·a⋆w (correlation).  Implemented with a single FFT.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    # Extract the central n entries (correlation at lags 0..n‑1)
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value for a normalised (∑a=1) vector."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ------------------------------------------------------------
# Analytic shape generators
# ------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^r − i^r  (0<r<1).  Returns a normalised probability vector."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_shape_p(p: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^‑p − (i+2)^‑p  (p>0).  Returns a normalised vector."""
    k = np.arange(1, n + 2, dtype=np.float64)          # 1 … n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _gen_shape_p_offset(p: float, n: int, offset: int) -> np.ndarray | None:
    """
    Shifted power‑law differences:
        a_i = (i+1+δ)^‑p − (i+2+δ)^‑p .
    δ is an integer offset (≥0).  Returns a normalised vector.
    """
    k = np.arange(1 + offset, n + 2 + offset, dtype=np.float64)
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


# ------------------------------------------------------------
# Main entry point
# ------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (probability vector) that
    approximately minimises `evaluate_sequence`.  The algorithm first
    scans analytic families, then refines the best candidate with an
    Adam optimiser using a soft‑max surrogate of the L∞‑objective.
    """
    rng = np.random.default_rng(seed)

    # -------- Problem size -------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # -------- Utility to keep the best solution ----------------------------
    best_a: np.ndarray | None = None
    best_score = np.inf

    def _consider(candidate: np.ndarray | None) -> None:
        nonlocal best_a, best_score
        if candidate is None:
            return
        sc = _score(candidate, fft_len)
        if sc < best_score:
            best_a = candidate
            best_score = sc
            print(f"[init] new best score = {sc:.6f}")

    # -------- 1) Scan analytic families ------------------------------------
    # r‑family (dense scan)
    r_vals = np.linspace(0.55, 0.96, 84)               # step ~0.0046
    r_shapes = []
    for r in r_vals:
        a = _gen_shape_r(r, n)
        if a is not None:
            r_shapes.append((r, a))
            _consider(a)

    # p‑family (dense scan)
    p_vals = np.linspace(0.05, 3.0, 291)               # step ~0.01
    p_shapes = []
    for p in p_vals:
        a = _gen_shape_p(p, n)
        if a is not None:
            p_shapes.append((p, a))
            _consider(a)

    # shifted‑power‑law families (a few offsets + a few p‑values)
    offsets = [0, 5, 10, 20, 50, 100]
    for offset in offsets:
        for p in [0.5, 0.8, 1.2, 1.5, 2.0]:
            a = _gen_shape_p_offset(p, n, offset)
            if a is not None:
                _consider(a)

    # -------- 2) Use the previously discovered construction if present -----
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    cand = cand / cand.sum()
                    _consider(cand)
                    n = cand.size
                    fft_len = _fft_len_for_n(n)
        except Exception:
            pass

    # -------- 3) Linear mixtures of the two best analytic shapes ----------
    # pick the best r‑shape and the best p‑shape
    best_r = None
    best_p = None
    if r_shapes:
        best_r = min(r_shapes, key=lambda tup: _score(tup[1], fft_len))[1]
    if p_shapes:
        best_p = min(p_shapes, key=lambda tup: _score(tup[1], fft_len))[1]

    if best_r is not None and best_p is not None:
        conv_r = _conv_self(best_r, fft_len)
        conv_p = _conv_self(best_p, fft_len)
        cross = np.convolve(best_r, best_p)[: 2 * n - 1]  # full linear conv
        ws = np.linspace(0.0, 1.0, 101)
        for w in ws:
            mixed_conv = (w * w) * conv_r + ((1 - w) * (1 - w)) * conv_p + (2 * w * (1 - w)) * cross
            sc = float(2 * n * mixed_conv.max())
            if sc < best_score:
                best_score = sc
                best_a = w * best_r + (1 - w) * best_p
                print(f"[mix] w={w:.2f} improved score = {sc:.6f}")

    # If nothing has been found yet, fall back to a uniform vector.
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # ------------------------------------------------------------
    # 4) Adam‑based soft‑max gradient descent on the log‑it space
    # ------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # keep a tiny safety margin

    # initialise logits φ such that softmax(φ) = best_a
    eps = 1e-15
    phi = np.log(best_a + eps)

    m = np.zeros_like(phi)   # first moment
    v = np.zeros_like(phi)   # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    it = 0
    last_improve = start_time
    restart_gap = 180.0    # seconds without improvement → restart

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # ----- schedules -------------------------------------------------
        lr = 0.5 * (1.0 - elapsed / total) + 0.003 * (elapsed / total)
        beta_temp = 0.5 + (4000.0 - 0.5) * (elapsed / total)

        # ----- obtain current probability vector -------------------------
        phi_shift = phi - phi.max()           # for numerical stability
        a = np.exp(phi_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / a_sum

        # ----- exact score (for bookkeeping) ----------------------------
        conv = _conv_self(a, fft_len)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)

        # ----- keep the best ------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.1f}s] new best score = {best_score:.6f}")

        # ----- gradient of the smooth surrogate ---------------------------
        w = _softmax(conv, beta_temp)               # soft‑max weights over convolutions
        grad_a = _gradient_via_correlation(a, w)    # ∂ E/∂a
        dot = np.dot(grad_a, a)                     # Σ a_i ∂E/∂a_i
        grad_phi = a * (grad_a - dot)               # chain rule for soft‑max

        # ----- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_phi
        v = beta2 * v + (1.0 - beta2) * (grad_phi ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        phi -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional stochastic exploration ---------------------------
        if rng.random() < 0.01:                      # small jitter on logits
            jitter = np.exp(rng.normal(0.0, 0.08, size=n))
            phi += np.log(jitter)

        if rng.random() < 0.03:                      # block scaling on the probabilities
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.2))
            a_block = a.copy()
            a_block[start:start + width] *= mult
            a_block = np.clip(a_block, 0.0, None)
            a_block /= a_block.sum()
            # translate back to logits for the next iteration
            phi = np.log(a_block + eps)

        if rng.random() < 0.02:                      # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and a[i] > 1e-14:
                delta = rng.random() * a[i] * 0.4
                a[i] -= delta
                a[j] += delta
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                phi = np.log(a + eps)

        # ----- restart if stuck -------------------------------------------
        if now - last_improve > restart_gap:
            # re‑initialise logits around the best solution found so far
            phi = np.log(best_a + eps) + rng.normal(0.0, 0.12, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.1f}s] restart – best score {best_score:.6f}")

    # --------------------------------------------------------------------
    # Return the best vector (already normalised, all entries non‑negative)
    # --------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
