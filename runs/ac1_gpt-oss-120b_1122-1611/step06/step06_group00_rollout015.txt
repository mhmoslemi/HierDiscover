def proj_simplex(v):
    # v is np array
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def solve_lp(f, M):
    n = len(f)
    k_len = 2*n -1
    # Build A_ub
    A = np.zeros((k_len, n))
    for k in range(k_len):
        # i ranges max(0, k-(n-1)) to min(n-1, k)
        i_start = max(0, k - (n-1))
        i_end = min(n-1, k)
        # For each i in [i_start, i_end], g_i contributes f_{k-i}
        A[k, i_start:i_end+1] = f[k - np.arange(i_start, i_end+1)]
    # Now we have A @ g <= M
    # Objective: maximize sum g -> minimize -sum g.
    c = -np.ones(n) # minimize c^T g
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(k_len, M), bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback: return uniform maybe
        return np.full(n, 1/n)

def proj_simplex(v):
    # using algorithm from ... (descending sort).
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
       theta = 0
    else:
       rho = rho[-1]
       theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _optimize_n(n, rng, time_budget):
    start = time.monotonic()
    deadline = start + time_budget

    # initial candidates
    candidates = _initial_candidates_fixed_n(n, rng) # maybe adapt

    # direct evaluation using self convolution (fast)
    fft_len = _next_pow2(2*n - 1)

    best = None
    best_score = np.inf
    for cand in candidates:
        conv = _self_conv(cand, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best = cand.copy()

    a = best.copy()
    c = _self_conv(a, fft_len)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    phase1_end = start + 0.25 * time_budget  # 25% gradient

    move_counter = 0
    RECOMPUTE_EVERY = 1000

    while time.monotonic() < deadline:
        now = time.monotonic()
        if now < phase1_end:
            # gradient descent
            frac = (now - start) / (phase1_end - start)
            eta = 2.0 * (1 - frac) + 0.01 * frac   # from 2.0 to 0.01

            k_max = int(c.argmax())
            # subgradient g
            lo = max(0, k_max - (n-1))
            hi = min(n-1, k_max)
            rev = np.zeros_like(a)
            rev[lo:hi+1] = a[k_max - np.arange(lo, hi+1)]
            g = 2.0 * rev

            a_temp = a - eta * g
            a = proj_simplex(a_temp)
            # recompute full conv
            c = _self_conv(a, fft_len)
            cur_score = 2.0 * n * c.max()
        else:
            # stochastic moves
            # compute contributions to current peak
            k_max = int(c.argmax())
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            # choose donor (index with largest contribution)
            if contrib.max() > 0:
                donor = int(np.argmax(contrib))
            else:
                donor = int(rng.integers(0, n))

            if a[donor] < 1e-12:
                continue

            # choose recipient among low mass
            low_thresh = np.percentile(a, 10)
            low_indices = np.where(a <= low_thresh)[0]
            if low_indices.size == 0:
                recipient = (donor + rng.integers(1, n)) % n
            else:
                recipient = int(rng.choice(low_indices))
                if recipient == donor:
                    # pick alternative
                    recipient = (donor + rng.integers(1, n)) % n

            # delta up to full donor
            delta = rng.random() * a[donor]  # uniform [0, a[donor]]
            if delta <= 0:
                continue

            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # temperature schedule
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = 0.05 * (1.0 - t_frac) + 1e-6 * t_frac
            temperature = max(temperature, 1e-12)

            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len)

"""
Propose a step‑function that improves the best known bound (≈1.5132 → ≈1.5025).

Algorithm
---------
1. Choose a modest length n (≈400–1024) – smaller n allows many cheap
   evaluations while the objective 2·n·max(conv) is independent of n
   (the constant C = max(conv)·n does not change with scaling).

2. Initialise a population of diverse candidates (uniform, linear,
   triangular, Dirichlet draws, beta‑shaped, sinusoidal, etc.) and keep
   the best one according to the exact objective 2·n·max(conv).

3. **Phase 1 – deterministic mirror descent** (≈25 % of the local budget)  
   Apply an exponentiated‑gradient (“entropy mirror”) update for the
   sub‑gradient of the current peak of the self‑convolution:
   `a ← a·exp(‑η·g)` with a linear schedule for η, followed by normalisation.

4. **Phase 2 – aggressive stochastic refinement** (≈75 % of the budget)  
   *Donor* = index with largest contribution `a[i]·a[k_max‑i]` to the
   current peak `k_max`.  
   *Recipient* = an index among the ≈10 % smallest heights.  
   Transfer a random fraction `δ∈[0,a[donor]]` from donor to recipient.
   The convolution is updated incrementally in O(n) using the exact
   formula, avoiding an FFT at every move.  
   Acceptance: Metropolis rule with a slowly cooling temperature.
   Every 1000 accepted moves the full convolution is recomputed to
   eliminate drift; a global log‑normal jitter is applied with 7 % probability.

5. **Soft restart** if no improvement for 45 s – re‑normalise the best
   solution with moderate multiplicative log‑normal noise.

6. The procedure is run for several values of n, each receiving a
   fraction of the global time‑budget.  The best sequence found overall
   is returned.

All helper functions are top‑level, no closures are used, and the code
respects the 5 s safety margin required by the harness.
"""

import time
import numpy as np
import math

# Helper utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum x = 1}.
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new a and c.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # linear part – shift of donor (subtraction) and recipient (addition)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic correction
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates_fixed_n(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a small diversified portfolio of length‑n candidates."""
    candidates = []

    # 1) uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) linear decreasing (down‑hill)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    candidates.append(v)

    # 3) triangular (peak in the middle)
    half = n // 2
    v = np.concatenate([np.linspace(1, 2, half, endpoint=False),
                        np.linspace(2, 1, n - half)])
    v /= v.sum()
    candidates.append(v)

    # 4) half‑exponential (right‑hand decay)
    for r in [0.99, 0.975, 0.95, 0.9]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # 5) sinusoidal (raised cosine) with random power
    for p in [1.0, 2.0, 3.0]:
        theta = np.pi * np.arange(n, dtype=np.float64) / (n - 1)
        v = (1 - np.cos(theta)) ** p
        v /= v.sum()
        candidates.append(v)

    # 6) beta‑shaped (Dirichlet with moderate concentration)
    for _ in range(5):
        alpha = rng.uniform(0.2, 1.0, size=n)
        v = rng.dirichlet(alpha)
        candidates.append(v.astype(np.float64))

    return candidates

def _optimise_fixed_n(n: int, rng: np.random.Generator,
                     time_budget: float) -> tuple[np.ndarray, float]:
    """
    Perform the two‑phase optimisation for a specific length ``n``.
    Returns the best sequence found and its exact objective value.
    """
    start = time.monotonic()
    deadline = start + time_budget

    # ---- initialise --------------------------------------------------
    candidates = _initial_candidates_fixed_n(n, rng)
    fft_len = _next_pow2(2 * n - 1)

    best_seq = None
    best_score = np.inf
    for cand in candidates:
        conv = _self_conv(cand, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best_seq = cand.copy()

    a = best_seq.copy()
    c = _self_conv(a, fft_len)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    # ---- schedule ----------------------------------------------------
    phase1_end = start + 0.25 * time_budget          # 25 % gradient
    move_counter = 0
    RECOMPUTE_EVERY = 1000

    # sub‑gradient step‑size schedule (mirrored‑entropy)
    eta_start = 4.0
    eta_end = 0.01

    # simulated‑annealing temperature schedule
    temp_start = 0.07
    temp_end = 1e-6

    # ---- main loop ---------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()

        # ------------------- Phase 1 (mirror descent) -----------------
        if now < phase1_end:
            frac = (now - start) / (phase1_end - start)
            eta = eta_start * (1.0 - frac) + eta_end * frac

            k_max = int(c.argmax())

            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            rev = np.zeros_like(a)
            rev[lo:hi + 1] = a[k_max - np.arange(lo, hi + 1)]

            g = 2.0 * rev
            factor = np.exp(-eta * g)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

            c = _self_conv(a, fft_len)
            cur_score = 2.0 * n * c.max()

        # ------------------- Phase 2 (stochastic moves) -------------
        else:
            # temperature (linear cooling)
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            k_max = int(c.argmax())

            # contributions of each index to the current peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]

            # donor = index with *largest* contribution to the peak
            if contrib.max() > 0.0:
                donor = int(np.argmax(contrib))
            else:
                donor = int(rng.integers(0, n))

            if a[donor] < 1e-14:
                continue

            # recipient = an index among the 10 % smallest heights
            low_thresh = np.percentile(a, 10)
            low_idxs = np.where(a <= low_thresh)[0]
            if low_idxs.size == 0:
                recipient = (donor + rng.integers(1, n)) % n
            else:
                recipient = int(rng.choice(low_idxs))
                if recipient == donor:
                    recipient = (donor + rng.integers(1, n)) % n

            # transfer a random fraction (up to the whole donor)
            delta = rng.random() * a[donor]
            if delta <= 0.0:
                continue

            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len)
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter to escape flat regions
            if rng.random() < 0.07:       # ~7 % of iterations
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len)
                cur_score = 2.0 * n * c.max()

        # ------------------- bookkeeping --------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n} new best = {best_score:.6f}")

        # ------------------- soft restart -------------------------------
        if now - last_improve > 45.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = _proj_simplex(a)
            c = _self_conv(a, fft_len)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n} restart – best score {best_score:.6f}")

    # final exact score (using the same formula as evaluate_sequence)
    final_score = 2.0 * n * np.max(_self_conv(best, fft_len))
    return best, final_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # safety margin for the harness (5 seconds)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # try a few different lengths – each gets a share of the remaining time
    length_options = [256, 384, 512, 768, 1024]
    best_global = None
    best_global_score = np.inf

    for n in length_options:
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break
        # allocate at most 200 s per length (or a fraction of what remains)
        allocation = min(remaining * 0.25, 200.0)
        seq, sc = _optimise_fixed_n(n, rng, allocation)
        if sc < best_global_score:
            best_global_score = sc
            best_global = seq.copy()
            elapsed = time.monotonic() - (deadline - budget_s)
            print(f"[{elapsed:6.2f}s] overall new best (n={n}) = {best_global_score:.6f}")

    # fallback – if everything failed return a uniform distribution of size 256
    if best_global is None:
        n = 256
        best_global = np.full(n, 1.0 / n, dtype=np.float64)

    # final clipping to respect the limits enforced by evaluate_sequence
    best_global = np.clip(best_global, 0.0, 1000.0)

    return [float(x) for x in best_global]
