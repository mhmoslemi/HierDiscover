"""search for a low‑maximum self‑convolution (minimise evaluate_sequence).

The optimiser proceeds as follows:

* Several families of initial distributions are tried – random Dirichlet,
  power‑law “U‑shaped’’ sine‑powers, Beta‑shapes, two‑block layouts etc.
* From each start a **combined local optimiser** refines the vector:
  – the first half of the iterations uses an exponentiated‑gradient step
    (exact sub‑gradient or smooth log‑sum‑exp gradient) on the
    probability simplex;
  – the second half runs a lightweight simulated‑annealing (SA) that
    performs random mass‑transfer moves and accepts worsening moves with
    a linearly decreasing temperature.
* The best candidate seen is kept globally and is periodically polished
  with a longer combined run.
* All heavy work (convolution, gradient) is performed with real FFTs for
  speed and the loop respects the supplied time budget.

The returned list consists of non‑negative floats ready for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. ``a``.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))          # safe, max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    grad_part = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Simple random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a random fraction of mass from i to j (accept only if improving)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a

# Score helper (fast, avoids evaluate_sequence overhead)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return the value 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()

def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponential variables."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Bell‑shaped family a_i ∝ sin^p(π·(i+0.5)/n),  p∈[0.5,4].
    """
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_sine_power_negative(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    U‑shaped family a_i ∝ sin^p(π·(i+0.5)/n) with p∈[-4,-0.5] (negative exponent).
    """
    p = rng.uniform(-4.0, -0.5)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    eps = 1e-6
    vals = np.clip(vals, eps, None)          # avoid division by zero
    a = np.power(vals, p)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_power_law(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Symmetric power‑law U‑shape:
    a_i ∝ (i+1)^{-p} + (n-i)^{-p},  p∈[0.5,2.0].
    """
    p = rng.uniform(0.5, 2.0)
    i = np.arange(n, dtype=np.float64)
    left = np.power(i + 1.0, -p)
    right = np.power(n - i, -p)
    a = left + right
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _init_beta(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Discrete Beta(α,α) shape on a regular grid, α∈[0.2,5].
    Small α gives U‑shape, large α gives bell‑shape.
    """
    alpha = rng.uniform(0.2, 5.0)
    x = (np.arange(n) + 0.5) / n
    vals = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    vals = np.clip(vals, 0.0, None)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _init_two_blocks(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Two uniform blocks placed at the two ends of the vector.
    The relative block size is sampled, as is the mass split between them.
    """
    block_frac = rng.uniform(0.05, 0.3)
    L = max(1, int(block_frac * n))
    if 2 * L > n:                     # ensure they do not overlap
        L = n // 2
    a = np.zeros(n, dtype=np.float64)
    left_mass = rng.uniform(0.3, 0.7)
    right_mass = 1.0 - left_mass
    a[:L] = left_mass / L
    a[-L:] = right_mass / L
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linear interpolation of an existing distribution to a different length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Combined local optimiser (exponentiated gradient + SA)
# ----------------------------------------------------------------------


def _local_optimise_combined(initial: np.ndarray,
                             rng: np.random.Generator,
                             deadline: float,
                             max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Refine ``initial`` (already on the simplex) by:
    – an exponentiated‑gradient phase (≈50 % of the iterations);
    – a light simulated‑annealing phase for the remaining iterations.
    Returns the best vector seen and its score.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    # exponentiated‑gradient phase
    beta = 0.5
    beta_mul = 1.07
    grad_iters = max_iters // 2
    for t in range(1, grad_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # choose gradient type
        if rng.random() < 0.55:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = min(1e5, beta * beta_mul)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # exponentiated gradient update
        lr = 0.7 / np.sqrt(t)
        raw_grad = np.clip(raw_grad, -20.0, 20.0)
        a = a * np.exp(-lr * raw_grad)
        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        # occasional random moves
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.02:
            a = _pairwise_move(a, rng, fft_len_conv)

    # simulated‑annealing phase
    # we continue from the last a (not from best_vec, to keep momentum)
    cur_score = 2.0 * n * _conv_self(a, fft_len_conv).max()
    remaining_iters = max_iters - grad_iters
    if remaining_iters > 0:
        # initial temperature – a small fraction of the current score
        T0 = max(1e-8, best_score * 0.02)
        for it in range(remaining_iters):
            if time.time() >= deadline:
                break
            # linear cooling schedule
            frac = it / remaining_iters
            T = T0 * (1.0 - frac)

            # propose a random mass‑transfer move
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)

            max_delta = min(a[i], 0.2 * a[i])
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            # compute new score
            conv2 = _conv_self(a2, fft_len_conv)
            new_score = 2.0 * n * conv2.max()
            dscore = new_score - cur_score
            if dscore < 0.0 or (T > 0.0 and rng.random() < np.exp(-dscore / T)):
                a = a2
                cur_score = new_score
                if new_score < best_score:
                    best_score = new_score
                    best_vec = a.copy()

    # final evaluation of the best candidate
    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # safety margin

    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Seed with a previously known construction, if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cand_score = _score(cand, _fft_len_for_len(2 * cand.size - 1))
                best_vec = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start:5.2f}s] Loaded previous best: {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths we shall explore (medium to large, but not huge)
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 3809, 4000,
               4500, 5000, 6000, 8000, 10000]

    # ------------------------------------------------------------------
    # Initialisation families and their relative probabilities
    # ------------------------------------------------------------------
    init_families = [
        _init_random_dirichlet,
        _init_sine_power,
        _init_sine_power_negative,
        _init_power_law,
        _init_beta,
        _init_two_blocks,
        _init_interpolate,      # requires the current best as a source
    ]
    # give a modest preference to the families that have proven useful
    init_weights = np.array([0.25, 0.20, 0.10, 0.10, 0.10, 0.10, 0.15])
    init_weights = init_weights / init_weights.sum()

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (uniform over the list)
        n = int(rng.choice(lengths))
        # pick an initialisation method
        which = rng.choice(len(init_families), p=init_weights)

        if which == 6:   # interpolation from the current global best
            if best_vec is not None:
                a0 = _init_interpolate(rng, best_vec, n)
            else:
                a0 = _init_random_dirichlet(rng, n)
        else:
            a0 = init_families[which](rng, n)

        # how many inner iterations? scale with n so we stay within the budget
        iters = int(2500 * (5000.0 / max(n, 500)))   # about 2500 for n≈5000
        a_opt, score_opt = _local_optimise_combined(a0, rng, deadline, max_iters=iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:5.2f}s] New global best: {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        # Polishing phase on the current global best (if time permits)
        # ------------------------------------------------------------------
        if best_vec is not None and time.time() < deadline - 1.0:
            # a longer combined optimisation (more iterations, slower beta growth)
            a_polished, score_polished = _local_optimise_combined(
                best_vec.copy(), rng, deadline, max_iters=4000)
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:5.2f}s] Polished best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – a simple uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (the harness does this as well)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
