Now implement _project_simplex:

```
def _project_simplex(v):
    # from Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a low‑max‑self‑convolution sequence.

Algorithm Overview
------------------
* Works on the probability simplex (∑a = 1, a_i ≥ 0).  Scaling does not affect
  the objective ``evaluate_sequence``, so we normalise to a simplex for
  convenience.
* Multi‑scale search:
    – Start with a moderate length (≈2000) and optimise.
    – Interpolate the current best solution to a larger length and continue
      optimisation.  Lengths grow up to ~12000 (adjustable by the remaining
      time budget).  Larger sequences tend to give a lower bound.
* Local optimiser (mirrored gradient descent with Nesterov‑style momentum):
    – At each step either the exact sub‑gradient of the worst convolution entry
      or a smooth log‑sum‑exp gradient (β grows exponentially) is used.
    – An additive update is performed and the vector is projected back onto the
      simplex.
    – Momentum (β₁≈0.9) stabilises the trajectory.
    – Occasional random block perturbations and cheap pairwise moves help
      escape plateaus.
* Several families of initialisations are tried (Dirichlet, sine‑power with a
  range of exponents, interpolation from the current best).  The best vector
  discovered is repeatedly polished until the time budget expires.
* If a previous construction is available via the global variable
  ``height_sequence_1`` it is used as an initial seed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT / convolution helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power of two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. ``a``.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))          # safe, max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    grad_part = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Simple random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a small random fraction of mass from i to j if it improves."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the probability simplex
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, Σx = 1} (Duchi et al., 2008)."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numeric issue – fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast score helper (identical to evaluate_sequence up to scaling)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponential variables."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power_fixed(n: int, p: float) -> np.ndarray:
    """
    Deterministic sine‑power shape: a_i ∝ sin(π·(i+0.5)/n)^p .
    p > 0 controls concentration (larger p → more central mass).
    """
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linearly interpolate an existing distribution to a different length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Local optimisation (mirror‑descent with momentum)
# ----------------------------------------------------------------------


def _local_optimize(initial: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float,
                    max_iters: int = 1500) -> tuple[np.ndarray, float]:
    """
    Refine ``initial`` (already simplex‑normalised) using an additive
    projected gradient descent with momentum.  Both exact sub‑gradients and a
    smooth log‑sum‑exp gradient are employed.
    Returns the best vector encountered and its score.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    # momentum parameters (Nesterov‑style)
    m = np.zeros_like(a)
    beta1 = 0.9

    # schedule for the smooth‑max temperature
    beta = 0.5
    beta_mul = 1.06

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ------------------------------------------------------------------
        # Choose a gradient (sub‑gradient vs. smooth‑max)
        # ------------------------------------------------------------------
        if rng.random() < 0.55:
            raw_grad = _subgradient_step(a, conv)
        else:
            beta = min(1e6, beta * beta_mul)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # momentum update (exponential moving average)
        m = beta1 * m + (1.0 - beta1) * raw_grad

        # learning rate decreasing ~ 1/√t (t starts at 1)
        lr = 0.8 / np.sqrt(t)

        # additive update then projection onto the simplex
        a_candidate = a - lr * m
        a = _project_simplex(a_candidate)

        # ------------------------------------------------------------------
        # Occasional exploratory moves
        # ------------------------------------------------------------------
        if rng.random() < 0.025:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.025:
            a = _pairwise_move(a, rng, fft_len_conv)

    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.  The search proceeds
    from small to larger sequences, using a momentum‑based mirror descent,
    occasional random perturbations and a handful of deterministic initialisers.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # Slight safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.35)

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Seed with a previously known construction if it exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / s
                cand_score = _score(cand, _fft_len_for_len(2 * cand.size - 1))
                best_vec = cand.copy()
                best_score = cand_score
                elapsed = time.time() - start
                print(f"[{elapsed:6.2f}s] Loaded previous best: {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Length schedule (grows geometrically, limited by remaining time)
    # ------------------------------------------------------------------
    length_schedule = [2048, 4096, 6144, 8192, 10240, 12288]  # up to ≈12k
    # We will adapt the maximum length later according to the remaining time.

    # ------------------------------------------------------------------
    # Main multi‑scale optimisation loop
    # ------------------------------------------------------------------
    for n in length_schedule:
        now = time.time()
        if now >= deadline:
            break

        # Determine iteration budget: roughly keep total FFT work constant.
        # 2000 iterations for n≈5000 -> scale inversely with n.
        iters = max(200, int(2000 * (5000.0 / max(n, 500.0))))

        # ---- initial candidate: interpolate from current best if available ----
        if best_vec is not None:
            a0 = _init_interpolate(rng, best_vec, n)
        else:
            a0 = _init_random_dirichlet(rng, n)

        # ---- try a few deterministic sine‑power shapes (quick local runs) ----
        sine_exponents = [0.8, 1.2, 2.0, 3.0]
        for p in sine_exponents:
            a_sine = _init_sine_power_fixed(n, p)
            a_opt, score_opt = _local_optimize(a_sine, rng, deadline,
                                               max_iters=int(0.4 * iters))
            if score_opt < best_score:
                best_score = score_opt
                best_vec = a_opt.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:6.2f}s] New global best from sine p={p}: "
                      f"{best_score:.6f} (n={n})")
                if time.time() >= deadline:
                    break

        # ---- main optimisation from the (possibly interpolated) start ----
        a_opt, score_opt = _local_optimize(a0, rng, deadline, max_iters=iters)
        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:6.2f}s] New global best (n={n}): {best_score:.6f}")

        # ---- occasional polishing of the current global best ----
        if best_vec is not None and time.time() < deadline - 2.0:
            a_polished, score_polished = _local_optimize(
                best_vec.copy(), rng, deadline, max_iters=int(1.5 * iters))
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:6.2f}s] Polished best (n={best_vec.size}): "
                      f"{best_score:.6f}")

    # ------------------------------------------------------------------
    # Final polishing (as much time as remains)
    # ------------------------------------------------------------------
    if best_vec is not None and time.time() < deadline - 0.5:
        remaining_iters = 5000
        a_final, score_final = _local_optimize(best_vec.copy(),
                                                rng,
                                                deadline,
                                                max_iters=remaining_iters)
        if score_final < best_score:
            best_score = score_final
            best_vec = a_final.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:6.2f}s] Final polishing gave: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 3000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Safety clipping (evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Return plain Python floats for the harness
    return [float(x) for x in best_vec]
