```
def project_onto_simplex(v):
    # Implementation from Duchi et al. 2008
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def max_conv_subgradient(a, conv):
    max_val = conv.max()
    idxs = np.where(conv >= max_val * 0.9999)[0]  # tolerance

        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        i_range = np.arange(lo, hi+1, dtype=np.int64)
        grad[i_range] += 2.0 * a[k - i_range]
```

def _subgradient_max_conv(a, conv, tol=0.999):
    max_val = conv.max()
    active = np.where(conv >= max_val * tol)[0]
    n = a.size
    grad = np.zeros_like(a)
    if active.size == 0:
        active = np.arange(conv.size)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx_range = np.arange(lo, hi+1, dtype=np.int64)
        grad[idx_range] += 2.0 * a[k - idx_range]
    grad = grad / max(active.size, 1)  # average
    return grad

def _project_onto_simplex(v):
    # Returns projection of v onto simplex {x>=0, sum x = 1}
    if v.ndim != 1:
        raise ValueError("v must be 1-D")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / (np.arange(1, len(v)+1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function (list of heights) that minimises
evaluate_sequence().  The algorithm works in two phases:

1)  A Frank‑Wolfe/conditional‑gradient style loop:
    * Start from a simple analytical shape (√‑difference) or a random
      Dirichlet draw, normalised to sum = 1.
    * Compute M = max(f∗f).
    * Solve the linear program   maximise 1·g   s.t.   f∗g ≤ M,   g ≥ 0 .
      The LP is built with a sparse Toeplitz matrix (2·n‑1 × n).
      The optimiser “highs” returns the extreme point g that puts mass on
      indices where f is small.
    * Normalise g to sum = 1 and mix it with the current f
      (convex combination with step t).  If the mixture reduces the
      objective 2·n·max(f∗f) the mixture is accepted; otherwise we try a
      smaller t or keep the best of {f, g}.
    * Repeat while time allows (typically a few dozen LP solves).

2)  A short projected sub‑gradient refinement:
    * The objective 2·n·max(f∗f) is convex in f (max of linear forms).
    * The sub‑gradient of max(f∗f) is obtained by averaging the gradients
      of the active convolution entries.
    * Perform a simple decreasing‑step gradient descent with Euclidean
      projection onto the probability simplex.
    * Occasional random mass‑transfer moves help escape tiny plateaus.

Both phases keep the globally best vector seen so far.  At termination the
vector is clipped to the limits required by ``evaluate_sequence`` and
returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize


def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a self‑convolution of size n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Exact linear convolution a∗a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a∗a) for a probability vector a (sum = 1)."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


def _normalise(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _build_A_ub(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (size (2·n‑1)×n) such that A @ g = f∗g.
    Each diagonal offset d (0 ≤ d < n) contains the constant value f[d].
    """
    n = f.size
    # one full‑length diagonal per offset
    diag_data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int64)
    A = sp.diags(diag_data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray, tol: float = 0.9995) -> np.ndarray:
    """
    Sub‑gradient of  max(a∗a)  at point a.
    Active indices are those where conv is within `tol` of the maximum.
    The gradient of (a∗a)_k w.r.t. a_i is 2·a_{k‑i}.
    """
    max_val = conv.max()
    active = np.where(conv >= max_val * tol)[0]
    if active.size == 0:                     # fallback (should not happen)
        active = np.arange(conv.size)

    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    grad /= max(active.size, 1)               # average over active set
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x≥0, Σx=1}.
    Implementation of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
