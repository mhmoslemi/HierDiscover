def softmax_weights(conv, beta):
    maxc = conv.max()
    exponent = beta * (conv - maxc)
    # Use numpy clip to avoid overflow
    w_raw = np.exp(exponent)
    sum_raw = w_raw.sum()
    if sum_raw == 0:
        # fallback uniform
        w = np.full_like(w_raw, 1.0 / w_raw.size)
    else:
        w = w_raw / sum_raw
    return w
```

def project_simplex(v):
    # ensure nonnegative after projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(1, len(v)+1)) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Propose a non‑negative step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. Build a diverse pool of initial candidates (analytic √‑difference,
   arcsine shape, power‑difference, exponential, random Dirichlet,
   previous best if available) and normalise them to sum = 1.
2. Pick the best candidate from the pool using a fast FFT‑based score.
3. Refine the current best by **mirror descent** on a smooth
   soft‑max surrogate of the L∞‑objective:
   * At each iteration compute the self‑convolution `a*a` with FFT.
   * Form soft‑max weights `w` on the convolution values with a
     temperature β that grows exponentially.
   * The gradient of the surrogate is `∇f = a + (w * a_rev)`,
     where `a_rev` is `a` reversed and `*` denotes linear convolution.
   * Perform an exponentiated‑gradient (multiplicative) update
     `a ← a·exp(‑η∇f)` and re‑normalise onto the probability simplex.
   * A momentum term stabilises the updates.  Occasionally apply
     random block perturbations and pairwise mass transfers to escape
     plateaus.
4. When only a short time budget (≈20 s) remains, switch to a **simulated
   annealing** phase that makes random mass‑transfer moves and accepts
   them with a Metropolis criterion.  Occasionally a tiny smoothing
   operation is applied.
5. Return the best sequence seen, clipped to the limits of
   `evaluate_sequence`.

The routine respects the overall 1000 s budget, prints progress stamps,
and returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: next power‑of‑two ≥ x
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# FFT‑based linear convolution (exact up to floating‑point noise)
# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Return np.convolve(a, b) using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)
    return conv[: a.size + b.size - 1]


# ----------------------------------------------------------------------
# Self‑convolution of a vector (used for the objective)
# ----------------------------------------------------------------------
def _self_convolve(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Convolution a*a of length 2·len(a)‑1 via FFT."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Fast L∞‑objective for a probability vector (sum = 1)
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, n: int, fft_len_self: int) -> float:
    """Return 2·n·max(a*a) (exact up to round‑off)."""
    conv = _self_convolve(a, fft_len_self)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Stable soft‑max weights for a vector of convolution values
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max on ``conv`` with temperature ``beta`` (beta≥0)."""
    maxc = conv.max()
    # exponent ≤ 0, no overflow risk
    w_raw = np.exp(beta * (conv - maxc))
    s = w_raw.sum()
    if s == 0.0:
        return np.full_like(w_raw, 1.0 / w_raw.size)
    return w_raw / s


# ----------------------------------------------------------------------

# ----------------------------------------------------------------------

    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

        new_a = cur_sa.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _score_of(new_a, n, fft_len_self)

        # Metropolis acceptance
        if new_score < cur_sa_score or rng.random() < np.exp(-(new_score - cur_sa_score) / max(T, 1e-12)):
            cur_sa = new_a
            cur_sa_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = new_a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        iteration += 1

        # occasional gentle smoothing (helps flatten local peaks)
        if iteration % 4000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            cur_sa = smoothed
            cur_sa_score = _score_of(cur_sa, n, fft_len_self)

    # --------------------------------------------------------------
    # Return the best sequence (respect clipping constraints)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
