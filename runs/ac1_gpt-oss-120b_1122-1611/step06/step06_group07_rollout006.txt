   a large one (≈2·10⁴) while the learning rate η is decayed.
   The gradient of the surrogate is obtained by a single
   correlation (FFT) and the iterate is projected onto the
   simplex (non‑negative, sum = √(2 n)).

3. **Exploration & restarts** – occasional block‑wise
   multiplicative perturbations, pairwise mass transfers and
   soft restarts (jittering the current best) keep the algorithm
   from stagnating.  The best vector seen during the whole run is
   returned (clipped to the limits imposed by `evaluate_sequence`).

The routine respects a 1000 s wall‑clock budget, prints progress
messages and finally returns a list of floats ready for
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of size 2·n‑1."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a * a (length 2·n‑1) via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max with temperature β (weights sum to 1)."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σ w_k (a ∗ a)_k  with respect to a.
    Implements 2·a⋆w (correlation) using a single FFT.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    # extract the n‑length slice centred at offset n‑1
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _score(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value (no normalisation needed because a is already scaled)."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Candidate generators (all return *unnormalised* vectors)
# ----------------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray:
    """a_i = (i+1)^r – i^r   (decreasing for 0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    return a


def _gen_shape_p(p: float, n: int) -> np.ndarray:
    """a_i = (i+1)^(-p) – (i+2)^(-p)   (p>0)."""
    k = np.arange(1, n + 2, dtype=np.float64)          # 1 … n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]                                 # length n
    return a


def _gen_beta_shape(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Discretised beta‑type density:
        a_i ∝ (i+1)^α · (n‑i)^β   (α,β ≥ 0)
    """
    i = np.arange(n, dtype=np.float64) + 1.0
    log_a = alpha * np.log(i) + beta * np.log(n - i + 1e-12)
    # shift for numerical stability
    log_a -= log_a.max()
    a = np.exp(log_a)
    return a


def _gen_exponential(lam: float, n: int) -> np.ndarray:
    """Geometric decay a_i = lam^i   (0<lam<1)."""
    i = np.arange(n, dtype=np.float64)
    a = lam ** i
    return a


def _gen_gaussian(sigma: float, n: int) -> np.ndarray:
    """Discrete Gaussian centred at the middle of the array."""
    mu = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    return a


# ----------------------------------------------------------------------
# Normalisation helper (sum → √(2 n))
# ----------------------------------------------------------------------
def _normalise(a: np.ndarray, n: int) -> np.ndarray:
    """Scale a to have ℓ₁‑norm √(2 n)."""
    target = np.sqrt(2.0 * n)
    s = a.sum()
    if s == 0.0:
        return np.full_like(a, target / n)
    return a * (target / s)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that
    approximately minimises `evaluate_sequence`.  The implementation
    respects the given time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Problem size – we will try a few nearby lengths.
    # ------------------------------------------------------------------
    base_n = int(kwargs.get("n", 3800))
    if base_n < 2:
        base_n = 2
    # a small set of lengths (the search will stop early if time runs out)
    n_candidates = [base_n, int(base_n * 0.9), int(base_n * 1.1)]
    n_candidates = [n for n in n_candidates if n >= 2]

    best_a = None
    best_score = np.inf
    best_n = None

    # ------------------------------------------------------------------
    # 1) Fast sweep of analytic families (time‑boxed)
    # ------------------------------------------------------------------
    for n in n_candidates:
        fft_len = _fft_len_for_n(n)

        def _evaluate_raw(raw: np.ndarray) -> float:
            a = _normalise(raw, n)
            # tiny safeguard – negative numbers are removed by the optimiser later
            a = np.clip(a, 0.0, None)
            return _score(a, fft_len)

        # ----- power‑difference (r‑family) -----
        for r in np.linspace(0.5, 0.95, 81):          # step≈0.0055
            a_raw = _gen_shape_r(r, n)
            sc = _evaluate_raw(a_raw)
            if sc < best_score:
                best_score, best_a, best_n = sc, _normalise(a_raw, n), n
                print(f"[init] r={r:.3f} → score={sc:.6f}")

        # ----- inverse‑difference (p‑family) -----
        for p in np.linspace(0.1, 2.0, 97):
            a_raw = _gen_shape_p(p, n)
            sc = _evaluate_raw(a_raw)
            if sc < best_score:
                best_score, best_a, best_n = sc, _normalise(a_raw, n), n
                print(f"[init] p={p:.3f} → score={sc:.6f}")

        # ----- beta‑type shapes (α,β grid) -----
        alphas = np.linspace(0.2, 4.0, 15)
        betas = np.linspace(0.2, 4.0, 15)
        for alpha in alphas:
            for beta in betas:
                a_raw = _gen_beta_shape(alpha, beta, n)
                sc = _evaluate_raw(a_raw)
                if sc < best_score:
                    best_score, best_a, best_n = sc, _normalise(a_raw, n), n
                    print(f"[init] beta(α={alpha:.2f},β={beta:.2f}) → {sc:.6f}")

        # ----- geometric decay -----
        for lam in np.linspace(0.90, 0.9995, 20):
            a_raw = _gen_exponential(lam, n)
            sc = _evaluate_raw(a_raw)
            if sc < best_score:
                best_score, best_a, best_n = sc, _normalise(a_raw, n), n
                print(f"[init] exp(λ={lam:.4f}) → {sc:.6f}")

        # ----- Gaussian shapes -----
        sigma_vals = np.linspace(n / 12.0, n / 3.0, 12)
        for sigma in sigma_vals:
            a_raw = _gen_gaussian(sigma, n)
            sc = _evaluate_raw(a_raw)
            if sc < best_score:
                best_score, best_a, best_n = sc, _normalise(a_raw, n), n
                print(f"[init] gauss(σ={sigma:.1f}) → {sc:.6f}")

        # ----- reversed r‑family (mass concentrated at the right side) -----
        for r in np.linspace(0.5, 0.95, 21):
            a_raw = _gen_shape_r(r, n)[::-1]   # mirror
            sc = _evaluate_raw(a_raw)
            if sc < best_score:
                best_score, best_a, best_n = sc, _normalise(a_raw, n), n
                print(f"[init] rev‑r={r:.3f} → {sc:.6f}")

        # ----- use any previously discovered sequence -----
        if "height_sequence_1" in globals():
            try:
                cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
                if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                    # adapt length if necessary (pad with zeros or truncate)
                    if cand.size != n:
                        if cand.size < n:
                            cand = np.concatenate([cand, np.zeros(n - cand.size)])
                        else:
                            cand = cand[:n]
                    sc = _evaluate_raw(cand)
                    if sc < best_score:
                        best_score, best_a, best_n = sc, _normalise(cand, n), n
                        print(f"[init] height_sequence_1 → {sc:.6f}")
            except Exception:
                pass

        # early exit if time is almost exhausted
        if time.monotonic() > deadline - 200.0:
            break

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_a is None:
        n = base_n
        best_n = n
        best_a = np.full(n, np.sqrt(2.0 * n) / n)
        best_score = _score(best_a, _fft_len_for_n(n))
        print("[fallback] uniform distribution")

    # ------------------------------------------------------------------
    # 2) Mirror‑descent (exponentiated‑gradient) on the smooth surrogate
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_n = best_n
    cur_fft_len = _fft_len_for_n(cur_n)

    # schedule parameters (tuned experimentally)
    eta_start, eta_end = 3.0, 0.001        # learning‑rate decay
    beta_start, beta_end = 0.4, 25000.0   # temperature rise

    # exploration parameters
    block_prob = 0.04
    pair_prob = 0.07
    restart_gap = 120.0   # seconds without improvement → jittered restart

    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time

        # linear schedules
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # ------- compute exact score (for bookkeeping) -------
        conv = _conv_self(cur_a, cur_fft_len)
        cur_max = conv.max()
        cur_score = float(2.0 * cur_n * cur_max)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ------- smooth max (soft‑max) -------
        w = _softmax(conv, beta)

        # ------- gradient of the surrogate -------
        grad = _gradient_via_correlation(cur_a, w)

        # ------- exponentiated‑gradient update -------
        # clip the exponent to avoid overflow/underflow
        step = -eta * grad
        step = np.clip(step, -50.0, 50.0)
        factor = np.exp(step)
        cur_a = cur_a * factor
        cur_a = np.clip(cur_a, 0.0, None)

        # renormalise to the proper ℓ₁‑norm
        cur_a = cur_a / cur_a.sum() * np.sqrt(2.0 * cur_n)

        # ------- occasional random perturbations -------
        # block multiplicative perturbation
        if rng.random() < block_prob:
            width = rng.integers(1, max(2, cur_n // 25))
            start = rng.integers(0, cur_n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.22))
            cur_a[start:start + width] *= mult
            cur_a = cur_a / cur_a.sum() * np.sqrt(2.0 * cur_n)

        # pairwise mass transfer
        if rng.random() < pair_prob:
            i, j = rng.integers(0, cur_n, size=2)
            if i != j and cur_a[i] > 1e-15:
                delta = rng.random() * cur_a[i] * 0.35
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = cur_a / cur_a.sum() * np.sqrt(2.0 * cur_n)

        # ------- soft restart on stagnation -------
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.12, size=cur_n))
            cur_a = best_a * jitter
            cur_a = cur_a / cur_a.sum() * np.sqrt(2.0 * cur_n)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (jitter) – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best vector found (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
```
