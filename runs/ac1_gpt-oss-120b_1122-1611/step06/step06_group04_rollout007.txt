"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
* Initialise a candidate on the probability simplex (∑a_i = 1).  
  The initialisation is the classic “√‑difference’’ shape,
  optionally perturbed by a random shift or a power‑law tail.
* Phase 1 – cheap global search: randomly generate a few families
  (shifted √‑differences, power‑law decays and geometric decays),
  normalise them to the simplex and keep the best one seen.
* Phase 2 – deterministic refinement: minimise the convex objective
  f(a)=max_k (a∗a)_k by projected sub‑gradient descent on the simplex.
  The sub‑gradient for the active convolution indices is obtained by a
  correlation via FFT.  A simple 1/√t stepsize together with occasional
  log‑normal jitter helps to escape shallow plateaux.
* Throughout the run keep the best vector (lowest 2·n·max_conv) and
  return it (clipped to the limits of ``evaluate_sequence``).

The routine respects the 1000 s budget and never fails to return a
candidate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length for a linear convolution of size n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for use with FFT."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (real‑valued inputs).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return linear convolution a*a (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the (weighted) convolution.
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w  (correlation).
    ``a`` is length n, ``w`` length 2·n‑1.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # Length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reversed a for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Euclidean projection onto the probability simplex.
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto {x≥0, Σx = 1}."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical clean‑up – enforce exact sum 1
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Candidate generators for Phase 1 (global random search).
# ----------------------------------------------------------------------
def _gen_sqrt_diff(n: int, shift: float) -> np.ndarray:
    """a_i = sqrt(i+1+shift) - sqrt(i+shift) , normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0 + shift) - np.sqrt(i + shift)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        return a
    return a / s


def _gen_power_law(n: int, p: float, shift: float) -> np.ndarray:
    """a_i = (i+shift+1)^{-p} , normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = (i + shift + 1.0) ** (-p)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        return a
    return a / s


def _gen_geo(n: int, lam: float) -> np.ndarray:
    """a_i = lam**i , normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = lam ** i
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        return a
    return a / s


# ----------------------------------------------------------------------
# Main optimisation routine.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Length of the step sequence.  Default is 3800 (a size that
        performed well in earlier experiments).

    Returns
    -------
    list[float]
        The best sequence found (values clipped to the limits imposed by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    # --------------------------------------------------------------
    # 2) Helper to evaluate the (scaled) objective quickly.
    #    Since the sequence will always be normalised to sum = 1,
    #    the true score is 2·n·max(conv).
    # --------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    def _score(a_vec: np.ndarray) -> float:
        conv = _conv_self(a_vec, fft_len)
        return 2.0 * n * float(conv.max())

    # --------------------------------------------------------------
    # 3) Initial candidate – classic √‑difference shape.
    # --------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    a0 = np.sqrt(i + 1.0) - np.sqrt(i)
    a0 = np.clip(a0, 0.0, None)
    a0 /= a0.sum()
    best = a0.copy()
    best_score = _score(best)

    # Use a previously discovered construction if it exists.
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    cand = cand / cand.sum()
                    cand_score = _score(cand)
                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score
        except Exception:
            pass

    # --------------------------------------------------------------
    # 4) Phase 1 – cheap random family search.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a small safety margin
    phase1_end = start_time + 0.25 * (deadline - start_time)  # ~25 % of remaining time

    # Random parameters for each family
    while time.monotonic() < phase1_end:
        shape_type = rng.choice(["sqrt", "power", "geo"])
        if shape_type == "sqrt":
            shift = rng.uniform(0.0, float(n) * 0.5)  # shift up to half the length
            a = _gen_sqrt_diff(n, shift)
        elif shape_type == "power":
            p = rng.uniform(0.30, 0.70)               # exponent around 0.5
            shift = rng.uniform(0.0, float(n) * 0.5)
            a = _gen_power_law(n, p, shift)
        else:  # geometric
            lam = rng.uniform(0.96, 0.9995)           # close to 1
            a = _gen_geo(n, lam)

        # Ensure feasibility
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        s = _score(a)
        if s < best_score:
            best_score = s
            best = a.copy()
            # occasional console feedback
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Phase 1 new best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Phase 2 – projected sub‑gradient descent on max‑convolution.
    # --------------------------------------------------------------
    a = best.copy()
    m = np.zeros_like(a)      # momentum terms – we keep them but may not use them
    v = np.zeros_like(a)
    t_iter = 0
    # hyper‑parameters
    step0 = 0.25                # initial stepsize multiplier
    jitter_prob = 0.03
    jitter_sigma = 0.12
    swap_prob = 0.07            # occasional pairwise transfer
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # ----------------------------------------------------------
        # a) evaluate current objective and keep global best
        # ----------------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # b) build sub‑gradient for the active set (max indices)
        # ----------------------------------------------------------
        # tolerance – treat values within 1e‑12 of the max as active
        tol = max(1e-12, cur_max * 1e-12)
        mask = conv >= (cur_max - tol)
        w = mask.astype(np.float64)
        if w.sum() == 0:
            w[conv.argmax()] = 1.0
        w /= w.sum()                     # distribution over active constraints
        grad = _gradient_correlation(a, w)   # shape (n,)

        # ----------------------------------------------------------
        # c) sub‑gradient step (simple 1/√t schedule)
        # ----------------------------------------------------------
        t_iter += 1
        step = step0 / np.sqrt(t_iter)
        a_candidate = a - step * grad
        a_candidate = _proj_simplex(a_candidate)

        # ----------------------------------------------------------
        # d) occasional stochastic jitter – helps escape plateaux
        # ----------------------------------------------------------
        if rng.random() < jitter_prob:
            factor = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a_candidate = a_candidate * factor
            a_candidate = _proj_simplex(a_candidate)

        # ----------------------------------------------------------
        # e) occasional pairwise mass transfer (swap)
        # ----------------------------------------------------------
        if rng.random() < swap_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and a_candidate[i] > 1e-15:
                delta = rng.random() * a_candidate[i] * 0.5
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = _proj_simplex(a_candidate)

        # ----------------------------------------------------------
        # f) accept the candidate if it does not worsen the objective;
        #    otherwise keep the old point (simple greedy acceptance).
        # ----------------------------------------------------------
        conv_cand = _conv_self(a_candidate, fft_len)
        cand_score = 2.0 * n * float(conv_cand.max())
        if cand_score <= cur_score:
            a = a_candidate
        else:
            # occasional restart if stuck for a while
            if now - last_improve > 120.0:   # 2 min without improvement
                a = best.copy()
                jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
                a = _proj_simplex(a * jitter)
                t_iter = 0
                last_improve = now
                print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (apply the clipping imposed by
    #    ``evaluate_sequence``).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
