"""
Improved search for a low‑maximum self‑convolution.

The algorithm
---------------
* Enumerates a few analytically‑motivated families of shapes on the simplex
  (symmetric Beta, edge‑power, sine‑power) for several lengths.
* The shape parameters are first scanned on a coarse grid, then locally
  refined around the best parameters found for each length.
* All candidates are evaluated with a fast FFT‑based score
  (exactly the value returned by ``evaluate_sequence`` after clipping).
* The best few candidates are *polished* with a momentum‑accelerated
  exponentiated‑gradient optimiser that alternates an exact sub‑gradient
  of the worst convolution entry with a smooth soft‑max gradient
  (temperature β grows exponentially).  Random block perturbations and
  cheap pairwise mass‑transfer moves help escape plateaus.
* After polishing, additional random Dirichlet restarts are performed
  until the time budget expires.
* The final distribution is returned as a plain Python list; it satisfies
  the non‑negative requirement and the score is (hopefully) below the
  target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self‑convolution a*a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. ``a``.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))           # safe: max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    # extract the n entries that correspond to derivative wrt a
    grad_part = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Simple random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    out = out / out_sum if out_sum > 0 else np.full_like(out, 1.0 / n)
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a random fraction of mass from i to j; keep the move if it improves."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a

def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()

def _shape_beta_distribution(n: int, a_param: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density on the discrete grid.
    α = a_param (>0).  a_i ∝ x^{α-1} (1‑x)^{α-1},  x = (i+0.5)/n .
    """
    idx = np.arange(n, dtype=np.float64) + 0.5
    x = idx / n
    with np.errstate(divide='ignore', invalid='ignore'):
        log_w = (a_param - 1.0) * (np.log(x) + np.log(1.0 - x))
    w = np.exp(log_w)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= s
    return w.astype(np.float64)


def _shape_power_edge(n: int, p: float) -> np.ndarray:
    """
    Edge‑heavy shape: a_i ∝ (i+0.5)^{-p} + (n‑0.5‑i)^{-p}.
    """
    idx = np.arange(n, dtype=np.float64) + 0.5
    left = idx
    right = n + 0.5 - idx
    w = (left ** -p) + (right ** -p)
    w = np.nan_to_num(w, nan=0.0, posinf=0.0, neginf=0.0)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= s
    return w.astype(np.float64)


def _shape_sine_power(n: int, p: float) -> np.ndarray:
    """
    Sine‑power shape: a_i ∝ sin^p(π·(i+0.5)/n).
    """
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    w = np.power(vals, p)
    s = w.sum()
    if s == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w /= s
    return w.astype(np.float64)

# Local optimiser (exponentiated gradient with momentum)
# ----------------------------------------------------------------------


def _local_opt_with_momentum(initial: np.ndarray,
                             rng: np.random.Generator,
                             deadline: float,
                             max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Refine ``initial`` on the simplex using an exponentiated‑gradient step
    with a first‑moment accumulator (Adam‑style momentum).  The routine
    mixes exact sub‑gradients of the worst convolution entry with a
    soft‑max gradient whose temperature β grows exponentially.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    # momentum accumulator
    m = np.zeros_like(a)
    beta1 = 0.9

    # soft‑max temperature
    beta = 0.5
    beta_mul = 1.07

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # Choose gradient type
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            grad = _subgradient_step(a, conv)
        else:
            beta = min(1e5, beta * beta_mul)
            grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # Momentum update (bias‑corrected)
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        m_hat = m / (1.0 - beta1 ** t)

        # --------------------------------------------------------------
        # Exponentiated‑gradient update
        # --------------------------------------------------------------
        lr = 0.7 / np.sqrt(t)
        grad_clip = np.clip(m_hat, -20.0, 20.0)
        a = a * np.exp(-lr * grad_clip)
        s = a.sum()
        a = a / s if s > 0 else np.full_like(a, 1.0 / n)

        # --------------------------------------------------------------
        # Occasional exploratory moves
        # --------------------------------------------------------------
        if rng.random() < 0.025:
            a = _random_block_perturb(a, rng)

        if rng.random() < 0.025:
            a = _pairwise_move(a, rng, fft_len_conv)

    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score

    best_score = float("inf")
    candidates = []                     # (vector, n, score)

    # ------------------------------------------------------------------
    # If a previous construction exists, use it as a seed
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s == 0.0:
                cand = np.full_like(cand, 1.0 / cand.size)
            else:
                cand = cand / s
            score = _score(cand, _fft_len_for_len(2 * cand.size - 1))
            candidates.append((cand, cand.size, score))
            if score < best_score:
                best_score = score
                best_vec = cand.copy()
                print(f"[{time.time() - start:.2f}s] Loaded previous best (score={score:.6f})")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to be explored
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 3809,
               4000, 4500, 5000, 6000, 8000, 10000]

    # ------------------------------------------------------------------
    # Coarse scan over the three analytic families
    # ------------------------------------------------------------------
    for n in lengths:
        fft_len = _fft_len_for_len(2 * n - 1)

        # ----- symmetric Beta family -----
        for a_param in np.linspace(0.15, 2.0, 18):
            a = _shape_beta_distribution(n, a_param)
            score = _score(a, fft_len)
            candidates.append((a, n, score))
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                print(f"[{time.time() - start:.2f}s] Beta n={n} α={a_param:.3f} → {score:.6f}")

        # ----- edge‑power family -----
        for p in np.linspace(0.2, 5.0, 18):
            a = _shape_power_edge(n, p)
            score = _score(a, fft_len)
            candidates.append((a, n, score))
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                print(f"[{time.time() - start:.2f}s] Edge n={n} p={p:.3f} → {score:.6f}")

        # ----- sine‑power family -----
        for p in np.linspace(0.5, 5.0, 18):
            a = _shape_sine_power(n, p)
            score = _score(a, fft_len)
            candidates.append((a, n, score))
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                print(f"[{time.time() - start:.2f}s] Sine n={n} p={p:.3f} → {score:.6f}")

        # stop early if we are close to the deadline
        if time.time() + 5.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # Local refinement of the best parameter for each family / length
    # ------------------------------------------------------------------
    # Take the 12 best coarse candidates and perform a fine grid around them.
    candidates.sort(key=lambda x: x[2])
    for a_coarse, n, _ in candidates[:12]:
        fft_len = _fft_len_for_len(2 * n - 1)

        # Refine Beta around its current α (if applicable)
        # (detect by checking if the vector is symmetric and strongly peaked at ends)
        if np.allclose(a_coarse, _shape_beta_distribution(n, 1.0), atol=1e-6) is False:
            # Try a few nearby α values
            centre = 1.0  # a reasonable centre; we just scan a small interval
            for a_param in np.linspace(max(0.05, centre - 0.3), centre + 0.3, 13):
                a = _shape_beta_distribution(n, a_param)
                score = _score(a, fft_len)
                if score < best_score:
                    best_score = score
                    best_vec = a.copy()
                    print(f"[{time.time() - start:.2f}s] Refined Beta n={n} α={a_param:.3f} → {score:.6f}")

        # Refine Edge‑power around its current p (simple heuristic)
        for p in np.linspace(0.1, 6.0, 13):
            a = _shape_power_edge(n, p)
            score = _score(a, fft_len)
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                print(f"[{time.time() - start:.2f}s] Refined Edge n={n} p={p:.3f} → {score:.6f}")

        # Refine Sine‑power similarly
        for p in np.linspace(0.3, 6.0, 13):
            a = _shape_sine_power(n, p)
            score = _score(a, fft_len)
            if score < best_score:
                best_score = score
                best_vec = a.copy()
                print(f"[{time.time() - start:.2f}s] Refined Sine n={n} p={p:.3f} → {score:.6f}")

        if time.time() + 5.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # Polishing phase – run the momentum optimiser on the best few seeds
    # ------------------------------------------------------------------
    candidates.sort(key=lambda x: x[2])
    top_seeds = candidates[:8]    # keep a handful for polishing
    for vec, n, _ in top_seeds:
        if time.time() >= deadline:
            break
        iters = 3000
        vec_opt, score_opt = _local_opt_with_momentum(vec, rng, deadline,
                                                      max_iters=iters)
        if score_opt < best_score:
            best_score = score_opt
            best_vec = vec_opt.copy()
            print(f"[{time.time() - start:.2f}s] Polished seed n={n} → {score_opt:.6f}")

    # ------------------------------------------------------------------
    # Random Dirichlet restarts for the remaining time budget
    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = int(rng.choice(lengths))
        a0 = _init_random_dirichlet(rng, n)
        iters = int(2000 * (5000.0 / max(n, 500)))   # ~2000 for n≈5000
        vec_opt, score_opt = _local_opt_with_momentum(a0, rng, deadline,
                                                      max_iters=iters)
        if score_opt < best_score:
            best_score = score_opt
            best_vec = vec_opt.copy()
            print(f"[{time.time() - start:.2f}s] Random restart improved → {score_opt:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]


# ----------------------------------------------------------------------
# Helper for random Dirichlet initialisation (used in the final loop)
# ----------------------------------------------------------------------
def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via independent exponentials."""
    a = rng.exponential(scale=1.0, size=n)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)
