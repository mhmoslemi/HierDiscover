"""
Simple Adam‑based mirror‑descent on the probability simplex.

The algorithm
-------------
* Work with the log‑weights v (a_i = softmax(v_i)).  This automatically
  enforces non‑negativity and Σa_i = 1.
* The objective is the smooth approximation
        f_β(a) = (1/β)·log Σ_i exp(β·conv_i(a))
  whose gradient w.r.t. a is computed by a single FFT‑based convolution.
* Adam (adaptive moment estimation) is used on the variables v; the
  chain‑rule for the soft‑max yields the gradient w.r.t. v.
* β starts very small (smooth objective) and is multiplied by 1.02 every
  iteration, capped at β_max = 1 000.  This “homotopy’’ gradually
  focuses the optimisation on the worst convolution entry.
* Several cheap stochastic moves (log‑normal block perturbations) are
  injected from time to time to escape plateaus.
* A tiny set of initialisation families (random Dirichlet, sine‑power,
  interpolation of the current best) is explored on many lengths
  (2000 – 10 000).  The best vector seen so far is continually
  polished with a longer Adam run.
* The whole search respects the supplied time budget (default 1000 s).

The function ``propose_candidate`` returns a list of non‑negative heights
that (approximately) minimise ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) via real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _softmax_gradient(
    a: np.ndarray,
    conv: np.ndarray,
    beta: float,
    fft_len_grad: int,
) -> np.ndarray:
    """
    Gradient of the smooth max
        f_β(a) = (1/β)·log Σ_i exp(β·conv_i)
    w.r.t. the simplex vector a.
    """
    cur_max = conv.max()
    # safe exponentiation – the largest exponent is 0
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    # convolution of the weight vector with the reversed a
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)  # length = 3·n‑2
    # extract the part that corresponds to ∂f/∂a_i
    n = a.size
    grad = conv_w_rev[n - 1 : n - 1 + n] / sum_w
    return grad


# ----------------------------------------------------------------------
# Simple random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Score helper (fast, avoids evaluate_sequence overhead)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return the value 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponential sampling."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    a_i ∝ sin^p( π·(i+0.5)/n ),   p ∈ [0.5, 4].
    """
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_interpolate(
    rng: np.random.Generator, source: np.ndarray, n_target: int
) -> np.ndarray:
    """Linearly interpolate an existing distribution to a different length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (log‑weights)
# ----------------------------------------------------------------------


def _adam_optimize(
    initial: np.ndarray,
    rng: np.random.Generator,
    deadline: float,
    max_iters: int = 3000,
    beta0: float = 0.5,
    beta_factor: float = 1.02,
    beta_max: float = 1000.0,
    lr0: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Optimise ``initial`` (already on the simplex) by mirror descent
    with Adam on the log‑weights.  Returns the best vector seen and its
    fast‑computed score.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # Log‑weights – a = softmax(v)
    v = np.log(a + 1e-15)

    # Adam moments
    m = np.zeros_like(v)
    v2 = np.zeros_like(v)

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    beta = beta0
    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- convert back to simplex -----
        max_v = np.max(v)
        expv = np.exp(v - max_v)
        a = expv / expv.sum()

        # ----- evaluate current score -----
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ----- smooth‑max temperature schedule -----
        beta = min(beta_max, beta * beta_factor)

        # ----- gradient of the smooth max w.r.t. a -----
        grad_a = _softmax_gradient(a, conv, beta, fft_len_grad)

        # ----- chain rule for the soft‑max (log‑weights) -----
        dot = np.dot(a, grad_a)
        grad_v = a * (grad_a - dot)

        # ----- Adam update -----
        m = 0.9 * m + 0.1 * grad_v
        v2 = 0.999 * v2 + 0.001 * (grad_v ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - 0.9 ** t)
        v2_hat = v2 / (1.0 - 0.999 ** t)

        # decaying learning‑rate (1/√t)
        lr = lr0 / np.sqrt(t)

        v = v - lr * m_hat / (np.sqrt(v2_hat) + 1e-8)

        # ----- occasional stochastic perturbation -----
        if rng.random() < 0.015:
            a_pert = _random_block_perturb(a, rng)
            v = np.log(a_pert + 1e-15)

    # final evaluation of the best candidate
    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Seed with a previously known construction, if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cand_score = _score(cand, _fft_len_for_len(2 * cand.size - 1))
                best_vec = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start:5.2f}s] Loaded previous best: {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths we will explore (bias towards medium/large)
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 3809, 4000, 4500, 5000, 6000, 8000, 10000]

    # probability for the three initialisation families
    init_weights = np.array([0.45, 0.40, 0.15])  # Dirichlet, sine‑power, interpolate

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (uniformly – we already bias through the list)
        n = int(rng.choice(lengths))
        # choose initialisation method
        which = rng.choice(3, p=init_weights)

        if which == 0:  # random Dirichlet
            a0 = _init_random_dirichlet(rng, n)
        elif which == 1:  # sine‑power shape
            a0 = _init_sine_power(rng, n)
        else:  # interpolate from the current global best (if any)
            if best_vec is not None:
                a0 = _init_interpolate(rng, best_vec, n)
            else:
                a0 = _init_random_dirichlet(rng, n)

        # give each restart a budget of iterations that roughly matches the
        # remaining time – we keep it modest for very large n.
        if n <= 4000:
            iters = 4000
        elif n <= 6000:
            iters = 3000
        else:
            iters = 2000

        a_opt, score_opt = _adam_optimize(a0, rng, deadline, max_iters=iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:5.2f}s] New global best: {best_score:.6f} (n={n})")

        # occasional polishing of the current best (more iterations, slower β‑growth)
        if best_vec is not None and time.time() < deadline - 0.8:
            a_polished, score_polished = _adam_optimize(
                best_vec.copy(),
                rng,
                deadline,
                max_iters=6000,
                beta_factor=1.01,  # slower temperature rise → finer control
                lr0=0.03,
            )
            if score_polished < best_score:
                best_score = score_polished
                best_vec = a_polished.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:5.2f}s] Polished best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 3000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
