"""
Propose a step‑function via logit‑Adam optimisation.

The algorithm works in two phases for a fixed length *n*:

* Phase 1 (≈ 60 % of the allotted time) – minimise a smooth
  approximation of the maximal self‑convolution.
  The objective `max(a*a)` is replaced by a soft‑max with a temperature
  β that grows from 0.5 to 3000.  Gradients are obtained by a single FFT‑based
  convolution.  The parameters are stored as logits *x* (a = softmax(x)),
  allowing us to apply Adam updates without an explicit simplex projection.

* Phase 2 – exact sub‑gradient descent on the true peak.
  The sub‑gradient of `max(a*a)` equals
  `g_i = 2·a_{k-i}` where *k* is the index of the current peak.
  The same Adam machinery is used, but now with a constant, small learning
  rate.

Stagnation (no improvement for 30 s) triggers a random log‑normal jitter of
the logits and a reset of the Adam moments.

The optimiser is run for several candidate lengths (including the previous
construction if available).  The best sequence found across all lengths is
returned.  All values are clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of prescribed length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_full(u: np.ndarray, v: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution u*v (full length) using an FFT of prescribed length."""
    fu = np.fft.rfft(u, n=fft_len)
    fv = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(fu * fv, n=fft_len)[: u.size + v.size - 1]
    return conv


# ----------------------------------------------------------------------
# Core optimisation routine for a single length
# ----------------------------------------------------------------------


def _optimise_one_length(init_a: np.ndarray,
                         deadline: float,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Optimise a step function of a given length.

    Parameters
    ----------
    init_a : np.ndarray
        Initial probability vector (sum ≈ 1, non‑negative).
    deadline : float
        Absolute time (monotonic) after which the optimisation must stop.
    rng : np.random.Generator
        Random number generator for jitter restarts.

    Returns
    -------
    best_a : np.ndarray
        Best probability vector discovered.
    best_score : float
        Corresponding value of ``2·n·max(a*a)``  (equal to
        ``evaluate_sequence`` when ``sum(a)=1``).
    """
    # ----- initialise logit representation -----
    eps = 1e-12
    x = np.log(init_a + eps)          # logits; softmax(x) yields the probabilities

    n = init_a.size
    # FFT lengths (powers of two, fixed for the whole run)
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # Adam state
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # Time‑budget handling
    start = time.monotonic()
    total = deadline - start
    phase1_end = start + 0.60 * total

    # hyper‑parameters (linear schedules)
    lr_start = 0.5
    lr_end = 0.001
    beta_s_start = 0.5
    beta_s_end = 3000.0

    # tracking best solution
    best_a = None
    best_score = np.inf
    last_improve = start
    it = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        it += 1

        # ----- build probability vector from logits -----
        max_x = np.max(x)
        a = np.exp(x - max_x)
        a_sum = a.sum()
        a = a / a_sum                     # ensures Σ a = 1

        # ----- self‑convolution and current score -----
        conv = _self_conv(a, fft_len_self)
        max_c = conv.max()
        cur_score = 2.0 * n * max_c / (a_sum ** 2)   # == evaluate_sequence when sum=1

        # ----- keep the best encountered -----
        if cur_score + 1e-12 < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} it={it:7d} new best {best_score:.6f}")

        # ----- gradient computation -----
        if now < phase1_end:
            # ----- phase 1: smooth soft‑max approximation -----
            frac = (now - start) / (phase1_end - start)
            lr = lr_start * (1.0 - frac) + lr_end * frac
            beta_s = beta_s_start + (beta_s_end - beta_s_start) * frac

            # soft‑max weights over the convolution entries
            scaled = beta_s * (conv - max_c)                     # max_c → 0
            # stable exponentiation
            max_scaled = np.max(scaled)                         # always 0, but kept for safety
            w = np.exp(scaled - max_scaled)
            w_sum = w.sum()
            p = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

            # gradient wrt a : 2·(p * rev(a))
            rev_a = a[::-1]
            grad_full = _conv_full(p, rev_a, fft_len_grad)[: 3 * n - 2]
            grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]
        else:
            # ----- phase 2: exact sub‑gradient of the current peak -----
            lr = lr_end
            k = int(conv.argmax())
            grad_a = np.zeros_like(a)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            idx_i = np.arange(i_min, i_max + 1, dtype=int)
            idx_j = k - idx_i
            grad_a[idx_i] = 2.0 * a[idx_j]

        # ----- map gradient to logits space (softmax Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)           # ∂L/∂x = Jᵀ·∂L/∂a

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        x = x - step

        # ----- stagnation handling -----
        if now - last_improve > 30.0:
            # random log‑normal jitter in logits space, reset moments
            m.fill(0.0)
            v.fill(0.0)
            noise = rng.normal(0.0, 0.5, size=n)   # moderate perturbation
            x = x + noise
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] n={n:5d} restart (stagnation)")

    # fallback in the unlikely case no improvement was recorded
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds (the function must return before the budget
        expires; a 5 s safety margin is subtracted automatically).

    Returns
    -------
    list[float]
        The best sequence of heights discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose candidate lengths.
    #    We always include the length of any previous construction and a few
    #    powers‑of‑two around the current champion (≈ 3800).
    # ------------------------------------------------------------------
    lengths = []

    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
            lengths.append(prev.size)

    # Additional lengths – a mix of small and medium sizes.
    for n in (1024, 2048, 3000, 3809, 4096, 5000, 6000):
        if n not in lengths:
            lengths.append(n)

    best_overall = None
    best_score_overall = np.inf

    # ------------------------------------------------------------------
    # 2) Optimise each length in turn, respecting the remaining time.
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # ----- initialise a probability vector for this length -----
        if "height_sequence_1" in globals():
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.size == n:
                a0 = prev.copy()
            else:
                # linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, prev.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(x_new, x_old, prev)
        else:
            a0 = rng.dirichlet(np.ones(n)).astype(np.float64)

        # normalise to the probability simplex (sum ≈ 1, non‑negative)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / n)

        # ----- run the optimiser for this length -----
        vec, score = _optimise_one_length(a0, deadline, rng)

        # ----- keep the global best -----
        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - (deadline - (budget_s - 5.0))
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

        # Warm‑start the next length with the current best (interpolated)
        # – this helps later runs start from a promising shape.
        if best_overall is not None:
            # store for the next iteration (will be interpolated if needed)
            globals()["height_sequence_1"] = best_overall.tolist()

    # ------------------------------------------------------------------
    # 3) Fallback – should never happen but guards against an empty result.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)

    # ------------------------------------------------------------------
    # 4) Clip to the limits enforced by ``evaluate_sequence`` and return.
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    return [float(x) for x in best_overall]
