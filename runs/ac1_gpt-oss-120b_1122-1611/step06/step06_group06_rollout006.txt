"""
Propose a step‑function (non‑negative heights) that minimises evaluate_sequence().

Algorithm (≈ 1000 s budget)
---------------------------
1. **Diverse seed generation** – builds a large pool of candidate vectors from
   analytic families (sqrt‑difference, power‑difference, exponential,
   Beta‑distribution, Kaiser window) plus many random Dirichlet draws.
   All candidates are projected onto the probability simplex.

2. **Fast pre‑selection** – evaluates every seed with a cheap FFT based
   computation of 2·n·max(convolution) (exact for normalised vectors) and
   keeps the 20 best seeds.

3. **Mirror‑descent refinement** – each of the elite seeds is improved by an
   exponentiated‑gradient (mirror‑descent) loop that works directly on the
   L∞‑objective.
   The true sub‑gradient of the max‑convolution is used (the gradient of the
   active shift(s)), a linear schedule drives the stepsize η from 2 → 1e‑3,
   a small momentum term stabilises the updates and occasional random
   block / pairwise perturbations keep the search from stagnating.

4. **Fine‑tuned simulated‑annealing** – after the time‑critical phase a short
   SA phase explores random mass‑transfer moves on the best vector found so
   far, accepting uphill moves with an exponentially cooling temperature.

5. **Return** – the best vector is clipped to the limits imposed by
   ``evaluate_sequence`` and returned as a plain list of floats.

The routine respects the 1000 s time budget, prints progress stamps and
returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1) Helper functions (FFT based convolution, scoring, normalisation)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a full convolution of size n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution (full) of a non‑negative vector a using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact value of evaluate_sequence() for a normalised (sum=1) vector."""
    conv = _conv_self(a, fft_len)
    return float(2.0 * n * conv.max())


def _normalise(v: np.ndarray) -> np.ndarray:
    """Project a non‑negative vector onto the probability simplex."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# 2) Candidate generators – a wide variety of analytic / random families
# ----------------------------------------------------------------------
def _generate_candidates(rng: np.random.Generator, n: int) -> list[np.ndarray]:
    """Return a list of diverse normalised candidate vectors."""
    candidates: list[np.ndarray] = []

    # a) Analytic √‑difference (classical start)
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    candidates.append(_normalise(a_sqrt))

    # b) Previous construction if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                candidates.append(_normalise(prev))
        except Exception:
            pass

    # c) Power‑difference families (p from 0.05 to 3)
    for p in np.linspace(0.05, 3.0, 15):
        a = ((np.arange(n, dtype=np.float64) + 1) ** (-p)
             - (np.arange(n, dtype=np.float64) + 2) ** (-p))
        candidates.append(_normalise(a))

    # d) Exponential families (λ from 5e‑4 to 3e‑2)
    for lam in np.linspace(5e-4, 3e-2, 15):
        a = (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))
        candidates.append(_normalise(a))

    # e) Beta‑distribution shapes (α,β ∈ [0.1,8])
    for _ in range(30):
        alpha = rng.uniform(0.1, 8.0)
        beta = rng.uniform(0.1, 8.0)
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        a = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
        candidates.append(_normalise(a))

    # f) Kaiser windows (β ∈ [0,20])
    for beta in np.linspace(0.0, 20.0, 12):
        a = np.kaiser(n, beta)
        candidates.append(_normalise(a))

    # g) Random Dirichlet / Gamma samples
    for _ in range(80):
        alpha = rng.uniform(0.1, 3.0)
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(_normalise(a))

    # h) Random mixtures of two previously generated candidates
    for _ in range(30):
        i, j = rng.integers(0, len(candidates), size=2)
        w = rng.random()
        mix = w * candidates[i] + (1.0 - w) * candidates[j]
        candidates.append(_normalise(mix))

    return candidates


# ----------------------------------------------------------------------
# 3) Mirror‑descent refinement (exponentiated‑gradient on the L∞‑objective)
# ----------------------------------------------------------------------
def _refine_candidate(initial: np.ndarray,
                      n: int,
                      fft_len: int,
                      deadline: float,
                      best_score_ref: list,
                      best_vec_ref: list) -> None:
    """
    Improves a candidate vector in place using a mirror‑descent loop.
    The global best (as mutable lists) is updated when a better vector is found.
    """
    cur = initial.copy()
    grad_prev = np.zeros_like(cur)

    # hyper‑parameters
    eta_start, eta_end = 2.0, 1e-3          # step‑size schedule
    rel_tol = 0.995                        # indices within 0.5 % of the max
    max_active = 12
    momentum = 0.9
    prob_block = 0.04                      # block perturbation probability
    prob_pair = 0.09                       # pairwise transfer probability

    start = time.monotonic()
    while time.monotonic() < deadline:
        # ----- convolution & score -----
        conv = _conv_self(cur, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ----- update global best if improved -----
        if cur_score < best_score_ref[0] - 1e-12:
            best_score_ref[0] = cur_score
            best_vec_ref[0] = cur.copy()
            now = time.monotonic()
            print(f"[{now - start:6.2f}s] new global best = {cur_score:.9f}")

        # ----- schedule -----
        now = time.monotonic()
        frac = (now - start) / (deadline - start) if deadline > start else 0.0
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # ----- active set (shifts close to the current maximum) -----
        active = np.where(conv >= cur_max * rel_tol)[0]
        if active.size > max_active:
            # keep the largest values only
            active = np.argsort(conv)[-max_active:]

        # ----- compute sub‑gradient of the (soft‑max) surrogate -----
        grad = np.zeros_like(cur)
        if active.size == 0:
            # empty (should not happen) – fall back to single max shift
            active = np.array([conv.argmax()], dtype=np.int64)

        # soft‑max weights over the active shifts
        w_raw = np.exp(10.0 * (conv[active] - cur_max))   # temperature ≈10
        w_sum = w_raw.sum()
        w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)

        for shift, weight in zip(active, w):
            lo = max(0, shift - (n - 1))
            hi = min(shift, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=np.int64)
            # contribution of a_i * a_{shift-i}
            grad[i_range] += weight * 2.0 * cur[shift - i_range]

        # ----- momentum -----
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # ----- exponentiated‑gradient update (mirror descent) -----
        factor = np.exp(-eta * grad)
        cur *= factor
        cur = np.clip(cur, 0.0, None)
        s = cur.sum()
        if s > 0.0:
            cur /= s
        else:
            cur[:] = 1.0 / n

        # ----- occasional random block perturbation -----
        if rng.random() < prob_block:
            width = rng.integers(1, max(2, n // 15))
            start_idx = rng.integers(0, n - width + 1)
            blk_factor = np.exp(rng.normal(0.0, 0.25))
            cur[start_idx:start_idx + width] *= blk_factor
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()

        # ----- occasional pairwise mass transfer -----
        if rng.random() < prob_pair:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur[i] > 1e-14:
                delta = rng.random() * cur[i] * 0.4
                cur[i] -= delta
                cur[j] += delta
                cur = np.clip(cur, 0.0, None)
                cur /= cur.sum()


# ----------------------------------------------------------------------
# 4) Simulated‑annealing fine‑tuning (remaining time)
# ----------------------------------------------------------------------
def _simulated_annealing(start_vec: np.ndarray,
                         n: int,
                         fft_len: int,
                         deadline: float,
                         best_score_ref: list,
                         best_vec_ref: list) -> None:
    """
    Small SA run that starts from the current best vector.
    """
    cur = start_vec.copy()
    cur_score = _score_of(cur, n, fft_len)

    # linear cooling from T_start to T_end
    T_start, T_end = 5e-3, 1e-7
    start = time.monotonic()
    it = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start) / (deadline - start) if deadline > start else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        # propose a random mass‑transfer move
        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] < 1e-14:
            continue
        delta = rng.random() * cur[i] * 0.5
        new_vec = cur.copy()
        new_vec[i] -= delta
        new_vec[j] += delta

        new_score = _score_of(new_vec, n, fft_len)

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new_vec
            cur_score = new_score
            if new_score < best_score_ref[0] - 1e-12:
                best_score_ref[0] = new_score
                best_vec_ref[0] = cur.copy()
                print(f"[{now - start:6.2f}s] SA improvement → {new_score:.9f}")

        # occasional gentle smoothing (helps escape tiny plateaus)
        if it % 4000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            cur = smoothed
            cur_score = _score_of(cur, n, fft_len)

        it += 1


# ----------------------------------------------------------------------
# 5) Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence().
    Returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # 1) Generate a large pool of seeds
    # ------------------------------------------------------------------
    raw_candidates = _generate_candidates(rng, n)

    # ------------------------------------------------------------------
    # 2) Quick evaluation of the pool & keep the best few
    # ------------------------------------------------------------------
    scores = np.empty(len(raw_candidates), dtype=np.float64)
    for i, cand in enumerate(raw_candidates):
        scores[i] = _score_of(cand, n, fft_len)

    # keep the 20 most promising seeds
    elite_idx = np.argpartition(scores, 20)[:20]
    elite_vectors = [raw_candidates[i].copy() for i in elite_idx]

    best_score = float('inf')
    best_vec = None
    for i, vec in zip(elite_idx, elite_vectors):
        sc = scores[i]
        if sc < best_score:
            best_score = sc
            best_vec = vec.copy()

    print(f"[init] pool best = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 3) Mirror‑descent refinement on the elite vectors
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # reserve roughly 20 s for SA (or less if budget is tight)
    refine_deadline = start_time + max(0.0, float(budget_s) - 20.0)

    # mutable holders for the global best (so helpers can update them)
    best_score_ref = [best_score]
    best_vec_ref = [best_vec]

    for vec in elite_vectors:
        if time.monotonic() >= refine_deadline:
            break
        _refine_candidate(vec, n, fft_len, refine_deadline,
                          best_score_ref, best_vec_ref)

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning on the final best vector
    # ------------------------------------------------------------------
    sa_deadline = start_time + float(budget_s)
    if time.monotonic() < sa_deadline:
        _simulated_annealing(best_vec_ref[0], n, fft_len,
                             sa_deadline, best_score_ref, best_vec_ref)

    # ------------------------------------------------------------------
    # 5) Return the best sequence (after clipping to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    final_vec = np.clip(best_vec_ref[0], 0.0, 1000.0)
    return [float(v) for v in final_vec]
