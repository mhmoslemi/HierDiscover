def _targeted_transfer(a, rng):
    n = a.shape[0]; conv = _conv_self(a, fft_len); i_max = conv.argmax()
    # compute partner vector (mirror index)
    partner = np.zeros_like(a)
    lo = max(0, i_max - (n-1))
    hi = min(i_max, n-1)
    # for j from lo to hi, partner[j] = a[i_max - j]
    idx = np.arange(lo, hi+1)
    partner[idx] = a[i_max - idx]
    contributions = a * partner
    # choose src among top 10% contribution entries
    top_k = max(1, int(0.05 * n))
    src_candidates = np.argpartition(contributions, -top_k)[-top_k:]
    src = rng.choice(src_candidates)
    # choose dst among bottom 10%
    bot_k = max(1, int(0.05 * n))
    dst_candidates = np.argpartition(contributions, bot_k)[:bot_k]
    dst = rng.choice(dst_candidates)
    if src == dst: return a  # fallback rarely
    max_delta = min(a[src], 0.2 * a[src])
    if max_delta <= 0: return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[src] -= delta
    a2[dst] += delta
    # No renorm needed, sum unchanged
    # Ensure nonnegative
    a2 = np.clip(a2, 0.0, None)
    return a2

def _adam_optimize(initial_a, rng, deadline, max_iters):
    n = len(initial_a)
    # logit representation
    epsilon = 1e-12
    z = np.log(initial_a + epsilon)
    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    beta = 0.5
    beta_mul = 1.005
    beta_max = 400.0

    # precompute FFT lengths
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    best_vec = initial_a.copy()
    best_score = _score(initial_a, fft_len_conv)

    for t in range(1, max_iters+1):
        now = time.time()
        if now >= deadline:
            break

        # compute current a from logits
        # for numerical stability: shift
        z_shift = z - np.max(z)
        a = np.exp(z_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(z, 1.0 / n)
        else:
            a = a / a_sum

        # convolution and loss
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # smooth max gradient
        beta = min(beta_max, beta * beta_mul)
        grad_a = _softmax_gradient(a, conv, beta, fft_len_grad)

        # convert to logit gradient
        avg = np.dot(grad_a, a)
        grad_z = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * (grad_z ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # occasional targeted move
        if rng.random() < 0.03:
            a2 = _targeted_transfer(a, rng)
            # evaluate
            score2 = _score(a2, fft_len_conv)
            if score2 < cur_score:
                # Accept move by moving logits accordingly
                # To reflect a2, we set logits to log of a2 (plus epsilon)
                z = np.log(a2 + epsilon)
                # optionally reset Adam moments?
                m = np.zeros_like(z)
                v = np.zeros_like(z)
                # update best if improved
                if score2 < best_score:
                    best_score = score2
                    best_vec = a2.copy()
        # occasional random block perturbation
        if rng.random() < 0.02:
            a2 = _random_block_perturb(a, rng)
            # update logits
            z = np.log(a2 + epsilon)

    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score

"""
A fast mixed Adam‑mirror‑descent optimizer for the step‑function problem.

The score to minimise is
        S(a) = 2·n·max(conv(a,a)) / (sum(a)**2) .
Because the expression is scale‑invariant we restrict the search to the
probability simplex (∑a = 1, a_i ≥ 0).  The optimiser works on the logits
z_i (so that a_i = softmax(z)_i) and minimises a smooth approximation of the
max using the log‑sum‑exp (soft‑max) surrogate

        f_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k) .

The gradient ∂f_β/∂a is obtained via a convolution with the weight vector
w_k = exp(β·((a*a)_k – max((a*a))) ) and the update is performed with an
Adam step in the log‑space (mirror descent).  β grows exponentially during a
run so that the optimisation starts smooth and finishes focusing on the true
maximum.

To escape plateaus we sprinkle cheap random moves:
* a log‑normal perturbation of a random block,
* a targeted mass‑transfer move that moves probability from indices that
  heavily contribute to the worst convolution entry towards indices that
  contribute little.

Various initial families are tried (Dirichlet, sine‑power, Kaiser, Hann,
Blackman, and interpolation of the current best).  The whole search respects
the 1000 s budget; the best vector found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    ∂/∂a of  (1/β)·log Σ_i exp(β·conv_i).
    Returned array has shape ``a`` and sums to 1.
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))          # safe: max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    # select the n entries that correspond to the derivative w.r.t. a
    grad = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad


# ----------------------------------------------------------------------
# Simple random moves
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _targeted_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a small amount of mass from an index that contributes heavily to the
    worst convolution entry towards one that contributes little.
    """
    n = a.size
    conv = _conv_self(a, _fft_len_for_len(2 * n - 1))
    i_max = int(conv.argmax())

    # partner value for each position (the value that appears together with a[j]
    # in the sum defining conv[i_max]).
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    partner = np.zeros_like(a)
    partner[idx] = a[i_max - idx]

    contrib = a * partner
    if contrib.max() == 0.0:
        return a

    # choose source among the top 5 % contributors, destination among the bottom 5 %
    top_k = max(1, int(0.05 * n))
    bot_k = max(1, int(0.05 * n))
    src_candidates = np.argpartition(contrib, -top_k)[-top_k:]
    dst_candidates = np.argpartition(contrib, bot_k)[:bot_k]
    src = rng.choice(src_candidates)
    dst = rng.choice(dst_candidates)
    if src == dst or a[src] == 0.0:
        return a

    # transfer a modest fraction of the source mass
    max_delta = min(a[src], 0.2 * a[src])
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[src] -= delta
    a2[dst] += delta
    return a2


# ----------------------------------------------------------------------
# Score helper (fast, avoids evaluate_sequence overhead)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Exact score 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initialisation families
# ----------------------------------------------------------------------


def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponentials."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """a_i ∝ sin^p(π·(i+0.5)/n) with random exponent p ∈ [0.5,4]."""
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_kaiser(rng: np.random.Generator, n: int) -> np.ndarray:
    """Kaiser window a_i = I0(π·β·√(1‑((2i/(n‑1)‑1)^2))) / I0(π·β)."""
    beta = rng.uniform(0.0, 12.0)          # typical β ∈ [0,12]
    a = np.kaiser(n, beta)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_hann(rng: np.random.Generator, n: int) -> np.ndarray:
    """Hann (raised‑cosine) window."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_blackman(rng: np.random.Generator, n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42
         - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linear interpolation of an existing distribution to a new length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam‑mirror‑descent optimiser on the simplex
# ----------------------------------------------------------------------


def _adam_optimize(initial_a: np.ndarray,
                   rng: np.random.Generator,
                   deadline: float,
                   max_iters: int = 1500) -> tuple[np.ndarray, float]:
    """
    Optimise a simplex vector using Adam on the log‑space (mirror descent)
    while minimising a smooth max of the self‑convolution.
    Returns the best vector encountered and its exact score.
    """
    n = initial_a.size
    # logit representation (add a tiny epsilon to avoid log(0))
    eps = 1e-12
    z = np.log(initial_a + eps)

    # Adam state
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05

    # smooth‑max schedule
    beta = 0.5
    beta_mul = 1.005
    beta_max = 400.0

    # FFT lengths (reused each iteration)
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # keep the best seen
    best_vec = initial_a.copy()
    best_score = _score(best_vec, fft_len_conv)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---------- recover a from logits ----------
        z_shift = z - np.max(z)                # numerical stability
        a = np.exp(z_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(z, 1.0 / n)
        else:
            a /= a_sum

        # ---------- evaluate ----------
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ---------- smooth‑max gradient ----------
        beta = min(beta_max, beta * beta_mul)
        grad_a = _softmax_gradient(a, conv, beta, fft_len_grad)

        # ---------- mirror‑descent (Adam) step ----------
        avg = np.dot(grad_a, a)                # ⟨grad_a , a⟩
        grad_z = a * (grad_a - avg)            # ∂L/∂z

        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ---------- occasional exploratory moves ----------
        # cheap random block perturbation
        if rng.random() < 0.02:
            a_tmp = _random_block_perturb(a, rng)
            z = np.log(a_tmp + eps)

        # targeted mass‑transfer move
        if rng.random() < 0.03:
            a_tmp = _targeted_transfer(a, rng)
            sc = _score(a_tmp, fft_len_conv)
            if sc < cur_score:
                # accept – reset momentum so we do not jump away wildly
                z = np.log(a_tmp + eps)
                m.fill(0.0)
                v.fill(0.0)
                if sc < best_score:
                    best_score = sc
                    best_vec = a_tmp.copy()

    # final exact evaluation of the best candidate
    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the provided time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 0.3)   # small safety margin

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Load a previously known construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:

                else:
                    cand = cand / cand.sum()
                cand_score = _score(cand, _fft_len_for_len(2 * cand.size - 1))
                best_vec = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start:5.2f}s] Loaded previous best: {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths to be explored (bias towards medium/large sizes)
    # ------------------------------------------------------------------
    lengths = [1800, 2400, 3000, 3600, 3809, 4200,

    # families and their probabilities
    families = ["dirichlet", "sine", "kaiser", "hann", "blackman", "interpolate"]
    fam_weights = np.array([0.30, 0.20, 0.15, 0.10, 0.10, 0.15])

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:

            best_vec = a_opt.copy()
            elapsed = time.time() - start

        # ------------------------------------------------------------------
        # Occasionally polish the current best with a longer run
        # ------------------------------------------------------------------
        if best_vec is not None and time.time() < deadline - 1.0:
