def _shape_r(r, n, offset=0.0):
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1.0) ** r - (i + offset) ** r
    if np.any(np.isnan(a)):
        return None
    # ensure nonnegative; due to rounding may have tiny negative near 0 for small offset; clip
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

def _shape_p(p, n, offset=0.0):
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1.0) ** (-p) - (i + offset + 2.0) ** (-p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (treated as a probability vector)
that minimises

    score = 2·n·max_k (a ∗ a)_k / (∑ a_i)^2

where `∗` is a linear convolution.  The routine combines three ideas:

1. **Analytic families** – generate candidates from two simple one‑parameter
   families (differences of power functions) and from a richer family that
   adds an offset to the argument.  Random mixtures of the two families are
   also explored.

2. **Intensive random search** – while time permits, draw random parameters
   (exponents, offsets and mixing ratios) and keep the best vector found.
   Convolution and the score are computed by FFT for speed.

3. **Soft‑max mirror‑descent refinement** – start from the best vector and
   perform an exponentiated‑gradient (mirror‑descent) optimisation on a
   smooth approximation of the max‑convolution using a log‑sum‑exp surrogate.
   The temperature β is increased gradually, the learning rate η decays, and
   occasional random perturbations keep the search from stagnating.

The function respects the supplied time budget (≈ 1000 s) and returns the
best sequence found, already clipped to the limits required by
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution helpers
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two that accommodates a linear convolution of length 2·n‑1."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_normalised(a: np.ndarray, fft_len: int) -> float:
    """
    Exact objective for a probability vector `a` (i.e. ∑a = 1).
    Returns 2·n·max(conv) which equals the value returned by evaluate_sequence.
    """
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Analytic families (including optional offset)
# ----------------------------------------------------------------------
def _shape_r(r: float, n: int, offset: float = 0.0) -> np.ndarray | None:
    """a_i = (i+offset+1)^r – (i+offset)^r  (decreasing for 0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1.0) ** r - (i + offset) ** r
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


def _shape_p(p: float, n: int, offset: float = 0.0) -> np.ndarray | None:
    """a_i = (i+offset+1)^(-p) – (i+offset+2)^(-p) (p>0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + offset + 1.0) ** (-p) - (i + offset + 2.0) ** (-p)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s <= 0.0:
        return None
    return a / s


# ----------------------------------------------------------------------
# Soft‑max and gradient utilities used in the mirror‑descent phase
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable softmax with temperature β."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a∗a)_k w.r.t. a.
    Returns 2·a⋆w (correlation) as a dense vector of length len(a).
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    return 2.0 * corr_full[n - 1 : n - 1 + n]


# ----------------------------------------------------------------------
# Main entry point required by the judging harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine respects the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – if a previous construction is available, honour its length
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
            n = int(cand0.size)
        else:
            n = int(kwargs.get("n", 3800))
    else:
        n = int(kwargs.get("n", 3800))

    if n < 2:
        n = 2

    fft_len = _fft_len_for_n(n)

    best_a = None          # best normalised vector (sum == 1)
    best_score = np.inf    # best objective value (identical to evaluate_sequence)

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and possibly update the incumbent
    # ------------------------------------------------------------------
    def _maybe_update(a: np.ndarray):
        nonlocal best_a, best_score
        if a is None or a.size != n:
            return
        s = a.sum()
        if s <= 0.0:
            return
        a_norm = a / s                    # enforce sum = 1
        sc = _score_normalised(a_norm, fft_len)
        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_norm.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 0) Use the supplied construction if present
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    if "height_sequence_1" in globals():
        try:
            cand0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand0.ndim == 1 and cand0.size > 0 and np.isfinite(cand0).all():
                # if the stored vector has a different length, resample it linearly
                if cand0.size != n:
                    x_old = np.linspace(0, 1, cand0.size)
                    x_new = np.linspace(0, 1, n)
                    cand0 = np.interp(x_new, x_old, cand0)
                _maybe_update(cand0)
                print("[init] using height_sequence_1")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Simple analytic families (no offset)
    # ------------------------------------------------------------------
    for r in np.linspace(0.5, 0.95, 91):
        a = _shape_r(r, n)
        _maybe_update(a)

    for p in np.linspace(0.1, 2.0, 96):
        a = _shape_p(p, n)
        _maybe_update(a)

    # ------------------------------------------------------------------
    # 2) Randomised search over offsets, exponents and mixtures
    # ------------------------------------------------------------------
    deadline = start_time + max(0.0, float(budget_s) - 10.0)  # keep ~10 s for final refinement
    while time.monotonic() < deadline:
        mode = rng.integers(0, 3)          # 0 – r‑family, 1 – p‑family, 2 – mixture
        if mode == 0:
            r = rng.uniform(0.30, 0.99)
            off = rng.uniform(0.0, n * 0.5)
            a = _shape_r(r, n, off)
        elif mode == 1:
            p = rng.uniform(0.05, 5.0)
            off = rng.uniform(0.0, n * 0.5)
            a = _shape_p(p, n, off)
        else:
            # mixture of the two families
            lam = rng.random()
            r = rng.uniform(0.30, 0.99)
            off_r = rng.uniform(0.0, n * 0.5)
            p = rng.uniform(0.05, 5.0)
            off_p = rng.uniform(0.0, n * 0.5)
            a1 = _shape_r(r, n, off_r)
            a2 = _shape_p(p, n, off_p)
            if a1 is None or a2 is None:
                continue
            a = lam * a1 + (1.0 - lam) * a2
        _maybe_update(a)

    # ------------------------------------------------------------------
    # 3) Mirror‑descent refinement on the best vector found so far
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)   # fallback: uniform
        best_score = _score_normalised(best_a, fft_len)

    cur_a = best_a.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    refine_start = time.monotonic()
    refine_deadline = refine_start + max(0.0, float(budget_s) - (time.monotonic() - start_time) - 2.0)

    # schedules – a slow linear annealing
    eta_start, eta_end = 0.8, 0.001
    beta_start, beta_end = 0.5, 8000.0

    while time.monotonic() < refine_deadline:
        now = time.monotonic()
        elapsed = now - refine_start
        total = refine_deadline - refine_start

        # linearly interpolated hyper‑parameters
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        conv = _conv_self(cur_a, fft_len)
        cur_max = conv.max()
        cur_score = float(2.0 * n * cur_max)   # sum(cur_a) == 1 by construction

        # record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = now
            print(f"[refine {elapsed:6.2f}s] new best = {best_score:.6f}")

        # soft‑max surrogate and gradient
        w = _softmax(conv, beta)
        grad = _gradient_via_correlation(cur_a, w)

        # exponentiated‑gradient (mirror descent) update
        factor = np.exp(-eta * grad)
        cur_a = cur_a * factor
        cur_a = np.clip(cur_a, 0.0, None)
        s = cur_a.sum()
        if s == 0.0:
            cur_a = np.full_like(cur_a, 1.0 / n)
        else:
            cur_a = cur_a / s

        # occasional stochastic perturbations to escape plateaus
        if rng.random() < 0.05:                               # block perturbation
            width = rng.integers(1, max(2, n // 50))
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.2))
            cur_a[start:start + width] *= mult
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = cur_a / cur_a.sum()

        if rng.random() < 0.10:                               # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > 1e-14:
                delta = rng.random() * cur_a[i] * 0.4
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = np.clip(cur_a, 0.0, None)
                cur_a = cur_a / cur_a.sum()

        # restart if no progress for a while
        if now - last_improve > 120.0:
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            cur_a = best_a * jitter
            cur_a = cur_a / cur_a.sum()
            last_improve = now
            print(f"[refine {elapsed:6.2f}s] restart after stagnation")

    # ------------------------------------------------------------------
    # Return best vector (already normalised, safe for evaluate_sequence)
    # ------------------------------------------------------------------
    final = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in final]
