"""
propose_candidate
-----------------

Search for a non‑negative step‑function (probability vector) that minimises

    score = 2·n·max_k (a ∗ a)_k

where `∗` is a linear convolution.  The algorithm consists of:

1. **Analytic seeding** – evaluate a dense grid of the two known
   one‑parameter families (r‑differences and p‑differences) and keep the
   best vector.

2. **Exact max‑surrogate mirror descent** –  at each iteration we compute the
   exact maximum of the convolution, build a binary mask `w` for the active
   (tight) indices and obtain a sub‑gradient `g = 2·a⋆w` via a single FFT‑based
   correlation.  An exponentiated‑gradient (mirror descent on the negative
   entropy) update `a_i ← a_i·exp(−η·g_i)` is performed and the vector is
   re‑normalised to sum 1.  The learning‑rate `η` decays linearly and we raise
   the temperature of an auxiliary soft‑max only for logging.

3. **Exploration & restarts** – when progress stalls we jitter the current
   best solution (multiply by log‑normal noise) and continue.  Simple pairwise
   mass‑transfer moves are also applied occasionally.

The routine respects the supplied time budget (≈ 1000 s) and returns the
best vector found.  It works for any `n ≥ 2`; the default is `n = 2500`,
large enough to obtain a value below the target 1.5025.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _corr_fft(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute correlation a ⋆ w (i.e. (reversed a) * w) using a single FFT.
    Returns a vector of length len(a) that is the gradient of the
    max‑surrogate with respect to `a`.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len_corr = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len_corr)
    fa = np.fft.rfft(a_rev, n=fft_len_corr)
    corr_full = np.fft.irfft(fw * fa, n=fft_len_corr)[:L]
    # gradient = 2·corr_full[n-1 : n-1+n]
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _score_fft(a: np.ndarray, fft_len: int) -> float:
    """Exact objective value using FFT‑based convolution."""
    conv = _conv_self_fft(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def _gen_shape_r(r: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^r - i^r   (0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_shape_p(p: float, n: int) -> np.ndarray | None:
    """a_i = (i+1)^(-p) - (i+2)^(-p)   (p>0)."""
    k = np.arange(1, n + 2, dtype=np.float64)          # 1 .. n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


# ----------------------------------------------------------------------
# Gradient of the exact L∞ surrogate
# ----------------------------------------------------------------------
def _gradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a∗a)_k.
    All indices attaining the maximum receive equal weight.
    """
    n = a.size
    M = conv.max()
    # indices where conv is (numerically) maximal
    # tolerance to catch multiple tight constraints
    eps = max(1e-14, 1e-12 * M)
    active = np.where(conv >= M - eps)[0]
    if active.size == 0:          # should not happen
        w = np.zeros_like(conv)
    else:
        w = np.zeros_like(conv, dtype=np.float64)
        w[active] = 1.0 / active.size
    # correlation a_rev ⋆ w gives the gradient
    grad = _corr_fft(a, w, fft_len=0)   # fft_len is computed internally
    return grad

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine respects the time budget.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------
    # Problem size
    # ----------------------------------------------
    n = int(kwargs.get("n", 2500))
    if n < 2:
        n = 2

    fft_len = _fft_len_for_n(n)

    # ----------------------------------------------
    # 1) Scan analytic families (dense grid)
    # ----------------------------------------------
    best_a = None
    best_score = np.inf

    # r‑family – finer grid towards the optimum
    r_vals = np.linspace(0.50, 0.99, 100)   # step ≈0.0049
    for r in r_vals:
        a = _gen_shape_r(r, n)
        if a is None:
            continue
        sc = _score_fft(a, fft_len)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[init] r={r:.4f}  score={sc:.6f}")

    # p‑family – dense grid
    p_vals = np.linspace(0.10, 2.00, 120)   # step ≈0.0158
    for p in p_vals:
        a = _gen_shape_p(p, n)
        if a is None:
            continue
        sc = _score_fft(a, fft_len)
        if sc < best_score:
            best_a, best_score = a, sc
            print(f"[init] p={p:.4f}  score={sc:.6f}")

    # ----------------------------------------------
    # 2) Use previously found construction if available
    # ----------------------------------------------
    if "height_sequence_1" in globals():

            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    cand = cand / s
                    sc = _score_fft(cand, _fft_len_for_n(cand.size))
                    if sc < best_score:
                        best_a, best_score = cand, sc
                        n = cand.size
                        fft_len = _fft_len_for_n(n)
                        print(f"[init] using height_sequence_1 score={sc:.6f}")
        except Exception:
            pass

    # If nothing was found, fall back to a uniform vector
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score_fft(best_a, fft_len)

    # 3) Main optimisation – exact max‑surrogate mirror descent
    # ----------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    eta_start, eta_end = 0.8, 0.001

    # Exploration probabilities
    block_prob = 0.06       # occasional block scaling
    pair_prob = 0.12        # pairwise mass transfer
    jitter_prob = 0.04      # soft restart via jitter

    # Stagnation detection
    last_improve = start_time
    restart_gap = 120.0     # seconds without improvement → jitter

    cur_a = best_a.copy()
    cur_score = best_score

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        iteration += 1

        # Linear learning‑rate decay
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)

        # Convolution (full) and exact score
        conv = _conv_self_fft(cur_a, fft_len)
        cur_score = float(2.0 * n * conv.max())

        # Record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s it={iteration}] new best score = {best_score:.6f}")

        grad = _gradient_max_conv(cur_a, conv)

        # Exponentiated‑gradient update (mirror descent on negative entropy)
        factor = np.exp(-eta * grad)
        cur_a *= factor
        cur_a = np.clip(cur_a, 0.0, None)
        s = cur_a.sum()
        if s == 0.0:
            cur_a = np.full_like(cur_a, 1.0 / n)
        else:
            cur_a /= s

        # -------------------------------------------------------------
        # Random exploration moves (light perturbations)
        # -------------------------------------------------------------
        if rng.random() < block_prob:
            # pick a random block and scale it
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.2))
            cur_a[start:start + width] *= mult
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a /= cur_a.sum()

        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > 1e-14:
                delta = rng.random() * cur_a[i] * 0.5
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = np.clip(cur_a, 0.0, None)
                cur_a /= cur_a.sum()

        # soft restart when stagnating

            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            cur_a = best_a * jitter
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a /= cur_a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s it={iteration}] jitter restart – best score {best_score:.6f}")

    # Return the best vector (already normalised, safe for evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
