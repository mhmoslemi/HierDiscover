    deadline = start + budget_s - 0.1

    best_seq = None
    best_score = np.inf

    for n in lengths:
        # initialize sequence
        a = initialize_sequence(n)

        fft_len = _fft_len_for_n(n)

        # gradient descent with exponentiated gradient (subgradient/hard and soft)
        # parameters...
        last_improve = time.monotonic()
        while time.monotonic() < deadline:
            # compute convolution
            conv = _conv_self(a, fft_len)
            cur_max = conv.max()
            cur_score = 2.0 * n * cur_max

            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = time.monotonic()
                print(f"[{time.monotonic() - start:.1f}s] n={n} new best {best_score:.6f}")

            # hard weight (mask)
            mask = conv == cur_max
            # handle tolerance for floating equality (maybe use np.isclose)
            mask = np.isclose(conv, cur_max, atol=1e-12)
            w = mask.astype(float)
            if w.sum() > 0:
                w = w / w.sum()
            else:
                w = np.ones_like(conv) / conv.size

            grad = _gradient_correlation(a, w)

            # schedule eta
            elapsed = time.monotonic() - start
            ratio = min(1.0, elapsed / budget_s)
            eta = 0.6 * (1 - ratio) + 0.005 * ratio

            # exponentiated gradient update
            # update in log space to avoid underflow
            log_a = np.log(a + 1e-30)
            log_a -= eta * grad
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum == 0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / a_sum

            # random perturbations every few steps
            if rng.random() < 0.02:
                # multiply by log-normal factor
                factor = np.exp(rng.normal(0.0, 0.2, size=n))
                a = a * factor
                a = a / a.sum()

            # occasional restart if no improvement
            if time.monotonic() - last_improve > 120.0:
                # reinitialize around best
                a = best_seq.copy()

Algorithm – multiplicative (exponentiated) gradient on a smooth surrogate

* The vector is kept on the probability simplex (∑a_i = 1).

  computed efficiently with FFTs.

  This respects the simplex constraints and often yields better progress

* If no improvement has been seen for ``restart_gap`` seconds the current

def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two ≥ 2·n‑1 for use with FFT."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length = 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute ∇_a Σ_k w_k (a∗a)_k = 2·a ⋆ w   (correlation).
    ``a`` length n, ``w`` length 2·n‑1.
    """
    n = a.size
    L = w.size                     # = 2·n‑1
    # length needed for linear correlation = L + n – 1 = 3·n – 2
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w = np.full_like(v, 1.0 / len(v))
    return w


# ----------------------------------------------------------------------
# Initialise a “sqrt‑difference’’ sequence (the previously known good start)
# ----------------------------------------------------------------------
def _initial_sqrt_seq(n: int) -> np.ndarray:
    """Return the normalised sqrt‑difference sequence of length n."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                      budget_s: float = 1000.0,
                      **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Time budget in seconds.
    n : int, optional
        Desired length of the sequence.  If omitted the routine tries several
        lengths (including the length of a previously discovered best
        construction).

    Returns
    -------
    list[float]
        The best sequence found (values clipped to the limits imposed by
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Decide on a set of lengths to explore.
    # ------------------------------------------------------------------
    lengths = []
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
            lengths.append(int(cand.size))
    # a few deterministic choices spanning a reasonable range
    lengths.extend([1800, 2800, 3800, 4800, 5800])
    lengths = sorted(set(lengths))

    best_seq = None
    best_score = np.inf

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # ------------------------------------------------------------------
    # Helper: compute the score of a normalised vector a (sum(a)=1)
    # ------------------------------------------------------------------
    def _score(a: np.ndarray) -> float:
        n = a.size
        fft_len = _fft_len_for_n(n)
        conv = _conv_self(a, fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Main loop over the chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # --------------------------------------------------------------
        # Initialise a.
        # --------------------------------------------------------------
        if best_seq is not None and best_seq.size == n:
            a = best_seq.copy()
        else:
            a = _initial_sqrt_seq(n)

        fft_len = _fft_len_for_n(n)

        # --------------------------------------------------------------
        # Optimisation state
        # --------------------------------------------------------------
        cur_score = _score(a)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{0.0:6.2f}s] init length {n} score {best_score:.6f}")

        # Adam‑like momentums are *not* used – we rely on the multiplicative
        # update which is naturally suited to the simplex.
        last_improve = time.monotonic()
        restart_gap = 150.0   # seconds without improvement before a soft restart

        # Schedules (linear in elapsed time)
        beta_start = 0.5
        beta_end = 3000.0
        lr_start = 0.9
        lr_end = 0.01

        # Random‑perturbation probabilities
        noise_prob = 0.02
        swap_prob = 0.015

        # --------------------------------------------------------------
        # Optimisation loop for this length
        # --------------------------------------------------------------
        while time.monotonic() < deadline:
            now = time.monotonic()
            elapsed = now - start_time
            ratio = min(1.0, elapsed / budget_s)

            # ---- dynamic hyper‑parameters --------------------------------
            beta = beta_start + (beta_end - beta_start) * ratio
            lr = lr_start * (1.0 - ratio) + lr_end * ratio   # learning rate η

            # ---- convolution and soft‑max weights ------------------------
            conv = _conv_self(a, fft_len)                     # length 2·n‑1
            cur_max = float(conv.max())
            cur_score = 2.0 * n * cur_max

            # ---- keep the global best ------------------------------------
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] length {n} new best score = {best_score:.6f}")

            # ---- soft‑max weights (stable) --------------------------------
            diff = conv - cur_max               # ≤ 0
            # exponent may under‑flow – that's fine
            w_raw = np.exp(beta * diff)
            w_sum = w_raw.sum()
            if w_sum == 0.0 or not np.isfinite(w_sum):
                # fallback: uniform weight on all maximal indices
                mask = conv == cur_max
                w_raw = np.where(mask, 1.0, 0.0)
                w_sum = w_raw.sum()
            w = w_raw / w_sum

            # ---- gradient of the smooth surrogate -------------------------
            grad = _gradient_correlation(a, w)

            # ---- exponentiated‑gradient step -------------------------------
            # work in log‑space for numerical stability
            eps = 1e-30
            log_a = np.log(a + eps)
            log_a -= lr * grad
            a = np.exp(log_a)
            a_sum = a.sum()
            if a_sum == 0.0 or not np.isfinite(a_sum):
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a /= a_sum

            # ---- occasional random multiplicative noise -------------------
            if rng.random() < noise_prob:
                factor = np.exp(rng.normal(0.0, 0.2, size=n))
                a = a * factor
                a /= a.sum()

            # ---- occasional mass‑transfer between two coordinates ----------
            if rng.random() < swap_prob:
                i, j = rng.integers(0, n, size=2)
                if i != j and a[i] > 1e-15:
                    delta = rng.random() * a[i] * 0.5
                    a[i] -= delta
                    a[j] += delta
                    a /= a.sum()

            # ---- soft restart if we have been stuck for a while ----------
            if now - last_improve > restart_gap:
                if best_seq is not None:
                    a = best_seq.copy()
                else:
                    a = _initial_sqrt_seq(n)
                jitter = np.exp(rng.normal(0.0, 0.12, size=n))
                a = a * jitter
                a /= a.sum()
                last_improve = now
                print(f"[{elapsed:6.2f}s] length {n} restart – best score {best_score:.6f}")

        # end while for this length
    # ------------------------------------------------------------------
    # Return the best sequence (clipped as required by ``evaluate_sequence``)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of a modest size
        n = kwargs.get("n", 2000)
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
