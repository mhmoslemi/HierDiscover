def _fft_len_for_n(n):
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    fa2 = fa * fa
    conv = np.fft.irfft(fa2, fft_len)[: 2 * a.size - 1]
    return conv

def _softmax_weights(conv, alpha):
    # Compute softmax of alpha*conv in numerically stable way
    # shift by max(alpha*conv) = alpha*conv_max
    conv_max = conv.max()
    # Avoid overflow: subtract conv_max
    exps = np.exp(alpha * (conv - conv_max))
    w = exps / exps.sum()
    return w

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\;\max\bigl(a*a\bigr)}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where `*` is the ordinary (full) convolution.  
Because the objective is *scale invariant* we may enforce  
\(\sum_i a_i = 1\); then  

\[
F(a)=2\,n\;\max\bigl(a*a\bigr).
\]

The problem is non‑convex (the convolution terms are quadratic) but the
maximum can be approximated by the **log‑sum‑exp** (soft‑max)  

\[
\phi_\alpha(a)=\frac{1}{\alpha}\log\Bigl(\sum_k e^{\alpha\,(a*a)_k}\Bigr)
\quad\Longrightarrow\quad
\lim_{\alpha\to\infty}\phi_\alpha(a)=\max_k (a*a)_k .
\]

`φ_α` is smooth; its gradient is  

\[
\nabla\phi_\alpha(a)=2\;a\star w,\qquad    
w_k=\frac{e^{\alpha\,(a*a)_k}}{\sum_j e^{\alpha\,(a*a)_j}},
\]

i.e. a correlation of `a` with the soft‑max weights `w`.  
We optimise the surrogate with **Adam** on the logits  
\(z_i=\log a_i\) (mirrored‑descent on the probability simplex).  
`α` is increased linearly from a modest value (≈5) to a large one
(≈ 5000) so that the optimisation gradually focuses on the true
maximum.

After the Adam phase a short **simulated‑annealing** polishing step
does cheap pairwise mass‑transfer moves and a periodic smoothing move.
Both phases keep the vector normalised to a probability distribution,
so the evaluated value is exactly the same as `evaluate_sequence`.

The whole routine respects a 1000 s wall‑time budget,
keeps the best sequence seen, prints progress, and finally returns the
clipped vector (0 ≤ v ≤ 1000) as a plain list of floats.

    size = 2 * n - 1

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a via FFT; returns length 2·len(a)‑1 array."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _normalize(v: np.ndarray) -> np.ndarray:
    """Project non‑negative vector onto the probability simplex (sum = 1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Numerically stable soft‑max of α·conv.
    Returns a probability vector w (sum = 1) of the same shape as conv.
    """
    # shift by the maximum to avoid overflow
    shift = conv.max()
    exps = np.exp(alpha * (conv - shift))
    w = exps / exps.sum()
    return w


def _grad_of_softmax_obj(a: np.ndarray, alpha: float, fft_len: int):
    """
    Gradient of the log‑sum‑exp surrogate (α fixed) w.r.t. a,
    together with the current convolution.
    Returns (grad, conv) where grad has shape a.shape.
    """

    w = _softmax_weights(conv, alpha)           # soft‑max of α·conv

    # correlation a ⋆ w  (gradient of Σ w_k (a*a)_k)
    n = a.size
    L = w.size

    return grad, conv
