"""
Gradient‑descent + soft‑max smoothing search for a low‑max convolution step‑function.

The algorithm normalises the candidate to sum = 1 and iteratively
minimises a smooth approximation of  maxₖ (a∗a)ₖ  using an
exponentiated‑gradient (mirror‑descent) update on the probability simplex.
The temperature (soft‑max β) is increased from a small to a large value,
while the learning‑rate η is decreased, allowing the method to first
smoothly explore the landscape and then focus on the true L∞ objective.
A short hill‑climbing phase with cheap pairwise mass‑transfer moves
finishes the search.  The routine respects the 1000 s time budget and
returns the best vector found.
"""

import time
import numpy as np

def _fft_len_for_n(n: int) -> int:
    """Smallest power of two length that can hold the linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT, result length 2*len(a)-1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k.
    Implements the correlation a ⋆ w (see the original code).
    Returns a vector of length len(a).
    """
    n = a.size
    L = w.size                     # should be 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a_sum = a.sum()
    if a_sum == 0:
        return np.full(n, 1.0 / n)
    return a / a_sum

def _maybe_use_global() -> np.ndarray | None:
    """If a previously found construction exists in the global namespace, return it normalised."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget elapses.

    The search consists of a smooth‑max gradient‑descent phase followed
    by a short hill‑climbing phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (default 3800)
    # ------------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Initial vector: use stored construction if available, else analytic.
    # ------------------------------------------------------------------
    a = _maybe_use_global()
    if a is None or a.size != n:
        a = _analytic_start(n)

    # Ensure normalisation
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    conv = _conv_self(a, fft_len)
    best_score = 2.0 * n * float(conv.max())
    best = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (exponentiated‑gradient)
    # ------------------------------------------------------------------
    # Hyper‑parameters (these have been tuned experimentally)
    beta_start = 2.0          # low temperature (smooth)
    beta_end = 120.0          # high temperature (approximates true max)
    eta_start = 0.45          # relatively large step for early iteration
    eta_end = 0.008           # small step for fine tuning
    max_exp_clip = 50.0       # for numerical stability in exp

    # Main optimisation loop
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start_time
        total = deadline - start_time
        progress = elapsed / total   # in [0,1]

        # Linear schedules for beta and eta
        beta = beta_start + (beta_end - beta_start) * progress
        eta = eta_start + (eta_end - eta_start) * (1.0 - progress)

        # Compute convolution and the soft‑max weights
        conv = _conv_self(a, fft_len)
        max_conv = float(conv.max())
        # soft‑max (weights sum to 1, numerically stable)
        shifted = conv - max_conv
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # Gradient of the smoothed objective
        grad = _gradient_correlation(a, w)   # shape (n,)

        # Exponentiated‑gradient update on the simplex
        exp_term = -eta * grad
        exp_term = np.clip(exp_term, -max_exp_clip, max_exp_clip)
        a = a * np.exp(exp_term)

        # Projection onto the simplex (non‑negative and sum = 1)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # Track best (true L∞) score
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best (gd) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short hill‑climbing (pairwise mass transfer)
    # ------------------------------------------------------------------
    # Use any remaining time (if any) for cheap local moves.
    hill_deadline = start_time + float(budget_s) - 0.05   # leave a tiny margin
    while time.monotonic() < hill_deadline:
        # Propose a simple pairwise mass transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if best[i] <= 1e-15:
            continue
        # Transfer up to 30 % of the source mass
        delta = rng.random() * best[i] * 0.3
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # Renormalise (should already sum to 1, but guard against rounding)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand = np.full(n, 1.0 / n)
        else:
            cand = cand / s

        # Evaluate
        conv_cand = _conv_self(cand, fft_len)
        cand_score = 2.0 * n * float(conv_cand.max())
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand
            now = time.monotonic()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best (hill) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # Return as a plain Python list of floats
    return [float(v) for v in best]

"""
Gradient‑descent + soft‑max smoothing search for a low‑max convolution
step‑function.

The algorithm normalises the candidate to sum = 1 and iteratively
minimises a smooth approximation of  maxₖ (a∗a)ₖ  using an
exponentiated‑gradient (mirror‑descent) update on the probability simplex.
The temperature (soft‑max β) is increased from a small to a large value,
while the learning‑rate η is decreased, allowing the method to first
smoothly explore the landscape and then focus on the true L∞ objective.
A short hill‑climbing phase with cheap pairwise mass‑transfer moves
finishes the search.  The routine respects the 1000 s time budget and
returns the best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via real FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k.
    Implements the correlation a ⋆ w and returns a vector of length len(a).
    """
    n = a.size
    L = w.size                     # should be 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return a / s


def _maybe_use_global() -> np.ndarray | None:
    """If a previous construction exists in the global namespace, return it normalised."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget elapses.

    The search consists of a smooth‑max gradient‑descent phase followed
    by a short hill‑climbing phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (default 3800)
    # ------------------------------------------------------------------
    n_default = int(kwargs.get("n", 3800))
    if n_default < 2:
        n_default = 2

    # ------------------------------------------------------------------
    # Prepare two candidate starts: analytic and a possible stored one
    # ------------------------------------------------------------------
    a_analytic = _analytic_start(n_default)
    fft_len_analytic = _fft_len_for_n(n_default)
    conv_analytic = _conv_self(a_analytic, fft_len_analytic)
    best_score = 2.0 * n_default * float(conv_analytic.max())
    best = a_analytic.copy()
    n = n_default
    fft_len = fft_len_analytic

    cand_global = _maybe_use_global()
    if cand_global is not None:
        # Evaluate the global candidate (its own length)
        cand_norm = np.clip(cand_global, 0.0, None)
        s = cand_norm.sum()
        if s > 0.0:
            cand_norm = cand_norm / s
            n_cand = cand_norm.size
            fft_len_cand = _fft_len_for_n(n_cand)
            conv_cand = _conv_self(cand_norm, fft_len_cand)
            cand_score = 2.0 * n_cand * float(conv_cand.max())
            if cand_score < best_score:
                best = cand_norm.copy()
                best_score = cand_score
                n = n_cand
                fft_len = fft_len_cand

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    a = best.copy()                     # current iterate
    conv = _conv_self(a, fft_len)       # convolution of current iterate

    start_time = time.monotonic()
    # keep a tiny safety margin for the final return
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent (exponentiated‑gradient)
    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned empirically)
    beta_start = 2.0          # start with a smooth approximation
    beta_end = 120.0          # end approximating the true L∞ norm
    eta_start = 0.45          # large step at the beginning
    eta_end = 0.008           # fine‑grained step at the end
    max_exp_clip = 50.0       # prevents overflow in exp()

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # Linear interpolation of schedules
        elapsed = now - start_time
        total = deadline - start_time
        progress = elapsed / total if total > 0 else 1.0
        beta = beta_start + (beta_end - beta_start) * progress
        eta = eta_start + (eta_end - eta_start) * (1.0 - progress)

        # --- compute soft‑max weights from the current convolution ---
        max_conv = float(conv.max())
        shifted = conv - max_conv                # numerical stabilisation
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # --- gradient of the smoothed objective ---
        grad = _gradient_correlation(a, w)

        # --- exponentiated‑gradient update on the simplex ---
        exp_term = -eta * grad
        exp_term = np.clip(exp_term, -max_exp_clip, max_exp_clip)
        a = a * np.exp(exp_term)

        # Project back onto the probability simplex
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # --- evaluate the new iterate ---
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * n * float(conv.max())
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best (gd) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short hill‑climbing with cheap pairwise mass transfers
    # ------------------------------------------------------------------
    hill_deadline = start_time + float(budget_s) - 0.05   # tiny final margin
    while time.monotonic() < hill_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j or best[i] <= 1e-15:
            continue
        # transfer up to 30 % of the source mass
        delta = rng.random() * best[i] * 0.3
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand = np.full(n, 1.0 / n)
        else:
            cand = cand / s

        conv_cand = _conv_self(cand, fft_len)
        cand_score = 2.0 * n * float(conv_cand.max())
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand
            now = time.monotonic()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best (hill) score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
