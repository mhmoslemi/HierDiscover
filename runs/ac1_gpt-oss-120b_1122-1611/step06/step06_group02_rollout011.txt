def _local_optimise_adam(initial, rng, deadline, max_iters=2000):
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2*n-1)
    # log domain
    x = np.log(np.maximum(a, 1e-30))
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.1
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_score = float('inf')
    best_vec = a.copy()
    for t in range(1, max_iters+1):
        now = time.time()
        if now >= deadline: break
        a = np.exp(x)
        a /= a.sum()
        # compute conv, score
        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
        # gradient
        if rng.random() < 0.6:
            grad_a = _subgradient_step(a, conv)
        else:
            beta_temp = min(1e5, 0.5 * (1.08**t)) # not necessary
            grad_a = _softmax_gradient(a, conv, beta_temp, _fft_len_for_len(3*n-2))
        # convert to log gradient
        grad_x = grad_a * a
        # Adam update
        m = beta1 * m + (1-beta1) * grad_x
        v = beta2 * v + (1-beta2) * (grad_x**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # periodic random perturbations
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            x = np.log(np.maximum(a, 1e-30))
        if rng.random() < 0.02:
            a = _pairwise_move(a, rng, fft_len_conv)
            x = np.log(np.maximum(a, 1e-30))
    # final best
    final_score = _score(best_vec, _fft_len_for_len(2*best_vec.size-1))
    return best_vec, final_score

import time, numpy as np
from scipy import signal

def _fft_len_for_len(length):
    return 1 << ((length-1).bit_length())

def _conv_self(a, fft_len):
    fa=np.fft.rfft(a, n=fft_len)
    conv=np.fft.irfft(fa*fa, n=fft_len)
    return conv[:2*a.size-1]

def _conv_fft(x,y,fft_len):
    fx=np.fft.rfft(x,n=fft_len)
    fy=np.fft.rfft(y,n=fft_len)
    conv=np.fft.irfft(fx*fy,n=fft_len)
    return conv[:len(x)+len(y)-1]

def _subgradient_step(a, conv):
    n=a.size
    i_max=int(conv.argmax())
    lo=max(0,i_max-(n-1))
    hi=min(i_max,n-1)
    idx=np.arange(lo,hi+1)
    grad=np.zeros_like(a)
    grad[idx]=2.0 * a[i_max-idx]
    return grad

def _softmax_gradient(a, conv, beta, fft_len_grad):
    cur_max=conv.max()
    w=np.exp(beta*(conv-cur_max))
    sum_w=w.sum()
    a_rev=a[::-1]
    conv_w_rev=_conv_fft(w,a_rev,fft_len_grad)
    grad_part=conv_w_rev[a.size-1:a.size-1+a.size]/sum_w
    return grad_part

def _random_block_perturb(a, rng):
    n=a.size
    width=rng.integers(1,max(2,n//30))
    start=rng.integers(0,n-width+1)
    factor=np.exp(rng.normal(0.0,0.2))
    out=a.copy()
    out[start:start+width]*=factor
    out=np.clip(out,0.0,None)
    out/=out.sum()
    return out

def _pairwise_move(a, rng, fft_len_conv):
    n=a.size
    i,j=rng.integers(0,n,size=2)
    if i==j or a[i]==0.0:
        return a
    max_delta=min(a[i],0.2*a[i])
    if max_delta<=0.0:
        return a
    delta=rng.random()*max_delta
    a2=a.copy()
    a2[i]-=delta
    a2[j]+=delta
    a2/=a2.sum()
    conv_old=_conv_self(a,fft_len_conv)
    conv_new=_conv_self(a2,fft_len_conv)
    if conv_new.max()<conv_old.max():
        return a2
    return a

def _score(a, fft_len_conv):
    n=a.size
    conv=_conv_self(a,fft_len_conv)
    return 2.0*n*conv.max()

def _init_random_dirichlet(rng,n):
    a=rng.exponential(scale=1.0,size=n)
    a/=a.sum()
    return a

def _init_sine_power(rng,n):
    p=rng.uniform(0.5,4.0)
    idx=np.arange(n,dtype=np.float64)
    vals=np.sin(np.pi*(idx+0.5)/n)
    vals=np.maximum(vals,0.0)
    a=np.power(vals,p)
    if a.sum()==0.0:
        a=np.full(n,1.0/n)
    else:
        a/=a.sum()
    return a

def _init_gaussian(rng,n):
    sigma=rng.uniform(0.05,0.5)*n
    idx=np.arange(n) - (n-1)/2.0
    a=np.exp(-0.5*(idx/sigma)**2)
    a_sum=a.sum()
    if a_sum==0.0:
        a=np.full(n,1.0/n)
    else:
        a/=a_sum
    return a

def _init_exponential(rng,n):
    rate=rng.uniform(0.001,0.05)
    idx=np.arange(n)
    a=np.exp(-rate*idx)

def _local_optimise(initial, rng, deadline, max_iters=1500):
    a=initial.copy()
    n=a.size
    fft_len_conv=_fft_len_for_len(2*n-1)
    fft_len_grad=_fft_len_for_len(3*n-2)
    best_vec=a.copy()
    best_score=_score(a,fft_len_conv)

    beta=0.5
    beta_mul=1.08

    for t in range(1, max_iters+1):
        now=time.time()
        if now>=deadline:
            break
        conv=_conv_self(a,fft_len_conv)
        cur_score=2.0*n*conv.max()
        if cur_score<best_score:
            best_score=cur_score
            best_vec=a.copy()
        # gradient choice
        if rng.random()<0.8:
            raw_grad=_subgradient_step(a,conv)
        else:
            beta=min(1e5, beta*beta_mul)
            raw_grad=_softmax_gradient(a,conv,beta,fft_len_grad)
        # exponentiated gradient step
        lr=0.7/np.sqrt(t)
        grad_clip=np.clip(raw_grad,-20.0,20.0)
        a=a*np.exp(-lr*grad_clip)
        total=a.sum()
        if total==0.0:
            a=np.full_like(a,1.0/n)
        else:
            a/=total
        # random perturbations
        if rng.random()<0.03:
            a=_random_block_perturb(a,rng)
        if rng.random()<0.03:
            a=_pairwise_move(a,rng,fft_len_conv)
    # final
    final_score=_score(best_vec,_fft_len_for_len(2*best_vec.size-1))
    return best_vec, final_score

def _local_optimise_adam(initial, rng, deadline, max_iters=2500):
    # Adam on log-probabilities
    n=initial.size
    fft_len_conv=_fft_len_for_len(2*n-1)
    fft_len_grad=_fft_len_for_len(3*n-2)
    # log variables
    eps=1e-30
    x=np.log(np.maximum(initial,eps))
    m=np.zeros_like(x)
    v=np.zeros_like(x)
    beta1=0.9
    beta2=0.999
    lr=0.05
    best_vec=np.exp(x)
    best_vec/=best_vec.sum()
    best_score=_score(best_vec,fft_len_conv)
    for t in range(1, max_iters+1):
        now=time.time()
        if now>=deadline:
            break
        a=np.exp(x)
        a_sum=a.sum()
        if a_sum==0.0:
            a=np.full(n,1.0/n)
        else:
            a/=a_sum
        conv=_conv_self(a,fft_len_conv)
        cur_score=2.0*n*conv.max()
        if cur_score<best_score:
            best_score=cur_score
            best_vec=a.copy()
        # gradient selection
        if rng.random()<0.75:
            grad_a=_subgradient_step(a,conv)
        else:
            beta=0.5*(1.08**t)
            beta=min(beta,1e5)
            grad_a=_softmax_gradient(a,conv,beta,fft_len_grad)
        # convert to log space
        grad_x=grad_a*a
        m=beta1*m+(1-beta1)*grad_x
        v=beta2*v+(1-beta2)*(grad_x**2)
        m_hat=m/(1-beta1**t)
        v_hat=v/(1-beta2**t)
        x-=lr*m_hat/(np.sqrt(v_hat)+1e-8)
        # occasional perturbations
        if rng.random()<0.02:
            a=_random_block_perturb(a,rng)
            x=np.log(np.maximum(a,eps))
        if rng.random()<0.02:
            a=_pairwise_move(a,rng,fft_len_conv)
            x=np.log(np.maximum(a,eps))
    final_score=_score(best_vec,_fft_len_for_len(2*best_vec.size-1))
    return best_vec, final_score

"""
Propose a step‑function (non‑negative heights) that minimises the
``evaluate_sequence`` score.

Strategy
--------
* A pool of diverse initialisers is used:
    – random Dirichlet,
    – sine‑power,
    – Gaussian,
    – exponential decay,
    – classic signal‑processing windows (Kaiser, Chebyshev,
      Hann, Hamming, Blackman).  If SciPy is unavailable the window
      families silently fall back to a Dirichlet start.
    – linear interpolation of the current best.
* Every start is refined by a fast exponentiated‑gradient (EG) descent
  that uses the exact sub‑gradient of the worst convolution entry most
  of the time (≈80 %).  Random block perturbations and cheap pairwise
  mass‑transfer moves help to escape plateaus.
* The current global best is periodically *polished* by an Adam‑style
  mirror‑descent on the log‑probabilities (i.e. natural gradient on the
  probability simplex).  This second‑order‑like stage uses a smooth
  “log‑sum‑exp’’ surrogate for the max during the early part of the
  run and switches to the exact sub‑gradient later.
* All work respects the provided wall‑clock budget (default 1000 s).

The routine returns a plain list of Python ``float`` values that can be
directly fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – window families are convenient but not mandatory.
# ----------------------------------------------------------------------
try:
    from scipy import signal
except Exception:  # pragma: no cover
    signal = None

# ----------------------------------------------------------------------
# FFT helpers (linear convolution via real FFT)
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length (sufficient for linear convolution)."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear self‑convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors ``x`` and ``y`` using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities (sub‑gradient & smooth max)
# ----------------------------------------------------------------------


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) w.r.t. ``a``.
    The worst index i_max contributes 2·a_{i_max‑j} for all admissible j.
    """
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i).
    Returns ∂f_β/∂a (no extra “+a” term).
    """
    cur_max = conv.max()
    w = np.exp(beta * (conv - cur_max))          # safe, max exponent = 0
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length = 3·n‑2
    # extract the n entries that correspond to derivative wrt a
    grad_part = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad_part


# ----------------------------------------------------------------------
# Simple random moves (used as occasional perturbations)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a small random fraction of mass from i to j (accept only if improving)."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a


# ----------------------------------------------------------------------
# Fast score helper (identical to what evaluate_sequence returns for a unit‑sum vector)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return the value 2·n·max(conv(a,a)) for a simplex vector."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()

# Initialisation families
# ----------------------------------------------------------------------


def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponentials."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """a_i ∝ sin^p(π·(i+0.5)/n) with random exponent p∈[0.5,4]."""
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    vals = np.maximum(vals, 0.0)
    a = np.power(vals, p)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_gaussian(rng: np.random.Generator, n: int) -> np.ndarray:
    """Centered Gaussian window with a random width."""
    sigma = rng.uniform(0.05, 0.5) * n
    idx = np.arange(n) - (n - 1) / 2.0
    a = np.exp(-0.5 * (idx / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _init_exponential(rng: np.random.Generator, n: int) -> np.ndarray:
    """Monotonically decaying exponential shape."""
    rate = rng.uniform(0.001, 0.05)
    idx = np.arange(n)
    a = np.exp(-rate * idx)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linearly interpolate an existing distribution to a different length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Local optimisation – exponentiated gradient (fast)
# ----------------------------------------------------------------------


def _local_optimise(initial: np.ndarray,
                    rng: np.random.Generator,
                    deadline: float,
                    max_iters: int = 1500) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using a multiplicative (EG)
    update.  The exact sub‑gradient of the worst convolution entry is used
    most of the time; a soft‑max surrogate is occasionally employed to
    provide a smoother descent direction.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    beta = 0.5
    beta_mul = 1.08

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # Gradient choice
        # --------------------------------------------------------------
        if rng.random() < 0.80:               # sub‑gradient most of the time
            raw_grad = _subgradient_step(a, conv)
        else:                                 # smooth max with growing temperature
            beta = min(1e5, beta * beta_mul)
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # Exponentiated‑gradient update (multiplicative)
        # --------------------------------------------------------------
        lr = 0.7 / np.sqrt(t)                 # 1/√t schedule
        grad_clip = np.clip(raw_grad, -20.0, 20.0)
        a = a * np.exp(-lr * grad_clip)

        total = a.sum()
        if total == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= total

        # --------------------------------------------------------------
        # Occasional random exploratory moves
        # --------------------------------------------------------------
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng)
        if rng.random() < 0.03:
            a = _pairwise_move(a, rng, fft_len_conv)

    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score


# ----------------------------------------------------------------------
# Local optimisation – Adam on the log‑probabilities (polishing)
# ----------------------------------------------------------------------


def _local_optimise_adam(initial: np.ndarray,
                         rng: np.random.Generator,
                         deadline: float,
                         max_iters: int = 2500) -> tuple[np.ndarray, float]:
    """
    Adam‑style mirror descent on the log‑probabilities.
    The update respects the simplex via the exponential map.
    """
    n = initial.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    eps = 1e-30
    x = np.log(np.maximum(initial, eps))       # log‑weights

    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    lr = 0.05

    best_vec = np.exp(x)
    best_vec /= best_vec.sum()
    best_score = _score(best_vec, fft_len_conv)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = np.exp(x)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= a_sum

        conv = _conv_self(a, fft_len_conv)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # --------------------------------------------------------------
        # Gradient selection (mostly exact sub‑gradient)
        # --------------------------------------------------------------
        if rng.random() < 0.75:
            grad_a = _subgradient_step(a, conv)
        else:
            # soft‑max gradient with a slowly increasing temperature
            beta = min(1e5, 0.5 * (1.08 ** t))
            grad_a = _softmax_gradient(a, conv, beta, fft_len_grad)

        # --------------------------------------------------------------
        # Transform to log‑space and apply Adam update
        # --------------------------------------------------------------
        grad_x = grad_a * a                     # ∂L/∂x_i = ∂L/∂a_i · a_i
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # --------------------------------------------------------------
        # Light random perturbations to avoid stagnation
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            x = np.log(np.maximum(a, eps))
        if rng.random() < 0.02:
            a = _pairwise_move(a, rng, fft_len_conv)
            x = np.log(np.maximum(a, eps))

    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score


# ----------------------------------------------------------------------
# Main entry point for the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # reserve a small safety margin for harness bookkeeping
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Global best container
    # --------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Load any previously known construction
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / s
                cand_score = _score(cand, _fft_len_for_len(2 * cand.size - 1))
                best_vec = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start:5.2f}s] Loaded previous best: {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # Lengths to explore (biased towards medium/large sizes)
    # --------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 3809, 4000,
               4500, 5000, 6000, 8000, 10000]

    # --------------------------------------------------------------
    # Initialisation method weights (must sum to 1)
    # --------------------------------------------------------------
    init_methods = [
        'dirichlet', 'sine', 'gaussian', 'exponential',
        'kaiser', 'cheb', 'hann', 'hamming', 'blackman', 'interpolate'
    ]
    init_weights = np.array([0.20, 0.12, 0.08, 0.08,
                             0.07, 0.07, 0.07, 0.07, 0.07, 0.15])
    init_weights = init_weights / init_weights.sum()

    # --------------------------------------------------------------
    # Main stochastic search loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        # Choose a length (uniform over the list – could be made smarter)
        n = int(rng.choice(lengths, p=np.ones(len(lengths)) / len(lengths)))
        method = rng.choice(init_methods, p=init_weights)

        # ----------------------------------------------------------
        # Generate a starting point
        # ----------------------------------------------------------
        if method == 'dirichlet':
            a0 = _init_random_dirichlet(rng, n)
        elif method == 'sine':
            a0 = _init_sine_power(rng, n)
        elif method == 'gaussian':
            a0 = _init_gaussian(rng, n)
        elif method == 'exponential':
            a0 = _init_exponential(rng, n)
        elif method == 'kaiser':
            a0 = _init_kaiser(rng, n)
        elif method == 'cheb':
            a0 = _init_chebwin(rng, n)
        elif method == 'hann':
            a0 = _init_hann(rng, n)
        elif method == 'hamming':
            a0 = _init_hamming(rng, n)
        elif method == 'blackman':
            a0 = _init_blackman(rng, n)
        else:  # interpolate from the current best
            if best_vec is not None:
                a0 = _init_interpolate(rng, best_vec, n)
            else:
                a0 = _init_random_dirichlet(rng, n)

        # ----------------------------------------------------------
        # Quick refinement with exponentiated‑gradient descent
        # ----------------------------------------------------------
        iters = int(1800 * (5000.0 / max(n, 500)))   # ≈1800 when n≈5000
        a_opt, score_opt = _local_optimise(a0, rng, deadline, max_iters=iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:5.2f}s] New global best (EG): {best_score:.6f} (n={n})")

        # ----------------------------------------------------------
        # Periodic polishing of the current best with Adam
        # ----------------------------------------------------------
        if best_vec is not None and time.time() < deadline - 2.0:
            # allocate a modest amount of iterations – scaled with size
            polish_iters = int(4000 * (5000.0 / max(best_vec.size, 500)))
            a_polish, score_polish = _local_optimise_adam(
                best_vec.copy(), rng, deadline, max_iters=polish_iters)
            if score_polish < best_score:
                best_score = score_polish
                best_vec = a_polish.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:5.2f}s] Polished best (Adam): {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 3000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
