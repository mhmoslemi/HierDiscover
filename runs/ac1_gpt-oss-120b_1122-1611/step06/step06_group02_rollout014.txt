   fft_len_grad = _fft_len_for_len(3 * n - 2)

   z = np.log(a + 1e-12)  # small epsilon

   m = np.zeros_like(z)
   v = np.zeros_like(z)
   beta1 = 0.9
   beta2 = 0.999
   eps = 1e-8

   beta = 0.5
   beta_mul = 1.02

   best_vec = a.copy()
   best_score = _score(a, fft_len_conv)
   step = 1

   for t in range(1, max_iters+1):
        now = time.time()
        if now >= deadline:
            break

        # softmax to get simplex a
        max_z = np.max(z)  # for numerical stability
        exp_z = np.exp(z - max_z)  # shape
        a = exp_z / exp_z.sum()

        # compute convolution
        conv = _conv_self(a, fft_len_conv)

        # compute smoothed gradient
        raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # compute grad w.r.t z via chain rule
        g_z = a * (raw_grad - np.dot(raw_grad, a))

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g_z
        v = beta2 * v + (1.0 - beta2) * (g_z ** 2)
        m_hat = m / (1 - beta1 ** step)
        v_hat = v / (1 - beta2 ** step)

        lr = 0.5 / (np.sqrt(step))   # decreasing LR
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # optionally random perturbation

"""
A fast stochastic optimiser for the “max self‑convolution” problem.

The routine works on the probability simplex (∑a = 1, a_i ≥ 0) and
optimises a smooth proxy of   max_i (a*a)_i   – the log‑sum‑exp of the
full self‑convolution – using an Adam‑style mirror‑descent on the
log‑variables (a_i = softmax(z_i)).  Several families of initial
distributions (Dirichlet, sine‑power, Beta, exponential decay and
interpolation of the current best) are tried, each run is limited only
by the remaining time budget.  The best vector found is periodically
polished with a longer Adam run.  The final sequence is clipped only
to satisfy the sanity checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers – linear convolution via real FFT
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """Smallest power‑of‑two ≥ length."""
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full self–convolution a*a (length 2·len(a)‑1) using a real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of real vectors x and y using a real FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: len(x) + len(y) - 1]


# ----------------------------------------------------------------------
# Gradient utilities
# ----------------------------------------------------------------------


def _softmax_gradient(a: np.ndarray,
                      conv: np.ndarray,
                      beta: float,
                      fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth max f_β(a) = (1/β)·log Σ_i exp(β·conv_i)
    w.r.t. a (the “log‑sum‑exp” surrogate).  Returns a vector of length n.
    """
    cur_max = conv.max()
    # safe exponentiation – the largest exponent becomes 0
    w = np.exp(beta * (conv - cur_max))
    sum_w = w.sum()
    a_rev = a[::-1]
    conv_w_rev = _conv_fft(w, a_rev, fft_len_grad)   # length 3·n‑2
    # the n entries that correspond to ∂/∂a_j
    grad = conv_w_rev[a.size - 1: a.size - 1 + a.size] / sum_w
    return grad


def _subgradient_step(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) w.r.t. a (uses the worst index)."""
    n = a.size
    i_max = int(conv.argmax())
    lo = max(0, i_max - (n - 1))
    hi = min(i_max, n - 1)
    idx = np.arange(lo, hi + 1)
    grad = np.zeros_like(a)
    grad[idx] = 2.0 * a[i_max - idx]
    return grad


# ----------------------------------------------------------------------
# Random moves used for occasional exploration
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_move(a: np.ndarray,
                   rng: np.random.Generator,
                   fft_len_conv: int) -> np.ndarray:
    """Transfer a small fraction of mass from i to j, keep the move if it improves."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    # limit the transferred amount to a modest fraction
    max_delta = min(a[i], 0.2 * a[i])
    if max_delta <= 0.0:
        return a
    delta = rng.random() * max_delta
    a2 = a.copy()
    a2[i] -= delta
    a2[j] += delta
    a2 /= a2.sum()
    # cheap check – if the max of the convolution does not increase, accept
    conv_old = _conv_self(a, fft_len_conv)
    conv_new = _conv_self(a2, fft_len_conv)
    if conv_new.max() < conv_old.max():
        return a2
    return a


# ----------------------------------------------------------------------
# Fast score (exact objective without the overhead of evaluate_sequence)
# ----------------------------------------------------------------------


def _score(a: np.ndarray, fft_len_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) for a vector on the simplex."""
    n = a.size
    conv = _conv_self(a, fft_len_conv)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Initialisation families (all return a point on the simplex)
# ----------------------------------------------------------------------


def _init_random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point on the simplex via exponential variables."""
    a = rng.exponential(scale=1.0, size=n)
    a /= a.sum()
    return a


def _init_sine_power(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    a_i ∝ sin^p(π·(i+0.5)/n)  with exponent p possibly negative.
    Negative p yields a “U‑shaped” distribution with higher mass near the ends.
    """
    p = rng.uniform(-0.8, 4.0)          # allows both concave and convex shapes
    idx = np.arange(n, dtype=np.float64)
    vals = np.sin(np.pi * (idx + 0.5) / n)
    # protect against exact zeros when p<0
    eps = 1e-12
    vals = np.clip(vals, eps, None)
    a = np.power(vals, p)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _init_beta_distribution(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    a_i ∝ x^{p-1} (1‑x)^{q-1}  with p,q sampled in (0.1,3).
    For p,q<1 the shape is U‑shaped (more mass near the boundaries).
    """
    p = rng.uniform(0.1, 3.0)
    q = rng.uniform(0.1, 3.0)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    vals = np.clip(vals, 0.0, None)
    if vals.sum() == 0.0:
        vals = np.full(n, 1.0 / n)
    else:
        vals /= vals.sum()
    return vals


def _init_exp_decay(rng: np.random.Generator, n: int) -> np.ndarray:
    """Symmetric exponential decay from the centre, parameter λ sampled."""
    lam = rng.uniform(0.0005, 0.02)
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    vals = np.exp(-lam * np.abs(idx - centre))
    vals = np.clip(vals, 0.0, None)
    vals /= vals.sum()
    return vals


def _init_interpolate(rng: np.random.Generator,
                     source: np.ndarray,
                     n_target: int) -> np.ndarray:
    """Linear interpolation of an existing distribution to a new length."""
    src_n = source.size
    if src_n == n_target:
        return source.copy()
    x_src = np.linspace(0.0, 1.0, src_n)
    x_tgt = np.linspace(0.0, 1.0, n_target)
    a = np.interp(x_tgt, x_src, source)
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n_target, 1.0 / n_target)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on the log‑variables (mirror descent on the simplex)
# ----------------------------------------------------------------------


def _adam_optimize(initial: np.ndarray,
                   rng: np.random.Generator,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Optimise the smoothed max (log‑sum‑exp) of the self‑convolution using
    Adam on the log‑variables.  Returns the best vector seen and its exact score.
    """
    a = initial.copy()
    n = a.size
    fft_len_conv = _fft_len_for_len(2 * n - 1)
    fft_len_grad = _fft_len_for_len(3 * n - 2)

    # initialise log‑variables (softmax parametrisation)
    z = np.log(a + 1e-12)

    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # parameters for the smooth‑max surrogate
    beta = 0.5
    beta_mul = 1.02

    best_vec = a.copy()
    best_score = _score(a, fft_len_conv)

    step = 1

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- simplex projection via softmax -----
        max_z = np.max(z)                # shift for numerical stability
        exp_z = np.exp(z - max_z)
        a = exp_z / exp_z.sum()

        # ----- convolution and smooth‑max gradient -----
        conv = _conv_self(a, fft_len_conv)

        # occasional exact sub‑gradient (helps when β is already large)
        if rng.random() < 0.25:
            raw_grad = _subgradient_step(a, conv)
        else:
            raw_grad = _softmax_gradient(a, conv, beta, fft_len_grad)

        # optional gradient clipping (prevents overflow in Adam updates)
        raw_grad = np.clip(raw_grad, -20.0, 20.0)

        # chain‑rule: ∂f/∂z = a * (∂f/∂a - (∂f/∂a·a))
        grad_z = a * (raw_grad - np.dot(raw_grad, a))

        # ----- Adam update on z -----
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        lr = 0.5 / np.sqrt(step)        # decreasing learning‑rate
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- occasional exploratory moves -----
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)
            # re‑initialise the log‑variables after a hard perturbation
            z = np.log(a + 1e-12)

        if rng.random() < 0.01:
            a = _pairwise_move(a, rng, fft_len_conv)
            z = np.log(a + 1e-12)

        # ----- keep track of the best exact score -----
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()

        # ----- increase smoothness -----
        beta = min(1e5, beta * beta_mul)

        step += 1

    final_score = _score(best_vec, _fft_len_for_len(2 * best_vec.size - 1))
    return best_vec, final_score


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    # keep a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 0.3)

    # ------------------------------------------------------------------
    # Global best container
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Load a previously known construction, if the variable exists
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                cand = np.clip(cand, 0.0, None)
                if cand.sum() == 0.0:
                    cand = np.full_like(cand, 1.0 / cand.size)
                else:
                    cand = cand / cand.sum()
                cand_score = _score(cand, _fft_len_for_len(2 * cand.size - 1))
                best_vec = cand.copy()
                best_score = cand_score
                print(f"[{time.time() - start:5.2f}s] Loaded previous best: {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Lengths that will be explored (bias towards larger vectors)
    # ------------------------------------------------------------------
    lengths = [1500, 2000, 2500, 3000, 3500, 3809, 4000,
               4500, 5000, 6000, 8000, 10000, 12000]

    # ------------------------------------------------------------------
    # Initialisation families and their selection probabilities
    # ------------------------------------------------------------------
    init_funs = [_init_random_dirichlet,
                 _init_sine_power,
                 _init_beta_distribution,
                 _init_exp_decay,
                 _init_interpolate]
    init_weights = np.array([0.30, 0.25, 0.20, 0.15, 0.10])
    init_weights /= init_weights.sum()

    # ------------------------------------------------------------------
    # Main stochastic search loop
    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ----- pick a length -----
        n = int(rng.choice(lengths))
        # ----- pick an initialisation method -----
        which = int(rng.choice(len(init_funs), p=init_weights))

        if init_funs[which] is _init_interpolate and best_vec is not None:
            a0 = _init_interpolate(rng, best_vec, n)
        else:
            a0 = init_funs[which](rng, n)

        # ----- decide on a modest number of Adam iterations -----
        # more iterations for smaller n, but always respect the remaining budget
        remaining = deadline - time.time()
        if remaining <= 0.0:
            break
        # heuristic: allocate at most 0.1 s per 1000 variables
        base_iters = int(min(5000, max(300, remaining * 2.0 / (n / 4000.0))))
        max_iters = max(300, base_iters)

        a_opt, score_opt = _adam_optimize(a0, rng, deadline, max_iters)

        if score_opt < best_score:
            best_score = score_opt
            best_vec = a_opt.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:5.2f}s] New global best: {best_score:.6f} (n={n})")

        # ----- occasional polishing of the current global best -----
        if best_vec is not None and time.time() < deadline - 2.0:
            # give the best a longer optimisation run
            polish_iters = int(min(12000, (deadline - time.time()) * 1.5))
            a_pol, score_pol = _adam_optimize(best_vec.copy(), rng, deadline, polish_iters)
            if score_pol < best_score:
                best_score = score_pol
                best_vec = a_pol.copy()
                elapsed = time.time() - start
                print(f"[{elapsed:5.2f}s] Polished best: {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 2000
        best_vec = np.full(n, 1.0 / n)
        best_score = _score(best_vec, _fft_len_for_len(2 * n - 1))
        print(f"Fallback uniform candidate: score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also does this)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Convert to plain Python floats for the harness
    return [float(x) for x in best_vec]
