"""
Advanced step‑function optimizer.

The algorithm proceeds in three stages:

1. **Diverse initialization** – many analytic families (√‑difference,
   power‑differences, exponentials) and random Dirichlet samples are
   generated, plus a simple 2‑component mixture family whose parameters are
   randomly sampled.  All candidates are normalised to sum = 1 and the best
   (lowest 2·n·max(conv)) is kept.

2. **Projected exponentiated‑gradient descent** – a soft‑max surrogate of the
   ‑∞‑norm is minimised.  The active set comprises the convolution indices
   whose value is within 0.5 % of the current maximum (capped at 15 indices).
   A momentum term stabilises the updates.  Random block perturbations and
   pairwise mass transfers keep the iterate from stagnating.

3. **Targeted local search** – for the remaining time a greedy move is
   performed that reduces the contribution of the actual maximum convolution
   entry.  The source index is sampled proportionally to its contribution
   `a_i * a_{k‑i}` to the current worst convolution index `k`.  A small mass
   δ is transferred to a random target.  Accept if the objective improves.
   Occasional smoothing further flattens spikes.

All operations respect the evaluation‐function constraints (non‑negative,
values ≤ 1000).  The routine prints progress timestamps and returns the best
sequence found before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: smallest power‑of‑two length that can hold the full linear
# convolution of two vectors of length n (i.e. length 2·n‑1).
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Fast (exact) L∞‑objective for a normalised (sum=1) non‑negative vector.
# Returns the numerator part 2·n·max(conv) (the denominator is 1 because we
# keep sum=1).
# ----------------------------------------------------------------------
def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    conv = _conv_self(a, fft_len)
    return 2.0 * n * float(conv.max())


# ----------------------------------------------------------------------
# Gradient of a soft‑max surrogate of the convolution max.
# Weight array w has length 2·n‑1 and is non‑negative, sum=1.
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    n = a.size
    L = w.size                     # = 2·n‑1
    # convolution a * a ↔ correlation a ⋆ w (a reversed)
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Normalise a raw vector to a probability vector (sum = 1, non‑negative).
# ----------------------------------------------------------------------
def _normalise(v: np.ndarray) -> np.ndarray:
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

            w = w_raw / w_sum if w_sum > 0 else np.full_like(w_raw, 1.0 / w_raw.size)

            grad = np.zeros_like(cur)
            for idx_k, weight in zip(active, w):
                lo = max(0, idx_k - (n - 1))
                hi = min(idx_k, n - 1)
                i_range = np.arange(lo, hi + 1, dtype=np.int64)
                grad[i_range] += weight * 2.0 * cur[idx_k - i_range]
        else:
            # Extremely unlikely; fall back to full soft‑max
            w_raw = np.exp(beta * (conv - cur_max))
            w = w_raw / w_raw.sum()
            grad = _gradient_correlation(cur, w)

        # Momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # Exponentiated‑gradient (mirror descent) update
        cur = cur * np.exp(-eta * grad)
        cur = np.clip(cur, 0.0, None)
        s = cur.sum()
        if s > 0.0:
            cur = cur / s
        else:
            cur = np.full_like(cur, 1.0 / n)

        # Occasional random block perturbation
        if rng.random() < prob_block:
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            blk_factor = np.exp(rng.normal(0.0, 0.2))
            cur[start:start + width] *= blk_factor
            cur = np.clip(cur, 0.0, None)
            cur = cur / cur.sum()

        # Occasional pairwise mass transfer
        if rng.random() < prob_pair:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur[i] > 1e-14:
                delta = rng.random() * cur[i] * 0.5
                cur[i] -= delta
                cur[j] += delta
                cur = np.clip(cur, 0.0, None)
                cur = cur / cur.sum()

    # ------------------------------------------------------------------
    # 2) Targeted local search focused on the worst convolution entry
    # ------------------------------------------------------------------
    cur_local = best.copy()
    cur_local_score = best_score
    deadline_total = start_time + float(budget_s)
    it = 0

    while time.monotonic() < deadline_total:
        conv = _conv_self(cur_local, fft_len)
        max_idx = int(conv.argmax())
        max_val = float(conv.max())

        # Indices that actually contribute to conv[max_idx]
        lo = max(0, max_idx - (n - 1))
        hi = min(max_idx, n - 1)
        i_arr = np.arange(lo, hi + 1, dtype=np.int64)
        j_arr = max_idx - i_arr
        contrib = cur_local[i_arr] * cur_local[j_arr]
        total_contrib = contrib.sum()

        # Choose a source proportional to its contribution
        if total_contrib > 0:
            prob = contrib / total_contrib
            src = rng.choice(i_arr, p=prob)
        else:
            src = rng.integers(0, n)

        # Random target (different from source)
        tgt = rng.integers(0, n)
        while tgt == src:
            tgt = rng.integers(0, n)

        # Transfer a modest amount of mass
        delta = rng.random() * cur_local[src] * 0.4
        new_seq = cur_local.copy()
        new_seq[src] -= delta
        new_seq[tgt] += delta
        if new_seq[src] < 0.0:
            # discard illegal move
            continue

        new_score = _score_of(new_seq, n, fft_len)

        if new_score < cur_local_score:
            cur_local = new_seq
            cur_local_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = new_seq.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] local improvement = {best_score:.9f}")

        it += 1

        # Gentle smoothing every few thousand iterations
        if it % 4000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            cur_local = np.convolve(cur_local, kernel, mode='same')
            cur_local = cur_local / cur_local.sum()
            cur_local_score = _score_of(cur_local, n, fft_len)

    # ------------------------------------------------------------------
    # Return the best sequence respecting the evaluate_sequence limits
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
