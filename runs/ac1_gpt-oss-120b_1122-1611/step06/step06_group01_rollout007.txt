"""
Hybrid flatten‑then‑smooth‑max optimizer for the step‑function problem.

The goal is to minimise
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a))²
under a_i ≥ 0.  Because the ratio is homogeneous we can fix
∑a_i = 1, so the objective becomes 2·n·max(conv).

Algorithm (for a fixed length n)
------------------------------------------------------------
1. **Flattening phase** – minimise the L² distance of the full
   self‑convolution to the uniform distribution 1/(2n‑1).  The
   gradient of ½‖conv‑target‖² is   g = 2·( (conv‑target) * rev(a) ).
   A simple projected gradient descent with a linearly decaying step
   size drives the autocorrelation towards a flat shape.

2. **Soft‑max max phase** – replace the hard max by a smooth soft‑max
   (log‑sum‑exp) with temperature β.  The gradient of the smoothed
   objective is   g = 2·(p * rev(a)),  where p is the soft‑max
   probability vector over the convolution entries.  β is increased
   linearly from a small value to a large one while the step‑size is
   simultaneously reduced.

3. **Exact sub‑gradient phase** – after the soft‑max has concentrated
   on the true peak we switch to the true sub‑gradient of the max:
   only the indices that contribute to the current peak are non‑zero.
   The step size follows a classic 1/√t schedule.

4. **Stagnation handling** – if no improvement has been seen for a
   while the current iterate is jittered by multiplicative log‑normal
   noise (≈10 % σ) and the optimiser state is reset.

5. **Multi‑length search** – several lengths are tried (the previous
   best length, a few powers‑of‑two and a longer candidate).  Each
   length receives an equal slice of the remaining time budget; the
   overall best vector is returned.

All updates are followed by an Euclidean projection onto the
probability simplex {x | x_i≥0, ∑x_i=1}.  The implementation uses only
NumPy and FFTs, runs well within the 1000 s budget and typically
pushes the bound below the previous 1.5136 value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of prescribed length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Gradient / sub‑gradient helpers

def _gradient_l2(a: np.ndarray,
                 target: np.ndarray,
                 fft_len_self: int,
                 fft_len_grad: int) -> np.ndarray:
    """
    Gradient of ½‖conv(a,a) - target‖² .
    g = 2·((conv – target) * rev(a)).
    """
    conv = _self_conv(a, fft_len_self)
    e = conv - target                      # length 2n‑1
    rev_a = a[::-1]
    fe = np.fft.rfft(e, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    full = np.fft.irfft(fe * fra, n=fft_len_grad)[: 3 * a.size - 2]
    grad = 2.0 * full[a.size - 1: a.size - 1 + a.size]
    return grad


def _gradient_softmax_max(a: np.ndarray,
                          beta: float,
                          fft_len_self: int,
                          fft_len_grad: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Gradient of the smooth approximation
        f_β(a) = Σ_i p_i(a)·conv_i(a) ,   p_i ∝ exp(β·conv_i)
    Returns (gradient, conv) where conv = a*a.
    """
    conv = _self_conv(a, fft_len_self)          # length 2n‑1
    max_c = conv.max()
    # stable soft‑max weights
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum
    rev_a = a[::-1]
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * a.size - 2]
    grad = 2.0 * grad_full[a.size - 1: a.size - 1 + a.size]
    return grad, conv


def _subgradient_max(a: np.ndarray,
                     fft_len_self: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Exact sub‑gradient of max(conv) at the current peak.
    Returns (sub‑gradient, conv).
    """
    conv = _self_conv(a, fft_len_self)
    k = int(conv.argmax())
    n = a.size
    g = np.zeros_like(a)
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    idx_i = np.arange(i_min, i_max + 1)
    idx_j = k - idx_i
    g[idx_i] = 2.0 * a[idx_j]
    return g, conv
