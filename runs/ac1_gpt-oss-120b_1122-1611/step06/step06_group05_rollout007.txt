def simplex_projection(v):
    """Project onto probability simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

Algorithm
---------
*   Several families of deterministic shapes (power‑law differences,
    geometric, linear decay, uniform) are enumerated for a handful of
    lengths (≈ 2000 … 8000).  The best candidate among them is kept.
*   From this starting point a projected sub‑gradient descent is run.
    The objective is  f(a) = max_k (a∗a)_k .  At each iteration we
    locate the index k where the convolution attains its maximum and
    use the exact sub‑gradient of that entry:
            ∂(a∗a)_k / ∂a_i = 2·a_{k−i}  (if 0≤k−i<n, else 0)
    The update is a simple gradient step followed by projection onto the
    probability simplex (a_i ≥ 0, Σ a_i = 1).  Step‑size decays like
    1/√t and occasional random jitter + restarts are performed when
    progress stalls.
*   The whole search respects the 1000 s budget, prints progress
    stamps and finally returns the best sequence found (clipped to the
    limits enforced by ``evaluate_sequence``).
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper to obtain a power‑of‑two length suitable for a full linear
# convolution of two vectors of length n.
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


# ----------------------------------------------------------------------
# Linear self‑convolution via FFT (returns length 2·len(a)‑1).
# ----------------------------------------------------------------------
def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Project a vector onto the probability simplex (non‑negative, sum = 1).
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Compute the objective value 2·n·max(a∗a) (the same quantity that
# evaluate_sequence returns when Σa = 1).
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, fft_len: int) -> float:
    conv = _conv_self(a, fft_len)
    return float(2.0 * a.size * conv.max())


# ----------------------------------------------------------------------
# Exact sub‑gradient of the maximal convolution entry.
# ----------------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray, k: int) -> np.ndarray:
    """Return ∂ (a∗a)_k / ∂a  (the sub‑gradient of the max entry)."""
    n = a.shape[0]
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    grad = np.zeros_like(a)
    # each valid pair contributes 2·a_{k−i}
    grad[idx] = 2.0 * a[k - idx]
    return grad


# ----------------------------------------------------------------------
# A few deterministic shape generators (always produce a non‑negative
# vector with sum = 1).
# ----------------------------------------------------------------------
def _shape_power_diff(n: int, p: float) -> np.ndarray:
    """a_i = (i+1)^p - i^p , normalised."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p - i ** p
    a /= a.sum()
    return a


def _shape_geometric(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i , truncated and normalised."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


def _shape_linear(n: int) -> np.ndarray:
    """Linearly decreasing shape a_i ∝ (n-i)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    a /= a.sum()
    return a


def _shape_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    a = np.ones(n, dtype=np.float64) / n
    return a


def _shape_sqrt_diff(n: int) -> np.ndarray:
    """The classic √‑difference shape used in earlier runs."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    evaluate_sequence() within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------- problem size -----------------------------
    # try a handful of lengths – larger lengths give more “degrees of
    # freedom” but also cost more time; we keep the range modest.
    n_candidates = [2000, 3000, 3809, 5000, 6000, 8000]
    # make sure we respect the user‑supplied size if any
    if "n" in kwargs:
        try:
            n_fixed = int(kwargs["n"])
            if n_fixed >= 2:
                n_candidates = [n_fixed]
        except Exception:
            pass

    best_a_global = None
    best_score_global = math.inf

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # safety margin

    # ------------------------------------------------------------------
    # 1) quick grid search over a few analytic families
    # ------------------------------------------------------------------
    for n in n_candidates:
        fft_len = _fft_len_for_n(n)

        # try a small collection of shape families
        shape_fns = [
            _shape_uniform,
            _shape_linear,
            _shape_sqrt_diff,
            lambda n: _shape_power_diff(n, 0.45),
            lambda n: _shape_power_diff(n, 0.50),
            lambda n: _shape_power_diff(n, 0.55),
            lambda n: _shape_power_diff(n, 0.60),
            lambda n: _shape_geometric(n, 0.95),
            lambda n: _shape_geometric(n, 0.98),
            lambda n: _shape_geometric(n, 0.99),
        ]

        for fn in shape_fns:
            a = fn(n).astype(np.float64)
            score = _objective(a, fft_len)

            if score < best_score_global:
                best_score_global = score
                best_a_global = a.copy()
                print(f"[grid] n={n:5d}  score={score:.6f}")

        # a few random Dirichlet starts (useful for later refinement)
        for _ in range(4):
            alpha = rng.uniform(0.1, 5.0)          # control concentration
            a = rng.dirichlet(np.full(n, alpha))
            score = _objective(a, fft_len)
            if score < best_score_global:
                best_score_global = score
                best_a_global = a.copy()
                print(f"[dir]  n={n:5d}  α={alpha:.2f}  score={score:.6f}")

        # abort early if budget is already exhausted
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) projected sub‑gradient descent from the best candidate found so far
    # ------------------------------------------------------------------
    if best_a_global is None:
        # fallback – uniform vector of length 2000
        n = 2000
        best_a_global = _shape_uniform(n)
        best_score_global = _objective(best_a_global, _fft_len_for_n(n))

    a = best_a_global.copy()
    n = a.size
    fft_len = _fft_len_for_n(n)

    # classic sub‑gradient schedule parameters
    step0 = 0.8          # initial step length (tuned by a few quick tests)
    max_iters = 10_000_000   # effectively “infinite”; loop stops by time
    iter_cnt = 0
    last_improve = time.monotonic()
    steps_without_improve = 0
    restart_gap = 150.0   # seconds without improvement → restart with jitter

    while time.monotonic() < deadline:
        iter_cnt += 1
        now = time.monotonic()

        # ---- convolution & maximal index --------------------------------
        conv = _conv_self(a, fft_len)
        k = int(np.argmax(conv))
        cur_score = float(2.0 * n * conv[k])

        # ---- keep the global best ---------------------------------------
        if cur_score < best_score_global - 1e-12:
            best_score_global = cur_score
            best_a_global = a.copy()
            last_improve = now
            steps_without_improve = 0
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best (subgrad) score = {best_score_global:.6f}")

        # ---- sub‑gradient of the maximal entry -------------------------
        g = _max_conv_subgrad(a, k)

        # ---- diminishing step size (1/√t) -------------------------------
        eta = step0 / math.sqrt(iter_cnt)

        # ---- gradient step + simplex projection -------------------------
        a = a - eta * g
        a = _simplex_projection(a)

        # ---- occasional random jitter to escape plateaus ---------------
        if rng.random() < 0.01:
            jitter = rng.normal(0.0, 0.02, size=n)
            a = _simplex_projection(a + jitter)

        # ---- restart logic ------------------------------------------------
        steps_without_improve += 1
        if now - last_improve > restart_gap:
            # re‑initialise from the best solution seen so far, add a little noise
            a = best_a_global.copy()
            a += rng.normal(0.0, 0.03, size=n)
            a = _simplex_projection(a)
            last_improve = now
            steps_without_improve = 0
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score still {best_score_global:.6f}")

        # safeguard against pathological runaway
        if iter_cnt >= max_iters:
            break

    # ------------------------------------------------------------------
    # Return the best sequence discovered (respect the clipping rules of
    # evaluate_sequence – values are already non‑negative and tiny).
    # ------------------------------------------------------------------
    best_a_global = np.clip(best_a_global, 0.0, 1000.0)
    return [float(v) for v in best_a_global]
