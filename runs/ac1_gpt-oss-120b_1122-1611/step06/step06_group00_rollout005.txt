"""
Gradient‑softmax optimiser for the self‑convolution bound.

The function builds a non‑negative sequence (a probability vector) that
minimises

        2·n·max(a*a) / (sum(a))²

which is exactly the value returned by ``evaluate_sequence`` (the sum is
kept equal to 1, so the denominator is fixed).  The optimiser works in two
phases inside a single time‑budget:

* **Phase 1 – continuation with a soft‑max objective**  
  The true objective ``max(a*a)`` is replaced by a smooth
  ``log‑sum‑exp`` (temperature τ).  The gradient of this surrogate can be
  obtained analytically and evaluated with two FFTs (self‑convolution and
  correlation).  An Adam optimiser drives the variables while the
  temperature τ is linearly annealed from 0.2 to 0.001, gradually turning
  the smooth surrogate into the true max.

* **Phase 2 – occasional stochastic jitter**  
  After every few hundred Adam updates a small multiplicative log‑normal
  perturbation is applied to a random block to help escape shallow
  plateaus.  The best sequence seen so far (according to the exact
  ``evaluate_sequence``) is stored and returned.

The algorithm starts from the previously discovered construction
``height_sequence_1`` when it exists; otherwise a uniform vector is used.
All operations respect the non‑negativity and simplex constraints, and
the routine finishes well before the 1000 s deadline, returning the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # ensure non‑negative (the algorithm works for any real vector)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be projected to zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _softmax_weights(c: np.ndarray, tau: float) -> np.ndarray:
    """
    Compute soft‑max probabilities of the convolution values ``c``
    with temperature ``tau`` in a numerically stable way.
    Returns a vector of the same length as ``c`` that sums to 1.
    """
    # shift by the maximum to avoid overflow
    shift = c.max()
    exp_shifted = np.exp((c - shift) / tau)
    total = exp_shifted.sum()
    if total == 0.0:
        # fallback – uniform weights (should not happen)
        return np.full_like(c, 1.0 / len(c))
    return exp_shifted / total


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    The algorithm uses a soft‑max surrogate and Adam optimisation,
    with occasional random jitter for exploration.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all():
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # default length – a power of two simplifies FFT handling
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)      # uniform start
    else:
        # use the provided construction, but rescale to sum = 1
        a = a0.astype(np.float64).copy()
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a[:] = 1.0 / a.size
        n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length (large enough for convolution + correlation)
    # ------------------------------------------------------------------
    fft_len = _next_pow2(3 * n)               # ≥ n + (2n‑1) – 1
    # Helper: indices to extract the “valid” correlation part
    corr_start = n - 1
    corr_end = corr_start + n                 # exclusive

    # ------------------------------------------------------------------
    # 3) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    # initial exact score (using the provided evaluate_sequence)
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    best_time = time.monotonic()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = beta1, beta2

    # Temperature schedule (soft‑max → true max)
    tau_start, tau_end = 0.20, 0.001

    # Learning‑rate schedule (larger at the beginning)
    lr_start, lr_end = 0.30, 0.001

    # Jitter schedule – every JITTER_EVERY iterations we perturb a block
    JITTER_EVERY = 500
    ITER_PRINT_EVERY = 200

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    start_time = time.monotonic()
    while time.monotonic() < deadline:
        # ---- time‑fraction for schedules --------------------------------
        now = time.monotonic()
        frac_total = (now - start_time) / (deadline - start_time)

        # ---- annealing of temperature and learning rate ------------------
        tau = tau_start * (1.0 - frac_total) + tau_end * frac_total
        lr = lr_start * (1.0 - frac_total) + lr_end * frac_total

        # ---- self‑convolution via FFT -----------------------------------
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ---- soft‑max weights -------------------------------------------
        w = _softmax_weights(conv, tau)            # length 2n‑1, sums to 1

        # ---- correlation (gradient of the soft‑max) ---------------------
        w_fft = np.fft.rfft(w, n=fft_len)
        # linear (non‑circular) correlation via conjugate multiplication
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        grad = (2.0 / tau) * corr_full[corr_start:corr_end]   # length n

        # ---- Adam update ------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- gradient descent step (project onto simplex) ---------------
        a = a - step
        # clip negatives, then project to the simplex
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = _project_onto_simplex(a)

        iteration += 1

        # ---- occasional jitter to increase exploration -------------------
        if iteration % JITTER_EVERY == 0:
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a_jit = a.copy()
            a_jit[start:start + width] *= factor
            a_jit = np.clip(a_jit, 0.0, None)
            a_jit = _project_onto_simplex(a_jit)
            a = a_jit

        # ---- periodic exact evaluation ---------------------------------
        if iteration % ITER_PRINT_EVERY == 0:
            # exact self‑convolution (numpy’s direct convolve) for a reliable score
            exact_score = evaluate_sequence(a.tolist())
            if exact_score < best_score - 1e-12:
                best_score = exact_score
                best = a.copy()
                best_time = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f} "
                      f"(iter={iteration}, τ={tau:.5f})")
        # ---- early‑restart if stagnated for a long time ----------------
        if now - best_time > 45.0:
            # re‑inject a modest amount of noise around the current best
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a *= noise
            a = _project_onto_simplex(a)
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = beta1, beta2
            best_time = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – injected noise")

    # ------------------------------------------------------------------
    # 6) Return the best sequence found (clipped to the limits of the
    #    evaluator for safety).  The evaluator is scale‑invariant, so we
    #    keep the vector as‑is (sum≈1) and simply enforce the allowed range.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
