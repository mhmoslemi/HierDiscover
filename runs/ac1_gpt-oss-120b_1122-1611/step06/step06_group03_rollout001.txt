"""
Propose a step‑function (non‑negative heights) that minimizes

    score = 2·n·max_k (a∗a)_k / (∑a_i)^2

The algorithm is a large‑scale mirror‑descent (exponentiated‑gradient)
optimiser applied to a smooth approximation of the max‑operator.
It works entirely on the simplex (∑a_i = 1) and uses full‑gradient
information via FFT‑based convolution/correlation.  A continuation
scheme slowly raises the soft‑max temperature (β) while simultaneously
decreasing the step‑size (η), allowing the optimiser to first move
smoothly and then focus on the true max.  Random “jitter” steps are
interleaved to escape shallow local minima.  The routine respects the
1000 s budget, prints progress whenever a new best score is found and
returns the best vector encountered (clipped to the limits used by
``evaluate_sequence``).

No external files are accessed; all helper functions are top‑level.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via a real‑to‑real FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum Σ_k w_k (a∗a)_k with respect to a.
    Implements the correlation a ⋆ w (full length 2·n‑1) via FFT.
    """
    n = a.size
    L = w.size                       # should be 2·n‑1
    # length sufficient for linear correlation (size = L + n - 1)
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    # reverse a for correlation
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start:  a_i = √(i+1)‑√i  (already normalised to sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _maybe_use_global() -> np.ndarray | None:
    """If a previously stored construction exists, return it normalised to sum = 1."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Core optimisation routine (mirror‑descent with continuation)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Initial vector (analytic start or a saved construction)
    # --------------------------------------------------------------
    a = _analytic_start(n)
    cand_global = _maybe_use_global()
    if cand_global is not None:
        # Use the better of the two starts (based on the fast internal score)
        # Fast score = 2·n·max(conv) because we keep sum = 1.
        conv_local = _conv_self(a, fft_len)
        score_local = 2.0 * n * float(conv_local.max())
        conv_global = _conv_self(cand_global, _fft_len_for_n(cand_global.size))
        score_global = 2.0 * cand_global.size * float(conv_global.max())
        if score_global < score_local:
            a = cand_global
            n = a.size
            fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len)
    best_score = 2.0 * n * float(conv_best.max())
    start_time = time.monotonic()
    # keep a small safety margin to avoid accidental overshoot
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Hyper‑parameters for the continuation schedule
    # --------------------------------------------------------------
    beta_start = 1.0          # soft‑max temperature (low → smooth)
    beta_end   = 5000.0       # final temperature (near‑max)
    eta_start  = 0.5          # initial step size (exponentiated‑gradient)
    eta_end    = 0.01         # final step size
    jitter_interval = 200     # steps between random jitter injections
    jitter_scale    = 0.02    # log‑normal jitter std‑dev

    step = 0
    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # Scheduling
        # ----------------------------------------------------------
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 0.0

        beta = beta_start + (beta_end - beta_start) * frac
        eta  = eta_start  * (1.0 - frac) + eta_end * frac

        # ----------------------------------------------------------
        # Current convolution and soft‑max weights
        # ----------------------------------------------------------
        conv = _conv_self(a, fft_len)           # length 2·n‑1
        cur_max = float(conv.max())
        # soft‑max (numerically stable)
        shift = cur_max
        w_raw = np.exp(beta * (conv - shift))
        w_sum = w_raw.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            # fallback to uniform weights (should rarely happen)
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w_raw / w_sum

        # ----------------------------------------------------------
        # Gradient of the smoothed objective
        # ----------------------------------------------------------
        grad = _gradient_correlation(a, w)   # shape (n,)

        # ----------------------------------------------------------
        # Exponentiated‑gradient update on the simplex
        # ----------------------------------------------------------
        a = a * np.exp(-eta * grad)
        # avoid complete underflow – keep a tiny floor
        a = np.clip(a, 1e-30, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ----------------------------------------------------------
        # Occasionally inject a tiny random jitter to help escape flat regions
        # ----------------------------------------------------------
        if step % jitter_interval == 0 and step > 0:
            jitter = np.exp(rng.normal(0.0, jitter_scale, size=n))
            a *= jitter
            a = np.clip(a, 1e-30, None)
            a /= a.sum()

        # ----------------------------------------------------------
        # Record improvement
        # ----------------------------------------------------------
        score = 2.0 * n * cur_max
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        step += 1

        # ----------------------------------------------------------
        # Safety break (a few ms before the deadline)
        # ----------------------------------------------------------
        if time.monotonic() > deadline - 0.05:
            break

    # --------------------------------------------------------------
    # Final post‑processing (clip to limits enforced by evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
