"""
Non‑negative step‑function optimizer.

The algorithm consists of two phases:

1. **Smoothed max‑gradient descent** – we minimise a smooth
   approximation of `max(conv)` using an exponentiated‑gradient
   (mirror‑descent) update on the simplex.  The smoothness parameter
   `theta` grows from a low to a high value, turning the softmax into a
   hard max, while the learning rate `eta` decays.  Momentum and occasional
   random jitter help escape plateaus.

2. **Active‑set polishing** – when the time budget is almost exhausted we
   switch to a deterministic mirror‑descent scheme that focuses on the
   (few) near‑maximal convolution entries (the “active set”), exactly as
   in the reference implementation.  This refines the best candidate
   found by the first phase.

Both phases use FFT‑based convolution/correlation for speed and honour the
`evaluate_sequence` constraints (non‑negative, bounded values).  The
function `propose_candidate` returns the best sequence found within the
given time budget (default 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution of two vectors of length n."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (real‑to‑real) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective for a vector already normalised to sum‑one."""
    return float(2.0 * n * conv.max())


def _compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Fast evaluation of the objective for a normalised vector a."""
    conv = _conv_self(a, fft_len)
    return _score_from_conv(conv, n)


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------
def _analytic_start(n: int) -> np.ndarray:
    """√‑difference start: a_i = √(i+1)‑√i, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0) - np.sqrt(i)
    a /= a.sum()
    return a


def _maybe_use_global() -> np.ndarray | None:
    """If a previous construction exists in the global namespace, return it normalised."""
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size >= 2 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                s = cand.sum()
                if s > 0.0:
                    return cand / s
        except Exception:
            pass
    return None


# ----------------------------------------------------------------------
# Gradient of a weighted autocorrelation (used in polishing)
# ----------------------------------------------------------------------
def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k .
    Implements the correlation a ⋆ w (cross‑correlation).
    """
    n = a.size
    L = 1 << ((w.size + n - 2).bit_length())  # power‑of‑two for FFT
    fw = np.fft.rfft(w, n=L)
    fa_rev = np.fft.rfft(a[::-1], n=L)
    corr_full = np.fft.irfft(fw * fa_rev, n=L)[: w.size + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Deterministic polishing (active‑set mirror‑descent)
# ----------------------------------------------------------------------
def _polish(a: np.ndarray, n: int, fft_len: int,
            deadline: float, global_start: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent polishing using an active set of near‑maximal
    convolution indices.  Returns the best vector and its score.
    """
    best = a.copy()
    conv = _conv_self(a, fft_len)
    best_score = 2.0 * n * conv.max()
    grad_prev = np.zeros_like(a)

    # hyper‑parameters (tuned for the final refinement phase)
    eta_start = 0.12
    eta_end = 0.002
    beta_start = 50.0
    beta_end = 2000.0
    momentum = 0.85
    max_active = 20

    while time.monotonic() < deadline:
        # current convolution & score
        conv = _conv_self(a, fft_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # keep the best ever seen during polishing
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - global_start
            print(f"[{elapsed:6.2f}s] polish: new best = {best_score:.6f}")

        # schedule for eta and beta
        now = time.monotonic()
        elapsed = now - global_start
        total = deadline - global_start
        if total <= 0:
            break
        eta = eta_start * (1.0 - elapsed / total) + eta_end * (elapsed / total)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total)

        # active set (indices whose convolution value is within 0.2 % of the max)
        tol = cur_max * 0.998
        active = np.where(conv >= tol)[0]
        if active.size > max_active:
            # keep only the top‑max_active entries
            active = np.argpartition(conv, -max_active)[-max_active:]

        # compute soft‑max weights on the active set
        if active.size == 0:
            # fall back to full soft‑max (should never happen)
            w = np.exp(beta * (conv - cur_max))
            w /= w.sum()
            grad = _gradient_correlation(a, w)
        else:
            w_raw = np.exp(beta * (conv[active] - cur_max))
            w = w_raw / w_raw.sum()
            grad = np.zeros_like(a)
            # each active index contributes a shifted copy of `a`
            for idx, weight in zip(active, w):
                lo = max(0, idx - (n - 1))
                hi = min(n - 1, idx)
                rng = np.arange(lo, hi + 1, dtype=np.int64)
                grad[rng] += weight * 2.0 * a[idx - rng]

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # exponentiated‑gradient update + simplex projection
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # early exit if we are almost out of time
        if time.monotonic() > deadline - 0.1:
            break

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` before the time budget elapses.

    The routine first performs a smoothed‑max gradient descent with an
    exponentially growing softmax temperature, then finishes with a short
    active‑set mirror‑descent polishing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    n = int(kwargs.get("n", 3800))
    if n < 2:
        n = 2
    fft_len = _fft_len_for_n(n)

    # --------------------------------------------------------------
    # Initial vector (analytic start or a previously stored construction)
    # --------------------------------------------------------------
    a = _analytic_start(n)
    cand_global = _maybe_use_global()
    if cand_global is not None and cand_global.size == n:
        if _compute_score(cand_global, _fft_len_for_n(n), n) < _compute_score(a, fft_len, n):
            a = cand_global

    # --------------------------------------------------------------
    # Book‑keeping
    # --------------------------------------------------------------
    best = a.copy()
    best_score = _compute_score(best, fft_len, n)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety margin

    # --------------------------------------------------------------
    # Smoothed‑max gradient descent parameters
    # --------------------------------------------------------------
    eta_start = 0.5      # initial learning rate
    eta_end = 0.001      # final learning rate
    theta_start = 5.0    # initial soft‑max temperature
    theta_end = 4000.0   # final temperature (hard max)
    momentum = 0.9
    grad_prev = np.zeros_like(a)
    no_improve = 0
    iteration = 0

    # --------------------------------------------------------------
    # Main gradient‑descent loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        prog = elapsed / total if total > 0 else 1.0

        # linear schedules for η and θ
        eta = eta_start * (1.0 - prog) + eta_end * prog
        theta = theta_start + (theta_end - theta_start) * prog

        # FFT of the current vector
        fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(fa * fa, n=fft_len)
        conv = conv_full[: 2 * n - 1]
        max_conv = conv.max()

        # soft‑max weights (stable: subtract max)
        w = np.exp(theta * (conv - max_conv))
        w_sum = w.sum()
        if w_sum == 0.0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient = 2 * (a ⋆ w)  (cross‑correlation)
        fw = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(fw * np.conj(fa), n=fft_len)
        grad = 2.0 * corr_full[n - 1:n - 1 + n]

        # momentum
        grad = momentum * grad_prev + (1.0 - momentum) * grad
        grad_prev = grad

        # safeguard against huge exponentials
        max_exp_arg = 80.0          # np.exp(80) ~ 5e34, still safe
        clip_val = max_exp_arg / max(eta, 1e-12)
        grad = np.clip(grad, -clip_val, clip_val)

        # exponentiated‑gradient update on the simplex
        factor = np.exp(-eta * grad)
        a = a * factor
        a = np.clip(a, 0.0, None)
        # enforce a tiny lower bound to avoid exact zeros (helps stability)
        a = np.maximum(a, 1e-15)
        a /= a.sum()

        # evaluate the true objective (max convolution) for tracking
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")
            no_improve = 0
        else:
            no_improve += 1

        # occasional random jitter to shake off stagnation
        if iteration % 500 == 0 and rng.random() < 0.15:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a = a * jitter
            a /= a.sum()

        # restart from the best if we have seen no improvement for a while
        if no_improve > 5000:
            jitter = np.exp(rng.normal(0.0, 0.05, size=n))
            a = best * jitter
            a = np.clip(a, 0.0, None)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            no_improve = 0

        iteration += 1

        # early break if we are within a short final window
        if time.monotonic() > deadline - 0.2:
            break

    # --------------------------------------------------------------
    # Polishing phase (active‑set mirror‑descent) if time permits
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        polish_deadline = time.monotonic() + remaining
        best, best_score = _polish(best, n, fft_len, polish_deadline, start_time)

    # --------------------------------------------------------------
    # Final clipping as required by evaluate_sequence()
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
