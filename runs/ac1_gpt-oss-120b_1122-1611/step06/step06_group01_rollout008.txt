def _objective_and_grad(u, beta, n, fft_len_self, fft_len_grad):
    # compute a
    max_u = np.max(u)  # for numerical stability
    exp_u = np.exp(u - max_u)
    a = exp_u / exp_u.sum()
    # compute self convolution
    conv = _self_conv(a, fft_len_self)
    max_c = conv.max()
    # compute softmax weights
    w = np.exp(beta * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum
    # value
    value = np.dot(p, conv)  # smoothed max

    # compute gradient wrt a
    rev_a = a[::-1]
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
    grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]  # shape n

    # gradient wrt u via chain rule
    dot = np.dot(grad_a, a)
    grad_u = a * (grad_a - dot)

    return float(value), grad_u.astype(np.float64)
```

            i_min = max(0, k_max - (n - 1))
            i_max = min(n - 1, k_max)
            idx_i = np.arange(i_min, i_max + 1)
            idx_j = k_max - idx_i
            g[idx_i] = 2.0 * a[idx_j]
            eta = 0.1 / np.sqrt(it + 1)
            a = a - eta * g
            # project onto simplex
            a = _project_simplex(a)
            # evaluate
            cur_score = 2.0 * n * _self_conv(a, fft_len_self).max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_overall = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] n={n} subgrad it={it} new best {best_score:.6f}")
            # stop early if below target
            if best_score < 1.5025:
                break

    # if best_overall is None fallback to length 1 etc.
    if best_overall is None:
        best_overall = np.array([1.0 / np.sqrt(2.0)], dtype=np.float64) # sum sqrt(2n)? Not needed.
        best_score = evaluate_sequence(best_overall.tolist())
        print(f"Fallback, got score {best_score}")

    # Scale to sum sqrt(2n) to match typical normalization
    n_best = best_overall.size
    scale = np.sqrt(2.0 * n_best) / best_overall.sum()
    best_overall = best_overall * scale
    # Clip
    best_overall = np.clip(best_overall, 0.0, 1000.0)

    return [float(x) for x in best_overall]

"""
Propose a step‑function with a lower bound on the evaluation function.

Algorithm (continuation + projected L‑BFGS)
------------------------------------------------
* Work on the probability simplex Σ a_i = 1 (the objective is scale‑invariant).
* Parametrise a_i = exp(u_i)/∑_j exp(u_j).  The optimisation variable u is
  unrestricted, which removes the simplex constraint.
* For a given temperature β we minimise the smooth approximation of the
  maximum of the self‑convolution:
        Φ_β(a) = ∑_k p_k·c_k  with  p_k ∝ exp(β·c_k)  and  c = a*a .
  Its gradient w.r.t. a is  g_a = 2·(p * rev(a))  (FFT‑based).
* Using the chain rule we obtain ∇_u Φ_β, allowing the use of a
  quasi‑Newton method (L‑BFGS‑B).  Starting from a low β we increase β
  geometrically (continuation), solving a modest‑size L‑BFGS sub‑problem
  at each stage.
* After the continuation phase we switch to a few exact sub‑gradient steps
  on the true max to polish the solution.
* The routine tries several lengths (including the previous construction
  `height_sequence_1`) while respecting the global time budget.  The best
  sequence found is returned (scaled to the conventional Σ a_i = √(2n) and
  clipped to the limits enforced by `evaluate_sequence`).
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of the prescribed length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  O(n log n) algorithm (Wang & Carreira‑Perpiñán).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.zeros_like(v)
        w[0] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_and_grad(u: np.ndarray,
                        beta: float,
                        n: int,
                        fft_len_self: int,
                        fft_len_grad: int):
    """
    Smoothed objective Φ_β(a) and its gradient w.r.t. the unconstrained
    variable u (where a = softmax(u)).  Returns (value, grad_u).
    """
    # ---- a = softmax(u) -------------------------------------------------
    max_u = np.max(u)
    exp_u = np.exp(u - max_u)               # stable exponentiation
    a = exp_u / exp_u.sum()                 # Σ a_i = 1, a_i ≥ 0

    # ---- self‑convolution ------------------------------------------------
    conv = _self_conv(a, fft_len_self)

    # ---- soft‑max weights for the current temperature --------------------
    max_c = conv.max()
    w = np.exp(beta * (conv - max_c))       # non‑negative, max at 0
    w_sum = w.sum()
    if w_sum == 0.0:
        p = np.full_like(w, 1.0 / w.size)
    else:
        p = w / w_sum                         # probability distribution

    # ---- value = 𝔼_p[c]  (smooth approximation of max) --------------------
    value = float(np.dot(p, conv))

    # ---- gradient w.r.t. a :  g_a = 2·(p * rev(a)) -----------------------
    rev_a = a[::-1]
    fp = np.fft.rfft(p, n=fft_len_grad)
    fra = np.fft.rfft(rev_a, n=fft_len_grad)
    grad_full = np.fft.irfft(fp * fra, n=fft_len_grad)[: 3 * n - 2]
    grad_a = 2.0 * grad_full[n - 1 : n - 1 + n]   # shape (n,)

    # ---- chain rule to obtain ∇_u ---------------------------------------
    dot = float(np.dot(grad_a, a))
    grad_u = a * (grad_a - dot)                # ∂Φ/∂u_j = a_j (g_j – Σ a_i g_i)

    return value, grad_u.astype(np.float64)


# ----------------------------------------------------------------------
# Core optimisation for a *single* length
# ----------------------------------------------------------------------


def _optimise_length(initial_a: np.ndarray,
                     deadline: float,
                     rng: np.random.Generator):
    """
    Optimises a probability vector for a fixed length using
    continuation (increasing β) with L‑BFGS‑B, followed by a short
    exact sub‑gradient phase.
    Returns the best vector found and its plain evaluation score.
    """
    n = initial_a.size
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # start from the probability vector (already normalised)
    eps = 1e-12
    u = np.log(initial_a + eps)               # free variable for optimisation

    # ----- continuation schedule (geometric increase of β) ---------------
    beta_vals = np.geomspace(0.5, 4000.0, num=12)

    best_a = initial_a.copy()
    best_score = float("inf")
    start = time.monotonic()

    for beta in beta_vals:
        if time.monotonic() >= deadline:
            break

        # a very modest max‑iteration cap – the outer continuation already
        # supplies many descent directions
        remaining = deadline - time.monotonic()
        # each L‑BFGS iteration costs a handful of FFTs; we conservatively cap
        # at 150 iterations or whatever fits into the remaining time.
        maxiter = int(min(150, max(5, remaining / 0.02)))

        res = scipy.optimize.minimize(
            _objective_and_grad,
            u,
            args=(beta, n, fft_len_self, fft_len_grad),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxiter, "disp": False},
        )
        u = res.x

        # evaluate exact objective (max of the self‑convolution)
        a = np.exp(u - np.max(u))
        a /= a.sum()
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] n={n:5d} β={beta:6.1f} new best {best_score:.6f}")

        # early exit if the target is already beaten
        if best_score < 1.5025:
            break

    # ----- final polishing with the exact sub‑gradient --------------------
    a = best_a.copy()
    it = 0
    while time.monotonic() < deadline and it < 3000:
        it += 1
        conv = _self_conv(a, fft_len_self)
        k_max = int(conv.argmax())

        # sub‑gradient of the active peak
        g = np.zeros_like(a)
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        j_range = k_max - i_range
        g[i_range] = 2.0 * a[j_range]

        eta = 0.12 / np.sqrt(it)               # decreasing step size
        a = a - eta * g
        a = _project_simplex(a)

        cur_score = 2.0 * n * _self_conv(a, fft_len_self).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] n={n:5d} subgrad it={it:5d} new best {best_score:.6f}")

        if best_score < 1.5025:
            break

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)

    # give a small safety margin so the harness never times out while we print
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Choose a few lengths to explore (always include the previous result).
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 3000, 3809, 4096]
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
            if seq0.size not in lengths:
                lengths.append(seq0.size)

    best_overall = None
    best_score_overall = float("inf")
    start_global = time.monotonic()

    # ------------------------------------------------------------------
    # Optimise each length in turn while time permits.
    # ------------------------------------------------------------------
    for n in sorted(lengths):
        if time.monotonic() >= deadline:
            break

        # --- initialise a probability vector for this length -------------
        if "height_sequence_1" in globals():
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq.size == n:
                a0 = seq.copy()
            else:
                # linear interpolation to the new size
                x_old = np.linspace(0.0, 1.0, seq.size, endpoint=False)
                x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(x_new, x_old, seq)
        else:
            a0 = rng.dirichlet(np.ones(n)).astype(np.float64)

        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s                                   # simplex normalisation

        # ---- run the optimiser for this length -------------------------
        vec, score = _optimise_length(a0, deadline, rng)

        if score < best_score_overall - 1e-12:
            best_score_overall = score
            best_overall = vec.copy()
            elapsed = time.monotonic() - start_global
            print(f"[{elapsed:6.2f}s] *** GLOBAL BEST n={n} score={best_score_overall:.6f}")

        # early exit if we already beat the target
        if best_score_overall < 1.5025:
            break

    # ------------------------------------------------------------------
    # Fallback – should never happen but guards against empty result.
    # ------------------------------------------------------------------
    if best_overall is None:
        best_overall = np.array([1.0], dtype=np.float64)
        best_score_overall = float("inf")

    # ------------------------------------------------------------------
    # Scale to the conventional normalisation Σ a_i = √(2 n) and clip.
    # The objective is invariant under scaling, this scaling only helps
    # readability and matches the notation used by the reference solution.
    # ------------------------------------------------------------------
    n_best = best_overall.size
    target_sum = np.sqrt(2.0 * n_best)
    scale = target_sum / best_overall.sum()
    best_overall = best_overall * scale
    best_overall = np.clip(best_overall, 0.0, 1000.0)

    return [float(x) for x in best_overall]
