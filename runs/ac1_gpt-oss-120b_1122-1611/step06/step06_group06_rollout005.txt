def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Grad of Σ_k w_k (a∗a)_k = 2·a ⋆ w (correlation)."""
    n = a.size
    L = w.size                # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad

assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (step‑function heights) with  
`score(a) = 2·n·max(conv(a,a)) / (sum(a)²)` as small as possible.  
Because the score is *scale invariant* we can restrict ourselves to vectors
with `sum(a)=1`; then the task is to minimise the maximal entry of the
self‑convolution.

The optimisation is carried out in three stages:

1. **Diverse initial pool** – analytic families (square‑root differences,
   power‑differences, exponentials, Beta‑like shapes) and random Dirichlet
   samples are normalised to sum = 1 and the best of them is selected.

2. **Smooth‑max L‑BFGS refinement** –  
   `f_β(a) = (1/β)·log Σ exp(β·(a∗a)_k)` is a smooth upper bound for
   `max(a∗a)`.  
   Using the soft‑max representation we obtain a closed‑form gradient
   `∇_x f_β` for the *softmax‑parameterised* vector
   `a = softmax(x)`.  A sequence of L‑BFGS runs with gradually increasing
   `β` (20 → 100 → 500 → 2000) drives the solution toward the true maximum
   while keeping the optimisation well‑behaved.

3. **Short simulated‑annealing polishing** – the best vector from step 2
   is fine‑tuned by cheap random mass‑transfer moves, accepted with a
   Metropolis rule.  A tiny smoothing step is performed every few thousand
   iterations.

All heavy work (convolutions, gradients) is performed with FFT‑based
`O(n log n)` kernels; the algorithm respects the 1000 s budget and prints
progress messages.

```python
"""
Propose a non‑negative step‑function (vector of heights) that minimises
evaluate_sequence().  The method builds a diverse pool of analytic and
random candidates, refines the best one with a smooth‑max L‑BFGS
optimiser (soft‑max parametrisation, explicit gradient via FFT), and finally
polishes the result with a brief simulated‑annealing phase.
"""

import time
import numpy as np

# optional: SciPy provides a fast L‑BFGS implementation
try:
    from scipy.optimize import minimize
except Exception:               # pragma: no cover
    minimize = None            # fallback to the mirror‑descent method


# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold the full linear
    convolution of two vectors of length n (i.e. length 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Exact a∗a using a real FFT (returns length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """Gradient of Σ_k w_k·(a∗a)_k = 2·a⋆w (correlation)."""
    n = a.size
    L = w.size                     # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)[: L + n - 1]
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a∗a) for a normalised (sum=1) non‑negative vector."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Simple helpers
# ----------------------------------------------------------------------
def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _normalise(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


# ----------------------------------------------------------------------
# Smooth‑max objective used by L‑BFGS
# ----------------------------------------------------------------------
class _SmoothMax:
    """
    Callable class  f(x) = (1/β)·log Σ exp(β·conv_i)
    together with its gradient w.r.t. the soft‑max parameters x.
    """

    def __init__(self, beta: float, n: int, fft_len: int):
        self.beta = beta
        self.n = n
        self.fft_len = fft_len

    def __call__(self, x: np.ndarray):
        """Return (objective, gradient) for optimizer."""
        a = _softmax(x)                     # a_i ≥0, Σa_i = 1
        conv = _conv_self(a, self.fft_len)  # length 2·n‑1

        # soft‑max weights over the convolution entries
        scaled = self.beta * conv
        m = np.max(scaled)
        exp_scaled = np.exp(scaled - m)     # stabilised
        sum_exp = exp_scaled.sum()
        w = exp_scaled / sum_exp

        # gradient w.r.t. a  (2·a⋆w)
        grad_a = _gradient_correlation(a, w)

        # chain rule through the soft‑max parametrisation
        dot = np.dot(a, grad_a)
        grad_x = a * (grad_a - dot)

        # smooth‑max value
        obj = (m + np.log(sum_exp)) / self.beta
        return obj, grad_x


# ----------------------------------------------------------------------
# Simulated‑annealing polishing (very cheap – uses exact max)
# ----------------------------------------------------------------------
def _simulated_annealing(best_a: np.ndarray,
                         best_score: float,
                         start_time: float,
                         deadline: float,
                         n: int,
                         fft_len: int,
                         rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """Short SA that moves mass between two positions; accepts with Metropolis rule."""
    cur = best_a.copy()
    cur_score = best_score
    T_start = 5e-3
    T_end = 1e-6
    it = 0

    while time.monotonic() < deadline:
        # linear cooling based on elapsed fraction of the SA window
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time) if deadline > start_time else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        i, j = rng.integers(0, n, size=2)
        if i == j or cur[i] < 1e-14:
            continue
        delta = rng.random() * cur[i] * 0.5  # move at most half of a[i]
        new_a = cur.copy()
        new_a[i] -= delta
        new_a[j] += delta

        new_score = _score_of(new_a, n, fft_len)

        improve = new_score < cur_score
        if improve or rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
            cur = new_a
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = new_a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        it += 1

        # gentle smoothing every few thousand steps
        if it % 5000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur, kernel, mode="same")
            smoothed = _normalise(smoothed)
            cur = smoothed
            cur_score = _score_of(cur, n, fft_len)

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 123,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises evaluate_sequence().
    The algorithm builds a rich initial pool, refines the best candidate
    with a smooth‑max L‑BFGS optimiser (increasing β) and finishes with a
    brief simulated‑annealing polish.  It respects the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size (use the length of a previously found solution if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = max(2, int(prev.size))
    else:
        n = kwargs.get("n", 3800)          # default size
        n = max(2, int(n))

    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Build a diverse pool of starting vectors (all normalised to sum = 1)
    # ------------------------------------------------------------------
    candidates = []

    # a) uniform
    candidates.append(np.full(n, 1.0 / n))

    # b) analytic √‑difference
    idx = np.arange(n, dtype=np.float64)
    a_sqrt = np.sqrt(idx + 1.0) - np.sqrt(idx)
    candidates.append(_normalise(a_sqrt))

    # c) power‑difference families
    for p in np.linspace(0.2, 2.0, 12):
        a = ((np.arange(n, dtype=np.float64) + 1) ** (-p)
             - (np.arange(n, dtype=np.float64) + 2) ** (-p))
        candidates.append(_normalise(a))

    # d) exponential families
    for lam in np.linspace(0.0005, 0.02, 12):
        a = (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))
        candidates.append(_normalise(a))

    # e) Beta‑like shapes  a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    for α in np.linspace(0.5, 4.0, 6):
        for β in np.linspace(0.5, 4.0, 6):
            i = np.arange(1, n + 1, dtype=np.float64)
            a = (i ** (α - 1.0)) * ((n + 1 - i) ** (β - 1.0))
            candidates.append(_normalise(a))

    # f) random Dirichlet samples
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0)
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(_normalise(a))

    # g) previous construction if it exists (pad / truncate to current n)
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1:
                if prev.size < n:
                    # pad with zeros
                    tmp = np.zeros(n, dtype=np.float64)
                    tmp[: prev.size] = prev
                    prev = tmp
                elif prev.size > n:
                    prev = prev[:n]
                candidates.append(_normalise(prev))
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Evaluate the pool (fast exact max‑convolution) and keep the best
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    for a in candidates:
        sc = _score_of(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    print(f"[init] best score from pool = {best_score:.9f}")

    # ------------------------------------------------------------------
    # L‑BFGS refinement using a smooth‑max surrogate
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # we allocate roughly the last 30 % of the budget to simulated annealing
    sa_deadline = start_time + budget_s
    lbf_deadline = start_time + 0.70 * budget_s

    # keep a sorted list of the most promising seeds (up to 12)
    pool_scores = [(_score_of(c, n, fft_len), c) for c in candidates]
    pool_scores.sort(key=lambda t: t[0])
    top_seeds = [c for (_, c) in pool_scores[:12]]

    # optimisation schedule (β values and iteration caps)
    beta_schedule = [(20.0, 80), (100.0, 60), (500.0, 40), (2000.0, 20)]

    epsilon = 1e-12  # to avoid log(0)

    for seed_a in top_seeds:
        if time.monotonic() >= lbf_deadline:
            break

        # initialise soft‑max parameters
        x = np.log(seed_a + epsilon)

        # run the cascade of L‑BFGS phases
        for beta, max_iter in beta_schedule:
            if time.monotonic() >= lbf_deadline:
                break

            obj = _SmoothMax(beta, n, fft_len)

            if minimize is not None:
                # L‑BFGS‑B with explicit gradient (returns (f, g))
                res = minimize(obj,
                               x,
                               method="L-BFGS-B",
                               jac=True,
                               options={"maxiter": max_iter,
                                        "ftol": 1e-12,
                                        "gtol": 1e-12,
                                        "disp": False})
                x = res.x
            else:
                # fallback: a few gradient‑descent steps (very crude)
                step = 0.2 / beta
                for _ in range(max_iter):
                    f, g = obj(x)
                    x -= step * g
                    step *= 0.995

        # evaluate the refined vector with the *exact* objective
        a_refined = _softmax(x)
        sc = _score_of(a_refined, n, fft_len)
        if sc < best_score - 1e-12:
            best_score = sc
            best = a_refined.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] L‑BFGS improvement → {best_score:.9f}")

    # ------------------------------------------------------------------
    # Final simulated‑annealing phase (uses the exact objective)
    # ------------------------------------------------------------------
    best, best_score = _simulated_annealing(best,
                                            best_score,
                                            start_time,
                                            sa_deadline,
                                            n,
                                            fft_len,
                                            rng)

    # ------------------------------------------------------------------
    # Return the best sequence (clip to the limits imposed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
```
