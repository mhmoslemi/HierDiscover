def _project_simplex(v):
    # v: np.ndarray
    # Return projection onto simplex {x >=0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

evaluate_sequence().

Algorithm
---------
1. Build a diverse pool of seed vectors:

   • power‑difference families a_i = (i+1)^‑p - (i+2)^‑p,
   • exponential families,
   • beta‑type families a_i ∝ (i+1)^{p-1}·(n-i)^{q-1},
   • random Dirichlet draws,
   • the previous solution if the global variable ``height_sequence_1`` exists.
   All seeds are normalised to sum = 1 (the objective is scale‑invariant).

2. Choose the best seed according to a fast FFT‑based evaluation
   2·n·max(conv(a,a)).

3. Refine the current best with a projected Adam optimiser on the
   *soft‑max* surrogate of the L∞‑objective:
   – at each iteration compute the self‑convolution,
   – convert it to a probability vector w by a soft‑max with a
     temperature β that grows from 0.1 to 5000,
   – the gradient of the surrogate is 2·a ⋆ w (correlation),
   – apply Adam’s adaptive momentum update,
   – project back onto the probability simplex.
   The step size η is linearly annealed from 0.5 to 0.001.
   Random block‑scaling and pairwise mass‑transfer perturbations are
   injected occasionally to escape stagnation.

4. When the main phase finishes (≈budget‑20 s), a short simulated‑annealing
   fine‑tuning is performed:
   – identify the lag k where the convolution attains its maximum,
   – move a small amount of mass from one of the two indices that
     contribute most to that lag to a random destination,
   – accept moves with a Metropolis rule (temperature decays to 1e‑6).

5. The best vector seen during the whole run is clipped to the limits

def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a convolution of length 2·n‑1."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Exact self‑convolution of a (length n) using FFT, result length 2·n‑1."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _score_of(a: np.ndarray, n: int, fft_len: int) -> float:
    """Objective for a normalised (sum=1) non‑negative vector."""
    conv = _conv_self(a, fft_len)
    return 2.0 * n * conv.max()


def _gradient_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k (a∗a)_k  =  2·a ⋆ w   (⋆ = correlation).
    a : shape (n,)
    w : shape (2·n‑1,)
    Returns an array of shape (n,).
    """
    n = a.shape[0]
    L = w.shape[0]               # = 2·n‑1
    fft_len = 1 << ((L + n - 2 - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a[::-1], n=fft_len)
    corr_full = np.fft.irfft(fw * fa_rev, n=fft_len)
    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero → return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _normalise(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if prev.ndim == 1 and prev.size > 0:
                candidates.append(_normalise(prev))
        except Exception:
            pass

    # c) Power‑difference families
    for p in np.linspace(0.2, 2.0, 15):
        a = ((np.arange(n, dtype=np.float64) + 1) ** (-p)
             - (np.arange(n, dtype=np.float64) + 2) ** (-p))
        candidates.append(_normalise(a))

    # d) Exponential families
    for lam in np.linspace(0.0005, 0.02, 15):
        a = (1.0 - np.exp(-lam)) * np.exp(-lam * np.arange(n, dtype=np.float64))
        candidates.append(_normalise(a))

    # e) Beta‑type families a_i ∝ (i+1)^{p-1}·(n-i)^{q-1}
    for _ in range(12):
        p = rng.uniform(0.2, 3.0)
        q = rng.uniform(0.2, 3.0)
        i = np.arange(n, dtype=np.float64) + 1.0
        a = (i ** (p - 1.0)) * ((n - i + 1.0) ** (q - 1.0))
        candidates.append(_normalise(a))

    # f) Random Dirichlet samples
    for _ in range(30):
        alpha = rng.uniform(0.1, 2.0)
        a = rng.gamma(alpha, 1.0, size=n)
        candidates.append(_normalise(a))

    # ------------------------------------------------------------------
    # 2) Pick the best seed (fast FFT‑based evaluation)
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    for a in candidates:
        sc = _score_of(a, n, fft_len)
        if sc < best_score:
            best_score = sc
            best = a.copy()
    print(f"[init] best seed score = {best_score:.9f}")

    # ------------------------------------------------------------------
    # 3) Adam‑based projected gradient descent on the soft‑max surrogate
    # ------------------------------------------------------------------
    cur = best.copy()
    m = np.zeros_like(cur)   # first moment
    v = np.zeros_like(cur)   # second moment
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # scheduling parameters
    step_start, step_end = 0.5, 0.001
    beta_start, beta_end = 0.1, 5000.0

    start_time = time.monotonic()
    main_deadline = start_time + max(0.0, float(budget_s) - 20.0)   # keep ~20 s for SA
    iteration = 0

    while time.monotonic() < main_deadline:
        # self‑convolution and current score
        conv = _conv_self(cur, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max

        # keep global best
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = cur.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.9f}")

        # schedule β and η
        now = time.monotonic()
        frac = (now - start_time) / (main_deadline - start_time) if main_deadline > start_time else 0.0
        beta = beta_start + (beta_end - beta_start) * frac
        step = step_start * (1.0 - frac) + step_end * frac

        # soft‑max weights over the full convolution
        w_raw = np.exp(beta * (conv - cur_max))   # stable because max shifted to zero
        w = w_raw / w_raw.sum()

        # gradient of the surrogate (correlation)
        grad = _gradient_correlation(cur, w)

        # Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        update = step * m_hat / (np.sqrt(v_hat) + eps)

        cur = cur - update
        cur = _project_simplex(cur)

        # occasional exploratory perturbations
        if rng.random() < 0.02:                      # random block scaling
            width = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.15))
            cur[start_idx:start_idx + width] *= factor
            cur = _project_simplex(cur)

        if rng.random() < 0.02:                      # pairwise mass transfer
            i, j = rng.integers(0, n, size=2)
            if i != j and cur[i] > 1e-12:
                delta = rng.random() * cur[i] * 0.3
                cur[i] -= delta
                cur[j] += delta
                cur = _project_simplex(cur)

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning (remaining budget)
    # ------------------------------------------------------------------
    sa_deadline = start_time + float(budget_s)
    cur_sa = best.copy()
    cur_sa_score = best_score
    T_start, T_end = 5e-3, 1e-6
    sa_iter = 0

    while time.monotonic() < sa_deadline:
        # exponential temperature cooling
        now = time.monotonic()
        frac = (now - main_deadline) / (sa_deadline - main_deadline) if sa_deadline > main_deadline else 0.0
        T = T_start * (1.0 - frac) + T_end * frac

        # ----- propose a move -----
        if rng.random() < 0.5:
            # targeted move guided by the current maximal lag
            conv = _conv_self(cur_sa, fft_len)
            k_max = int(conv.argmax())
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx_range = np.arange(lo, hi + 1, dtype=np.int64)
            contrib = cur_sa[idx_range] * cur_sa[k_max - idx_range]
            best_pair = idx_range[np.argmax(contrib)]
            i_src = best_pair
            j_src = k_max - best_pair
            src = i_src if cur_sa[i_src] > cur_sa[j_src] else j_src
            dest = rng.integers(0, n)
            while dest == src or dest == (k_max - src):
                dest = rng.integers(0, n)
            delta = rng.random() * cur_sa[src] * 0.4
            new_a = cur_sa.copy()
            new_a[src] -= delta
            new_a[dest] += delta
        else:
            # generic random transfer
            i, j = rng.integers(0, n, size=2)
            if i == j or cur_sa[i] < 1e-12:
                continue
            delta = rng.random() * cur_sa[i] * 0.4
            new_a = cur_sa.copy()
            new_a[i] -= delta
            new_a[j] += delta

        # evaluate the candidate (exact objective)
        new_score = _score_of(new_a, n, fft_len)

        # Metropolis acceptance
        improve = new_score < cur_sa_score
        if improve or rng.random() < np.exp(-(new_score - cur_sa_score) / max(T, 1e-12)):
            cur_sa = new_a
            cur_sa_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best = new_a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] SA improvement → {best_score:.9f}")

        sa_iter += 1

        # occasional gentle smoothing (helps flatten tiny spikes)
        if sa_iter % 4000 == 0:
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)
            smoothed = np.convolve(cur_sa, kernel, mode='same')
            smoothed = smoothed / smoothed.sum()
            cur_sa = smoothed
            cur_sa_score = _score_of(cur_sa, n, fft_len)

    # ------------------------------------------------------------------
    # 5) Return the best sequence (respect the clipping limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
