def _fft_len_for_n(n: int) -> int:
    """Smallest power‑of‑two length that can hold a linear convolution
    of two length‑n vectors (full length = 2·n‑1)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length = 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max with temperature β (numerically stable)."""
    mx = x.max()
    z = np.exp(beta * (x - mx))
    s = z.sum()
    if s == 0.0:
        return np.full_like(z, 1.0 / z.size)
    return z / s


def _gradient_via_correlation(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute 2·a ⋆ w  (linear correlation) where w has length 2·n‑1.
    This matches the sub‑gradient of Σ_k w_k (a∗a)_k.
    """
    n = a.size
    a_rev = a[::-1]
    L = w.size + n - 1
    fft_len = 1 << ((L - 1).bit_length())
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr_full = np.fft.irfft(fw * fa, n=fft_len)[:L]
    # slice the part that corresponds to the gradient (length n)
    return 2.0 * corr_full[n - 1 : n - 1 + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _gen_shape_r(r: float, n: int) -> np.ndarray:
    """Shape a_i = (i+1)^r - i^r  (0<r<1)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) ** r - i ** r
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a


def _gen_shape_p(p: float, n: int) -> np.ndarray:
    """Shape a_i = (i+1)^(-p) - (i+2)^(-p)  (p>0)."""
    k = np.arange(1, n + 2, dtype=np.float64)          # 1 .. n+1
    f = k ** (-p)
    a = f[:-1] - f[1:]
    s = a.sum()
    if s <= 0.0:
        return None
    a /= s
    return a

            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                cand = np.clip(cand, 0.0, None)
                if cand.sum() > 0.0:
                    cand = cand / cand.sum()
                    cur_score = 2.0 * cand.size * _conv_self(cand, _fft_len_for_n(cand.size)).max()
                    if cur_score < best_score:
                        best_a = cand.copy()
                        best_score = cur_score
                        n = cand.size
                        fft_len = _fft_len_for_n(n)
                        print(f"[init] height_sequence_1 score = {cur_score:.6f}")
        except Exception:
            pass

    # Ensure we start from a proper probability vector
    cur_a = best_a.copy()
    cur_a = _project_simplex(cur_a)

    # ------------------------------------------------------------------
    # 1) Soft‑max mirror‑descent phase (≈30 % of the budget)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep ~2 s safety margin

    total_budget = deadline - start_time
    soft_phase_frac = 0.30
    soft_deadline = start_time + soft_phase_frac * total_budget

    # scheduling constants
    eta_start, eta_end = 1.0, 0.005
    beta_start, beta_end = 0.2, 5000.0

    # auxiliary random‑move probabilities
    block_prob = 0.04
    pair_prob = 0.08
    restart_gap = 180.0      # seconds without improvement → jitter restart

    last_improve = start_time
    iteration = 0

    while time.monotonic() < soft_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear schedules for η and β
        eta = eta_start * (1.0 - elapsed / total_budget) + eta_end * (elapsed / total_budget)
        beta = beta_start + (beta_end - beta_start) * (elapsed / total_budget)

        # convolution and exact objective
        conv = _conv_self(cur_a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] soft‑phase new best = {best_score:.6f}")

        # soft‑max weights (smooth approximation of max)
        w = _softmax(conv, beta)

        # gradient of smooth surrogate (2·a⋆w)
        grad = _gradient_via_correlation(cur_a, w)

        # exponentiated‑gradient update on the simplex (mirror descent)
        factor = np.exp(-eta * grad)
        cur_a = cur_a * factor
        cur_a = np.clip(cur_a, 0.0, None)
        cur_a = cur_a / cur_a.sum()

        # -------- random exploration moves ----------
        if rng.random() < block_prob:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.25))
            cur_a[start:start + width] *= mult
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = cur_a / cur_a.sum()

        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > 1e-14:
                delta = rng.random() * cur_a[i] * 0.5
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = np.clip(cur_a, 0.0, None)
                cur_a = cur_a / cur_a.sum()

        # -------- stagnation restart ----------
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            cur_a = best_a * jitter
            cur_a = _project_simplex(cur_a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] soft‑phase restart (stagnation)")

        iteration += 1

    # ------------------------------------------------------------------
    # 2) Hard‑max projected‑gradient descent phase (remaining budget)
    # ------------------------------------------------------------------
    # reset some schedules for the second phase
    hard_start_iter = iteration
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # decay step size as 1/sqrt(iteration)
        step = 0.5 / np.sqrt(iteration - hard_start_iter + 1)

        # convolution and objective
        conv = _conv_self(cur_a, fft_len)
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] hard‑phase new best = {best_score:.6f}")

        # identify all active maximal indices (within a tiny tolerance)
        tol = 1e-14
        active = np.where(conv >= cur_max - tol)[0]
        w = np.zeros_like(conv)
        if active.size > 0:
            w[active] = 1.0 / active.size
        else:
            # fallback (should never happen)
            w[np.argmax(conv)] = 1.0

        # sub‑gradient of the true max (averaged over active indices)
        grad = _gradient_via_correlation(cur_a, w)

        # gradient step & simplex projection
        tentative = cur_a - step * grad
        cur_a = _project_simplex(tentative)

        # ----------------- random jumps (same as before) -----------------
        if rng.random() < block_prob:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            mult = np.exp(rng.normal(0.0, 0.25))
            cur_a[start:start + width] *= mult
            cur_a = np.clip(cur_a, 0.0, None)
            cur_a = cur_a / cur_a.sum()

        if rng.random() < pair_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_a[i] > 1e-14:
                delta = rng.random() * cur_a[i] * 0.5
                cur_a[i] -= delta
                cur_a[j] += delta
                cur_a = np.clip(cur_a, 0.0, None)
                cur_a = cur_a / cur_a.sum()

        # ---------- stagnation based restart ----------
        if now - last_improve > restart_gap:
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            cur_a = best_a * jitter
            cur_a = _project_simplex(cur_a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] hard‑phase restart (stagnation)")

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best candidate (already normalised, safe for evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)    # respect evaluate_sequence limits
    return [float(v) for v in best_a]
