import time, numpy as np, scipy.sparse as sp, scipy.optimize as opt

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Construct a non‑negative step‑function that (approximately) minimises
    ``evaluate_sequence``.
    
    The algorithm uses a simple Frank‑Wolfe style iteration based on a
    linear‐programming sub‑problem:
    
    * Start from a simple power‑law distribution a (∑a = 1).
    * Compute M = max(a*a) (the ℓ∞‑norm of the self‑convolution).
    * Solve the LP   maximise 1ᵀb   subject to   A·b ≤ M ,  b ≥ 0,
      where A is the convolution matrix of a (so A·b = a∗b).
      Any feasible b has the same ℓ∞‑bound as a.  If the optimum
      sum(b) = s > 1 then the normalised vector g = b / s satisfies
      max(g∗g) = M / s² < M, i.e. we obtain a strict improvement.
    * Replace a ← g and repeat until s is ≈ 1 or the time budget expires.
    
    Multiple values of ``n`` and random initial exponents are tried;

* a is a probability vector (sum = 1, a_i ≥ 0) of length n.
