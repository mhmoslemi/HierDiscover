        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # --------------------------------------------------------------
        # 4.5 – occasional global jitter (≈0.5 % of iterations)
        # --------------------------------------------------------------
        if rng.random() < 0.005:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 4.6 – keep the best solution seen so far
        # --------------------------------------------------------------
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 4.7 – soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # jitter the best solution and continue
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits imposed by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
