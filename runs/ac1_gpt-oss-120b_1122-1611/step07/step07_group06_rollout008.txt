def simulated_annealing_sparse(n, k, time_limit, seed, initial_support=None):
    rng = np.random.default_rng(seed)
    if initial_support is None:
        support = set(rng.choice(n, size=k, replace=False))
    else:
        support = set(initial_support)
    # Represent as binary vector
    a = np.zeros(n, dtype=np.float64)
    a[list(support)] = 1.0/k
    # Compute conv indicator for binary vector? Use binary 0/1.
    # We'll compute eigen conv each loop.
    best_a = a.copy()
    best_obj = compute_obj(a)
    # SA schedule:
    t0 = time.monotonic()
    end_time = t0 + time_limit
    # parameters:
    temp_start = best_obj * 0.5  # initial temperature based on objective magnitude
    temp_end = best_obj * 0.001
    total_iter = 0
    while time.monotonic() < end_time:
        # compute time fraction
        frac = (time.monotonic() - t0) / (time_limit)
        temp = temp_start * (1-frac) + temp_end * frac
        # propose move: randomly pick i in support, j not in support
        i = rng.choice(list(support))
        # choose j random not in support
        # may choose from complement
        j = rng.integers(0, n)
        while j in support:
            j = rng.integers(0, n)
        # apply swap
        support_new = support.copy()
        support_new.remove(i)
        support_new.add(j)
        a_new = np.zeros_like(a)
        a_new[list(support_new)] = 1.0/k
        obj_new = compute_obj(a_new)
        delta = obj_new - best_obj # Actually compare to current
        # acceptance rule:
        if obj_new < best_obj or rng.random() < np.exp(-delta / temp):
            # accept
            a = a_new
            support = support_new
            if obj_new < best_obj:
                best_obj = obj_new
                best_a = a_new.copy()
        total_iter += 1
        # occasional logging:

"""
Simulated‑annealing + short Adam refinement for the step‑function bound.

Stage 1 (discrete) – we search for a set of indices S (|S| = k) such that
the integer convolution count   c = 1_S * 1_S   has the smallest possible
maximum entry.  For a binary support the evaluation function reduces to

        obj = 2·n·max(c) / k²

so minimising max(c) suffices.  A simple SA loop repeatedly swaps a
support element with a non‑support element and accepts the move
according to a linearly‑annealed temperature.  Several values of k are
tried until the global time budget (≈ 1000 s) is exhausted.

Stage 2 (continuous) – the best binary support is turned into a real
vector (initially uniform weights 1/k) and a few hundred Adam updates
with the usual soft‑max surrogate of the “max‑convolution” are applied.
During these updates the support is kept fixed (all non‑support entries
are forced to zero and the remaining weights are renormalised).

The best exact score seen (via the provided evaluate_sequence) is kept
and finally returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only works for 1‑D arrays.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w.sum()
    return w


def _softmax_weights(c: np.ndarray, tau: float) -> np.ndarray:
    """Numerically stable soft‑max on a vector c with temperature tau."""
    shift = c.max()
    exp_shifted = np.exp((c - shift) / tau)
    total = exp_shifted.sum()
    if total == 0.0:
        return np.full_like(c, 1.0 / len(c))
    return exp_shifted / total


def _max_pair_count(binary: np.ndarray, fft_len: int) -> int:
    """
    Return max_{s} (binary * binary)[s] – the integer maximum of the
    ordered‑pair convolution.  The result is rounded to the nearest
    integer to avoid floating‑point noise.
    """
    a_fft = np.fft.rfft(binary, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * len(binary) - 1]
    return int(np.rint(conv.max()))


def _objective_from_maxcount(max_cnt: int, n: int, k: int) -> float:
    """Exact objective for a binary support of size k."""
    return 2.0 * n * max_cnt / (k * k)


# ----------------------------------------------------------------------
# Stage 1 – discrete simulated annealing on a binary support
# ----------------------------------------------------------------------


def _simulated_annealing_binary(
    n: int,
    k: int,
    time_budget: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Perform SA for a given support size k.
    Returns the best binary vector (0/1) found and its objective value.
    """
    # FFT length for convolutions of binary vectors (full length = 2n‑1)
    fft_len = _next_pow2(2 * n)

    # initialise a random support
    support = list(rng.choice(n, size=k, replace=False))
    binary = np.zeros(n, dtype=np.float64)
    binary[support] = 1.0

    cur_max = _max_pair_count(binary, fft_len)
    cur_obj = _objective_from_maxcount(cur_max, n, k)

    best_binary = binary.copy()
    best_obj = cur_obj
    best_max = cur_max

    start = time.monotonic()
    deadline = start + time_budget

    # temperature schedule (linear)
    temp_start = 1.0
    temp_end = 0.001

    # SA main loop
    iter_cnt = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        frac = elapsed / time_budget
        temp = temp_start * (1.0 - frac) + temp_end * frac

        # propose a swap (one element leave support, one enters)
        src_idx = rng.integers(k)            # index in support list
        src = support[src_idx]                # actual position to remove
        dst = rng.integers(n)
        while binary[dst] == 1.0:
            dst = rng.integers(n)

        # perform swap in place
        binary[src] = 0.0
        binary[dst] = 1.0

        new_max = _max_pair_count(binary, fft_len)
        new_obj = _objective_from_maxcount(new_max, n, k)

        delta = new_obj - cur_obj
        accept = False
        if delta <= 0.0:
            accept = True
        else:
            # Metropolis acceptance
            if rng.random() < math.exp(-delta / (temp + 1e-12)):
                accept = True

        if accept:
            # keep the change
            support[src_idx] = dst
            cur_max = new_max
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_binary = binary.copy()
                best_max = new_max
        else:
            # revert swap
            binary[dst] = 0.0
            binary[src] = 1.0

        iter_cnt += 1

        # occasional cheap logging
        if iter_cnt % 5000 == 0:
            now = time.monotonic()
            elapsed = now - start
            print(
                f"[{elapsed:6.2f}s] SA(k={k}) – cur obj {cur_obj:.6f}, "
                f"best obj {best_obj:.6f}"
            )

        # early exit if we already hit a theoretically optimal bound
        # (max count cannot be lower than ceil(k²/(2n))) – stop when we reach it
        theoretical_min = math.ceil(k * k / (2 * n))
        if best_max <= theoretical_min:
            break

    return best_binary, best_obj


# ----------------------------------------------------------------------
# Stage 2 – continuous refinement with Adam (soft‑max surrogate)
# ----------------------------------------------------------------------


def _adam_refine(
    a: np.ndarray,
    mask: np.ndarray,
    time_budget: float,
    rng: np.random.Generator,
    initial_best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Short Adam optimisation that keeps the mask fixed.
    Returns the refined vector and the best exact score seen.
    """
    n = len(a)
    fft_len = _next_pow2(3 * n)  # enough for correlation length
    corr_start = n - 1
    corr_end = corr_start + n

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = 1.0, 1.0

    # temperature / learning‑rate schedules (linear)
    tau_start, tau_end = 0.20, 0.001
    lr_start, lr_end = 0.30, 0.001

    start = time.monotonic()
    deadline = start + time_budget

    best_vec = a.copy()
    best_score = initial_best_score
    best_obj = float("inf")
    it = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac_total = (now - start) / (deadline - start)

        # anneal
        tau = tau_start * (1.0 - frac_total) + tau_end * frac_total
        lr = lr_start * (1.0 - frac_total) + lr_end * frac_total

        # forward convolution (self‑convolution)
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # soft‑max weights over the convolution entries
        w = _softmax_weights(conv, tau)

        # correlation = gradient of the smooth surrogate
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        grad = (2.0 / tau) * corr_full[corr_start:corr_end]

        # Adam update (projected onto the mask later)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # keep non‑negative and respect the mask
        a = np.where(mask, np.maximum(a, 0.0), 0.0)

        # renormalise the non‑zero part to sum = 1 (simple simplex projection)
        total = a.sum()
        if total == 0.0:
            a[mask] = 1.0 / mask.sum()
        else:
            a /= total

        it += 1

        # occasional exact evaluation
        if it % 200 == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score:
                best_score = exact
                best_vec = a.copy()
                elapsed = time.monotonic() - start
                print(
                    f"[{elapsed:6.2f}s] Adam refinement – new exact = {best_score:.6f}"
                )

    return best_vec, best_score

        if remaining < 10.0:  # not enough time for another run
            break
        run_time = min(sa_per_k, remaining - 5.0)  # keep a small buffer
        print(f"[{time.monotonic():.2f}s] SA start – n={n}, k={k}, time={run_time:.1f}s")
        binary_best, obj_best = _simulated_annealing_binary(
            n=n, k=k, time_budget=run_time, rng=rng
        )
        # convert binary to a real vector (uniform weights)
        a_candidate = binary_best.astype(np.float64) / k
        # quick exact evaluation
        try:
            exact = evaluate_sequence(a_candidate.tolist())
        except Exception:
            exact = float("inf")
        if exact < best_score:
            best_score = exact
            best_seq = a_candidate.copy()
            print(
                f"[{time.monotonic():.2f}s] New best after SA (k={k}) – exact = {best_score:.6f}"
            )
        # keep the binary mask for possible refinement later
        mask_best = binary_best.astype(bool)

        # ------------------------------------------------------------------
        # Stage 2 – short Adam refinement on this promising support
        # ------------------------------------------------------------------
        time_left = overall_deadline - time.monotonic()
        if time_left < 8.0:
            break
        refine_time = min(8.0, time_left)  # a few seconds per support
        refined_vec, refined_score = _adam_refine(
            a_candidate,
            mask_best,
            time_budget=refine_time,
            rng=rng,
            initial_best_score=best_score,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_vec.copy()
            print(
                f"[{time.monotonic():.2f}s] New best after refinement – exact = {best_score:.6f}"
            )
        # if we already beat the target, we can stop early
        if best_score < 1.5025:
            print("[early exit] Target bound reached.")
            break

    # ------------------------------------------------------------------
    # Final safety: if nothing was found, fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Return as a plain Python list of floats
    return [float(x) for x in best_seq]
