def project_onto_simplex(v):
    # v is np.ndarray
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

import time, math, numpy as np

def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def _fft_convolve(a, b, size):
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    conv = np.fft.irfft(fa * fb, size)
    return conv[: a.size + b.size - 1]

def _self_conv(a, size):
    return _fft_convolve(a, a, size)

def _project_simplex(v):
    # v : np.ndarray
    # Return projection onto simplex sum=1, v_i>=0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv/ind > 0
    if not np.any(cond):
        # all <=0, return uniform
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_from_uniform_psd(n):
    # Construct initial a using spectral sqrt of target uniform convolution.
    L = _next_pow2(2 * n - 1)
    target = np.concatenate([np.full(2*n-1, 1.0/(2*n-1)), np.zeros(L - (2*n-1))])
    # DFT
    Tf = np.fft.rfft(target)
    # sqrt magnitude (complex sqrt)
    Af = np.sqrt(Tf)
    # Inverse transform
    a_full = np.fft.irfft(Af, n=L)
    # take first n entries
    a = a_full[:n]
    # ensure nonnegativity
    a = np.clip(a, 0.0, None)
    if a.sum() == 0.0:
        a = np.full(n, 1.0/n)
    else:
        a /= a.sum()
    return a
```

    best = a.copy()
    best_score = cur_score
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0
    last_improve = start
    # hyper schedule params
    alpha_start = 0.5
    alpha_end = 80.0
    lr_start = 0.5
    lr_end = 0.001
    # loop
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / (deadline - start)  # relative progress
        # schedules
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lr = lr_start * (1 - frac) + lr_end * frac

"""
Propose a step‑function using a projected‑Adam optimiser on a smooth‑max
approximation of the peak of the self‑convolution.

Algorithm
---------
1. **Initialisation** – for each candidate length `n` we generate a start
   vector by taking the square‑root of the DFT of a *flat* target
   convolution (so that the target autocorrelation is (almost) constant).
   The vector is clipped to non‑negative values and projected onto the
   probability simplex.

2. **Smooth‑max + Adam** – at each iteration we
   * compute the full self‑convolution `c = a*a` (FFT length ≥ 3n‑2);
   * form soft‑max weights `w ∝ exp(α·c)` (α grows from 0.5 to ≈80);
   * obtain the gradient of the smoothed peak
         g = 2·(a * w)  (linear convolution via FFT);
   * perform an Adam step `a ← a – η·ĝ` with a learning‑rate η that
     decays from about 0.5 to 0.001;
   * project the result back onto the simplex (sum = 1, a_i ≥ 0).

3. **Exploration** – with a tiny probability we apply a multiplicative
   log‑normal block perturbation; if no improvement is observed for
   ≈25 s we restart from the current best with a small random jitter.

4. **Multi‑scale search** – several lengths (1024 … 16384) are tried
   sequentially; the best sequence over all lengths is kept.

5. **Budget handling** – the loop runs until the global deadline
   (1000 s – 5 s safety margin).  The best vector found so far is
   returned as a plain Python list of floats.

The implementation relies only on NumPy (FFT‑based O(n log n) operations)
and works comfortably within the allowed time budget on a single CPU.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _linear_convolution_fft(a: np.ndarray, b: np.ndarray, L: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of size
    ``L`` (must be ≥ len(a)+len(b)-1).  Returns the full convolution of
    length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)
    return conv[: a.size + b.size - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a re‑normalisation onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_from_uniform_psd(n: int) -> np.ndarray:
    """
    Construct a starting vector of length ``n`` by taking the square‑root of
    the DFT of a flat target convolution (length 2n‑1).  The result is
    projected onto the simplex.
    """
    L = _next_pow2(2 * n - 1)
    # target linear convolution (flat, normalised)
    target = np.concatenate(
        [np.full(2 * n - 1, 1.0 / (2 * n - 1)), np.zeros(L - (2 * n - 1))]
    )
    Tf = np.fft.rfft(target)
    Af = np.sqrt(Tf)               # spectral square‑root
    a_full = np.fft.irfft(Af, n=L)
    a = a_full[:n]
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a.astype(np.float64)

# ----------------------------------------------------------------------


def _optimize_length(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Optimise a vector of length ``n`` (sum = 1, a_i ≥ 0) to minimise the
    evaluation function.  Returns the best vector found and its score.
    """
    start = time.monotonic()

    L = _next_pow2(3 * n - 2)

    a = init_a.copy()

    c_full = np.fft.irfft(fa * fa, L)
    c = c_full[: 2 * n - 1]

    best_score = cur_score

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    eps = 1e-8
    t_iter = 0

    lr_start, lr_end = 0.5, 0.001

        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        frac = elapsed / total if total > 0 else 1.0

        # smooth‑max temperature and learning rate
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        lr = lr_start * (1.0 - frac) + lr_end * frac

        c_max = c.max()

        # Adam update (additive)
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a_new = _project_simplex(a_new)

        # evaluate candidate
        fa_new = np.fft.rfft(a_new, L)
        c_full_new = np.fft.irfft(fa_new * fa_new, L)
        c_new = c_full_new[: 2 * n - 1]
        new_score = 2.0 * n * c_new.max()

        # simple Metropolis acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # occasional uphill move (linear cooling)
            temperature = max(0.1 * (1.0 - frac), 1e-12)
            if rng.random() < math.exp(-(new_score - cur_score) / temperature):
                accept = True

        if accept:
            a = a_new
            fa = fa_new
            c = c_new
            cur_score = new_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] n={n} new best score = {best_score:.6f}")

        # occasional random block perturbation
        if rng.random() < 0.015:
            a = _random_block_perturb(a, rng)
            fa = np.fft.rfft(a, L)
            c_full = np.fft.irfft(fa * fa, L)
            c = c_full[: 2 * n - 1]
            cur_score = 2.0 * n * c.max()

        # restart after stagnation
        if now - last_improve > 25.0:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = best * noise
            a = _project_simplex(a)
            fa = np.fft.rfft(a, L)
            c_full = np.fft.irfft(fa * fa, L)
            c = c_full[: 2 * n - 1]
            cur_score = 2.0 * n * c.max()
            last_improve = now
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            print(f"[{elapsed:6.2f}s] n={n} restart (stagnation)")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the allotted time budget.  The routine
    tries several problem sizes, runs a projected‑Adam optimiser with a
    smooth‑max surrogate of the peak autocorrelation, and keeps the best
    sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # 5 s safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_overall = None
    best_score = float("inf")

    # Candidate lengths (powers of two, but not too large for the time budget)
    lengths = [1024, 2048, 4096, 8192, 16384]

    # If a previous construction is available, try its length first
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq.ndim == 1 and seq.size > 0 and np.isfinite(seq).all():
            seq = np.clip(seq, 0.0, None)
            s = seq.sum()
            if s > 0.0:
                seq = seq / s
            else:
                seq = np.full(seq.size, 1.0 / seq.size)
            # prepend its length
            lengths = [seq.size] + [l for l in lengths if l != seq.size]
            init_for_len = {seq.size: seq}
        else:
            init_for_len = {}
    else:
        init_for_len = {}

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 12.0:  # not enough time for a useful run
            break

        # initial vector for this length
        if n in init_for_len:
            a0 = init_for_len[n].copy()
        else:
            a0 = _initial_from_uniform_psd(n)

        # optimise for the current length using whatever time is left
        sub_deadline = deadline  # run until global deadline
        best_n, score_n = _optimize_length(n, a0, sub_deadline, rng)

        if score_n < best_score:
            best_score = score_n
            best_overall = best_n.copy()
            print(f"[global] new overall best = {best_score:.6f} (n={n})")

    # Fallback – should never happen, but keep a uniform distribution just in case
    if best_overall is None:
        n = 1024
        best_overall = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by ``evaluate_sequence`` and return a Python list
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    return [float(x) for x in best_overall]
