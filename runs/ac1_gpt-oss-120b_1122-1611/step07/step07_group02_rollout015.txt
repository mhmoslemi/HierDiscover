        # ------------------------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        src = a[i]
        if src <= 0.0:
            continue
        # transfer at most `step_scale * src` (keeps a[i] non‑negative)
        delta = rng.random() * step_scale * src
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        # ------------------------------------------------------------------
        # Compute the new score (exact via FFT)
        # ------------------------------------------------------------------
        prop_score = _score_from_a(a_prop, fft_len)

        # ------------------------------------------------------------------
        # Acceptance test (Metropolis)
        # ------------------------------------------------------------------
        accept = False
        if prop_score < cur_score:
            accept = True
        else:
            # uphill move – accept with probability exp(-(Δ)/T)
            if temperature > 0.0:
                prob = np.exp(-(prop_score - cur_score) / temperature)
                if rng.random() < prob:
                    accept = True

        if accept:
            a = a_prop
            cur_score = prop_score

        # ------------------------------------------------------------------
        # Update global best
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}   "
                  f"(it={it})")

        # ------------------------------------------------------------------
        # Temperature schedule
        # ------------------------------------------------------------------
        temperature *= cooling_rate
        # gradually shrink the step size with temperature
        step_scale = max(0.001, step_scale * cooling_rate)

        # ------------------------------------------------------------------
        # Hard restart if we have been stagnant for a while
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > RESTART_WAIT:
            # re‑initialise around the current best point with a jitter
            a = best_a.copy()
            jitter = rng.normal(0.0, 0.02, size=n)
            a = np.maximum(0.0, a + jitter)
            a /= a.sum()
            cur_score = _score_from_a(a, fft_len)
            temperature = 0.15
            step_scale = 0.08
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # ------------------------------------------------------------------
    # OPTIONAL: short Adam refinement on the log‑softmax parametrisation.
    # This often removes the few remaining “rough edges’’ produced by SA.
    # ------------------------------------------------------------------
    # The refinement is performed only if we still have a few seconds left.
    # ------------------------------------------------------------------
    refinement_deadline = deadline - 5.0
    if time.monotonic() < refinement_deadline:
        # initialise logits from the current best vector
        eps = 1e-12
        x = np.log(best_a + eps)               # logits → softmax probabilities

        # Adam state
        m = np.zeros_like(x)
        v = np.zeros_like(x)

        # hyper‑parameters
        lr = 0.5
        beta1 = 0.9
        beta2 = 0.999
        adam_eps = 1e-8

        beta_start = 5.0
        beta_end = 200.0
        beta_ramp_iters = 2000

        it_ref = 0
        while time.monotonic() < refinement_deadline:
            it_ref += 1

            # logits → probabilities (sum = 1)
            a_ref = np.exp(x - np.max(x))
            a_ref /= a_ref.sum()

            # convolution via FFT
            a_fft = np.fft.rfft(a_ref, n=fft_len)
            conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

            # exact score (used only for bookkeeping)
            cur = 2.0 * n * conv_full.max()
            if cur < best_score - 1e-12:
                best_score = cur
                best_a = a_ref.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] refined best = {best_score:.6f}")

            # smooth surrogate of max via log‑sum‑exp
            beta = (beta_end if it_ref > beta_ramp_iters
                    else beta_start + (beta_end - beta_start) * it_ref / beta_ramp_iters)
            scaled = beta * conv_full
            lse = np.logaddexp.reduce(scaled)      # log‑sum‑exp
            w = np.exp(scaled - lse)                # weights, sum to 1

            # gradient w.r.t. a – correlation w ⋆ a
            w_fft = np.fft.rfft(w, n=_next_pow2(3 * n - 2))
            a_fft_corr = np.fft.rfft(a_ref, n=_next_pow2(3 * n - 2))
            corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr),
                                     n=_next_pow2(3 * n - 2))
            start_idx = n - 1
            grad_a = 2.0 * corr_full[start_idx:start_idx + n]

            # chain rule: gradient w.r.t. logits
            dot = np.dot(grad_a, a_ref)
            grad_x = a_ref * (grad_a - dot)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** it_ref)
            v_hat = v / (1.0 - beta2 ** it_ref)
            x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # occasional jitter to avoid plateaus
            if it_ref % 400 == 0:
                x += rng.normal(0.0, 0.02, size=n)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
