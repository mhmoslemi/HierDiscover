def _grad_max_convolution(a, mask):
    # mask shape length 2n-1.
    # reverse mask and convolve with a
    n = a.shape[0]
    L = _next_pow2(n + mask.size - 1)  # likely ~ next pow2(3n)
    # Use rfft for real
    fa = np.fft.rfft(a, L)
    fm = np.fft.rfft(mask[::-1], L)  # reversed
    grad_full = np.fft.irfft(fa * fm, L)
    # The part needed: indices from mask.size-1 to mask.size-1 + n-1
    offset = mask.size - 1
    grad = 2.0 * grad_full[offset: offset + n]
    return grad

"""
Hybrid optimisation for the step‑function problem.

The objective to minimise is

    J(a) = 2·n·max(a*a) / (∑a)²

Since the scaling of *a* does not change J we keep the L¹‑norm equal to 1.
The algorithm works in three stages

0. initialise from the best known construction (``height_sequence_1``) or a
   uniform start,
1. early phase – smooth soft‑max approximation of the max together with an
   exponentiated‑gradient update (fast reduction of the dominant peak),
2. late phase – exact sub‑gradient of the max (the mask of the maximal
   convolution entries) with a back‑tracking line‑search and projection onto
   the probability simplex.

Occasional random block perturbations and a simple restart strategy avoid
getting stuck in plateaus.  All operations are O(n log n) thanks to FFTs, so
the routine can perform many thousands of iterations inside the 1000 s
budget.

The function ``propose_candidate`` returns the best sequence found (as a plain
list of floats) and is ready to be called by the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # uniform vector is a safe fallback
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by L¹‑normalisation.
    """
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10) + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# ----------------------------------------------------------------------
