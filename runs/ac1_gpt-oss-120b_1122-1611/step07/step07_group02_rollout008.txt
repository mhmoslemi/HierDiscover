import time
import numpy as np
from scipy.special import logsumexp

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    m = np.max(logits)
    exps = np.exp(logits - m)
    s = exps.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exps / s

def _conv_max(a: np.ndarray, fft_len: int) -> float:
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2 * a.size - 1]
    return float(conv.max())

def _objective(a: np.ndarray, fft_len: int) -> float:
    s = a.sum()
    if s < 0.01:
        return np.inf
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2 * a.size - 1]
    return float(2.0 * a.size * conv.max() / (s * s))

                best_a = a.copy()

            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)

                best_a = a_local.copy()
                last_improve = time.monotonic()

                print(f"[{elapsed:6.2f}s] new best (hill‑climb) = {best_score:.6f}")

        # ------------------------------------------------------------------

        m = beta1 * m + (1.0 - beta1) * grad_x

    # ----------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
