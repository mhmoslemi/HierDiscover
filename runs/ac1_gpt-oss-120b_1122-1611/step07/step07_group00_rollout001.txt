def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)

    # initial size n: from init sequence if provided else 5000
    init_seq = globals().get('height_sequence_1')
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            n = a0.size
            a0 = np.clip(a0, 0., None)
            # Normalize to sum 1
            if a0.sum() > 0:
                a0 /= a0.sum()
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            x = np.log(a0 + 1e-12)
        else:
            # fallback
            n = 5000
            x = np.zeros(n)
    else:
        n = 5000
        x = np.zeros(n)

    # Precompute FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n - 2)   # maybe not needed; but for gradient we need conv of len (2n-1) and reversed a length n; length = n+(2n-1)-1 = 3n-2

    # Adam parameters
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; eps=1e-8; t=0

    # Track best
    def a_from_x(x):
        return _softmax(x)

    a = a_from_x(x)
    c = _self_conv(a, fft_len_self)
    best_a = a.copy()
    best_score = 2.0 * n * c.max() / (a.sum()**2)
    last_improve = time.monotonic()
    start = time.monotonic()
    total_time = deadline - start

    # hyperparams
    lr_start=1.0
    lr_end=0.01

    # We can precompute target average of conv:
    # Since sum a = 1, avg = 1/(2n-1)

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total_time

        a = a_from_x(x)

        # compute conv
        c = _self_conv(a, fft_len_self)

        # compute loss J = sum((c - avg)^2)
        avg = 1.0/(2*n - 1)  # sum a =1
        r = c - avg

        # compute gradient w.r.t a (quartic)
        # use _gradient_wrt_a returns 2*conv; we need 4*conv = 2*_gradient_wrt_a
        g_a = 2.0 * _gradient_wrt_a(a, r)   # shape n

        # convert to gradient w.r.t x via softmax Jacobian
        avg_grad = np.dot(a, g_a)
        g_x = a * (g_a - avg_grad)

        # Adam step
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # compute score (exact objective)
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best score: {best_score:.6f}")

        # stagnation restart
        if now - last_improve > 30.0:
            # restart from best + noise
            noise = rng.normal(0.0, 0.3, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0); v.fill(0.0); t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart due to stagnation")

        # occasional jitter: random block perturb
        if rng.random() < 0.02:
            # perturb best_a
            a_jit = _random_block_perturb(best_a, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0); v.fill(0.0); t = 0

    # after loop, return best_a as list floats
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]

"""
Improved search for a step‑function that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm has two phases:

1. **Variance‑flattening phase** – minimise the L²‑distance of the
   self‑convolution to the constant value 1/(2·n‑1).  This drives the
   convolution towards a flat shape and typically reduces the maximum
   far below the previous bound (≈1.513).

2. **Smoothed‑max phase** – starting from the best result of phase 1 we
   switch to a log‑sum‑exp surrogate of the true max (the same idea used
   in the reference code) but now with a much flatter starting point.
   A short fine‑tuning run finishes the optimisation.

Both phases use an Adam optimiser on the soft‑max parametrisation
`a = softmax(x)`, which automatically enforces non‑negativity and unit
sum.  Restarts on stagnation, occasional block‑jitter and a simple
learning‑rate schedule keep the search robust.  The routine respects the
global time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that gives a
    low value of `evaluate_sequence`.  Returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Choose a length and initialise log‑weights
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)            # start from log‑weights
        else:
            n = 5000
            x = np.zeros(n, dtype=np.float64)
    else:
        n = 5000
        x = np.zeros(n, dtype=np.float64)

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (used repeatedly)
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)               # for a*a
    # length needed for correlation a_rev * w (w has length 2n-1)
    fft_len_corr = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 3) Adam optimiser state (shared by both phases)
    # --------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # --------------------------------------------------------------
    # 4) Helper converting log‑weights to probabilities
    # --------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # --------------------------------------------------------------
    # 5) Tracking the best solution seen so far
    # --------------------------------------------------------------
    a = _a_from_x(x)
    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * c.max() / (a.sum() ** 2)
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 6) Phase split (variance‑flattening → smoothed‑max)
    # --------------------------------------------------------------
    # 60 % of the budget for the first phase, the rest for fine‑tuning.
    phase2_start = start_time + 0.6 * total_time
    phase2 = False

    # --------------------------------------------------------------
    # 7) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # switch phase if needed
        if not phase2 and now >= phase2_start:
            phase2 = True
            # re‑initialise Adam (helps when the loss landscape changes)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] Switching to smoothed‑max refinement")

        # ----------------------------------------------------------
        # common: obtain current a and its self‑convolution
        # ----------------------------------------------------------
        a = _a_from_x(x)                     # probability vector, sum = 1
        c = _self_conv(a, fft_len_self)      # length 2·n‑1

        # ----------------------------------------------------------
        # evaluate exact score (the quantity we finally care about)
        # ----------------------------------------------------------
        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # ----------------------------------------------------------
        # record improvement
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # choose the appropriate loss and its gradient
        # ----------------------------------------------------------
        if not phase2:
            # ------- Phase 1 : minimise variance of the convolution -------
            # target (average) value of the convolution
            avg = 1.0 / (2 * n - 1)               # because sum(a)=1
            r = c - avg                           # residual vector (length 2n‑1)

            # gradient w.r.t. a of J = Σ (c‑avg)²  = 2·_gradient_wrt_a(r)
            g_a = 2.0 * _gradient_wrt_a(a, r)     # shape (n,)

        else:
            # ------- Phase 2 : smooth approximation of max(c) -------------
            # linearly increase γ from a small to a larger value
            phase_elapsed = now - phase2_start
            phase_total = deadline - phase2_start
            phase_frac = phase_elapsed / max(phase_total, 1e-12)

            gamma_start = 0.5
            gamma_end = 30.0
            gamma = gamma_start + (gamma_end - gamma_start) * phase_frac

            # stable log‑sum‑exp weights
            c_shift = c - np.max(c)
            w = np.exp(gamma * c_shift)
            w /= w.sum()                           # probability weights

            # gradient of the smooth max
            g_a = _gradient_wrt_a(a, w)            # shape (n,)

        # ----------------------------------------------------------
        # chain rule to obtain gradient w.r.t. log‑weights x
        # ----------------------------------------------------------
        avg_g = np.dot(a, g_a)                    # ⟨a,∂F/∂a⟩
        g_x = a * (g_a - avg_g)                  # ∂F/∂x (softmax Jacobian)

        # ----------------------------------------------------------
        # Adam step
        # ----------------------------------------------------------
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning‑rate schedule (linear decay over the whole budget)
        lr_start = 0.5
        lr_end = 0.02
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----------------------------------------------------------
        # stagnation handling – soft restart if no improvement for 30 s
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # ----------------------------------------------------------
        # occasional global jitter to escape shallow minima
        # ----------------------------------------------------------
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # --------------------------------------------------------------
    # 8) Return the best sequence (clip to the limits used by evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
