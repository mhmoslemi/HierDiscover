"""
Propose a step‑function with a lower bound for ``evaluate_sequence``.

Algorithm
---------

The search works in two phases on a single length (taken from the previous
best construction if available, otherwise a uniform vector of length ≈ 3800).

* Phase 1 (≈ first half of the time) – a “soft‑max mirror descent”.
  The full self‑convolution is turned into a probability distribution by a
  temperature‑controlled soft‑max.  The resulting weighted gradient
  (  g = 2·conv(w, a) ) is applied with an exponentiated‑gradient update
  that stays on the simplex.

* Phase 2 (≈ second half) – a pure sub‑gradient step for the *worst* lag.
  The index `k_max` where the convolution attains its maximum is found,
  the exact gradient of that term (  g_i = 2·a_{k_max‑i} ) is used and an
  exponentiated‑gradient update is performed with a rapidly decaying step
  size.

* Exploration – with a small probability a random contiguous block is
  multiplicatively perturbed (log‑normal factor) and the vector is
  re‑normalised.

* Restarts – if no improvement is seen for a few seconds the current best
  solution is jittered (log‑normal noise) and optimisation continues.
  This prevents the optimiser from staying stuck in a local minimum.

* Book‑keeping – after every iteration the exact objective
  ``2·n·max(conv) / sum(a)^2`` is computed (via an FFT‑based convolution) and
  the best sequence seen so far is stored.

The whole routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution using a real FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _objective(a: np.ndarray, fft_len_self: int) -> float:
    """
    Exact value of ``evaluate_sequence`` (but without the safety checks).
    ``a`` is assumed to be non‑negative.
    """
    n = a.size
    c = _self_conv(a, fft_len_self)
    max_c = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (sum_a * sum_a))

        # default length – similar to the previous best construction
        n = 3800
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths needed throughout the run
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_wa = _next_pow2(3 * n)               # for w*a (w length = 2n‑1)

    # ------------------------------------------------------------------
    # 3) Book‑keeping for the current state
    # ------------------------------------------------------------------
    cur_score = _objective(a, fft_len_self)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = (now - start_time) / total   # 0 → 1

        # --------------------------------------------------------------
        # 4.1 – Choose hyper‑parameters according to the phase
        # --------------------------------------------------------------
        if frac < 0.5:
            # ------- Phase 1 : soft‑max mirror descent ----------------
            beta = 0.5 + (1500.0 - 0.5) * (frac / 0.5)            # 0.5 → 1500
            eta = 5.0 * (1.0 - (frac / 0.5)) + 0.01 * (frac / 0.5)  # 5 → 0.01

            # self‑convolution and soft‑max weights
            c = _self_conv(a, fft_len_self)                     # length 2n‑1
            max_c = c.max()
            w_raw = np.exp(beta * (c - max_c))
            w = w_raw / w_raw.sum()

            # gradient of the smoothed objective:  g = 2·(w*a)   (slice)
            conv_wa = _conv_fft(w, a, fft_len_wa)               # length 3n‑2
            grad = 2.0 * conv_wa[n - 1:n - 1 + n]               # slice of length n
        else:
            # ------- Phase 2 : pure max‑shift sub‑gradient ------------
            # self‑convolution to locate the worst lag
            c = _self_conv(a, fft_len_self)
            max_c = c.max()
            k_max = int(np.argmax(c))

            # gradient of the *single* max term (sub‑gradient)
            grad = np.zeros_like(a)
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx = np.arange(lo, hi + 1)
            grad[idx] = 2.0 * a[k_max - idx]

            # learning‑rate schedule – decays quickly in phase 2
            eta = 0.5 * (1.0 - (frac - 0.5) / 0.5) + 0.001 * ((frac - 0.5) / 0.5)

        # --------------------------------------------------------------
        # 4.2 – Exponentiated‑gradient (mirror descent) update
        # --------------------------------------------------------------
        exponent = -eta * grad
        exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
        factor = np.exp(exponent)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # --------------------------------------------------------------
        # 4.3 – Occasional stochastic jitter (exploration)
        # --------------------------------------------------------------
        if rng.random() < 0.01:    # 1 % of iterations
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 4.4 – Exact evaluation and best‑track
        # --------------------------------------------------------------
        cur_score = _objective(a, fft_len_self)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 4.5 – Soft restart after stagnation
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.10, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            # recompute a fresh convolution to reset the stale ``c`` variable
            c = _self_conv(a, fft_len_self)
            cur_score = _objective(a, fft_len_self)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped for safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
