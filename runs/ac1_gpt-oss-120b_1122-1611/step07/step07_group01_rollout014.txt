def project_simplex(v):
    # Implementation based on sorting.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative height sequence) that tries to
minimise the evaluation function

    evaluate_sequence(a) = 2·n·max(a*a) / (sum(a)²)

where n = len(a) and * denotes linear self‑convolution.
The code implements a fast projected Adam optimiser on the simplex.
Two complementary objectives are combined:

* a smooth approximation of the maximum (soft‑max with a temperature β);
* an L²‑flattening term that pushes the whole convolution toward a flat
  (≈ uniform) distribution.

Both terms are blended with a linear schedule (α) that gradually
focuses the search on the true maximum.  The vector a is kept on the
probability simplex by a fast Euclidean projection after each update.
Occasional random “jitter” moves and a restart after stagnation help to
escape local minima.  The best sequence seen during the run is returned
(and clipped to the limits required by `evaluate_sequence`).

The routine runs for the whole allotted time budget (default 1000 s)
and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT of length ``fft_len``.
    Returns the full convolution with length ``len(x)+len(y)-1``.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenient wrapper for the linear self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex.
    """
    n = a.size
    # block width at most 1 % of n, at least 1
    max_width = max(2, n // 100)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))  # moderate jitter
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
#  Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget expires.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total wall‑clock time allowance (seconds).  The routine stops a
        few seconds before the limit to avoid a forced termination.

    Returns
    -------
    List[float]
        The best sequence discovered (already clipped to the limits
        imposed by ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a length and initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            a = np.clip(arr, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
        else:
            a = None
    else:
        a = None

    if a is None:
        # start from a uniform distribution of moderate size
        n = 4000
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # If the current length is modest we can safely enlarge it – more
    # degrees of freedom usually help flatten the autocorrelation.
    target_n = max(8000, n)                # try a larger search space
    if target_n != n:
        # linear interpolation to the new resolution
        xs = np.linspace(0, n - 1, n)
        xt = np.linspace(0, n - 1, target_n)
        a = np.interp(xt, xs, a)
        a = _project_simplex(a)
        n = target_n

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (powers of two for speed)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_wa = _next_pow2(3 * n)                # for w*a and diff*a

    # ------------------------------------------------------------------
    # 3) Targets for the flatten‑the‑convolution phase
    # ------------------------------------------------------------------
    target_vec = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # ------------------------------------------------------------------
    # 4) Adam state and bookkeeping
    # ------------------------------------------------------------------
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best = a.copy()
    best_score = float("inf")
    last_improve = start_time
    iter_counter = 0

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iter_counter += 1

        # ----- compute self‑convolution and the current objective -----
        c = _self_conv(a, fft_len_self)       # length 2n‑1
        max_c = c.max()
        cur_score = 2.0 * n * max_c

        # ----- keep the best solution seen so far -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- progress fractions for scheduling -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total          # 0 → 1

        # ----- schedule for learning rate, soft‑max temperature and blending -----
        # learning rate decays from 0.1 to 0.001
        lr = 0.1 * (1.0 - frac) + 0.001 * frac
        # soft‑max concentration β goes from 0.5 to 2500
        beta = 0.5 * (1.0 - frac) + 2500.0 * frac
        # blending weight α starts at 1 (full flattening) and linearly drops to 0
        alpha = max(0.0, 1.0 - 2.0 * frac)   # flattening dominates early phases

        # ----- soft‑max weights approximating the max -----
        w_raw = np.exp(beta * (c - max_c))   # shifted for numerical stability
        w = w_raw / w_raw.sum()

        # ----- gradients -------------------------------------------------
        # Gradient of the soft‑max surrogate of the max
        grad_soft = 2.0 * _conv_fft(w, a, fft_len_wa)[n - 1:n - 1 + n]

        # Gradient trying to flatten the whole convolution toward a constant
        diff = c - target_vec
        grad_flat = 2.0 * _conv_fft(diff, a, fft_len_wa)[n - 1:n - 1 + n]

        # Blended gradient
        grad = (1.0 - alpha) * grad_soft + alpha * grad_flat

        # ----- Adam update on the simplex (additive) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** iter_counter)
        v_hat = v / (1.0 - beta2 ** iter_counter)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.clip(a, 0.0, None)          # keep non‑negative
        a = _project_simplex(a)            # enforce sum = 1

        # ----- occasional random jitter to explore -----
        if rng.random() < 0.004:           # ≈ 0.4 % of iterations
            a = _random_block_perturb(a, rng)

        # ----- restart after long stagnation -----
        if now - last_improve > 30.0:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * jitter
            a = _project_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final local random search (uses remaining budget if any)
    # ------------------------------------------------------------------
    # The loop above already used almost the whole budget, but we keep a
    # tiny final phase that tries cheap random block moves.
    while time.monotonic() < deadline + 2.0:   # a few extra seconds are harmless
        cand = _random_block_perturb(best, rng)
        c = _self_conv(cand, fft_len_self)
        cand_score = 2.0 * n * c.max()
        if cand_score < best_score:
            best_score = cand_score
            best = cand
            now = time.monotonic()
            print(f"[{now - start_time:6.2f}s] random‑jitter improvement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clip to limits required by eval)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
