def compute_score_and_grad(a, p, fft_len_self, fft_len_grad):
    # Self conv
    A = np.fft.rfft(a, n=fft_len_self)
    b = np.fft.irfft(A * A, n=fft_len_self)[:2*n-1]
    # Compute weights w for p-norm surrogate (softmax variant)
    b_pow_p = b**p
    sum_b_pow_p = b_pow_p.sum()
    if sum_b_pow_p == 0:
        # handle degeneracy
        w = np.full_like(b, 1.0/(b.size))
    else:
        denom = sum_b_pow_p**((p-1)/p)
        w = b**(p-1) / denom
    # Compute gradient: conv of w and a_rev
    W = np.fft.rfft(w, n=fft_len_grad)
    a_rev = a[::-1]
    Fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv_full = np.fft.irfft(W * Fa_rev, n=fft_len_grad)[:3*n-1]
    grad = 2.0 * conv_full[n-1:n-1+n]
    # Score = 2 * n * b.max()
    score = 2.0 * a.size * b.max()
    return score, grad

"""
Propose a non‑negative step‑function that (empirically) beats the previous
bound.  The optimizer consists of two stages:

1.  **Smooth‑Lp descent** (≈70 % of the time).  We minimise the
    Lp‑norm of the self‑convolution  b = a*a  with  p slowly increasing
    from 2 to ~80.  The Lp‑norm is a smooth surrogate for the max‑norm,
    and its gradient can be written as a convolution, which we evaluate
    with FFTs.  The update is performed in log‑space with an Adam‑style
    adaptive step‑size, guaranteeing positivity of the coefficients.
    After each step the vector is projected back onto the simplex.

2.  **Stochastic fine‑tuning** (≈30 % of the time).  After the surrogate
    becomes very sharp we switch to a Metropolis‑style mass‑transfer
    move that directly reduces the peak of the convolution.  Periodic
    block‑perturbations keep the search from stagnating.

Restarts are triggered after 20 s of no improvement: the current best
solution is perturbed with log‑normal noise and the optimiser continues.
The routine respects the 1000 s budget (leaving a 5 s safety margin) and
returns the best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _grad_lp_norm(a: np.ndarray, p: float,
                  fft_len_self: int, fft_len_grad: int) -> tuple[np.ndarray, np.ndarray]:
    """
    Compute the Lp‑norm of the self‑convolution and its gradient w.r.t. ``a``.
    Returns (score, grad) where ``score`` is 2·n·max(b) (the true objective)
    and ``grad`` is the gradient of the smooth proxy
        (∑ b_k**p)^{1/p}
    with respect to ``a``.
    """
    n = a.size
    # self‑convolution b = a*a
    A = np.fft.rfft(a, n=fft_len_self)
    b = np.fft.irfft(A * A, n=fft_len_self)[: 2 * n - 1]

    # ----- smooth surrogate weights (gradient of Lp‑norm) -----
    if p == 2.0:
        # for p = 2 the weights are simply proportional to b
        w = b.copy()
    else:
        b_pow = b ** p
        sum_pow = b_pow.sum()
        if sum_pow == 0.0:
            w = np.full_like(b, 1.0 / b.size)
        else:
            denom = sum_pow ** ((p - 1.0) / p)
            w = b ** (p - 1.0) / denom

    # ----- gradient of the surrogate -----
    # grad = 2 * (w * a_rev)  (central n entries of convolution)
    W = np.fft.rfft(w, n=fft_len_grad)
    a_rev = a[::-1]
    A_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    conv_full = np.fft.irfft(W * A_rev, n=fft_len_grad)[: 3 * n - 1]
    grad = 2.0 * conv_full[n - 1:n - 1 + n]

    # true objective (used for bookkeeping)
    score = 2.0 * n * b.max()
    return score, grad

# ----------------------------------------------------------------------

            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Switching to a stochastic fine‑tuning phase near the end
        # --------------------------------------------------------------
        if frac > 0.70:
            # simple Metropolis move on the simplex
            # locate the current peak of the convolution
            k_max = int(b_new.argmax())
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()
            if total_contrib > 0.0:
                donor = rng.choice(np.arange(n), p=contrib / total_contrib)
            else:
                donor = rng.integers(0, n)
            if a[donor] > 1e-14:
                # recipient prefers small entries
                recv_weights = 1.0 - a
                recv_weights[donor] = 0.0
                recv_sum = recv_weights.sum()
                if recv_sum > 0.0:
                    recipient = rng.choice(np.arange(n), p=recv_weights / recv_sum)
                else:
                    recipient = (donor + rng.integers(1, n)) % n
                max_delta = 0.5 * a[donor]
                delta = rng.random() * max_delta

                # fast incremental update of convolution (same code as the
                # reference implementation)
                a_candidate = a.copy()
                a_candidate[donor] -= delta
                a_candidate[recipient] += delta

                # recompute convolution for the candidate (cheap via FFT)
                A_cand = np.fft.rfft(a_candidate, n=fft_len_self)
                b_cand = np.fft.irfft(A_cand * A_cand, n=fft_len_self)[: 2 * n - 1]
                cand_score = 2.0 * n * b_cand.max()

                # temperature schedule for Metropolis acceptance
                temp_start, temp_end = 0.07, 1e-6
                temperature = temp_start * (1.0 - frac) + temp_end * frac
                temperature = max(temperature, 1e-12)

                if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / temperature):
                    a = a_candidate
                    cur_score = cand_score

        # --------------------------------------------------------------
        # Occasional global jitter to escape flat regions
        # --------------------------------------------------------------
        if rng.random() < 0.015:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # Restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.2, size=n))
            a *= noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            # reset Adam moments so the new region can be explored afresh
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
