            last_improve = time.monotonic()

        remaining_time = deadline - time.monotonic()

            k_max = int(np.argmax(conv))

                best_score = cur_score
                best_vec = a.copy()
                last_improve = time.monotonic()
                print(f"[{time.monotonic()-start_time:6.2f}s] cand {cand_idx} phase2 new best {best_score:.6f}")

            shift = k_max - (n - 1)
            if shift >= 0:
                grad = np.concatenate([np.zeros(shift), 2.0 * a[:n - shift]])
            else:
                grad = np.concatenate([2.0 * a[-shift:], np.zeros(-shift)])

            # Learning rate schedule (smaller)
            eta = 0.1 * (0.5 ** (iter_phase2 / 500.0))  # decaying

            # occasional jitter

            iter_phase2 += 1

        cand_idx += 1

    if best_vec is None:

    # Clip for safety
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]

"""
Propose a candidate sequence that minimises ``evaluate_sequence``.

The algorithm works in two stages.
*Stage 1* – a smoothed‐gradient descent.  The convolution
  ``c = a*a`` is turned into a soft‑max weight vector ``w``.
  The gradient of the smooth surrogate ``φ(a)=w·c`` is
  ``∇φ = 2·(a * w_rev)`` where ``w_rev`` is the reversed weight
  vector.  An exponentiated‑gradient step followed by a projection onto
  the probability simplex keeps the parameters non‑negative and with
  unit L¹‑norm (the objective is scale‑invariant).

*Stage 2* – a hard‑max sub‑gradient refinement.  Once the soft‑max has
  flattened the convolution we switch to the exact max index
  ``k* = argmax(a*a)``.  The sub‑gradient of this max is simply the
  reversed slice of ``a``:
  ``grad_j = 2·a_{k*−j}`` (zero outside the support).  A few hundred
  exponentiated‑gradient updates with a decreasing step‑size finish the
  optimisation.

Both stages are run on a portfolio of start vectors
(uniform, exponential decays, Dirichlet draws, etc.) and the best
solution seen before the time‑budget expires is returned.
Random block perturbations and short restarts are used to avoid
stagnation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (>=0, sum=1)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A small, diverse portfolio of feasible start vectors (sum=1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a handful of Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


def _hard_max_grad(a: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient of max(a*a) at the current peak index."""
    n = a.shape[0]
    idx = k_max - np.arange(n)               # needed source indices
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    # 2 * a[idx] where idx is valid
    grad[mask] = 2.0 * a[idx[mask]]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine performs a soft‑max gradient descent
    (stage 1) followed by a hard‑max sub‑gradient polish (stage 2) on a set
    of diversified initial vectors.  Random block jitter and short restarts
    help escape plateaus.  The best sequence found before the time budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) time‑budget handling (keep a few seconds for the harness)
    # --------------------------------------------------------------
    safety_margin = 5.0
    deadline = time.monotonic() + max(0.0, float(budget_s) - safety_margin)

    # --------------------------------------------------------------
    # 1) Determine length and initialise candidates
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
            n = prev.size
            init_vecs = [np.clip(prev, 0.0, None)]
        else:
            n = 4096
            init_vecs = []
    else:
        n = 4096
        init_vecs = []

    # add the diversified portfolio
    init_vecs.extend(_initial_candidates(n, rng))

    # Pre‑compute FFT size for the self‑convolution (used many times)
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 2) Book‑keeping of the global best solution
    # --------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 3) Optimise each start vector while time permits
    # --------------------------------------------------------------
    for cand_id, a in enumerate(init_vecs):
        if time.monotonic() >= deadline:
            break

        # ensure feasibility: non‑negative and on the simplex
        a = np.clip(a, 0.0, None)
        a = proj_simplex(a)

        # quick evaluation of the start point
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = time.monotonic()
            print(f"[init {cand_id}] score {best_score:.6f}")

        # ----------------------------------------------------------
        # Stage 1 – soft‑max gradient descent (up to ~4000 iters)
        # ----------------------------------------------------------
        max_iters_stage1 = 4000
        for it in range(max_iters_stage1):
            if time.monotonic() >= deadline:
                break

            # linear schedules for learning rate and temperature
            frac = (time.monotonic() - start_time) / total_time
            eta = 0.8 * (1.0 - frac) + 0.001 * frac          # learning rate
            alpha = 5.0 * (1.0 - frac) + 1200.0 * frac       # soft‑max temperature

            # self‑convolution
            fa = np.fft.rfft(a, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]

            # soft‑max weights (shifted for numerical stability)
            cmax = conv.max()
            shifted = conv - cmax
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient = 2 * (a * w_rev)   (cross‑correlation)
            w_rev = w[::-1]
            grad_len = _next_pow2(n + w_rev.size - 1)   # = next_pow2(3n‑2)
            fa = np.fft.rfft(a, n=grad_len)
            fw = np.fft.rfft(w_rev, n=grad_len)
            grad_full = np.fft.irfft(fa * fw, n=grad_len)
            offset = w_rev.size - 1
            grad = 2.0 * grad_full[offset: offset + n]

            # exponentiated‑gradient update and projection
            a = a * np.exp(-eta * grad)
            a = proj_simplex(a)

            # occasional evaluation (every 10 iters)
            if it % 10 == 0:
                conv = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * conv.max()
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_vec = a.copy()
                    last_improve = time.monotonic()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] cand {cand_id} stage1 new best {best_score:.6f}")

            # occasional random block jitter (≈1 % of iterations)
            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)

            # short restart if nothing improved for ~30 s
            if time.monotonic() - last_improve > 30.0:
                a = best_vec.copy()
                noise = np.exp(rng.normal(0.0, 0.08, size=n))
                a = a * noise
                a = proj_simplex(a)
                last_improve = time.monotonic()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] cand {cand_id} stage1 restart after stagnation")

        # ----------------------------------------------------------
        # Stage 2 – hard‑max sub‑gradient polish (up to ~2000 iters)
        # ----------------------------------------------------------
        max_iters_stage2 = 2000
        for it in range(max_iters_stage2):
            if time.monotonic() >= deadline:
                break

            conv = _self_conv(a, fft_len_self)
            k_max = int(np.argmax(conv))
            cur_score = 2.0 * n * conv.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_vec = a.copy()
                last_improve = time.monotonic()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] cand {cand_id} stage2 new best {best_score:.6f}")

            # exact sub‑gradient w.r.t. the current peak
            grad = _hard_max_grad(a, k_max)

            # decaying step‑size (simple exponential decay)
            eta = 0.2 * (0.7 ** (it / 400.0))

            a = a * np.exp(-eta * grad)
            a = proj_simplex(a)

            # tiny jitter occasionally
            if rng.random() < 0.005:
                a = _random_block_perturb(a, rng)

            # restart on prolonged stagnation
            if time.monotonic() - last_improve > 20.0:
                a = best_vec.copy()
                noise = np.exp(rng.normal(0.0, 0.05, size=n))
                a = a * noise
                a = proj_simplex(a)
                last_improve = time.monotonic()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] cand {cand_id} stage2 short restart")

    # --------------------------------------------------------------
    # 4) Fallback (should never happen) and final sanitisation
    # --------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
