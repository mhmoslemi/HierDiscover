def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    # Determine n:
    if 'height_sequence_1' in globals():
        a0 = np.array(globals()['height_sequence_1'], dtype=np.float64)
        # clip and renormalize
        a0 = np.clip(a0, 0, None)
        if a0.sum() > 0:
            a0 = a0 / a0.sum()
        else:
            n = 4096
            a0 = np.full(n, 1.0/n)
    else:
        n = 4096
        a0 = np.full(n, 1.0/n)
    n = a0.shape[0]

    # If desired, could also try different n (maybe round up to power of two)
    # We'll keep current length.

    cur = a0.copy()
    cur_score = compute_score(cur)
    best = cur.copy()
    best_score = cur_score

    # set time
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    start = time.monotonic()
    iter_cnt = 0
    # parameters
    T_start = 0.05  # temperature initial
    T_end = 1e-4
    # schedule: exponential
    while time.monotonic() < deadline:
        # compute fraction of time elapsed
        elapsed = time.monotonic() - start
        frac = elapsed / (deadline - start)
        # temperature
        T = T_start * (T_end/T_start)**frac

        # decide move type
        if rng.random() < 0.7:
            # transfer move
            # select source i weighted by a (so bigger mass)
            i = rng.choice(n, p=cur)  # weighted selection
            # ensure i not all zero (if any zero)
            # Choose destination j uniformly
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            # amount to transfer
            max_transfer = cur[i]
            if max_transfer <= 0:
                # choose another iteration
                continue
            # amount proportional to T: early bigger moves
            frac_transfer = rng.random() * min(0.5, T*10)  # scaling factor
            amount = max_transfer * frac_transfer
            if amount <= 0:
                continue
            new = cur.copy()
            new[i] -= amount
            new[j] += amount
            # no negative because amount <= cur[i]
        else:
            # block jitter move
            block_len = max(2, int(n * 0.02 * (T/T_start + 0.1)))  # bigger early
            start_idx = rng.integers(0, n - block_len + 1)
            factor = np.exp(rng.normal(0.0, sigma=0.2 * (T/T_start + 0.01)))  # lognormal variation
            new = cur.copy()
            new[start_idx:start_idx+block_len] *= factor
            # ensure positivity
            new = np.maximum(new, 0.0)
            # renormalize to sum=1
            new_sum = new.sum()
            if new_sum == 0:
                continue
            new = new / new_sum

        # evaluate
        new_score = compute_score(new)
        # acceptance
        delta = new_score - cur_score
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = new
            cur_score = new_score

            # update best
            if new_score < best_score:
                best_score = new_score
                best = new.copy()
                print(f"[{elapsed:6.2f}s] new best {best_score:.6f} iter={iter_cnt}")
        iter_cnt += 1

        # occasional restart if stagnation
        if iter_cnt % 20000 == 0:
            # reinitialize near best with noise
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            cur = best * noise
            cur = cur / cur.sum()
            cur_score = compute_score(cur)
            print(f"[{elapsed:6.2f}s] restart after {iter_cnt} it")
    # After loop
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

    # ------------------------------------------------------------------

    return [float(v) for v in best]

"""
Simulated‑annealing + sub‑gradient mirror descent for the
self‑convolution bound.

The routine works in two stages:

* Phase 1 (≈15 % of the time budget) – random Dirichlet draws
  (concentration < 1) are sampled and the best among them is kept.
  This quickly breaks the symmetry of a uniform start.

* Phase 2 – starting from the best candidate a mirror‑descent step
  is performed on the simplex.  At each iteration the convolution
  `c = a*a` is computed (FFT), the index `k` attaining `max(c)` is
  located and the sub‑gradient of that entry w.r.t. `a` is
  evaluated analytically.  A multiplicative update
  `a ← a·exp(‑η·g)` (η is a linearly annealed step‑size) is applied
  and the vector is re‑normalised to sum = 1.  Every few hundred
  iterations a small block is multiplied by a log‑normal factor
  (jitter) and the whole vector is re‑normalised, while a restart
  with modest Gaussian noise is triggered if no improvement has been
  seen for a while.

All operations respect the non‑negativity and simplex constraints.
The best exact score (computed with the supplied ``evaluate_sequence``
function) is stored and finally returned as a plain list of floats.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    The algorithm combines a short random‑sampling phase with a
    sub‑gradient mirror‑descent phase, enriched by occasional jitter
    and restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Determine length and obtain an initial candidate
    # ------------------------------------------------------------------
    # Use the previously discovered construction if available.
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() > 0.0:
            a0 = a0 / a0.sum()
        else:
            a0 = None
    else:
        a0 = None

    # Choose a length – if we have a previous construction keep it,
    # otherwise work with a power‑of‑two size for cheap FFTs.
    if a0 is not None:
        a = a0.copy()
    else:
        n = 4096
        idx = np.arange(n)
        centre = (n - 1) / 2.0
        x = (idx - centre) / centre               # in [‑1, 1]
        a = 0.5 * (1.0 + np.cos(np.pi * x))       # Hann window, non‑negative
        a = np.maximum(a, 0.0)
        a = a / a.sum()

    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)                # length for FFT‑based convolution

    # ------------------------------------------------------------------
    # 2) Initial exact evaluation (using the public evaluator)
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – random Dirichlet sampling (≈15 % of total budget)
    # ------------------------------------------------------------------
    phase1_deadline = start + (deadline - start) * 0.15
    while time.monotonic() < phase1_deadline:
        # Dirichlet with concentration < 1 → sparse candidates
        alpha = 0.4
        cand = rng.gamma(alpha, 1.0, size=n)
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best = cand.copy()
            print(f"[{time.monotonic() - start:6.2f}s] phase‑1 improvement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – mirror‑descent on the maximal convolution entry
    # ------------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score
    iteration = 0
    idx = np.arange(n)                     # reusable index array

    # Schedules
    eta_start = 0.80                       # initial multiplicative step
    eta_end = 0.001                        # final step
    jitter_every = 300                     # iterations between jitter moves
    jitter_sigma_start = 0.45              # log‑normal jitter std at start
    jitter_sigma_end = 0.03                # at the end
    restart_after = 45.0                   # seconds without improvement

    last_improve_time = time.monotonic()

    while time.monotonic() < deadline:
        # ---- self‑convolution via FFT ---------------------------------
        A = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
        k = int(conv.argmax())                         # index attaining the max

        # ---- sub‑gradient of the active convolution entry ------------
        t = k - idx
        mask = (t >= 0) & (t < n)
        g = np.zeros_like(cur)
        g[mask] = 2.0 * cur[t[mask]]
        if (k % 2) == 0:                               # correct the double‑count on the diagonal
            j_center = k // 2
            if 0 <= j_center < n:
                g[j_center] = cur[j_center]

        # ---- step‑size (linear annealing) ----------------------------
        elapsed = time.monotonic() - start
        frac = elapsed / (deadline - start)
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # ---- mirror‑descent update (multiplicative) -----------------
        cur = cur * np.exp(-eta * g)
        cur_sum = cur.sum()
        if cur_sum == 0.0:
            cur = np.full_like(cur, 1.0 / n)
        else:
            cur = cur / cur_sum

        iteration += 1

        # ---- periodic exact evaluation -------------------------------
        if iteration % 250 == 0:
            score = evaluate_sequence(cur.tolist())
            if score < best_score - 1e-12:
                best_score = score
                best = cur.copy()
                last_improve_time = time.monotonic()
                print(f"[{time.monotonic() - start:6.2f}s] new best = {best_score:.6f}  iter={iteration}")

        # ---- jitter – occasional random block perturbation -----------
        if iteration % jitter_every == 0:
            sigma = jitter_sigma_start * (1.0 - frac) + jitter_sigma_end * frac
            # block size shrinks from about 3 % → 0.5 % of the vector
            block_len = max(2, int(n * (0.03 * (1.0 - frac) + 0.005 * frac)))
            start_idx = rng.integers(0, n - block_len + 1)
            factor = np.exp(rng.normal(0.0, sigma))
            cur[start_idx:start_idx + block_len] *= factor
            cur = cur / cur.sum()

        # ---- restart if stuck for a while ----------------------------
        if time.monotonic() - last_improve_time > restart_after:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            cur = best * noise
            cur = cur / cur.sum()
            last_improve_time = time.monotonic()
            iteration = 0
            print(f"[{time.monotonic() - start:6.2f}s] restart – injected noise")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
