def subgradient_step(a, lr):

    conv = # using FFT
    max_b = conv.max()

import scipy.sparse as sp

def convolution_constraint_matrix(a):
    n = a.shape[0]
    diags = []
    offsets = []
    for d in range(n):
        # diagonal offset = -d
        offsets.append(-d)
        length = n - d
        diag_vals = np.full(length, a[d])
        diags.append(diag_vals)
    A = sp.diags(diags, offsets, shape=(2*n-1, n), format='csr')
    return A
```

"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The algorithm is a hybrid of three ideas:

1. **Smooth‑max Adam optimisation** – the true max in the evaluation
   is approximated by a soft‑max (log‑sum‑exp).  The gradient of this
   surrogate is computed analytically with two FFTs, and an Adam
   optimiser updates the vector while staying on the simplex
   (sum = √(2 n)).

2. **Periodic LP refinement** – every few minutes we solve a linear
   program
       maximise   Σ b_j
       s.t.       b_j ≥ 0
                 (a * b)_k ≤ M   for all k,
   where ``M`` is the current maximum self‑convolution of ``a``.
   The solution ``b`` is scaled back to the target sum and mixed with
   ``a``.  The mix is accepted only if it improves the exact score.

3. **Stagnation handling** – if no improvement is seen for a while we
   re‑inject a small multiplicative log‑normal noise around the best
   vector and reset the Adam moments.

All heavy lifting (FFT, LP matrix) uses NumPy / SciPy and runs well
within the 1000 s budget.  The routine returns the best sequence
encountered.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Returns a new array with sum exactly 1.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be projected to zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)

    # Renormalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _project_to_target_sum(v: np.ndarray, target: float) -> np.ndarray:
    """
    Project ``v`` onto the simplex and then scale it so that its sum equals
    ``target``.  The result stays non‑negative.
    """
    w = _project_onto_simplex(v)
    return w * target


def _build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2 n‑1) × n sparse matrix ``A`` such that
        (A @ b) = a * b   (full linear convolution)
    for non‑negative vectors ``a``, ``b`` of length n.
    Each row k of ``A`` contains the coefficients a_{k‑j}
    for the appropriate indices j.
    """
    n = a.shape[0]
    # Offsets are negative because row = col + d (d >= 0)
    offsets = [-d for d in range(n)]
    data = [np.full(n - d, a[d], dtype=np.float64) for d in range(n)]
    A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csr")
    return A


def _softmax_weights(c: np.ndarray, tau: float) -> np.ndarray:
    """
    Numerically stable soft‑max of ``c`` with temperature ``tau``.
    The result sums to 1.
    """
    shift = c.max()
    exp_shifted = np.exp((c - shift) / tau)
    total = exp_shifted.sum()
    if total == 0.0:
        return np.full_like(c, 1.0 / len(c))
    return exp_shifted / total


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` within the supplied time budget.
    The returned list respects the evaluator's safety caps.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2) Determine problem size and initialise ``a``
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0 and np.isfinite(seq0).all():
            n = seq0.size
            a = np.clip(seq0, 0.0, None)
        else:
            seq0 = None
    else:
        seq0 = None

    if seq0 is None:
        # default length – a power of two makes FFT cheap
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    # Normalise to the standard sum √(2 n) (the evaluator is scale‑invariant)
    target_sum = np.sqrt(2.0 * n)
    a_sum = a.sum()
    if a_sum > 0.0:
        a = a * (target_sum / a_sum)
    else:
        a = np.full(n, target_sum / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT length (next power of two ≥ 2 n‑1)
    # ------------------------------------------------------------------
    fft_len = 1 << ((2 * n - 2).bit_length())

    # ------------------------------------------------------------------
    # 4) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    best_time = time.monotonic()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Adam hyper‑parameters
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = beta1, beta2

    # ------------------------------------------------------------------
    # 6) Schedules and auxiliary parameters
    # ------------------------------------------------------------------
    JITTER_EVERY = 400          # iterations
    JITTER_SIGMA = 0.10         # log‑normal std‑dev for jitter
    ITER_EVAL_EVERY = 250       # exact evaluation interval
    LP_INTERVAL = 180.0         # seconds between LP refinements
    STAGNATION_TIME = 60.0      # seconds without improvement → restart
    last_lp = time.monotonic()
    start_time = time.monotonic()
    iteration = 0

    # ------------------------------------------------------------------
    # 7) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)

        # ----- annealing of temperature and learning rate ---------------
        tau = 0.20 * (1.0 - frac) + 0.0005 * frac   # ends very close to 0
        lr = 0.30 * (1.0 - frac) + 0.002 * frac    # small final step‑size

        # ----- convolution and soft‑max weights -------------------------
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        w = _softmax_weights(conv, tau)            # length 2n‑1, sums to 1

        # ----- correlation (gradient of the soft‑max) -------------------
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)

        corr_start = n - 1
        grad = (2.0 / tau) * corr_full[corr_start:corr_start + n]

        # ----- Adam update ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.clip(a, 0.0, None)
        a = _project_to_target_sum(a, target_sum)

        iteration += 1

        # ----- occasional jitter ---------------------------------------
        if iteration % JITTER_EVERY == 0:
            width = rng.integers(1, max(2, n // 15))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, JITTER_SIGMA))
            a_jit = a.copy()
            a_jit[start:start + width] *= factor
            a_jit = np.clip(a_jit, 0.0, None)
            a = _project_to_target_sum(a_jit, target_sum)

        # ----- exact evaluation from time to time ----------------------
        if iteration % ITER_EVAL_EVERY == 0:
            exact_score = evaluate_sequence(a.tolist())
            if exact_score < best_score - 1e-12:
                best_score = exact_score
                best = a.copy()
                best_time = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f} "
                      f"(iter={iteration}, τ={tau:.5f})")

        # ----- LP refinement (if enough time has passed) -------------
        if (now - last_lp) >= LP_INTERVAL and n <= 4000:
            # Build constraint matrix only once per LP call
            M = conv.max()
            try:
                A_ub = _build_convolution_matrix(a)
                b_ub = np.full(2 * n - 1, M, dtype=np.float64)

                # maximise Σ b  →  minimise -Σ b
                c_obj = -np.ones(n, dtype=np.float64)
                bounds = [(0.0, None) for _ in range(n)]

                lp_res = opt.linprog(
                    c_obj,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True, "time_limit": 30},
                )
                if lp_res.success:
                    b0 = lp_res.x
                    if b0.sum() > 0.0:
                        b = _project_to_target_sum(b0, target_sum)

                        # try a few mixing fractions and keep the best one
                        best_mix = a
                        best_mix_score = best_score
                        for t in (0.2, 0.4, 0.6, 0.8):
                            cand = (1.0 - t) * a + t * b
                            cand = _project_to_target_sum(cand, target_sum)
                            sc = evaluate_sequence(cand.tolist())
                            if sc < best_mix_score:
                                best_mix_score = sc
                                best_mix = cand.copy()
                        if best_mix_score < best_score - 1e-12:
                            best_score = best_mix_score
                            best = best_mix.copy()
                            print(f"[{elapsed:6.2f}s] LP‑mix improved score = {best_score:.6f}")
                        a = best_mix.copy()
                else:
                    # LP failed – not fatal, just continue
                    pass
            except Exception as exc:
                # Defensive programming: LP may fail for numerical reasons
                print(f"[{elapsed:6.2f}s] LP exception: {exc}")

            last_lp = now

        # ----- stagnation handling --------------------------------------
        if now - best_time > STAGNATION_TIME:
            # Re‑inject gentle noise around the current best and reset Adam
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            a = a * noise
            a = _project_to_target_sum(a, target_sum)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = beta1, beta2
            best_time = now
            print(f"[{elapsed:6.2f}s] restart after stagnation")

    # ------------------------------------------------------------------
    # 8) Return the best sequence found (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
