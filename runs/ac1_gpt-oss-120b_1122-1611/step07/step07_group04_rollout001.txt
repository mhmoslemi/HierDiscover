"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------

* Work on the simplex (L¹‑norm = 1); the objective is scale‑invariant.
* For a given length `n` we minimise the peak of the self‑convolution
  `c = a * a`.  The true objective is `2 * n * max(c)`.
* Instead of the smooth soft‑max used previously we use a *sub‑gradient*
  of the max: all indices where `c` is within a tiny tolerance of the
  current maximum receive equal weight.  The gradient of the weighted
  sum is `2 * (a * w_rev)`, which we obtain by a single FFT‑based
  convolution.
* The update is performed with an Adam‑style adaptive step on the
  *additive* domain, followed by a projection onto the probability simplex
  (non‑negative, sum = 1).  Adam gives momentum and per‑coordinate
  scaling, helping to escape shallow plateaus.
* Several lengths (`n = 2000, 4000, 6000`) are tried sequentially.  For each
  length a small portfolio of diverse initial vectors is generated
  (uniform, symmetric exponential, Dirichlet draws).  The best vector
  across all runs is kept.
* A tiny amount of random block perturbation is injected every few
  iterations to explore new directions; if no improvement is seen for
  30 s the optimiser is re‑initialised from the current global best.
* After the main loop a short “fine‑tuning” phase runs with a very small
  learning rate to polish the best solution.

The whole routine respects the given time budget (default 1000 s) and
returns a plain ``list[float]`` that is already clipped to the bounds
enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all defined at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    # block width: between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Uses an Adam‑style sub‑gradient method on the
    max of the self‑convolution, plus occasional stochastic perturbations.
    """
    rng = np.random.default_rng(seed)

    # 0) time budget – keep a 5 s safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # 1) bookkeeping of the best solution found so far
    best_seq = None
    best_score = np.inf
    best_time = None

    # 2) lengths to try – a few values give a good trade‑off between
    #    expressive power and computational cost
    lengths = [2000, 4000, 6000]

    # 3) main outer loop over lengths
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # Pre‑compute FFT size for self‑convolution (used many times)
        fft_self = _next_pow2(2 * n - 1)

        # ------------------------------------------------------------------
        # 3a) generate a diversified portfolio of start vectors (all sum=1)
        # ------------------------------------------------------------------
        starters = []

        # uniform
        starters.append(np.full(n, 1.0 / n, dtype=np.float64))

        # symmetric exponential (high at both ends, low in the middle)
        r = 0.97
        half = np.arange((n + 1) // 2, dtype=np.float64)
        exp_half = r ** half
        if n % 2 == 0:
            sym = np.concatenate([exp_half, exp_half[::-1]])
        else:
            sym = np.concatenate([exp_half, exp_half[-2::-1]])
        sym /= sym.sum()
        starters.append(sym)

        # linear decay from the left side (mirrored to keep symmetry)
        lin = np.arange(n, 0, -1, dtype=np.float64)
        lin /= lin.sum()
        starters.append(lin)

        # a couple of Dirichlet draws (moderately concentrated)
        for _ in range(4):
            vec = rng.dirichlet(np.ones(n) * 0.3).astype(np.float64)
            starters.append(vec)

        # ------------------------------------------------------------------
        # 3b) run a short optimisation from each starter
        # ------------------------------------------------------------------
        for a0 in starters:
            if time.monotonic() >= deadline:
                break

            a = proj_simplex(a0)  # enforce simplex
            # Adam moments
            m = np.zeros_like(a)
            v_vec = np.zeros_like(a)
            beta1, beta2 = 0.9, 0.999
            eps = 1e-8

            # schedule for learning rate (linear decay over this run)
            # we compute it on‑the‑fly based on overall progress
            start_run = time.monotonic()
            run_deadline = deadline  # we never exceed global deadline

            # last improvement timestamp (for possible restart)
            last_improve = time.monotonic()

            # main inner loop
            while time.monotonic() < run_deadline:
                # ----- self‑convolution (FFT) ---------------------------------
                fa = np.fft.rfft(a, n=fft_self)
                conv = np.fft.irfft(fa * fa, n=fft_self)[: 2 * n - 1]

                max_c = conv.max()

                # ----- sub‑gradient of the max (active set) --------------------
                # Use a relative tolerance of 1e‑12 (or absolute 1e‑12)
                tol = max(1e-12, max_c * 1e-12)
                active = conv >= max_c - tol
                w = active.astype(np.float64)
                weight_sum = w.sum()
                if weight_sum == 0:
                    # fallback – uniform weighting
                    w[:] = 1.0 / w.size
                else:
                    w /= weight_sum

                # ----- gradient = 2 * (a * w_rev) -----------------------------
                w_rev = w[::-1]                     # reverse for correlation
                grad_len = _next_pow2(n + w_rev.size - 1)  # ≈ next_pow2(3n‑2)
                fa = np.fft.rfft(a, n=grad_len)
                fw = np.fft.rfft(w_rev, n=grad_len)
                grad_full = np.fft.irfft(fa * fw, n=grad_len)
                offset = w_rev.size - 1
                grad = 2.0 * grad_full[offset:offset + n]

                # ----- Adam adaptive step ------------------------------------
                # progress fraction of the whole budget
                now = time.monotonic()
                overall_frac = (now - start_time) / (deadline - start_time)
                # learning rate from 0.15 → 0.001 linearly
                lr_start, lr_end = 0.15, 0.001
                lr = lr_start * (1.0 - overall_frac) + lr_end * overall_frac

                v_hat = v_vec / (1.0 - beta2 ** t)
                a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

                a = np.clip(a, 0.0, None)

                cur_score = 2.0 * n * max_c

                if cur_score < best_score - 1e-12:

                    best_seq = a.copy()
                    last_improve = now
                    best_time = now - start_time
                    print(f"[{best_time:6.2f}s] new best score = {best_score:.6f} (n={n})")

                    a = proj_simplex(a)
                    last_improve = now
                    print(f"[{now - start_time:6.2f}s] restart – best still {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4) final polishing on the best sequence (if any)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback: uniform of modest size
        n = 2000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * (1.0 / n)   # = 2

    # Fine‑tune with a tiny constant learning rate for a few hundred steps
    a = best_seq.copy()
    n = a.size

    m = np.zeros_like(a)
    v_vec = np.zeros_like(a)
