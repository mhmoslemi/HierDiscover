def proj_simplex(v, s):
    """
    Projects vector v onto the simplex { x >= 0, sum(x) = s }.
    Based on algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - s
    # Find rho: max i such that u[i] > cssv[i] / (i+1)
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all zeros case
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative list of heights) that minimizes
evaluate_sequence.  The method combines:

* a small set of deterministic shapes (uniform, 1/√i, etc.);
* many random Dirichlet initialisations;
* a projected sub‑gradient descent on the exact max‑convolution
  (no soft‑max surrogate).  The sum of the coefficients is kept fixed
  to √(2 n) by projecting onto the simplex after each update.
* a simple schedule for the learning‑rate together with occasional
  Gaussian jitter to escape shallow plateaus.

The routine runs until the supplied time‑budget (default 1000 s) is
exhausted and returns the best sequence found.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
    { x >= 0 , sum(x) = s }.
    Implementation follows Duchi et al. (2008).
    """
    if s <= 0:
        return np.zeros_like(v)

    n = v.shape[0]
    u = np.sort(v)[::-1]                     # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all components will be zero
        theta = cssv[-1] / n
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear self‑convolution a*a using a pre‑allocated FFT length.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm explores several
    vector lengths, random Dirichlet initialisations and a projected
    sub‑gradient descent on the exact max‑convolution.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best_score = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # Candidate lengths – we try a few powers of two up to 8192.
    # Larger lengths give a lower theoretical lower‑bound, but cost
    # more per iteration.  The loop will automatically stop when the
    # time budget is exhausted.
    # --------------------------------------------------------------
    candidate_lengths = [1024, 2048, 4096, 8192]

    # --------------------------------------------------------------
    # Helper to evaluate the exact objective (no security checks)
    # --------------------------------------------------------------
    def _objective(a: np.ndarray) -> float:
        """Return 2·n·max(a*a)/(sum(a)**2).  For sum = √(2n) this is just max."""
        n = a.size
        fft_len = _next_pow2(2 * n - 1)
        conv = _conv_self(a, fft_len)
        return conv.max()

    # --------------------------------------------------------------
    # Main search loop – iterate over lengths and several restarts
    # --------------------------------------------------------------
    length_index = 0
    while time.monotonic() < deadline:
        # Pick a length; we cycle through the candidate list, but can
        # also randomise a bit to avoid deterministic ordering.
        n = candidate_lengths[length_index % len(candidate_lengths)]
        length_index += 1

        target_sum = math.sqrt(2.0 * n)          # fixed L1‑norm for this length

        # ----------------------------------------------------------
        # Prepare a few deterministic seeds (uniform, 1/√i, √i)
        # ----------------------------------------------------------
        deterministic_inits = []

        # uniform
        uniform = np.full(n, target_sum / n, dtype=np.float64)
        deterministic_inits.append(uniform)

        # decreasing 1/√i shape (known to flatten convolution)
        idx = np.arange(1, n + 1, dtype=np.float64)
        inv_sqrt = 1.0 / np.sqrt(idx)
        inv_sqrt = inv_sqrt / inv_sqrt.sum() * target_sum
        deterministic_inits.append(inv_sqrt)

        # increasing √i shape (just for completeness)
        sqrt_shape = np.sqrt(idx)
        sqrt_shape = sqrt_shape / sqrt_shape.sum() * target_sum
        deterministic_inits.append(sqrt_shape)

        # ----------------------------------------------------------
        # Random Dirichlet initialisations – the number of samples
        # scales with the remaining time.
        # ----------------------------------------------------------
        remaining = deadline - time.monotonic()
        # roughly allow 0.1 s per random sample (very cheap) and a few
        # gradient steps afterwards – adjust on the fly.
        max_random_samples = max(1, int(remaining / 0.12))

        for _ in range(max_random_samples):
            # Dirichlet with a random concentration (sparseness) parameter
            alpha = rng.uniform(0.05, 2.0)
            a = rng.gamma(alpha, 1.0, size=n)
            a = a / a.sum() * target_sum
            deterministic_inits.append(a)

        # ----------------------------------------------------------
        # Process each initial vector by a short projected gradient run
        # ----------------------------------------------------------
        for init_a in deterministic_inits:
            if time.monotonic() >= deadline:
                break

            a = init_a.copy()

            # Parameters for the projected sub‑gradient descent
            max_iters = 2000                     # total iterations for this start
            lr = 0.8                              # initial learning‑rate
            lr_decay = 0.5                        # decay every 500 iters
            jitter_std = 0.02 * target_sum / n   # small stochastic jitter

            # Pre‑compute FFT length (same for all iters with this n)
            fft_len = _next_pow2(2 * n - 1)

            # Main optimisation loop for this start point
            for it in range(max_iters):
                now = time.monotonic()
                if now >= deadline:
                    break

                # ---- self‑convolution -------------------------------------------------
                conv = _conv_self(a, fft_len)
                M = conv.max()                     # max of the convolution (objective)

                # ---- keep the best solution -------------------------------------------
                if M < best_score - 1e-12:
                    best_score = M
                    best_seq = a.copy()
                    elapsed = now - start
                    print(f"[{elapsed:6.2f}s] n={n:5d} it={it:4d}  new best = {best_score:.6f}")

                # ---- sub‑gradient of the max‑convolution --------------------------------
                k = int(np.argmax(conv))           # an index where the max is attained

                # Indices that actually contribute to conv[k]
                i_start = max(0, k - (n - 1))
                i_end = min(n - 1, k)
                idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
                idx_j = k - idx_i

                # grad_M = d/d a of conv[k] (the active shift)
                grad_M = np.zeros_like(a)
                grad_M[idx_i] += a[idx_j]
                grad_M[idx_j] += a[idx_i]          # includes the i=j case automatically

                # ---- full gradient of the objective (factor = 1 because S² = 2n) ---
                c = 2.0 * M / target_sum            # scalar term from denominator
                grad_f = grad_M - c                # ∇f(a)

                # ---- gradient step ----------------------------------------------------
                a = a - lr * grad_f

                # ---- projection onto the simplex (keeps sum = target_sum) ------------
                a = _proj_simplex(a, target_sum)

                # ---- learning‑rate schedule & occasional jitter -----------------------
                if (it + 1) % 500 == 0:
                    lr *= lr_decay

                if (it + 1) % 250 == 0:
                    # Add a little random noise to help escape plateaus
                    a += rng.normal(0.0, jitter_std, size=n)
                    a = np.maximum(a, 0.0)          # keep non‑negative
                    a = _proj_simplex(a, target_sum)

        # ----------------------------------------------------------
        # If the remaining time is tiny, break early.
        # ----------------------------------------------------------
        if deadline - time.monotonic() < 1.0:
            break

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        # a simple uniform construction of moderate size
        n = 1024
        best_seq = np.full(n, math.sqrt(2.0 * n) / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Return the best sequence, clipped to the limits enforced by the
    # evaluator (0 ≤ value ≤ 1000).
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq.tolist()]
