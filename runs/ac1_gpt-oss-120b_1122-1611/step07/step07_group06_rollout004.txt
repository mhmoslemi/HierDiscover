        n = rng.integers(2000, 8000)   # maybe small for speed

        sum_counts = np.zeros(2*n-1, dtype=np.int16)  # enough size

        cand_order = rng.permutation(n)

        for idx in cand_order:

            if len(indices) == 0:
                # only self-pair
                if sum_counts[2*idx] + 1 > c_allowed:

    indices = []

"""
B₂[2] (Sidon‑type) set construction.

The evaluation function is

    score = 2·n·max(conv(a,a)) / (sum(a))²

For a non‑negative vector a with total mass 1 and support of size k,
max(conv) = (max number of ordered index pairs i+j = s) / k².
If we keep the number of ordered pairs per sum ≤ C (C = 2 here) we obtain

    score ≤ 2·n·C / k² .

Choosing a small C and a large support k (≈ √(C·(2n–1))) therefore gives a
very good bound.  The routine below repeatedly builds random B₂[2] sets
(i.e. sets where each sum occurs at most twice) with a greedy algorithm,
keeps the best construction seen and finally returns the corresponding
step‑function (all non‑zero entries have the same height 1/k).
The search is performed inside the allotted time budget, evaluates candidates
analytically (no costly convolution) and calls ``evaluate_sequence`` only
for promising vectors.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _build_b2_set(n: int, max_per_sum: int, rng: np.random.Generator):
    """
    Greedy construction of a set S ⊆ {0,…,n‑1} such that for every sum s
    the ordered pair count |{(i,j)∈S² : i+j = s}| ≤ max_per_sum.
    Returns the numpy array of chosen indices and the array of pair counters.
    """
    # ordered‑pair counters for all possible sums 0 … 2·n‑2
    pair_counts = np.zeros(2 * n - 1, dtype=np.int16)
    chosen = []

    # random order – different runs give different sizes
    order = rng.permutation(n)

    for idx in order:
        # self‑pair (i,i) contributes 1 to sum 2·idx
        s_self = idx + idx
        if pair_counts[s_self] + 1 > max_per_sum:
            continue

        if chosen:
            # sums with all previously selected indices
            sums = np.add(chosen, idx)          # vectorised
            # each unordered pair creates two ordered pairs → +2
            if np.any(pair_counts[sums] + 2 > max_per_sum):
                continue
            # all constraints satisfied → update counters
            pair_counts[sums] += 2
        # finally add the self‑pair
        pair_counts[s_self] += 1
        chosen.append(idx)

    return np.asarray(chosen, dtype=np.int32), pair_counts


def _score_from_counts(n: int, k: int, max_cnt: int) -> float:
    """
    Analytical upper‑bound on the evaluator's value for a vector with:
        - length n,
        - support size k,
        - maximal ordered‑pair count per sum = max_cnt,
        - equal weights 1/k on the support.
    """
    # max(conv) = max_cnt / k²  (total mass = 1)
    return 2.0 * n * max_cnt / (k * k)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The algorithm repeatedly builds random B₂[2] sets (max 2 ordered pairs
    per sum) of various lengths, keeps the best construction found and
    returns it as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # Leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) optional starting point (the previous best, if any)
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if np.isfinite(cand).all():
                sc = evaluate_sequence(cand.tolist())
                if sc < best_score:
                    best_score, best_seq = sc, cand.copy()
                    print(f"[init] known sequence score = {sc:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) main random‑set search
    # ------------------------------------------------------------------
    # parameters of the construction
    MAX_PER_SUM = 2                     # allow at most 2 ordered pairs per sum
    MIN_N = 2000                        # smallest length we try
    MAX_N = 6000                        # largest length we try

    # counters for statistics
    it = 0
    while time.monotonic() < deadline:
        it += 1

        # random length – uniform on a reasonable interval
        n = rng.integers(MIN_N, MAX_N + 1)

        # several independent greedy builds for the same n
        for _ in range(3):
            support, counts = _build_b2_set(n, MAX_PER_SUM, rng)
            k = support.size
            if k < 2:                         # degenerate, skip
                continue

            max_cnt = int(counts.max())       # ≤ MAX_PER_SUM
            approx_score = _score_from_counts(n, k, max_cnt)

            # quick filter – only compute the exact score if we might beat the best
            if approx_score < best_score * 0.99:
                a = np.zeros(n, dtype=np.float64)
                weight = 1.0 / k
                a[support] = weight
                exact_score = evaluate_sequence(a.tolist())
                if exact_score < best_score:
                    best_score = exact_score
                    best_seq = a.copy()
                    print(f"[{time.monotonic():.2f}s] n={n:4d} k={k:3d} "
                          f"max_cnt={max_cnt} score={exact_score:.6f}")

        # stop early if we already reached the target bound
        if best_score <= 1.5025:
            break

    # ------------------------------------------------------------------
    # 3) fallback – uniform vector (never used in practice)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[fallback] uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) final safety clipping (the evaluator already does this)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return as plain Python list
    return [float(x) for x in best_seq]
