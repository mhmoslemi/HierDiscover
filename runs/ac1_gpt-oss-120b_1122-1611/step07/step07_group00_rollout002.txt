        r = rng.random()

The following `propose_candidate` implements a *hybrid* optimizer that mixes three ideas  

1. **Adam on the probability simplex** – the sequence is stored as logits `x`;
   `a = softmax(x)` guarantees non‑negativity and unit sum.  
2. **A smooth approximation of the max‑convolution** – for most of the run a
   log‑sum‑exp surrogate is used (parameter γ increasing from 0.5 to a large
   value). In the last 20  % of the time the surrogate is replaced by a true
   sub‑gradient (binary mask of the currently maximal convolution entries).
3. **Simulated‑annealing style perturbations** – occasional random block
   scalings, pairwise mass transfers and global log‑normal jitter help the
   search escape shallow local minima.  Stagnation is detected and a restart
   with moderate noise is performed.

All operations are O(N log N) thanks to FFT based convolutions, allowing
hundreds of thousands of iterations within the 1000 s budget.  The best
sequence found is returned as a plain list of floats (already clipped to the
limits imposed by `evaluate_sequence`).  The algorithm is completely
self‑contained and does not rely on any external files.

```python
"""
Hybrid Adam + simulated‑annealing optimizer on the simplex.

* The sequence `a` is represented by logits `x` (soft‑max → probability vector).
* A smooth max–convolution surrogate Fγ(a) = (1/γ)·log Σk exp(γ·(a*a)k) is
  minimised with Adam.  γ grows from 0.5 to a large value (≈150); after 80 % of
  the time the surrogate is replaced by the exact sub‑gradient (binary mask
  of the current maximal convolution entries).
* Random block perturbations, pairwise mass transfers and occasional global
  jitter are applied; if no improvement has been seen for >30 s a restart
  (log‑normal noise around the current best) is performed.

The routine respects the 1000 s budget (5 s safety margin) and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns a vector of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_grad_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of  Σk w_k·(a*a)_k  with respect to a.
    ``w`` must have length 2·len(a)‑1 and be non‑negative.
    The returned gradient has the same shape as ``a``.
    """
    n = a.size
    a_rev = a[::-1]                     # reversed a
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(x, 1.0 / x.size)
    return e / s


def _fast_score(a: np.ndarray, fft_len: int) -> float:
    """Fast evaluation of the objective using FFT (sum(a) is assumed = 1)."""
    n = a.size
    conv = _self_conv_fft(a, fft_len)
    return 2.0 * n * conv.max()


def _exact_score(a: np.ndarray) -> float:
    """
    Exact objective (identical to `evaluate_sequence` apart from
    clipping / safety checks which are already satisfied by our iterates).
    """
    n = a.size
    sum_a = a.sum()
    if sum_a < 1e-8:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max() / (sum_a ** 2)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    max_width = max(1, n // 20)               # ≤ 5 % of the sequence
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _pairwise_transfer(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Move a small fraction of mass from a random source index to a random
    target index (preserving the total sum).
    """
    n = a.size
    src = rng.integers(0, n)
    dst = rng.integers(0, n)
    if src == dst or a[src] == 0.0:
        return a
    # transfer up to 15 % of the source weight
    delta = rng.random() * 0.15 * a[src]
    out = a.copy()
    out[src] -= delta
    out[dst] += delta
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **_: object) -> list[float]:
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds (5 s safety margin is
    reserved for the surrounding harness).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) time budget (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) initial sequence
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)) and len(init_seq) > 0:
        a0 = np.asarray(init_seq, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
        x = np.log(a0 + 1e-12)                     # logits
    else:
        # fallback – uniform distribution of length ≈ 3800
        n = 3800
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        x = np.log(a0 + 1e-12)

    # --------------------------------------------------------------
    # 2) pre‑compute FFT length for self‑convolution
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 3) Adam state
    # --------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # --------------------------------------------------------------
    # 4) bookkeeping
    # --------------------------------------------------------------
    a = _softmax(x)
    cur_score = _fast_score(a, fft_len_self)
    best_a = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 5) hyper‑parameter schedules
    # --------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 150.0               # larger than the previous 30
    lr_start = 0.5
    lr_end = 0.015                  # a bit lower final learning‑rate

    # --------------------------------------------------------------
    # 6) main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # ---- current iterate -------------------------------------------------
        a = _softmax(x)                       # probability vector (sum = 1)

        # ---- self‑convolution ------------------------------------------------
        conv = _self_conv_fft(a, fft_len_self)   # length 2·n‑1
        max_conv = conv.max()

        # ---- choose smoothing / sub‑gradient ---------------------------------
        # after 80 % of the time we switch to a hard mask (true sub‑gradient)
        if frac < 0.80:
            gamma = gamma_start + (gamma_end - gamma_start) * (frac ** 1.5)
            # log‑sum‑exp weights
            shifted = conv - max_conv          # stabilise exponentials
            w = np.exp(gamma * shifted)
            w_sum = w.sum()
            # avoid division by zero (should never happen)
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum
        else:
            # hard mask of the “almost maximal’’ entries
            tol = max(1e-12, max_conv * 1e-6)   # tolerance relative to max
            mask = (conv >= max_conv - tol).astype(float)
            m_sum = mask.sum()
            if m_sum == 0.0:
                mask = np.ones_like(conv) / conv.size
                m_sum = 1.0
            w = mask / m_sum                    # normalised mask

        # ---- gradient w.r.t. a ------------------------------------------------
        g_a = _conv_grad_wrt_a(a, w)            # length n

        # ---- chain rule to logits x (mirror‑descent on simplex) -------------
        avg = np.dot(a, g_a)                    # <a,∇F>
        g_x = a * (g_a - avg)                   # ∂F/∂x  (natural gradient)

        # ---- Adam update -----------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ---- exact objective (occasionally for safety) -----------------------
        # we recompute the exact objective only when the fast score improves,
        # because the FFT based score is already a very good proxy.
        fast_score = 2.0 * n * max_conv
        if fast_score < best_score - 1e-12:
            exact = _exact_score(a)               # cheap – linear convolution
            if exact < best_score - 1e-12:
                best_score = exact
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- stochastic perturbations ----------------------------------------
        # 2 % chance of a global jitter (log‑normal noise around the current best)
        if rng.random() < 0.02:
            pert = _random_block_perturb(best_a, rng)
            x = np.log(pert + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] global jitter – best score {best_score:.6f}")

        # occasional pairwise mass transfer (≈3 % of iterations)
        if rng.random() < 0.03:
            a_tmp = _pairwise_transfer(a, rng)
            # just replace the current iterate (no restart of Adam state)
            x = np.log(a_tmp + 1e-12)

        # ---- stagnation handling ---------------------------------------------
        if now - last_improve > 30.0:
            # re‑initialise from the best solution with moderate log‑normal noise
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

    # -----------------------------------------------------------------------
    # 7) Return best sequence (clipped to the limits of `evaluate_sequence`)
    # -----------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
```
