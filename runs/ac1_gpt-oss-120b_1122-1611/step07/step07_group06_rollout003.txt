"""
Gradient‑subgradient optimiser for the self‑convolution bound.

The routine builds a non‑negative vector a on the probability simplex
(∑a = 1) and minimises the exact objective  

        f(a) = 2·n·max(a*a)               (since the denominator is 1)

using a projected Adam optimiser driven by the sub‑gradient of the
max‑convolution (the true, non‑smooth objective).  The algorithm works
in three phases inside the given time budget:

1. **Random initialisation** – a Dirichlet draw gives a non‑uniform
   starting point, breaking the symmetry of the uniform stationary
   point.
2. **Projected Adam on the exact sub‑gradient** – each iteration
   computes the self‑convolution by FFT, extracts the active set
   (indices attaining the current maximum), forms the sub‑gradient,
   applies Adam updates and projects back onto the simplex.
   The learning‑rate and Adam moments are annealed; every few hundred
   steps a small multiplicative jitter on a random block is applied
   to escape shallow plateaux.
3. **Local stochastic refinement** – after the main loop the best
   vector is perturbed randomly (log‑normal multiplicative noise on a
   random block) and kept if it improves the exact score.  This stage
   runs until the overall deadline.

All heavy work (convolution, correlation) uses FFTs, making an
iteration cheap even for sequences with tens of thousands of entries.
The best sequence found is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be projected to zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to avoid drift
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    The optimiser uses a projected Adam method on the exact max‑convolution
    sub‑gradient, supplemented by stochastic jitter and a final
    local refinement stage.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # 1) Choose a working length (power of two) – larger gives tighter bound
    # --------------------------------------------------------------
    # start from a decent size; we have plenty of time, so use 16384.
    # If a previous solution exists and is longer, honour that length.
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev_len = prev.size
    else:
        prev_len = 0

    # Desired length (power of two, at least 8192)
    target_len = max(8192, 1 << (prev_len - 1).bit_length())
    # Cap to a reasonable size (to keep FFT cheap)
    target_len = min(target_len, 32768)

    n = target_len
    print(f"[0.00s] Using sequence length n = {n}")

    # --------------------------------------------------------------
    # 2) Initialise a probability vector on the simplex
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64).copy()
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        if a0.size > n:
            a = a0[:n]
        else:
            pad = n - a0.size
            a = np.concatenate([a0, np.full(pad, 0.0)])
        # re‑project to guarantee exact sum = 1
        a = _project_onto_simplex(a)
    else:
        # random Dirichlet initialisation (alpha < 1 gives more variability)
        alpha = 0.5
        a = rng.dirichlet(np.full(n, alpha))
        a = _project_onto_simplex(a)

    # --------------------------------------------------------------
    # 3) Pre‑compute FFT length (large enough for convolution + correlation)
    # --------------------------------------------------------------
    fft_len = _next_pow2(3 * n)  # ≥ n + (2n‑1) – 1
    corr_start = n - 1
    corr_end = corr_start + n  # exclusive slice for correlation result

    # --------------------------------------------------------------
    # 4) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    # Exact score = 2·n·max(conv) because sum = 1
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    best_max = conv.max()
    best_score = 2.0 * n * best_max
    best = a.copy()
    best_time = time.monotonic()
    print(f"[0.00s] Initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Optimisation hyper‑parameters
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = beta1, beta2

    # Learning‑rate schedule (large at start, decays)
    lr_start, lr_end = 0.20, 0.001

    # Jitter schedule – every JITTER_EVERY iterations we perturb a block
    JITTER_EVERY = 400
    # How often we print a progress line
    PRINT_EVERY = 200
    # Restart if we see no improvement for this many seconds
    RESTART_AFTER = 30.0

    iteration = 0

    # --------------------------------------------------------------
    # 6) Main optimisation loop (projected Adam on exact sub‑gradient)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # ---- FFT of current vector ----
        a_fft = np.fft.rfft(a, n=fft_len)

        # ---- self‑convolution (linear) ----
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ---- locate active set (indices attaining the maximum) ----
        max_val = conv.max()
        # tolerance protects against floating‑point noise
        tol = max(1e-12, max_val * 1e-12)
        mask = (conv >= max_val - tol).astype(np.float64)
        mask_sum = mask.sum()  # number of active indices (≥1)

        # ---- correlation = ∑_{k∈I} a_{k‑i}  (linear, non‑circular) ----
        w_fft = np.fft.rfft(mask, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        corr = corr_full[corr_start:corr_end]  # length n

        # ---- sub‑gradient of f(a) = 2·n·max(conv) (since ∑a = 1) ----
        # ∂M/∂a_i = corr_i + mask_sum·a_i
        grad_M = corr + mask_sum * a
        # ∇f = 2·n·grad_M - 4·n·max_val (the latter is a constant vector)
        grad = 2.0 * n * (grad_M - 2.0 * max_val)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- gradient step + projection onto simplex ----
        a = a - step
        a = np.maximum(a, 0.0)
        a = _project_onto_simplex(a)

        iteration += 1

        # ---- occasional jitter to shake the system ----
        if iteration % JITTER_EVERY == 0:
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.09, size=width))
            a_jit = a.copy()
            a_jit[start:start + width] *= factor
            a_jit = np.clip(a_jit, 0.0, None)
            a_jit = _project_onto_simplex(a_jit)
            a = a_jit

        # ---- periodic exact‑objective assessment (cheap) ----
        if iteration % PRINT_EVERY == 0:
            # score = 2·n·max(conv) because sum(a)=1
            score = 2.0 * n * max_val
            if score < best_score - 1e-12:
                best_score = float(score)
                best = a.copy()
                best_time = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f} "
                      f"(iter={iteration}, max={max_val:.6e})")

        # ---- restart if stagnated for a while ----
        if now - best_time > RESTART_AFTER:
            # re‑inject a modest amount of noise around the current best
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a *= noise
            a = _project_onto_simplex(a)
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = beta1, beta2
            best_time = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – injected noise")
            continue

    # --------------------------------------------------------------
    # 7) Final local stochastic refinement (cheap objective)
    # --------------------------------------------------------------
    # Use the fast objective (2·n·max(conv)) for many cheap tries.
    while time.monotonic() < deadline:
        cand = best.copy()
        width = rng.integers(1, max(2, n // 15))
        start = rng.integers(0, n - width + 1)
        factor = np.exp(rng.normal(0.0, 0.10, size=width))
        cand[start:start + width] *= factor
        cand = np.clip(cand, 0.0, None)
        cand = _project_onto_simplex(cand)

        # cheap score
        a_fft = np.fft.rfft(cand, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = float(score)
            best = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] refinement improvement = {best_score:.6f}")

    # --------------------------------------------------------------
    # 8) Return the best sequence (clipped for safety)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
