def project_simplex(v):
    """Euclidean projection of v onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid optimiser for the self‑convolution problem.

Phase 1 – Adam optimisation on a soft‑max surrogate of the max‑convolution
           (identical in spirit to the reference code but with a slower
           learning‑rate schedule and a longer β‑ramp).  This quickly finds a
           good basin.

Phase 2 – Projected sub‑gradient descent on the exact max‑convolution.
           After the soft‑max has essentially converged the exact max is
           used; the gradient of the active offset is cheap to compute and a
           simple Euclidean projection onto the probability simplex (non‑negative,
           unit sum) drives the iterate towards a flatter convolution.

Multiple random restarts, occasional jitter and an adaptive step size are
used to escape plateaus.  The best sequence seen over the whole budget is
returned (clipped to the limits imposed by `evaluate_sequence`).

The implementation avoids closures, uses only top‑level helpers and works
within the 1000 s time budget.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (real‑valued)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (sum(a) is not forced to 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # Fast path for already feasible vectors
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be projected to zero – fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """One projected sub‑gradient step on max‑convolution."""
    n = a.shape[0]
    conv = _conv_self(a)
    kmax = int(np.argmax(conv))                     # active offset
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]                   # ∂ max / ∂ a_i
    return _project_simplex(a - step * grad)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method first runs an Adam optimiser on a
    smooth approximation of the infinity‑norm of the self‑convolution and then
    refines the result with a projected sub‑gradient descent on the exact
    max‑convolution.  Random restarts and jitter help to avoid stagnation.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start + total_budget
    phase1_end = start + total_budget * 0.6        # 60 % of time for phase 1

    # --------------------------------------------------------------
    # Choose a length n and an initial distribution.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        base = base / base.sum()
        n = max(3000, base.size)                 # work with a sufficiently large vector
        if base.size != n:
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(new_x, old_x, base)
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
        else:
            a = base.copy()
    else:
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Keep the best solution seen so far
    best_a = a.copy()
    best_score = _objective_score(a)
    print(f"initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Adam on a smooth surrogate of max(conv)
    # --------------------------------------------------------------
    eps = 1e-12
    x = np.log(a + eps)                # logits → simplex via soft‑max

    m = np.zeros_like(x)               # Adam first moment
    v = np.zeros_like(x)               # Adam second moment

    lr = 0.4
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    beta_start = 5.0
    beta_end = 400.0
    beta_ramp_iters = 3000

    RESTART_WAIT = 40.0                # seconds without improvement ⇒ hard restart
    last_improve = start

    fft_len_conv = _next_pow2(2 * n - 1)
    it = 0
    while time.monotonic() < phase1_end:
        it += 1

        # logits → probability vector (sum = 1)
        a = _softmax_logits(x)

        # linear self‑convolution a * a via FFT
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]

        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max          # denominator is 1 (sum(a)=1)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase1 new best {best_score:.6f}")

        # soft‑max over the convolution → smooth surrogate of max
        beta = (beta_end if it > beta_ramp_iters
                else beta_start + (beta_end - beta_start) * it / beta_ramp_iters)

        scaled = beta * conv_full
        lse = logsumexp(scaled)               # robust log‑sum‑exp
        w = np.exp(scaled - lse)               # shape (2n‑1,), sums to 1

        # gradient of the surrogate w.r.t. a (correlation w ⋆ a)
        fft_len_corr = _next_pow2(3 * n - 2)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)

        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # chain rule → gradient w.r.t. logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to escape plateaus
        if it % 400 == 0:
            x += rng.normal(0.0, 0.015, size=n)

        # hard restart if we have been stagnant for a while
        if time.monotonic() - last_improve > RESTART_WAIT:
            x = np.log(best_a + eps) + rng.normal(0.0, 0.12, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase1 restart")

    # --------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    # start from the best sequence discovered in phase 1
    a = best_a.copy()
    step = 0.1
    sub_start = time.monotonic()

    while time.monotonic() < deadline:
        # exact convolution and the active offset
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # gradient of the active offset (see analysis)
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # tentative update and projection onto the simplex
        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step *= 1.05                      # enlarge step after a success
            print(f"[{time.monotonic() - start:6.2f}s] subgrad improvement {best_score:.6f}")
        else:
            step *= 0.7                       # shrink step after a failure
            # occasional random jitter to shake out of flat regions
            if step < 1e-8 or rng.random() < 0.001:
                jitter = rng.normal(0.0, 0.002, size=n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.01)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
