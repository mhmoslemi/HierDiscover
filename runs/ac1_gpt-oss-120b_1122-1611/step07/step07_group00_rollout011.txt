def _project_simplex(v):
    """Projection of vector v onto the probability simplex (nonnegative entries sum to 1)."""

"""
Hybrid gradient–annealing search for a step‑function that minimises
    2·n·max(conv(a,a)) / (sum(a)²)

* a is constrained to be non‑negative and to sum to 1 (the objective is
  scale‑invariant, so we work on the probability simplex).
* A smooth‑max surrogate (log‑sum‑exp) is optimised with Adam; the
  smoothing parameter γ is increased from 0.5 up to a large value so the
  landscape eventually matches the true max.
* Stagnation triggers a full random restart from a low‑concentration
  Dirichlet distribution (explores very non‑uniform shapes).
* Occasional “jitter’’ mutates a random block multiplicatively.
* After the main Adam phase we perform a short projected sub‑gradient
  descent on the exact max‑conv objective to polish the best solution.
* All convolutions are done with a single FFT (O(n log n)).

The routine respects the 1000 s budget and returns the best sequence
found as a list of Python floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns a vector of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient ∂F/∂a for the smoothed‑max surrogate.
    ``w`` is a probability vector of length 2·n‑1 (the soft‑max weights).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    return 2.0 * conv[start:start + n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max producing a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex:
        {x | x ≥ 0, sum(x) = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main optimisation routine
# ----------------------------------------------------------------------

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(np.maximum(a0, 1e-12))
        else:
            n = 3809
            x = np.zeros(n, dtype=np.float64)  # uniform on simplex (log‑weights)
    else:
        n = 3809
        x = np.zeros(n, dtype=np.float64)      # uniform log‑weights

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Track the best solution
    # ------------------------------------------------------------------
    a = _softmax(x)
    c = _self_conv(a, fft_len_self)
    best_a = a.copy()
    best_score = 2.0 * n * c.max()
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 5) Hyper‑parameter schedules
    # ------------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 200.0          # final smoothing → true max
    lr_start = 0.5
    lr_end = 0.001             # final learning‑rate
    stagnation_limit = 12.0    # seconds without improvement → restart
    jitter_prob = 0.015        # probability per iteration

    # ------------------------------------------------------------------
    # 6) Main optimisation loop (Adam on smoothed max)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time   # ∈[0,1]

        # ----- current iterate on the simplex -----
        a = _softmax(x)

        # ----- self‑convolution (full) -----
        c = _self_conv(a, fft_len_self)

        # ----- smooth‑max weights -----
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        c_shift = c - np.max(c)               # improve numerical stability
        w = np.exp(gamma * c_shift)
        w /= w.sum()                          # now a probability vector

        # ----- gradient w.r.t. a -----
        g_a = _gradient_wrt_a(a, w)

        # ----- chain rule to log‑weights x (softmax reparameterisation) -----
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- exact objective value (no smoothing) -----
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- restart if we have been stagnant for a while -----
        if now - last_improve > stagnation_limit:
            # Dirichlet with low concentration (spiky candidates)
            alpha = np.full(n, 0.2)
            a_restart = rng.dirichlet(alpha)
            rng.shuffle(a_restart)                 # random ordering matters
            x = np.log(np.maximum(a_restart, 1e-12))
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # ----- occasional jitter to escape shallow minima -----
        if rng.random() < jitter_prob:
            if best_a is not None:
                a_jit = _random_block_perturb(best_a, rng)
                x = np.log(np.maximum(a_jit, 1e-12))
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                print(f"[{elapsed:6.2f}s] jitter applied")

    # ------------------------------------------------------------------
    # 7) Optional short projected sub‑gradient refinement on the exact max
    # ------------------------------------------------------------------
    # Use any remaining time (if any) for a few hundred iterations of
    # sub‑gradient descent on the true max‑conv objective.
    remaining = deadline - time.monotonic()
    if remaining > 2.0 and best_a is not None:
        a = best_a.copy()
        # Simple linear decay of step size
        sub_iters = int(5000 * (remaining / total_time)) + 1
        for it in range(sub_iters):
            c = _self_conv(a, fft_len_self)
            k = int(np.argmax(c))                 # index of maximal entry
            # gradient of conv_k w.r.t. a (see analysis)
            g = np.zeros_like(a)
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            idx = np.arange(lo, hi + 1)
            g[idx] = 2.0 * a[k - idx]
            # step size – decay with iteration count
            eta = 0.02 * (1.0 - it / sub_iters)
            a = a - eta * g
            a = _project_simplex(a)               # stay on the simplex
            # keep the best during refinement
            cur_score = 2.0 * n * np.max(_self_conv(a, fft_len_self))
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
        print(f"[refine] final refined best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = _softmax(x)

    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]
