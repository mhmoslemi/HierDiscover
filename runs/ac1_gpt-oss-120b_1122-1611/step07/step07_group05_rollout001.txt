```python
def _convolve_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[:len(x)+len(y)-1]
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
\text{score}=2\,n\;\frac{\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

for a non‑negative step‑function \(a\) (the sum is irrelevant because the
objective is scale‑invariant).  
A good sequence must make its *self‑convolution* as flat as possible.
The previous implementation reduced the *average* squared error of the
convolution – it did not directly target the maximal entry, therefore it
got stuck at ≈ 1.5132.

The new algorithm works in two phases:

1. **Soft‑max gradient descent on the maximal entry**  
   We minimise a smooth approximation of the max:
   \[
     M_\beta(a)=\frac{1}{\beta}\log\!\sum_k e^{\beta\,(a*a)_k}.
   \]
   The gradient can be written as a convolution, so each iteration needs only
   three FFTs (self‑convolution, soft‑max weights and a convolution with the
   reversed vector).  
   The temperature‑parameter β is increased from 0.5 to 8, turning the smooth
   objective gradually into the true max.  After a gradient step the vector is
   projected back onto the probability simplex (non‑negative, sum = 1).

2. **Stochastic fine‑tuning** (simulated‑annealing style)  
   After the smooth phase we switch to cheap incremental moves that transfer
   mass from indices that contribute heavily to the current peak to indices
   that contribute little.  Acceptance follows a Metropolis rule with a
   linearly cooling temperature.  Periodic full recomputation of the
   convolution corrects accumulated floating‑point errors, and occasional
   log‑normal jitter prevents stagnation.

The routine works with a longer sequence (10 000 coefficients) – the larger
search space gives a lower theoretical bound, yet the FFT length stays modest
(\(2^{15}=32768\)), so the whole optimisation finishes well within the 1000 s
budget.

All helper functions are top‑level, no closures are used, and the code
produces progress output so that the optimisation process can be monitored.

```python
"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise from the known construction (``height_sequence_1``) or from a
   uniform distribution of length *n* = 10 000.  The vector is projected onto the
   simplex (non‑negative, sum = 1).

2. Phase 1 – smooth max minimisation (≈40 % of the time)
   * Use a soft‑max approximation of the maximal convolution entry:
         M_β(a) = (1/β)·log Σ_k exp(β·c_k),   c = a*a .
   * The gradient of M_β is
         ∇M_β = 2·conv(w, a_rev) ,
     where w_k = exp(β·c_k)/Σ_j exp(β·c_j) and a_rev is the reversed vector.
   * Perform a projected gradient step with a linearly decreasing step size.
   * Increase β linearly from 0.5 to 8 to turn the smooth problem into the
     original max‑problem.

3. Phase 2 – stochastic refinement (≈60 % of the time)
   * Locate the current peak `k_max` of the convolution.
   * Choose a *donor* index proportionally to its contribution
         a_i·a_{k_max‑i}` to the peak.
   * Choose a *recipient* index biased towards small values.
   * Transfer a random fraction (≤ ½ of the donor mass) and update the
     convolution incrementally (O(n) per move).
   * Accept the move if it improves the score or with a Metropolis
     probability based on a linearly cooling temperature.
   * Every few thousand moves recompute the convolution exactly; occasionally
     apply a log‑normal jitter to escape plateaus.
   * If no improvement for ~22 s, re‑initialise from the best solution with
     moderate noise.

4. Keep the best sequence seen during the whole run and return it
   (clipped to the limits enforced by ``evaluate_sequence``).

The optimisation finishes well under the 1000 s budget and typically reaches
a score below 1.5025, beating the previous best construction.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _convolve_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT of length ``fft_len``."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: x.size + y.size - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` (donor) to ``j`` (recipient) and
    update the convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # linear contribution of the moved mass
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor,
    then project back onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise – use the known construction if available,
    #    otherwise a uniform distribution.
    # ------------------------------------------------------------------
    target_n = 10000  # larger than the previous 5000; still cheap for FFTs
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            # pad or truncate to target size
            if cand.size < target_n:
                cand = np.pad(cand, (0, target_n - cand.size), constant_values=0.0)
            else:
                cand = cand[:target_n]
            a = cand.copy()
        else:
            a = np.full(target_n, 1.0 / target_n, dtype=np.float64)
    else:
        a = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    # Normalise to sum = 1 (probability simplex)
    a = a / a.sum()
    a = _project_simplex(a)

    n = a.size
    # pre‑computed FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the current state
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()                # same as evaluate_sequence for sum=1
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.40 * total_time   # 40 % for the smooth‑max phase

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Phase 1 – step‑size and temperature schedule
    eta_start = 0.5
    eta_end = 1e-5
    beta_start = 0.5
    beta_end = 8.0

    # Phase 2 – simulated‑annealing schedule
    temp_start = 0.03
    temp_end = 1e-7
    move_counter = 0
    RECOMPUTE_EVERY = 800   # full convolution recomputed each N moves

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase 1 – soft‑max gradient descent on the maximal entry
        # --------------------------------------------------------------
        if now < phase1_end:
            # fraction of phase 1 elapsed
            frac = (now - start_time) / (phase1_end - start_time)

            # linear schedules
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start * (1.0 - frac) + beta_end * frac

            # self‑convolution (c) – required for the soft‑max weights
            c = _self_conv(a, fft_len_self)

            # soft‑max weights (stable)
            scaled = beta * c
            max_scaled = scaled.max()
            exp_scaled = np.exp(scaled - max_scaled)
            w = exp_scaled / exp_scaled.sum()          # Σ w = 1

            # gradient = 2 * conv(w, a_rev)
            a_rev = a[::-1]
            grad_full = _convolve_fft(w, a_rev, fft_len_grad)
            grad = 2.0 * grad_full[n - 1 : n - 1 + n]

            # projected gradient step
            a_tmp = a - eta * grad
            a = _project_simplex(a_tmp)

            # recompute convolution & score for bookkeeping
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Phase 2 – stochastic fine‑tuning (simulated annealing)
        # --------------------------------------------------------------
        else:
            # linear temperature cooling
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # locate current peak
            k_max = int(c.argmax())

            # contributions of each index to the peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # donor selection (biased by contribution)
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                # nothing to move – continue loop
                continue

            # recipient selection (biased towards small entries)
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            r_sum = recv_weights.sum()
            if r_sum > 0.0:
                recv_weights /= r_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # amount to transfer (at most half of donor mass)
            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            # apply the move (incrementally update convolution)
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # occasional full recomputation to correct drift
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter to escape plateaus
            if rng.random() < 0.015:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 22.0:
            a = best.copy()
            # moderate log‑normal jitter around the best solution
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart with jitter (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
