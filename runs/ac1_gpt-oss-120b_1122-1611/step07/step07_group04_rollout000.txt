"""
Propose a candidate step‑function by stochastic hill‑climbing with simulated
annealing.  The method starts from the best known sequence (if available) or
from a diversified set of simple initialisations.  It repeatedly proposes
tiny mass transfers between two random positions (or a small block‑wise
perturbation), evaluates the true objective

    J(a) = 2·n·max(a*a) / (sum(a)**2)

using an FFT‑based convolution, and accepts the move if it improves the
score or with a probability that decreases with a temperature schedule.
The current best solution is kept throughout the run and is returned
as a plain Python list when the time budget expires.

All helper functions are defined at module level (no closures) and the
algorithm respects the 1000 s budget with a 5 s safety margin.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute the exact objective used by ``evaluate_sequence``.
    The vector a is assumed non‑negative; the function does not apply
    clipping or the 0.01 sum guard, those are handled by the caller.
    """
    conv = _conv_self(a, fft_len)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:                     # safety – mirrors evaluate_sequence
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit L1‑norm.
    """
    n = a.size
    # block width between 1 and max(2, n//15) – relatively small
    width = rng.integers(1, max(2, n // 15))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_portfolio(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Create a small set of diversified starting vectors (all normalised to sum=1).
    """
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # Linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # Two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    # Half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # Dirichlet draws with various concentrations
    for concentration in [0.05, 0.2, 0.5, 2.0, 5.0]:
        v = rng.dirichlet(np.full(n, concentration))
        cand.append(v.astype(np.float64))

    return cand

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm is a stochastic hill‑climbing
    optimiser with simulated‑annealing acceptance, occasional block
    perturbations and restarts.  It runs for the allotted time budget and
    returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (reserve a few seconds for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem size and initialise a feasible a
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0 and np.isfinite(a0).all():
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s == 0.0:
                a0 = np.full_like(a0, 1.0 / a0.size)
            else:
                a0 /= s
        else:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # Default length derived from previous best run
        n = 3809
        # Build a small portfolio of seeds and keep the best according to the exact objective
        candidates = _initial_portfolio(n, rng)
        fft_len = _next_pow2(2 * n - 1)
        best_local = None
        best_score = np.inf
        for v in candidates:
            sc = _objective(v, n, fft_len)
            if sc < best_score:
                best_score = sc
                best_local = v.copy()
        a = best_local
    else:
        a = a0.copy()
        n = a.size
        # Normalise to sum = 1 (objective invariant to scaling)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

    # 2) Pre‑compute constants
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # Evaluate the starting point
    cur_score = _objective(a, n, fft_len_self)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_budget = deadline - start_time

    # ------------------------------------------------------------------
    # 3) Main stochastic optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        progress = elapsed / total_budget          # ∈[0,1]

        # ----- annealing schedules -----
        # Learning step for the mass transfer (fraction of the source entry)
        step_frac = 0.12 * (1.0 - progress) + 0.004 * progress

        # Temperature for Metropolis acceptance (controls uphill moves)
        temp_start, temp_end = 5e-4, 5e-6
        temperature = temp_start * (1.0 - progress) + temp_end * progress

        # ----- propose a move -----
        if rng.random() < 0.15:
            # Block perturbation (larger, less frequent)
            a_new = _random_block_perturb(a, rng)
        else:
            # Small transfer of mass from i -> j
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1   # ensure i != j
            src_mass = a[i]
            if src_mass == 0.0:
                continue  # nothing to move, try next iteration
            eps = step_frac * src_mass * rng.random()
            a_new = a.copy()
            a_new[i] = src_mass - eps
            a_new[j] += eps

        # ----- evaluate -----
        sc = _objective(a_new, n, fft_len_self)

        # ----- acceptance rule -----
        accept = False
        if sc < cur_score:
            accept = True
        else:
            # Metropolis condition
            if temperature > 0.0:
                prob = np.exp(-(sc - cur_score) / temperature)
                accept = rng.random() < prob

        if accept:
            a = a_new
            cur_score = sc
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        if now - last_improve > 45.0:

            noise = np.exp(rng.normal(0.0, 0.07, size=n))

            if a_sum == 0.0:
                a[:] = 1.0 / n
            else:
                a /= a_sum
            cur_score = _objective(a, n, fft_len_self)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score still {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best solution (clipped to evaluate_sequence bounds)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
