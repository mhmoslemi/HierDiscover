def simplex_projection(v):
    """Project v onto simplex {x >= 0, sum x = 1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho = max_i { u_i - cssv_i / i > 0 }
    # Actually condition u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # then all values negative, just distribute uniformly? But rarely.
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # ensure sum to 1 (due to numeric)
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w[:] = 1.0 / len(v)
    return w

assistantfinal**Solution Overview**

The task is to minimise  

\[
\text{score}= \frac{2n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

for a non‑negative vector `a`.  
Because the score is invariant under a global scaling of `a`, we can (and do) work with
vectors normalised to `sum(a)=1`.  
The problem then is to make the self‑convolution `c = a*a` as flat as possible:
the lower bound is the average value of `c`,
\(\bar c = 1/(2n-1)\), giving a theoretical lower score of \(2n/(2n-1)\approx 1\).

The code below implements a **smooth‑max mirror‑descent** optimiser
with an **Adam** style adaptive learning rate.
The main ideas are

* **Smooth‑max surrogate** –  replace the non‑smooth `max` by  

  \[
  \operatorname{smax}_\alpha(c) = \frac1\alpha\log\!\sum_k e^{\alpha c_k},
  \]

  whose gradient is a weighted average of the autocorrelation terms.
  The temperature `α` is increased from a small to a large value,
  so that the optimisation first sees a smooth landscape and later
  tightens to the true maximum.

* **Additive updates + simplex projection** – Adam updates are applied to the
  (unconstrained) vector, afterwards it is projected onto the probability
  simplex `{a_i ≥ 0, Σ a_i = 1}` using the O(n) algorithm of
  Duchi et al. (2008).  This keeps the iterates feasible without
  the numerical issues of multiplicative updates.

* **Heavy‑tail initialisation** – when no previous solution is available,
  the vector is initialised with the coefficients of the power series  

  \[
  a_i \propto \frac{\binom{2i}{i}}{4^i}\qquad(i=0,\dots ,n-1),
  \]

  i.e. the first `n` terms of the expansion of \((1-z)^{-1/2}\).
  This yields a good starting point (score ≈ π/2) that is already far
  better than the uniform distribution.

* **Exploration & restarts** – a small random log‑normal block perturbation
  is applied every few hundred iterations, and a “soft restart’’ (adding
  a little noise to the current best) is performed after 30 s without
  improvement.

* **Time‑budget handling** – the loop stops 5 s before the supplied budget,
  guaranteeing a return value even if the optimiser has not converged.

The optimiser works entirely with NumPy and FFTs (linear‑time convolution),
so each iteration costs `O(n log n)` and thousands of iterations comfortably
fit into the 1000 s budget.

```python
"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence().  The optimiser uses a smooth‑max surrogate of the
self‑convolution peak, Adam‑style adaptive steps, and a projection onto the
probability simplex.  A good analytic initialisation (coefficients of
(1‑z)^‑½) is used when no previous solution is available.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution (len = len(a)+len(b)-1).
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT of size ``fft_len``."""
    return _fft_convolve(a, a, fft_len)


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("projection works on 1‑d arrays")
    n = v.shape[0]

    # Quick return if already feasible
    if v.min() >= 0 and np.isclose(v.sum(), 1.0):
        return v.copy()

    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # condition u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform vector
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]          # last true index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)

    # renormalise to avoid tiny numerical drift
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w[:] = 1.0 / n
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection.
    """
    n = a.size
    # block width between 1 and max(2, n//20)
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.25))  # larger variance for exploration
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    # project back onto simplex
    out = _simplex_projection(out)
    return out


def _initial_coeffs(n: int) -> np.ndarray:
    """
    Return the first ``n`` coefficients of the series
    (2i choose i) / 4^i  (i.e. the expansion of (1‑z)^‑½).
    These are positive, decreasing and give a good starting point.
    """
    coeffs = np.empty(n, dtype=np.float64)
    coeffs[0] = 1.0
    for i in range(1, n):
        coeffs[i] = coeffs[i - 1] * (2 * i - 1) / (2 * i)
    # normalise to a probability vector
    s = coeffs.sum()
    if s == 0.0:
        coeffs[:] = 1.0 / n
    else:
        coeffs /= s
    return coeffs


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.  The routine
    uses a smooth‑max / Adam optimiser with occasional random block
    perturbations and soft restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose problem size and initialise a on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            # fallback
            n = 4096
            a = _initial_coeffs(n)
    else:
        # no previous candidate – use the analytical initialisation
        n = 4096                     # power‑of‑two size for fast FFTs
        a = _initial_coeffs(n)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)      # for a * w  (len 2n‑1)

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * c.max()
    cur_score = best_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0

    # Hyper‑parameter schedules
    alpha_start = 0.2
    alpha_end = 80.0                # final smooth‑max temperature
    lr_start = 0.5
    lr_end = 0.001

    RESTART_AFTER = 30.0            # seconds without improvement
    PERTURB_PROB = 0.04             # probability of a random block perturb

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        frac_total = elapsed / total_time

        # -------------------- schedules --------------------
        alpha = alpha_start + (alpha_end - alpha_start) * frac_total
        lr = lr_start + (lr_end - lr_start) * (1.0 - frac_total)

        # -------------------- smooth‑max weights --------------------
        c = _self_conv(a, fft_len_self)           # a*a
        c_max = c.max()
        # stable soft‑max
        w = np.exp(alpha * (c - c_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # -------------------- gradient of the smoothed peak --------------------
        conv_full = _fft_convolve(a, w, fft_len_grad)   # length = n + (2n-1) -1 = 3n-2
        g = 2.0 * conv_full[n - 1: 2 * n - 1]           # slice of length n

        # -------------------- Adam update --------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = _simplex_projection(a_new)

        # -------------------- evaluate exact objective --------------------
        c_new = _self_conv(a_new, fft_len_self)
        new_score = 2.0 * n * c_new.max()

        # -------------------- accept if improvement --------------------
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            c = c_new

        # -------------------- occasional random block perturb --------------------
        if rng.random() < PERTURB_PROB:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        # -------------------- keep the best solution --------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # -------------------- soft restart after stagnation --------------------
        if now - last_improve > RESTART_AFTER:
            # add gentle log‑normal noise around the best found vector
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = best * noise
            a = _simplex_projection(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            it = 0
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
