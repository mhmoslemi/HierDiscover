"""Search for a low‑peak self‑convolution step function.

The algorithm starts from an analytically motivated “arcsine” shape  
a_i ∝ 1/√((i+1)(N‑i)).  The vector is represented in log‑space (u = log a) and
optimised with Adam on the simplex (softmax(u)).  A smooth approximation
to the `max` in the objective is obtained by a temperature‑controlled
soft‑max over the convolution entries; the temperature β is annealed
from 0.5 to a few thousand.  Learning‑rate η decays from ~0.5 to 1e‑3.
Occasional random block perturbations and a restart after stagnation
keep the search from getting stuck.

The loop runs until the time budget expires and the best sequence found
so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the simplex.
    """
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that minimise ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds (a small safety margin
    is subtracted) and returns the best vector seen.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose length and initialise a probability vector
    # ------------------------------------------------------------------
    # Prefer the length of the previously found construction if it exists,
    # otherwise use a larger length that still fits comfortably in the budget.
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
            base_n = max(prev.size, 6000)          # at least 6000 entries
        else:
            base_n = 8000
    else:
        base_n = 8000

    n = int(base_n)

    # Initialise with the analytical “arcsine” shape:
    # a_i ∝ 1 / sqrt((i+1) * (n-i))
    i = np.arange(n, dtype=np.float64)
    a0 = 1.0 / np.sqrt((i + 1.0) * (n - i))
    a0_sum = a0.sum()
    a = a0 / a0_sum

    # Work in log‑space (logits) so that the simplex constraint is satisfied
    # automatically via the soft‑max.
    u = np.log(a + 1e-30)   # add epsilon to avoid log(0)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    fft_len_wa   = _next_pow2(3 * n)          # for w*a   (w length = 2n‑1)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps   = 1e-8
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    step_counter = 0

    # ------------------------------------------------------------------
    # 4) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    # Evaluate initial point
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best_u   = u.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total   = deadline - start_time
        frac    = elapsed / total   # 0 → 1

        # --------------------------------------------------------------
        # 5.1 – schedules for learning‑rate η and soft‑max concentration β
        # --------------------------------------------------------------
        # Learning‑rate decays from ≈0.5 to 1e‑3
        eta  = 0.5 * (1.0 - frac) + 0.001 * frac
        # β grows from 0.5 to ~3000 (more aggressive than the previous code)
        beta = 0.5 * (1.0 - frac) + 3000.0 * frac

        # --------------------------------------------------------------
        # 5.2 – recover the probability vector from logits
        # --------------------------------------------------------------
        # Numerical stability: subtract max before exponentiation
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a_sum

        # --------------------------------------------------------------
        # 5.3 – self‑convolution and soft‑max weights
        # --------------------------------------------------------------
        c = _self_conv(a, fft_len_self)           # length 2n‑1
        max_c = c.max()
        # soft‑max over the convolution entries (shifted by max_c for stability)
        w_raw = np.exp(beta * (c - max_c))
        w = w_raw / w_raw.sum()

        # --------------------------------------------------------------
        # 5.4 – gradient of the smoothed objective (with respect to a)
        # --------------------------------------------------------------
        conv_wa = _conv_fft(w, a, fft_len_wa)     # length 3n‑2
        grad_a = 2.0 * conv_wa[n - 1:n - 1 + n]   # slice of length n

        # --------------------------------------------------------------
        # 5.5 – convert gradient ∂L/∂a to gradient w.r.t. logits u
        # --------------------------------------------------------------
        # ∂L/∂u_i = a_i * (grad_i - ⟨a, grad⟩)
        dot_ag = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot_ag)

        # --------------------------------------------------------------
        # 5.6 – Adam update on logits
        # --------------------------------------------------------------
        step_counter += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** step_counter)
        v_hat = v / (1.0 - beta2 ** step_counter)

        # parameter update (gradient descent on the negative direction)
        u -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5.7 – occasional random block jitter (≈0.7 % of iterations)
        # --------------------------------------------------------------
        if rng.random() < 0.007:
            a = _random_block_perturb(a, rng)
            # re‑project back to logits
            u = np.log(a + 1e-30)

        # --------------------------------------------------------------
        # 5.8 – keep the best exact solution (using the true max)
        # --------------------------------------------------------------
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_u = u.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.9 – soft restart after a period of stagnation
        # --------------------------------------------------------------
        if now - last_improve > 20.0:           # shorter stagnation horizon
            # jitter the best logits and continue
            u = best_u.copy()
            jitter = np.exp(rng.normal(0.0, 0.06, size=n))
            u += np.log(jitter)                 # multiplicative perturbation in log‑space
            # reset Adam moments to avoid carrying over stale information
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Construct the final sequence from the best logits,
    #    clamp to the limits imposed by ``evaluate_sequence``.
    # ------------------------------------------------------------------
    u_shift = best_u - np.max(best_u)
    best_a = np.exp(u_shift)
    best_a_sum = best_a.sum()
    if best_a_sum != 0.0:
        best_a /= best_a_sum

    best_a = np.clip(best_a, 0.0, 1000.0)

    # Return as a plain Python list of floats
    return [float(x) for x in best_a]

        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.
    The routine respects the given time budget (a small safety margin is
    subtracted) and returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose length and initialise a probability vector
    # ------------------------------------------------------------------
    # Prefer the length of a previously found construction if available,
    # otherwise use a generous length that still fits into the time budget.
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.ndim == 1 and prev.size > 0 and np.isfinite(prev).all():
            base_n = max(prev.size, 6000)   # at least 6 k entries
        else:
            base_n = 8000
    else:
        base_n = 8000

    n = int(base_n)

    # Analytic “arcsine’’ initialise: a_i ∝ 1/√((i+1)(n‑i)).
    i = np.arange(n, dtype=np.float64)
    a0 = 1.0 / np.sqrt((i + 1.0) * (n - i))
    a0 /= a0.sum()
    # Work in log‑space so that the simplex constraint is implicit.
    u = np.log(a0 + 1e-30)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    fft_len_wa   = _next_pow2(3 * n)          # for w*a (w length = 2n‑1)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    step_counter = 0

    # ------------------------------------------------------------------
    # 4) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    c = _self_conv(a0, fft_len_self)
    best_score = 2.0 * n * c.max()
    best_u = u.copy()
    last_improve = time.monotonic()
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total  # 0 → 1

        # --------------------------------------------------------------
        # 5.1 – schedules for learning‑rate η and soft‑max concentration β
        # --------------------------------------------------------------
        eta = 0.5 * (1.0 - frac) + 0.001 * frac          # 0.5 → 1e‑3
        beta = 0.5 * (1.0 - frac) + 3000.0 * frac       # 0.5 → 3000

        # --------------------------------------------------------------
        # 5.2 – recover the probability vector from logits
        # --------------------------------------------------------------
        u_shift = u - np.max(u)        # numerical stability
        a = np.exp(u_shift)
        a_sum = a.sum()
        if a_sum == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a_sum

        # --------------------------------------------------------------
        # 5.3 – self‑convolution and soft‑max weights
        # --------------------------------------------------------------
        c = _self_conv(a, fft_len_self)           # length 2n‑1
        max_c = c.max()
        w_raw = np.exp(beta * (c - max_c))        # shifted for stability
        w = w_raw / w_raw.sum()

        # --------------------------------------------------------------
        # 5.4 – gradient of the smoothed objective (w.r.t. a)
        # --------------------------------------------------------------
        conv_wa = _conv_fft(w, a, fft_len_wa)     # length 3n‑2
        grad_a = 2.0 * conv_wa[n - 1:n - 1 + n]   # slice of length n

        # --------------------------------------------------------------
        # 5.5 – transform to gradient w.r.t. logits u
        # --------------------------------------------------------------
        dot_ag = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot_ag)            # ∂L/∂u_i

        # --------------------------------------------------------------
        # 5.6 – Adam update on logits
        # --------------------------------------------------------------
        step_counter += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v = beta2 * v + (1.0 - beta2) * (grad_u * grad_u)

        m_hat = m / (1.0 - beta1 ** step_counter)
        v_hat = v / (1.0 - beta2 ** step_counter)

        u -= eta * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5.7 – occasional random block perturbation (≈0.7 % of iterations)
        # --------------------------------------------------------------
        if rng.random() < 0.007:
            a = _random_block_perturb(a, rng)
            u = np.log(a + 1e-30)   # re‑project to log‑space

        # --------------------------------------------------------------
        # 5.8 – keep the best exact solution (using the true max)
        # --------------------------------------------------------------
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_u = u.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.9 – soft restart after stagnation
        # --------------------------------------------------------------
        if now - last_improve > 20.0:          # shorter stagnation window
            # jitter the best logits and discard accumulated moments
            u = best_u.copy()
            jitter = np.exp(rng.normal(0.0, 0.06, size=n))
            u += np.log(jitter)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Construct the final sequence from the best logits and clamp it.
    # ------------------------------------------------------------------
    u_shift = best_u - np.max(best_u)
    best_a = np.exp(u_shift)
    s = best_a.sum()
    if s != 0.0:
        best_a /= s
    best_a = np.clip(best_a, 0.0, 1000.0)

    return [float(x) for x in best_a]
