    beta1 = 0.9
    beta2 = 0.999

    t_iter = 0

    best_a = a0.copy()
    # compute initial J
    conv0 = _self_conv(best_a, fft_len_self)
    best_score = 2.0 * n * conv0.max()
    cur_score = best_score

    start_time = time.monotonic()
    total_time = deadline - start_time

    # Phase 1: Adam gradient descent
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start_time
        frac = (now - start_time) / total_time
        # schedule
        # alpha goes from 5 to 200
        alpha = 5.0 + 195.0 * frac  # linearly
        # learning rate schedule: from 0.3 to 0.01
        lr = 0.3 * (1 - frac) + 0.01 * frac

        c_max = conv.max()

        shifted = conv - c_max
        w = np.exp(alpha * shifted)

"""
Propose a step‑function minimizing the evaluation function.

Algorithm
---------
1.  Choose a length `n`.  If a previous construction (`height_sequence_1`)
    exists we reuse its length, otherwise we use a power‑of‑two (4096) for
    FFT efficiency.
2.  Initialise the vector either from the previous best or from a uniform
    distribution.
3.  Optimisation proceeds in two phases within the time budget.

   Phase A – Adam‑based smooth‑max gradient descent
   ------------------------------------------------
   *   The objective `max(a*a)` is approximated by a soft‑max with a
       temperature `α` that grows from 5 to 200.
   *   The gradient of the smooth surrogate is `∇φ(a)=2·a * w_rev`,
       where `w` are the soft‑max weights of the self‑convolution.
   *   We parametrize the simplex point as `a = softmax(p)` and apply
       Adam updates on `p`.  The learning rate decays from 0.30 to 0.01.
   *   The best vector seen is stored.

   Phase B – Simulated‑annealing fine‑tuning
   -----------------------------------------
   *   Starting from the best vector of Phase A we perform random
       block‑perturbations (multiplicative log‑normal factor on a random
       contiguous block) followed by a projection onto the probability
       simplex.
   *   A temperature schedule exponentially decays from `T_start` to
       `T_end`; worse moves are accepted with probability `exp(-Δ/T)`.
   *   Whenever a new best score is obtained it is printed.

4.  The final best sequence (clipped to the bounds enforced by
    `evaluate_sequence`) is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (sum=1, non‑neg)."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    # block width between 1 and max(2, n//20)
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))   # σ≈0.15
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_self_conv(a: np.ndarray, w: np.ndarray,
                    fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the weighted convolution φ(a)=w·(a*a) w.r.t. a.
    Returns a vector of length len(a).
    """
    # w_rev = reverse(w) for the cross‑correlation a * w_rev
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    conv_full = np.fft.irfft(fa * fw, n=fft_len_grad)
    offset = w_rev.size - 1               # = (2n‑2)
    # extract the n relevant entries and multiply by 2
    grad = 2.0 * conv_full[offset: offset + a.size]
    return grad


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over the convolution vector with temperature α."""
    c_max = conv.max()
    shifted = conv - c_max                # avoid overflow
    w = np.exp(alpha * shifted)
    s = w.sum()
    if s == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= s
    return w


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective J = 2·n·max(conv) (sum(a)=1)."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence of heights that minimizes
    `evaluate_sequence`.  The routine respects a 1000 s time budget,
    uses an Adam‑based soft‑max gradient descent followed by simulated‑
    annealing fine‑tuning, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ---- time budget (keep 5 s safety margin) ----
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---- determine length and initialise ----
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if init.ndim == 1 and init.size > 0 and np.isfinite(init).all():
            n = init.size
            a0 = proj_simplex(np.clip(init, 0.0, None))
        else:
            init = None

    if "height_sequence_1" not in globals() or init is None:
        # use a power‑of‑two length for FFT convenience
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ---- Adam parameters ----
    p = np.log(a0 + 1e-30)                     # log‑space (softmax)
    m = np.zeros_like(p)                       # first moment
    v = np.zeros_like(p)                       # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t_iter = 0

    # pre‑compute FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)       # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)       # for a * w_rev

    # ---- bookkeeping for the best solution ----
    a = a0.copy()
    conv = _self_conv(a, fft_len_self)
    best_score = _objective_from_conv(conv, n)
    best_a = a.copy()
    cur_score = best_score

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # Phase A – Adam based smooth‑max gradient descent
    # ------------------------------------------------------------------
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start_time
        frac = (now - start_time) / total_time          # ∈[0,1]

        # --- schedules -------------------------------------------------
        # temperature for soft‑max
        alpha = 5.0 + 195.0 * frac          # 5 → 200
        # learning‑rate for Adam (decays)
        lr = 0.30 * (1.0 - frac) + 0.01 * frac

        # --- compute a from p (softmax) --------------------------------
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        a = exp_p / exp_p.sum()            # guarantees a∈simplex

        # --- self‑convolution and soft‑max weights --------------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
        w = _softmax_weights(conv, alpha)

        # --- gradient of the smooth surrogate -------------------------
        grad = _grad_self_conv(a, w, fft_len_grad)

        # --- convert to gradient w.r.t. p (mirror descent) ------------
        avg = np.dot(grad, a)                      # scalar ⟨grad,a⟩
        grad_p = a * (grad - avg)                  # ∇_p φ

        # --- Adam update -----------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad_p
        v = beta2 * v + (1.0 - beta2) * (grad_p * grad_p)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --- evaluate current objective ---------------------------------
        # (re‑compute a after the update)
        maxp = np.max(p)
        exp_p = np.exp(p - maxp)
        a = exp_p / exp_p.sum()
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
        cur_score = _objective_from_conv(conv, n)

        # --- keep best -------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{elapsed:6.2f}s] Adam new best = {best_score:.6f}")

        # --- occasional jitter to break possible plateaus -------------
        if rng.random() < 0.01:
            a_jit = _random_block_perturb(a, rng)
            # map back to p (log‑space) – this keeps the representation consistent
            maxp_jit = np.max(np.log(a_jit + 1e-30))
            p = np.log(a_jit + 1e-30) - maxp_jit

            # recompute objective for the jittered point
            fa = np.fft.rfft(a_jit, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
            cur_score = _objective_from_conv(conv, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_jit.copy()
                print(f"[{elapsed:6.2f}s] Adam jitter improved best = {best_score:.6f}")

        # stop Adam after about 60 % of the budget (reserve time for SA)
        if elapsed > 0.60 * total_time:
            break

    # ------------------------------------------------------------------
    # Phase B – Simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    # initialise SA state from the best point obtained in Phase A
    a = best_a.copy()
    # recompute its score (already known, but keep locally)
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
    cur_score = _objective_from_conv(conv, n)

    # temperature schedule (exponential decay)
    T_start = max(0.05 * best_score, 1e-4)   # a modest start temperature
    T_end = 1e-7

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time
        # exponential temperature decay
        T = T_start * (T_end / T_start) ** frac

        # propose a random block perturbation
        a_new = _random_block_perturb(a, rng)

        # evaluate new candidate
        fa = np.fft.rfft(a_new, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
        new_score = _objective_from_conv(conv, n)

        delta = new_score - cur_score
        if delta < 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            # accept move
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{elapsed:6.2f}s] SA new best = {best_score:.6f}")

        # occasional global jitter to keep diversity
        if rng.random() < 0.02:
            a = _random_block_perturb(best_a, rng)
            # recompute score after jitter (use it as the new current point)
            fa = np.fft.rfft(a, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
            cur_score = _objective_from_conv(conv, n)

    # ------------------------------------------------------------------
    # Return the best sequence found (bounded to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
