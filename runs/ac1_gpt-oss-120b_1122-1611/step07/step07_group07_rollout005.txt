def max_conv_fft(a):
    n = a.shape[0]
    L = _next_pow2(2*n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[:2*n - 1]

    return float(conv.max())

"""
Propose a step‑function that beats the current bound.

Algorithm
---------
1. **Exploration phase** (≈30 % of the time):
   Generate candidates from the family of discrete Beta‑shaped
   distributions a_i ∝ x_i^{α‑1}(1‑x_i)^{β‑1} (with x_i = (i+½)/n).
   The special cases α=2,β=1 (a linear “ramp”) and its mirror are
   included because the continuous ramp has a self‑convolution whose
   L∞‑norm is 2/3 of the uniform case, giving a theoretical score
   2·2/3≈1.33 < 1.50.  Random (α,β) pairs are also sampled.

2. **Refinement phase** (remaining time):
   Starting from the best candidate, perform a simple stochastic
   hill‑climbing that repeatedly moves a small amount of mass from a
   donor index (chosen proportionally to a_i) to a recipient index
   (chosen preferentially where a_i is small).  Moves are accepted
   only if they improve the objective 2 n·max(conv(a,a)),
   and occasional “jitter’’ perturbations are applied.
   If no improvement is observed for a few seconds a noisy restart
   is performed.

All operations keep the vector non‑negative and on the simplex, which
matches the expectations of ``evaluate_sequence``.  The routine stops
a few seconds before the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np

# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((int(x) - 1).bit_length())


def _max_self_convolution(a: np.ndarray) -> float:
    """Linear self‑convolution a*a using an FFT and return its maximum."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    # Numerical noise may create tiny negatives – clamp.
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (without its clamping)."""
    n = a.size
    max_b = _max_self_convolution(a)
    s = a.sum()
    if s < 0.01:                      # protection present in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def gen_beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) distribution on {0,…,n‑1}:
    a_i ∝ x_i^{α‑1}(1‑x_i)^{β‑1} with x_i = (i+½)/n.
    The result is normalised to sum = 1.
    """
    # map indices to (0,1)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    raw = np.clip(raw, 0.0, None)
    s = raw.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / s


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def local_optimize(a: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that moves a small amount of mass
    between indices.  Stops when ``deadline`` is reached.
    Returns the best sequence seen and its score.
    """
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)

    last_improve = time.monotonic()
    stagnation_limit = 5.0  # seconds without improvement before a restart

    while time.monotonic() < deadline:
        # donor ‑ weighted by current mass (more mass ⇒ more chance)
        if best_a.max() == 0.0:
            break
        donor = rng.choice(n, p=best_a)               # a is already normalised
        if best_a[donor] < 1e-12:
            continue

        # recipient ‑ prefers small entries
        inv_w = 1.0 - best_a
        inv_w[donor] = 0.0
        inv_sum = inv_w.sum()
        if inv_sum > 0.0:
            recipient = rng.choice(n, p=inv_w / inv_sum)
        else:
            recipient = rng.integers(n)

        # transfer a modest fraction of the donor’s mass
        delta = rng.random() * 0.5 * best_a[donor]
        a_new = best_a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta
        a_new /= a_new.sum()        # keep sum = 1

        new_score = compute_score(a_new)
        if new_score < best_score:
            best_a = a_new
            best_score = new_score
            last_improve = time.monotonic()
            print(f"    [{time.monotonic():.2f}s] local improvement → {best_score:.6f}")

        # occasional global jitter to escape plateaus
        if rng.random() < 0.01:
            a_jit = _random_block_perturb(best_a, rng)
            jit_score = compute_score(a_jit)
            if jit_score < best_score:
                best_a = a_jit
                best_score = jit_score
                last_improve = time.monotonic()
                print(f"    [{time.monotonic():.2f}s] jitter improvement → {best_score:.6f}")

        # restart if we have been stagnant too long
        if time.monotonic() - last_improve > stagnation_limit:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a_restart = best_a * noise
            a_restart = np.clip(a_restart, 0.0, None)
            a_restart /= a_restart.sum()
            restart_score = compute_score(a_restart)
            if restart_score < best_score:
                best_a = a_restart
                best_score = restart_score
                print(f"    [{time.monotonic():.2f}s] restart improvement → {best_score:.6f}")
            last_improve = time.monotonic()

    return best_a, best_score
