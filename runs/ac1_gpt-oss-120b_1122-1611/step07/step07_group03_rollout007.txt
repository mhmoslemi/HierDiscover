def build_Q_matrices(n):
    from scipy.sparse import csr_matrix, lil_matrix
    Qs = []
    for k in range(2*n-1):
        rows = []
        cols = []
        data = []
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # i runs over these inclusive
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(i)
            cols.append(j)
            data.append(1.0)
        Qs.append(csr_matrix((data, (rows, cols)), shape=(n, n)))
    return Qs
```

def proj_simplex(v):
    # https://arxiv.org/pdf/1309.1541.pdf
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def build_conv_matrix(f):
    n = f.size
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_min = max(0, k-(n-1))
        i_max = min(k, n-1)
        # for each i, j = k - i
        # f_i multiplies b_j
        for i in range(i_min, i_max+1):
            j = k - i
            rows.append(k)
            cols.append(j)
            data.append(f[i])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
```

"""
Propose a step‑function by mirror‑descent with a hard‑max (binary) weight.
The algorithm works on the simplex (sum = 1) and repeatedly:

1.   Computes the self‑convolution a*a.
2.   Builds a weight vector w that is 1 on the (few) lag(s) attaining the
     current maximum and 0 elsewhere (soft‑max with a very large temperature
     is used early on to improve stability).
3.   Obtains the sub‑gradient g_i = 2·∑_k w_k·a_{k‑i} via a single FFT‑based
     convolution.
4.   Performs an exponentiated‑gradient update
     a_i ← a_i·exp(‑η·g_i) and projects back onto the probability simplex.
5.   Accepts the move if the objective improves, otherwise with a tiny
     Metropolis probability that slowly cools down.
6.   Periodically perturbs a random block (log‑normal factor) to escape
     plateaus and restarts from the current best after long stagnation.

The schedule for the smooth‑max temperature α grows from 0.5 to 80,
while the step size η decays from 5 to 0.001.  The routine runs for the
whole allotted time (≤ budget_s seconds) and returns the best sequence
found.  All numbers are clipped to the limits enforced by
``evaluate_sequence`` (≥ 0, ≤ 1000).
"""

import time
import numpy as np
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT length ``fft_len``."""
    return _fft_convolve(a, a, fft_len)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 12))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.18))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initialise a on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)

    else:
        # No prior candidate – start from a Dirichlet draw.
        # A moderate length (≈ 1k) gives enough degrees of freedom while
        # keeping each iteration cheap.
        n = 1024

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths

    # ------------------------------------------------------------------

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------

        # --------------------------------------------------------------

        s = a_new.sum()
        if s == 0.0:

        else:
            a_new /= s

        # --------------------------------------------------------------
        c_new = _self_conv(a_new, fft_len_self)
        new_score = evaluate_sequence(a_new.tolist())

        # --------------------------------------------------------------
        temp_frac = (now - start_time) / total_time

        temperature = max(temperature, 1e-12)

        accept = False

        else:

                accept = True

        if accept:

            c = c_new
            cur_score = new_score
            accepted += 1

            # Periodic exact recomputation to wipe numerical drift
            if accepted % RECOMPUTE_EVERY == 0:
                c = _self_conv(a, fft_len_self)
                cur_score = evaluate_sequence(a.tolist())

        # --------------------------------------------------------------
        # 5.7  Random block jitter (occasional)
        # --------------------------------------------------------------
        if rng.random() < jitter_prob:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = evaluate_sequence(a.tolist())

        # --------------------------------------------------------------
        # 5.8  Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:

            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # 5.9  Soft restart after long stagnation
        # --------------------------------------------------------------

            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.12, size=n))

            last_improve = now
            accepted = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
