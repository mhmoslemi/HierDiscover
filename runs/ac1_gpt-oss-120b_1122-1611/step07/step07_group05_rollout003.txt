"""
Optimisation of the max‑self‑convolution (the quantity used in
``evaluate_sequence``).

Algorithm
---------
* Work on a probability vector (sum = 1) – scaling does not affect the
  score, so the simplex projection is safe.
* At each iteration compute the self‑convolution c = a*a.
* Replace the non‑smooth objective ``max(c)`` by a smooth soft‑max:
      w_k = exp(β·(c_k − max(c)))   (β > 0, w_max = 1, 0≤w_k≤1).
  For β→∞ the soft‑max approaches the true max.
* The gradient of the weighted sum  Σ w_k·c_k w.r.t. a is
      g = 2·(w ⋆ a_rev)    (cross‑correlation, obtained by one FFT).
* Perform a projected gradient step   a ← Π_simplex(a − η·g).
  η is slowly reduced, β is slowly increased (annealing).
* If a step does not improve the true score we sometimes accept it
  with a tiny Metropolis probability (temperature ≈ 1/β) to escape
  shallow plateaux.
* When no improvement occurs for ≈20 s the current vector is jittered
  (log‑normal multiplicative noise) and the optimisation continues.
* The whole search respects a 1000 s budget (5 s safety margin) and
  returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)
    return conv[: 2 * a.size - 1]


def _cross_corr(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for the gradient:
        g_i = 2·Σ_k w_k·a_{k‑i}
    Returns the array of length ``len(a)``.
    """
    a_rev = a[::-1]
    wf = np.fft.rfft(w, n=fft_len)
    ar = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(wf * ar, n=fft_len)
    start = a.size - 1
    return full[start : start + a.size]


def _jitter(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.08) -> np.ndarray:
    """Multiplicative log‑normal jitter followed by simplex projection."""
    out = a * np.exp(rng.normal(0.0, sigma, size=a.shape))
    out = _project_simplex(out)
    return out

    """
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Initial vector -------------------------------------------------
    # Use the known construction if present, otherwise start from a random Dirichlet.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        a = _project_simplex(a)
    else:
        # start from a modest‑size uniform vector (you can change the size)
        n = 5000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) FFT planning ---------------------------------------------------
    # Length required for self‑convolution: 2n‑1.
    # Length required for cross‑correlation (w • a_rev): (2n‑1)+n‑1 = 3n‑2.
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 3) Book‑keeping ----------------------------------------------------
    # best solution seen so far:
    # (the score is 2·n·max(c) because we keep Σa = 1)
    # ------------------------------------------------------------------
    c = _self_convolution(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    # ------------------------------------------------------------------
    # 4) Annealing schedules ---------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety
    total_time = deadline - start_time

    # η – step size (decreases), β – soft‑max sharpness (increases)
    eta_start, eta_end = 0.5, 1e-6
    beta_start, beta_end = 5.0, 500.0
    temp_start, temp_end = 0.1, 1e-6        # for occasional Metropolis moves

    # ------------------------------------------------------------------
    # 5) Main optimisation loop -----------------------------------------
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time                # 0 → 1 over the run

        # linear schedules
        eta = eta_start * (1.0 - frac) + eta_end * frac
        beta = beta_start * (1.0 - frac) + beta_end * frac
        temperature = temp_start * (1.0 - frac) + temp_end * frac
        temperature = max(temperature, 1e-12)

        # -------- Self‑convolution ------------------------------------
        c = _self_convolution(a, fft_len_self)
        M = c.max()                                # true max
        cur_score = 2.0 * n * M

        # -------- Soft‑max weights ------------------------------------
        # w_i = exp(β·(c_i − M))  ≤ 1 , w_max = 1
        w = np.exp(beta * (c - M))

        # -------- Gradient of Σ w_k·c_k -------------------------------
        grad = _cross_corr(w, a, fft_len_grad) * 2.0

        # -------- Gradient descent step -------------------------------
        a_candidate = a - eta * grad
        a_candidate = _project_simplex(a_candidate)

        # -------- Evaluate candidate ----------------------------------
        # (fast recompute of the max via FFT)
        c_cand = _self_convolution(a_candidate, fft_len_self)
        cand_score = 2.0 * n * c_cand.max()

        # -------- Acceptance -----------------------------------------
        improved = cand_score < best_score - 1e-12
        if improved or rng.random() < np.exp(-(cand_score - cur_score) / temperature):
            a = a_candidate
            cur_score = cand_score
            if improved:
                best = a_candidate.copy()
                best_score = cand_score
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # -------- Stagnation handling --------------------------------
        if now - last_improve > 20.0:
            # inject a mild log‑normal perturbation around the best point
            a = _jitter(best, rng, sigma=0.12)
            last_improve = now
            print(f"[{elapsed:6.2f}s] jitter (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits enforced by
    #    ``evaluate_sequence``)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
