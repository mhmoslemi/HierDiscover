    Uses a two‑phase approach:

    2. Refinement by Adam optimisation on a soft‑max surrogate of the max

    time budget.
    """

        # Adam update

        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = np.maximum(a, 0.0)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = _project_onto_simplex(a)

        iteration += 1

            start = rng.integers(0, n - width + 1)

            a_j = _project_onto_simplex(a_j)
            a = a_j

        # periodic exact evaluation
        if iteration % PRINT_EVERY == 0:
            exact = evaluate_sequence(a.tolist())
            if exact < best_score - 1e-12:
                best_score = exact
                best_seq = a.copy()
                last_improve_time = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best ({iteration} iter) = {best_score:.6f}")

        # restart if stagnation long

            a = best_seq.copy()

            print(f"[{elapsed:6.2f}s] stagnation restart")
    # final best
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

"""
Two‑phase optimizer for the self‑convolution bound.

Phase 1 – fast random exploration on a low‑dimensional parametrisation
          (mixture of Gaussian bumps, exponential decays, simple
          deterministic shapes).  Candidates are projected onto the
          probability simplex and the cheap FFT‑based estimate
          `2·n·max(conv)` is used to keep the best few.

Phase 2 – refinement of the best candidate with Adam on a smooth
          log‑sum‑exp surrogate of the max‑convolution (the same
          surrogate that was used in the reference implementation,
          but with a slower temperature/learning‑rate schedule,
          less aggressive jitter and a longer stagnation timeout).

Both phases respect the non‑negativity and simplex constraints.
The routine stays inside the given time budget and returns the
best sequence (converted to a plain list of floats) found so far.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _softmax_weights(c: np.ndarray, tau: float) -> np.ndarray:
    """
    Numerically stable soft‑max of the convolution values *c* with
    temperature *tau*.  Returns a vector that sums to 1.
    """
    shift = c.max()
    exp_shifted = np.exp((c - shift) / tau)
    total = exp_shifted.sum()
    if total == 0.0:
        return np.full_like(c, 1.0 / len(c))
    return exp_shifted / total

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    The algorithm combines a cheap random parametrisation search with a
    refined Adam optimisation on the log‑sum‑exp surrogate of the
    convolution max.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a working length
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
    else:
        n = 4096
        a0 = None

    # ------------------------------------------------------------------
    # 2) Cheap convolution–max helper (FFT based)
    # ------------------------------------------------------------------
    def _conv_max_fft(vec: np.ndarray) -> float:
        """Return max of linear self‑convolution of *vec* (FFT implementation)."""
        N = 1 << ((2 * n - 1).bit_length())
        v_fft = np.fft.rfft(vec, n=N)
        conv = np.fft.irfft(v_fft * v_fft, n=N)[: 2 * n - 1]
        return float(conv.max())

    # 3) Phase 1 – random parametric exploration
    # ------------------------------------------------------------------

    K_KEEP = 15
    best_candidates: list[tuple[float, np.ndarray]] = []

    idx = np.arange(n, dtype=np.float64)

    def _add_candidate(arr: np.ndarray, label: str = ""):
        """Project *arr* onto the simplex, evaluate cheap objective and keep it."""
        proj = _project_onto_simplex(arr)
        approx = 2.0 * n * _conv_max_fft(proj)
        if len(best_candidates) < K_KEEP:
            best_candidates.append((approx, proj))
        else:
            worst_i = max(range(len(best_candidates)),

            if approx < best_candidates[worst_i][0]:
                best_candidates[worst_i] = (approx, proj)

    # deterministic seeds

        # occasional exponential‑decay style candidate

    # ------------------------------------------------------------------
    # 4) Exact evaluation of the (few) promising candidates
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

        exact = evaluate_sequence(cand.tolist())
        if exact < best_score:
            best_score = exact
            best_seq = cand.copy()
            print(f"[Phase 1] exact = {best_score:.6f} (approx = {approx_obj:.6f})")
    # fallback – should never happen
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[Phase 1] fallback uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase 2 – soft‑max Adam refinement
    # ------------------------------------------------------------------
    a = best_seq.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # FFT length large enough for both convolution and correlation
    fft_len = _next_pow2(3 * n)
    corr_start = n - 1
    corr_end = corr_start + n

    # schedules (slow annealing)
    tau_start, tau_end = 0.5, 0.001
    lr_start, lr_end = 0.5, 0.001

    # control parameters
    JITTER_EVERY = 2000
    PRINT_EVERY = 1000
    STALL_TIMEOUT = 90.0      # seconds without improvement → inject noise

    iteration = 0

    start_refine = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_refine) / (deadline - start_refine)

        # anneal temperature and learning rate linearly
        tau = tau_start * (1.0 - frac) + tau_end * frac
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # ----- convolution & soft‑max weights -----
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        w = _softmax_weights(conv, tau)

    # ------------------------------------------------------------------

    return [float(x) for x in best_seq]
