    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(budget_s - 5.0, 0.0)

    # choose n adaptively: maybe 1200
    n = 1200

    # init a

    last_improve = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline:

        M = c.max()
        cur_score = evaluate_sequence(a.tolist()) # could compute simpler but fine

            break

        # maybe also compute new max after mixing

def lp_step(a, sum_target):
    n = len(a)
    # compute current conv and bound
    c = convolve_fft(a, a)
    M = c.max()
    # LP matrix
    A = build_conv_matrix(a)   # shape (2n-1, n)
    # Solve LP

                                bounds=[(0, None)]*n,
                                method='highs')
    if not res.success:
        return a, M, False
    b = res.x
    sum_b = b.sum()
    # scale to target sum
    g = b * (sum_target / sum_b)
    # compute three convolutions
    ff = c
    fg = convolve_fft(a, g)
    gg = convolve_fft(g, g)
    # line search for best t
    best_t = 0.0
    best_max = ff.max()
    # coarse scan
    for t in np.linspace(0.0, 1.0, 41): # step 0.025
        cur = (1-t)*(1-t)*ff + 2*t*(1-t)*fg + t*t*gg
        cur_max = cur.max()
        if cur_max < best_max:
            best_max = cur_max
            best_t = t
    # refine around best_t
    left = max(0.0, best_t - 0.05)

        cur1 = (1-t1)*(1-t1)*ff + 2*t1*(1-t1)*fg + t1*t1*gg

"""
Hybrid linear‑programming + entropy‑mirror descent optimizer.

Phase 1 (≈35 % of the time) – solve a small LP that maximises the total
mass while keeping the convolution with the current iterate below the
current worst entry.  The LP solution is mixed with the current vector
using a cheap line‑search that minimises the new maximal self‑convolution.
Phase 2 (≈65 % of the time) – fine‑tune the best vector with an
entropy‑mirror (exponentiated‑gradient) descent on a soft‑max surrogate of
the max‑convolution.  The routine respects the 1000 s budget and returns
the best sequence found.
"""

import time
import numpy as np
from scipy import linalg
from scipy import optimize

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def convolve_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution using an FFT of length that is a power of two."""
    n = len(x) + len(y) - 1
    L = _next_pow2(n)
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(fx * fy, n=L)[:n]
    return conv


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A representing the linear map
    b ↦ f * b (full convolution, zero padding).
    Shape of A: (2*len(f)-1, len(f)).
    """
    n = f.size
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    # scipy.linalg.toeplitz returns a dense (2n‑1)×n array.
    return linalg.toeplitz(col, row)


def _softmax_weights(c: np.ndarray, eps: float) -> np.ndarray:
    """Numerically stable soft‑max weights for the surrogate max."""
    max_c = c.max()
    w = np.exp((c - max_c) / eps)
    s = w.sum()
    if s == 0.0:                     # fallback – exact max
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
        s = 1.0
    return w / s


def _mirror_descent_step(a: np.ndarray, eps: float, eta: float) -> np.ndarray:
    """
    One entropy‑mirror (exponentiated‑gradient) step on the smooth‑max
    surrogate of max(a*a).  Returns the updated vector, already
    renormalised to the target sum.
    """
    n = a.size
    # current self‑convolution
    c = convolve_fft(a, a)                     # length 2n‑1
    # soft‑max weights
    w = _softmax_weights(c, eps)                # length 2n‑1, sum 1

    # gradient of  w·(a*a)  w.r.t. a  (see analysis)
    a_rev = a[::-1]
    grad_full = convolve_fft(w, a_rev)          # length 3n‑2
    grad = 2.0 * grad_full[n - 1:n - 1 + n]     # length n

    # exponentiated‑gradient update on the simplex
    factor = np.exp(-eta * grad)
    factor = np.clip(factor, 1e-40, 1e40)       # avoid numerical extremes
    a = a * factor
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _lp_update(a: np.ndarray, sum_target: float) -> tuple[np.ndarray, float, bool]:
    """
    Perform one LP‑based improvement step.
    Returns (new_a, new_max, success_flag).
    """
    n = a.size
    # current self‑convolution and bound M
    ff = convolve_fft(a, a)
    M = ff.max()

    # build dense convolution matrix A (size (2n‑1, n))
    A = build_conv_matrix(a)

    # linear program:
    #   maximise   sum(b)   ⇔ minimise   -sum(b)
    #   subject to  A @ b <= M,   b >= 0
    c_obj = -np.ones(n, dtype=np.float64)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # solve with HiGHS (fast for moderate size)
    res = optimize.linprog(
        c=c_obj,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        return a, M, False

    b = res.x
    sum_b = b.sum()
    if sum_b == 0.0:
        return a, M, False

    # scale the LP solution to the required simplex sum
    g = b * (sum_target / sum_b)

    # three convolutions needed for the quadratic line‑search
    fg = convolve_fft(a, g)
    gg = convolve_fft(g, g)

    # ------------------------------------------------------------------
    # line‑search for mixing coefficient t ∈ [0,1] that minimises
    # max( (1‑t)²·ff + 2t(1‑t)·fg + t²·gg )
    # ------------------------------------------------------------------
    def mixed_max(t_: float) -> float:
        return ((1 - t_) ** 2 * ff + 2 * t_ * (1 - t_) * fg + t_ ** 2 * gg).max()

    # coarse grid
    t_best = 0.0
    max_best = ff.max()
    for t in np.linspace(0.0, 1.0, 41):
        cur = mixed_max(t)
        if cur < max_best:
            max_best = cur
            t_best = t

    # ternary refinement around the coarse optimum
    left = max(0.0, t_best - 0.05)
    right = min(1.0, t_best + 0.05)
    for _ in range(12):
        t1 = left + (right - left) / 3.0
        t2 = right - (right - left) / 3.0
        m1 = mixed_max(t1)
        m2 = mixed_max(t2)
        if m1 < m2:
            right = t2
            if m1 < max_best:
                max_best, t_best = m1, t1
        else:
            left = t1
            if m2 < max_best:
                max_best, t_best = m2, t2

    # convex combination
    a_new = (1.0 - t_best) * a + t_best * g
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 0.0:
        a_new *= sum_target / s
    else:
        a_new = np.full_like(a_new, sum_target / n)

    return a_new, max_best, True


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return the best non‑negative height sequence found within the given time
    budget.  The algorithm mixes a linear‑programming based coarse optimiser
    (Phase 1) with a fast entropy‑mirror descent fine tuner (Phase 2).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Decide on the problem size (n) and initialise a
    # ------------------------------------------------------------------
    # Use a previous construction if it exists and is not too large.
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            # cap the length to keep the LP tractable
            n = min(cand.size, 1500)
            a = cand[:n].copy()
        else:
            n = 1200
            a = rng.dirichlet(np.ones(n))
    else:
        n = 1200
        a = rng.dirichlet(np.ones(n))

    a = np.clip(a, 0.0, None)
    sum_target = np.sqrt(2.0 * n)
    a = a / a.sum() * sum_target

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    # The score under the chosen normalisation equals the maximal entry
    # of the self‑convolution.
    cur_max = convolve_fft(a, a).max()
    best = a.copy()
    best_score = cur_max
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.35 * total_time   # about one third for LP

    # ------------------------------------------------------------------
    # 3) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – LP based coarse improvement
        # --------------------------------------------------------------
        if now < phase1_end:
            a, cur_max, ok = _lp_update(a, sum_target)
            if not ok:
                # fall back to mirror descent if LP fails
                phase1_end = now
                continue
        else:
            # ----------------------------------------------------------
            # Phase 2 – entropy‑mirror descent on a smooth surrogate
            # ----------------------------------------------------------
            # Linear schedules for step size and soft‑max temperature
            frac = (now - phase1_end) / (deadline - phase1_end)
            eta_start, eta_end = 2.0, 1e-4
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # current convolution and its dynamic range
            c = convolve_fft(a, a)
            range_c = c.max() - c.min()
            eps = max(range_c * max(0.2 * (1.0 - frac), 0.01), 1e-12)

            a = _mirror_descent_step(a, eps, eta)

            # recompute max after the mirror step
            cur_max = convolve_fft(a, a).max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_max < best_score - 1e-12:
            best_score = cur_max
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Simple restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # perturb the current best a little and continue
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a / a.sum() * sum_target
            cur_max = convolve_fft(a, a).max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to the limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
