"""
Propose a step‑function (non‑negative heights on the simplex) that
lowers the value of ``evaluate_sequence``.

Algorithm
---------

* Parameterise the sequence `a` by unconstrained log‑weights `x` and use a
  soft‑max to enforce `a_i ≥ 0` and `∑ a_i = 1`.  The objective for a
  normalised sequence is `score = 2·n·max(a*a)`, i.e. we only have to
  minimise the maximum entry of the self‑convolution.

* Instead of a smooth surrogate we use a *sub‑gradient* of the true
  maximum:
      – let `c = a*a` be the linear convolution,
      – pick all indices `k` where `c_k` is (almost) maximal,
      – the derivative of `c_k` w.r.t. `a_i` equals `a_i + a_{k-i}`
        (with out‑of‑range terms treated as `0`),
      – the sub‑gradient for the whole vector `a` is therefore
        `|I|·a + (a ⋆ w)`,
        where `I` is the set of maximal indices and `w` is a binary mask
        (`w_k = 1` iff `k∈I`).  The correlation `a ⋆ w` is computed by an
        FFT (O(n log n)).

* The chain‑rule for the soft‑max gives the gradient w.r.t. the
  log‑weights `x`.  An Adam optimiser with a slowly decaying learning
  rate updates `x`.

* To avoid stagnation we restart from the current best solution after
  30 s without improvement, adding moderate log‑normal noise.  Occasionally
  a random block of the current best sequence is jittered.

* The outer loop samples several problem sizes (`n` from {2000,4000,8000})
  and repeats the optimisation until the time budget (≈ 1000 s) is
  exhausted.  The best sequence seen globally is returned (clipped to the
  limits imposed by `evaluate_sequence`).

The method works directly on the true objective (no surrogate) and
exploits fast FFT‑based convolutions, allowing many thousands of
iterations within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be defined at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _grad_max_convolution(a: np.ndarray, c: np.ndarray, eps: float = 1e-12):
    """
    Sub‑gradient of max(c) w.r.t. the vector `a`.

    Parameters
    ----------
    a : np.ndarray
        Current probability vector (sum = 1).
    c : np.ndarray
        Linear self‑convolution of `a`.
    eps : float
        Tolerance for selecting the active set of maximal indices.

    Returns
    -------
    g_a : np.ndarray
        Gradient of ``max(c)`` w.r.t. ``a`` (same length as ``a``).
    """
    max_val = c.max()
    # active set – indices whose value is within `eps` of the maximum
    active = np.where(c >= max_val - max(eps, max_val * eps))[0]

    # binary mask w of length len(c)
    w = np.zeros_like(c)
    w[active] = 1.0
    sum_w = w.sum()

    # correlation a ⋆ w  (definition: (a⋆w)[i] = Σ_j a_j w_{i+j})
    N = _next_pow2(len(a) + len(w) - 1)
    fa = np.fft.rfft(a, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * np.conj(fw), n=N)[: len(a) + len(w) - 1]

    # extract the part corresponding to shifts i=0..n‑1
    offset = len(w) - 1
    cross = corr[offset: offset + a.size]

    # gradient = |I|·a + cross‑correlation term
    g_a = sum_w * a + cross
    return g_a

# Main optimisation routine
# ----------------------------------------------------------------------
