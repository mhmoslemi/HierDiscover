def _simplex_projection(v, s):
    # Project onto simplex {x: x>=0, sum x = s}
    n = v.shape[0]
    if v.sum() == 0:
        return np.full(n, s/n)
    # shift so that sum = s after projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Improved search for a low‑peak autocorrelation step function.

The algorithm is a variant of smooth‑max mirror descent with a much
more aggressive temperature schedule (α → 500) and a slowly decaying
step‑size (η → 0.001).  It is complemented by occasional random
block perturbations and a simple simulated‑annealing acceptance rule
to escape shallow plateaus.  The current best solution is always kept
and used as a restart point if no improvement is seen for a while.
All operations use FFT‑based convolutions for O(n log n) cost per
iteration, allowing many thousands of updates within the 1000 s budget.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of two real vectors ``a`` and ``b`` using an FFT
    of size ``length`` (must satisfy length ≥ len(a)+len(b)-1).
    Returns the full convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using an FFT of appropriate length."""
    return _fft_convolve(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex of a given target sum.
    """
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return out


def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex {x ≥ 0, Σx = s}.
    Implements the algorithm of [Wang & Carreira‑Perpinán 2013].
    """
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    n = v.shape[0]
    # Fast path for already non‑negative with correct sum
    if (v >= 0).all() and abs(v.sum() - s) <= 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform vector
        return np.full(n, s / n, dtype=v.dtype)
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # Numerical fix: renormalise exact sum if needed
    w_sum = w.sum()
    if w_sum > 0:
        w *= s / w_sum
    else:
        w[:] = s / n
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.

    The routine first uses the best known sequence if available,
    otherwise starts from a Dirichlet draw.  It then runs a smooth‑max
    mirror‑descent loop with aggressive temperature growth and occasional
    random block perturbations.  The best vector observed is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initialise a on the simplex with sum = √(2n)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
        else:
            # fallback – random Dirichlet
            n = 3840
            a = rng.dirichlet(np.ones(n) * 0.5).astype(np.float64)
    else:
        n = 3840
        a = rng.dirichlet(np.ones(n) * 0.5).astype(np.float64)

    # If we loaded a candidate, honour its length
    n = a.size
    target_sum = math.sqrt(2.0 * n)

    # Normalise onto the simplex of the required sum
    a = _simplex_projection(a, target_sum)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)                # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)                # for a * w

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    # Helper to compute the current peak of the self‑convolution
    def _current_peak(vec: np.ndarray) -> float:
        return _self_conv(vec, fft_len_self).max()

    cur_peak = _current_peak(a)
    best = a.copy()
    best_score = cur_peak                      # because sum = √(2n) ⇒ score = peak
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters (linear schedules)
    # ------------------------------------------------------------------
    # smooth‑max temperature (α)
    alpha_start = 0.5
    alpha_end = 500.0          # much larger than before

    # exponentiated‑gradient step size (η)
    eta_start = 1.5
    eta_end = 0.001

    # Simulated‑annealing “temperature” for occasional uphill moves
    temp_start = 0.08
    temp_end = 1e-6

    # Restart control
    RESTART_AFTER = 20.0       # seconds without improvement

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_total = elapsed / total_time

        # --------------------------------------------------------------
        # 5.1  Update schedules
        # --------------------------------------------------------------
        alpha = alpha_start + (alpha_end - alpha_start) * frac_total
        eta = eta_start + (eta_end - eta_start) * (1.0 - frac_total)  # decay
        # temperature for Metropolis acceptance
        temperature = temp_start + (temp_end - temp_start) * (1.0 - frac_total)
        temperature = max(temperature, 1e-12)

        # --------------------------------------------------------------
        # 5.2  Compute self‑convolution and soft‑max weights
        # --------------------------------------------------------------
        c = _self_conv(a, fft_len_self)
        c_max = c.max()
        # stable soft‑max
        w = np.exp(alpha * (c - c_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 5.3  Gradient of the smooth‑max: g = 2 * (a * w)
        # --------------------------------------------------------------
        conv_full = _fft_convolve(a, w, fft_len_grad)          # length 3n‑2
        g = 2.0 * conv_full[n - 1: 2 * n - 1]                 # slice length n

        # --------------------------------------------------------------
        # 5.4  Exponentiated‑gradient update on the simplex
        # --------------------------------------------------------------
        log_factor = -eta * g
        # Clip to safe range for exp
        log_factor = np.clip(log_factor, -700, 700)
        factor = np.exp(log_factor)

        a_new = a * factor
        a_new = np.clip(a_new, 0.0, None)
        a_new = _simplex_projection(a_new, target_sum)

        # --------------------------------------------------------------
        # 5.5  Evaluate new peak (exact self‑convolution)
        # --------------------------------------------------------------
        new_peak = _current_peak(a_new)

        # --------------------------------------------------------------
        # 5.6  Metropolis acceptance (allows occasional uphill steps)
        # --------------------------------------------------------------
        accept = False
        if new_peak < cur_peak:
            accept = True
        else:
            if rng.random() < math.exp(-(new_peak - cur_peak) / temperature):
                accept = True

        if accept:
            a = a_new
            cur_peak = new_peak

        # --------------------------------------------------------------
        # 5.7  Occasional global jitter – random block perturbation
        # --------------------------------------------------------------
        if rng.random() < 0.02:          # ≈2 % of iterations
            a = _random_block_perturb(a, rng)
            a = _simplex_projection(a, target_sum)
            cur_peak = _current_peak(a)

        # --------------------------------------------------------------
        # 5.8  Keep the best solution observed
        # --------------------------------------------------------------
        if cur_peak < best_score - 1e-12:
            best_score = cur_peak
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best peak = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.9  Soft restart after stagnation
        # --------------------------------------------------------------
        if now - last_improve > RESTART_AFTER:
            # restart from best with a gentle log‑normal jitter
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _simplex_projection(a, target_sum)
            cur_peak = _current_peak(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best peak {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best.tolist()]
