def _random_block_perturb(a, rng):
    n = a.size
    width = rng.integers(1, max(2, n//10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2)) # stronger?
    out = a.copy()
    out[start:start+width] *= factor
    out = np.clip(out, 0.0, None)
    tot = out.sum()
    if tot == 0.0:
        out[:] = 1.0 / n
    else:
        out /= tot
    return out

"""...
Algorithm: multi‑start subgradient descent on the exact ℓ∞‑norm of the self‑convolution.
Each trial starts either from a random Dirichlet draw or a jittered copy of the current best,
runs a short sub‑gradient descent where the sub‑gradient is obtained from the indicator
of the maximal convolution entry (i.e. true max, not a soft‑max), projects onto the probability simplex,
and keeps the best solution seen.  Restarts and occasional random perturbations avoid
getting stuck in local minima.  The code respects the 1000 s budget and returns a list of
non‑negative heights.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv

def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Return the objective 2·n·max(conv) / sum_a²."""
    return 2.0 * n * conv.max() / (sum_a * sum_a)

def _run_subgrad(a: np.ndarray,
                 steps: int,
                 eta_start: float,
                 eta_end: float,
                 fft_len_self: int,
                 grad_len: int,
                 n: int,
                 rng: np.random.Generator,
                 deadline: float) -> tuple[np.ndarray, float]:
    """One short sub‑gradient descent run."""
    for step in range(steps):
        if time.monotonic() >= deadline:
            break
        # self‑convolution
        conv = _self_conv(a, fft_len_self)
        max_val = conv.max()
        # mask of maximal entries (binary weight)
        mask = (conv == max_val).astype(np.float64)
        # reversed mask for cross‑correlation
        mask_rev = mask[::-1]
        # gradient = 2·a * mask_rev (cross‑correlation)
        fa = np.fft.rfft(a, n=grad_len)
        fw = np.fft.rfft(mask_rev, n=grad_len)
        grad_full = np.fft.irfft(fa * fw, n=grad_len)
        offset = mask_rev.size - 1  # = 2n‑2
        grad = 2.0 * grad_full[offset: offset + n]

        # step‑size schedule (linear decay)
        eta = eta_start * (1.0 - step / steps) + eta_end * (step / steps)

        # multiplicative update, then project onto the simplex
        a = a * np.exp(-eta * grad)
        a = proj_simplex(a)

    # final score
    conv = _self_conv(a, fft_len_self)
    final_score = _score_from_conv(conv, n, a.sum())
    return a, final_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

            a0 = np.clip(a0, 0.0, None)

        else:
            a0 = None
    else:

    if a0 is None:

    else:

    # ------------------------------------------------------------------
