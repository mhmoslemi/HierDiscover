    else:
        a = np.full_like(a, 1.0/a.size)

    n = a.size

    # time tracking
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)

    # Helper: FFT length for conv/corr
    fft_len = 1 << ((3*n-2).bit_length())  # next pow2

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9

    eps = 1e-8
    beta1_pow = beta1
    beta2_pow = beta2

    # initial best

    best_score = evaluate_sequence(best.tolist())
    best_time = time.monotonic()

    iteration = 0
    # main loop
    while time.monotonic() < deadline:

"""
Propose a step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.

The search consists of three stages (all respect the simplex constraint
∑a_i = 1, which keeps the objective scale‑invariant):

1. **Initialisation** – use the previously discovered construction
   ``height_sequence_1`` if available, otherwise a uniform vector.
2. **Smooth‑max Adam optimisation** – replace the hard max in the
   objective by a temperature‑controlled log‑sum‑exp, compute the exact
   gradient via FFTs and perform Adam updates while projecting onto the
   probability simplex.  The temperature is linearly annealed from 0.3
   down to 1e‑5 and the learning rate from 0.25 down to 5e‑4.
3. **Stochastic local refinement** – after the annealing phase we perform
   many tiny mass‑transfer moves (pick two indices, move a small
   fraction of the source mass to the target) and keep the move if it
   improves the exact score.  This fine‑tunes the solution and usually
   pushes the bound a few percent further.

The routine stops a few seconds before the full budget so that the
harness can safely collect the result.  Progress is printed to stdout.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    The algorithm uses a smooth‑max Adam optimiser followed by a
    stochastic local search.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Determine initial sequence and normalise to sum = 1
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        a = a0.copy()
    else:
        # default: power‑of‑two length simplifies FFT handling
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # --------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    best_time = time.monotonic()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – smooth‑max Adam optimisation
    # --------------------------------------------------------------
    # FFT length – enough for convolution + correlation
    fft_len = _next_pow2(3 * n)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # Annealing schedule (soft‑max temperature) and learning‑rate schedule
    TAU_START, TAU_END = 0.30, 5e-05          # 0.30 → 5 × 10⁻⁵
    LR_START, LR_END = 0.25, 5e-04          # 0.25 → 5 × 10⁻⁴

    # Number of iterations devoted to the Adam phase (≈60 % of budget)
    phase1_deadline = start_time + 0.60 * (deadline - start_time)

    iteration = 0
    while time.monotonic() < phase1_deadline:
        now = time.monotonic()
        frac = (now - start_time) / (phase1_deadline - start_time)

        # linearly anneal temperature and learning rate
        tau = TAU_START * (1.0 - frac) + TAU_END * frac
        lr = LR_START * (1.0 - frac) + LR_END * frac

        # ----- self‑convolution via FFT -----
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ----- soft‑max weights (smooth surrogate for max) -----
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)   # shift avoids overflow
        w = exp_shifted / exp_shifted.sum()          # sums to 1, length 2n‑1

        # ----- gradient of the smooth surrogate (correlation) -----
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projected gradient step (simplex) -----
        a = a - step
        a = _project_onto_simplex(a)

        iteration += 1

        # ----- exact evaluation from time to time -----
        if iteration % 250 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                best_time = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f} "
                      f"(iter={iteration})")
                # early stop if we already beat the target
                if best_score <= 1.5025:
                    print("[early stop] target reached during Adam phase")
                    break

    # --------------------------------------------------------------
    # 4) Phase 2 – stochastic local refinement (tiny mass transfers)
    # --------------------------------------------------------------
    # Use the best vector found so far as the starting point
    a = best.copy()
    iteration_local = 0
    while time.monotonic() < deadline:
        # propose a tiny mass transfer between two random indices
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # fraction of source mass to move (log‑uniform between 0.1 % and 5 %)
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[i] * frac_move

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta   # total sum stays 1, non‑negativity guaranteed

        sc = evaluate_sequence(a_candidate.tolist())
        iteration_local += 1

        if sc < best_score - 1e-12:
            best_score = sc
            best = a_candidate.copy()
            a = a_candidate               # keep the improvement
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} "
                  f"(move #{iteration_local})")
            if best_score <= 1.5025:
                print("[early stop] target reached during local phase")
                break

        # Occasionally inject a very small jitter to avoid getting stuck
        if iteration_local % 4000 == 0:
            width = rng.integers(1, max(2, n // 50))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            a_j = a.copy()
            a_j[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_j)

    # --------------------------------------------------------------
    # 5) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
