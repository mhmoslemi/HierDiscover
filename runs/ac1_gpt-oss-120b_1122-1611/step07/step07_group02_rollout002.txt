import time
import numpy as np
from scipy.special import logsumexp

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable softmax."""
    m = np.max(logits)
    shifted = logits - m
    exp = np.exp(shifted)
    sum_exp = np.sum(exp)
    if sum_exp == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / sum_exp

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step‑function minimizing ``evaluate_sequence``.

    The method starts from an interpolated version of the best known
    construction (``height_sequence_1`` if available) and runs a soft‑max
    parametrised Adam optimisation on a smooth approximation of the
    infinity‑norm of the self‑convolution.  Temperature for the soft‑max
    is increased slowly up to a very large value so the surrogate approaches
    the true max.  A few tiny stochastic jitter steps are added to escape
    plateaus.  The best vector encountered during the run is returned.
    """
    # Random generator
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Time budget (5 s reserve for the harness)
    # ----------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # Choose problem size and initialise a probability vector ``a``.
    # ----------------------------------------------------------------------
    target_n = 8000                     # a size that still allows fast FFTs
    base_seq = globals().get("height_sequence_1")
    if base_seq is not None:
        base = np.asarray(base_seq, dtype=np.float64)
        # Interpolate to target_n points (linear)
        old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
        new_x = np.linspace(0.0, 1.0, target_n, endpoint=False)
        a = np.interp(new_x, old_x, base)
    else:
        a = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    # Normalise to a probability distribution (sum = 1) and clip tiny negatives
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / a.size
    else:
        a /= s

    n = a.size

    # ----------------------------------------------------------------------
    # Logits parametrisation (softmax ensures non‑negativity and unit sum)
    # ----------------------------------------------------------------------
    eps = 1e-12
    x = np.log(a + eps)                # shape (n,)

    # Adam optimiser state
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # ----------------------------------------------------------------------
    # FFT planning – we work with powers of two for speed.
    # ----------------------------------------------------------------------
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n - 2)

    # ----------------------------------------------------------------------
    # Best‑so‑far bookkeeping
    # ----------------------------------------------------------------------
    a_fft = np.fft.rfft(a, n=fft_len_conv)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[:2 * n - 1]
    best_score = float(2.0 * n * conv.max())
    best_a = a.copy()
    last_improve = start

    # ----------------------------------------------------------------------
    lr_start = 0.6

    beta_end   = 4000.0
    jitter_every = 1000               # iterations

        # --------------------------------------------------------------

        # --------------------------------------------------------------

        conv_full = np.fft.irfft(a_fft * a_fft,
                                 n=fft_len_conv)[:2 * n - 1]

        # --------------------------------------------------------------

        # Record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Temperature for the smooth max approximation (increases with time)
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start
        total   = deadline - start
        ratio   = elapsed / total
        # Use a slightly accelerating schedule
        beta = beta_start + (beta_end - beta_start) * (ratio ** 1.3)

        # --------------------------------------------------------------
        # Soft‑max over the convolution values → weight vector w (sums to 1)
        # --------------------------------------------------------------
        scaled = beta * conv_full
        lse = logsumexp(scaled)
        w = np.exp(scaled - lse)               # shape (2n‑1,)

        # --------------------------------------------------------------
        # Gradient of the surrogate w.r.t. a (correlation w ⋆ a)
        # --------------------------------------------------------------
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr),
                                 n=fft_len_corr)          # length 3n‑2
        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # Chain rule → gradient w.r.t. logits x (soft‑max Jacobian)
        # --------------------------------------------------------------

        # --------------------------------------------------------------
        # Optional early‑exit if we have exhausted the time budget
        # --------------------------------------------------------------

    # ----------------------------------------------------------------------
    # Return the best sequence seen during the run.
    # ----------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]

"""
search for a non‑negative step function that minimises ``evaluate_sequence``.

The algorithm works in three phases:

1. **Initialisation** – a probability vector is built from the previously
   discovered construction (``height_sequence_1``) if it exists, otherwise a
   uniform distribution.  The vector is interpolated to a fixed length
   (8000) and normalised so that its sum is 1.  The vector is parametrised by
   logits ``x`` and a soft‑max maps the logits back to a probability vector
   (hence the constraints are automatically satisfied).

2. **Continuous optimisation** – we minimise a smooth surrogate of the
   infinity‑norm of the self‑convolution.  For a temperature parameter ``β``
   the weights ``w`` are the soft‑max of ``β·(a*a)``; when ``β`` grows the
   surrogate converges to the true ``max(a*a)``.  Using FFT we obtain the
   gradient of the surrogate w.r.t. the logits and perform an Adam update
   with a linearly decaying learning‑rate.  ``β`` is increased according to
   a slightly accelerating schedule; a tiny Gaussian jitter is injected every
   thousand iterations to avoid plateaus.

3. **Tracking** – the best vector (lowest exact objective) found so far is
   stored and finally returned (clipped to the limits enforced by the
   evaluator).

The whole search respects the 1000 s time budget (5 s are kept as a safety
margin).  No file or network I/O is performed.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities – top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (logits → probability simplex)."""
    m = np.max(logits)
    shifted = logits - m
    exp = np.exp(shifted)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of heights) with a small
    ``evaluate_sequence`` value.

    The optimisation runs for the allotted time budget, keeping the
    best sequence seen and finally returning it.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budget (keep a 5‑second safety margin for the runner)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Choose a problem size and initialise a probability vector ``a``.
    # ------------------------------------------------------------------
    target_n = 8000                     # large enough for a fine‑grained shape
    base_seq = globals().get("height_sequence_1")
    if base_seq is not None:

        # Linear interpolation to the target size
        old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
        new_x = np.linspace(0.0, 1.0, target_n, endpoint=False)
        a = np.interp(new_x, old_x, base)
    else:
        a = np.full(target_n, 1.0 / target_n, dtype=np.float64)

    # Normalise to a probability distribution (sum = 1) and enforce non‑negativity
    a = np.clip(a, 0.0, None)
    s = a.sum()
