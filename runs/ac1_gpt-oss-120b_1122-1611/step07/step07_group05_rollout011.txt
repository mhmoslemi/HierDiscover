"""
Optimisation of the step‑function objective 2·n·max(a*a)/(∑a)².

The algorithm consists of two stages:

1️⃣  *Smooth max descent* (≈80 % of the time budget)  
    – minimise the smooth approximation  
      fα(a)=α⁻¹·log∑ₖexp(α·cₖ) with c=a∗a.  
    – the gradient of fα is obtained by a single extra FFT:  
      grad fα = 2·conv( softmaxα(c), a[::-1] ).  
    – a projected (onto the probability simplex) gradient step
      with a linearly decaying step‑size η and a linearly increasing
      smoothing parameter α (from 2 to 200).

2️⃣  *Stochastic fine‑tuning* (≈20 % of the budget)  
    – a classic simulated‑annealing move that transfers a random
      amount δ from a “donor’’ index (chosen proportionally to its
      contribution aᵢ·a_{kmax‑i} to the current peak) to a “recipient’’
      that is preferably small.  
    – the convolution is updated incrementally (O(n)).  
    – Metropolis acceptance with a linearly cooling temperature.

Throughout the run the best sequence (lowest objective) is stored.
If no improvement is seen for 30 s a mild log‑normal jitter
re‑initialises the search around the current best.

The routine respects a 5 s safety margin and returns the best
sequence found before the deadline.
"""

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update
    the convolution ``c`` incrementally. Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # subtract donor contribution, add recipient contribution
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms (the diagonal of the convolution)
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a simplex projection.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out
