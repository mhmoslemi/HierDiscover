"""
Propose a step‑function that beats the previous bound (≈1.5132).

Algorithm
---------

1.  **Parameterisation** – work with logits `x` and use the soft‑max
    `a = softmax(x)` to guarantee non‑negativity and the simplex
    constraint `sum(a)=1`.  The evaluation function is scale‑invariant,
    therefore fixing the ℓ₁‑norm to one is sufficient.

2.  **Smooth surrogate for the max‑convolution** – the true objective
    is `2·n·max(c)` with `c = a⋆a`.  The non‑differentiable max is
    replaced by a temperature‑controlled soft‑max
    `S_ε(c) = ε·log(∑exp(c_i/ε))`.  Its gradient wrt `a` can be written
    analytically and is obtained by an FFT‑based convolution.

3.  **Adam optimiser on the logits** – we minimise the smooth
    surrogate using the Adam update rule (first‑ and second‑moment
    adaptivity).  The learning‑rate and the temperature decay
    linearly from a relatively large value (exploration) to a tiny
    value (exploitation).  The whole budget (≈1000 s) is spent on this
    descent; intermittent restarts (adding log‑normal noise to the
    logits) avoid stagnation.

4.  **Best‑ever bookkeeping** – after each iteration the true bound
    `2·n·max(a⋆a)` is computed.  The best vector seen so far is stored
    and finally returned.

The method keeps the whole optimisation on the simplex, uses only a
few FFTs per iteration (≈3), and therefore can perform many thousands
of updates within the time budget, typically improving the bound by a
few percent and reaching the target < 1.5025.
"""

import time
import numpy as np

# --------------------------------------------------------------
# FFT helper utilities
# --------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_full(w: np.ndarray, a_rev: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full convolution w * a_rev (linear, not circular) using an FFT.
    The result is trimmed to length ``len(w) + len(a_rev) - 1``.
    """
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    conv = np.fft.irfft(fw * fa, n=fft_len)
    # Desired length = len(w) + len(a_rev) - 1 = (2n-1) + n - 1 = 3n-2
    return conv[: len(w) + len(a_rev) - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(x, 1.0 / x.size)
    return e / s

# Main optimisation routine
# --------------------------------------------------------------

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.

    The routine runs a temperature‑controlled Adam optimisation on the
    soft‑max parametrisation of the step‑function.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 0) Time‑budget handling (few seconds safety margin)
    # -----------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------

                a = a / s
            else:

        else:
            a = None
    else:
        a = None

    if a is None:
        # default length – a little larger than the previous run, still cheap for FFTs
        n = 5000
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # -----------------------------------------------------------------

    eps_init = 1e-15
    logits = np.log(a + eps_init)  # shape (n,)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    beta1_pow = 1.0
    beta2_pow = 1.0

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a⋆a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w⋆a_rev

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    # Temperature for the smooth‑max (ε).  Starts relatively large to
    # provide a global view, ends near machine precision.
    eps_start = 0.15
    eps_end   = 1e-12

    # Adam learning‑rate schedule – start aggressive, finish gentle.
    lr_start = 0.8

    best = a.copy()
    best_score = cur_score

    total_time = deadline - start_time

    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / total_time   # 0 … 1

        # ---- schedules ----
        eps = max(eps_start * (1.0 - frac) + eps_end * frac, eps_end)
        lr  = lr_start * (1.0 - frac) + lr_end * frac

        # ---- obtain probability vector from logits ----
        a = _softmax(logits)
        sum_a = a.sum()          # should be 1.0 up to rounding

        # ---- self‑convolution (current objective) ----
        c = _self_conv(a, fft_len_self)
        max_c = c.max()
        cur_score = 2.0 * n * max_c / (sum_a * sum_a)

        # ---- keep global best ----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- compute gradient of smooth‑max ----
        # soft‑max (weights) over convolution entries

        w_sum = w.sum()
        if w_sum == 0.0:                       # numerical guard
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
            w_sum = 1.0
        w /= w_sum

        #   grad_a = 2 * (w ⋆ a_rev)[n‑1 : n‑1+n]

        conv_full = _conv_full(w, a_rev, fft_len_grad)   # length≈3n‑2
        grad_a = 2.0 * conv_full[n - 1:n - 1 + n]        # shape (n,)

        # ---- gradient wrt logits (soft‑max chain rule) ----
        avg = np.dot(a, grad_a)          # scalar = ⟨a,∂L/∂a⟩
        grad_logits = a * (grad_a - avg)

        # ---- Adam update on logits ----
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits * grad_logits)

        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            noise = rng.normal(0.0, 0.07, size=n)
            logits += noise
            # reset Adam moments to avoid stale direction
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = beta2_pow = 1.0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – jitter applied (best {best_score:.6f})")

    # -----------------------------------------------------------------
    # 7) Return the best vector found (clipped to the limits of the
    #    evaluation routine for safety).
    # -----------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
