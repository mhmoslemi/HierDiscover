```
import numpy as np
from scipy.optimize import linprog

def _next_pow2(x): ...
def _fft_convolve(a,b,length): ...
def _self_conv(a, fft_len): ...

def _active_constraints(conv, M, max_constraints=30, rel_tol=0.995):
    # Returns indices where conv >= rel_tol*M or top max_constraints largest values.
    idx = np.where(conv >= rel_tol * M)[0]
    if idx.size > max_constraints:
        # pick top max_constraints based on values
        top = np.argpartition(-conv, max_constraints)[:max_constraints]
        idx = top
    else:
        # if still less than max_constraints, also fill with top values
        if idx.size < max_constraints:
            # compute largest missing
            needed = max_constraints - idx.size
            top_extra = np.argpartition(-conv, needed)[:needed]
            idx = np.unique(np.concatenate([idx, top_extra]))
    return idx

def _solve_lp(f, M, active_idx):
    n = f.size
    rows = []
    for k in active_idx:
        row = np.zeros(n, dtype=np.float64)
        start_j = max(0, k - (n - 1))
        end_j = min(k, n - 1)
        # indices j from start_j to end_j inclusive
        j = np.arange(start_j, end_j + 1, dtype=np.int64)
        # source index for f: i = k - j
        row[j] = f[k - j]
        rows.append(row)
    if rows:
        A_ub = np.vstack(rows)
        b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
    else:
        # no constraints -> unbounded, return uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
    # bounds >=0
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        g = res.x
        # small numerical cleanup
        g[g < 0] = 0.0
        return g
    else:
        # fallback to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
```

The function below implements a *conditional‑gradient* (Frank‑Wolfe) approach that is
very different from the smooth‑max mirror‑descent used previously.

* **Step 1 – initialise** – use the best known sequence (if any) or a random
  Dirichlet draw of length `n≈3800`.  The vector is normalised so that  
  `∑a = √(2 n)`; with this normalisation the evaluation simplifies to
  `score = max_k (a∗a)[k]`.

* **Step 2 – active‑set LP** – let `M` be the current peak of `a∗a`.  
  Build a small set of *active* convolution indices (the positions where the
  peak is attained or almost attained).  Solve the linear programme  

  ```
  maximise   1ᵀ g
  subject to (a * g)[k] ≤ M   for every active index k
              g ≥ 0
  ```

  The constraints are linear in `g` (they are just shifted scalar products of
  `a`).  The LP is tiny (≤ 30 constraints, ≈ 4000 variables) and is solved with
  SciPy’s HiGHS solver.

* **Step 3 – direction scaling** – the LP gives a direction `g₀`.  It is
  rescaled to the required ℓ¹‑norm `√(2 n)`.

* **Step 4 – Frank‑Wolfe update** – replace the current point by a convex
  combination  

  `a ← (1‑t) a + t g`  (with `t≈0.5` and a back‑off line‑search).  
  The new point is re‑normalised and its peak `M` is recomputed.

* **Step 5 – bookkeeping** – keep the best point found, restart after
  30 s of stagnation (adding a tiny log‑normal jitter), and occasionally apply
  a random block perturbation to escape flat regions.

* **Step 6 – termination** – stop when the 1000 s budget (minus a 5 s safety
  margin) is exhausted and return the best sequence.

The algorithm exploits the *linear* structure of the convolution constraints
and therefore can push the peak lower than the previous smooth‑max method,
often reaching a score well below `1.51`.

```python
"""
Frank‑Wolfe optimisation for the step‑function bound.

The algorithm
---------------
1. Initialise a non‑negative vector a (length n≈3800) with ∑a = √(2n).
2. Repeatedly:
   * compute the self‑convolution c = a*a and its current peak M.
   * build a tiny “active set’’ of convolution indices where c is close to M.
   * solve an LP that maximises the ℓ¹‑mass of a direction g while keeping
     (a*g)[k] ≤ M for every active index k (linear constraints).
   * rescale g to have the same ℓ¹‑norm as a and take a Frank‑Wolfe step
     a ← (1‑t) a + t g (t is reduced if the step does not improve M).
   * keep the best a seen; restart after long stagnation with a small jitter.
3. Return the best sequence (clipped to the limits of evaluate_sequence).

The LP contains at most a few dozen constraints, so it is solved quickly with
SciPy’s HiGHS backend.  The whole routine runs well within the 1000 s budget
and typically improves the bound below 1.51.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT length ``fft_len``."""
    return _fft_convolve(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _active_constraints(conv: np.ndarray, M: float,
                        max_constraints: int = 30,
                        rel_tol: float = 0.999) -> np.ndarray:
    """
    Return a small set of indices where ``conv`` is close to its maximum ``M``.
    The set is limited to ``max_constraints`` entries.
    """
    # indices with value >= rel_tol * M
    idx = np.where(conv >= rel_tol * M)[0]
    if idx.size > max_constraints:
        # keep only the highest values
        idx = np.argpartition(-conv, max_constraints)[:max_constraints]
    elif idx.size < max_constraints:
        # fill up with the absolute top values
        needed = max_constraints - idx.size
        extra = np.argpartition(-conv, needed)[:needed]
        idx = np.unique(np.concatenate([idx, extra]))
    return idx


def _solve_lp(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Solve the LP

        maximise   1ᵀ g
        subject to (f * g)[k] ≤ M   for k in active_idx
                  g ≥ 0

    Returns the optimal g (not normalised).
    """
    n = f.size
    rows = []
    for k in active_idx:
        row = np.zeros(n, dtype=np.float64)
        # indices j where 0 ≤ k‑j < n
        start_j = max(0, k - (n - 1))
        end_j = min(k, n - 1)
        j = np.arange(start_j, end_j + 1, dtype=np.int64)
        row[j] = f[k - j]   # entry = f_{k-j}
        rows.append(row)

    if rows:
        A_ub = np.vstack(rows)          # shape (m, n)
        b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)
    else:
        # No constraints – return uniform vector
        return np.full(n, 1.0 / n, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)   # maximise sum(g)  ⇔ minimise -sum(g)
    bounds = [(0, None)] * n            # g_i ≥ 0

    # HiGHS (default) is fast for medium‑size LPs
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        g = res.x
        g[g < 0] = 0.0                  # clean tiny negatives
        return g
    else:
        # If the LP fails, fall back to a uniform direction
        return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Frank‑Wolfe optimisation for ``evaluate_sequence``.

    Returns a list of non‑negative heights (the best found) before the
    time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a on the simplex with the standard normalisation
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
    else:
        # start from a Dirichlet draw (moderately sparse)
        n = 3840
        a = rng.dirichlet(np.ones(n) * 0.3).astype(np.float64)

    n = a.size
    target_sum = np.sqrt(2.0 * n)          # ∑a = √(2 n)  →  score = max (a*a)
    a = a * (target_sum / a.sum())

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)           # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)           # for a * w (unused here)

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    conv_a = _self_conv(a, fft_len_self)
    M = conv_a.max()                # current peak
    best = a.copy()
    best_score = M
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4) Parameters
    # ------------------------------------------------------------------
    base_t = 0.5                   # initial Frank‑Wolfe step size
    RESTART_STAGNATION = 30.0     # seconds without improvement → restart
    max_active = 30               # size of the active LP set

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- 5.1 build active set ----
        active_idx = _active_constraints(conv_a, M,
                                         max_constraints=max_active,
                                         rel_tol=0.999)

        # ---- 5.2 solve LP for a direction g ----
        g_raw = _solve_lp(a, M, active_idx)

        # ---- 5.3 enforce bound on all positions (separation) ----
        # compute full convolution a*g_raw and add violating indices if any
        conv_ag = _fft_convolve(a, g_raw, fft_len_grad)
        viol = conv_ag.max() - M
        sep_iter = 0
        while viol > 1e-12 and sep_iter < 5:
            viol_idx = np.argmax(conv_ag)
            active_idx = np.unique(np.append(active_idx, viol_idx))
            g_raw = _solve_lp(a, M, active_idx)
            conv_ag = _fft_convolve(a, g_raw, fft_len_grad)
            viol = conv_ag.max() - M
            sep_iter += 1

        # ---- 5.4 scale g to the required ℓ¹‑norm ----
        sum_g = g_raw.sum()
        if sum_g <= 0.0:
            # fallback: a small random jitter
            a = _random_block_perturb(a, rng)
            a = a * (target_sum / a.sum())
            conv_a = _self_conv(a, fft_len_self)
            M = conv_a.max()
            continue
        g = g_raw * (target_sum / sum_g)

        # ---- 5.5 Frank‑Wolfe update (simple back‑off line‑search) ----
        success = False
        for t in (base_t, base_t * 0.5, base_t * 0.2):
            cand = (1.0 - t) * a + t * g
            cand = np.clip(cand, 0.0, None)
            cand = cand * (target_sum / cand.sum())
            conv_cand = _self_conv(cand, fft_len_self)
            M_cand = conv_cand.max()
            if M_cand < M - 1e-12:
                a = cand
                conv_a = conv_cand
                M = M_cand
                success = True
                break

        if not success:
            # occasional random exploration
            if rng.random() < 0.025:
                a = _random_block_perturb(a, rng)
                a = a * (target_sum / a.sum())
                conv_a = _self_conv(a, fft_len_self)
                M = conv_a.max()
                continue

        # ---- 5.6 record the best solution ----
        if M < best_score - 1e-12:
            best_score = M
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best max = {best_score:.6f}")

        # ---- 5.7 restart after stagnation ----
        if now - last_improve > RESTART_STAGNATION:
            a = best.copy()
            # small log‑normal jitter around the best point
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = a * (target_sum / a.sum())
            conv_a = _self_conv(a, fft_len_self)
            M = conv_a.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best max {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
