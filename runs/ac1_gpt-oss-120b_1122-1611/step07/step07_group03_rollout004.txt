def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    # set up random generator
    rng = np.random.default_rng(seed)
    # get initial a
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        total = a0.sum()
        if total > 0:
            a0 /= total
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        n = 3840 # could be different
        a0 = rng.dirichlet(np.ones(n) * 0.3)
    
    a = a0.copy()
    n = a.size
    # precompute fft length for convolution
    fft_len = _next_pow2(2 * n - 1)
    # compute conv
    c = _self_conv(a, fft_len)
    cur_score = 2.0 * n * c.max() / (a.sum() **2)
    best = a.copy()
    best_score = cur_score
    #time budget
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    start_time = time.monotonic()
    last_improve = start_time

    # hyperparameters
    eta_start = 1.5
    eta_end = 0.005
    temp_start = 0.1
    temp_end = 1e-6
    epsilon = 1e-30

    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        fraction = elapsed / (deadline - start_time)
        # schedule
        eta = eta_start * (1 - fraction) + eta_end * fraction
        temperature = temp_start * (1 - fraction) + temp_end * fraction

        # compute max index
        k_max = int(np.argmax(c))

        # compute gradient for peak
        g = np.zeros_like(a)
        start_i = max(0, k_max - (n - 1))
        end_i = min(k_max, n - 1)
        # indices start_i..end_i inclusive
        idx_range = np.arange(start_i, end_i + 1)
        g[idx_range] = 2.0 * a[k_max - idx_range]

        # exponentiated gradient via log space
        log_a = np.log(a + epsilon) - eta * g
        # normalize on simplex
        max_log = np.max(log_a)
        a_new = np.exp(log_a - max_log)
        s_new = a_new.sum()
        if s_new == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s_new

        # evaluate new via convolution
        c_new = _self_conv(a_new, fft_len)
        new_score = 2.0 * n * c_new.max() / (a_new.sum() **2)  # sum should be 1

        # acceptance
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # metropolis

"""
Mirror‑descent optimiser for the autocorrelation peak problem.

The algorithm works with a probability‑simplex representation of the
step‑function heights `a`.  It alternates two complementary update
schemes:

1. **Smooth‑max phase** (first 30 % of the time budget) – exactly the
   soft‑max mirror‑descent used in the previous state‑of‑the‑art.
   This quickly finds a reasonably flat autocorrelation.

2. **Peak‑focused phase** (remaining time) – a sub‑gradient update that
   targets the *currently* maximal convolution entry only.  The update
   is performed in log‑space (exponentiated‑gradient / KL‑projection) to
   keep the iterate on the simplex without numerical under‑flow.

The two phases are combined with a simple schedule for the learning
rate `η` and a simulated‑annealing acceptance rule.  Random block
perturbations and a soft restart after 30 s of stagnation help to escape
local minima.  Finally very small entries are trimmed before returning
the sequence.

The routine respects the 1000 s budget, prints progress, and returns a
list of non‑negative heights ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT length ``fft_len``."""
    return _fft_convolve(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _gradient_for_peak(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry `k`.
    For a self‑convolution a*a the gradient is
        g_i = 2 * a_{k-i}   if 0 ≤ k-i < n
               0            otherwise
    """
    n = a.size
    g = np.zeros_like(a)
    # valid i satisfy 0 ≤ k-i < n  ⇔  k-n+1 ≤ i ≤ k
    start_i = max(0, k - (n - 1))
    end_i = min(k, n - 1)
    if start_i <= end_i:
        idx = np.arange(start_i, end_i + 1, dtype=np.int64)
        g[idx] = 2.0 * a[k - idx]
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted runtime.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1) Initialise a on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            a = cand / s
        else:
            a = np.full_like(cand, 1.0 / cand.size)
    else:
        n_init = 3840
        a = rng.dirichlet(np.ones(n_init) * 0.3).astype(np.float64)

    n = a.size
    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)          # current self‑convolution
    cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters & schedules
    # ------------------------------------------------------------------
    # Smooth‑max phase (first 30 % of the total time)
    smooth_phase_frac = 0.30
    phase_switch = start_time + smooth_phase_frac * total_time

    # Smooth‑max schedules
    alpha_start, alpha_end = 0.5, 18.0
    eta_smooth_start, eta_smooth_end = 2.0, 0.001

    # Peak‑focused schedules
    eta_peak_start, eta_peak_end = 3.0, 0.001

    # Simulated‑annealing temperature schedule (global)
    temp_start, temp_end = 0.08, 1e-6

    # Counters
    accepted = 0
    RECOMPUTE_EVERY = 400

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_total = elapsed / total_time

        # ------------------------------------------------------------------
        # 5.1  Decide which phase we are in
        # ------------------------------------------------------------------
        if now < phase_switch:
            # --------------------------------------------------------------
            #   Smooth‑max mirror descent (same spirit as the previous code)
            # --------------------------------------------------------------
            alpha = alpha_start + (alpha_end - alpha_start) * frac_total
            eta = eta_smooth_start + (eta_smooth_end - eta_smooth_start) * (1.0 - frac_total)

            # soft‑max weights over the convolution values
            c_max = c.max()
            w = np.exp(alpha * (c - c_max))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient = 2 * a * w  (via convolution)
            fft_len_grad = _next_pow2(3 * n - 2)
            conv_full = _fft_convolve(a, w, fft_len_grad)          # length = 3n‑2
            g = 2.0 * conv_full[n - 1: 2 * n - 1]                  # slice length n
        else:
            # --------------------------------------------------------------
            #   Peak‑focused sub‑gradient descent
            # --------------------------------------------------------------
            eta = eta_peak_start + (eta_peak_end - eta_peak_start) * (1.0 - frac_total)

            # locate the current maximal convolution entry
            k_max = int(np.argmax(c))

            # sub‑gradient for that peak
            g = _gradient_for_peak(a, k_max)

        # --------------------------------------------------------------
        # 5.2  Exponentiated‑gradient (log‑space) update on the simplex
        # --------------------------------------------------------------
        eps = 1e-30                     # to avoid log(0)
        log_a = np.log(a + eps) - eta * g

        # renormalise onto the simplex in a numerically stable way
        max_log = np.max(log_a)
        a_new = np.exp(log_a - max_log)
        s_new = a_new.sum()
        if s_new == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s_new

        # --------------------------------------------------------------
        # 5.3  Evaluate the new objective (exact self‑convolution)
        # --------------------------------------------------------------
        c_new = _self_conv(a_new, fft_len_self)
        new_score = 2.0 * n * c_new.max() / (a_new.sum() ** 2)

        # --------------------------------------------------------------
        # 5.4  Metropolis acceptance (helps to escape plateaus)
        # --------------------------------------------------------------
        temperature = temp_start + (temp_end - temp_start) * (1.0 - frac_total)
        temperature = max(temperature, 1e-12)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                accept = True

        if accept:
            a = a_new
            c = c_new
            cur_score = new_score
            accepted += 1

            # periodic clean‑up of numerical drift
            if accepted % RECOMPUTE_EVERY == 0:
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # --------------------------------------------------------------
        # 5.5  Occasionally apply a random block perturbation (exploration)
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # --------------------------------------------------------------
        # 5.6  Keep the best solution observed so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.7  Soft restart after stagnation (≈30 s without improvement)
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            # gentle log‑normal jitter around the best
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
            last_improve = now
            accepted = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Post‑processing – trim negligible entries
    # ------------------------------------------------------------------
    # Very small components do not affect the score but inflate the
    # length (and therefore the 2·n factor).  Removing them is safe.
    threshold = 1e-8
    mask = best > threshold
    if mask.sum() >= 2:                     # keep at least two entries
        best = best[mask]
        best /= best.sum()                  # re‑normalise

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
