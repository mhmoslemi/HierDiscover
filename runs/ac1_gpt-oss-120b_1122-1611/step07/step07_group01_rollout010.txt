def project_simplex(v):
    if v.sum() == 1 and np.all(v>=0):
        return v
    sorted_v = np.sort(v)[::-1]
    cs = np.cumsum(sorted_v) - 1
    rho = np.where(sorted_v - cs/(np.arange(1,len(v)+1)) > 0)[0][-1]
    theta = cs[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_simplex(v):
    # projection onto simplex {x: x>=0, sum x = 1}
    # v is ndarray
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho = last true index
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
A second‑generation optimiser for the “max self‑convolution” problem.

Main ideas
----------
* Work on a probability vector a (∑a = 1, a_i ≥ 0).  The objective to be
  minimised is 2·n·max(a∗a) (the normalisation factor in evaluate_sequence
  cancels because it is scale‑invariant).
* Replace the non‑smooth maximum by a soft‑max with a temperature β.
  The gradient of   L_β(a)= (1/β)·log Σ_k exp(β·c_k) ,  c=a∗a,
  w.r.t. a is   g = 2·(w∗a)   (w is the soft‑max weight vector).
* Use Adam (adaptive moment estimation) on this gradient and project the
  result back onto the simplex after each step.  This is an additive‑
  gradient version of the exponentiated‑gradient method used previously.
* β grows from a small value (smooth objective) to a large value (tight
  approximation of the true max).  The learning‑rate η decays in parallel.
* Small random block perturbations and “soft restarts” (jitter the best
  solution after a period of stagnation) keep the search from getting
  trapped.
* The implementation works with FFT‑based convolutions, so a single
  iteration costs only two FFTs (≈ O(n log n)).  With a length of about
  6000–8000 elements many thousands of iterations fit comfortably
  into the 1000 s budget.

The function returns the best sequence found before the time budget
expires.  All numbers are clipped to the limits enforced by
evaluate_sequence (≥0, ≤1000).  The algorithm is deliberately
deterministic apart from the RNG seed, making it easy to reproduce.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection requires a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise precisely
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def _hard_grad(c: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Gradient of the *actual* maximum (cimax) with respect to a.
    For k = argmax c_k  we have  ∂c_k/∂a_i = 2·a_{k-i} (if index in range).
    """
    n = a.size
    k = int(np.argmax(c))
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[idx[mask]]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget expires.

    The routine uses Adam on a soft‑max approximation of the max
    self‑convolution, together with simplex projection, occasional
    random jitter and a stagnation‑restart mechanism.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a problem size and initialise a probability vector
    # ------------------------------------------------------------------
    # If a previous construction exists, reuse its length; otherwise pick
    # a comfortable size (≈ 6500) which gives a good trade‑off between
    # expressiveness and speed.
    if "height_sequence_1" in globals():
        src = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if src.ndim == 1 and src.size > 0 and np.isfinite(src).all():
            a = np.clip(src, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            a = None
    else:
        a = None

    if a is None:
        n = 6500
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_wa   = _next_pow2(3 * n)              # for w*a (len(w)=2n‑1)

    # ------------------------------------------------------------------
    # 3) Adam hyper‑parameters
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps   = 1e-8
    m = np.zeros(n, dtype=np.float64)   # first moment
    v = np.zeros(n, dtype=np.float64)   # second moment
    beta1_pow = 1.0
    beta2_pow = 1.0

    # ------------------------------------------------------------------
    # 4) Scheduling parameters
    # ------------------------------------------------------------------
    lr_start = 0.3          # initial learning‑rate (additive)
    lr_end   = 0.001        # final learning‑rate
    beta_start = 0.5        # start of soft‑max concentration
    beta_end   = 3500.0     # final concentration (very sharp)

    # ------------------------------------------------------------------
    # 5) Book‑keeping for the best solution
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    it = 0

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)   # 0 → 1

        # --------------------------------------------------------------
        # 6.1 – schedules for learning‑rate and soft‑max concentration
        # --------------------------------------------------------------
        lr   = lr_start * (1.0 - frac) + lr_end * frac
        # use a slightly convex schedule for β (slow start, fast finish)
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.6)

        # --------------------------------------------------------------
        # 6.2 – current self‑convolution and soft‑max weights
        # --------------------------------------------------------------
        c = _self_conv(a, fft_len_self)                 # length 2n‑1
        max_c = c.max()
        # soft‑max (shift by max_c to avoid overflow)
        w_raw = np.exp(beta * (c - max_c))
        w = w_raw / w_raw.sum()                         # still length 2n‑1

        # --------------------------------------------------------------
        # 6.3 – gradient of the smoothed objective (soft‑max version)
        # --------------------------------------------------------------
        conv_wa = _conv_fft(w, a, fft_len_wa)           # length 3n‑2
        grad_soft = 2.0 * conv_wa[n - 1:n - 1 + n]      # slice of length n

        # --------------------------------------------------------------
        # 6.4 – optionally blend in the exact‑max gradient
        # --------------------------------------------------------------
        # Gradients are blended more heavily towards the end of the run.
        hard_weight = frac
        if hard_weight > 0.0:
            grad_hard = _hard_grad(c, a)
            grad = (1.0 - hard_weight) * grad_soft + hard_weight * grad_hard
        else:
            grad = grad_soft

        # --------------------------------------------------------------
        # 6.5 – Adam update (additive) followed by projection onto simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        a = _project_simplex(a)

        # --------------------------------------------------------------
        # 6.6 – occasional stochastic jitter (helps escape plateaux)
        # --------------------------------------------------------------
        if rng.random() < 0.004:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 6.7 – evaluate the new candidate
        # --------------------------------------------------------------
        c_new = _self_conv(a, fft_len_self)
        new_score = 2.0 * n * c_new.max() / (a.sum() ** 2)

        # --------------------------------------------------------------
        # 6.8 – maintain the best solution seen so far
        # --------------------------------------------------------------
        if new_score < best_score - 1e-12:
            best_score = new_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6.9 – stagnation handling: soft restart after 25 s of no gain
        # --------------------------------------------------------------
        if now - last_improve > 25.0:
            # Jitter the current best and discard momentum
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * jitter
            a = _project_simplex(a)
            # reset Adam moments
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow = 1.0
            beta2_pow = 1.0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – re‑jitter best (score {best_score:.6f})")

        # --------------------------------------------------------------
        # 6.10 – prepare for next iteration
        # --------------------------------------------------------------
        c = c_new

    # ------------------------------------------------------------------
    # 7) Return the best sequence (respect the bounds enforced by the
    #    evaluation function).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
