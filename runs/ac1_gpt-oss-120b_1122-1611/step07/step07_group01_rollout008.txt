"""
Two‑phase optimization for the step‑function bound.

Phase 1 – “soft” mirror descent.
    The self‑convolution c = a*a is turned into a probability
    distribution by a temperature‑controlled soft‑max.
    The gradient of the smoothed objective is 2·(w*a) (slice of size n)
    and an exponentiated‑gradient (mirror‑descent on the simplex) update
    keeps the vector a non‑negative and normalised.

Phase 2 – “hard” sub‑gradient descent.
    After 70 % of the time budget we switch to the true objective:
    k* = argmax(a*a) and the gradient of b_{k*}=a*a at a is
        g_i = 2·a_{k*‑i}    (zero outside the support).
    An exponentiated‑gradient step with a decreasing step size
    reduces the peak of the convolution directly.

Exploration & restarts.
    * Occasionally a random contiguous block is multiplicatively perturbed.
    * If no improvement is seen for a while the current best vector is
      jittered (log‑normal factor) and optimisation resumes.
    * All convolutions are performed with real FFTs for speed.

The routine returns the best sequence found before the time budget expires.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(arr, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # fall back to a uniform distribution of length ≈ 3800
        n0 = 3800
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_wa   = _next_pow2(3 * n)              # for w*a (w length = 2n‑1)

    # ------------------------------------------------------------------
    # 3) Book‑keeping of the best solution seen so far
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    best_score = 2.0 * n * c.max()
    best = a.copy()
    last_improve = time.monotonic()
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 4) Main optimisation loop (two phases)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        prog = (now - start_time) / total          # 0 → 1

        # --------------------------------------------------------------
        # Phase selection
        # --------------------------------------------------------------
        if prog < 0.70:           # -----------  Phase 1  -----------------
            # learning‑rate η decays, temperature β rises
            eta  = 2.0 * (1.0 - prog / 0.70) + 0.001 * (prog / 0.70)
            beta = 0.5 * (1.0 - prog / 0.70) + 1500.0 * (prog / 0.70)

            # self‑convolution and soft‑max weights
            c = _self_conv(a, fft_len_self)
            max_c = c.max()
            w_raw = np.exp(beta * (c - max_c))
            w = w_raw / w_raw.sum()

            # gradient of the smoothed objective (2·(w*a) slice)
            conv_wa = _conv_fft(w, a, fft_len_wa)
            grad = 2.0 * conv_wa[n - 1:n - 1 + n]

            # exponentiated‑gradient update
            exponent = -eta * grad
            exponent = np.clip(exponent, -50.0, 50.0)
            factor = np.exp(exponent)
            a = a * factor
            a_sum = a.sum()
            if a_sum == 0.0:
                a[:] = 1.0 / n
            else:
                a /= a_sum

        else:                     # -----------  Phase 2  -----------------
            # compute current self‑convolution
            c = _self_conv(a, fft_len_self)
            max_c = c.max()
            k_star = int(np.argmax(c))

            # gradient of the exact max (sub‑gradient 2·a_{k*‑i})
            idx = np.arange(n)
            j = k_star - idx
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[j[mask]]

            # step size schedule (from ~0.5 down to 1e‑3)
            eta = 0.5 * (1.0 - (prog - 0.70) / 0.30) + 0.001 * ((prog - 0.70) / 0.30)

            exponent = -eta * grad
            exponent = np.clip(exponent, -50.0, 50.0)
            factor = np.exp(exponent)
            a = a * factor
            a_sum = a.sum()
            if a_sum == 0.0:
                a[:] = 1.0 / n
            else:
                a /= a_sum

        # --------------------------------------------------------------
        # Occasional global jitter (≈0.5 % of iterations)
        # --------------------------------------------------------------
        if rng.random() < 0.005:
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # Re‑evaluate exact score after the update
        # --------------------------------------------------------------
        c_new = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * c_new.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Restart after stagnation (no improvement for ~25 s)
        # --------------------------------------------------------------
        if now - last_improve > 25.0:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits imposed by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
