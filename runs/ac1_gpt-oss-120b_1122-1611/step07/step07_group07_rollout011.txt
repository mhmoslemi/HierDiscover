def project_to_simplex(v):

    # v is array; we want to find w >=0, sum(w)=1 minimizing ||w-v||^2 (Euclidean). Standard algorithm.

    u = np.sort(v)[::-1]

    cssv = np.cumsum(u) - 1

    ind = np.arange(len(v)) + 1

    cond = u - cssv / ind > 0

    rho = ind[cond][-1]

    theta = cssv[cond][-1] / rho

    w = np.maximum(v - theta, 0)

    return w

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – work with a power‑of‑two length `n` (default 8192)
   and start from the uniform distribution on the simplex.

2. **Phase 1 – smoothed max mirror‑descent** (≈70 % of the budget)  
   * Compute the full self‑convolution `c = a*a` with FFT.  
   * Form a soft‑max weight vector `w` that approximates the
     ``max(c)`` (temperature `eps` decays to ~1e‑12).  
   * The gradient of the smooth max w.r.t. `a` is
     ``grad = 2·conv(w, a[::-1])`` – evaluated with another FFT.  
   * Apply an exponentiated‑gradient (entropy mirror) update
     ``a ← a·exp(-η·grad)`` with a linearly decaying step‑size `η`,
     clip negatives, and renormalise to the simplex.

3. **Phase 2 – stochastic refinement** (≈30 % of the budget)  
   * Keep the current convolution `c`.  
   * Repeatedly pick a *donor* index proportional to its contribution
     to the current maximal entry `c[k_max]` and a *recipient* index
     favouring small coefficients.  
   * Transfer a random `Δ≤0.5·a[donor]` mass, update `c` incrementally
     (`_apply_move`).  
   * Accept the move if it improves the score, otherwise with a
     Metropolis probability using a temperature that cools to zero.  
   * Periodically recompute the full convolution to avoid drift and
     occasionally apply a block‑wise log‑normal jitter.

4. **Restarts** – if no improvement for 20 s, perturb the best vector
   with mild log‑normal noise and continue.

Both phases keep the vector non‑negative and on the simplex, which is
exactly what ``evaluate_sequence`` expects.  The routine runs until the
time budget (minus a 5 s safety margin) expires and returns the best
sequence seen.

The method substantially improves the uniform bound (score = 2.0) and
is able to push the score below the target 1.5025 within the given

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new a and c."""
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # incremental convolution update (see original programme)
    c_new = c.copy()
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a problem size.
    #    Use a power of two (FFT‑friendly). 8192 works well for the budget.
    # ------------------------------------------------------------------
    n = 8192
    # If a previous construction is available, copy its length if reasonable.
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            n = int(_next_pow2(cand.size))
    a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start, sum=1

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)                # current self‑convolution
    cur_score = 2.0 * n * c.max()                  # current objective value
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # allocate 70 % of the time to the mirror‑descent phase
    phase1_end = start_time + 0.70 * total_time

    # Hyper‑parameters
    eta_start = 2.0
    eta_end = 1e-3
    eps_min = 1e-12

    # Phase‑2 (stochastic) schedule
    temp_start = 0.07
    temp_end = 1e-6
    RECOMPUTE_EVERY = 500
    move_counter = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – smoothed max mirror descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedule for step size and temperature
            prog = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - prog) + eta_end * prog

            # compute full self‑convolution
            c = _self_conv(a, fft_len_self)

            # soft‑max temperature
            range_c = c.max() - c.min()
            eps = max(range_c * max(0.2 * (1.0 - prog), 0.01), eps_min)

            # stable soft‑max weights
            max_c = c.max()
            w = np.exp((c - max_c) / eps)
            ws = w.sum()
            if ws == 0.0:
                w = np.zeros_like(c)
                w[int(c.argmax())] = 1.0
                ws = 1.0
            w /= ws

            # gradient of the smooth max (entropy mirror)
            a_rev = a[::-1]
            fw = np.fft.rfft(w, n=fft_len_grad)
            fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: 3 * n - 1]
            grad = 2.0 * conv_full[n - 1:n - 1 + n]   # length n

            # exponentiated‑gradient update
            factor = np.exp(-eta * grad)
            factor = np.clip(factor, 1e-40, 1e40)
            a = a * factor
            a = np.clip(a, 0.0, None)

            # project onto the simplex (simple renormalisation works well)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a[:] = 1.0 / n

            # recompute score
            cur_score = 2.0 * n * c.max()

        else:
            # --------------------------------------------------------------
            # Phase 2 – fast stochastic refinement (simulated annealing)
            # --------------------------------------------------------------
            # temperature schedule (linear cooling)
            prog = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - prog) + temp_end * prog
            temperature = max(temperature, 1e-14)

            # locate current maximum of the convolution
            k_max = int(c.argmax())

            # contributions of each index to that peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # donor selection (biased by contribution)
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                continue

            # recipient prefers small entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # random transfer amount (up to half of donor mass)
            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            # apply move incrementally
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # full recomputation occasionally to avoid drift
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter (~2 % of iterations)
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Book‑keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after stagnation (20 s without improvement)
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a = _project_to_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clip to safety limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
