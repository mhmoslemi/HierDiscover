"""
Propose a step‐function (non‑negative sequence) that minimises
evaluate_sequence().

Algorithm
---------
* Initialise a probability vector a (sum = 1) – a mix of simple shapes and
  Dirichlet draws.
* Optimise a by minimising the smooth surrogate of the max‑convolution:
      φ(a) = Σ w_k·c_k ,   c = a * a  (linear self‑convolution)
  where w is a soft‑max over c with a temperature α that grows
  exponentially (5 → 2000).  The gradient of φ is
      ∇φ = 2·(a * rev(w)) .
* Updates are performed on the log‑domain (multiplicative update) with
  momentum (Adam‑style RMS scaling) and a projection onto the probability
  simplex after each step.
* Every few iterations a random block perturbation is applied and after
  long stagnation the current best solution is jittered to escape plateaus.
* The loop runs until the time budget (≈ 995 s) is exhausted, tracking
  the best sequence seen.  The final sequence is clipped to the bounds
  enforced by evaluate_sequence() and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (non‑negative,
    sum = 1).  Implements the algorithm of Wang & Carreira‑Perpinán (2013).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by L1‑normalisation."""
    n = a.size
    # block width between 1 and max(2, n/20)
    width = rng.integers(1, max(2, n // 20) + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Diverse portfolio of initial probability vectors (sum = 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # symmetric double‑spike (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # few Dirichlet draws (moderately concentrated)
    for _ in range(8):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative sequence that minimises ``evaluate_sequence``.
    Implements a momentum‑accelerated soft‑max gradient descent on the
    self‑convolution surrogate, with occasional random block perturbations
    and stagnation restarts.
    Returns the best sequence found (list of Python floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Budget handling – keep a 5 s safety margin for the harness
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initialise a feasible a
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0 and np.isfinite(a0).all():
            a0 = np.clip(a0, 0.0, None)
            a0 = proj_simplex(a0)
            n = a0.size
        else:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # pick a size that balances expressiveness and speed
        n = 8192  # power of two simplifies FFT length choices
        candidates = _initial_candidates(n, rng)
        # quick scan to pick the most promising start (exact score)
        fft_len0 = _next_pow2(2 * n - 1)
        best_score = np.inf
        best_vec = None
        for v in candidates:
            conv = _self_conv(v, fft_len0)
            score = 2.0 * n * conv.max() / (v.sum() ** 2)
            if score < best_score:
                best_score = score
                best_vec = v.copy()
        a = best_vec
    else:
        a = a0

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Book‑keeping of the current best solution
    # ------------------------------------------------------------------
    conv = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # Gradient‑momentum state
    m = np.zeros_like(a)          # first moment (exponential moving avg)
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    it = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - start_time) / total_time      # ∈[0,1]

        # ----- schedules -------------------------------------------------
        # learning‑rate schedule (Adam‑style decay)
        eta_start, eta_end = 0.5, 0.001
        eta = eta_start * ((eta_end / eta_start) ** frac)

        # soft‑max temperature (exponential growth)
        alpha_start, alpha_end = 5.0, 2000.0
        alpha = alpha_start * ((alpha_end / alpha_start) ** frac)

        # ----- self‑convolution -----------------------------------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]

        # ----- soft‑max weights -----------------------------------------
        c_max = conv.max()
        shifted = conv - c_max                     # shift for numeric stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient (a * rev(w)) ------------------------------------
        w_rev = w[::-1]                            # reverse for cross‑correlation
        grad_len = _next_pow2(n + w_rev.size - 1)   # = next_pow2(3n‑2)
        fa = np.fft.rfft(a, n=grad_len)
        fw = np.fft.rfft(w_rev, n=grad_len)
        grad_full = np.fft.irfft(fa * fw, n=grad_len)
        offset = w_rev.size - 1                     # aligns convolution result
        grad = 2.0 * grad_full[offset: offset + n]   # same shape as a

        # ----- Adam moments ---------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # ----- multiplicative update on the log‑domain -------------------
        update = -eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a * np.exp(update)                     # exponentiated‑gradient step

        # ----- project back onto the simplex -----------------------------
        a = proj_simplex(a)

        # ----- evaluate current objective ---------------------------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)

        # ----- keep the best solution ------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- occasional random jitter ----------------------------------
        if rng.random() < 0.015:  # ≈1.5 % of iterations
            a = _random_block_perturb(a, rng)
            # recompute score after jitter
            fa = np.fft.rfft(a, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
            cur_score = 2.0 * n * conv.max() / (a.sum() ** 2)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{elapsed:6.2f}s] jitter improved best to {best_score:.6f}")

        # ----- stagnation restart (no improvement > 20 s) ----------------
        if now - last_improve > 20.0:
            # restart from the best seen, add a mild log‑normal noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best still {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clip to bounds enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
