"""
Adam‑based smooth‑max mirror descent.

We optimise the simplex vector a (non‑negative, sum = 1) that minimises  

    f(a) = 2·n·max_k (a*a)[k]                         (the same as
    evaluate_sequence after the sum‑normalisation).

A smooth approximation of the max is used:
    ϕ(a) = (1/α)·log(∑_k exp(α·c_k)),   c = a*a.

∂ϕ/∂a = 2·conv(a, w)   with   w_k ∝ exp(α·c_k) (soft‑max weights).

We work in the log‑domain (logits ℓ) so that the simplex constraint is
handled by the soft‑max transform a_i = exp(ℓ_i)/∑_j exp(ℓ_j).  Adam
updates are performed on ℓ, a schedule raises α (smoothness) from a
small value to a large one, and a learning‑rate schedule decays η.
Occasional jitter and block‑perturbations prevent stagnation.
The best vector seen during the run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1). Returns the full
    convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using the provided FFT length."""
    return _fft_convolve(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Main optimisation routine
# ----------------------------------------------------------------------
