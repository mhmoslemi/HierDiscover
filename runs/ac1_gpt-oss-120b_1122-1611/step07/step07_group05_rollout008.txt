    # If there is an existing sequence variable, use it (maybe length differs)

            if existing.size != n:

                x_old = np.linspace(0, 1, existing.size)
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, existing)
                a = np.clip(a, 0.0, None)
                a = _project_simplex(a)
            else:
                a = existing.copy()
                a = np.clip(a, 0.0, None)
                a = _project_simplex(a)

        np.linspace(30.0, 100.0, 15),

    for alpha in alphas:

        # allocate approx equal time to each alpha.

            # update best

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.
--------------------------------------------------------------------
Algorithm
---------

We treat the objective  
    F(a) = 2·n·max(a*a) / (∑a)²                      (∑a is kept = 1)

as the (scale‑invariant) maximum of the autocorrelation.
To reduce the peak we minimise a smooth approximation of the max
by a soft‑max (log‑sum‑exp) and perform projected gradient descent.
The temperature of the soft‑max (parameter ``alpha``) is increased
from a small value (smooth) to a large one (sharp) while the step‑size
is simultaneously decreased – a classic continuation method.

After the gradient phase a short simulated‑annealing (SA) phase
performs cheap mass‑transfer moves with an incremental update of the
convolution, allowing the algorithm to escape tiny local minima.
Both phases keep the best sequence seen so far and finally return it.

The code respects the 1000 s budget (5 s safety margin) and uses only
top‑level helper functions (no closures).  All arithmetic is done with
FFT‑based convolutions (O(n log n)) and the simplex projection
maintains non‑negativity and a unit sum.  The final sequence is clipped
to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally (O(n)).  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # subtract donor contribution, add recipient contribution
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms (exact contribution of delta*delta)
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a simplex projection.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 12))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def _gradient_softmax_max(
    a: np.ndarray,
    fft_len_self: int,
    fft_len_grad: int,
    alpha: float,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns the self‑convolution ``c`` and the gradient of the
    soft‑max approximation of ``max(c)``.
    The smooth approximation is
        M̂ = (1/α)·log( Σ_i exp(α·c_i) ),
    whose gradient w.r.t. ``a`` is 2·(w * a_rev) restricted to the
    original indices, where w_i = exp(α·c_i) / Σ_j exp(α·c_j).

    Parameters
    ----------
    a : np.ndarray
        Current step heights (sum = 1, non‑negative).
    fft_len_self, fft_len_grad : int
        FFT lengths pre‑computed for the self‑convolution and the
        gradient convolution, respectively.
    alpha : float
        Soft‑max temperature (larger → sharper approximation of the max).

    Returns
    -------
    c : np.ndarray
        Full linear self‑convolution (length 2·n‑1).
    grad : np.ndarray
        Gradient vector (length n).
    """
    # ----- self‑convolution -------------------------------------------------
    fa = np.fft.rfft(a, n=fft_len_self)
    c = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * a.size - 1]

    # ----- soft‑max weights -------------------------------------------------
    # shift by the current maximum to avoid overflow
    shift = c.max()
    w = np.exp(alpha * (c - shift))
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        # fall back to the pure max (single active index)
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    # ----- gradient (2·conv(w, a_rev)) ------------------------------------
    a_rev = a[::-1]
    w_fft = np.fft.rfft(w, n=fft_len_grad)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len_grad)
    conv_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len_grad)
    grad = 2.0 * conv_full[a.size - 1 : a.size - 1 + a.size]
    return c, grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Returns a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    The routine combines a continuation‑style projected gradient descent
    on a smooth max‑approximation with a final simulated‑annealing
    refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # 1) Initialise – use a known construction if available, otherwise
    #               a random Dirichlet vector.
    # --------------------------------------------------------------
    # work with a length that is a power of two (FFT‑friendly)
    n = 4096                     # can be changed – 4096 gives good trade‑off
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            # resample / interpolate to the chosen length
            if cand.size != n:
                x_old = np.linspace(0, 1, cand.size)
                x_new = np.linspace(0, 1, n)
                a = np.interp(x_new, x_old, cand)
            else:
                a = cand.copy()
        else:
            a = rng.dirichlet(np.full(n, 1.2))
    else:
        a = rng.dirichlet(np.full(n, 1.2))

    # make sure we obey the simplex constraints (non‑negative, sum = 1)
    a = np.clip(a, 0.0, None)
    a = _project_simplex(a)

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 3) Book‑keeping for the current best solution
    # --------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best_a = a.copy()
    best_score = cur_score
    last_improve = start_time

    print(f"[{0.0:6.2f}s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Gradient‑descent phase (≈70 % of the budget)
    # --------------------------------------------------------------
    grad_deadline = start_time + 0.70 * (deadline - start_time)

    # schedule parameters
    alpha_max = 250.0            # final soft‑max sharpness
    eta_start = 0.30             # initial step size
    eta_end = 0.001              # final step size

    while time.monotonic() < grad_deadline:
        now = time.monotonic()
        frac = (now - start_time) / (grad_deadline - start_time)   # 0 → 1

        # increase soft‑max sharpness (slow start, faster towards the end)
        alpha = 0.5 + (alpha_max - 0.5) * (frac ** 1.5)

        # linearly decreasing learning rate
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # gradient of the smooth max approximation
        c, grad = _gradient_softmax_max(a, fft_len_self, fft_len_grad, alpha)

        # current objective value (exact, not the smoothed one)
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] grad best score = {best_score:.6f}")

        # projected gradient step
        a_tmp = a - eta * grad
        a = _project_simplex(a_tmp)

        # occasional global jitter to avoid stagnation
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] jitter improved to {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Simulated‑annealing fine‑tuning (≈30 % of the budget)
    # --------------------------------------------------------------
    sa_deadline = deadline
    a = best_a.copy()                                 # start SA from the best found
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    move_counter = 0
    RECOMPUTE_EVERY = 2000

    while time.monotonic() < sa_deadline:
        now = time.monotonic()
        # locate the current convolution peak
        k_max = int(c.argmax())
        # contribution of each index to that peak
        partners = k_max - np.arange(n)
        mask = (partners >= 0) & (partners < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[partners[mask]]
        total_contrib = contrib.sum()

        # choose a donor index proportionally to its contribution
        if total_contrib > 0.0:
            prob = contrib / total_contrib
            donor = rng.choice(np.arange(n), p=prob)
        else:
            donor = rng.integers(0, n)

        if a[donor] < 1e-14:
            continue  # nothing to move

        # recipient – we bias towards small entries
        recv_weights = 1.0 - a
        recv_weights[donor] = 0.0
        w_sum = recv_weights.sum()
        if w_sum > 0.0:
            recv_weights /= w_sum
            recipient = rng.choice(np.arange(n), p=recv_weights)
        else:
            recipient = (donor + rng.integers(1, n)) % n

        # amount of mass to transfer (up to half of the donor)
        max_delta = 0.5 * a[donor]
        delta = rng.random() * max_delta

        # apply the move (incrementally update the convolution)
        a_new, c_new = _apply_move(a, c, donor, recipient, delta)
        new_score = 2.0 * n * c_new.max()

        # simple temperature schedule (linear cooling)
        elapsed = now - start_time
        total_elapsed = deadline - start_time
        temperature = 0.08 * (1.0 - elapsed / total_elapsed) + 1e-7

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            a = a_new
            c = c_new
            cur_score = new_score
            move_counter += 1

            # keep the global best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] SA new best = {best_score:.6f}")

            # occasional full recomputation to correct numerical drift
            if move_counter % RECOMPUTE_EVERY == 0:
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # sporadic global perturbation (helps escape plateaus)
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        # restart on long stagnation
        if now - last_improve > 25.0:
            a = best_a.copy()
            a = a * np.exp(rng.normal(0.0, 0.08, size=n))
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart after stagnation (best {best_score:.6f})")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
