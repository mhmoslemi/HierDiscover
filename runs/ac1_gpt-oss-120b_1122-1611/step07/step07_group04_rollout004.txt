    # Determine length n.
    if 'height_sequence_1' in globals():
        base = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if base.ndim==1 and base.size>0:

        else:

    best_vec = None

    # define helper to evaluate vector
    def score_of(a):

    # multiple restarts

        conc = rng.uniform(0.1, 2.0)  # random concentration to vary sparsity

    for restart in range(restarts):

        t = 0

"""
Hybrid hard‑max sub‑gradient descent with simplex projection.

We minimise
    J(a) = 2·n·max_k (a⋆a)[k]            (sum(a) is kept = 1)

The gradient of the non‑smooth max is the sub‑gradient obtained from all
indices where the convolution attains its maximum.  For a mask w that is 1
exactly on those indices we have

    ∂max/∂a = 2· a * w_rev .

The algorithm repeatedly
* evaluates the self‑convolution by FFT,
* builds the mask of maximal entries,
* computes the sub‑gradient by a second FFT,
* performs an exponentiated‑gradient step  a←a·exp(−η·grad)  and
* projects back onto the probability simplex.

A few stochastic “jitter’’ moves (log‑normal block perturbations) are
applied occasionally and after long stagnation the current best solution
is perturbed and the descent restarts.  Several random restarts are tried
within the time budget; the best sequence encountered is returned.

The implementation is fully self‑contained, uses only NumPy and the
provided ``evaluate_sequence`` for the final check.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projection of a vector onto the probability simplex (non‑negative,
    sum = 1).  Works in O(n log n) time.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    (a 5‑second safety margin is kept) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine sequence length and initial feasible vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if base.ndim == 1 and base.size > 0 and np.isfinite(base).all():
            n = base.size
        else:
            n = 3800
    else:
        n = 3800

    # Pre‑computed FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)          # for a⋆a
    fft_len_grad = _next_pow2(3 * n - 2)          # for a * mask_rev (mask length ≤2n‑1)

    # --------------------------------------------------------------
    # 2) Book‑keeping of the global best solution
    # --------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    last_global_improve = start_time

    # --------------------------------------------------------------
    # 3) Helper to evaluate the current objective (fast, no call to
    #    evaluate_sequence – scaling does not matter, sum(a)=1)
    # --------------------------------------------------------------
    def _objective(a: np.ndarray) -> float:
        conv = _self_conv(a, fft_len_self)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 4) Random‑restart loop
    # --------------------------------------------------------------
    RESTARTS = 5
    for restart in range(RESTARTS):
        if time.monotonic() >= deadline:
            break

        # initialise a – sometimes from the current global best (perturbed)
        if best_vec is not None and rng.random() < 0.5:
            a = best_vec.copy()
            a *= np.exp(rng.normal(0.0, 0.09, size=n))
            a = proj_simplex(a)
        else:
            # Dirichlet with a random concentration parameter (0.1 … 2.0)
            conc = rng.uniform(0.1, 2.0)
            a = rng.dirichlet(np.ones(n) * conc)

        # inner optimisation variables
        it = 0
        last_improve = time.monotonic()
        eta0 = 0.8                     # initial step size for exponentiated update
        while time.monotonic() < deadline:
            # ------------------------------------------------------
            # a) self‑convolution and current objective
            # ------------------------------------------------------
            fa = np.fft.rfft(a, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
            max_c = conv.max()
            cur_score = 2.0 * n * max_c

            # keep the best ever seen
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_vec = a.copy()
                last_global_improve = time.monotonic()
                elapsed = last_global_improve - start_time
                print(f"[{elapsed:6.2f}s] restart {restart} new global best = {best_score:.6f}")

            # ------------------------------------------------------
            # b) mask of maximal indices (hard max)
            # ------------------------------------------------------
            # a tiny tolerance avoids numerical mishaps
            tol = max(1e-12, max_c * 1e-12)
            mask = (conv >= max_c - tol).astype(np.float64)
            # normalise the mask (otherwise the gradient magnitude drifts)
            mask_sum = mask.sum()
            if mask_sum > 1.0:
                mask /= mask_sum
            # reverse for the cross‑correlation used in the sub‑gradient
            mask_rev = mask[::-1]

            # ------------------------------------------------------
            # c) sub‑gradient via a second FFT
            # ------------------------------------------------------
            fa = np.fft.rfft(a, n=fft_len_grad)
            fm = np.fft.rfft(mask_rev, n=fft_len_grad)
            grad_full = np.fft.irfft(fa * fm, n=fft_len_grad)
            offset = mask_rev.size - 1            # = 2n‑2
            grad = 2.0 * grad_full[offset: offset + n]

            # ------------------------------------------------------
            # d) exponentiated‑gradient update + simplex projection
            # ------------------------------------------------------
            # simple schedule:  η = η0 / (1 + it/500)
            eta = eta0 / (1.0 + it / 500.0)
            a = a * np.exp(-eta * grad)
            a = proj_simplex(a)

            # ------------------------------------------------------
            # e) occasional stochastic jitter
            # ------------------------------------------------------
            if rng.random() < 0.01:               # ~1 % of iterations
                a = _random_block_perturb(a, rng)

            # ------------------------------------------------------
            # f) stagnation handling
            # ------------------------------------------------------
            now = time.monotonic()
            if now - last_improve > 30.0:         # 30 s without improvement
                # restart from the current global best with a small perturbation
                if best_vec is not None:
                    a = best_vec.copy()
                    a *= np.exp(rng.normal(0.0, 0.07, size=n))
                    a = proj_simplex(a)
                else:
                    a = rng.dirichlet(np.ones(n) * rng.uniform(0.1, 2.0))
                last_improve = now
                print(f"[{now - start_time:6.2f}s] restart {restart} – stagnation reset")

            it += 1

        # end of this restart – if we have been idle for a long time break
        if time.monotonic() - last_global_improve > 120.0:   # 2 min without any global improvement
            break

    # --------------------------------------------------------------
    # 5) Polishing phase on the best vector found so far
    # --------------------------------------------------------------
    if best_vec is not None:
        a = best_vec.copy()
        it = 0
        eta0 = 0.4
        while time.monotonic() < deadline:
            fa = np.fft.rfft(a, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
            max_c = conv.max()
            cur_score = 2.0 * n * max_c
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_vec = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] polishing improved best = {best_score:.6f}")

            # build mask (soft version: a few top entries)
            tol = max(1e-12, max_c * 0.5e-12)
            mask = (conv >= max_c - tol).astype(np.float64)
            mask_sum = mask.sum()
            if mask_sum > 1.0:
                mask /= mask_sum
            mask_rev = mask[::-1]

            # gradient
            fa = np.fft.rfft(a, n=fft_len_grad)
            fm = np.fft.rfft(mask_rev, n=fft_len_grad)
            grad_full = np.fft.irfft(fa * fm, n=fft_len_grad)
            offset = mask_rev.size - 1
            grad = 2.0 * grad_full[offset: offset + n]

            eta = eta0 / (1.0 + it / 800.0)
            a = a * np.exp(-eta * grad)
            a = proj_simplex(a)

            if rng.random() < 0.007:
                a = _random_block_perturb(a, rng)

            it += 1

    # --------------------------------------------------------------
    # 6) Return the best sequence (clipped to the bounds enforced by
    #    evaluate_sequence for safety).  If nothing was found we fall back
    #    to a uniform distribution.
    # --------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
