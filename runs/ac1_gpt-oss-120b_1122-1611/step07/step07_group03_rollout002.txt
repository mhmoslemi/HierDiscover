The goal is to minimise  

\[
\text{score}= \frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear self‑convolution of a non‑negative vector `a`.
The score is *scale‑invariant*, therefore we may normalise every candidate to
\(\sum a = 1\) and simply minimise  

\[
\text{score}=2n\;\max (a*a).
\]

A flat autocorrelation (all convolution entries close to the average
\(\frac{1}{2n-1}\)) gives a low maximum and thus a low score.
The algorithm below is a **simulated‑annealing hill‑climber** that repeatedly
moves a small amount of mass from one position to another while keeping the
vector on the probability simplex.  

*   **Targeted moves** – the current maximal convolution entry is located.
    Mass is taken from indices that contribute most to that peak and moved
    to a random location, directly attacking the worst part of the objective.  
*   **Random moves** – occasional unbiased transfers explore the space.  
*   **Temperature schedule** – a simple linear cooling controls the
    Metropolis acceptance probability.  
*   **Restarts & jitter** – after 30 s without improvement the best known
    solution is slightly perturbed, which helps to escape plateaus.  
*   **Multiple start‑points** – if a previous construction (`height_sequence_1`)
    exists it is used; otherwise a fresh Dirichlet draw is taken.

All heavy work (the convolution) is performed with a single FFT per
iteration, giving an \(O(n\log n)\) step that easily fits inside the 1000 s
budget even for \(n\approx4000\).

The program follows the required interface:

```python

    ...
```

```python
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length 2·n‑1)."""
    return _fft_convolve(a, a, fft_len)


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact score for a normalised vector (∑a = 1)."""
    return 2.0 * n * conv.max()


def _initial_sequence(rng: np.random.Generator, n: int) -> np.ndarray:
    """Return a random Dirichlet start vector of length n (∑ = 1)."""
    a = rng.dirichlet(np.ones(n) * 0.3)   # moderate concentration
    return a.astype(np.float64)


def _perturb_sequence(a: np.ndarray,
                      rng: np.random.Generator,
                      sigma: float = 0.09) -> np.ndarray:
    """
    Gentle multiplicative log‑normal jitter around the current vector
    (used after stagnation).  The result is re‑normalised to sum = 1.
    """
    noise = np.exp(rng.normal(0.0, sigma, size=a.size))
    a2 = a * noise
    a2 = np.clip(a2, 0.0, None)
    s = a2.sum()
    if s == 0.0:
        a2[:] = 1.0 / a2.size
    else:
        a2 /= s
    return a2


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Simulated‑annealing hill‑climber that minimises
    ``evaluate_sequence`` by flattening the self‑convolution of a
    non‑negative step function.  The algorithm uses targeted mass‑transfer
    moves that directly attack the current convolution peak, a simple
    temperature schedule, and occasional global jitter to avoid stagnation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1) Choose length and initialise the current vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0 else np.full_like(a, 1.0 / a.size)
        else:
            n = 3800
            a = _initial_sequence(rng, n)
    else:
        n = 3800
        a = _initial_sequence(rng, n)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length (nearest power‑of‑two ≥ 2 n‑1)
    # ------------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Initial evaluation
    # ------------------------------------------------------------------
    conv = _self_conv(a, fft_len)
    cur_score = _score_from_conv(conv, n)

    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters (linear schedules)
    # ------------------------------------------------------------------
    # Simulated‑annealing temperature – high → low
    temp_start = 0.15
    temp_end   = 1e-6

    # Maximal transferred mass (as a fraction of the source entry)
    step_start = 0.10
    step_end   = 0.001

    # Frequency of a pure random move (vs. a targeted move)
    prob_targeted = 0.70

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    iterations = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        fraction = elapsed / total_time

        # --------------------------------------------------------------
        # 5.1  Temperature & step‑size schedules
        # --------------------------------------------------------------
        temperature = temp_start * (1.0 - fraction) + temp_end * fraction
        max_transfer = step_start * (1.0 - fraction) + step_end * fraction

        # --------------------------------------------------------------
        # 5.2  Decide move type
        # --------------------------------------------------------------
        if rng.random() < prob_targeted:
            # ---- Targeted move -------------------------------------------------
            k_max = int(np.argmax(conv))                     # index of the peak
            # indices that actually participate in this peak
            lo = max(0, k_max - (n - 1))
            hi = min(n - 1, k_max)
            idx = np.arange(lo, hi + 1, dtype=np.int64)
            partner = k_max - idx
            contrib = a[idx] * a[partner]
            total_contrib = contrib.sum()
            if total_contrib > 0.0:
                # pick a source index proportional to its contribution
                prob = contrib / total_contrib
                src = rng.choice(idx, p=prob)
                # destination – any index different from src (uniform)
                dst = rng.integers(0, n)
                while dst == src:
                    dst = rng.integers(0, n)
            else:
                # fallback to a pure random transfer
                src = rng.integers(0, n)
                dst = rng.integers(0, n)
                while dst == src:
                    dst = rng.integers(0, n)
        else:
            # ---- Simple random transfer -----------------------------------------
            src = rng.integers(0, n)
            dst = rng.integers(0, n)
            while dst == src:
                dst = rng.integers(0, n)

        # --------------------------------------------------------------
        # 5.3  Choose amount to move (never exceeds the source mass)
        # --------------------------------------------------------------
        src_mass = a[src]
        if src_mass <= 0.0:
            continue                        # nothing to move
        eps = rng.random() * min(max_transfer * src_mass, src_mass)
        if eps <= 0.0:
            continue

        # --------------------------------------------------------------
        # 5.4  Generate trial vector
        # --------------------------------------------------------------
        a_trial = a.copy()
        a_trial[src] -= eps
        a_trial[dst] += eps

        # --------------------------------------------------------------
        # 5.5  Evaluate trial vector (exact convolution)
        # --------------------------------------------------------------
        conv_trial = _self_conv(a_trial, fft_len)
        trial_score = _score_from_conv(conv_trial, n)

        # --------------------------------------------------------------
        # 5.6  Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if trial_score < cur_score:
            accept = True
        else:
            # probability = exp( -(Δ) / T )
            if temperature > 0.0:
                prob_accept = np.exp(-(trial_score - cur_score) / temperature)
                if rng.random() < prob_accept:
                    accept = True

        if accept:
            a = a_trial
            conv = conv_trial
            cur_score = trial_score
            iterations += 1

            # --------------------------------------------------------------
            # 5.7  Record best solution
            # --------------------------------------------------------------
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.8  Small global jitter (≈2 % of iterations)
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            a = _perturb_sequence(a, rng, sigma=0.07)
            conv = _self_conv(a, fft_len)
            cur_score = _score_from_conv(conv, n)

        # --------------------------------------------------------------
        # 5.9  Restart after stagnation (≈30 s without improvement)
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            a = _perturb_sequence(best, rng, sigma=0.12)
            conv = _self_conv(a, fft_len)
            cur_score = _score_from_conv(conv, n)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best still {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.10  Early exit if target reached (optional)
        # --------------------------------------------------------------
        if best_score < 1.5025:
            print(f"[{elapsed:6.2f}s] target achieved (score {best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # 6) Return the best vector (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
