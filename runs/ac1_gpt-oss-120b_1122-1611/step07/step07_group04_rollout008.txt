"""
propose_candidate – hybrid random‑sampling + softmax/one‑hot mirror descent

We search for a non‑negative step‑function (a probability vector on the
simplex) that minimises

        J(a) = 2·n·max(a*a) / (sum a)²

(the scaling of `a` does not matter, we keep `sum(a)=1`).  The objective
is dominated by the maximum entry of the linear self‑convolution
`c = a*a`.  The algorithm has three stages

1. **Broad random exploration** – draw many Dirichlet vectors with a
   range of concentration parameters (including the uniform vector and
   the previously found best `height_sequence_1`).  Each candidate is
   evaluated via a fast FFT‑based convolution and the best one is kept.

2. **Mirror‑descent refinement** – starting from the best candidate we
   perform projected mirror descent on the simplex.  The gradient of the
   smooth surrogate `ϕ(a)=∑ w_k c_k` is `∇ϕ = 2·(a * rev(w))`.  The
   weight vector `w` is a soft‑max over `c`.  During the last part of the
   run the temperature `α` is ramped up to a very large value (≥ 5000)
   so that `w` becomes a one‑hot vector located at the current maximum
   lag – this exactly yields the sub‑gradient of `max(c)`.

   The update is multiplicative (exponentiated‑gradient) followed by a
   Euclidean projection onto the probability simplex (`proj_simplex`).
   Learning‑rate `η` decays linearly from 0.5 to 0.001.

3. **Escape from stagnation** – with a small probability we apply a
   multiplicative log‑normal perturbation to a random block (function
   `_random_block_perturb`).  If no improvement has been observed for
   > 30 s we restart from the current best with a modest amount of noise.

All steps are O(n log n); with n≈3800 the whole search finishes well
within the 1000 s budget.  The function returns the best sequence found
as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy import sparse

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution `a*a` using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the probability simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best list of heights found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem size and initialise a feasible a
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = init_seq.size
        a = np.clip(init_seq, 0.0, None)
        a = proj_simplex(a)
    else:
        n = 3800                                   # default length similar to prior runs
        a = np.full(n, 1.0 / n, dtype=np.float64)  # uniform start

    fft_len_self = _next_pow2(2 * n - 1)           # reused for all self‑convolutions

    # ------------------------------------------------------------------
    # 2) Small utilities for fast objective evaluation
    # ------------------------------------------------------------------
    def objective_val(vec: np.ndarray) -> float:
        """Return J = 2·n·max(conv) for a probability vector (sum=1)."""
        conv = _self_conv(vec, fft_len_self)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 3) Random‑sampling phase (Dirichlet draws with various concentrations)
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = objective_val(best)
    print(f"[init] score = {best_score:.6f}")

    # include a few handcrafted seeds
    seeds = []

    # uniform (already in best)
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        seeds.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    seeds.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    seeds.append(v)

    # Dirichlet draws with a variety of concentration parameters
    conc_params = [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]
    draws_per_alpha = 150

    # evaluate handcrafted seeds first
    for v in seeds:
        if time.monotonic() >= deadline:
            break
        sc = objective_val(v)
        if sc < best_score:
            best_score = sc
            best = v.copy()
            print(f"[seed] new best {best_score:.6f}")

    # random Dirichlet sampling
    for alpha in conc_params:
        if time.monotonic() >= deadline:
            break
        for _ in range(draws_per_alpha):
            if time.monotonic() >= deadline:
                break
            v = rng.dirichlet(np.ones(n) * alpha)
            sc = objective_val(v)
            if sc < best_score:
                best_score = sc
                best = v.copy()
                print(f"[dirichlet α={alpha:.3f}] new best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Mirror‑descent refinement (soft‑max → one‑hot)
    # ------------------------------------------------------------------
    a = best.copy()
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # parameters for the stochastic perturbation
    jitter_prob = 0.015
    stagnation_time = 30.0  # seconds without improvement before a soft restart

    while time.monotonic() < deadline:
        now = time.monotonic()
        progress = (now - start_time) / total_time          # ∈[0,1]

        # ----- schedules -------------------------------------------------
        # learning rate (multiplicative / exponentiated‑gradient)
        eta = 0.5 * (1.0 - progress) + 0.001 * progress

        # temperature for the soft‑max (grow to a very large value)
        alpha_start, alpha_end = 5.0, 5000.0
        alpha = alpha_start * (1.0 - progress) + alpha_end * progress

        # ----- linear self‑convolution ---------------------------------
        conv = _self_conv(a, fft_len_self)

        # ----- construct weight vector w --------------------------------
        # In the final 20 % of the budget we switch to a strict one‑hot
        # weight (exact sub‑gradient of the max).
        if progress > 0.80:
            k_star = np.argmax(conv)
            w = np.zeros_like(conv)
            w[k_star] = 1.0
        else:
            c_max = conv.max()
            shifted = conv - c_max                     # stability shift
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:                           # numeric safety
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

        # ----- gradient (convolution of a with reversed w) -------------
        if w.sum() == 1.0 and np.count_nonzero(w) == 1:
            # one‑hot – use the cheap O(n) formula
            k_star = int(np.argmax(w))
            idx = k_star - np.arange(n)                # reversed index mapping
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[idx[mask]]
        else:
            w_rev = w[::-1]
            grad_len = _next_pow2(n + w_rev.size - 1)
            fa = np.fft.rfft(a, n=grad_len)
            fw = np.fft.rfft(w_rev, n=grad_len)
            grad_full = np.fft.irfft(fa * fw, n=grad_len)
            offset = w_rev.size - 1
            grad = 2.0 * grad_full[offset: offset + n]

        # ----- exponentiated‑gradient update & simplex projection -------
        a = a * np.exp(-eta * grad)
        a = proj_simplex(a)

        # ----- evaluate current objective --------------------------------
        cur_score = objective_val(a)

        # ----- keep the best solution ------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- occasional random jitter ----------------------------------
        if rng.random() < jitter_prob:
            a = _random_block_perturb(a, rng)
            # recompute objective after jitter
            cur_score = objective_val(a)
            if cur_score < best_score:
                best_score = cur_score
                best = a.copy()
                print(f"[{now - start_time:6.2f}s] jitter improved best to {best_score:.6f}")

        # ----- soft restart after stagnation -----------------------------
        if now - last_improve > stagnation_time:
            a = best.copy()
            # add a little isotropic log‑normal noise before restarting
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – best still {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (safeguard clipping as in evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
