from scipy.optimize import linprog

def lp_optimize(a, M):
    n = a.shape[0]
    # Build sparse matrix for A_ub
    data = []
    rows = []
    cols = []
    for k in range(2*n-1):
        i_low = max(0, k-(n-1))
        i_high = min(k, n-1)
        for i in range(i_low, i_high+1):
            # coefficient a[i] for b_{k-i}
            j = k - i
            data.append(a[i])
            rows.append(k)
            cols.append(j)
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(b); linprog minimizes.
    res = linprog(c, A_ub=A, b_ub=b_ub, A_eq=np.ones((1, n)), b_eq=np.array([1.0]), bounds=(0, None), method='highs')
    return res.x

  renormalised to sum‑to‑one.  The new candidate is accepted with the
  usual Metropolis rule.  The temperature decays linearly from ``0.15``
  to ``1e‑8``.

* **Stage 2 – local refinement (≈70 % of the time).**  A single
  “mass‑transfer’’ move is performed: a donor index *i* (chosen
  proportionally to its contribution to the current worst
  convolution lag) gives a random amount of mass to a recipient index
  *j* (chosen among the smallest entries).  The convolution is updated
  incrementally (O(n)), and the move is accepted with a Metropolis rule
  whose temperature now decays from ``0.09`` to ``1e‑6``.  Every few
  hundred accepted moves the full convolution is recomputed to avoid
  drift; occasional global jitteres keep the search from stagnating.

The routine respects the 1000 s budget (5 s safety margin) and
returns the best sequence found.
"""

import time
import numpy as np
from scipy import fftpack  # used only for self‑convolution via FFT

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new a and c.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # shift of donor i (subtract) and recipient j (add)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ part
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine the length of the vector we shall work with
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            n = cand.size
        else:
            n = 3840
    else:
        n = 3840

    # ------------------------------------------------------------------
    # 2) Build an initial pool of candidate vectors (Beta‑family + uniform)
    # ------------------------------------------------------------------
    x = (np.arange(n, dtype=np.float64) + 0.5) / n                     # mid‑point grid
    alpha_vals = np.concatenate([
        np.linspace(0.10, 0.90, 9),
        np.linspace(1.00, 2.00, 11)
    ])
    candidates = []

    # uniform starter
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previously found vector (if any)
    if "height_sequence_1" in globals():
        prev = np.clip(np.asarray(globals()["height_sequence_1"], dtype=np.float64),
                       0.0, None)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = np.full(n, 1.0 / n, dtype=np.float64)
        candidates.append(prev)

    # Beta‑family (including the limit α→0 which corresponds to a very
    # spiked distribution – we cap the extreme values by normalising)
    for a in alpha_vals:
        raw = (x ** (a - 1.0)) * ((1.0 - x) ** (a - 1.0))
        raw = np.nan_to_num(raw, nan=0.0, posinf=0.0, neginf=0.0)
        tot = raw.sum()
        if tot == 0.0:
            continue
        candidates.append(raw / tot)

    # ------------------------------------------------------------------
    # 3) Pick the best of the initial candidates (using the exact evaluator)
    # ------------------------------------------------------------------
    best = None
    best_score = np.inf
    for vec in candidates:
        # direct evaluation – the formulation is homogeneous, we keep the
        # sum exactly one, therefore the score is 2·n·max(conv).
        conv = _self_conv(vec, _next_pow2(2 * n - 1))
        score = 2.0 * n * conv.max()
        if score < best_score:
            best_score = score
            best = vec.copy()

    # ------------------------------------------------------------------
    # 4) Prepare data structures for the main loop
    # ------------------------------------------------------------------
    a = best.copy()
    fft_len_self = _next_pow2(2 * n - 1)
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    last_improve = start_time

    # time split between the two phases
    total_time = deadline - start_time
    phase1_end = start_time + 0.30 * total_time          # 30 % for global noise

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    move_counter = 0
    RECOMPUTE_EVERY = 250        # full recompute to curb drift

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        if now < phase1_end:
            # --------------------- Phase 1 : global perturbations ----------
            # Linear temperature schedule (high → low)
            frac = (now - start_time) / (phase1_end - start_time)
            temperature = 0.15 * (1.0 - frac) + 1e-8 * frac
            sigma = 0.30 * (1.0 - frac) + 0.02 * frac    # log‑normal width

            # propose a noisy version of the current vector
            log_noise = rng.normal(0.0, sigma, size=n)
            a_new = a * np.exp(log_noise)
            a_new = np.clip(a_new, 0.0, None)
            a_new_sum = a_new.sum()
            if a_new_sum == 0.0:
                a_new = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a_new /= a_new_sum

            # full convolution for the candidate
            c_new = _self_conv(a_new, fft_len_self)
            new_score = 2.0 * n * c_new.max()

        else:
            # ------------------ Phase 2 : local mass‑transfer -------------
            # Temperature schedule for the Metropolis test
            frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = 0.09 * (1.0 - frac) + 1e-6 * frac
            temperature = max(temperature, 1e-12)

            # locate the lag where the convolution attains its maximum
            k_max = int(c.argmax())

            # contribution of each index to that maximum
            idx = np.arange(n)
            partners = k_max - idx
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total = contrib.sum()

            # donor – preferably a heavy contributor
            if total > 0.0:
                prob = contrib / total
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-12:
                continue

            # recipient – favour very small entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # random amount of mass to transfer (up to the whole donor)
            delta = rng.random() * a[donor]

            # apply the move incrementally
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

        # ------------------------------ acceptance test --------------------
        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            a = a_new
            c = c_new
            cur_score = new_score
            move_counter += 1

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

            # periodic exact recomputation to avoid drift of the incremental update
            if move_counter % RECOMPUTE_EVERY == 0:
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # occasional global jitter to escape flat plateaux (≈2 % of iterations)
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        # --------------------------- soft restart -------------------------
        if now - last_improve > 20.0:          # stagnation → restart
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```

This implementation starts from a diverse set of analytically motivated initial vectors, then applies a two‑phase stochastic optimiser (global log‑normal perturbations followed by focused mass‑transfer moves) with a temperature schedule and periodic restarts. The routine respects the 1000 s budget and returns the best step‑function it discovers.
