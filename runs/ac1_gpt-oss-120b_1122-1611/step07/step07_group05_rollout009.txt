"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from the global best ``height_sequence_1`` if
   it exists, otherwise from a random point on the probability simplex
   (sum = 1).  The length ``n`` is taken from the seed sequence or set to
   5000.

2. **Phase 1 – smooth‑max gradient descent** (≈70 % of the time budget)  
   We minimise a smooth approximation of the peak of the self‑convolution:
   f(a)=log∑_k exp(α·c_k) with c = a∗a.  The gradient is
       ∇f = 2·conv(w, a_rev) ,   w_i ∝ exp(α·c_i)   (L2‑normalised),
   where ``a_rev`` is ``a`` reversed.  After each gradient step we project
   back onto the probability simplex.  The smoothness parameter α and the
   step size η are increased / decreased linearly during the phase.

3. **Phase 2 – stochastic local refinement** (≈30 % of the budget)  
   A Metropolis‑annealing loop moves a random fraction of mass from a
   “donor’’ (chosen proportionally to its contribution to the current
   convolution peak) to a “recipient’’ (biased towards small entries).  The
   convolution is updated incrementally in O(n).  Moves are accepted if
   they improve the true objective or with probability exp(−Δ/T) where the
   temperature T is cooled linearly.

4. **Restarts** – after 30 s without improvement the current best solution
   is perturbed by a mild log‑normal jitter and optimisation continues.

All helper functions are top‑level, no closures are used and the routine
respects a 5 s safety margin before the supplied ``budget_s`` expires.
The returned list is clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _apply_move(a: np.ndarray,
                c: np.ndarray,
                i: int,
                j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # linear part of the update
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms (i,i), (j,j) and (i,j)
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ---------- 0. time‑budget handling (5 s safety margin) ----------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ---------- 1. initialise sequence ----------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = cand.size
        a = np.clip(cand, 0.0, None)
        a = _project_simplex(a)
    else:
        n = 5000
        a = rng.random(n)
        a = _project_simplex(a)

    # ---------- 2. pre‑compute FFT lengths ----------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for w * a_rev

    # ---------- 3. initial score ----------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()

    # ---------- 4. hyper‑parameters ----------
    total_time = deadline - start_time
    phase1_end = start_time + 0.70 * total_time   # 70 % for gradient phase

    # schedules (linear interpolation)
    eta_start = 0.5          # initial step size
    eta_end = 1e-5
    alpha_start = 0.5        # smooth‑max starts almost L2‑like
    alpha_end = 10.0

    temp_start = 0.02        # Simulated‑annealing temperature
    temp_end = 1e-6

    move_counter = 0
    RECOMPUTE_EVERY = 1000   # full FFT recomputation interval in phase‑2

    # ---------- 5. main optimisation loop ----------
    while time.monotonic() < deadline:
        now = time.monotonic()

        # ---------- Phase 1 – smooth‑max gradient descent ----------
        if now < phase1_end:
            # linear schedules
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac
            alpha = alpha_start * (1.0 - frac) + alpha_end * frac

            # 1) self‑convolution a*a
            a_fft = np.fft.rfft(a, n=fft_len_self)
            c_full = np.fft.irfft(a_fft * a_fft, n=fft_len_self)
            c = c_full[: 2 * n - 1]

            # 2) smooth weights approximating the max of c
            max_c = c.max()
            # shift by the maximum for numerical stability
            w_raw = np.exp(alpha * (c - max_c))
            w_sum = w_raw.sum()
            if w_sum == 0.0:
                w = np.full_like(w_raw, 1.0 / w_raw.size)
            else:
                w = w_raw / w_sum

            # 3) gradient of log‑sum‑exp(c) w.r.t. a
            a_rev = a[::-1]
            w_fft = np.fft.rfft(w, n=fft_len_grad)
            a_rev_fft = np.fft.rfft(a_rev, n=fft_len_grad)
            full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len_grad)
            grad = 2.0 * full[n - 1 : n - 1 + n]

            # 4) projected gradient step
            a = _project_simplex(a - eta * grad)

            # 5) new score (re‑compute convolution)
            a_fft = np.fft.rfft(a, n=fft_len_self)
            c_full = np.fft.irfft(a_fft * a_fft, n=fft_len_self)
            c = c_full[: 2 * n - 1]
            cur_score = 2.0 * n * c.max()

        # ---------- Phase 2 – stochastic fine‑tuning ----------
        else:
            # linear temperature schedule
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # locate current convolution peak
            k_max = int(c.argmax())

            # contribution of each index to that peak (a_i * a_{k_max-i})
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # pick donor proportionally to contribution, or uniformly if none
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                continue  # nothing to move

            # pick recipient – bias towards small entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            rw_sum = recv_weights.sum()
            if rw_sum > 0.0:
                recv_weights /= rw_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # move a random fraction of the donor’s mass (≤ 50 %)
            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            # incremental update of a and its convolution
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # occasional full recomputation to curb drift
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter (helps escaping plateaus)
            if rng.random() < 0.02:
                jitter_factor = np.exp(rng.normal(0.0, 0.05, size=n))
                a = _project_simplex(a * jitter_factor)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # ---------- keep the best solution ----------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---------- soft restart after prolonged stagnation ----------
        if now - last_improve > 30.0:
            # perturb the best solution with mild log‑normal noise
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            a = _project_simplex(best * noise)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart with jitter (best {best_score:.6f})")

    # ---------- 6. return the best sequence ----------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
