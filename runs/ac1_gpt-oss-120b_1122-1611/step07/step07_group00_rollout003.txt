def _initial_from_uniform_conv(n):
    P = _next_pow2(2*n - 1)
    target = np.zeros(P, dtype=np.float64)
    target[:2*n-1] = 1.0/(2*n-1)
    # Compute FFT via rfft
    F = np.fft.rfft(target)
    # sqrt of complex numbers
    A = np.sqrt(F)
    a_full = np.fft.irfft(A, n=P)
    # keep first n entries
    a = a_full[:n]
    # Ensure nonnegative:
    a = np.clip(a, 0, None)
    total = a.sum()
    if total == 0:
        a = np.full(n, 1.0/n)
    else:
        a /= total
    return a

"""
Improved search for a step‑function that minimises
    evaluate_sequence(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm works directly on the probability simplex by
representing the sequence `a` as a soft‑max of log‑weights `x`.
A smooth “log‑sum‑exp” surrogate of the max is used initially;
later the optimisation switches to the exact max (hard max) to
fine‑tune the solution.

Key ideas
---------
* **Analytic initialization** – start from a distribution whose
  self‑convolution is (approximately) flat.  This is obtained by
  taking the square‑root of the Fourier transform of a uniform
  convolution vector (the “uniform‑conv” init).  As a fallback we
  also offer an arcsine‑shaped init or a uniform init.
* **Adam optimiser** on the unconstrained log‑weights.
* **Two‑phase schedule** – a smoothed max (γ from 0.5 to 500) for
  the first ~80 % of the time budget, then a hard‑max gradient
  (sub‑gradient) for the remaining time.
* **Stagnation handling** – if no improvement is seen for 30 s,
  the state is re‑initialised from the current best with moderate
  log‑normal noise.
* **Occasional jitter** – a random block perturbation helps to
  escape shallow basins.
* **Best‑ever tracking** – the exact objective (via
  ``evaluate_sequence``) is evaluated periodically and the best
  sequence is kept.

The routine is limited to the given time‑budget (≈1000 s) and
returns the best sequence found as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient ∂F/∂a for a surrogate where ``w`` are the (soft) weights
    applied to each lag of the convolution (length 2·n‑1).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _arcsine_init(n: int) -> np.ndarray:
    """Arcsine‑shaped start (a_i ∝ 1/√{(i+1)(n‑i)})."""
    i = np.arange(n, dtype=np.float64)
    raw = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    raw = np.clip(raw, 0.0, None)
    total = raw.sum()
    return raw / total


def _uniform_conv_init(n: int) -> np.ndarray:
    """
    Initialise a distribution whose *linear* self‑convolution is (approximately)
    uniform on its support.  This is obtained by taking the square‑root of the
    Fourier transform of a uniform convolution vector.
    """
    P = _next_pow2(2 * n - 1)
    target = np.zeros(P, dtype=np.float64)
    target[: 2 * n - 1] = 1.0 / (2 * n - 1)          # flat target convolution
    F = np.fft.rfft(target)                         # complex spectrum
    A = np.sqrt(F)                                   # element‑wise complex sqrt
    a_full = np.fft.irfft(A, n=P)                    # real signal of length P
    a = a_full[:n]                                   # keep first n entries
    a = np.clip(a, 0.0, None)                        # enforce non‑negativity
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _random_init(n: int, rng: np.random.Generator) -> np.ndarray:
    """Pick a random sensible initial distribution."""
    choice = rng.integers(0, 3)
    if choice == 0:
        a = np.full(n, 1.0 / n)               # uniform
    elif choice == 1:
        a = _arcsine_init(n)                 # arcsine‑like
    else:
        a = _uniform_conv_init(n)            # flat‑convolution approx.
    # small log‑normal perturbation to avoid exact symmetry
    a *= np.exp(rng.normal(0.0, 0.02, size=n))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    a /= s if s > 0 else 1.0 / n
    return a

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initial log‑weights
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)
        else:
            n = 3809
            a0 = _uniform_conv_init(n)
            x = np.log(a0 + 1e-12)
    else:
        n = 3809
        a0 = _uniform_conv_init(n)
        x = np.log(a0 + 1e-12)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for the self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Tracking best solution
    # ------------------------------------------------------------------
    a = _softmax(x)
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()                 # exact objective for sum≈1
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 5) Hyper‑parameters (schedules)
    # ------------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 500.0          # large → hard max approximation
    lr_start = 0.8
    lr_end = 0.02

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # ---- current iterate on the simplex ----
        a = _softmax(x)                     # probability vector (sum ≈ 1)

        # ---- self‑convolution (linear) ----
        c = _self_conv(a, fft_len_self)     # length 2·n‑1

        # ---- decide which surrogate to use ----
        if frac < 0.80:
            # ----- smooth max via log‑sum‑exp -----
            gamma = gamma_start + (gamma_end - gamma_start) * frac
            c_shift = c - np.max(c)                # stabilise exp
            w = np.exp(gamma * c_shift)
            w_sum = w.sum()
            w /= w_sum                              # probability weights
        else:
            # ----- hard max (sub‑gradient) -----
            # pick a random index among the maximisers to avoid bias
            max_val = c.max()
            max_inds = np.flatnonzero(c == max_val)
            k = rng.choice(max_inds)
            w = np.zeros_like(c)
            w[k] = 1.0                               # delta weight

        # ---- gradient w.r.t. a (smooth or hard) ----
        g_a = _gradient_wrt_a(a, w)                 # length n

        # ---- chain rule to log‑weights x ----
        avg = np.dot(a, g_a)                        # scalar = a·grad_a
        g_x = a * (g_a - avg)                       # ∂F/∂x (projected onto simplex)

        # ---- Adam step ----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ---- occasional exact evaluation (every ~0.3 s) ----
        if (t % 30) == 0:                               # cheap periodic check
            cur_score = 2.0 * n * c.max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- soft restart on stagnation ----
        if now - last_improve > 30.0:
            # re‑initialise from the current best with moderate log‑normal noise
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # ---- occasional global jitter ----
        if rng.random() < 0.02:

            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 7) Final evaluation and return
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best.tolist())
    print(f"Final best score (exact evaluate_sequence) = {final_score:.6f}")

    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
