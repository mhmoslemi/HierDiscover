def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    # seed for reproducibility.

    import time, numpy as np

    rng = np.random.default_rng(seed)

    # Determine start sequence
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # ensure positivity and non-zero sum
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        n = 3840
        a = rng.dirichlet(np.ones(n) * 0.3).astype(np.float64)

    n = a.size
    # compute convolution
    c = np.convolve(a, a)
    cur_score = 2.0 * n * c.max()
    best_a = a.copy()
    best_score = cur_score
    best_c = c.copy()
    last_improve = time.monotonic()
    start = last_improve

    deadline = start + max(0.0, budget_s - 5.0)

    # parameters
    max_frac = 0.15   # max fraction of source mass moved each step
    recompute_every = 5000
    iteration = 0
    # temperature schedule
    T0 = 0.12
    T1 = 1e-6

    # loop
    while time.monotonic() < deadline:
        # compute schedule
        elapsed = time.monotonic() - start
        total = deadline - start
        progress = elapsed / total
        temperature = T0 * (1 - progress) + T1 * progress
        # pick source index with non-zero mass
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        # pick target index different
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # epsilon fraction
        frac_move = rng.random() * max_frac
        epsilon = a[i] * frac_move
        if epsilon <= 0:
            continue

        # compute new convolution
        c_new = c.copy()
        # linear updates
        c_new[i:i+n] -= 2.0 * epsilon * a
        c_new[j:j+n] += 2.0 * epsilon * a
        # quadratic adjustments
        idx_2i = 2*i
        idx_2j = 2*j
        idx_ij = i + j
        # ensure indices within bound
        c_new[idx_2i] += epsilon*epsilon
        c_new[idx_2j] += epsilon*epsilon
        c_new[idx_ij] -= 2.0 * epsilon * epsilon

        new_score = 2.0 * n * c_new.max()
        delta = new_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                accept = True

        if accept:
            # apply move to a and c
            a[i] -= epsilon
            a[j] += epsilon
            c = c_new
            cur_score = new_score
            iteration += 1

            # keep best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_c = c.copy()
                last_improve = time.monotonic()
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # periodic recompute
        if iteration % recompute_every == 0:
            c = np.convolve(a, a)
            cur_score = 2.0 * n * c.max()
            # optionally update best if this recompute improved
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_c = c.copy()
                last_improve = time.monotonic()
                print(f"[{elapsed:6.2f}s] recompute -> new best {best_score:.6f}")

        # stagnation restart
        if time.monotonic() - last_improve > 30.0:
            # apply a random jitter
            a = best_a.copy()
            # multiplicative log-normal noise

    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]

"""
Local simulated‑annealing search for a non‑negative step function.

The objective of `evaluate_sequence` (with sum(a)=1) is 2·n·max(a∗a).
We therefore minimise the peak of the self‑convolution while keeping the
vector on the simplex.  Starting from the current best construction (or a
Dirichlet draw) we repeatedly transfer a small amount of mass from one
coordinate to another.  The effect on the full convolution can be updated
in O(n) time using the analytical Δc derived from
   c = a∗a,   Δc = -2·ε·a shifted by i  + 2·ε·a shifted by j
          + ε² at 2i, 2j  – 2·ε² at i+j .
A simulated‑annealing acceptance rule (temperature linearly decays) allows
occasional uphill moves, helping the search escape plateaus.  When no
improvement is seen for ~30 s a gentle log‑normal perturbation (“restart”)
is applied around the best solution seen so far.

All operations are pure NumPy; each iteration costs only a few
vectorised slices and a max‑reduction, so millions of moves fit easily
into the 1000 s budget.
"""

import time
import numpy as np


def _initial_sequence(rng: np.random.Generator) -> np.ndarray:
    """Return a starting distribution on the simplex.
    Uses the global `height_sequence_1` if present, otherwise a Dirichlet draw."""
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        n = 3840                     # a reasonable default size
        a = rng.dirichlet(np.ones(n) * 0.3).astype(np.float64)
    return a


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Exact linear convolution a*a (full length 2·len(a)-1)."""
    return np.convolve(a, a)


def _apply_pair_move(
    c: np.ndarray,
    a: np.ndarray,
    i: int,
    j: int,
    eps: float,
) -> np.ndarray:
    """Return a copy of `c` after moving mass `eps` from index i to j.
    The update follows the analytical formula described in the module docstring."""
    # make a temporary copy (re‑use pre‑allocated array outside the function)
    c_new = c.copy()
    # linear part: -2·eps·a shifted by i  + 2·eps·a shifted by j
    c_new[i : i + a.size] -= 2.0 * eps * a
    c_new[j : j + a.size] += 2.0 * eps * a
    # quadratic corrections at the three special offsets
    idx_2i = 2 * i
    idx_2j = 2 * j
    idx_ij = i + j
    c_new[idx_2i] += eps * eps
    c_new[idx_2j] += eps * eps
    c_new[idx_ij] -= 2.0 * eps * eps
    return c_new


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine runs for roughly ``budget_s`` seconds
    (leaving a 5 s safety margin) and returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a on the simplex
    # ------------------------------------------------------------------
    a = _initial_sequence(rng)               # shape (n,)
    n = a.size

    # ------------------------------------------------------------------
    # 2) Initial convolution and bookkeeping
    # ------------------------------------------------------------------
    c = _full_convolution(a)                  # length 2n‑1
    cur_score = 2.0 * n * c.max()             # because sum(a) == 1
    best_a = a.copy()
    best_c = c.copy()
    best_score = cur_score

    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters (linear schedules)
    # ------------------------------------------------------------------
    # temperature for Metropolis acceptance (high → low)
    T_start = 0.12
    T_end = 1e-6

    # size of a single mass transfer – fraction of the source entry
    frac_start = 0.30    # up to 30 % of the source mass at the beginning
    frac_end = 0.02      # later only tiny tweaks

    # how often we recompute the convolution from scratch (to avoid drift)
    RECOMPUTE_EVERY = 5000

    # ------------------------------------------------------------------
    # 4) Main simulated‑annealing loop
    # ------------------------------------------------------------------
    accepted = 0
    iterations = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        prog = elapsed / total

        # temperature schedule (linear)
        temperature = T_start * (1.0 - prog) + T_end * prog
        # fraction of source mass to move (linear decay)
        max_frac = frac_start * (1.0 - prog) + frac_end * prog

        # ---- 4.1  Choose a random source i with non‑zero mass ----------
        i = rng.integers(0, n)
        if a[i] < 1e-14:          # practically zero – pick another
            continue

        # ---- 4.2  Choose a distinct target index j --------------------
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # ---- 4.3  Choose how much mass to move -----------------------
        eps = a[i] * rng.random() * max_frac
        if eps <= 0.0:
            continue

        # ---- 4.4  Compute the updated convolution --------------------
        # (re‑use a pre‑allocated buffer to avoid repeated allocations)
        c_new = c.copy()
        c_new[i : i + n] -= 2.0 * eps * a
        c_new[j : j + n] += 2.0 * eps * a
        idx_2i = 2 * i
        idx_2j = 2 * j
        idx_ij = i + j
        c_new[idx_2i] += eps * eps
        c_new[idx_2j] += eps * eps
        c_new[idx_ij] -= 2.0 * eps * eps

        new_score = 2.0 * n * c_new.max()
        delta = new_score - cur_score

        # ---- 4.5  Metropolis acceptance --------------------------------
        accept = False
        if delta < 0.0:
            accept = True
        else:
            # occasional uphill moves, temperature may become extremely small
            if temperature > 0.0 and rng.random() < np.exp(-delta / temperature):
                accept = True

        if accept:
            # apply the move permanently
            a[i] -= eps
            a[j] += eps
            c = c_new
            cur_score = new_score
            accepted += 1

            # keep the best solution seen so far
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_c = c.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        iterations += 1

        # ---- 4.6  Periodic exact recomputation ------------------------
        if accepted % RECOMPUTE_EVERY == 0:
            c = _full_convolution(a)
            cur_score = 2.0 * n * c.max()
            # (maybe the exact recompute yields a new best)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                best_c = c.copy()
                last_improve = time.monotonic()
                print(f"[{time.monotonic() - start_time:6.2f}s] recomputed – new best {best_score:.6f}")

        # ---- 4.7  Stagnation handling (restart after 30 s no improve) --
        if now - last_improve > 30.0:
            # gentle log‑normal jitter around the current best
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a *= noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _full_convolution(a)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            accepted = 0
            print(f"[{elapsed:6.2f}s] restart – score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)   # respect the evaluator's caps
    return [float(x) for x in best_a]
