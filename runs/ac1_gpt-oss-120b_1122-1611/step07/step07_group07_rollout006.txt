```
def _project_simplex(v):
    # ensures nonnegative, sum = 1.
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
    eta_end = 1e-5

    eps_end = 1e-12

    # variables for momentum

    beta = 0.9

    move_counter = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # Phase1 gradient descent
        if now < phase1_end:
            # compute c
            c = _self_conv(a, fft_len_self)
            max_c = c.max()
            # compute eps schedule: start relatively large, decay linearly
            frac = (now - start_time) / (phase1_end - start_time)

            w = np.exp((c - max_c) / eps)
```

    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv

def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new a and c."""
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    c_new = c.copy()
    # donor i (subtract) and recipient j (add)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a
    # quadratic ΔΔ part
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta
    return a_new, c_new

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # algorithm from https://arxiv.org/abs/1101.6081
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    # ensure numerical sum exactly 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.

    The method consists of two phases:

    1️⃣  A deterministic smoothed‑max gradient descent.  The L∞ norm of the
        self‑convolution is approximated by a soft‑max with a temperature that
        is annealed.  The gradient of the smooth surrogate is computed via FFTs
        and an additive projected‑gradient step (with momentum) is taken.
        The iterate is always projected back onto the probability simplex.

    2️⃣  A stochastic refinement that repeatedly moves a small amount of mass
        from a “donor’’ index that contributes heavily to the current peak to a
        “recipient’’ index with low mass.  Moves are accepted by a Metropolis
        rule with a slowly decreasing temperature.  Occasionally a random
        block‑wise perturbation is applied to escape plateaus.

    Restarts are triggered when no improvement is observed for a while; the
    current best solution is perturbed with modest log‑normal noise and the
    optimisation continues.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise the distribution on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = None
        else:
            a = None
    else:
        a = None

    if a is None:
        # default length – similar to previous best but a power of two is convenient
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT sizes
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)               # current self‑convolution
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    # allocate ~35 % of the time to the deterministic phase
    phase1_end = start_time + 0.35 * total_time

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters
    # ------------------------------------------------------------------
    eta_start = 0.8            # initial additive step size
    eta_end   = 1e-5           # final step size
    eps_start_frac = 0.5       # initial temperature as fraction of current max
    eps_end = 1e-12

    beta = 0.9                 # momentum coefficient
    momentum = np.zeros_like(a)

    move_counter = 0
    RECOMPUTE_EVERY = 250

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – smoothed‑max projected gradient descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # recompute convolution (needed for soft‑max weights)
            c = _self_conv(a, fft_len_self)
            max_c = c.max()
            # annealing schedule for the soft‑max temperature
            frac = (now - start_time) / (phase1_end - start_time)
            eps = max(max_c * (eps_start_frac * (1.0 - frac) + 0.01 * frac),
                      eps_end)

            # soft‑max weights (stable formulation)
            w = np.exp((c - max_c) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:
                # fall back to hard max
                w = np.zeros_like(c)
                w[int(c.argmax())] = 1.0
                w_sum = 1.0
            w /= w_sum

            # gradient of the smooth surrogate via FFTs
            a_rev = a[::-1]
            fw = np.fft.rfft(w, n=fft_len_grad)
            fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: 3 * n - 1]
            grad = 2.0 * conv_full[n - 1:n - 1 + n]

            # additive step with momentum
            eta = eta_start * (1.0 - frac) + eta_end * frac
            momentum = beta * momentum + (1.0 - beta) * grad
            a_tmp = a - eta * momentum

            # projection onto the simplex (non‑negative, sum‑to‑1)
            a = _project_simplex(a_tmp)

            # re‑evaluate the true objective (no need to recompute c here,
            # we will do it at the top of the next iteration)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        else:
            # --------------------------------------------------------------
            # Phase 2 – fast stochastic refinement (simulated annealing)
            # --------------------------------------------------------------
            # locate the current peak of the convolution
            k_max = int(c.argmax())

            # contribution of each index to that peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            # recipient favours small entries (inverse probability)
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            max_delta = 0.5 * a[donor]
            if max_delta > 0.0:
                delta = rng.random() * max_delta
                a_new, c_new = _apply_move(a, c, donor, recipient, delta)
                new_score = 2.0 * n * c_new.max()

                # annealing temperature schedule
                t_frac = (now - phase1_end) / (deadline - phase1_end)
                temperature = max(0.05 * (1.0 - t_frac) + 1e-7 * t_frac, 1e-12)

                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                    c = c_new
                    cur_score = new_score
                    move_counter += 1
                    if move_counter % RECOMPUTE_EVERY == 0:
                        c = _self_conv(a, fft_len_self)
                        cur_score = 2.0 * n * c.max()

            # occasional global jitter to shake the system
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # ------------------------------------------------------------------
        # Keep the best solution seen so far
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Soft restart after stagnation
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            momentum = np.zeros_like(a)
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            max_delta = 0.5 * a[donor]
            if max_delta > 0.0:
                delta = rng.random() * max_delta
                a_new, c_new = _apply_move(a, c, donor, recipient, delta)
                new_score = 2.0 * n * c_new.max()

                # annealing temperature (slowly decreasing)
                t_frac = (now - phase1_end) / (deadline - phase1_end)
                temperature = max(0.05 * (1.0 - t_frac) + 1e-7 * t_frac,
                                  1e-12)

                if (new_score < cur_score or
                        rng.random() < np.exp(-(new_score - cur_score) / temperature)):
                    a = a_new
                    c = c_new
                    cur_score = new_score
                    move_counter += 1
                    if move_counter % RECOMPUTE_EVERY == 0:
                        c = _self_conv(a, fft_len_self)
                        cur_score = 2.0 * n * c.max()

            # occasional global jitter to shake the system out of plateaus
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            momentum = np.zeros_like(a)
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
