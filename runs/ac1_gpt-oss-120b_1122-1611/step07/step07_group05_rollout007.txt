                a = _init_random(n, rng)
            # compute FFT lengths
            fft_len_self = _next_pow2(2 * n - 1)
            fft_len_grad = _next_pow2(3 * n - 2)

            # iteration counter
            it = 0
            # schedule parameters
            eta = 0.5
            alpha = 1.0
            while time.monotonic() < sub_deadline:
                # gradient descent on smooth max
                c, grad = _gradient_smooth_max(a, alpha, fft_len_self, fft_len_grad)
                # step size schedule decreasing with it
                eta = max(1e-5, eta * 0.9995)  # decay slowly
                a_tmp = a - eta * grad
                a = _project_simplex(a_tmp)
                # update alpha slowly up to max
                alpha = min(200.0, alpha * 1.005)

                # compute current score
                cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
                if cur_score < best_score - 1e-9:
                    best_score = cur_score
                    best_seq = a.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] new best (n={n}) score = {best_score:.6f}")
                it += 1
                # occasional restart
                if it % 1000 == 0:
                    # jitter
                    a = _random_block_perturb(a, rng)
            # after gradient loops, optionally perform simulated annealing refinement for leftover time
            # compute convolution
            c = _self_conv(a, _next_pow2(2*n-1))
            cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
            # annealing
            temperature = 0.1
            while time.monotonic() < sub_deadline:
                # choose donor, recipient as before
                k_max = int(c.argmax())
                partners = k_max - np.arange(n)
                mask = (partners >= 0) & (partners < n)
                contrib = np.zeros_like(a)
                contrib[mask] = a[mask] * a[partners[mask]]
                total_contrib = contrib.sum()
                donor = rng.choice(np.arange(n), p=contrib/total_contrib) if total_contrib > 0 else rng.integers(n)
                if a[donor] < 1e-12:
                    continue
                recv_weights = 1.0 - a
                recv_weights[donor] = 0.0
                rw_sum = recv_weights.sum()
                recipient = rng.choice(np.arange(n), p=recv_weights/rw_sum) if rw_sum > 0 else (donor+1)%n
                max_delta = 0.5 * a[donor]
                delta = rng.random() * max_delta
                a_new, c_new = _apply_move(a, c, donor, recipient, delta)
                new_score = 2.0 * n * c_new.max() / (a.sum() ** 2)  # sum unchanged
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                    c = c_new
                    cur_score = new_score
                    if cur_score < best_score - 1e-9:
                        best_score = cur_score
                        best_seq = a.copy()
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] new best (anneal n={n}) score = {best_score:.6f}")
                temperature = max(1e-8, temperature * 0.9999)
                # periodic jitter
                if rng.random() < 0.01:
                    a = _random_block_perturb(a, rng)
                    c = _self_conv(a, _next_pow2(2*n-1))
                    cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
        # after each length, we have best_seq possibly for that length
    # final return: best_seq or fallback uniform if none
    if best_seq is None:
        # fallback uniform
        n = 500
        best_seq = np.full(n, 1.0/n)
    # clip to bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

"""
Algorithm: Smooth‑max gradient descent + simulated‑annealing refinement.

*   Each candidate length `n` is searched independently.
*   The objective `2·n·max(conv(a,a)) / (sum(a)^2)` is homogeneous, so we
    keep the vector on the probability simplex (sum = 1) which makes the
    projection step cheap.
*   A smooth approximation of the max – the log‑sum‑exp with a temperature
    parameter `α` – is minimised by gradient descent.  The gradient can be
    obtained analytically by one extra FFT (convolution of the soft‑max
    weights with the reversed signal).  `α` is slowly increased so that the
    smoothed problem becomes closer to the true `ℓ_∞` problem.
*   After a short gradient phase a lightweight simulated‑annealing move
    (transfer a small amount of mass from a “donor’’ that contributes most
    to the current peak to a low‑value “recipient’’) is applied.  The
    convolution is updated incrementally, making the refinement very fast.
*   Several initialisations (exponential decay, Dirichlet‑sampled,
    perturbations of a known solution) are tried.  The best sequence seen
    before the time budget expires is returned.

The routine respects a 5 s safety margin and prints intermediate best
scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # subtract donor contribution, add recipient contribution
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a simplex projection.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Gradient of the smooth‑max (log‑sum‑exp) of the self‑convolution
# ----------------------------------------------------------------------


def _gradient_smooth_max(
    a: np.ndarray,
    alpha: float,
    fft_len_self: int,
    fft_len_grad: int,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns the self‑convolution ``c`` and the gradient of the
    smoothed max (log‑sum‑exp) with respect to ``a``.
    """
    n = a.shape[0]

    # ----- self‑convolution -------------------------------------------------
    a_fft = np.fft.rfft(a, n=fft_len_self)
    c = np.fft.irfft(a_fft * a_fft, n=fft_len_self)[: 2 * n - 1]

    # ----- smooth max (log‑sum‑exp) weights -------------------------------
    max_c = c.max()
    # stable exponentiation
    w = np.exp(alpha * (c - max_c))
    p = w / w.sum()          # probability vector that approximates the argmax

    # ----- gradient = 2·(p * a_rev) ---------------------------------------
    a_rev = a[::-1]
    p_fft = np.fft.rfft(p, n=fft_len_grad)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len_grad)
    conv_full = np.fft.irfft(p_fft * a_rev_fft, n=fft_len_grad)
    # Only the first (3·n‑2) entries are the true linear convolution
    conv = conv_full[: 3 * n - 2]
    grad = 2.0 * conv[n - 1 : n - 1 + n]  # slice corresponding to the original a

    return c, grad


# ----------------------------------------------------------------------
# Initialisation helpers
# ----------------------------------------------------------------------


def _init_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    """Decaying exponential vector, normalised to the simplex."""
    r = rng.uniform(0.5, 0.99)               # decay factor
    a = np.array([r ** i for i in range(n)], dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a


def _init_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet‑sampled vector (concentration drawn uniformly)."""
    conc = rng.uniform(0.1, 5.0)
    a = rng.gamma(conc, 1.0, size=n)
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Lengths to try – include the known construction if available
    # ------------------------------------------------------------------
    lengths = [200, 400, 800, 1600, 3200, 3809]
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            lengths.append(cand.size)

    # ------------------------------------------------------------------
    # 2) Global bookkeeping
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_vec = None
    last_improve = start_time

    # ------------------------------------------------------------------
    # 3) Main loop over candidate lengths
    # ------------------------------------------------------------------
    for n in lengths:
        # stop if we are out of time
        if time.monotonic() >= deadline:
            break

        # allocate a slice of the remaining budget to this length
        remaining = deadline - time.monotonic()
        # 60 % of the remaining time for the current length,
        # the rest will be used by later lengths.
        sub_deadline = time.monotonic() + remaining * 0.60

        # pre‑compute FFT lengths for this ``n``
        fft_len_self = _next_pow2(2 * n - 1)
        fft_len_grad = _next_pow2(3 * n - 2)

        # ----------------------------------------------------------------
        # 3a) Try a few different initial vectors for the same length
        # ----------------------------------------------------------------
        for init_id in range(3):
            if time.monotonic() >= sub_deadline:
                break

            # ----------------------------------------------------------------
            # Initialise a
            # ----------------------------------------------------------------
            if init_id == 0:
                a = _init_exponential(n, rng)
            elif init_id == 1:
                a = _init_dirichlet(n, rng)
            else:
                # jitter around a known construction (if any)
                if best_vec is not None and best_vec.size == n:
                    a = best_vec.copy()
                    a = _random_block_perturb(a, rng)
                else:
                    a = _init_exponential(n, rng)

            # ----------------------------------------------------------------
            # Gradient descent on the smoothed max (log‑sum‑exp)
            # ----------------------------------------------------------------
            eta = 0.6                 # initial step size
            alpha = 1.0               # start with a very smooth approximation
            max_alpha = 250.0
            eta_decay = 0.9995
            alpha_inc = 1.005

            iter_cnt = 0
            while time.monotonic() < sub_deadline:
                # compute convolution and gradient of smooth max
                c, grad = _gradient_smooth_max(
                    a, alpha, fft_len_self, fft_len_grad
                )

                # projected gradient step
                a_tmp = a - eta * grad
                a = _project_simplex(a_tmp)

                # update schedules
                eta = max(1e-6, eta * eta_decay)
                alpha = min(max_alpha, alpha * alpha_inc)

                # current (exact) score – cheap because we already have ``c``
                cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_vec = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] new best (grad, n={n}) = {best_score:.6f}")

                # occasional global jitter to shake out of plateaus
                if rng.random() < 0.01:
                    a = _random_block_perturb(a, rng)

                iter_cnt += 1
                # stop after a reasonable number of iterations for this length
                if iter_cnt >= 4000:
                    break

            # ----------------------------------------------------------------
            # 3b) Fine‑grained stochastic refinement (simulated annealing)
            # ----------------------------------------------------------------
            # recompute the exact convolution for the current ``a``
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

            # annealing schedule
            temp_start = 0.08
            temp_end = 1e-6
            anneal_start = time.monotonic()
            while time.monotonic() < sub_deadline:
                now = time.monotonic()
                t_frac = (now - anneal_start) / max(1e-12, sub_deadline - anneal_start)
                temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
                temperature = max(temperature, 1e-9)

                # locate the current peak of the convolution
                k_max = int(c.argmax())

                # contribution of each index to this peak
                partners = k_max - np.arange(n)
                mask = (partners >= 0) & (partners < n)
                contrib = np.zeros_like(a)
                contrib[mask] = a[mask] * a[partners[mask]]
                total_contrib = contrib.sum()

                # pick donor (prefer high contribution)
                if total_contrib > 0.0:
                    prob = contrib / total_contrib
                    donor = rng.choice(np.arange(n), p=prob)
                else:
                    donor = rng.integers(0, n)

                if a[donor] < 1e-14:
                    continue

                # pick recipient (prefer low values)
                recv_w = 1.0 - a
                recv_w[donor] = 0.0
                rw_sum = recv_w.sum()
                if rw_sum > 0.0:
                    recv_w /= rw_sum
                    recipient = rng.choice(np.arange(n), p=recv_w)
                else:
                    recipient = (donor + rng.integers(1, n)) % n

                # amount to move (up to half of the donor)
                max_delta = 0.5 * a[donor]
                delta = rng.random() * max_delta

                # apply move and update convolution incrementally
                a_new, c_new = _apply_move(a, c, donor, recipient, delta)
                new_score = 2.0 * n * c_new.max() / (a.sum() ** 2)

                # Metropolis acceptance
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                    c = c_new
                    cur_score = new_score
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_vec = a.copy()
                        last_improve = time.monotonic()
                        elapsed = last_improve - start_time
                        print(f"[{elapsed:6.2f}s] new best (anneal, n={n}) = {best_score:.6f}")

                # occasional jitter
                if rng.random() < 0.02:
                    a = _random_block_perturb(a, rng)
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

                # break if we spent too long on this length
                if time.monotonic() >= sub_deadline:
                    break

    # ------------------------------------------------------------------
    # 4) Fallback – uniform distribution if nothing was found (should not happen)
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 300
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
