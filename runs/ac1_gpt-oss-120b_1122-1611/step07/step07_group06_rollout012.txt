"""
Propose a step‑function (non‑negative heights) that minimizes the value
returned by ``evaluate_sequence``.

The optimizer works as a projected‑Adam scheme on the *exact* max‑convolution
objective.  At each iteration the convolution `c = a*a` is computed with an
FFT, the indices where `c` attains its maximum are identified and an
(averaged) sub‑gradient of the max‑operator is obtained via a correlation
step (also using FFTs).  The update is projected onto the probability simplex
so the sum of the coefficients stays equal to 1.  A small multiplicative
log‑normal jitter is applied every few hundred steps, and the best exact
solution seen so far (according to ``evaluate_sequence``) is stored and
returned.  The whole procedure respects the 1000 s time budget (5 s safety
margin) and never violates the security checks of the evaluator.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – fall back to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safeguard
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w

# Main optimisation routine
# ----------------------------------------------------------------------

    Return a list of non‑negative heights that (hopefully) yields a score

    descent on the max‑convolution objective using an Adam optimiser,
