(`height_sequence_1` when it exists) and run a **dual‑phase** optimiser:

  Gradients are obtained analytically by a single FFT‑based convolution.

* Phase 2 – a pure stochastic “mass‑transfer” annealing that moves a
  random amount of probability mass from a *donor* index to a *receiver*.
  Moves are accepted with a Metropolis rule and the convolution is
  updated incrementally (O(n) per move).  Periodic full recomputation
  prevents drift.

The routine keeps the best vector seen so far and performs occasional
restarts when progress stalls.  All operations respect the security
checks of `evaluate_sequence`.  When the time budget is exhausted the

The implementation is fully self‑contained, uses only the standard

import numpy as np

# Helper utilities – top‑level (no closures)

def _next_power_of_two(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _incremental_move(
    a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution incrementally.  Returns the new (a, conv) pair.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # self‑convolution contribution of the donor (subtract) and the
    # receiver (add).  Both updates are O(n).
    conv_new = conv.copy()
    conv_new[i : i + n] -= 2.0 * delta * a
    conv_new[j : j + n] += 2.0 * delta * a

    # quadratic ‑Δ² terms (exact correction)
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < conv_new.size:
        conv_new[ii] += delta * delta
    if jj < conv_new.size:
        conv_new[jj] += delta * delta
    if ij < conv_new.size:
        conv_new[ij] -= 2.0 * delta * delta

    return a_new, conv_new


def _smooth_max_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """Soft‑max weights for a surrogate of max(conv) with temperature eps."""
    # stable soft‑max
    c_max = conv.max()
    w = np.exp((conv - c_max) / eps)
    w_sum = w.sum()
    if w_sum == 0.0:                # should never happen, but be safe
        w = np.zeros_like(conv)
        w[int(conv.argmax())] = 1.0
        w_sum = 1.0
    return w / w_sum


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate (using weights ``w``) with
    respect to the probability vector ``a``.
    """
    n = a.size
    fft_len = _next_power_of_two(3 * n - 1)

    a_rev = a[::-1]                       # reversed a
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    conv_full = np.fft.irfft(fw * fa, n=fft_len)[: 3 * n - 1]

    # entries n‑1 … 2n‑2 correspond to the gradient components
    return 2.0 * conv_full[n - 1 : n - 1 + n]


def _entropy_mirror_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient (entropy‑mirror) update on the simplex."""
    factor = np.exp(-eta * grad)
    # clamp extreme values to avoid under/overflow
    factor = np.clip(factor, 1e-40, 1e40)
    a_new = a * factor
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / a_new.size
    else:
        a_new /= s
    return a_new


def _random_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a modest log‑normal perturbation to a random block of
    entries and re‑normalise.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)

    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

            w = _smooth_max_weights(conv, eps)
            grad = _gradient_wrt_a(a, w)
            a = _entropy_mirror_update(a, grad, eta)

            # (re)compute convolution (still cheap – O(n log n))
            conv = _self_convolution(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()

        else:
            # ----------------------------------------------------------
            # Phase 2 – stochastic mass‑transfer annealing
            # ----------------------------------------------------------
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # Choose a donor index roughly proportional to its contribution
            # to the current peak of the convolution.
            peak_idx = int(conv.argmax())
            partners = peak_idx - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            if total_contrib > 0.0:
                donor = rng.choice(np.arange(n), p=contrib / total_contrib)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                continue  # nothing to move

            # Prefer giving mass to a relatively empty index
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            max_delta = 0.45 * a[donor]          # keep enough mass at donor
            delta = rng.random() * max_delta

            a_new, conv_new = _incremental_move(a, conv, donor, recipient, delta)
            new_score = 2.0 * n * conv_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a, conv, cur_score = a_new, conv_new, new_score
                move_counter += 1

                if move_counter % RECOMPUTE_EVERY == 0:
                    conv = _self_convolution(a, fft_len_self)
                    cur_score = 2.0 * n * conv.max()

            # Occasional global jitter (≈2 % of iterations)
            if rng.random() < 0.02:
                a = _random_perturb(a, rng)
                conv = _self_convolution(a, fft_len_self)
                cur_score = 2.0 * n * conv.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Restart if we have been idle for a while
        # --------------------------------------------------------------
        if now - last_improve > 22.0:
            a = best.copy()
            # modest log‑normal jitter around the best point
            jitter = np.exp(rng.normal(0.0, 0.06, size=n))
            a *= jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _self_convolution(a, fft_len_self)
            cur_score = 2.0 * n * conv.max()
            last_improve = now
            move_counter = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence, clipped to the limits enforced by
    #    ``evaluate_sequence`` for safety.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
