assistantfinal```python
"""
Propose a step‑function that minimises ``evaluate_sequence``.

The algorithm combines two ideas from the literature:

1.  **Linear‑programming smoothing** (the “f‑g update” from the statement).
    For a current distribution *f* we compute its self‑convolution peak M.
    The LP

        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k
                   g_i ≥ 0

    returns a vector *g* that spreads the mass as much as possible while
    respecting the current peak bound.  After normalising *g* to have unit
    sum we mix it into *f* with a decreasing step size and re‑project onto
    the probability simplex.

2.  **Local stochastic refinement** – a short simulated‑annealing style
    fine‑tuning that moves a tiny amount of mass from a “donor’’ (an index
    that contributes heavily to the current peak) to a “recipient’’ that
    contributes little.  The convolution is updated incrementally, and a
    Metropolis rule accepts improving or slightly worsening moves.

Several independent restarts with different lengths *n* and random
initialisations are performed until the time budget (minus a 5 s safety
margin) expires.  The best sequence seen is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _eval_score(a: np.ndarray) -> float:
    """
    Returns the value of ``evaluate_sequence`` for a vector with unit sum.
    (The function itself is homogeneous, so fixing the sum to 1 is safe.)
    """
    n = a.size
    c = np.convolve(a, a)
    return float(2.0 * n * c.max())


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ g = f * g  (linear convolution).
    Each row k contains the coefficients f_i that multiply g_{k‑i}.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for k in range(rows):
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # i runs over the support of f that participates in position k
        for i in range(i_start, i_end + 1):
            j = k - i
            A[k, j] = f[i]
    return A


def _lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_i   subject to   (f * g)_k ≤ M,   g_i ≥ 0.
    Returns the optimal g (may be zero if the LP fails).
    """
    A = _build_conv_matrix(f)          # shape (2n‑1, n)
    rows = A.shape[0]
    b = np.full(rows, M, dtype=np.float64)

    # maximise Σg_i  →  minimise −Σg_i
    c = -np.ones(f.size, dtype=np.float64)

    bounds = [(0.0, None)] * f.size
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success or res.x is None:
        # a safe fallback – return a tiny uniform vector
        return np.full(f.size, 1.0 / f.size, dtype=np.float64)
    return res.x


def _incremental_move(a: np.ndarray, conv: np.ndarray,
                      i: int, j: int, delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` (donor) to index ``j`` (recipient)
    and update the convolution *conv* incrementally.
    Returns the updated (a, conv).
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    conv_new = conv.copy()
    # removing contribution of donor
    conv_new[i:i + n] -= 2.0 * delta * a
    # adding contribution of recipient
    conv_new[j:j + n] += 2.0 * delta * a

    # quadratic correction terms (see the reference code)
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < conv_new.size:
        conv_new[ii] += delta * delta
    if jj < conv_new.size:
        conv_new[jj] += delta * delta
    if ij < conv_new.size:
        conv_new[ij] -= 2.0 * delta * delta

    return a_new, conv_new


def _stochastic_refine(a: np.ndarray, rng: np.random.Generator,
                       steps: int = 5_000) -> np.ndarray:
    """
    A short simulated‑annealing style refinement that attempts to lower the
    convolution peak.  The temperature schedule is very aggressive because the
    number of steps is small – it is only a final polish.
    """
    n = a.size
    conv = np.convolve(a, a)
    cur_score = 2.0 * n * conv.max()
    temp_start = 0.01
    temp_end = 1e-6

    for s in range(steps):
        # linearly cooled temperature
        frac = s / steps
        temp = temp_start * (1.0 - frac) + temp_end * frac

        k_max = int(conv.argmax())
        # contribution of each index to the current peak
        partners = k_max - np.arange(n)
        mask = (partners >= 0) & (partners < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[mask] * a[partners[mask]]
        total = contrib.sum()
        if total <= 0.0:
            break
        donor = rng.choice(np.arange(n), p=contrib / total)

        if a[donor] < 1e-12:
            continue

        # favour small entries as recipients
        recv_weights = 1.0 - a
        recv_weights[donor] = 0.0
        if recv_weights.sum() == 0.0:
            recipient = (donor + rng.integers(1, n)) % n
        else:
            recipient = rng.choice(np.arange(n), p=recv_weights / recv_weights.sum())

        max_delta = 0.5 * a[donor]
        if max_delta <= 0.0:
            continue
        delta = rng.random() * max_delta

        a_new, conv_new = _incremental_move(a, conv, donor, recipient, delta)
        new_score = 2.0 * n * conv_new.max()

        # Metropolis acceptance
        if new_score < cur_score or rng.random() < math.exp(-(new_score - cur_score) / temp):
            a, conv, cur_score = a_new, conv_new, new_score

    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    # ------------------------------------------------------------------
    # 0) start from the previously discovered candidate, if present
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float("inf")
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        if cand.size > 0:
            cand = _project_simplex(cand)
            sc = _eval_score(cand)
            best_seq, best_score = cand, sc
            print(f"[{time.monotonic() - start_time:.2f}s] "
                  f"Starting best from global variable: score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Search parameters
    # ------------------------------------------------------------------
    # lengths to explore – small enough to keep the LP cheap, large enough
    # to benefit from the 2n/(2n‑1) → 1 theoretical trend.
    lengths = [256, 512, 1024, 2048]
    # adapt the list if time is already low
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # number of independent restarts per length
        restarts = 2
        for r in range(restarts):
            if time.monotonic() >= deadline:
                break

            # ---- initialise a random probability vector -----------------
            a = rng.random(n)
            a = _project_simplex(a)

            # ---- main LP‑mixing loop ---------------------------------
            max_lp_iters = 15                     # a modest number – LP solves dominate time
            for it in range(max_lp_iters):
                if time.monotonic() >= deadline:
                    break

                # current peak
                conv = np.convolve(a, a)
                M = conv.max()

                # solve the LP
                g0 = _lp_max_sum(a, M)
                sum_g0 = g0.sum()
                if sum_g0 <= 0.0:
                    break
                # normalise to unit sum (the algorithm in the statement rescales
                # after the LP – the exact factor does not affect the mixing)
                g = g0 / sum_g0

                # decreasing mixing coefficient – early steps are aggressive
                t = 0.5 * (0.8 ** it)
                a = (1.0 - t) * a + t * g
                a = _project_simplex(a)          # keep unit sum and non‑negative

            # ---- short stochastic polishing ---------------------------
            a = _stochastic_refine(a, rng, steps=2000)

            # ---- evaluate candidate ------------------------------------
            sc = _eval_score(a)
            if sc < best_score - 1e-12:
                best_score, best_seq = sc, a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] New global best = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 2) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector of length 256 (never happens in practice)
        best_seq = _project_simplex(np.ones(256))

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
