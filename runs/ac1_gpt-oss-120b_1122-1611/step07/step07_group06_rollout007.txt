def max_conv_fft(a):
    n = len(a)
    fft_len = 1 << ((2*n-1).bit_length())  # minimal power of 2 >= 2n-1
    a_fft = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(a_fft * a_fft, fft_len)[:2*n-1]
    return float(conv.max())

"""
Simulated‑annealing + targeted‑transfer optimiser for the self‑convolution
bound.

* Initialise from the best known construction (`height_sequence_1`) if it
  exists, otherwise fall back to a uniform probability vector of a power‑of‑2
  length.

* During the allotted time we repeatedly propose a new candidate by one of
  three cheap perturbations:
    – a random local transfer of a small fraction of mass from one index to another
    – a multiplicative log‑normal jitter on a random block
    – a *targeted* move that shifts mass away from the indices that contribute
      most to the current maximal convolution entry (the exact sub‑gradient of
      the “max‑conv” objective).

* A simple simulated‑annealing acceptance rule lets occasional worsening
  moves early on, helping the search escape shallow plateaus.  The temperature
  drops linearly to zero as the deadline approaches.

* The objective used for the search is the fast FFT‑based estimator
  `2·n·max(a*a) / (sum(a)²)` – the same quantity returned by
  ``evaluate_sequence`` (the sum is kept equal to 1, so the denominator is
  fixed).  Exact scores are computed only when a new best solution is found,
  keeping the run fast.

The routine stops a few seconds before the overall 1000 s budget to give
the harness a safety margin, and finally returns the best sequence encountered
(clipped to the evaluator’s allowed range).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _max_conv_fft(a: np.ndarray) -> tuple[float, int]:
    """
    Return (max_convolution, argmax_index) of the linear self‑convolution
    ``a * a`` using a real FFT.  The length of ``a`` is assumed to be modest
    (few thousand), so this is fast enough for the inner loop.
    """
    n = a.size
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.  Implements the O(n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D vector")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would be projected to zero – return uniform.
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Guard against numerical drift.
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _targeted_transfer(a: np.ndarray, max_idx: int, frac: float = 0.2) -> np.ndarray:
    """
    Move a fraction of the mass from the index that contributes most to the
    currently maximal convolution entry to the index that contributes the
    least (i.e. a cheap sub‑gradient step).  ``frac`` controls the maximal
    relative amount that may be moved.
    """
    n = a.size
    # Build the sub‑gradient of the max‑convolution entry.
    g = np.zeros_like(a)
    if max_idx < n:
        # Overlap region = [0 … max_idx]
        g[: max_idx + 1] = 2.0 * a[max_idx :: -1]
    else:
        # max_idx >= n: overlap region = [max_idx‑(n‑1) … n‑1]
        start = max_idx - (n - 1)
        length = n - start
        g[start:] = 2.0 * a[:length][::-1]

    # Source: index with largest gradient (most responsible for the peak)
    src = int(np.argmax(g))
    # Destination: index with smallest gradient (ideally zero)
    dst = int(np.argmin(g))

    if src == dst or a[src] == 0.0:
        return a  # nothing to do

    # Transfer up to ``frac`` of the source mass.
    eps = a[src] * np.random.random() * frac
    new_a = a.copy()
    new_a[src] -= eps
    new_a[dst] += eps
    # Re‑normalise (numerical drift may appear).
    return _project_onto_simplex(new_a)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence`` within the supplied time budget.  The routine returns
    the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the surrounding harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all():
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # default length – a power of two is convenient for FFTs
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)  # uniform start
    else:
        a = a0.astype(np.float64).copy()
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a[:] = 1.0 / a.size
        n = a.size

    # ------------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())          # exact score
    best_time = time.monotonic()
    print(f"[0.00 s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Simulated‑annealing parameters
    # ------------------------------------------------------------------
    # Temperature (linear cooling) – start at a few percent of the current score.
    init_temp = max(0.02 * best_score, 1e-4)
    # Move‑size parameters
    block_sigma = 0.30          # STD of log‑normal block jitter
    transfer_frac = 0.25       # max fraction of mass moved in a local transfer
    targeted_frac = 0.20       # max fraction moved in a targeted transfer
    # Frequencies of the three move types
    prob_local  = 0.30
    prob_block  = 0.30
    prob_target = 0.40

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    start_time = time.monotonic()
    cur_a = a.copy()
    cur_val, cur_max_idx = _max_conv_fft(cur_a)          # fast surrogate score
    cur_score = 2.0 * n * cur_val                       # since sum(cur_a) == 1

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)        # 0 → 1

        # Linear temperature schedule
        temperature = init_temp * (1.0 - frac)

        # --------------------------------------------------------------
        # Propose a new candidate
        # --------------------------------------------------------------
        move_type = rng.random()
        if move_type < prob_local:          # ---- simple local transfer ----
            i = rng.integers(0, n)
            j = rng.integers(0, n - 1)
            while j == i:
                j = rng.integers(0, n - 1)
            eps = cur_a[i] * rng.random() * transfer_frac
            cand = cur_a.copy()
            cand[i] -= eps
            cand[j] += eps
            cand = _project_onto_simplex(cand)

        elif move_type < prob_local + prob_block:   # ---- block jitter ----
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, block_sigma))
            cand = cur_a.copy()
            cand[start:start + width] *= factor
            cand = np.clip(cand, 0.0, None)
            cand = _project_onto_simplex(cand)

        else:                                 # ---- targeted transfer (sub‑gradient) ----
            cand = _targeted_transfer(cur_a, cur_max_idx, frac=targeted_frac)

        # --------------------------------------------------------------
        # Evaluate candidate (fast surrogate)
        # --------------------------------------------------------------
        cand_val, cand_max_idx = _max_conv_fft(cand)
        cand_score = 2.0 * n * cand_val

        # --------------------------------------------------------------
        # Acceptance test (simulated annealing)
        # --------------------------------------------------------------
        delta = cand_score - cur_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # With probability exp(-Δ / T) we accept a worsening move.
            if temperature > 0 and rng.random() < np.exp(-delta / temperature):
                accept = True

        if accept:
            cur_a = cand
            cur_score = cand_score
            cur_val = cand_val
            cur_max_idx = cand_max_idx

            # ---------------------------------------------------------
            # Record best solution (using the exact evaluator)
            # ---------------------------------------------------------
            if cand_score < best_score - 1e-12:
                # compute exact score for the new incumbent
                exact = evaluate_sequence(cand.tolist())
                if exact < best_score:
                    best_score = exact
                    best = cand.copy()
                    best_time = now
                    print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # Periodic progress report
        # --------------------------------------------------------------
        iteration += 1
        if iteration % 2000 == 0:
            print(f"[{elapsed:6.2f}s] iter={iteration}, cur={cur_score:.6f}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence found (clipped for safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
