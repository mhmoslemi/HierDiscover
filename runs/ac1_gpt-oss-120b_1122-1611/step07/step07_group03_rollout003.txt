def _mass_transfer(a, grad, rng):
    # Choose source i with probabilities proportional to grad
    prob = grad.copy()
    total = prob.sum()
    if total > 0:
        prob = prob / total
        i = rng.choice(len(a), p=prob)
    else:
        i = rng.integers(len(a))
    # Choose destination j randomly
    j = rng.integers(len(a))
    if i == j:
        return a  # no change
    # Determine transfer amount δ: sample fraction f ~ Beta(2,2) or random small
    max_transfer = a[i]
    if max_transfer <= 0:
        return a
    # sample delta fraction between 0 and 0.5 maybe
    f = rng.random() * 0.5 # up to 50% of source
    delta = f * max_transfer
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # Clip small negatives
    a_new = np.clip(a_new, 0.0, None)
    # Renormalize sum to 1 (to avoid rounding error)
    s = a_new.sum()
    if s == 0:
        a_new[:] = 1.0 / len(a_new)
    else:
        a_new /= s
    return a_new
```

def _exponentiated_gradient_step(a, n, fft_len_self, steps=2000, alpha=30.0, eta=0.1):
    for _ in range(steps):
        # compute conv
        fa = np.fft.rfft(a, fft_len_self)
        conv = np.fft.irfft(fa * fa, fft_len_self)[:2*n-1]
        max_val = conv.max()
        # compute softmax weights
        w = np.exp(alpha * (conv - max_val))
        w /= w.sum()
        # gradient
        # conv of a and w
        fft_len_grad = 1 << ((len(a) + len(w) - 2).bit_length())
        fa = np.fft.rfft(a, fft_len_grad)
        fw = np.fft.rfft(w, fft_len_grad)
        grad_full = np.fft.irfft(fa * fw, fft_len_grad)
        grad = 2.0 * grad_full[n-1:2*n-1]
        # update
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -700, 700)
        factor = np.exp(log_factor)
        a = a * factor
        a = np.clip(a, 0, None)
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0 / len(a))
        else:
            a /= s
    return a
```

"""
Simulated‑annealing + gradient‑informed mass‑transfer optimisation.

The algorithm works on a non‑negative vector `a` (normalised to sum = 1)
and repeatedly proposes cheap local moves:
* a mass transfer from a source index to a destination index,
  where the source is biased by the gradient of the current
  peak of the self‑convolution;
* a multiplicative log‑normal perturbation of a random contiguous block;
* a tiny global jitter.

Moves are accepted according to the Metropolis rule with a
temperature that decays exponentially from 0.1 to 1e‑6.
Every 30 s without improvement a gentle random restart around the best
solution so far is performed.  After the annealing phase a short
exponentiated‑gradient polishing stage (smooth‑max mirror descent)
is executed to fine‑tune the result.

All heavy work (self‑convolution and its maximum) is performed with an
FFT of length a power‑of‑two, giving O(n log n) per iteration.
The routine respects the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_max(a: np.ndarray, fft_len: int):
    """
    Linear self‑convolution a*a using an FFT of size ``fft_len``.
    Returns a tuple ``(max_value, argmax_index)`` of the full convolution.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    conv = conv[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx


def _gradient_from_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the (unsmoothed) peak ``conv[k_max]`` with respect to ``a``.
    For a linear convolution the derivative is 2·a[k_max‑i] when the index
    is in range, otherwise 0.
    """
    n = a.size
    grad = np.zeros_like(a)
    # valid i satisfy 0 ≤ k_max‑i < n  ⇔  k_max‑(n‑1) ≤ i ≤ k_max
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    i = np.arange(lo, hi + 1, dtype=np.int64)
    grad[i] = 2.0 * a[k_max - i]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _mass_transfer_move(a: np.ndarray,
                        grad: np.ndarray,
                        rng: np.random.Generator,
                        temperature: float) -> np.ndarray:
    """
    Transfer a fraction of mass from a source index (biased by ``grad``)
    to a uniformly chosen destination.  The amount moved scales with the
    current temperature so that early moves can be large.
    """
    n = a.size
    # choose source i with probability proportional to gradient magnitude
    prob = grad.copy()
    prob_sum = prob.sum()
    if prob_sum > 0:
        prob /= prob_sum
        i = int(rng.choice(n, p=prob))
    else:
        i = rng.integers(n)

    j = rng.integers(n)
    if i == j:
        return a  # nothing to do

    # maximum transferable fraction (temperature‑driven)
    max_frac = min(0.5, temperature / 0.1)        # at start up to 0.5, later tiny
    if max_frac <= 0.0:
        return a

    delta_frac = rng.random() * max_frac
    delta = delta_frac * a[i]

    if delta <= 0.0:
        return a

    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta
    # numerical safety
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new


def _objective(a: np.ndarray, fft_len: int) -> float:
    """
    Exact value of the evaluation function (without the clipping that
    ``evaluate_sequence`` performs).  The sum of ``a`` is kept >0.01.
    """
    n = a.size
    max_b, _ = _self_conv_max(a, fft_len)
    sum_a = a.sum()
    # sum_a is never zero in our runs
    return 2.0 * n * max_b / (sum_a * sum_a)


def _smooth_max_mirror_descent(a: np.ndarray,
                               n: int,
                               fft_len_self: int,
                               steps: int = 2000,
                               alpha: float = 30.0,
                               eta: float = 0.1) -> np.ndarray:
    """
    A short polishing phase: exponentiated‑gradient (smooth‑max) updates.
    The update shrinks entries that contribute most to large convolution
    values, flattening the autocorrelation.
    """
    fft_len_grad = _next_pow2(3 * n - 2)

    for _ in range(steps):
        # self‑convolution
        fa = np.fft.rfft(a, fft_len_self)
        conv = np.fft.irfft(fa * fa, fft_len_self)[: 2 * n - 1]
        c_max = conv.max()

        # soft‑max weights (stable)
        w = np.exp(alpha * (conv - c_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient = 2·a∗w (slice the middle part)
        fa = np.fft.rfft(a, fft_len_grad)
        fw = np.fft.rfft(w, fft_len_grad)
        grad_full = np.fft.irfft(fa * fw, fft_len_grad)
        grad = 2.0 * grad_full[n - 1: 2 * n - 1]          # length n

        # exponentiated‑gradient step on the simplex
        log_factor = -eta * grad
        log_factor = np.clip(log_factor, -700, 700)
        factor = np.exp(log_factor)

        a = a * factor
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The strategy combines simulated annealing with gradient‑informed mass
    transfers and a final smooth‑max mirror‑descent polishing phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length and initialise a on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            n = 3000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # No prior candidate – start from a Dirichlet draw
        n = 3000
        a = rng.dirichlet(np.ones(n) * 0.3).astype(np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)    # for a*a

    # ------------------------------------------------------------------
    # 3) Initial objective and bookkeeping
    # ------------------------------------------------------------------
    cur_score = _objective(a, fft_len_self)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4) Temperature schedule (exponential)
    # ------------------------------------------------------------------
    T_start = 0.12
    T_end = 1e-6

    # ------------------------------------------------------------------
    # 5) Main optimisation loop (simulated annealing)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_total = elapsed / total_time

        # --------------------------------------------------------------
        # 5.1   current temperature
        # --------------------------------------------------------------
        temperature = T_start * np.exp(np.log(T_end / T_start) * frac_total)
        temperature = max(temperature, 1e-15)

        # --------------------------------------------------------------
        # 5.2   compute current peak index (used for gradient‑biased move)
        # --------------------------------------------------------------
        _, k_max = _self_conv_max(a, fft_len_self)
        grad = _gradient_from_max(a, k_max)

        # --------------------------------------------------------------
        # 5.3   propose a move
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.60:
            a_new = _mass_transfer_move(a, grad, rng, temperature)
        elif r < 0.90:
            a_new = _random_block_perturb(a, rng)
        else:
            # tiny global jitter
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            a_new = a * noise
            a_new = np.clip(a_new, 0.0, None)
            a_new /= a_new.sum()

        new_score = _objective(a_new, fft_len_self)

        # --------------------------------------------------------------
        # 5.4   Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability of uphill move
            prob_up = np.exp(-(new_score - cur_score) / temperature)
            if rng.random() < prob_up:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score

            # ----------------------------------------------------------
            # 5.5   keep the best solution
            # ----------------------------------------------------------
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.6   soft restart after stagnation (≈30 s without improvement)
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            # add a gentle log‑normal noise around the best
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective(a, fft_len_self)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final polishing with smooth‑max mirror descent
    # ------------------------------------------------------------------
    best = _smooth_max_mirror_descent(best, n, fft_len_self,
                                      steps=2500, alpha=30.0, eta=0.08)
    best_score = _objective(best, fft_len_self)
    print(f"Final polished score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
