```python
"""
Propose a candidate step sequence via Adam‑style mirror descent on the
probability simplex.  The non‑smooth max in the objective is approximated
by a soft‑max with a temperature that is raised linearly to a large value
so that the surrogate converges to the true max.  The algorithm runs a
few gradient‑descent phases on several initialisations and lengths,
periodically injects random block perturbations to escape plateaus
and keeps the best sequence seen within the allotted time budget.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex."""
    n = a.size
    # Choose a block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over the convolution vector with temperature 1/alpha."""
    c_max = conv.max()
    shifted = conv - c_max   # prevent overflow
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum

def _gradient(a: np.ndarray, w_rev: np.ndarray, grad_len: int) -> np.ndarray:
    """Return 2·(a * w_rev) where * is linear convolution (valid part)."""
    # a length = n, w_rev length = 2n‑1, result length = 3n‑2
    fa = np.fft.rfft(a, n=grad_len)
    fw = np.fft.rfft(w_rev, n=grad_len)
    conv_full = np.fft.irfft(fa * fw, n=grad_len)
    offset = w_rev.size - 1          # = 2n‑2
    n = a.size
    # slice the central n entries
    grad = 2.0 * conv_full[offset: offset + n]
    return grad

def _optimize(initial_a: np.ndarray,
              start_time: float,
              deadline: float,
              rng: np.random.Generator,
              debug: bool = False):
    """Run Adam‑style mirror descent on ``initial_a`` (already on the simplex)."""
    a = initial_a.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    grad_len = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # bookkeeping of best solution seen in this run
    conv = _self_conv(a, fft_len_self)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / max(1e-12, (deadline - start_time))

        # learning‑rate schedule (exponential‑decay like)
        eta = 0.5 * (1.0 - frac) + 0.001 * frac

        # temperature schedule – ramp up fast, then stay high
        alpha_max = 500.0
        if frac < 0.9:
            alpha = 5.0 + (alpha_max - 5.0) * (frac / 0.9)
        else:
            alpha = alpha_max

        # ----- self‑convolution -----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]

        # ----- soft‑max weights (approx. max) -----
        w = _softmax_weights(conv, alpha)

        # ----- gradient -----
        w_rev = w[::-1]
        grad = _gradient(a, w_rev, grad_len)

        # ----- Adam update (additive) -----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ----- project onto simplex -----
        a = proj_simplex(a)

        # ----- evaluate current objective -----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            if debug:

            m.fill(0.0)

    # final safety clipping (evaluate_sequence will also clip)
    best_a = np.clip(best_a, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_a.tolist())
    except Exception:   # evaluate_sequence may not be importable in offline test
        final_score = float('inf')
    return best_a, final_score

            v /= v.sum()
            starts.append(v)

        v /= v.sum()
        starts.append(v)

        v = np.zeros(n, dtype=np.float64)

"""
Propose a candidate step function via Adam‑style mirror descent on the
simplex.  The non‑smooth max in the objective is approximated by a
soft‑max whose temperature is raised linearly to a large value, turning
the surrogate into the true max.  Several initialisations (different
lengths, geometric/linear/Dirichlet starts) are optimised, occasional
random block perturbations are injected to escape plateaus, and the best
sequence seen before the deadline is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]          # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]           # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max over a convolution vector with temperature 1/α."""
    c_max = conv.max()
    shifted = conv - c_max               # numerical stability
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _gradient(a: np.ndarray, w_rev: np.ndarray, grad_len: int) -> np.ndarray:
    """Compute 2·(a * w_rev) where * is linear convolution (valid part)."""
    # a length = n, w_rev length = 2n‑1, full convolution length = 3n‑2
    fa = np.fft.rfft(a, n=grad_len)
    fw = np.fft.rfft(w_rev, n=grad_len)
    conv_full = np.fft.irfft(fa * fw, n=grad_len)
    offset = w_rev.size - 1          # = 2n‑2
    n = a.size
    grad = 2.0 * conv_full[offset: offset + n]
    return grad


def _optimize(initial_a: np.ndarray,
              start_time: float,
              deadline: float,
              rng: np.random.Generator,
              debug: bool = False):
    """
    Adam‑style mirror descent on the probability simplex.
    Returns the best vector found and its exact score (via evaluate_sequence).
    """
    a = initial_a.copy()
    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    grad_len = _next_pow2(3 * n - 2)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # best solution seen in this optimisation run
    conv = _self_conv(a, fft_len_self)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / max(1e-12, deadline - start_time)

        # learning‑rate schedule (large → small)
        eta = 0.5 * (1.0 - frac) + 0.001 * frac

        # temperature schedule – quickly rise then stay high
        alpha_max = 500.0
        if frac < 0.9:
            alpha = 5.0 + (alpha_max - 5.0) * (frac / 0.9)
        else:
            alpha = alpha_max

        # ---- self‑convolution ----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]

        # ---- soft‑max weights (smooth max) ----
        w = _softmax_weights(conv, alpha)

        # ---- gradient (ignoring ∂w/∂a) ----
        w_rev = w[::-1]
        grad = _gradient(a, w_rev, grad_len)

        # ---- Adam update (additive) ----
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)
        step = eta * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step

        # ---- project onto simplex ----
        a = proj_simplex(a)

        # ---- evaluate current objective ----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            if debug:
                print(f"[iter {t_iter}] new best {best_score:.6f}")

        # ---- occasional jitter to inject noise ----
        if rng.random() < 0.015:
            a = _random_block_perturb(a, rng)
            m.fill(0.0)
            v.fill(0.0)

        # ---- restart after stagnation (>20 s) ----
        if now - last_improve > 20.0:
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            a = a * noise
            a = proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            if debug:
                print(f"[{now - start_time:.2f}s] restart (best {best_score:.6f})")

    # final clipping to stay inside the evaluator's safe range
    best_a = np.clip(best_a, 0.0, 1000.0)

    # obtain the exact score via the provided evaluator (optional safety)
    try:
        final_score = evaluate_sequence(best_a.tolist())
    except Exception:               # evaluate_sequence may be unavailable in some local tests
        final_score = float('inf')
    return best_a, final_score
