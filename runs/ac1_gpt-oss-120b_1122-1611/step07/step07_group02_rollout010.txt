"""
Search for a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  The method combines a smooth LSE surrogate for the
max‑convolution with a limited‑memory BFGS optimiser and a final short Adam
refinement.  The surrogate temperature (β) is increased in stages so the
optimiser first sees a smooth landscape and then focuses on the true maximum.
The algorithm respects the 1000 s time‑budget and works for any length `n`
(the default is 5000 or the length of a previously discovered best
construction).
"""

import time
import numpy as np
from scipy import special, optimize

# ----------------------------------------------------------------------
# Helper utilities – no closures, top‑level only
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    e = np.exp(logits - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return e / s


def _smooth_max_loss_and_grad(x: np.ndarray,
                              n: int,
                              beta: float,
                              eps: float = 1e-12) -> tuple[float, np.ndarray]:
    """
    Smooth approximation of 2·n·max(conv) using a log‑sum‑exp surrogate.
    Returns (loss, grad) where the gradient is taken w.r.t. the logits `x`
    (the softmax of `x` yields the probability vector `a` that sums to one).
    """
    # a = softmax(x) – the current probability vector
    a = _softmax_logits(x)

    # ---- convolution a*a (via FFT) ------------------------------------
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

    # ---- smooth max via log‑sum‑exp ------------------------------------
    scaled = beta * conv
    lse = special.logsumexp(scaled)          # log‑sum‑exp of β·conv
    loss = lse / beta                        # smooth approximation of max

    # ---- gradient of the surrogate w.r.t. a ---------------------------
    # softmax weights over the convolution entries
    w = np.exp(scaled - lse)                 # shape (2n‑1,); sums to 1

    # correlation w ⋆ a gives ∂L/∂a (see derivation in the docstring)
    fft_len_corr = _next_pow2(3 * n - 2)
    w_fft = np.fft.rfft(w, n=fft_len_corr)
    a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
    corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)

    # valid part of the correlation has length n, centred at index n‑1
    start_idx = n - 1
    grad_a = 2.0 * corr_full[start_idx:start_idx + n]   # ∂L/∂a

    # ---- chain rule: logits → probabilities ---------------------------
    dot = np.dot(grad_a, a)                # ⟨grad_a, a⟩
    grad_x = a * (grad_a - dot)            # ∂L/∂x

    return float(loss), grad_x.astype(np.float64)


def _evaluate_exact(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv)/(sum(a)**2) – used for logging."""
    conv = np.convolve(a, a)
    maxb = conv.max()
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * maxb / (s ** 2)

    """Internal exception used to abort optimiser callbacks when time runs out."""
    pass
