def lp_improve(a, cur_score):
    n = a.shape[0]
    c_self = np.convolve(a, a)
    M = c_self.max()
    # Solve LP
    A = diags([np.full(n, a[i]) for i in range(n)], offsets=np.arange(n), shape=(2*n-1, n), format='csc')
    b = np.full(2*n-1, M)
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        return a, cur_score
    g = res.x
    # Normalize g to sum 1
    sum_g = g.sum()
    if sum_g == 0:
        return a, cur_score
    g = g / sum_g
    # compute convolutions
    conv_f_g = np.convolve(a, g)
    conv_g_g = np.convolve(g, g)
    # try mixing t in grid
    best_score = cur_score
    best_a = a
    for t in [0.0, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0]:
        comb = (1-t) * a + t * g
        conv_mix = (1-t)*(1-t) * c_self + 2*t*(1-t)*conv_f_g + t*t * conv_g_g
        score = 2 * n * conv_mix.max()
        if score < best_score:
            best_score = score
            best_a = comb
    return best_a, best_score
```

"""
Hybrid optimiser for the step‑function bound.

The algorithm consists of three stages that run sequentially inside the
time budget (≈1000 s):

1. **Weighted mirror descent** – a smooth soft‑max surrogate of the
   self‑convolution max is minimised by an exponentiated‑gradient (entropy)
   step on the probability simplex.  This gives a fast global descent.

2. **Linear‑programming refinement** – with the current vector `a` we fix
   `M = max(a*a)`.  The linear programme
        maximise   ∑ g_i
        subject to a ⊛ g ≤ M ,   g ≥ 0
   (⊛ = convolution) finds a direction `g` that can increase the total
   mass without raising the peak.  After normalising `g` to sum 1 we
   search a small mixing parameter `t∈[0,1]` and replace `a←(1‑t)a+tg`
   whenever the bound improves.  A few LP iterations typically give a
   noticeable drop.

3. **Fast stochastic refinement** – a simple Metropolis‑like mass‑transfer
   move (donor → recipient) is performed many times.  The move is accepted
   if the bound decreases or with a temperature‑controlled probability.
   Occasional global jitter prevents stagnation.

All vectors are kept non‑negative and normalised to unit sum, which matches
the expectations of `evaluate_sequence`.  The routine prints progress and
returns the best sequence found before the budget expires.
"""
import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import diags

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of two equal‑length vectors using FFT."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)[: 2 * x.size - 1]
    return conv


def _build_conv_matrix(f: np.ndarray) -> "scipy.sparse.csc_matrix":
    """
    Return the (2n‑1)×n matrix that maps a vector g to the convolution
    (f ⊛ g).  The matrix consists of the diagonals of f, i.e.
    row k, column j contains f[k‑j] when 0 ≤ k‑j < n, else 0.
    """
    n = f.size
    # each diagonal is a constant vector equal to the corresponding entry of f
    data = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)            # 0,1,…,n‑1
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the evaluation score (the quantity we minimise)."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose problem size and an initial feasible point on the simplex
    # ------------------------------------------------------------------
    # moderate size – small enough for the LP step, large enough for a
    # fine‑grained shape.
    n = 2048

    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0:
            if cand.size >= n:
                a = cand[:n].copy()
            else:
                a = np.pad(cand, (0, n - cand.size))
        else:
            a = rng.random(n)
    else:
        a = rng.random(n)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    a = a / s if s > 0 else np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    c_self = _self_conv(a, fft_len_self)
    cur_score = _score_from_conv(c_self, n)

    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # proportion of the total budget for the three phases
    phase1_frac = 0.20          # mirror descent
    phase2_frac = 0.55          # LP refinement
    phase1_end = start_time + phase1_frac * total_time
    phase2_end = phase1_end + phase2_frac * total_time

    # mirror‑descent schedules
    eta_start = 2.0
    eta_end = 1e-4
    eps_min = 1e-12

    # stochastic phase schedules
    temp_start = 0.08
    temp_end = 1e-6
    RECOMPUTE_EVERY = 400

    move_counter = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase 1 – weighted mirror descent (entropy‑mirror)
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedules
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # soft‑max temperature (scaled to the current dynamic range)
            range_c = c_self.max() - c_self.min()
            eps = max(range_c * max(0.2 * (1.0 - frac), 0.01), eps_min)

            # log‑sum‑exp (stable) → weights w_k = exp((c_k - max_c)/eps)
            max_c = c_self.max()
            w = np.exp((c_self - max_c) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:          # extremely unlikely – fall back to hard max
                w = np.zeros_like(c_self)
                w[int(c_self.argmax())] = 1.0
                w_sum = 1.0
            w = w / w_sum               # shape (2n‑1,)

            # gradient of the smooth max w.r.t. a.
            a_rev = a[::-1]             # reversed a, length n
            fw = np.fft.rfft(w, n=fft_len_grad)
            fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: 3 * n - 1]
            grad = 2.0 * conv_full[n - 1:n - 1 + n]   # length n

            # exponentiated‑gradient (entropy mirror) update
            factor = np.exp(-eta * grad)
            factor = np.clip(factor, 1e-40, 1e40)      # guard against under/overflow
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

            # recompute convolution and current score
            c_self = _self_conv(a, fft_len_self)
            cur_score = _score_from_conv(c_self, n)

        # --------------------------------------------------------------
        # Phase 2 – LP refinement (alternating maximise‑mass step)
        # --------------------------------------------------------------
        elif now < phase2_end:
            # Build the convolution matrix A from the current a (sparse)
            A = _build_conv_matrix(a)
            M = c_self.max()
            b_ub = np.full(2 * n - 1, M)

            # Linear program: maximise ∑ g_i  ↔ minimise –∑ g_i
            lp_res = linprog(
                c=-np.ones(n),
                A_ub=A,
                b_ub=b_ub,
                bounds=[(0.0, None)] * n,
                method="highs",
                options={"presolve": True},
            )

            if not lp_res.success:
                # if LP fails, fall back to a short stochastic exploration
                phase2_end = now  # skip the rest of LP phase
                continue

            g = lp_res.x
            sum_g = g.sum()
            if sum_g == 0.0:
                # degenerate – abandon LP for this round
                phase2_end = now
                continue

            # normalise g to keep the simplex invariant
            g = g / sum_g

            # Pre‑compute the three convolutions needed for mixing
            conv_f_g = _conv_fft(a, g, _next_pow2(2 * n - 1))
            conv_g_g = _self_conv(g, _next_pow2(2 * n - 1))

            # try a small set of mixing parameters and keep the best
            best_local_score = cur_score
            best_mix = a.copy()
            for t in (0.0, 0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9, 1.0):
                # convex combination
                a_mix = (1.0 - t) * a + t * g
                # convolution of the mix (quadratic expansion)
                conv_mix = ((1.0 - t) ** 2) * c_self \
                           + 2.0 * t * (1.0 - t) * conv_f_g \
                           + (t ** 2) * conv_g_g
                score_mix = _score_from_conv(conv_mix, n)
                if score_mix < best_local_score - 1e-12:
                    best_local_score = score_mix
                    best_mix = a_mix

            # accept improvement, if any
            if best_local_score < cur_score - 1e-12:
                a = best_mix
                c_self = _self_conv(a, fft_len_self)
                cur_score = best_local_score

        # --------------------------------------------------------------
        # Phase 3 – fast stochastic refinement (annealed mass transfer)
        # --------------------------------------------------------------
        else:
            # linear temperature schedule
            t_frac = (now - phase2_end) / (deadline - phase2_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # locate a current peak of the convolution
            k_max = int(c_self.argmax())

            # contribution of each index to that peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # donor – pick proportionally to its contribution (or uniform)
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                continue

            # recipient – bias towards small entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # propose a transfer
            max_delta = 0.5 * a[donor]
            delta = rng.random() * max_delta

            a_new = a.copy()
            a_new[donor] -= delta
            a_new[recipient] += delta
            c_new = _self_conv(a_new, fft_len_self)
            new_score = _score_from_conv(c_new, n)

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c_self = c_new
                cur_score = new_score
                move_counter += 1

                if move_counter % RECOMPUTE_EVERY == 0:
                    c_self = _self_conv(a, fft_len_self)
                    cur_score = _score_from_conv(c_self, n)

            # occasional global jitter (≈2 % of iterations)
            if rng.random() < 0.02:
                # multiplicative log‑normal perturbation of a random block
                width = rng.integers(1, max(2, n // 10))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.15))
                a[start:start + width] *= factor
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                c_self = _self_conv(a, fft_len_self)
                cur_score = _score_from_conv(c_self, n)

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:          # shorter than the original 25 s
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c_self = _self_conv(a, fft_len_self)
            cur_score = _score_from_conv(c_self, n)
            last_improve = now
            move_counter = 0
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
