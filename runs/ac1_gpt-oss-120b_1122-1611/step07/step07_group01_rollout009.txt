"""
Propose a step‑function that (hopefully) beats the current bound.
-----------------------------------------------------------------

The search is built around a *soft‑max* approximation of the true
objective  

    F(a) = 2·n·max_k (a*a)[k] ,

where `a` lives on the probability simplex (non‑negative, sum = 1).
For a temperature parameter β the smoothed loss  

    L_β(a) = Σ_k w_k·(a*a)[k] ,   w_k ∝ exp(β·((a*a)[k]–max(a*a)))

has gradient  

    ∇_a L_β = 2·(w * a)               (central slice of the convolution)

which is cheap to obtain with FFTs.

We perform **mirror descent on the simplex** by optimising the
log‑probabilities `z = log a` with an Adam optimiser; the
gradient on `z` follows from the chain‑rule:

    ∂L/∂z = a * (∇_a L – ⟨∇_a L, a⟩) .

The schedule slowly raises β (0.5 → 5 000) so that the loss
focuses on the true maximum, while the learning‑rate decays
(0.2 → 1e‑3).  To avoid stagnation we occasionally jitter the current
vector with a log‑normal block perturbation and restart after a long
period without improvement.

Multiple target lengths are tried (the length of the previously best
construction and a few alternatives).  The best sequence found before
the time budget expires is returned.

The implementation uses only NumPy, no external solvers, and stays
well within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine runs until the time budget
    expires (a 5 s safety margin is kept).  The best sequence found is
    returned as a plain list of Python floats.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_total = time.monotonic()
    deadline = start_total + max(0.0, float(budget_s) - 5.0)
    total_budget = deadline - start_total

    # ------------------------------------------------------------------
    # 1) Choose a set of candidate lengths
    # ------------------------------------------------------------------
    # length of the previously discovered best sequence (if any)
    base_len = None
    if "height_sequence_1" in globals():
        base_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if base_seq.ndim == 1 and base_seq.size > 0 and np.isfinite(base_seq).all():
            base_len = int(base_seq.size)

    # Fixed list – we always test the previous best length and a few
    # alternatives.  The numbers are chosen to stay well within the
    # allowed runtime.
    lengths = []
    if base_len is not None:
        lengths.append(base_len)
    # other lengths to explore (roughly the same order of magnitude)
    for L in (2000, 3000, 3800, 5000, 6000):
        if L != base_len:
            lengths.append(L)

    # ------------------------------------------------------------------
    # 2) Global bookkeeping
    # ------------------------------------------------------------------
    best_global_seq = None
    best_global_score = float("inf")

    # ------------------------------------------------------------------
    # 3) Main search over the chosen lengths
    # ------------------------------------------------------------------
    for n in lengths:
        # Check remaining time – break early if we cannot afford a full
        # run for another length.
        now = time.monotonic()
        if now >= deadline - 2.0:
            break

        # --------------------------------------------------------------
        # 3.1 – initialise a probability vector a on the simplex
        # --------------------------------------------------------------
        if best_global_seq is not None and best_global_seq.size == n:
            a = best_global_seq.copy()
        elif base_len == n and base_len is not None:
            # use the previously known construction (already normalised)
            a = np.clip(base_seq, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / n)
        else:
            # start from uniform distribution
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # --------------------------------------------------------------
        # 3.2 – pre‑compute FFT lengths for the two convolutions we need
        # --------------------------------------------------------------
        fft_len_self = _next_pow2(2 * n - 1)          # for a*a
        fft_len_wa   = _next_pow2(3 * n)              # for w*a

        # --------------------------------------------------------------
        # 3.3 – Adam state on the log‑probabilities (mirror descent)
        # --------------------------------------------------------------
        z = np.log(a + 1e-12)              # work in log‑space (soft‑max)
        m = np.zeros_like(z)                # 1st‑moment estimate
        v = np.zeros_like(z)                # 2nd‑moment estimate
        beta1 = 0.9
        beta2 = 0.999
        eps_adam = 1e-8
        step = 0

        # --------------------------------------------------------------
        # 3.4 – local best for this length
        # --------------------------------------------------------------
        cur_score = 2.0 * n * _self_conv(a, fft_len_self).max()
        best_seq_len = a.copy()
        best_score_len = cur_score
        last_improve = time.monotonic()

        # --------------------------------------------------------------
        # 3.5 – optimisation loop for this length
        # --------------------------------------------------------------
        while time.monotonic() < deadline:
            now = time.monotonic()
            elapsed_total = now - start_total
            frac = min(1.0, elapsed_total / total_budget)      # 0 → 1

            # ---- schedule -------------------------------------------------
            # learning‑rate for Adam, decays from 0.2 to 1e‑3
            eta = 0.2 * (1.0 - frac) + 0.001 * frac
            # soft‑max temperature, grows from 0.5 to 5 000
            beta = 0.5 * (1.0 - frac) + 5000.0 * frac

            # ---- self‑convolution and soft‑max weights --------------------
            c = _self_conv(a, fft_len_self)          # length 2n‑1
            max_c = c.max()
            w_raw = np.exp(beta * (c - max_c))       # shift by max_c → ≤1
            w = w_raw / w_raw.sum()                  # probabilities, length 2n‑1

            # ---- gradient of the smoothed objective ----------------------
            conv_wa = _conv_fft(w, a, fft_len_wa)    # length 3n‑2
            grad_a = 2.0 * conv_wa[n - 1:n - 1 + n]   # slice of length n

            # ---- transform gradient to the log‑probability space ---------
            dot = np.dot(grad_a, a)
            grad_z = a * (grad_a - dot)

            # ---- Adam update ---------------------------------------------
            step += 1
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
            m_hat = m / (1.0 - beta1 ** step)
            v_hat = v / (1.0 - beta2 ** step)
            z = z - eta * m_hat / (np.sqrt(v_hat) + eps_adam)

            # ---- map back to the simplex ----------------------------------
            max_z = np.max(z)                # numerical stability
            exp_z = np.exp(z - max_z)
            a = exp_z / exp_z.sum()

            # ---- occasional random block jitter (exploration) ------------
            if rng.random() < 0.002:         # ~0.2 % of iterations
                a = _random_block_perturb(a, rng)
                # keep Adam state consistent with the new point
                z = np.log(a + 1e-12)

            # ---- evaluate the *exact* objective every few steps ----------
            if step % 5 == 0:
                cur_score = 2.0 * n * max_c
                if cur_score < best_score_len - 1e-12:
                    best_score_len = cur_score
                    best_seq_len = a.copy()
                    last_improve = now
                    print(f"[{now - start_total:6.2f}s] "
                          f"len={n:5d} new best = {best_score_len:.6f}")

            # ---- restart after stagnation --------------------------------
            if now - last_improve > 25.0:
                # jitter the current best and continue
                a = best_seq_len.copy()
                jitter = np.exp(rng.normal(0.0, 0.12, size=n))
                a = a * jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                z = np.log(a + 1e-12)
                m.fill(0.0)
                v.fill(0.0)
                step = 0
                last_improve = now
                print(f"[{now - start_total:6.2f}s] len={n:5d} restart – "
                      f"best {best_score_len:.6f}")

            # --------------------------------------------------------------
            # check if we have exhausted the global budget
            # --------------------------------------------------------------
            if time.monotonic() >= deadline:
                break

        # --------------------------------------------------------------
        # 3.6 – update the global best
        # --------------------------------------------------------------
        if best_score_len < best_global_score:
            best_global_score = best_score_len
            best_global_seq = best_seq_len.copy()
            print(f"=== Global new best after length {n}: {best_global_score:.6f} ===")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    if best_global_seq is None:
        # Very unlikely – fall back to a short uniform vector
        best_global_seq = np.full(10, 0.1, dtype=np.float64)

    best_global_seq = np.clip(best_global_seq, 0.0, 1000.0)
    return [float(x) for x in best_global_seq]
