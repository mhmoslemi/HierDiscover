"""
Differential‑Evolution optimiser for the min‑max self‑convolution problem.

We minimise
    J(a) = 2·n·max(a * a) / (sum(a)**2)

Because the objective is homogeneous we keep every candidate on the
probability simplex (sum = 1, a_i ≥ 0).  The algorithm maintains a small
population, creates mutants by the classic DE/rand/1 scheme and performs
binomial crossover.  After mutation / crossover each trial vector is
projected back onto the simplex.  The fitness is the exact value returned
by ``evaluate_sequence`` for a normalised candidate, i.e.

    score(a) = 2·n·max(conv(a,a))

The search runs until the given time budget expires (a 5 s safety margin
is kept) and the best sequence found is returned as a plain Python list.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (sum=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]        # last index where condition holds
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective value for a candidate that is already normalised."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _initial_population(n: int, popsize: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse starting population (all vectors sum to 1)."""
    pop = []

    # uniform vector
    pop.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few exponential decays
    for rate in (0.99, 0.975, 0.95, 0.9):
        v = rate ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        pop.append(v)

    # Dirichlet‑like draws using exponential sampling (fast)
    while len(pop) < popsize:
        v = rng.exponential(scale=1.0, size=n)
        v_sum = v.sum()
        if v_sum == 0.0:
            v[:] = 1.0 / n
        else:
            v /= v_sum
        pop.append(v)

    # if we have too many vectors truncate
    return pop[:popsize]


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Differential‑Evolution search for a step‑function with a low
    ``evaluate_sequence`` score.  Returns the best sequence found before the
    time limit.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time budget (leave 5 s margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) decide the problem size
    # ------------------------------------------------------------------
    # If a previous construction is available, reuse its length as a
    # baseline; otherwise start with a moderately large length.
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base_n = int(prev.size)
    else:
        base_n = 0

    # A longer sequence gives a lower theoretical bound (≈1).  We pick
    # a size that still fits comfortably into the budget.
    n = max(5000, base_n)          # at least 5 000 entries
    # limit the size to keep per‑iteration cost reasonable
    n = min(n, 8000)               # cap at 8 000

    fft_len = _next_pow2(2 * n - 1)   # length for the self‑convolution FFT

    # ------------------------------------------------------------------
    # 2) initialise a population (DE)
    # ------------------------------------------------------------------
    popsize = 30
    population = _initial_population(n, popsize, rng)

    # if a previous good solution exists, insert it (projected onto the simplex)
    if "height_sequence_1" in globals():
        old = np.clip(prev, 0.0, None)
        old = proj_simplex(old)
        if old.size == n:
            population[0] = old.copy()

    scores = np.empty(popsize, dtype=np.float64)
    for i, vec in enumerate(population):
        scores[i] = compute_score(vec, fft_len, n)

    best_idx = int(np.argmin(scores))
    best = population[best_idx].copy()
    best_score = float(scores[best_idx])
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 3) main DE loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        progress = (now - start_time) / total_time   # ∈[0,1]

        # adaptation of DE parameters
        # crossover probability slowly reduces from 0.9 → 0.3
        Cr = 0.9 * (1.0 - progress) + 0.3 * progress
        # differential weight from 0.9 → 0.5
        F = 0.9 * (1.0 - progress) + 0.5 * progress

        # pre‑compute a mask of indices for fast exclusion of the target
        all_idx = np.arange(popsize)

        for i in range(popsize):
            # ------------------------------------------------------------
            # mutation: DE/rand/1
            # ------------------------------------------------------------
            # choose three distinct vectors different from i
            idxs = rng.choice(all_idx[all_idx != i], 3, replace=False)
            r1, r2, r3 = idxs
            mutant = population[r1] + F * (population[r2] - population[r3])

            # enforce non‑negativity and simplex constraint
            mutant = np.maximum(mutant, 0.0)
            mutant = proj_simplex(mutant)

            # ------------------------------------------------------------
            # crossover (binomial)
            # ------------------------------------------------------------
            cross_mask = rng.random(n) < Cr
            # guarantee at least one component from mutant
            if not np.any(cross_mask):
                cross_mask[rng.integers(0, n)] = True
            trial = np.where(cross_mask, mutant, population[i])

            # project again to be safe (numerical drift)
            trial = np.maximum(trial, 0.0)
            trial = proj_simplex(trial)

            # ------------------------------------------------------------
            # evaluation
            # ------------------------------------------------------------
            trial_score = compute_score(trial, fft_len, n)

            # ------------------------------------------------------------
            # selection
            # ------------------------------------------------------------
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score - 1e-12:
                    best_score = trial_score
                    best = trial.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------
        # stagnation handling – occasional random injection
        # ------------------------------------------------------------
        if now - last_improve > 30.0:
            # replace a fraction of the worst individuals with fresh draws
            n_replace = max(1, popsize // 5)
            worst = np.argsort(scores)[-n_replace:]
            for idx in worst:
                new_vec = rng.exponential(scale=1.0, size=n)
                new_vec /= new_vec.sum()
                population[idx] = new_vec
                scores[idx] = compute_score(new_vec, fft_len, n)
            last_improve = now
            print(f"[{elapsed:6.2f}s] stagnation restart – refreshed {n_replace} individuals")

    # ------------------------------------------------------------------
    # 4) final polishing – clip to the hard limits of evaluate_sequence
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
