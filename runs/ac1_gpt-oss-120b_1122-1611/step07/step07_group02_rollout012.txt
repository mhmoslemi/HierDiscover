"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

The algorithm is a lightweight simulated‑annealing / stochastic
local‑search that works directly on the probability vector
(a_i ≥ 0, Σ a_i = 1).  The objective to minimise is

    f(a) = 2·n·max(conv(a,a)) / (Σ a_i)²

which, because Σ a_i = 1, reduces to ``2·n·max(conv(a,a))``.
The search starts from the previously discovered construction
``height_sequence_1`` (if it exists) or from a uniform distribution.
At every iteration a tiny amount of mass is transferred between two
random positions, the new score is evaluated using an FFT based
convolution (O(n log n)), and the move is accepted either because it
improves the score or with a probability that follows an exponential
cooling schedule.  Periodic restarts from the best solution seen so
far avoid stagnation.

The routine respects the 1000 s time‑budget, prints progress updates,
and finally returns the best sequence (clipped to the limits enforced
by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fast_score(a: np.ndarray) -> float:
    """
    Compute the evaluation score for a probability vector ``a`` (Σa = 1)
    using the same formula as ``evaluate_sequence`` but without the
    security‑check steps.  The convolution is performed by FFT for speed.
    """
    n = a.shape[0]
    # length for a linear convolution via FFT
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_b = conv_full.max()
    # Σa = 1, but we keep the general formula for completeness
    sum_a = a.sum()
    if sum_a < 0.01:          # same safeguard as the official evaluator
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def _initial_vector(seed: int = 12345) -> np.ndarray:
    """
    Return an initial probability vector.  If a global
    ``height_sequence_1`` exists it is used (normalised to sum to 1);
    otherwise a uniform vector of length 3000 is returned.
    """
    rng = np.random.default_rng(seed)
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)            # protect against negatives
        s = base.sum()
        if s == 0.0:
            base[:] = 1.0 / base.size
        else:
            base /= s                               # normalise to a probability vector
        n = max(2000, base.size)                    # keep the vector reasonably large
        if base.size != n:                          # linear interpolation if we changed the size
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            base = np.interp(new_x, old_x, base)
            base = np.clip(base, 0.0, None)
            base /= base.sum()
        return base
    else:
        n = 3000
        return np.full(n, 1.0 / n, dtype=np.float64)

    T_end   = 1e-6
    total_time = deadline - start

    # typical amount of mass moved in one proposal (as a fraction of average mass)
    move_fraction = 0.02                 # ≈ 2 % of the average weight (≈ 1/n)

    # ------------------------------------------------------------------
    # Main stochastic loop
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start
        # exponential temperature decay
        T = T_start * ((T_end / T_start) ** (elapsed / total_time))

        # --------------------------------------------------------------
        # Propose a random mass transfer between two distinct positions
        # --------------------------------------------------------------
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # a[i] is the donor; we never remove more than a[i] itself
        max_delta = min(a[i], move_fraction)
        delta = rng.uniform(-max_delta, max_delta)   # negative = move the other way
        if delta == 0.0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        # protect against accidental negative values caused by numerical noise
        if a_new[i] < 0.0 or a_new[j] < 0.0:
            continue

        new_score = _fast_score(a_new)

        # --------------------------------------------------------------
        # Metropolis acceptance criterion
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # probability to accept a worsening move
            prob = np.exp((cur_score - new_score) / max(T, 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Periodic “large‑jitter” restart from the best point
        # --------------------------------------------------------------
        if it % 5000 == 0:
            # add a small random perturbation to the best solution and
            # resume the walk from there – helps escape shallow plateaus
            a = best_a + 0.01 * rng.normal(size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()                     # re‑normalise
            cur_score = _fast_score(a)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the
    # official evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
