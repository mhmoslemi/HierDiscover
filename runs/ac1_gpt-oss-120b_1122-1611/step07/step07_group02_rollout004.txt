"""
Hybrid optimizer for the step‑function evaluation problem.

The objective to minimise is
    2·n·max_k (a * a)[k]   with   a_i ≥ 0,  Σ a_i = 1.

The code combines two complementary phases:

1. **Smooth surrogate optimisation** – the infinity‑norm of the
   self‑convolution is approximated by a temperature‑controlled
   log‑sum‑exp.  The distribution is parametrised by logits x → softmax(x),
   which guarantees non‑negativity and the unit‑sum constraint.
   Adam updates are applied to the logits.  The temperature is raised
   slowly so that the optimisation first sees a smooth landscape and
   finally concentrates on the true maximum.

2. **Local simulated‑annealing refinement** – after a gradient run the best
   distribution found so far is fine‑tuned by repeatedly moving a small
   amount of probability mass between two random positions.  The move is
   accepted if it improves the exact objective or with a probability that
   decreases with a linear cooling schedule.

Multiple restarts with randomly chosen lengths (2000‑5000) are performed
until the allotted time budget (default 1000 s) is exhausted.  The best
sequence seen overall is returned (clipped to the limits enforced by the
evaluator).
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        # fallback – uniform distribution
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_max(a: np.ndarray) -> float:
    """
    Exact maximum of the linear self‑convolution a*a.
    a must be a 1‑D non‑negative array with Σ a = 1.
    """
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def _objective(a: np.ndarray) -> float:
    """
    Return the value of the evaluation function for a (normalised to sum=1):
    2·n·max(a*a).
    """
    n = a.shape[0]
    return 2.0 * n * _conv_max(a)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm runs a few Adam‑optimised smooth
    phases followed by a simulated‑annealing fine‑tuning, restarting with
    different lengths until the time budget is exhausted.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time‑budget handling (a few seconds safety margin for the harness)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 4.0)

    # ------------------------------------------------------------------
    # Book‑keeping for the best solution
    # ------------------------------------------------------------------
    best_score = np.inf
    best_a = None

    # ------------------------------------------------------------------
    # Core optimisation loop – performed until time runs out
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # Choose a problem size (length of the step function)
        # --------------------------------------------------------------
        n = int(rng.integers(2000, 5001))

        # --------------------------------------------------------------
        # Initialise a probability vector a (sum == 1)
        # --------------------------------------------------------------
        if "height_sequence_1" in globals():
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            base = np.clip(base, 0.0, None)
            # Interpolate / truncate to the desired length
            if base.size != n:
                old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
                new_x = np.linspace(0.0, 1.0, n, endpoint=False)
                a = np.interp(new_x, old_x, base)
            else:
                a = base.copy()
            # Normalise to sum = 1
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)

        # Add a small random jitter to avoid exact symmetry at start
        a += rng.normal(0.0, 0.01, size=n)
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a_sum

        # --------------------------------------------------------------
        # Adam optimisation on the log‑softmax parametrisation
        # --------------------------------------------------------------
        # logits such that softmax(logits)=a
        eps = 1e-12
        x = np.log(a + eps)

        m = np.zeros_like(x)
        v = np.zeros_like(x)

        # hyper‑parameters
        lr = 0.55
        beta1, beta2 = 0.9, 0.999
        adam_eps = 1e-8

        # temperature schedule for the smooth max surrogate
        beta_start = 4.0
        beta_end = 250.0
        beta_ramp_iters = 2500

        # FFT sizes (pre‑computed for speed)
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n - 2)

        # run a bounded number of iterations; break early if time runs out
        for it in range(3000):
            # ----------------------------------------------------------
            # a = softmax(x)  (guarantees non‑negativity and Σa = 1)
            # ----------------------------------------------------------
            a = _softmax_logits(x)

            # ----------------------------------------------------------
            # Exact self‑convolution (for scoring only)
            # ----------------------------------------------------------
            a_fft = np.fft.rfft(a, n=fft_len_conv)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]
            cur_max = float(conv.max())
            cur_score = 2.0 * n * cur_max

            # ----------------------------------------------------------
            # Record best solution
            # ----------------------------------------------------------
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (Adam) score = {best_score:.6f}")

            # ----------------------------------------------------------
            # Smooth surrogate of max via softmax (temperature β)
            # ----------------------------------------------------------
            beta = (beta_end if it >= beta_ramp_iters
                    else beta_start + (beta_end - beta_start) * it / beta_ramp_iters)

            scaled = beta * conv
            # w = softmax(scaled)  (sums to 1)
            lse = logsumexp(scaled)
            w = np.exp(scaled - lse)

            # ----------------------------------------------------------
            # Gradient of the surrogate w.r.t. a :  ∇_a  Σ w_k (a*a)_k
            # This equals 2·(w ⋆ a) where ⋆ is the correlation.
            # ----------------------------------------------------------
            w_fft = np.fft.rfft(w, n=fft_len_corr)
            a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
            corr = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)
            # The valid part of the correlation (length n) starts at index n‑1
            grad_a = 2.0 * corr[n - 1 : n - 1 + n]

            # ----------------------------------------------------------
            # Chain rule through the softmax parametrisation
            # ----------------------------------------------------------
            dot = np.dot(grad_a, a)          # ⟨grad_a , a⟩
            grad_x = a * (grad_a - dot)      # ∂L/∂x

            # ----------------------------------------------------------
            # Adam update on the logits
            # ----------------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

            # ----------------------------------------------------------
            # Small stochastic jitter every few hundred steps (helps escape plateaus)
            # ----------------------------------------------------------
            if (it + 1) % 450 == 0:
                x += rng.normal(0.0, 0.015, size=n)

            # ----------------------------------------------------------
            # Time‑budget check – abort the inner loop if we are out of time
            # ----------------------------------------------------------
            if time.monotonic() >= deadline:
                break

        # --------------------------------------------------------------
        # Simulated‑annealing fine‑tuning on the best vector found so far
        # --------------------------------------------------------------
        if best_a is not None:
            a = best_a.copy()
            # Ensure we are on the probability simplex
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a[:] = 1.0 / a.shape[0]
            else:
                a /= a_sum

            cur_score = _objective(a)
            # Annealing schedule
            T_start = 0.02
            T_end = 0.0001
            sa_steps = 7000
            for sa_it in range(sa_steps):
                # linear temperature decay
                T = T_start + (T_end - T_start) * (sa_it / sa_steps)

                # propose a small redistribution of mass between two indices
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                delta = rng.uniform(-0.03, 0.03)   # move up to 3 % of the total mass
                if a[i] + delta < 0.0 or a[j] - delta < 0.0:
                    continue
                a_new = a.copy()
                a_new[i] += delta
                a_new[j] -= delta

                new_score = _objective(a_new)

                # Metropolis acceptance
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (T + 1e-12)):
                    a = a_new
                    cur_score = new_score
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_a = a.copy()
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] new best (SA) score = {best_score:.6f}")

                # time guard
                if time.monotonic() >= deadline:
                    break

        # --------------------------------------------------------------
        # End of one outer restart – loop continues if time permits
        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    # Fallback: if nothing was ever found (should not happen) return uniform
    # ------------------------------------------------------------------
    if best_a is None:
        n_fallback = 3000
        best_a = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = _objective(best_a)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
