Implementation details for projection onto simplex:

```
def proj_simplex(v):
    """Projection of vector v onto the probability simplex (sum=1, v>=0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights on the simplex) that beats the
previous bound.  

The optimiser works on the probability simplex (∑a_i = 1, a_i ≥ 0) and
minimises a *variance* surrogate of the convolution peak:

    var(a) = 1/(2n‑1)·∑_k ( (a*a)_k – μ )² ,   μ = 1/(2n‑1).

The gradient of this smooth functional is obtained analytically by a single
FFT‑based correlation, giving a cheap O(n log n) update.  After each gradient
step the iterate is projected back onto the simplex (Euclidean projection).
A simple learning‑rate schedule (linear decay) together with occasional
random “jitter” and a stagnation‑restart strategy provides global exploration.
The exact objective (the value returned by ``evaluate_sequence``) is
computed from the convolution max after every iteration, and the best
sequence seen so far is retained.

The routine respects the 1000‑second budget by stopping a few seconds early
and returns the best sequence found (clipped to the limits enforced by
``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute the (unnormalised) correlation Σ_j a_j w_{i+j} for all i.
    The returned vector has length ``len(a)``.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return conv[start:end]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (components non‑negative and summing to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find the rho value (last index where the condition holds)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    # renormalise
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Deadline handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a (probability vector)
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a = np.asarray(init_seq, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            a = None
    else:
        a = None

    if a is None:
        # default length – a power of two for cheap FFTs
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / a.size
        else:
            a /= s
        n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)   # for a*a
    # ``_gradient_wrt_a`` needs a length at least n + (2n‑1) - 1 = 3n‑2
    fft_len_corr = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 3) Helper to evaluate the *exact* objective (no extra clipping)

    def _exact_score(vec: np.ndarray) -> float:

    # ------------------------------------------------------------------

    best_score = cur_score

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    # learning‑rate schedule for the variance‑gradient descent
    lr_start = 0.8          # fairly aggressive at the beginning

    # stagnation detection (seconds)
    stagnation_time = 30.0

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    iteration = 0

        mu = 1.0 / (2 * n - 1)                # the mean of the convolution (sum = 1)
        w = conv - mu                         # centred convolution
        # gradient_var = (2/(2n‑1)) * _gradient_wrt_a(a, w)    (derived analytically)
        grad = (2.0 / (2 * n - 1)) * _gradient_wrt_a(a, w)

        # ------- step‑size (linear decay) -------
        lr = lr_start + (lr_end - lr_start) * frac_time

        # ------- projected gradient descent step -------
        a_candidate = _proj_simplex(a - lr * grad)

        # occasional random “jitter’’ to escape shallow minima
        if rng.random() < 0.02:

        # ------- exact evaluation -------
        cand_score = _exact_score(a_candidate)

        # ------- keep the best -------
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = a_candidate.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (var‑grad) score = {best_score:.6f}")

        # ------- prepare next iterate -------
        a = a_candidate

        # ------- stagnation restart -------
        if now - last_improve > stagnation_time:
            # re‑initialise from the current best with moderate log‑normal noise
            noise = rng.normal(0.0, 0.3, size=n)
            a = np.log(best + 1e-12) + noise
            a = np.exp(a)
            a = _proj_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after {stagnation_time:.0f}s stagnation")

        # ------- occasional large random restart (exploration) -------
        if iteration % 500 == 0 and iteration > 0:

            else:
                # otherwise use it as a temporary perturbation

        iteration += 1

    # ------------------------------------------------------------------
    # 7) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
