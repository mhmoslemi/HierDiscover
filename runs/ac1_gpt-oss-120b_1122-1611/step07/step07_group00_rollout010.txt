"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the bound on ``evaluate_sequence``.

Algorithm
---------
1.  Choose a length *n* (use the length of the previous best construction if
    available, otherwise a moderate size such as 2048).  The target ℓ₁‑norm is
    ``sqrt(2·n)`` – this is the normalisation used by the original benchmark.

2.  Initialise the vector ``a`` uniformly so that ``sum(a)=sqrt(2·n)``.  This
    gives a starting value of exactly ``2.0`` for the evaluation metric.

3.  Repeatedly improve the vector by solving a linear programme:
        maximise   Σ b_j
        subject to (a  *  b)_k ≤ M   for every convolution shift *k*
                   b_j ≥ 0
    where ``M`` is the current maximal self‑convolution of ``a``.
    The constraints are linear in ``b`` because each entry of the convolution
    ``a*b`` is a linear combination of the unknowns ``b_j``.

    The optimum is attained at an extreme point; intuitively it fills the
    positions where ``a`` is small while keeping the convolution under the
    current bound.

4.  Rescale the solution ``b`` to satisfy the ℓ₁‑norm constraint
    ``sum(b)=sqrt(2·n)`` and mix it with the current iterate:
        a ← (1‑α)·a + α·b,
    where the step size *α* decays slowly (≈1/√iteration) so that the
    optimisation behaves like a projected sub‑gradient method.

5.  Keep the best vector seen (according to the exact ``evaluate_sequence``
    function).  If no improvement is observed for 30 s, restart from the best
    vector with a moderate log‑normal perturbation to escape possible stalls.

6.  The loop runs until the time budget (default 1000 s) is exhausted.
    The final best vector is clipped to the limits imposed by
    ``evaluate_sequence`` and returned as a plain list of ``float`` values.

The LP is solved with ``scipy.optimize.linprog`` (method *highs*).  For
moderate dimensions (n≈2000–3000) the problem size (≈2·n constraints, n
variables) fits comfortably in memory and a handful of LP solves easily fit
into the time budget.

The routine is completely deterministic apart from the random seed, uses only
standard scientific Python libraries and respects all constraints imposed by
the given ``evaluate_sequence`` implementation.

def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix ``A`` that implements the linear map
    ``b ↦ a * b`` (full linear convolution) as a set of inequality constraints.
    Shape of ``A`` is (2·n‑1) × n, where ``n = len(a)``.
    The k‑th row of ``A`` contains ``a`` shifted down by ``k`` rows.
    """
    n = a.size
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    # each column i carries the vector a placed at rows i … i+n‑1
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _solve_lp_max_sum(a: np.ndarray, M: float, sum_target: float):
    """
    Solve   maximise Σ b_j   s.t.   (a * b)_k ≤ M  (all k),
           b_j ≥ 0.
    Returns the optimal ``b`` rescaled to have ℓ₁‑norm ``sum_target``.
    If SciPy is unavailable or the solver fails, a simple heuristic is used.
    """
    try:
        from scipy.optimize import linprog
    except Exception:  # pragma: no cover – SciPy is expected to be present
        # Heuristic: put all mass where a is smallest.
        idx = np.argmin(a)
        b = np.zeros_like(a)
        b[idx] = sum_target
        return b

    n = a.size
    A = _build_conv_matrix(a)
    rows = A.shape[0]

    # Linear programme: minimise -Σ b  <=> maximise Σ b
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(rows, M, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # SciPy’s HiGHS solver is fast and robust for dense problems of this size.
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method="highs", options={"presolve": True})
    if not res.success:
        # Fallback – concentrate the mass on the smallest entries of ``a``.
        # This respects the convolution bound because those entries contribute
        # the least to a*b.
        order = np.argsort(a)
        b = np.zeros_like(a)
        remaining = sum_target
        for idx in order:
            # the most we can allocate without breaking the bound is M / a[idx]
            # but we keep it simple and just distribute uniformly among the
            # smallest positions.
            add = remaining / (n - np.where(order == idx)[0][0])
            b[idx] = add
            remaining -= add
            if remaining <= 0:
                break
        return b

    b = res.x
    # Rescale to the required ℓ₁‑norm.
    total = b.sum()
    if total > 0:
        b *= (sum_target / total)
    else:  # degenerate case – fall back to uniform
        b = np.full_like(a, sum_target / n)
    return b

    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The function respects the 1000 s budget, returns the best sequence found,
    and prints progress information.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time management – keep a 5 s safety margin
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine the length of the trial vector
    # ------------------------------------------------------------------
    # If a previous construction is available, reuse its length.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        n = len(init_seq)
    else:
        n = 2048                     # a moderate size that fits easily in memory

    # The benchmark uses the normalisation Σ a_i = √(2·n)
    sum_target = float(np.sqrt(2.0 * n))

    # ------------------------------------------------------------------
    # 2) Initialise ``a`` uniformly on the simplex with the required ℓ₁‑norm
    # ------------------------------------------------------------------
    a = np.full(n, sum_target / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3) Track the best sequence seen so far (according to the exact evaluation)
    # ------------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    iteration = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop (LP‑based improvement)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time
        frac = (elapsed / total_time)

        # ---- current maximal self‑convolution (exact) ----
        conv = np.convolve(a, a)                 # exact linear convolution
        M = conv.max()

        # ---- solve the LP to obtain a beneficial direction g ----
        g = _solve_lp_max_sum(a, M, sum_target)

        # ---- convex combination step (step size slowly decreases) ----
        #  α = 0.5 / sqrt(iteration) gives a diminishing step size.
        alpha = 0.5 / np.sqrt(iteration + 1.0)
        a = (1.0 - alpha) * a + alpha * g

        # ---- enforce the hard box constraints and renormalise ----
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s > 0.0:
            a *= (sum_target / s)
        else:                                   # extremely unlikely – fall back
            a = np.full_like(a, sum_target / n)

        # ---- exact evaluation (the metric used by the harness) ----
        cur_score = evaluate_sequence(a.tolist())

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- stagnation handling – restart after 30 s without improvement ----
        if now - last_improve > 30.0:
            # Add moderate log‑normal noise to the best vector and restart.
            noise = rng.normal(0.0, 0.3, size=n)
            a = best * np.exp(noise)
            a = np.clip(a, 0.0, 1000.0)
            a_sum = a.sum()
            if a_sum > 0.0:
                a *= (sum_target / a_sum)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # ---- occasional global jitter (≈2 % of iterations) ----
        if rng.random() < 0.02:
            # Perturb a random contiguous block – helps escape shallow minima.
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.15))
            a_jit = best.copy()
            a_jit[start:start + width] *= factor
            a_jit = np.clip(a_jit, 0.0, 1000.0)
            a_jit_sum = a_jit.sum()
            if a_jit_sum > 0.0:
                a_jit *= (sum_target / a_jit_sum)
            a = a_jit
            print(f"[{elapsed:6.2f}s] jitter applied")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits imposed by ``evaluate_sequence``)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
