def self_convolve(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2*a.size - 1]
    return conv

from scipy.optimize import linprog
import scipy.sparse as sp

def solve_lp(a, M):
    n = len(a)
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # for each i in max(0,k-(n-1))..min(k,n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # for i in [i_start, i_end] inclusive:
        # coefficient for b_{i} is a_{k-i}
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            data.append(a[k - i])
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # objective maximize sum b; linprog minimizes; use c = -1
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    b = res.x
    return b
```

search for a non‑negative step‑function minimizing ``evaluate_sequence``.

Algorithm
---------

* Initialise a feasible sequence `a` (use the best known sequence if the
  global ``height_sequence_1`` is present, otherwise a random Dirichlet
  draw).  The vector is normalised to L¹‑norm = 1, which leaves the
  objective invariant.

* At every iteration we compute the linear self‑convolution `c = a*a`
  with an FFT (O(n log n)).  The current objective is
  ``J = 2·n·max(c)`` because the L¹‑norm is 1.

* The index `k_max` where `c` attains its maximum is identified.
  Contributions to this peak are the products
  ``a[i]·a[k_max-i]``.  A source index `i_src` is sampled proportionally
  to those contributions – i.e. we preferentially move mass that hurts
  the peak most.

* A target index `i_tgt` is chosen preferentially where `a` is small
  (probability proportional to ``1‑a``).  A fraction (up to 20 %) of the
  source mass is transferred to the target, the vector is re‑normalised
  and the new objective `J'` is evaluated.

* The move is accepted if it improves the objective, otherwise it is
  accepted with a small, linearly decreasing temperature (simulated
  annealing) to allow occasional uphill steps.

* To avoid long stagnation the current state is periodically perturbed
  (log‑normal multiplicative noise on a random contiguous block) and,
  after 15 s without improvement, we restart from the best solution seen
  so far adding mild random noise.

* The loop runs until the prescribed time budget (with a 5 s safety
  margin).  The best sequence encountered is returned as a plain list;
  values are clipped to the bounds enforced by ``evaluate_sequence`` for
  safety.

The method is deliberately simple, cheap per iteration and focuses
directly on reducing the maximal convolution entry, which is the only
term appearing in the evaluation function.

import numpy as np

# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution ``a*a`` using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # keep only the linear part (length 2·n‑1)
    return conv[: 2 * a.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit L¹‑norm."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))               # block width ≤ 5 % of n
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))                  # moderate noise
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

        if s == 0.0:
            continue
        a_candidate /= s

        # ----- 3.3) Evaluate the candidate -----
        conv_cand = _self_conv(a_candidate, fft_len)
        cand_J = 2.0 * n * float(conv_cand.max())

        # ----- 3.4) Acceptance test (monotone + simulated annealing) -----
        now = time.time()
        elapsed_frac = (now - start_time) / (deadline - start_time)
        temp = temp_start * (1.0 - elapsed_frac) + temp_end * elapsed_frac

        accept = False
        if cand_J < cur_J:
            accept = True
        else:
            # occasional uphill moves; probability decays with temperature
            if temp > 0.0 and rng.random() < np.exp(-(cand_J - cur_J) / temp):
                accept = True

        if accept:
            a = a_candidate
            cur_J = cand_J
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best J = {best_J:.6f}")

        # ----- 3.5) Occasional global jitter (helps escaping plateaus) -----
        if rng.random() < 0.01:               # ~1 % of iterations
            a = _random_block_perturb(a, rng)

        # ----- 3.6) Restart after stagnation -----
        if now - last_improve > 15.0:
            # return to the best candidate and inject mild noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a /= a.sum()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – still best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence seen (clipped to allowed bounds)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    # Convert to plain Python list of floats
    return [float(x) for x in best]
