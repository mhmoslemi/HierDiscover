"""
Propose a step‑function that beats the previous 1.5132 bound.

Algorithm
---------

* Start from the best known sequence (global ``height_sequence_1``) if it is
  available, otherwise use a uniform distribution.  The sequence is always
  kept on the simplex (non‑negative, sum = 1).

* Repeatedly perform **targeted mass‑transfer moves** that directly attack the
  current peak of the self‑convolution:
    – locate the index `k_max` where the convolution `c = a*a` attains its
      maximum,
    – compute the contributions `a[i]*a[k_max-i]` of all pairs that build this
      peak,
    – pick a donor index `i` proportionally to its contribution,
    – pick a recipient index `j` that currently has a small value,
    – transfer a random fraction of the donor’s mass to the recipient and
      update the convolution incrementally (O(n) slice operations).

* Accept a move if it lowers the true objective
  `score = 2·n·max(c)`.  Otherwise accept with a Metropolis probability
  `exp(-(Δscore)/T)`, where the temperature `T` decays linearly from an
  initial value (≈0.05) to zero.

* Every few thousand accepted moves recompute the convolution from scratch
  (FFT) to remove accumulated numerical drift.

* If no improvement is seen for a while, apply a light random “jitter”
  (log‑normal perturbation of a random block) and continue.

* The whole routine runs for the allotted time (≈ 995 s, 5 s safety margin)
  and finally returns the best sequence found, clipped to the limits imposed
  by ``evaluate_sequence`` (0 ≤ x ≤ 1000).

The method focuses the search on the actual bottleneck (the peak of the
convolution) and therefore converges much faster than a generic mirror‑descent
or simulated‑annealing scheme.  It is completely self‑contained, uses only
NumPy and the provided ``evaluate_sequence`` function and respects the required
API.

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to index ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    # new a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # incremental update of convolution (all operations are vectorised)
    c_new = c.copy()
    # contributions of the donor (subtract) and recipient (add)
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ part (the i,i and j,j self‑pairs and the i,j cross‑pair)
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

        a_new, c_new = _apply_move(a, c, donor, recipient, delta)
        new_score = 2.0 * n * c_new.max()

        # --------------------------------------------------------------
        # Metropolis acceptance with a linearly decaying temperature
        # --------------------------------------------------------------
        frac = (now - start_time) / (deadline - start_time)
        temperature = INIT_TEMP * (1.0 - frac)
        temperature = max(temperature, 1e-12)

        if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
            a = a_new
            c = c_new
            cur_score = new_score
            move_counter += 1

            # Periodic full recomputation to keep numerical errors bounded
            if move_counter % RECOMPUTE_EVERY == 0:
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > STAGNATION_LIMIT:
            # Perturb the current best a little and continue
            a = _random_block_perturb(best, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

        # --------------------------------------------------------------
        # Occasionally add a tiny global jitter to keep diversity
        # --------------------------------------------------------------
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to limits enforced by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
