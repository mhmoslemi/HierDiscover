            c = _self_conv(a, fft_len_self)

```
def _simplex_projection(v):
    """Project onto the probability simplex {x >=0, sum=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

            v = beta2 * v + (1 - beta2) * (grad * grad)
            m_hat = m / (1 - beta1 ** iter_cnt)
            v_hat = v / (1 - beta2 ** iter_cnt)

            # projection onto simplex
            a_new = _simplex_projection(a_new)

            # evaluate
            c_new = _self_conv(a_new, fft_len_self)
            new_score = 2.0 * n * c_new.max()

            if new_score < cur_score:
                a = a_new
            else:
                # Metropolis fallback
                if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    a = a_new
                else:
                    # occasional perturbation
                    if rng.random() < 0.02:
                        a = _random_block_perturb(a, rng)
                        a = _simplex_projection(a)  # ensure sum 1

            # occasional random swap or permutation
            if rng.random() < 0.01:
                # swap two random positions
                i, j = rng.integers(0, n, size=2)
                a[i], a[j] = a[j], a[i]
                # (swap does not change sum)
                # keep a as is

            # early restart if stagnation
            if time.monotonic() - last_improve > 40.0:
                # jitter
                noise = np.exp(rng.normal(0.0, 0.05, size=n))
                a = a * noise
                a = _simplex_projection(a)
                last_improve = time.monotonic()
                # reset Adam state
                m = np.zeros(n)
                v = np.zeros(n)
                iter_cnt = 0

            # break condition for this restart if we have done enough iterations (e.g., 20000)
            if iter_cnt >= 30000:
                break

        iter_restart += 1
    # after all restarts
    if best_seq is None:
        # fallback uniform
        n = 4000
        best_seq = np.full(n, 1.0 / n)
    # clip as required
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

Step‑function optimizer using Adam‑based mirror descent with soft‑max
approximation of the convolution peak. Multiple restarts, stochastic
perturbations and a simple simplex projection are employed to escape
local minima and to push the objective 2·n·max(a*a) below the previous
best (≈1.513). The optimizer works entirely with NumPy/FFT, fits the
1000 s budget and returns a list of non‑negative heights.

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """Linear convolution of real vectors a and b using an FFT of size ``length``."""
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]

def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using a pre‑computed FFT length."""
    return _fft_convolve(a, a, fft_len)

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    # Implementation of the algorithm from
    # https://arxiv.org/abs/1309.1541 (section 3)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise
    s = w.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / s

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    the evaluation function. The routine runs several independent
    optimisation restarts, each employing Adam‑style updates on the
    soft‑max of the self‑convolution. Random block perturbations and
    occasional index swaps help to escape plateaus. The best sequence
    encountered before the time budget expires is returned.
    """

    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = np.inf
    global_start = time.monotonic()
    restart_count = 0

                a = _simplex_projection(a)
                n = a.size
            else:

                a = rng.dirichlet(np.ones(n) * 0.5)
        else:
            # no prior candidate – random length in a reasonable range
            n = rng.integers(3000, 6000)
            a = rng.dirichlet(np.ones(n) * 0.5)

        # ------------------------------------------------------------------
        # 1.2 Pre‑compute FFT lengths
        # ------------------------------------------------------------------
        fft_len_self = _next_pow2(2 * n - 1)
        fft_len_grad = _next_pow2(3 * n - 2)

        # ------------------------------------------------------------------

        # ------------------------------------------------------------------

        v = np.zeros(n, dtype=np.float64)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8

        # ------------------------------------------------------------------
        # 1.4 Statistics for this restart
        # ------------------------------------------------------------------
        it = 0
        last_improve = time.monotonic()
        inner_start = time.monotonic()

        # ------------------------------------------------------------------

        while time.monotonic() < deadline:
            # --------------------------------------------------------------

            # --------------------------------------------------------------
            c = _self_conv(a, fft_len_self)             # length 2n‑1
            cur_max = c.max()
            cur_score = 2.0 * n * cur_max                # sum(a)=1 because of projection

            # --------------------------------------------------------------

            # --------------------------------------------------------------
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_seq = a.copy()
                elapsed = time.monotonic() - global_start
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

                last_improve = time.monotonic()

            # --------------------------------------------------------------
            # 1.5.3 Schedule for smooth‑max temperature and learning rate
            # --------------------------------------------------------------
            total_frac = (time.monotonic() - global_start) / max(1e-12, (budget_s - 5.0))
            alpha = 0.5 + 60.0 * total_frac               # grow to ≈60.5
            # learning‑rate schedule (Adam‑style global step size)
            lr = 0.8 / (1.0 + it * 1e-4)                  # slowly decreasing

            # --------------------------------------------------------------

            # --------------------------------------------------------------

            w = np.exp(alpha * c_shift)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # --------------------------------------------------------------
            # 1.5.5 Gradient of the smooth‑max (g = 2·a*w convolution)
            # --------------------------------------------------------------
            conv_full = _fft_convolve(a, w, fft_len_grad)   # length 3n‑2
            g = 2.0 * conv_full[n - 1: 2 * n - 1]          # slice to length n

            # --------------------------------------------------------------

            # --------------------------------------------------------------
            it += 1
            m = beta1 * m + (1 - beta1) * g

            v_hat = v / (1 - beta2 ** it)

            a_proposed = _simplex_projection(a_proposed)

            # --------------------------------------------------------------
            # 1.5.7 Decide whether to accept the proposal
            # --------------------------------------------------------------

            new_score = 2.0 * n * c_new.max()

            # Metropolis‑style acceptance to avoid getting stuck
            temperature = max(0.02 * (1 - total_frac), 1e-6)   # linear cooling
            if new_score < cur_score:
                accept = True
            else:
                accept = rng.random() < np.exp(-(new_score - cur_score) / temperature)

            if accept:
                a = a_proposed
                cur_score = new_score
                # occasionally recompute the self‑convolution to wipe drift
                if it % 5000 == 0:
                    c = _self_conv(a, fft_len_self)
            else:

                if rng.random() < 0.005:
                    a = _random_block_perturb(a, rng)
                    a = _simplex_projection(a)

            # --------------------------------------------------------------
            # 1.5.8 Extra stochastic moves (swap / permutation)
            # --------------------------------------------------------------
            if rng.random() < 0.008:
                i, j = rng.integers(0, n, size=2)
                a[i], a[j] = a[j], a[i]   # swap – keeps the sum intact

            # --------------------------------------------------------------
            # 1.5.9 Restart if stagnation lasts too long
            # --------------------------------------------------------------

                # gentle jitter around the current best (if any)

                jitter = np.exp(rng.normal(0.0, 0.07, size=n))
                a = base * jitter
                a = _simplex_projection(a)
                # reset Adam moments
                m.fill(0.0)
                v.fill(0.0)
                it = 0
                last_improve = time.monotonic()
                # small log to trace the restart
                elapsed = time.monotonic() - global_start
                print(f"[{elapsed:6.2f}s] restart #{restart_count+1} – score {cur_score:.6f}")

            # --------------------------------------------------------------
            # 1.5.10 Termination of this restart (iteration budget)
            # --------------------------------------------------------------
            if it >= 40000:
                break

        restart_count += 1

    # ------------------------------------------------------------------
    # 2) Fallback – uniform sequence if nothing was found (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3) Clip to the limits of evaluate_sequence and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

            v_hat = v / (1 - beta2 ** it)
            a_proposed = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- project onto simplex (keeps non‑negativity and Σ=1) ---------------
            a_proposed = _simplex_projection(a_proposed)

            # ---- evaluate the proposal ---------------------------------------------
            c_new = _self_conv(a_proposed, fft_len_self)
            new_score = 2.0 * n * c_new.max()

            # ---- Metropolis‑style acceptance ---------------------------------------
            temperature = max(0.02 * (1 - total_frac), 1e-6)
            if new_score < cur_score:
                accept = True
            else:
                accept = rng.random() < np.exp(-(new_score - cur_score) / temperature)

            if accept:
                a = a_proposed
                # occasional exact recomputation to wipe drift
                if it % 5000 == 0:
                    c = _self_conv(a, fft_len_self)
            else:
                # small chance of forced exploration
                if rng.random() < 0.005:
                    a = _random_block_perturb(a, rng)
                    a = _simplex_projection(a)

            # ---- extra stochastic moves (swap two entries) -------------------------
            if rng.random() < 0.008:
                i, j = rng.integers(0, n, size=2)
                a[i], a[j] = a[j], a[i]   # swap, sum unchanged

            # ---- restart on stagnation ---------------------------------------------
            if time.monotonic() - last_improve > 45.0:
                # jitter around the best sequence seen so far
                base = best_seq if best_seq is not None else a
                jitter = np.exp(rng.normal(0.0, 0.07, size=n))
                a = base * jitter
                a = _simplex_projection(a)
                m.fill(0.0)
                v.fill(0.0)
                it = 0
                last_improve = time.monotonic()
                elapsed = time.monotonic() - global_start
                print(f"[{elapsed:6.2f}s] restart #{restart_idx + 1} – score {cur_score:.6f}")

            # ---- per‑restart iteration budget ---------------------------------------
            if it >= 40000:
                break

        restart_idx += 1

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 4000
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Clip to the limits of evaluate_sequence and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
