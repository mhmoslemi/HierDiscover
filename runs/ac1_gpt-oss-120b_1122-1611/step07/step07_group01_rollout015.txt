"""
Search for a non‑negative step‑function that minimises the provided
``evaluate_sequence`` objective.

The search consists of three stages:

1. **Hard‑max mirror descent** – a simplex‑constrained exponentiated‑gradient
   optimisation where the loss is the (soft) maximum of the self‑convolution.
   The gradient is obtained as the convolution of a mask that selects the
   current maximal entries with the current vector.  The mask tolerance is
   annealed from a wide band (≈ 50 % of the peak) to a narrow one (≈ 1 %).

2. **Exploration & restarts** – every few seconds a random contiguous block
   is multiplicatively perturbed (log‑normal factor) and, if the algorithm has
   stagnated for > 30 s, a jittered copy of the best‑so‑far vector is used as
   a new starting point.

3. **Polishing (simulated annealing)** – after the main descent a short
   annealing phase moves a small fraction of mass between two random indices.
   Worsening moves are occasionally accepted according to a temperature that
   decays to zero.

The routine tries several lengths (including the length of the known best
construction, if the global variable ``height_sequence_1`` is present) and
keeps the globally best sequence discovered before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return conv_fft(a, a, fft_len)


def compute_score(a: np.ndarray, fft_len_self: int) -> float:
    """
    Compute the exact evaluation score (without the clipping performed by
    ``evaluate_sequence``) for a normalized non‑negative vector ``a``.
    The vector is assumed to sum to 1; the returned value equals
    ``2·n·max(a*a)``.
    """
    n = a.size
    c = self_conv(a, fft_len_self)
    return 2.0 * n * c.max()


def random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def interpolate_sequence(seq: np.ndarray, target_len: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length, normalised to sum 1."""
    src_len = seq.size
    if src_len == target_len:
        out = seq.copy()
    else:
        x_src = np.linspace(0.0, 1.0, src_len)
        x_tgt = np.linspace(0.0, 1.0, target_len)
        out = np.interp(x_tgt, x_src, seq)
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / target_len
    else:
        out /= s
    return out

def hardmax_descent(
    n: int,
    init: np.ndarray | None,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent optimisation using a hard‑max mask on the self‑convolution.
    Returns the best vector (sum = 1) and its score.
    """
    fft_len_self = next_pow2(2 * n - 1)
    fft_len_wa = next_pow2(3 * n)

    # initialise a (probability vector)
    if init is not None:
        a = init.copy()
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s
    else:
        # Dirichlet(0.5) gives a moderately sparse start
        a = rng.dirichlet(np.full(n, 0.5))
    best = a.copy()
    best_score = compute_score(a, fft_len_self)

    start = time.monotonic()
    last_improve = start
    iter_cnt = 0

    while time.monotonic() - start < time_limit:
        iter_cnt += 1
        now = time.monotonic()
        elapsed = now - start

        # ------------------------------------------------------------------
        # scheduling
        # ------------------------------------------------------------------
        frac = elapsed / time_limit  # 0 → 1
        eta = 2.0 * (1.0 - frac) + 0.001 * frac           # learning rate
        tol = 0.5 * (1.0 - frac) + 0.01 * frac            # mask tolerance

        # ------------------------------------------------------------------
        # convolution and mask
        # ------------------------------------------------------------------
        c = self_conv(a, fft_len_self)            # length 2n‑1
        max_c = c.max()
        mask = c >= max_c * (1.0 - tol)
        w = np.zeros_like(c)
        w[mask] = 1.0 / mask.sum()                 # probability mass on the
                                                   # currently maximal entries

        # ------------------------------------------------------------------
        # gradient of the smoothed loss
        # ------------------------------------------------------------------
        conv_wa = conv_fft(w, a, fft_len_wa)       # length 3n‑2
        grad = 2.0 * conv_wa[n - 1:n - 1 + n]      # slice of length n

        # ------------------------------------------------------------------
        # exponentiated‑gradient update (mirror descent on the simplex)
        # ------------------------------------------------------------------
        exponent = -eta * grad
        exponent = np.clip(exponent, -50.0, 50.0)   # avoid overflow
        a = a * np.exp(exponent)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ------------------------------------------------------------------
        # occasional exploratory perturbation
        # ------------------------------------------------------------------
        if rng.random() < 0.01:
            a = random_block_perturb(a, rng)

        # ------------------------------------------------------------------
        # keep the best solution seen so far
        # ------------------------------------------------------------------
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] n={n} new best = {best_score:.6f}")

        # ------------------------------------------------------------------
        # soft restart after stagnation
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            # jitter around the current best
            jitter = np.exp(rng.normal(0.0, 0.09, size=n))
            a = best * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            last_improve = now
            print(f"[{elapsed:6.2f}s] n={n} restart after stagnation; best {best_score:.6f}")

    return best, best_score


def annealing_polish(
    a: np.ndarray,
    rng: np.random.Generator,
    time_limit: float,
) -> tuple[np.ndarray, float]:
    """
    Small simulated‑annealing polishing phase that moves a tiny amount of mass
    between two random positions.  The temperature decays linearly to zero.
    Returns the (possibly) improved vector and its score.
    """
    n = a.size
    fft_len_self = next_pow2(2 * n - 1)
    start = time.monotonic()
    best = a.copy()
    best_score = compute_score(best, fft_len_self)

    while time.monotonic() - start < time_limit:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / time_limit
        # linear cooling schedule
        temperature = max(0.0, 0.5 * (1.0 - frac))  # max temperature ≈0.5

        # propose a move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of mass from i to j
        delta = best[i] * rng.uniform(0.0, 0.1)  # up to 10 % of i's mass
        if delta <= 0.0:
            continue
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta

        # evaluate
        cand_score = compute_score(cand, fft_len_self)

        # accept if better, otherwise accept with Metropolis probability
        if cand_score < best_score:
            best, best_score = cand, cand_score
        else:
            if temperature > 0.0:
                prob = np.exp(-(cand_score - best_score) / temperature)
                if rng.random() < prob:
                    best, best_score = cand, cand_score

    return best, best_score

        # allocate a slice of the remaining budget to the next length
        remaining = deadline - time.monotonic()
        # give each remaining length at most 1/len(lengths) of the time,
        # but never less than 30 s (to allow a meaningful search)
        per_len_budget = max(30.0, remaining / len(lengths))

        n = lengths.pop(0)                     # take the smallest remaining

        # initialise from the reference if the sizes match (or interpolate)
        init_vec = None
        if ref_seq is not None:
            init_vec = interpolate_sequence(ref_seq, n)

        # ------------------------------------------------------------------
        # Stage 1 – hard‑max mirror descent
        # ------------------------------------------------------------------
        best_local, best_local_score = hardmax_descent(
            n=n,
            init=init_vec,
            rng=rng,
            time_limit=per_len_budget * 0.8,   # 80 % of the slice for descent
        )

        # ------------------------------------------------------------------
        # Stage 2 – polishing with simulated annealing
        # ------------------------------------------------------------------
        polish_time = per_len_budget * 0.2
        if polish_time > 5.0:                 # need a minimal time slice
            best_local, best_local_score = annealing_polish(
                a=best_local,
                rng=rng,
                time_limit=polish_time,
            )

        # ------------------------------------------------------------------
        # Update the global best
        # ------------------------------------------------------------------
        if best_local_score < global_best_score:
            global_best_score = best_local_score
            global_best = best_local.copy()
            elapsed = time.monotonic() - (deadline - budget_s)
            print(f"[{elapsed:6.2f}s] GLOBAL NEW BEST (n={n}) = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # If nothing was found (unlikely) fall back to a uniform distribution
    # ------------------------------------------------------------------
    if global_best is None:
        n = 3800
        global_best = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the
    # evaluation function for safety).
    # ------------------------------------------------------------------
    global_best = np.clip(global_best, 0.0, 1000.0)
    return [float(x) for x in global_best]
