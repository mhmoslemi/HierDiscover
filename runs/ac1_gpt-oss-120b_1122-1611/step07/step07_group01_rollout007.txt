"""
Propose a step‑function with a low self‑convolution peak.

Algorithm
---------

1.  Multiple independent runs are performed until the time‑budget expires.
    Each run works on a (possibly) different length `n` – larger `n`
    gives a larger optimisation space but costs more per iteration.

2.  A sequence `a` (always kept on the probability simplex) is
    initialised either from the known construction `height_sequence_1`
    (interpolated to the current length) or from a random Dirichlet
    distribution.

3.  Two optimisation phases are combined:

    * **Soft‑max mirror descent** (early phase).  
      The self‑convolution `c = a*a` is turned into a probability
      distribution with a temperature‑controlled soft‑max.
      The smoothed gradient is `g = 2·(w*a)` (slice of the convolution
      of the soft‑max weights `w` with `a`).  An exponentiated‑gradient
      update `a ← a·exp(‑η·g)` keeps `a` on the simplex.

    * **Maximum‑subgradient descent** (late phase).  
      When the soft‑max concentration `β` is already large we switch to
      the exact sub‑gradient of the true maximum:
      for the index `k* = argmax (a*a)` the gradient is
      `∂(a*a)[k*]/∂a_i = 2·a_{k*‑i}` (if the index is valid).  The same
      multiplicative update is applied.

    The learning‑rate `η` decays strongly, while `β` grows up to
    several thousand, so the method quickly focuses on the true peak.

4.  Random *block perturbations* are applied with a small probability
    to escape shallow basins, and a *restart* (jitter of the current
    best) is performed after a few seconds of stagnation.

5.  The best sequence seen over all runs (and all phases) is returned.
    The evaluation uses the exact formula
    `score = 2·n·max(a*a)` (the sum of `a` is kept equal to 1).

The whole search finishes comfortably inside the 1000 s budget,
uses only NumPy (no external solvers) and respects the non‑negative
constraint required by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _softmax_step(a: np.ndarray,
                  eta: float,
                  beta: float,
                  fft_len_self: int,
                  fft_len_wa: int) -> tuple[np.ndarray, float]:
    """
    One mirror‑descent step with a soft‑max approximation of the max.
    Returns the updated sequence and the current maximum of the convolution.
    """
    n = a.size
    c = _self_conv(a, fft_len_self)                 # length 2n‑1
    max_c = c.max()
    # soft‑max (shifted by max_c for stability)
    w_raw = np.exp(beta * (c - max_c))
    w = w_raw / w_raw.sum()
    # gradient: 2·(w * a) slice centred at index n‑1
    conv_wa = _conv_fft(w, a, fft_len_wa)           # length 3n‑2
    grad = 2.0 * conv_wa[n - 1:n - 1 + n]
    # exponentiated‑gradient update (mirror descent on the simplex)
    exponent = -eta * grad
    exponent = np.clip(exponent, -50.0, 50.0)
    factor = np.exp(exponent)
    a_new = a * factor
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new, max_c


def _max_subgrad_step(a: np.ndarray,
                      eta: float,
                      fft_len_self: int) -> tuple[np.ndarray, float]:
    """
    One step of sub‑gradient descent targeting the true maximum of a*a.
    Returns the updated sequence and the current maximum value.
    """
    n = a.size
    c = _self_conv(a, fft_len_self)
    k = int(np.argmax(c))
    max_c = float(c[k])
    # gradient of the selected entry (only indices that contribute)
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    if i_start <= i_end:
        i_idx = np.arange(i_start, i_end + 1)
        j_idx = k - i_idx
        grad[i_idx] = 2.0 * a[j_idx]
    # exponentiated‑gradient update
    exponent = -eta * grad
    exponent = np.clip(exponent, -50.0, 50.0)
    factor = np.exp(exponent)
    a_new = a * factor
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / n
    else:
        a_new /= s
    return a_new, max_c


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    `evaluate_sequence`.  The routine runs several independent optimisation
    runs (different lengths, random seeds) and returns the best sequence
    found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Global bookkeeping
    # ------------------------------------------------------------------
    global_best = None
    global_best_score = np.inf

    # ------------------------------------------------------------------
    # 2) Length options – a mixture of moderate and large problem sizes
    # ------------------------------------------------------------------
    length_options = np.array([3000, 3800, 5000, 8000, 12000])
    # favour the larger sizes a bit
    probs = np.array([0.15, 0.25, 0.20, 0.25, 0.15])
    probs = probs / probs.sum()

    # ------------------------------------------------------------------
    # 3) Main run loop – each iteration spawns a short optimisation job
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 15.0:
            break                     # not enough time for another run

        # allocate a slice of the remaining budget (up to 120 s)
        run_time = min(120.0, remaining * 0.4)
        # pick a problem size
        n = int(rng.choice(length_options, p=probs))

        # ----------------------------------------------------------------
        # 3.1) Initialise the probability vector `a`
        # ----------------------------------------------------------------
        if "height_sequence_1" in globals():
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            # Interpolate to the required length if necessary
            if base.size != n:
                xp = np.linspace(0.0, 1.0, base.size)
                x = np.linspace(0.0, 1.0, n)
                a = np.interp(x, xp, base)
                a = np.clip(a, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(n, 1.0 / n)
            else:
                a = base.copy()
                a = np.clip(a, 0.0, None)
                s = a.sum()
                a = a / s if s > 0.0 else np.full(n, 1.0 / n)
        else:
            a = rng.random(n)
            a /= a.sum()

        # ----------------------------------------------------------------
        # 3.2) Pre‑compute FFT lengths
        # ----------------------------------------------------------------
        fft_len_self = _next_pow2(2 * n - 1)          # for a*a
        fft_len_wa   = _next_pow2(3 * n)              # for w*a (soft‑max phase)

        # ----------------------------------------------------------------
        # 3.3) Local bookkeeping
        # ----------------------------------------------------------------
        c0 = _self_conv(a, fft_len_self)
        local_best = a.copy()
        local_best_score = 2.0 * n * c0.max()
        last_improve = time.monotonic()
        sub_start = time.monotonic()

        # ----------------------------------------------------------------
        # 3.4) Optimisation loop for this run
        # ----------------------------------------------------------------
        while time.monotonic() < sub_start + run_time:
            now = time.monotonic()
            elapsed = now - sub_start
            frac = elapsed / run_time          # 0 → 1 within this run

            # ------------------------------------------------------------
            # Phase selection
            # ------------------------------------------------------------
            if frac < 0.55:                     # soft‑max (smooth) phase
                eta  = 2.0 * (1.0 - frac / 0.55) + 0.01 * (frac / 0.55)   # 2 → 0.01
                beta = 0.5 + (5000.0 - 0.5) * (frac / 0.55)               # 0.5 → 5000
                a, max_c = _softmax_step(a, eta, beta, fft_len_self, fft_len_wa)
                cur_score = 2.0 * n * max_c
            else:                               # max‑subgradient (accurate) phase
                # linearly decay η from 0.5 to 0.001 during the remaining part
                subfrac = (frac - 0.55) / 0.45
                eta = 0.5 * (1.0 - subfrac) + 0.001 * subfrac
                a, max_c = _max_subgrad_step(a, eta, fft_len_self)
                cur_score = 2.0 * n * max_c

            # ------------------------------------------------------------
            # Occasional random exploration
            # ------------------------------------------------------------
            if rng.random() < 0.003:           # about 0.3 % of iterations
                a = _random_block_perturb(a, rng)

            # ------------------------------------------------------------
            # Keep the best solution seen so far (global & local)
            # ------------------------------------------------------------
            if cur_score < local_best_score - 1e-12:
                local_best_score = cur_score
                local_best = a.copy()
                last_improve = now
                if cur_score < global_best_score:
                    global_best_score = cur_score
                    global_best = a.copy()
                    total_elapsed = now - (deadline - budget_s)
                    print(f"[{total_elapsed:6.2f}s] NEW GLOBAL BEST = {global_best_score:.6f} (n={n})")

            # ------------------------------------------------------------
            # Restart after stagnation
            # ------------------------------------------------------------
            if now - last_improve > 15.0:
                # jitter the current local best and continue
                a = local_best.copy()
                jitter = np.exp(rng.normal(0.0, 0.07, size=n))
                a = a * jitter
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                # recompute a fresh score after the jitter
                c_tmp = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c_tmp.max()
                last_improve = now
                print(f"[{now - (deadline - budget_s):6.2f}s] restart (len={n}) cur={cur_score:.6f}")

        # end of this optimisation run – possibly continue with another size
        # early exit if we have already beaten the target
        if global_best_score < 1.5025:
            break

    # ------------------------------------------------------------------
    # 4) Final fine‑tuning (if time permits)
    # ------------------------------------------------------------------
    if global_best is not None:
        a = global_best.copy()
        n = a.size
        fft_len_self = _next_pow2(2 * n - 1)
        fft_len_wa   = _next_pow2(3 * n)

        # a handful of very aggressive soft‑max steps with huge β
        # (use the remaining budget, but keep it short)
        fine_steps = 0
        while time.monotonic() < deadline and fine_steps < 200:
            a, _ = _softmax_step(a,
                                 eta=0.001,
                                 beta=1e6,
                                 fft_len_self=fft_len_self,
                                 fft_len_wa=fft_len_wa)
            fine_steps += 1
        global_best = a
        # recompute the exact score for reporting
        c_final = _self_conv(global_best, fft_len_self)
        global_best_score = 2.0 * n * c_final.max()
        print(f"FINAL SCORE = {global_best_score:.6f} (len={n})")

    # ------------------------------------------------------------------
    # 5) Fallback (should never happen) and return
    # ------------------------------------------------------------------
    if global_best is None:
        # uniform fallback of a modest length
        n = 2000
        global_best = np.full(n, 1.0 / n, dtype=np.float64)

    # Safety clipping as required by `evaluate_sequence`
    global_best = np.clip(global_best, 0.0, 1000.0)

    return [float(x) for x in global_best]
