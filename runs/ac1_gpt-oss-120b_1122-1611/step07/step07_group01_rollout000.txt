"""
Hybrid Adam‑mirrored descent + stochastic local refinement for minimizing
the max self‑convolution of a non‑negative step function.
The algorithm works on the probability simplex (∑a_i = 1) and uses a
soft‑max (log‑sum‑exp) surrogate for the ∞‑norm.  An Adam‐style
adaptive step size on the log‑coordinates is combined with occasional
random block perturbations and a simple hill‑climbing phase that moves
tiny amounts of mass between two positions if it lowers the exact
objective.  Multiple restarts are triggered after a period of stagnation.
The routine respects the 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Basic convolution utilities (FFT based, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT of length ``fft_len``.
    Returns the full convolution of length ``len(x)+len(y)-1``.
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len_self: int) -> np.ndarray:
    """Self‑convolution a*a (full length)."""
    return _conv_fft(a, a, fft_len_self)


def _conv_wa(w: np.ndarray, a: np.ndarray, fft_len_wa: int) -> np.ndarray:
    """Convenient wrapper for w*a convolution used in the gradient."""
    return _conv_fft(w, a, fft_len_wa)


# ----------------------------------------------------------------------
#  Simple utilities
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a multiplicative log‑normal factor to a random contiguous block and
    re‑normalise to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Guarantees non‑negativity and sum equals 1.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.size
    # shift by max to avoid negative numbers before sorting
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _score_from_a(a: np.ndarray, fft_len_self: int, n: int) -> float:
    """
    Exact objective value 2·n·max(a*a) for a probability vector a.
    """
    c = _self_conv(a, fft_len_self)
    return 2.0 * n * c.max()


# ----------------------------------------------------------------------
#  Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 2) Initial probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            a = np.clip(arr, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            a = None
    else:
        a = None

    if a is None:
        n = 3800  # length used in the previous best construction
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT lengths (rounded up to a power of two)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    fft_len_wa = _next_pow2(3 * n)            # for w*a (w length = 2n‑1)

    # ------------------------------------------------------------------
    # 4) Adam‑style state on log‑coordinates
    # ------------------------------------------------------------------
    log_a = np.log(a + 1e-16)                 # avoid log(0)
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # ------------------------------------------------------------------
    # 5) Book‑keeping of the best solution
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = _score_from_a(a, fft_len_self, n)
    last_improve = time.monotonic()
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)   # 0 → 1

        # --------------------------------------------------------------
        # 6.1 – schedules for learning‑rate η and soft‑max concentration β
        # --------------------------------------------------------------
        lr = 0.5 * (1.0 - frac) + 0.001 * frac                # 0.5 → 0.001
        beta = 0.5 * (1.0 - frac) + 5000.0 * frac            # 0.5 → 5000

        # --------------------------------------------------------------
        # 6.2 – compute self‑convolution and soft‑max weights
        # --------------------------------------------------------------
        c = _self_conv(a, fft_len_self)                      # length 2n‑1
        max_c = c.max()
        # soft‑max (shifted by max_c for numerical stability)
        w_raw = np.exp(beta * (c - max_c))
        w = w_raw / w_raw.sum()                              # still length 2n‑1

        # --------------------------------------------------------------
        # 6.3 – gradient of the smoothed objective (w.r.t. a)
        # --------------------------------------------------------------
        conv_wa = _conv_wa(w, a, fft_len_wa)                 # length 3n‑2
        grad_a = 2.0 * conv_wa[n - 1:n - 1 + n]              # slice of length n

        # --------------------------------------------------------------
        # 6.4 – Adam update on log‑coordinates (mirror descent on simplex)
        # --------------------------------------------------------------
        t_iter += 1
        g_log = a * grad_a                                   # chain rule
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log ** 2)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # update
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # map back to probability vector (exponentiate and renormalise)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a[:] = 1.0 / n
        else:
            a /= a_sum

        # --------------------------------------------------------------
        # 6.5 – occasional random block perturbation (exploration)
        # --------------------------------------------------------------
        if rng.random() < 0.01:      # ~1 % of iterations
            a = _random_block_perturb(a, rng)
            log_a = np.log(a + 1e-16)   # keep log_a consistent

        # --------------------------------------------------------------
        # 6.6 – exact scoring check
        # --------------------------------------------------------------
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6.7 – stagnation handling
        # --------------------------------------------------------------
        if now - last_improve > 25.0:
            # restart from the best found so far, adding a modest jitter
            a = best_a.copy()
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            a /= a_sum if a_sum > 0 else (1.0 / n)
            log_a = np.log(a + 1e-16)
            # reset Adam statistics to avoid bias from the previous run
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart after stagnation – best {best_score:.6f}")

        # --------------------------------------------------------------
        # 6.8 – simple hill‑climbing refinement (every few seconds)
        # --------------------------------------------------------------
        if int((now - start_time) * 2) % 7 == 0:   # cheap periodic check
            # propose a tiny transfer of mass from i → j
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j and a[i] > 1e-8:
                delta = min(a[i] * 0.002, 0.001)   # small amount
                a_tmp = a.copy()
                a_tmp[i] -= delta
                a_tmp[j] += delta

                # exact score for the tentative move
                score_tmp = _score_from_a(a_tmp, fft_len_self, n)
                if score_tmp < best_score - 1e-12:
                    a = a_tmp
                    log_a = np.log(a + 1e-16)
                    best_score = score_tmp
                    best_a = a.copy()
                    last_improve = now
                    print(f"[{now - start_time:6.2f}s] hill‑climb improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to limits imposed by the
    #    evaluation function for safety).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
