"""
Propose a step‑function by a two‑stage mirror‑descent / hard‑max
gradient method with multi‑length exploration.

Stage 1 – for each chosen length `n` we run a fast mirror‑descent that
optimises a *soft‑max* approximation of the convolution peak.  The
temperature `α` is increased from 0.5 to a large value (≈80) while the
step size `η` is decayed from a few units to ≈1e‑3.  After a few
seconds the soft‑max is replaced by the exact peak (hard‑max) and we
continue with a standard exponentiated‑gradient update that directly
reduces the maximal autocorrelation entry.

Stage 2 – whenever progress stalls for >15 s a gentle “restart” is
performed (random Dirichlet draw).  Occasional block‑perturbations
(≈2 % of iterations) help escape flat regions.

The routine cycles through several problem sizes
`[128, 256, 512, 1024, 2048, 3809]` and keeps the best sequence
encountered over the whole budget.  All updates keep the ℓ₁‑norm
`∑a_i = √(2 n)`, which makes the evaluation score exactly the maximal
self‑convolution value, i.e. the quantity we minimise.

The implementation is pure NumPy, uses FFT‑based convolutions
(`O(n log n)`) and fits comfortably into the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT length ``fft_len``."""
    return _fft_convolve(a, a, fft_len)


def _grad_from_weights(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  ∑_k w_k·(a*a)_k  w.r.t. a.
    Using the identity  ∂/∂a_i ∑_k w_k (a*a)_k = 2·(a∗w)_{i},
    where the convolution (a∗w) is taken and the slice
    n‑1 … 2n‑2 corresponds to the gradient entries.
    """
    conv_full = _fft_convolve(a, w, fft_len)      # length = 3n‑2
    return 2.0 * conv_full[a.size - 1: 2 * a.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
        s = 1.0
    out *= (np.sqrt(2.0 * n) / s)
    return out


# ----------------------------------------------------------------------
# Core optimisation for a single length
# ----------------------------------------------------------------------


def _optimise_length(
    n: int,
    rng: np.random.Generator,
    time_budget: float,
    start_seq: np.ndarray | None = None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length `n` for at most `time_budget` seconds.
    Returns the best sequence (as a NumPy array) and its score
    (maximal self‑convolution, i.e. the evaluation value).
    """
    target_sum = np.sqrt(2.0 * n)

    # initialise ---------------------------------------------------------
    if start_seq is not None and start_seq.size == n:
        a = start_seq.copy()
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full(n, target_sum / n)
        else:
            a = a * (target_sum / a_sum)
    else:
        a = rng.dirichlet(np.ones(n) * 0.7).astype(np.float64)
        a *= target_sum

    # FFT lengths ---------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)   # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)   # for a*w

    # bookkeeping ---------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + time_budget
    best = a.copy()
    # compute initial score (max of self‑convolution)
    c = _self_conv(a, fft_len_self)
    best_score = c.max()
    cur_score = best_score
    last_improve = start_time
    accepted = 0

    # hyper‑parameters (linear schedules) --------------------------------
    eta_start = 5.0      # aggressive start
    eta_end = 0.001      # fine‑tuning
    alpha_start = 0.5
    alpha_end = 80.0     # very sharp soft‑max at the end of the warm‑up

    # optimisation loop --------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / time_budget                     # 0 … 1

        # ---------- compute convolution and weight vector ----------------
        c = _self_conv(a, fft_len_self)
        max_c = c.max()

        # warm‑up uses a soft‑max; after 30 % of time switch to hard‑max
        if frac < 0.30:
            alpha = alpha_start + (alpha_end - alpha_start) * (frac / 0.30)
            w = np.exp(alpha * (c - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum
        else:
            # hard‑max: weight is a Kronecker delta at the maximal index
            k_max = int(np.argmax(c))
            w = np.zeros_like(c)
            w[k_max] = 1.0

        # ---------- gradient of the (soft)‑max -------------------------
        g = _grad_from_weights(a, w, fft_len_grad)

        # ---------- step‑size schedule ---------------------------------
        eta = eta_start + (eta_end - eta_start) * frac

        # ---------- exponentiated‑gradient update -----------------------
        log_factor = -eta * g
        # protect against overflow/underflow in the exponential
        log_factor = np.clip(log_factor, -700, 700)
        factor = np.exp(log_factor)
        a_new = a * factor
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, target_sum / n)
        else:
            a_new = a_new * (target_sum / s)

        # ---------- evaluate candidate ----------------------------------
        c_new = _self_conv(a_new, fft_len_self)
        new_score = c_new.max()

        # ---------- simple acceptance rule (monotone) ------------------
        if new_score < cur_score:
            a = a_new
            cur_score = new_score
            accepted += 1
        else:
            # a tiny probabilistic acceptance to escape plateaus
            if rng.random() < 0.001:
                a = a_new
                cur_score = new_score
                accepted += 1

        # ---------- occasional random block perturb --------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            cur_score = _self_conv(a, fft_len_self).max()

        # ---------- keep best so far -----------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(
                f"[{n:4d}] {elapsed:6.2f}s – new best score = {best_score:.6f}"
            )

        # ---------- restart after stagnation ---------------------------
        if now - last_improve > 15.0:
            a = rng.dirichlet(np.ones(n) * 0.7).astype(np.float64) * target_sum
            cur_score = _self_conv(a, fft_len_self).max()
            last_improve = now
            accepted = 0
            print(f"[{n:4d}] {elapsed:6.2f}s – restart (stagnation)")

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine explores several sequence lengths, runs a fast mirror‑descent /
    hard‑max gradient optimiser for each, and returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global time budget (5 s safety margin)
    # ------------------------------------------------------------------
    end_time = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Lengths to explore – a mix of small and large problem sizes
    # ------------------------------------------------------------------
    lengths = [128, 256, 512, 1024, 2048, 3809]

    # If a previous candidate exists, keep it as a possible start for the
    # 3809‑length run (the last entry in ``lengths``).
    start_big = None
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            start_big = cand

    best_global_seq: np.ndarray | None = None
    best_global_score = float("inf")

    # ------------------------------------------------------------------
    # 2) Allocate time per length (roughly equal, but respect remaining)
    # ------------------------------------------------------------------
    remaining_lengths = len(lengths)
    for idx, n in enumerate(lengths):
        now = time.monotonic()
        remaining_time = end_time - now
        if remaining_time <= 5.0:               # safety cut‑off
            break

        # Guarantee a minimum of 30 s for each remaining problem, unless
        # we are close to the deadline.
        min_per_len = 30.0
        alloc = max(min_per_len, remaining_time / (remaining_lengths - idx))
        alloc = min(alloc, remaining_time - (remaining_lengths - idx - 1) * min_per_len)

        start_seq = None
        if n == lengths[-1] and start_big is not None:
            # use the previous best sequence as a starting point for the
            # longest run (down‑sample / up‑sample if necessary)
            if start_big.size == n:
                start_seq = start_big
            else:
                # simple interpolation / truncation to obtain the right size
                rise = np.linspace(0, start_big.size - 1, n)
                start_seq = np.interp(rise, np.arange(start_big.size), start_big)

        print(f"--- Optimising length {n} for {alloc:.1f}s ---")
        seq, score = _optimise_length(n, rng, alloc, start_seq)

        if score < best_global_score:
            best_global_score = score
            best_global_seq = seq.copy()
            print(
                f"***** New overall best (len={n}) – score = {best_global_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 3) Final fine‑tuning on the overall best (if any time left)
    # ------------------------------------------------------------------
    now = time.monotonic()
    if best_global_seq is not None and now < end_time - 5.0:
        leftover = end_time - now
        print(f"Fine‑tuning final sequence for {leftover:.1f}s")
        seq, score = _optimise_length(
            best_global_seq.size, rng, leftover, best_global_seq
        )
        if score < best_global_score:
            best_global_score = score
            best_global_seq = seq.copy()
            print(f"*** Improved final score = {best_global_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    if best_global_seq is None:
        # fallback – uniform sequence of a modest length
        n = 256
        a = np.full(n, np.sqrt(2.0 * n) / n)
        best_global_seq = a

    best_global_seq = np.clip(best_global_seq, 0.0, 1000.0)
    return [float(x) for x in best_global_seq]
