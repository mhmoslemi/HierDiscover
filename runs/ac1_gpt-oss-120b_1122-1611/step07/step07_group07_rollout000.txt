"""
Propose a step‑function that (hopefully) beats the previous 1.513208 bound.

Algorithm – Adam on a smoothed max‑convolution objective
---------------------------------------------------------

* The coefficients a_i are kept on the probability simplex via a soft‑max
  parametrisation a = softmax(y).  The objective 2·n·max(a*a) is invariant
  to scaling, so we may safely enforce Σ a_i = 1.

* At every iteration we form the self‑convolution c = a*a and a smooth
  approximation of its maximum:
        L_α(c) = (1/α)·log Σ_k exp(α·c_k)      (α – “inverse temperature”)

  The weights w_k = exp(α·(c_k−max_c))/Σ exp(·) are the soft‑max
  distribution over the convolution entries.  The gradient ∂L/∂a is
  the convolution of w with the reversed a (factor 2 because
  c_k = Σ_i a_i a_{k−i}).

* We back‑propagate through the soft‑max to obtain the gradient w.r.t.
  the logits y (the usual soft‑max Jacobian).  An Adam optimiser
  (β1=0.9, β2=0.999) updates y.  The temperature α is linearly increased
  from a modest value to a large one so the optimisation starts with a
  global, smooth landscape and finishes by approximating the true max.

* After the Adam phase we perform a short stochastic “mass‑transfer”
  refinement using the incremental convolution update from the reference
  implementation.  The best vector seen during the whole run is returned.

The routine respects the 1000 s budget (a 5 s safety margin) and
produces a list of non‑negative heights ready for evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to index ``j`` and update
    the convolution ``c`` incrementally.  Returns the new a and c.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # incremental update of the convolution (see reference implementation)
    c_new = c.copy()
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new

    DEFAULT_N = 1024                # a size that is fast yet expressive
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            # possibly down‑sample a too‑large vector
            if cand.size > 2000:
                step = max(1, cand.size // DEFAULT_N)
                cand = cand[::step]
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0.0:
                a0 = cand / s
            else:
                a0 = np.full_like(cand, 1.0 / cand.size)
            n = a0.size
            # initialise logits as log‑probabilities (add epsilon for stability)
            y = np.log(a0 + 1e-12)
        else:
            n = DEFAULT_N
            y = np.zeros(n, dtype=np.float64)
    else:
        n = DEFAULT_N
        y = np.zeros(n, dtype=np.float64)           # uniform start

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)     # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)     # for w * a_rev

    # ------------------------------------------------------------------
    # 3) Adam state and schedules
    # ------------------------------------------------------------------
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr_start = 0.5
    lr_end = 0.02

    alpha_start = 1.0       # soft‑max “temperature” at start
    alpha_end = 5000.0      # final (almost exact) max

    best_score = np.inf
    best_a = None
    it = 0
    last_print = 0.0

    # ------------------------------------------------------------------
    # 4) Main Adam optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # ----- a = soft‑max(y) (probability simplex) -----
        max_y = np.max(y)
        exp_y = np.exp(y - max_y)                 # subtract max for stability
        a = exp_y / np.sum(exp_y)

        # ----- self‑convolution and current score -----
        c = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * c.max()

        # ----- keep global best -----
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- temperature (smooth‑max) schedule -----
        alpha = alpha_start + frac * (alpha_end - alpha_start)

        # ----- soft‑max weights over the convolution -----
        c_max = c.max()
        w = np.exp(alpha * (c - c_max))           # max entry gets weight 1
        w_sum = w.sum()
        if w_sum == 0.0:                           # pathological – fall back
            w = np.zeros_like(w)
            w[int(np.argmax(c))] = 1.0
            w_sum = 1.0
        w = w / w_sum

        # ----- gradient w.r.t. a (smooth max) -----
        a_rev = a[::-1]
        fw = np.fft.rfft(w, n=fft_len_grad)
        fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
        conv_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: 3 * n - 2]
        grad_a = 2.0 * conv_full[n - 1: n - 1 + n]   # length n

        # ----- gradient w.r.t. logits y (soft‑max Jacobian) -----
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)

        # ----- Adam update on y -----
        lr = lr_start * (1.0 - frac) + lr_end * frac
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional jitter to escape plateaus -----
        if it % 5000 == 0:
            y += rng.normal(scale=0.02, size=n)

        # ----- lightweight progress output (once per 30 s) -----
        if elapsed - last_print > 30.0:
            last_print = elapsed
            print(f"[{elapsed:6.2f}s] current score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Local stochastic refinement (mass‑transfer moves)
    # ------------------------------------------------------------------
    if best_a is None:                     # safety fallback
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = 2.0 * n * _self_conv(best_a, fft_len_self).max()

    a = best_a.copy()
    c = _self_conv(a, fft_len_self)
    cur_score = best_score
    move_counter = 0
    RECOMPUTE_EVERY = 250

    while time.monotonic() < deadline:
        # pick a donor with non‑trivial mass
        donor = rng.integers(0, n)
        if a[donor] < 1e-12:
            continue
        # pick a recipient (different from donor)
        recipient = rng.integers(0, n - 1)
        if recipient >= donor:
            recipient += 1

        # random fraction of donor mass to move
        delta = rng.random() * 0.5 * a[donor]
        if delta <= 0.0:
            continue

        a_new, c_new = _apply_move(a, c, donor, recipient, delta)
        new_score = 2.0 * n * c_new.max()

        if new_score < cur_score:
            a, c, cur_score = a_new, c_new, new_score
            move_counter += 1
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] refinement improved score = {best_score:.6f}")

            if move_counter % RECOMPUTE_EVERY == 0:
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
