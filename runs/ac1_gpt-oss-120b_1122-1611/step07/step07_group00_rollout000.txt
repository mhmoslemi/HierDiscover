   # If previous global variable height_sequence_1 exists, we can upsample:
   def resize_seq(seq, new_n):
       old_n = len(seq)
       # use linear interpolation
       x_old = np.linspace(0, 1, old_n, endpoint=False)
       x_new = np.linspace(0, 1, new_n, endpoint=False)
       seq_res = np.interp(x_new, x_old, seq)
       seq_res = np.clip(seq_res, 0, None)
       s = seq_res.sum()
       if s <= 0:

       else:
           seq_res /= s
       return seq_res

   v = np.zeros_like(x)

   while time.monotonic() < deadline:

"""
Propose a non‑negative step‑function (probability vector) that minimizes
evaluate_sequence.  

The algorithm works on the probability simplex by parametrising the heights
with log‑weights `x` (soft‑max of `x` gives the vector `a`).  A smooth
approximation of the max‑operator – the log‑sum‑exp with a temperature
parameter `γ` – is used together with a simple Adam optimiser.  The
temperature is increased linearly, so the optimisation first sees a smooth
landscape and later concentrates on the true maximum of the self‑convolution.
 
To avoid getting stuck in local minima the routine performs
*periodic restarts* (adding Gaussian noise to the log‑weights) and
*occasional jitter* (multiplicative log‑normal perturbation of a random
contiguous block).  When the allocated time is almost exhausted a short
simulated‑annealing phase (random block/exponent perturbations) is run to
refine the best solution found.

A larger problem size (default `n = 8192`) is used – theoretically the
optimal value is the ratio `max(conv) / avg(conv)`, which tends to 1 as the
sequence length grows; with enough degrees of freedom the method can push
the bound below the previous best 1.5132.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – keep them at module level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns a real array of length 2*len(a)-1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed max‑objective w.r.t. the probability vector ``a``.
    ``w`` is a probability vector over the convolution indices (length 2*n-1).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(a_rev.size + w.size - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    # choose a block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10) + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _exponent_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Raise the vector to a random exponent p∈[0.5,1.5] and renormalise.
    This explores more global shape changes.
    """
    p = rng.uniform(0.5, 1.5)
    out = np.clip(a, 0.0, None) ** p
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / a.size
    else:
        out /= s
    return out


def _resize_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linearly interpolate ``seq`` to length ``new_n`` and renormalise to sum=1.
    """
    old_n = seq.size
    if old_n == new_n:
        out = seq.copy()
    else:
        x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
        out = np.interp(x_new, x_old, seq)
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / new_n
    else:
        out /= s
    return out

# Main optimisation routine

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose problem size
    # ------------------------------------------------------------------
    # Larger n gives more degrees of freedom – we aim for n≈8192 when
    # the budget permits.  For very short budgets we fall back to smaller
    # sizes to allow enough optimisation steps.
    if budget_s >= 600:
        n = 8192
    elif budget_s >= 300:
        n = 4096
    else:
        n = 2048

    # ------------------------------------------------------------------
    # 2) Initialise from a previously found construction if available,
    #    otherwise start from uniform (or a random Dirichlet draw).
    # ------------------------------------------------------------------
    base_seq = globals().get("height_sequence_1")
    if isinstance(base_seq, (list, tuple, np.ndarray)):
        base_arr = np.asarray(base_seq, dtype=np.float64)
        if base_arr.ndim == 1 and base_arr.size > 0:
            a = _resize_seq(base_arr, n)
        else:
            a = np.full(n, 1.0 / n)

        a = np.full(n, 1.0 / n)

    # Log‑weights representation (soft‑max will enforce the simplex)
    x = np.log(a + 1e-12)

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT length for the self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 4) Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 5) Tracking the best solution
    # ------------------------------------------------------------------
    a_curr = _softmax(x)
    c_curr = _self_conv(a_curr, fft_len_self)
    cur_score = 2.0 * n * c_curr.max()
    best = a_curr.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 6) Hyper‑parameters (schedules)
    # ------------------------------------------------------------------
    gamma_start = 0.4               # low smoothing at the beginning
    gamma_end = 120.0                # high smoothing → true max
    lr_start = 0.8
    lr_end = 0.03

    # ------------------------------------------------------------------
    # 7) Main optimisation loop (gradient descent on the smoothed max)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_time = elapsed / total_time

        # ----- probability vector from log‑weights -----
        a = _softmax(x)

        # ----- self‑convolution -----
        c = _self_conv(a, fft_len_self)          # length 2·n‑1

        # ----- smooth‑max weights (log‑sum‑exp) -----
        gamma = gamma_start + (gamma_end - gamma_start) * frac_time
        c_shift = c - np.max(c)                  # for numerical stability
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a -----
        g_a = _gradient_wrt_a(a, w)              # shape (n,)

        # ----- chain rule to log‑weights (soft‑max Jacobian) -----
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr_start * (1.0 - frac_time) + lr_end * frac_time
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        x -= step

        # ----- exact evaluation (no smoothing) -----
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----- restart on stagnation (30 s without improvement) -----
        if now - last_improve > 30.0:
            # Re‑initialise log‑weights from the current best plus moderate noise
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # ----- occasional global jitter (≈2 % of iterations) -----
        if rng.random() < 0.02:
            # apply a block‑perturbation to the best solution and restart
            pert = _random_block_perturb(best, rng)
            x = np.log(pert + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 8) Final short simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 1.0 and best_score < np.inf:
        # start from the best solution found so far
        a_cur = best.copy()
        cur_score = 2.0 * n * _self_conv(a_cur, fft_len_self).max()
        T0 = 0.015                     # initial temperature
        anneal_start = time.monotonic()
        while time.monotonic() < deadline:
            now = time.monotonic()
            frac = (now - anneal_start) / remaining
            T = T0 * (1.0 - frac)          # linear cooling

            # propose a random move: block perturb or exponent perturb
            if rng.random() < 0.5:
                a_new = _random_block_perturb(a_cur, rng)
            else:
                a_new = _exponent_perturb(a_cur, rng)

            new_score = 2.0 * n * _self_conv(a_new, fft_len_self).max()
            delta = new_score - cur_score
            if delta < 0.0 or rng.random() < np.exp(-delta / (T + 1e-16)):
                a_cur = a_new
                cur_score = new_score
                if cur_score < best_score:
                    best_score = cur_score
                    best = a_cur.copy()
                    print(f"[{now - start_time:6.2f}s] annealing improvement: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 9) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
