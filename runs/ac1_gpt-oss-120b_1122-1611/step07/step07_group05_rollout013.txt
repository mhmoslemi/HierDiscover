def _linear_conv(x, y, fft_len):
    fx = np.fft.rfft(x, n=fft_len)

"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm (≈1000 s total)
-------------------------
* **Initialisation** – start from a known solution (global variable
  ``height_sequence_1``) if it exists, otherwise use a uniform simplex
  vector of length *n* (default 4000).

* **Phase 1 – smoothed‑max gradient descent** (≈40 % of the time)  
  The objective  ``max(a*a)`` is approximated by a smooth log‑sum‑exp
  ``μ·log(∑exp(c/μ))`` where ``c = a*a``.  Its gradient is
  ``2·(w_rev * a)`` with ``w = softmax(c/μ)``.  A step
  ``a ← a – η·grad`` is taken and projected back onto the probability
  simplex.  The step‑size ``η`` and the smoothing parameter ``μ`` are
  decreased linearly during this phase.

* **Phase 2 – focussed simulated annealing** (≈60 % of the time)  
  The current peak index ``k_max`` of the convolution is located.
  The index ``i`` that contributes most to this peak
  (maximising ``a_i·a_{k_max‑i}``) is chosen as *donor*.
  The *recipient* is taken among the indices whose partner value
  ``a_{k_max‑j}`` is smallest (i.e. moving mass there has little impact
  on the peak).  A random fraction of the donor mass is transferred,
  the convolution is updated incrementally, and the new state is
  accepted by a Metropolis rule with a linearly‑cooling temperature.
  Full recomputation of the convolution corrects drift, and occasional
  log‑normal jitter escapes plateaus.

The best sequence seen so far is returned (clipped to the limits imposed
by ``evaluate_sequence``).  All helper functions are top‑level and the
routine respects a 5 s safety margin.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``."""
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution ``a*a`` using an FFT."""
    return _conv_fft(a, a, fft_len)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally. Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # linear part (subtract donor contribution, add recipient contribution)
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a simplex projection.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out

                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

                # occasional global jitter to escape plateaus
                if rng.random() < 0.02:
                    a = _random_block_perturb(a, rng)
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

        # ----------------------------------------------------------
        # Keep the best solution seen so far
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Soft restart after prolonged stagnation
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # re‑initialise from the best with moderate log‑normal noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart with jitter (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
