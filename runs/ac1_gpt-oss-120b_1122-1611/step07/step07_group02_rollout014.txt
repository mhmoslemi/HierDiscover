def objective(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    fft_len = 1 << ((2*n - 2).bit_length())  # next pow2 >= 2n-1
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
    maxb = conv.max()
    sum_a = a.sum()
    return 2.0 * n * maxb / (sum_a * sum_a)

import time, numpy as np, cvxpy as cp

def _next_pow2(x): return 1 << ((x-1).bit_length())

def objective(a):
  a = np.asarray(a, dtype=np.float64)
  n = a.shape[0]
  fft_len = _next_pow2(2*n-1)
  a_fft = np.fft.rfft(a, n=fft_len)
  conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
  maxb = conv.max()
  sa = a.sum()
  if sa < 0.01:
    return np.inf
  return float(2.0 * n * maxb / (sa*sa))

def build_convolution_matrix(a):
  n = a.shape[0]
  mat = np.zeros((2*n-1, n), dtype=np.float64)
  for j in range(n):
      mat[j:j+n, j] = a
  return mat

def perturb(a, rng, sigma):
  # Add gaussian noise proportional to a
  n = a.shape[0]
  # generate noise, allow moving mass between coordinates
  noise = rng.normal(scale=sigma, size=n)
  a_new = a + noise * a
  a_new = np.clip(a_new, 0.0, None)
  total = a_new.sum()
  if total <= 0:
    a_new = np.full_like(a, 1.0/n)
  else:
    a_new /= total
  return a_new

def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
   """Search for a step-function that minimizes evaluate_sequence.
   
   The algorithm performs a stochastic local search followed by an
   exact linear‑programming refinement based on the iterative scheme
   described in the problem statement.
   """
   rng = np.random.default_rng(seed)
   start = time.monotonic()
   deadline = start + max(0.0, float(budget_s) - 5.0)

   # initialise a
   if 'height_sequence_1' in globals():
       base = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
       base = np.clip(base, 0.0, None)
       s = base.sum()
       if s == 0:
           base[:] = 1.0 / base.size
       else:
           base = base / s
       n = max(1024, base.size)
       if base.size != n:
           # linear interpolation to size n
           old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
           new_x = np.linspace(0.0, 1.0, n, endpoint=False)
           base = np.interp(new_x, old_x, base)
           base = np.clip(base, 0.0, None)
           base = base / base.sum()
       a = base.copy()
   else:
       n = 1024
       a = np.full(n, 1.0 / n, dtype=np.float64)

   # ensure sum=1
   a = np.clip(a, 0.0, None)
   a = a / a.sum()

   best_a = a.copy()
   best_score = objective(a)
   print(f"[0.00s] initial score = {best_score:.6f}")

   # ---- Stage 1: stochastic local search ----
   # allocate 30% of time for this stage
   stage1_deadline = start + 0.3 * (deadline - start)
   sigma = 0.5   # starting perturbation scale
   sigma_decay = 0.9995   # per iteration
   while time.monotonic() < stage1_deadline:
       a_candidate = perturb(a, rng, sigma)
       sigma *= sigma_decay
       cand_score = objective(a_candidate)
       if cand_score < best_score - 1e-12:
           best_score = cand_score
           best_a = a_candidate.copy()
           print(f"[{time.monotonic()-start:6.2f}s] new best (stage1) = {best_score:.6f}")
       # accept if better or otherwise keep the best as current
       if cand_score < objective(a):
           a = a_candidate

   # ---- Stage 2: LP refinement ----
   # hyper‑parameters
   t = 0.2                # convex‑combination weight
   restart_wait = 30.0    # seconds without improvement
   last_improve = time.monotonic()
   while time.monotonic() < deadline:
       # compute current max convolution
       n = a.shape[0]
       fft_len = _next_pow2(2*n-1)
       a_fft = np.fft.rfft(a, n=fft_len)

       for j in range(n):

       b = cp.Variable(n, nonneg=True)

       prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
       try:
           prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
       except Exception as e:
           print(f"LP solve failed: {e}")
           break

       if b.value is None:
           print("LP returned no solution, terminating.")
           break

       b_val = np.array(b.value, dtype=np.float64)
       # combine
       a_new = (1.0 - t) * a + t * b_val
       # renormalise
       s = a_new.sum()
       if s <= 0:

           a_new = a_new / s

       cur_score = objective(a_new)

           best_score = cur_score
           best_a = a_new.copy()

   # final clipping
   best_a = np.clip(best_a, 0.0, 1000.0)
   return [float(v) for v in best_a]

def max_convolution(a):
    n = a.shape[0]
    fft_len = 1 << ((2*n - 2).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    c = np.fft.irfft(A * A, n=fft_len)[:2*n-1]
    return float(c.max())

def compute_score(a):
    maxc = max_convolution(a)

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The search is a three‑stage hybrid:

1. **Heuristic seeding** – try a few analytically motivated windows
   (uniform, Hann, Blackman, Gaussian) and a handful of random Dirichlet
   draws.  Keep the best of these as a starting point.

2. **Simulated‑annealing hill climbing** –  
   repeatedly move a small amount of mass from one index to another,
   accepting worsening moves with a temperature that cools linearly.
   This quickly explores the neighbourhood of the best seed.

3. **Linear‑programming refinement** – an iteration of the scheme from the
   problem statement.  For a current vector ``a`` we solve

        maximise   sum(b)
        subject to  conv(b, a) ≤ M   ,  b ≥ 0

   where ``M = max(conv(a, a))``.  The solution ``b`` is mixed with ``a``
   (convex combination) and renormalised.  Repeating this step strictly
   reduces the objective (or leaves it unchanged) and drives the sequence
   to a better local optimum.

The algorithm works with a moderate length (default 1024) – large enough
to obtain a good bound while keeping each LP cheap (≈ 2 MiB matrix).  The
total running time respects the 1000 s budget with a 5 s safety margin.
"""
import time
import numpy as np
import cvxpy as cp
from scipy.signal import windows

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (no clipping, no checks).

    a must be a 1‑D non‑negative array, sum(a) > 0.
    Returns 2·n·max(conv(a, a)) / (sum(a)**2).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    max_conv = float(conv.max())
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a, a)) – the part inside the objective."""
    a = np.asarray(a, dtype=np.float64)
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    return float(conv.max())


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix C such that C @ b = conv(b, a).
    Entry C[k, j] = a[k‑j] if 0 ≤ k‑j < n else 0.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    # Each column j is a shifted copy of ``a``.
    for j in range(n):
        C[j : j + n, j] = a
    return C


def _perturb_move(a: np.ndarray, rng: np.random.Generator,
                  strength: float = 0.1) -> np.ndarray:
    """
    Move a random fraction of mass from one index to another.
    ``strength`` controls the typical fraction transferred (0 < strength ≤ 1).
    """
    n = a.shape[0]
    # pick donor with some mass
    donor = rng.integers(n)
    while a[donor] == 0.0:
        donor = rng.integers(n)
    # amount to move: a fraction of the donor's current mass
    delta = a[donor] * rng.uniform(0.0, strength)
    recipient = rng.integers(n)
    while recipient == donor:
        recipient = rng.integers(n)

    a_new = a.copy()
    a_new[donor] -= delta
    a_new[recipient] += delta
    # re‑normalise to keep sum exactly 1 (numerical drift may appear)
    total = a_new.sum()
    if total > 0.0:
        a_new /= total
    else:
        a_new[:] = 1.0 / n
    return a_new


def _random_dirichlet(rng: np.random.Generator, n: int,
                     concentration: float = 0.6) -> np.ndarray:
    """
    Sample a Dirichlet vector of length ``n`` with the given concentration
    (all alphas equal).  Returned vector sums to 1.
    """
    # Gamma shape == concentration, scale == 1
    x = rng.gamma(concentration, 1.0, size=n)
    total = x.sum()
    if total <= 0.0:
        return np.full(n, 1.0 / n)
    return x / total


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method combines deterministic seeding,
    simulated‑annealing hill‑climbing and a linear‑programming refinement
    inspired by the iterative scheme described in the problem statement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Choose a length n and an initial distribution.
    # ------------------------------------------------------------------
    DEFAULT_N = 1024
    # If a previously discovered construction exists, down‑sample it.
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        old = np.clip(old, 0.0, None)
        # linear interpolation to the new size
        n = max(DEFAULT_N, old.size)
        old_x = np.linspace(0.0, 1.0, old.size, endpoint=False)
        new_x = np.linspace(0.0, 1.0, n, endpoint=False)
        a0 = np.interp(new_x, old_x, old)
    else:
        n = DEFAULT_N
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise to a probability vector (sum = 1)
    a0 = np.clip(a0, 0.0, None)
    if a0.sum() == 0.0:
        a0[:] = 1.0 / n
    else:
        a0 /= a0.sum()

    best_a = a0.copy()
    best_score = _objective(best_a)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – deterministic seeds (window functions & few Dirichlet draws)
    # ------------------------------------------------------------------
    seeds = []

    # uniform (already in a0)
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Hann window
    hann = windows.hann(n, sym=False)  # non‑negative, sum ≈ n/2
    hann = np.maximum(hann, 0.0)
    seeds.append(hann / hann.sum())

    # Blackman window (more aggressive taper)
    black = windows.blackman(n, sym=False)
    black = np.maximum(black, 0.0)
    seeds.append(black / black.sum())

    # Gaussian‑like window
    x = np.arange(n) - (n - 1) / 2.0
    sigma = 0.3 * n
    gauss = np.exp(-0.5 * (x / sigma) ** 2)
    seeds.append(gauss / gauss.sum())

    # A few random Dirichlet samples with varied concentration
    for conc in (0.3, 0.6, 1.2, 2.0):
        for _ in range(5):
            d = _random_dirichlet(rng, n, concentration=conc)
            seeds.append(d)

    # Evaluate all seeds and keep the best
    for s in seeds:
        sc = _objective(s)
        if sc < best_score:
            best_score = sc
            best_a = s.copy()
            print(f"[{time.monotonic() - start:6.2f}s] seed improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    # Allocate roughly 30 % of the remaining budget for SA.
    sa_deadline = start + 0.30 * (deadline - start)
    temperature_start = 0.05
    temperature_end = 1e-6

    cur_a = best_a.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    it = 0
    while time.monotonic() < sa_deadline:
        it += 1
        # linear cooldown
        progress = (time.monotonic() - start) / (sa_deadline - start)
        T = temperature_start * (1.0 - progress) + temperature_end * progress

        cand_a = _perturb_move(cur_a, rng, strength=0.12)
        cand_score = _objective(cand_a)

        delta = cand_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            cur_a = cand_a
            cur_score = cand_score

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = cur_a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] SA new best → {best_score:.6f}")

        # occasional restart if we have been stuck for a while
        if time.monotonic() - last_improve > 30.0:
            cur_a = _perturb_move(best_a, rng, strength=0.30)
            cur_score = _objective(cur_a)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] SA restart")

    # ------------------------------------------------------------------
    # Stage 3 – linear‑programming refinement (the scheme from the statement)
    # ------------------------------------------------------------------
    # Parameters for the LP phase
    t_mix = 0.25          # convex‑combination weight (1‑t)*a + t*b
    restart_wait = 35.0   # seconds without improvement → restart

    # Warm‑start the LP phase with the best vector found so far.
    cur_a = best_a.copy()
    cur_score = best_score
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        # Compute current maximal self‑convolution value M
        M = _max_convolution(cur_a)

        # Build the convolution matrix for the current ``a``
        C = _build_conv_matrix(cur_a)

        # Linear programme: maximise sum(b) subject to C @ b ≤ M and b ≥ 0
        nvar = cur_a.shape[0]
        b = cp.Variable(nvar, nonneg=True)
        constraints = [C @ b <= M + 1e-12]   # a tiny slack for numerical safety
        prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

        try:
            prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
        except Exception as exc:            # pragma: no cover – solver failures are rare
            print(f"LP solve failed: {exc}")
            break

        if b.value is None:
            # Solver could not find a feasible point – abort the LP phase.
            print("LP returned no solution; terminating LP refinement.")
            break

        b_val = np.asarray(b.value, dtype=np.float64)

        # Convex combination and re‑normalisation
        new_a = (1.0 - t_mix) * cur_a + t_mix * b_val
        s = new_a.sum()
        if s > 0.0:
            new_a /= s
        else:
            new_a = np.full_like(new_a, 1.0 / nvar)

        new_score = _objective(new_a)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = new_a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] LP improvement → {best_score:.6f}")

        # Accept the new point (even if not better – the LP step never hurts)
        cur_a = new_a
        cur_score = new_score

        # Restart if stagnant
        if time.monotonic() - last_improve > restart_wait:
            cur_a = _perturb_move(best_a, rng, strength=0.30)
            cur_score = _objective(cur_a)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] LP restart")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
