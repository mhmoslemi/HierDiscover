    b_max = b.max()

"""
Propose a step‑function vector that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Problem reformulation** – we keep the vector on the probability simplex
   (sum = 1).  For a simplex‑normalised vector the evaluation reduces to  
   ``2·n·max(conv(a,a))``.  Hence we minimise the (weighted) max of the
   self‑convolution.

2. **Smooth max** – replace ``max(b)`` by a smooth maximum
   ``smooth_max(b, λ) = max(b) + log Σ exp(λ·(b‑max(b))) / λ``.
   Its gradient is a convex combination of the individual gradients
   of the convolution entries, which can be computed by a single FFT.

3. **Projected Adam** – we perform Adam optimisation on the smooth
   objective, projecting the iterates onto the probability simplex after each
   step.  The smoothing parameter λ grows exponentially from a small start
   value to a large final value, thereby turning the smooth objective into
   the true max‑objective.

4. **Stagnation handling** – if no improvement is observed for a while,
   the current best vector is jittered multiplicatively (log‑normal noise)
   and optimisation continues with a reset Adam state.

The routine respects the 1000 s time budget (keeps a 5 s safety margin)
and prints progress.  The final list is clipped to the limits enforced by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(u: np.ndarray, v: np.ndarray) -> np.ndarray:
    """Linear convolution of two 1‑D real vectors using FFT."""
    n = u.size + v.size - 1
    size = _next_pow2(n)
    fu = np.fft.rfft(u, n=size)
    fv = np.fft.rfft(v, n=size)
    conv = np.fft.irfft(fu * fv, n=size)
    return conv[:n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _smooth_max_and_grad(a: np.ndarray, lam: float) -> tuple[float, np.ndarray]:
    """
    For a non‑negative vector ``a`` (∑a = 1) compute

        smooth_max = max(a∗a)   (smooth approximation with parameter lam)
        grad        = ∂smooth_max/∂a   (vector of same length as a)

    Both quantities are obtained via FFTs.  The gradient uses the fact that
    ∂(a∗a)_k / ∂a_i = 2·a_{k‑i} when the index is valid.
    """
    n = a.size
    # full self‑convolution
    b = _conv_full(a, a)                     # length 2n‑1
    b_max = b.max()
    # smooth maximum (log‑sum‑exp trick for stability)
    shifted = b - b_max
    exp_vals = np.exp(lam * shifted)
    sum_exp = exp_vals.sum()
    # weights that sum to 1 and concentrate on large entries of b
    w = exp_vals / sum_exp                    # length 2n‑1
    smooth_max = b_max + np.log(sum_exp) / lam

    # gradient: 2 * (w * a_rev) convolved, sliced to length n
    a_rev = a[::-1]
    size = _next_pow2(w.size + a_rev.size - 1)
    fw = np.fft.rfft(w, n=size)
    fa_rev = np.fft.rfft(a_rev, n=size)
    conv = np.fft.irfft(fw * fa_rev, n=size)
    grad = 2.0 * conv[n - 1 : n - 1 + n]     # shape (n,)

    return smooth_max, grad


def _jitter(v: np.ndarray, rng: np.random.Generator, sigma: float = 0.03) -> np.ndarray:
    """Multiplicative log‑normal jitter followed by simplex projection."""
    out = v * np.exp(rng.normal(0.0, sigma, size=v.shape))
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initialise – use a known construction if present, otherwise uniform
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a.ndim != 1 or a.size == 0 or not np.isfinite(a).all():
            n = 6000
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
    else:
        n = 6000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # --------------------------------------------------------------
    # 2) Adam state for the smooth‑max objective
    # --------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v_acc = np.zeros_like(a)      # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12

    # --------------------------------------------------------------
    # 3) Scheduling parameters
    # --------------------------------------------------------------
    total_time = deadline - time.monotonic()
    start_time = time.monotonic()

    lam_start = 0.5               # start of smoothing parameter
    lam_end = 200.0               # final (approaches true max)

    lr_start = 0.5                # initial learning rate
    lr_end = 0.001               # final learning rate

    # --------------------------------------------------------------
    # 4) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    last_improve = start_time
    it = 0

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        frac_time = elapsed / total_time if total_time > 0 else 1.0

        # ----- smoothing parameter (exponential schedule) -----
        lam = lam_start * (lam_end / lam_start) ** frac_time

        # ----- smooth max and its gradient -----
        smooth_val, grad = _smooth_max_and_grad(a, lam)

        # ----- learning‑rate schedule (linear) -----
        lr = lr_start * (1.0 - frac_time) + lr_end * frac_time

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v_acc = beta2 * v_acc + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_acc / (1.0 - beta2 ** it)

        step = lr / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat

        # ----- keep a on the probability simplex -----
        a = _project_simplex(a)

        # ----- periodic bookkeeping (every ~0.2 s of wall‑time) -----
        if it % 200 == 0:
            cur_score = evaluate_sequence(a.tolist())
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] iteration {it:6d}  new best = {best_score:.6f}")

        # ----- stagnation handling -----
        if now - last_improve > 30.0:
            # reset from best with gentle jitter
            a = _jitter(best, rng, sigma=0.05)
            m.fill(0.0)
            v_acc.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
