Define:

def _project_onto_simplex(v):
    # ensure sum=1 and nonnegative
    # algorithm as in https://arxiv.org/pdf/1101.6081.pdf

    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    v = np.zeros_like(a)

    # schedule

    # temperature schedule for surrogate
    eps_start = None # to be initial set later

    # loop
    iter_count = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        if now < phase1_end:
            # compute conv
            c = _self_conv(a, fft_len_self)
            # update temperature eps
            # maybe linearly decay from max(c) * 0.2 to 1e-12.
            # compute dynamic range: we can use safety.
            range_c = c.max() - c.min()

            phase_progress = (now - start_time) / (phase1_end - start_time)

            # compute softmax weight

            v = beta2 * v + (1 - beta2) * (grad**2)

            else:

            if recv_sum > 0:

            else:

                continue

            max_delta = 0.5 * a[donor]
            delta = rng.random() * max_delta

            a_new, c_new = _apply_move(a, c, donor, recipient, delta)

            # acceptance schedule

            # but sometimes accept slightly worse to escape local minima.
            temp_progress = (now - phase1_end) / (deadline - phase1_end)

                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

                    best = a.copy()
                    last_improve = now
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] new best (stoch) = {best_score:.6f}")

            # optional jitter:

                # apply random block perturbation as before
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        if now - last_improve > 30.0:

            a = best.copy()

            last_improve = now

    best = np.clip(best, 0.0, 1000.0)
    # Ensure sum > 0.01 constraint: if sum too small, scale up to 1 (but sum already 1)
    return [float(x) for x in best]

"""
Propose a step‑function that (hopefully) beats the previous bound.

Algorithm
---------
* Initialise a probability vector `a` (the step heights) – use the
  previously found best sequence if it exists, otherwise start with a
  uniform distribution of length 4096.

* **Phase 1 – Adam mirror descent on a smooth max surrogate**.
  The self‑convolution `c = a*a` is softened with a soft‑max
  temperature `eps`.  The gradient of the surrogate
  `eps*log Σ exp(c/eps)` w.r.t. `a` can be written as a convolution of
  the soft‑max weights with the reversed `a`.  An Adam optimiser
  (first‑ and second‑moment adaptation) follows this gradient.  After
  each step the vector is projected onto the probability simplex.

* **Phase 2 – Stochastic mass‑transfer refinement**.
  A Metropolis‑style move transfers a random amount `δ` from a “donor”
  index (chosen proportionally to its contribution to the current
  maximal convolution entry) to a “recipient” index (chosen favouring
  small entries).  The move is accepted if it improves the true score
  or with a temperature‑dependent probability.

* **Restarts** – if no improvement is seen for 30 s, the current best
  solution is perturbed with log‑normal noise and optimisation resumes.

The whole routine respects the 1000 s budget (5 s safety margin) and
always returns a feasible, non‑negative sequence that can be fed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new (a, c)."""
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # subtract contribution of donor i, add contribution of recipient j
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ terms
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            cand = None
    else:
        cand = None

    if cand is None:
        # Default length – large enough to give a fine resolution
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = cand.size
        a = np.full(n, 1.0 / n, dtype=np.float64) if n == 0 else a

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 3) Book‑keeping for the current state
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)                # current self‑convolution
    cur_score = 2.0 * n * c.max()                  # sum(a)=1 ⇒ denominator = 1
    best = a.copy()
    best_score = cur_score
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4) Optimisation parameters
    # ------------------------------------------------------------------
    # Phase allocation (≈80 % gradient descent, ≈20 % stochastic)
    total_time = deadline - start_time
    phase1_end = start_time + 0.80 * total_time

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Stochastic‐refinement temperature schedule
    temp_start = 0.20
    temp_end = 1e-6
    recompute_every = 250
    move_counter = 0

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase 1 – Adam on soft‑max surrogate of the max‑convolution
        # --------------------------------------------------------------
        if now < phase1_end:
            # recompute convolution (needed for the surrogate)
            c = _self_conv(a, fft_len_self)

            # temperature for the soft‑max (smooth approximation of max)
            range_c = c.max() - c.min()
            prog = (now - start_time) / (phase1_end - start_time)
            eps = max(1e-12,
                      range_c * (0.4 * (1.0 - prog) + 0.02))

            # soft‑max weights
            c_max = c.max()
            w = np.exp((c - c_max) / eps)
            w /= w.sum()

            # gradient of the surrogate w.r.t. a
            a_rev = a[::-1]
            fw = np.fft.rfft(w, n=fft_len_grad)
            fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv_full = np.fft.irfft(fw * fa_rev,
                                     n=fft_len_grad)[: 3 * n - 1]
            grad = 2.0 * conv_full[n - 1:n - 1 + n]

            # Adam update
            it += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # linear decay of the nominal step size
            lr = 0.6 * (1.0 - prog) + 0.01 * prog
            step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            a = a - step
            # keep a strictly positive before projecting
            a = np.maximum(a, 1e-12)
            a = _project_onto_simplex(a)

            # true score (no surrogate)
            cur_score = 2.0 * n * c.max()

            # keep best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best (grad) = {best_score:.6f}")

        # --------------------------------------------------------------
        # Phase 2 – fast stochastic refinement (simulated annealing)
        # --------------------------------------------------------------
        else:
            # we keep `c` from the last iteration; recompute periodically
            if move_counter % recompute_every == 0:
                c = _self_conv(a, fft_len_self)

            # locate current peak of the convolution
            k_max = int(c.argmax())

            # contribution of each index to that peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                # donor essentially empty – skip this iteration
                continue

            # recipient favours small entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            max_delta = 0.5 * a[donor]
            delta = rng.random() * max_delta

            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # temperature schedule for Metropolis acceptance
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = max(temp_start * (1.0 - t_frac) + temp_end * t_frac,
                              1e-12)

            accept = False
            if new_score < cur_score:
                accept = True
            else:
                if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                    accept = True

            if accept:
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = now
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] new best (stoch) = {best_score:.6f}")

            # occasional global jitter to escape plateaus
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * noise
            a = _project_onto_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            it = 0
            m.fill(0.0)
            v.fill(0.0)
            print(f"[{now - start_time:6.2f}s] restart after stagnation – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped for safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
