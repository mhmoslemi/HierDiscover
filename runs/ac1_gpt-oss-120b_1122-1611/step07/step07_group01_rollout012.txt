def simplex_projection(v):
    # assumes v is a 1D array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that improves the bound on
evaluate_sequence().  The method combines two complementary
gradient directions:

* **Uniform‑flattening gradient** – minimise the squared L2‑distance
  between the self‑convolution and a perfectly flat target (the ideal
  uniform distribution of the sum‑variable).  This spreads the mass
  and reduces the global autocorrelation level.

* **Maximum‑focused gradient** – a soft‑max approximation of the true
  max‑convolution (the quantity that is actually reported).  When the
  flattening phase is almost finished, the algorithm concentrates on
  the remaining peaks.

Both gradients are blended with a linear schedule (λ) that starts at 1
(only the uniform‑flattening term) and ends at 0 (only the max‑focused
term).  The update is a projected gradient step on the simplex
(non‑negative entries that sum to 1).  Momentum is introduced by a
running average of the gradient, and occasional random block
perturbations help to escape shallow local minima.  If no improvement
has been observed for a while the current best solution is jittered
and the optimisation continues (soft restart).

The routine respects the 1000 s time budget, prints progress, and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if np.all(v == 0):
        # uniform fallback
        return np.full_like(v, 1.0 / v.size)

    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safeguard
        theta = 0.0
    else:
        rho = ind[cond].max()
        theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    # Occasionally rounding errors can make the sum slightly off
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= s
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Random multiplicative log‑normal perturbation of a contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_onto_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (probability vector)
    that minimises ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found as a list of floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        arr = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
            a = np.clip(arr, 0.0, None)
            a = _project_onto_simplex(a)
        else:
            a = None
    else:
        a = None

    if a is None:
        # default length – similar to the previously best construction
        n = 3800
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT lengths
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_ea   = _next_pow2(3 * n - 2)          # for (c‑target)*a and w*a

    # --------------------------------------------------------------
    # 3) Target flat convolution (uniform over 2n‑1 entries)
    # --------------------------------------------------------------
    target_c = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    # --------------------------------------------------------------
    # 4) Book‑keeping for the current best solution
    # --------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # 5) Running averages for momentum / smoothing
    # --------------------------------------------------------------
    grad_avg = np.zeros_like(a)          # exponential moving average of gradients

    # --------------------------------------------------------------
    # 6) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = (now - start_time) / total          # 0 → 1

        # ------------------------------------------------------------------
        # 6.1 – schedules for learning‑rate η, soft‑max concentration β,
        #        and blending factor λ (uniform‑vs‑max gradient)
        # ------------------------------------------------------------------
        eta  = 1.0 * (1.0 - frac) + 0.001 * frac    # 1 → 1e‑3
        beta = 0.5 + (2500.0 - 0.5) * frac          # 0.5 → 2500
        lam  = max(0.0, 1.0 - 1.5 * frac)          # start 1, decay to 0

        # ------------------------------------------------------------------
        # 6.2 – compute self‑convolution and the soft‑max weights
        # ------------------------------------------------------------------
        c = _self_conv(a, fft_len_self)            # length 2n‑1
        max_c = c.max()

        w_raw = np.exp(beta * (c - max_c))        # shifted to avoid overflow
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            w = np.full_like(c, 1.0 / c.size)
        else:
            w = w_raw / w_sum

        # ------------------------------------------------------------------
        # 6.3 – gradient of the two objectives
        # ------------------------------------------------------------------
        # 6.3.1 – max‑focused gradient (soft‑max approximation)
        conv_wa = _conv_fft(w, a, fft_len_ea)      # length 3n‑2
        grad_max = 2.0 * conv_wa[n - 1:n - 1 + n]

        # 6.3.2 – uniform‑flattening gradient
        e = c - target_c
        conv_ea = _conv_fft(e, a, fft_len_ea)
        grad_uni = 2.0 * conv_ea[n - 1:n - 1 + n]

        # blended gradient
        grad = lam * grad_uni + (1.0 - lam) * grad_max

        # ------------------------------------------------------------------
        # 6.4 – exponential moving average (momentum) of the gradient
        # ------------------------------------------------------------------
        gamma = 0.9          # momentum coefficient (you can tune)
        grad_avg = gamma * grad_avg + (1.0 - gamma) * grad

        # ------------------------------------------------------------------
        # 6.5 – projected gradient step on the simplex
        # ------------------------------------------------------------------
        # Normalise the step size so that updates are roughly of the same
        # magnitude irrespective of the raw gradient norm.
        norm = np.linalg.norm(grad_avg) + 1e-12
        step = eta * grad_avg / norm
        a_new = a - step
        a = _project_onto_simplex(a_new)

        # ------------------------------------------------------------------
        # 6.6 – occasional stochastic jitter (exploration)
        # ------------------------------------------------------------------
        if rng.random() < 0.01:          # 1 % of iterations
            a = _random_block_perturb(a, rng)

        # ------------------------------------------------------------------
        # 6.7 – keep the best solution seen so far
        # ------------------------------------------------------------------
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 6.8 – soft restart after prolonged stagnation
        # ------------------------------------------------------------------
        if now - last_improve > 20.0:   # 20 s without improvement
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * jitter
            a = _project_onto_simplex(a)
            # recompute the convolution for the new starting point
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 7) Return the best sequence (clipped to limits imposed by the
    #    evaluation function for safety).
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
