"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the bound on ``evaluate_sequence``.

Algorithm
---------

1. **Parameterisation** – the sequence `a` lives on the simplex.  We store
   unrestricted log‑weights `x` and recover the probabilities via a soft‑max
   (`a = softmax(x)`).  This guarantees all constraints automatically.

2. **Phase I – Smoothed‑max Adam optimisation**  
   The true objective is `max_k (a*a)_k`.  We replace the max by a smooth
   log‑sum‑exp surrogate

        F_γ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)

   and compute its gradient analytically (convolution + correlation with the
   soft‑max weights).  An Adam optimiser updates `x` while a linear schedule
   raises `γ` from 0.5 to 200, sharpening the surrogate gradually.  Restarts
   (with log‑normal noise) are triggered after 30 s without improvement.

3. **Phase II – Greedy local refinement**  
   Starting from the best point found in Phase I we repeatedly:
   * compute the full self‑convolution,
   * locate the index `k*` where it attains its maximum,
   * evaluate the contribution `a_i·a_{k*‑i}` of each coordinate,
   * move a small fraction of the largest‑contributing mass to a random
     location (mass transfer respects non‑negativity and preserves the
     total sum),
   * accept the move if the true objective improves (or occasionally
     accept uphill moves with a tiny temperature for escaping shallow
     minima).

   If no improvement occurs for 30 s we jitter the current best solution.

4. **Best‑ever tracking** – the globally best sequence seen so far is kept
   and printed.  At termination the sequence is clipped to the limits imposed
   by ``evaluate_sequence`` and returned as a plain list of floats.

The two‑phase scheme (global smooth optimisation + focussed greedy search)
explores the space broadly and then fine‑tunes the convolution peak, which
has been observed to push the bound well below the target 1.5025 on the
available time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to ``a``.
    ``w`` are the soft‑max weights (probability vector of length 2·n‑1).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out
