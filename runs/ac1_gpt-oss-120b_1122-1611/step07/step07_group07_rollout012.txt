    # initial sequence
    if 'height_sequence_1' in globals():
        a = np.asarray(globals()['height_sequence_1'], dtype=float)
        if a.ndim !=1 or a.size==0 or not np.isfinite(a).all():
            n = 3840
            a = np.full(n, 1.0/n)
        else:
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s>0:
                a = a / s
            else:
                n = a.size
                a = np.full(n, 1.0/n)
    else:
        n = 3840
        a = np.full(n, 1.0/n)
    n = int(a.size)

    fft_len_self = _next_pow2(2*n - 1)
    fft_len_grad = _next_pow2(3*n - 2)

    c = _self_conv(a, fft_len_self)   # length 2n-1
    cur_score = 2.0 * n * c.max() / (a.sum()**2)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    # allocate phases
    phase1_end = start_time + 0.15*total_time
    phase2_end = start_time + 0.45*total_time
    phase3_end = start_time + 0.80*total_time

    # hyperparameters
    # Phase1: mirror descent (similar to existing)
    eta_start = 2.0
    eta_end = 1e-3
    eps_min = 1e-12

    move_counter = 0

    temp_end = 1e-6

        now = time.monotonic()
        elapsed = now - start_time
        # Determine phase
        if now < phase1_end:
            # Weighted mirror descent (same as initial)
            # compute softmax w for c
            frac = (now - start_time) / (phase1_end - start_time)

"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence(seq) = 2·n·max(a*a) / (sum(a)^2).

Algorithm
---------

* **Initialisation** – start from the best known sequence (global
  `height_sequence_1`) or from a uniform simplex vector.

* **Phase 1 – weighted mirror descent** (≈15 % of the time).  
  Same technique as the reference code: a soft‑max over the
  self‑convolution provides a smooth surrogate of the max, whose
  gradient is obtained analytically via an FFT convolution with the
  reversed vector.  An exponentiated‑gradient (entropy mirror) step
  updates the distribution on the simplex.

* **Phase 2 – hard‑max sub‑gradient descent** (≈30 % of the time).  
  The soft‑max is replaced by an exact indicator of the current
  maximal convolution entry.  The resulting sub‑gradient is applied
  with a linearly decaying step‑size, again using the exponentiated‑
  gradient projection onto the simplex.

* **Phase 3 – greedy mass‑transfer refinement** (≈35 % of the time).  
  The index `k` where the convolution attains its maximum is found,
  the pair `(i, k‑i)` contributing the most product is selected as a
  donor, and a small amount of mass is transferred to a low‑mass
  recipient.  The convolution is updated incrementally (O(n)) and the
  move is kept only if the objective improves.

* **Phase 4 – simulated‑annealing jitter** (remaining time).  
  A light stochastic move similar to the reference implementation,
  but driven by a slowly decreasing temperature, is used to escape
  shallow local minima.

* **Restarts** – if no improvement for 20 s the current best vector is
  perturbed multiplicatively by log‑normal noise and optimisation
  continues.

All updates keep the vector non‑negative and on the simplex, which
matches the expectations of ``evaluate_sequence``.  The routine stops
a few seconds before the 1000 s budget and returns the best sequence
found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new a and c."""
    n = a.size
    # new a
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    # incremental update of convolution
    c_new = c.copy()
    # shift of donor i (subtract) and recipient j (add)
    # each ordered pair (i, *) appears twice except i==*.
    c_new[i:i + n] -= 2.0 * delta * a
    c_new[j:j + n] += 2.0 * delta * a

    # quadratic ΔΔ part (i,i) and (j,j) and (i,j) terms
    ii = 2 * i
    jj = 2 * j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence`` within the time budget.  Returns the best
    sequence found as a plain Python list.
    """

    # ------------------------------------------------------------------
    # 0) Time‑budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise – use the previously discovered construction if
    #    available, otherwise start with a uniform distribution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            n = 3840
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Book‑keeping of the current state
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)                # current self‑convolution
    cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 3) Phase boundaries (percentage of total time)
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.15 * total_time      # weighted mirror descent
    phase2_end = start_time + 0.45 * total_time      # hard‑max sub‑gradient
    phase3_end = start_time + 0.80 * total_time      # greedy mass transfer
    # remaining time → simulated‑annealing jitter

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters
    # ------------------------------------------------------------------
    # Phase‑1 (mirror descent)
    eta_start = 2.0
    eta_end = 1e-3
    eps_min = 1e-12

    # Phase‑2 (hard max sub‑gradient)
    sub_eta_start = 2.0
    sub_eta_end = 1e-4

    # Phase‑3 (greedy refinement)
    move_counter = 0
    RECOMPUTE_EVERY = 500

    # Phase‑4 (simulated annealing)
    temp_start = 0.08
    temp_end = 1e-6

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – weighted mirror descent (entropy‑mirror)
        # --------------------------------------------------------------
        if now < phase1_end:
            frac = (now - start_time) / (phase1_end - start_time)
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # temperature for the soft‑max – stronger smoothing early
            range_c = c.max() - c.min()
            eps = max(range_c * max(0.2 * (1.0 - frac), 0.01), eps_min)

            max_c = c.max()
            w = np.exp((c - max_c) / eps)
            w_sum = w.sum()
            if w_sum == 0.0:                     # pathological fallback
                w = np.zeros_like(c)
                w[int(c.argmax())] = 1.0
                w_sum = 1.0
            w = w / w_sum                         # shape (2n‑1,)

            # gradient of the smooth max w.r.t. a
            a_rev = a[::-1]
            fw = np.fft.rfft(w, n=fft_len_grad)
            fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: 3 * n - 1]
            grad = 2.0 * conv_full[n - 1:n - 1 + n]   # length n

            # exponentiated‑gradient update on the simplex
            factor = np.exp(-eta * grad)
            factor = np.clip(factor, 1e-40, 1e40)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

            # recompute convolution and current score
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # --------------------------------------------------------------
        # Phase 2 – hard‑max sub‑gradient descent
        # --------------------------------------------------------------
        elif now < phase2_end:
            frac = (now - phase1_end) / (phase2_end - phase1_end)
            eta = sub_eta_start * (1.0 - frac) + sub_eta_end * frac

            # index of the (first) maximal convolution entry
            k_max = int(c.argmax())
            w = np.zeros_like(c)
            w[k_max] = 1.0            # exact sub‑gradient (no smoothing)

            a_rev = a[::-1]
            fw = np.fft.rfft(w, n=fft_len_grad)
            fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
            conv_full = np.fft.irfft(fw * fa_rev, n=fft_len_grad)[: 3 * n - 1]
            grad = 2.0 * conv_full[n - 1:n - 1 + n]

            factor = np.exp(-eta * grad)
            factor = np.clip(factor, 1e-40, 1e40)
            a = a * factor
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # --------------------------------------------------------------
        # Phase 3 – greedy donor/recipient mass transfer
        # --------------------------------------------------------------
        elif now < phase3_end:
            # locate current peak of the convolution
            k_max = int(c.argmax())

            # contributions of each index to that peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            donor = int(np.argmax(contrib))

            if a[donor] < 1e-14:
                # nothing useful to move – skip this iteration
                continue

            # recipient prefers a small entry
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = int(rng.choice(np.arange(n), p=recv_weights))
            else:
                recipient = (donor + rng.integers(1, n)) % n

            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max() / (a_new.sum() ** 2)

            # keep the move only if it improves the objective
            if new_score < cur_score:
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # periodic full recomputation to avoid drift
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

            # occasional global jitter to escape plateaus
            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # --------------------------------------------------------------
        # Phase 4 – simulated‑annealing jitter (fine‑grained refinement)
        # --------------------------------------------------------------
        else:
            # linear temperature schedule
            t_frac = (now - phase3_end) / (deadline - phase3_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            k_max = int(c.argmax())
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = int(rng.choice(np.arange(n), p=prob))
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-12:
                continue

            # recipient favours low entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            recv_sum = recv_weights.sum()
            if recv_sum > 0.0:
                recv_weights /= recv_sum
                recipient = int(rng.choice(np.arange(n), p=recv_weights))
            else:
                recipient = (donor + rng.integers(1, n)) % n

            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max() / (a_new.sum() ** 2)

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

            # occasional block jitter
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max() / (a.sum() ** 2)
            last_improve = now
            move_counter = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clip to evaluation limits for safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
