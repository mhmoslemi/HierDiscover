def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _self_conv_direct(a: np.ndarray) -> np.ndarray:
    """Fallback direct convolution (used only for exact evaluation)."""
    return np.convolve(a, a)


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the LSE surrogate w.r.t. a.
    ``w`` is a probability vector (same length as the linear conv of a with a).
    """
    n = a.size
    # correlation: rev(a) * w  (linear)
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    # slice the part that corresponds to Σ_j a_j w_{i+j}
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 8))           # block size up to 12 % of n
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.

    The search works on the simplex by optimising the log‑softmax of the
    coefficients with an Adam optimiser and a log‑sum‑exp surrogate of
    the max‑convolution.  A multi‑restart schedule with occasional block‑jitter
    is used to escape plateaus.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise – use the provided construction if available,
    #    otherwise fall back to a uniform start.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)                     # log‑weights
        else:
            n = 2048
            x = np.zeros(n, dtype=np.float64)
    else:
        n = 2048
        x = np.zeros(n, dtype=np.float64)  # uniform log‑weights

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (used for all gradient steps)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)               # for a*a
    # (the weight vector w has the same length as the self‑conv)
    # ------------------------------------------------------------------
    # 3) Adam state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Helper: obtain probability vector from log‑weights
    # ------------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # ------------------------------------------------------------------
    # 5) Track the best solution seen (exact evaluation)
    # ------------------------------------------------------------------
    a = _a_from_x(x)
    cur_conv = _self_conv_direct(a)       # exact conv for the first score
    best_score = 2.0 * n * cur_conv.max()  # == evaluate_sequence for sum=1
    best = a.copy()
    last_improve = time.monotonic()
    start = time.monotonic()
    total_time = deadline - start

    # ------------------------------------------------------------------
    # 6) Hyper‑parameters (schedules)
    # ------------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 200.0          # very sharp surrogate for the true max
    lr_start = 0.8
    lr_end = 0.03              # linear decay

    # ------------------------------------------------------------------
    # 7) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total_time

        # ---- current iterate ------------------------------------------------
        a = _a_from_x(x)                  # simplex (sum = 1)

        # ---- self‑convolution (FFT, cheap) ---------------------------------
        c = _self_conv_fft(a, fft_len_self)   # length 2·n‑1

        # ---- smooth‑max weights --------------------------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        c_shift = c - np.max(c)                     # stabilise exp
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        w /= w_sum                                   # w is a prob. vector

        # ---- gradient wrt a (analytic) ----------------------------------------
        g_a = _gradient_wrt_a(a, w)                  # length n

        # ---- chain rule to log‑weights x (∂F/∂x) ------------------------------
        avg = np.dot(a, g_a)                         # ⟨a,∇F⟩
        g_x = a * (g_a - avg)                        # sees that a stays on simplex

        # ---- Adam update ------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # linear learning‑rate decay
        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ---- exact evaluation (true objective) -------------------------------
        cur_score = 2.0 * n * c.max()                # same as evaluate_sequence for sum=1
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best exact score = {best_score:.6f}")

        # ---- stagnation handling (soft restart) ------------------------------
        if now - last_improve > 30.0:
            # re‑initialise log‑weights from the current best with moderate noise
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # ---- occasional block‑jitter (global jump) ---------------------------
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 8) Return the best sequence (clipped to the bounds of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
