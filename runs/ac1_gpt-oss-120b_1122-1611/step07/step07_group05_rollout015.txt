def _grad_max_smooth(a, beta):
    n = a.size
    fft_len = _next_pow2(2*n - 1) # for conv
    a_fft = np.fft.rfft(a, n=fft_len)
    c = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
    # softmax weights
    max_c = c.max()
    exp_term = np.exp(beta * (c - max_c))
    w = exp_term / exp_term.sum()
    # Gradient:
    # Compute conv of a and w_rev: w_rev = w[::-1]
    w_rev = w[::-1]
    # compute convolution using FFT of length at least len(a)+len(w)-1
    conv_len = _next_pow2(a.size + w_rev.size - 1)
    a_fft2 = np.fft.rfft(a, n=conv_len)
    w_rev_fft = np.fft.rfft(w_rev, n=conv_len)
    conv_full = np.fft.irfft(a_fft2 * w_rev_fft, n=conv_len)
    # slice out needed part: indices from len(w_rev)-1 to len(w_rev)-1 + n

Let's craft final answer with code in python fences.assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\,\max (a*a)}{(\sum a)^2},
\qquad a_i\ge 0 ,
\]

which, after fixing the sum to 1, is simply the maximum entry of the
self‑convolution `c = a*a`.  
The algorithm works in two stages:

1. **Smoothed \(L_{\infty}\) gradient descent** –  
   The non‑smooth `max` is replaced by a soft‑max
   \(\operatorname{softmax}_\beta(c)\) with a temperature parameter β.
   For a given β the gradient is  

   \[
   \nabla_\!a\; \operatorname{softmax}_\beta(c)
   =2\;a * \operatorname{softmax}_\beta(c)^{\;\!{\rm rev}},
   \]

   which can be obtained by a single FFT‑based convolution.
   β is increased linearly from a small value (smooth landscape) to a
   large value (almost the true max).  After each gradient step the
   vector is projected onto the probability simplex.

2. **Stochastic refinement** –  
   Once β is large we switch to a cheap **mass‑transfer move**:
   transfer a random fraction of the mass of a “donor’’ index (chosen
   proportionally to its contribution to the current peak) to a
   “recipient’’ with a small value.  The convolution is updated
   incrementally ( O(n) ) and the move is accepted by a simple
   simulated‑annealing rule.  Periodic full recomputation of the
   convolution removes accumulated numerical drift.

Both phases keep the best‑ever vector, restart from the best after a
period of stagnation (log‑normal jitter) and respect the 1000 s time
budget.

The implementation uses only `numpy` and pure Python – no external
solvers – and is fully self‑contained.

```python
# -------------------  Step‑function optimiser  -------------------
"""
Proposes a non‑negative height sequence that minimises
``evaluate_sequence``.

Algorithm
---------
1. Initialise a on the simplex (random Dirichlet or the supplied
   global solution).
2. Phase 1 (≈ 70 % of time):
      – minimise a smooth approximation of the max‑convolution:
        soft‑max weight w = exp(β·(c‑max(c))) / Σ .
      – gradient = 2·conv(a , rev(w))  (FFT, O(n log n)).
      – projected gradient step with a linearly decaying step‑size.
3. Phase 2 (≈ 30 % of time):
      – stochastic mass‑transfer moves (donor → recipient) with an
        incremental update of the convolution (O(n)).
      – Metropolis acceptance with a linearly cooling temperature.
4. Restart after 30 s of no improvement: multiply a random block by a
   log‑normal factor and re‑project.
5. Return the best vector found (clipped to the limits of
   ``evaluate_sequence``).

The routine works for any time budget ≤ 1000 s and prints intermediate
progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # subtract donor contribution, add recipient contribution
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms (retain exactness for the moved mass)
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal multiplicative perturbation of a random contiguous block,
    followed by a simplex projection.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ----- time budget (5 s safety margin) -----
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----- initialise -------------------------------------------------
    # target length – a few thousand is a good trade‑off
    n = 5000

    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            a = _project_simplex(a)
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # random Dirichlet start – gives a non‑trivial initial shape
        a = rng.random(n).astype(np.float64)
        a = _project_simplex(a)

    # small jitter to break exact symmetry
    a += 0.001 * rng.random(n)
    a = _project_simplex(a)

    # ----- bookkeeping ------------------------------------------------
    # initial convolution and score
    fft_len_full = _next_pow2(3 * n - 2)          # large enough for all FFTs
    a_fft_full = np.fft.rfft(a, n=fft_len_full)
    c_full = np.fft.irfft(a_fft_full * a_fft_full, n=fft_len_full)
    c = c_full[: 2 * n - 1]                     # linear self‑convolution
    cur_score = 2.0 * n * c.max()               # sum(a) == 1 → denominator = 1
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ----- schedule: 70 % phase 1, 30 % phase 2 -----------------------
    phase1_end = start_time + 0.70 * total_time

    # phase‑1 hyper‑parameters
    beta_start = 0.5
    beta_end = 2000.0
    eta_start = 0.3
    eta_end = 1e-6

    # phase‑2 hyper‑parameters
    temp_start = 0.05
    temp_end = 1e-7
    RECOMPUTE_EVERY = 2000   # full FFT recompute to avoid drift

    move_counter = 0
    it = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase 1 – smoothed max gradient descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # fraction of phase 1 elapsed
            f = (now - start_time) / (phase1_end - start_time)

            # step size and temperature schedules
            eta = eta_start * (1.0 - f) + eta_end * f
            beta = beta_start * (1.0 - f) + beta_end * f

            # ----- convolution a*a (FFT) --------------------------------
            a_fft_full = np.fft.rfft(a, n=fft_len_full)
            c_full = np.fft.irfft(a_fft_full * a_fft_full, n=fft_len_full)
            c = c_full[: 2 * n - 1]

            # ----- soft‑max weights ------------------------------------
            max_c = c.max()
            # stability: shift by max_c, all exponents ≤ 0
            w_raw = np.exp(beta * (c - max_c))
            w = w_raw / w_raw.sum()          # shape (2n‑1,)

            # ----- gradient = 2·conv(a , rev(w)) ----------------------
            w_rev = w[::-1]
            w_fft = np.fft.rfft(w_rev, n=fft_len_full)
            grad_full = np.fft.irfft(a_fft_full * w_fft, n=fft_len_full)
            # slice that corresponds to indices 0 … n‑1
            start_idx = w_rev.size - 1
            grad = 2.0 * grad_full[start_idx : start_idx + n]

            # ----- projected gradient step ------------------------------
            a_tmp = a - eta * grad
            a = _project_simplex(a_tmp)

            # current objective (max of current convolution)
            cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Phase 2 – stochastic refinement (mass‑transfer moves)
        # --------------------------------------------------------------
        else:
            # temperature for Metropolis rule
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # locate current peak of the convolution
            k_max = int(c.argmax())

            # contribution of each index to this peak (a_i·a_{k_max‑i})
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # ----- choose donor (probability ∝ contribution) ------------
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                # nothing to move – skip this iteration
                continue

            # ----- choose recipient (bias towards small entries) -------
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            rw_sum = recv_weights.sum()
            if rw_sum > 0.0:
                recv_weights /= rw_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # ----- amount of mass to transfer ---------------------------
            max_delta = 0.5 * a[donor]
            delta = rng.random() * max_delta

            # ----- apply move (incremental update of convolution) -------
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # ----- Metropolis acceptance --------------------------------
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # occasional full recomputation to keep numerical accuracy
                if move_counter % RECOMPUTE_EVERY == 0:
                    a_fft_full = np.fft.rfft(a, n=fft_len_full)
                    c_full = np.fft.irfft(a_fft_full * a_fft_full, n=fft_len_full)
                    c = c_full[: 2 * n - 1]
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter to escape plateaus
            if rng.random() < 0.02:
                a = _random_block_perturb(a, rng)
                a_fft_full = np.fft.rfft(a, n=fft_len_full)
                c_full = np.fft.irfft(a_fft_full * a_fft_full, n=fft_len_full)
                c = c_full[: 2 * n - 1]
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Record the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # reset from the best with a mild log‑normal perturbation
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _project_simplex(a)
            a_fft_full = np.fft.rfft(a, n=fft_len_full)
            c_full = np.fft.irfft(a_fft_full * a_fft_full, n=fft_len_full)
            c = c_full[: 2 * n - 1]
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart after stagnation (best {best_score:.6f})")

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
```
