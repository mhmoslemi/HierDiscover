assistantfinal**Solution Overview**

The objective  

\[
F(a)=\frac{2\,n\;\max_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2}
\]

is **scale‑invariant** – we may restrict the search to the probability simplex  
\(\sum_i a_i = 1,\;a_i\ge 0\).  
For a fixed vector the term \(\max_k (a*a)_k\) is the maximum of a family of
convex quadratic forms, therefore \(F\) is a convex function on the simplex.
A simple **sub‑gradient method** converges to the global optimum.

The implementation below

* works directly with the exact sub‑gradient of the *max* (no soft‑max
  smoothing);
* uses a log‑softmax parametrisation \(a=\operatorname{softmax}(x)\) so the
  simplex constraint is satisfied automatically;
* employs Adam‑style adaptive steps;
* injects a small Gaussian jitter every few hundred steps to escape plateaus;
* restarts from the current best after a period of stagnation;
* starts from a random Dirichlet‑like distribution (instead of the uniform
  stationary point) and, if a previous solution is present, also tries an
  interpolation of it.

The routine respects the 1000 s budget, prints progress and finally returns the
best coefficient vector found.

python
"""
propose_candidate – convex sub‑gradient optimisation for the max‑self‑convolution
objective.

The algorithm works on the probability simplex (sum = 1) by parametrising the
step‑function heights with logits `x` and the softmax mapping
`a = softmax(x)`.  In each iteration the exact sub‑gradient of the max
convolution is obtained by taking an (uniform) convex combination of the
gradients of all active convolution entries.  Adam updates are applied to the
logits, occasional Gaussian jitter is added and hard restarts are performed
when no improvement has been seen for a while.

The routine can be called repeatedly; it respects a time‑budget (default 1000 s)
and returns the best sequence discovered.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        # should never happen, but keep the function robust
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Project a (possibly negative) vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true index
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The optimisation is a sub‑gradient Adam scheme on
    the exact max‑convolution surrogate, with periodic restarts and a small
    stochastic jitter.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Allowed wall‑clock time (seconds).  The routine stops a few seconds
        before the limit to avoid a forced termination.

    Returns
    -------
    List[float]
        The best sequence found (clipped to the evaluator’s limits).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time budget handling (keep a small safety margin)
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Choose a length n and an initial distribution.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        s = base.sum()
        if s == 0.0:
            base[:] = 1.0 / base.size
        else:
            base /= s
        # keep at least 2000 entries, otherwise up‑sample linearly
        n = max(2000, base.size)
        if base.size != n:
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a0 = np.interp(new_x, old_x, base)
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()
        else:
            a0 = base.copy()
    else:
        # no previous solution – start from a random Dirichlet‑like vector
        n = 3800                           # a size that gives a good trade‑off
        # exponential(1) draws give a Dirichlet(1,…,1) distribution
        a0 = rng.exponential(1.0, size=n)
        a0 /= a0.sum()

    # --------------------------------------------------------------
    # Parameterisation: logits x → probabilities a = softmax(x)
    # --------------------------------------------------------------
    eps = 1e-12
    x = np.log(a0 + eps)                 # initial logits

    # Adam bookkeeping
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # --------------------------------------------------------------
    # Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    fft_len_conv = _next_pow2(2 * n - 1)
    a = _softmax_logits(x)
    a_fft = np.fft.rfft(a, n=fft_len_conv)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[:2 * n - 1]
    best_max = conv_full.max()
    best_score = 2.0 * n * best_max          # exact objective for sum(a)=1
    best_a = a.copy()
    last_improve = start

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    lr = 0.5                     # Adam base learning rate
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    RESTART_WAIT = 25.0          # seconds without improvement → restart
    JITTER_EVERY = 400          # steps
    jitter_scale = 0.025

    # FFT length for the correlation needed for the sub‑gradient
    fft_len_corr = _next_pow2(3 * n - 2)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ---------------------------------------------------------
        # logits → probabilities (simplex)
        # ---------------------------------------------------------
        a = _softmax_logits(x)                 # shape (n,)

        # ---------------------------------------------------------
        # Full self‑convolution via FFT
        # ---------------------------------------------------------
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft,
                                 n=fft_len_conv)[:2 * n - 1]   # length 2n‑1

        # ---------------------------------------------------------
        # Current exact score
        # ---------------------------------------------------------
        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max

        # Keep the best record
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---------------------------------------------------------
        # Build the sub‑gradient of the max‑convolution.
        # ---------------------------------------------------------
        # indices where the maximum is attained (within a tiny tolerance)
        tol = max(1e-12, 1e-9 * cur_max)
        active_idx = np.where(conv_full >= cur_max - tol)[0]
        w = np.zeros_like(conv_full)
        w[active_idx] = 1.0 / active_idx.size          # uniform weight over the active set

        # correlation w ⋆ a  (the gradient of Σ w_k * (a*a)_k )
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr),
                                 n=fft_len_corr)          # length 3n‑2

        start_idx = n - 1                      # the relevant window
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]   # shape (n,)

        # ---------------------------------------------------------
        # Chain rule: gradient w.r.t. logits x (softmax Jacobian)
        # ---------------------------------------------------------
        dot = np.dot(grad_a, a)                # scalar ⟨grad_a, a⟩
        grad_x = a * (grad_a - dot)            # shape (n,)

        # ---------------------------------------------------------
        # Adam update on the logits
        # ---------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # ---------------------------------------------------------
        # Small stochastic jitter every few hundred steps (helps escape plateaus)
        # ---------------------------------------------------------
        if it % JITTER_EVERY == 0:
            x += rng.normal(0.0, jitter_scale, size=n)

        # ---------------------------------------------------------
        # Hard restart if we have been stagnant for a while
        # ---------------------------------------------------------
        if time.monotonic() - last_improve > RESTART_WAIT:
            # Re‑initialise logits around the current best point
            x = np.log(best_a + eps) + rng.normal(0.0, 0.07, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(val) for val in best_a]
