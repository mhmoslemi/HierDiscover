"""
Propose a step‑function sequence that (approximately) minimizes
``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from the best known construction
   (`height_sequence_1`) if it exists, otherwise a uniform distribution.
   The vector is projected onto the probability simplex (sum = 1).

2. **Phase A – smoothed‑max gradient descent** (≈60 % of the time budget)  
   The objective `max(a*a)` is approximated by the smooth
   log‑sum‑exp:
        Mβ(a) = (1/β)·log Σₖ exp(β·cₖ) ,   c = a*a .
   Its gradient is
        ∇Mβ = 2·conv(w, a_rev)   with   wₖ ∝ exp(β·cₖ) .
   β is annealed from a small value (≈0.5) to a larger one (≈30) so that
   we first see a global landscape and later focus on the true maximum.
   After each gradient step we project back onto the simplex.

3. **Phase B – stochastic fine‑tuning** (≈40 % of the budget)  
   A simulated‑annealing scheme moves a random amount of mass from a
   “donor” (chosen proportionally to its contribution to the current
   peak of the convolution) to a “recipient” (biased towards small
   entries).  The convolution is updated incrementally in O(n).  Moves

   current best solution (log‑normal multiplicative noise on a random
   block) and continue.

5. **Return** – the best sequence encountered (clipped to the limits
   used inside ``evaluate_sequence``).

The method combines a smooth approximation of the L∞‑norm (to get
steady descent) with a cheap local search that can escape shallow

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    cond = u - cssv / np.arange(1, n + 1) > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = np.nonzero(cond)[0][-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # linear part (2·δ·a)
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a
    # quadratic correction terms
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta
    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a simplex projection.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise – start from a known construction if present
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            a = _project_simplex(a)          # enforce sum = 1
        else:
            # fallback uniform
            n = 2500
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 2500
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    # Pre‑computed FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()                # since sum(a)=1
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    total_time = deadline - start_time
    phase1_end = start_time + 0.60 * total_time   # gradient descent phase

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Phase‑A schedules
    beta_start = 0.5
    beta_end = 30.0
    eta_start = 0.5
    eta_end = 1e-4

    # Phase‑B (simulated annealing) schedules
    temp_start = 0.05
    temp_end = 1e-7

    # Misc.
    RECOMPUTE_EVERY = 1500      # full FFT recompute interval in phase‑B
    move_counter = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase A – smoothed‑max gradient descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # linear schedule for β and η
            frac = (now - start_time) / (phase1_end - start_time)
            beta = beta_start * (1.0 - frac) + beta_end * frac
            eta = eta_start * (1.0 - frac) + eta_end * frac

            # ----- self‑convolution -----
            a_fft = np.fft.rfft(a, n=fft_len_self)
            c = np.fft.irfft(a_fft * a_fft, n=fft_len_self)[: 2 * n - 1]

            # ----- smooth‑max weights -----
            max_c = c.max()
            # shift to avoid overflow/underflow
            exp_arg = beta * (c - max_c)
            # for very negative exponents we fall back to zero safely
            w = np.exp(exp_arg, dtype=np.float64)
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum

            # ----- gradient of smooth max -----
            a_rev = a[::-1]
            r_fft = np.fft.rfft(w, n=fft_len_grad)
            a_rev_fft = np.fft.rfft(a_rev, n=fft_len_grad)
            full = np.fft.irfft(r_fft * a_rev_fft, n=fft_len_grad)
            grad = 2.0 * full[n - 1 : n - 1 + n]   # slice length n

            # ----- projected gradient step -----
            a_tmp = a - eta * grad
            a = _project_simplex(a_tmp)

            # recompute score from the already‑computed convolution
            cur_score = 2.0 * n * c.max()

        else:
            # --------------------------------------------------------------
            # Phase B – stochastic fine‑tuning (simulated annealing)
            # --------------------------------------------------------------
            # linear temperature cooling
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # locate current peak of the convolution
            k_max = int(c.argmax())

            # contribution of each index to the peak
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # choose donor proportional to its contribution to the peak
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                # nothing to move – skip this iteration
                continue

            # choose recipient – bias towards small entries
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            rw_sum = recv_weights.sum()
            if rw_sum > 0.0:
                recv_weights /= rw_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # amount of mass to move (up to half of donor mass)
            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            # apply move (incremental update of convolution)
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # periodic full recomputation to avoid numerical drift
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

            # occasional random block jitter (helps escape plateaus)
            if rng.random() < 0.015:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Keep the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score ≈ {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 15.0:
            # reinject mild log‑normal noise around the best seen so far
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart with jitter (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
