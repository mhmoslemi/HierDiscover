def proj_simplex(v):
    if np.sum(v) == 1 and np.all(v >= 0): return v
    v = np.asarray(v)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # Find rho = max{i | u_i - cssv_i / i > 0}
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho)==0:
        # all points non positive
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""Optimise a non‑negative step‑function to minimise
   evaluate_sequence(seq) = 2·n·max(conv(seq,seq))  (seq is normalised to sum=1).

   The optimiser mixes two complementary phases:
   * 0‑85 % of the time – projected sub‑gradient descent on the
     max‑convolution (the non‑smooth objective).  The sub‑gradient at a
     maximising shift `k` is 2·a shifted by `k`.  After each step the
     vector is projected back onto the probability simplex.
   * 85‑100 % – a soft‑max mirror‑descent refinement (the same exponentiated‑gradient
     scheme used in the previous baseline).  This concentrates the
     optimisation on the true maximum after the distribution has been
     roughly shaped by the sub‑gradient phase.

   Random block perturbations and periodic restarts add stochastic
   exploration.  The best vector seen is kept throughout the run and
   returned at the end.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two that is ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w | w_i ≥ 0, Σ w_i = 1 }.
    """
    if np.sum(v) == 1 and np.all(v >= 0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                         # all entries are non‑positive
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Renormalise numerically in case of tiny drift
    w_sum = w.sum()
    if w_sum > 0:
        w /= w_sum
    else:
        w[:] = 1.0 / n
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# ----------------------------------------------------------------------

            w_raw = np.exp(beta * (c - max_c))
            w = w_raw / w_raw.sum()

            # Gradient of the smoothed objective: 2·(w * a) (central slice)
            conv_wa = _conv_fft(w, a, fft_len_wa)        # length 3n‑2
            grad = 2.0 * conv_wa[n - 1:n - 1 + n]

            # Exponentiated‑gradient (mirror) update on the simplex
            exponent = -eta * grad
            exponent = np.clip(exponent, -50.0, 50.0)
            factor = np.exp(exponent)
            a = a * factor
            a = _proj_simplex(a)

        # ----------------------------------------------------------
        # 4.4 – occasional stochastic perturbation
        # ----------------------------------------------------------
        if rng.random() < perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----------------------------------------------------------
        # 4.5 – restart after prolonged stagnation
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * jitter
            a = _proj_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best {best_score:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to limits of evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    print(f"Finished – best score = {best_score:.6f}")
    return [float(x) for x in best]
