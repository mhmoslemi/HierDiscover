"""
Propose‑candidate – low‑peak self‑convolution via a two‑phase
mirror‑descent / hard‑max sub‑gradient scheme.

Phase 1 (0‑≈ 60 % of the time)  
    – smooth‑max (soft‑max) weights w_k ∝ exp(α·c_k) with α growing up
      to 30.  Gradient g = 2·a∗w is obtained by a single FFT.
    – exponentiated‑gradient update a_i ← a_i·exp(‑η·g_i) on the simplex
      (η decays from 1.5 to 0.03).

Phase 2 (≈ 60‑90 % of the time)  
    – keep the very sharp soft‑max (α≈30) but lower the step size
      (η ≈ 0.03→0.001) to refine the solution.

Phase 3 (final ≈ 10 % of the budget)  
    – switch to a hard‑max sub‑gradient: w is a Kronecker delta at the
      current maximal convolution index.  The sub‑gradient is exactly
      the slice of the convolution a∗δ, i.e. g_i = 2·a_{k_max‑i} (if
      the index lies inside the vector).  An Adam–style adaptive
      multiplicative update is applied for very fine tuning.

Exploration / restarts  
    – every few hundred accepted moves the exact convolution is
      recomputed (removes numerical drift).
    – a small probability (2 %) of a random log‑normal block perturbation
      (as in the reference implementation) is applied.
    – if > 20 s pass without improvement the current best is jittered
      (multiplicative log‑normal noise) and the search continues.

All operations are O(n log n) via FFTs, the routine stays well
within the 1000 s budget on a modern CPU.  The returned list is
already clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution of length ``len(a)+len(b)-1``.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (full length) via FFT."""
    return _fft_convolve(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _hard_max_gradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of the (non‑smooth) max at index ``k_max``.
    Returns the vector g_i = 2·a_{k_max‑i} for the overlapping region,
    zero elsewhere.
    """
    n = a.size
    g = np.zeros_like(a)
    # Overlap region where 0 ≤ i ≤ n‑1 and 0 ≤ k_max‑i ≤ n‑1
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    # Slice indices i = lo … hi, corresponding j = k_max‑i
    g[lo:hi + 1] = 2.0 * a[k_max - np.arange(lo, hi + 1)]
    return g


def _softmax_weights(c: np.ndarray, alpha: float) -> np.ndarray:
    """Stable soft‑max with temperature α."""
    c_max = c.max()
    w = np.exp(alpha * (c - c_max))
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= s
    return w


def _compute_score(a: np.ndarray, fft_len_self: int) -> float:
    """Exact value of evaluate_sequence for a (sum is assumed to be non‑zero)."""
    n = a.size
    c = _self_conv(a, fft_len_self)
    max_b = c.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """

    # ------------------------------------------------------------------

            total = a.sum()
            a = a / total if total > 0.0 else np.full_like(a, 1.0 / a.size)

    # ------------------------------------------------------------------
    # 3) Book‑keeping variables

    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    t_iter = 0                    # iteration counter for bias correction

    accepted = 0

        # 6.1  Choose schedule parameters for the current phase
        # --------------------------------------------------------------
        if frac < phase1_end:                      # Phase 1 – smooth‑max
            # linear ramp of α and η
            phase_frac = frac / phase1_end
            alpha = alpha_start + (alpha_peak - alpha_start) * phase_frac
            eta = eta_start + (eta_mid - eta_start) * phase_frac
            use_soft = True
        elif frac < phase2_end:                   # Phase 2 – sharp soft‑max
            phase_frac = (frac - phase1_end) / (phase2_end - phase1_end)
            alpha = alpha_peak                     # keep very sharp
            eta = eta_mid + (eta_end - eta_mid) * phase_frac
            use_soft = True
        else:                                     # Phase 3 – hard max + Adam
            alpha = alpha_peak

            use_soft = False

        # --------------------------------------------------------------

        # --------------------------------------------------------------

        else:
            w = np.zeros_like(c)

        # 6.4  Gradient of the (smoothed) peak
        # --------------------------------------------------------------
        # Convolution a * w  (full length = 3n‑2)
        conv_full = _fft_convolve(a, w, fft_len_grad)
        # Slice the n elements that correspond to the linear convolution a*a
        g = 2.0 * conv_full[n - 1: 2 * n - 1]

        # --------------------------------------------------------------
        # 6.5  Adam‑style multiplicative update (exponentiated‑gradient)
        # --------------------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * g

        a_new = np.clip(a_new, 0.0, None)

        if s == 0.0:

        else:
            a_new /= s

        # 6.6  Evaluate the new candidate (exact max)
        # --------------------------------------------------------------
        new_score = _compute_score(a_new, fft_len_self)

        # 6.7  Simple deterministic acceptance (only improvements)
        #      – a mild Metropolis criterion is also kept for robustness
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:

                accept = True

            cur_score = new_score
            accepted += 1

                c = _self_conv(a, fft_len_self)
                cur_score = _compute_score(a, fft_len_self)

        # --------------------------------------------------------------
        # 6.8  Occasional random block perturbation (≈2 % of iterations)
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            cur_score = _compute_score(a, fft_len_self)

        # --------------------------------------------------------------
        # 6.9  Record the best solution seen so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 6.10  Soft restart after stagnation (≥20 s without improvement)
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            # jitter the current best with a small log‑normal noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _compute_score(a, fft_len_self)
            last_improve = now
            accepted = 0
            # reset Adam moments to avoid stale gradients
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
