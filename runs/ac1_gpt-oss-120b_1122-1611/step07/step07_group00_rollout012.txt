"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the bound obtained by the previous gradient‑descent code.

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1) – either from the
   global `height_sequence_1` or from a uniform start.
2. **Phase I – smooth‑max Adam optimisation**  
   We minimise a smooth approximation of `max_k (a*a)_k`:
   * `γ` (the log‑sum‑exp temperature) is increased from a small value to a
     large one, turning the surrogate into the true maximum.
   * Adam updates are performed on the *log‑weights* `x` such that
     `a = softmax(x)`.  The learning‑rate decays linearly.
   * If no improvement is seen for a configurable time, a soft restart
     (log‑normal noise around the current best) is performed.
3. **Phase II – local hill‑climbing**  
   Starting from the best vector found in Phase I we repeatedly
   move a small amount of mass from a randomly chosen source index to a
   randomly chosen destination index.  The move is kept only if it reduces
   the exact evaluation `2·n·max(conv)`.  The step size shrinks
   gradually, allowing finer adjustments near the end.
4. Return the best sequence (clipped to the limits imposed by
   `evaluate_sequence`).

Both phases respect the time budget (≈ 1000 s) – a safety margin of a few
seconds is kept for the harness.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a 1‑D array using an FFT of length ``fft_len``.
    Result length is 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient ∂F/∂a for the smooth‑max surrogate.
    ``a`` – current probability vector (length n).
    ``w`` – soft‑max weights over the 2·n‑1 convolution positions (sum = 1).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]                       # reverse for correlation
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    # The relevant slice corresponds to indices n‑1 … 2·n‑2
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor,
    then re‑normalise to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _move_mass(a: np.ndarray, rng: np.random.Generator, step_frac: float) -> np.ndarray:
    """
    Transfer a small fraction of mass from a random source index to a random
    destination index, then re‑normalise.
    ``step_frac`` controls the maximal share of the source weight that can be
    moved (typical values 0.01–0.3).  The function returns a new array.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = step_frac * rng.random() * a[i]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    # Re‑normalise to avoid tiny drift from the exact sum of 1.
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out

    # 4) Tracking the best solution
    # ------------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    a = _a_from_x(x)
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()          # exact objective (sum(a)=1)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 5) Hyper‑parameters (schedules)
    # ------------------------------------------------------------------
    gamma_start = 0.2
    gamma_end = 80.0          # final γ – closer to the true max
    lr_start = 0.4
    lr_end = 0.01

    # ------------------------------------------------------------------
    # 6) Phase I – smoothed‑max Adam optimisation
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.6 * total_time   # 60 % of the budget

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start_time
        frac_time = elapsed / total_time

        # current iterate on the simplex
        a = _a_from_x(x)

        # self‑convolution
        c = _self_conv(a, fft_len_self)

        # smooth‑max weights (log‑sum‑exp surrogate)
        gamma = gamma_start + (gamma_end - gamma_start) * frac_time
        c_shift = c - np.max(c)          # for numerical stability
        w = np.exp(gamma * c_shift)
        w /= w.sum()                     # probability vector over 2·n‑1 spots

        # gradient w.r.t. a (for the surrogate)
        g_a = _gradient_wrt_a(a, w)

        # chain rule to log‑weights x (∂F/∂x = a·(g_a – ⟨a,g_a⟩))
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # Adam step
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr_start * (1.0 - frac_time) + lr_end * frac_time
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # exact objective – keep the best ever seen
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best (phase I) = {best_score:.6f}")

        # soft restart on stagnation (no improvement for ~20 s)
        if now - last_improve > 20.0:
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # occasional global jitter (≈2 % of iterations)
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 7) Phase II – simple hill‑climbing / simulated annealing
    # ------------------------------------------------------------------
    a = best.copy()                     # start local search from the best
    cur_score = best_score
    last_improve = time.monotonic()
    phase2_start = time.monotonic()
    phase2_total = deadline - phase2_start

    # decaying step size: start relatively large, shrink to ~0.01
    def _step_fraction(elapsed):
        """Linear decay of the move‑size fraction."""
        return 0.3 * (1.0 - elapsed / phase2_total) + 0.01

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        step_frac = _step_fraction(elapsed)

        # propose a simple mass‑move
        cand = _move_mass(a, rng, step_frac)

        # evaluate candidate (exact objective)
        c_cand = _self_conv(cand, fft_len_self)
        cand_score = 2.0 * n * c_cand.max()

        # accept only improving moves (greedy hill‑climb)
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(
                    f"[{now - start_time:6.2f}s] new best (phase II) = {best_score:.6f}"
                )
            # reset stagnation timer
            last_improve = now

        # occasional block‑perturbation (larger jumps)
        if rng.random() < 0.015:
            a_block = _random_block_perturb(a, rng)
            c_block = _self_conv(a_block, fft_len_self)
            block_score = 2.0 * n * c_block.max()
            if block_score < cur_score:
                a = a_block
                cur_score = block_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = now
                    print(
                        f"[{now - start_time:6.2f}s] new best (block) = {best_score:.6f}"
                    )

        # safety exit if we have been stagnant for a while
        if now - last_improve > 30.0:
            # re‑initialize locally around the current best
            noise = rng.normal(0.0, 0.4, size=n)
            x = np.log(best + 1e-12) + noise
            a = _a_from_x(x)               # fresh start for the hill‑climb
            cur_score = 2.0 * n * _self_conv(a, fft_len_self).max()
            last_improve = now
            print(
                f"[{now - start_time:6.2f}s] local‑restart (stagnation) – score {cur_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 8) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
