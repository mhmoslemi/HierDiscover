"""
propose_candidate
=================

Search for a non‑negative step‑function (a vector `a`) that minimizes

    2·n·max(a*a) / (sum(a)**2)

where `*` denotes linear convolution.  The objective is the
`evaluate_sequence` score, which is invariant to scaling, so we fix
`sum(a)=1` and optimise the *∞‑norm* of the self‑convolution.

The optimiser consists of two stages (all operations respect the
time‑budget and run on the CPU only).

1. **L2‑error descent** – minimise the squared ℓ₂‑distance between the
   current convolution and its average value.  This gives a fast global
   flattening of the autocorrelation.

2. **Max‑subgradient descent** – directly attack the quantity we really
   care about: the maximum entry of the convolution.  The sub‑gradient
   of the max‑operator at the current peak `k*` is known analytically:
   `g_i = 2·a_{k*‑i}` (zero when the index is out of range).  A
   projected gradient step on the probability simplex therefore reduces
   the peak efficiently.

The two phases run sequentially inside the allotted 1000 s (5 s safety
margin).  Periodic “jitter’’ restarts avoid stagnation: the best
solution found so far is perturbed by a mild log‑normal factor and the
search continues from there.

All helper functions are top‑level (no closures) and the routine returns
the best sequence it has discovered before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_l2_error(
    a: np.ndarray, fft_len_self: int, fft_len_grad: int
) -> tuple[np.ndarray, np.ndarray]:
    """
    Returns the self‑convolution ``c`` and the gradient of the L2‑error
    (∑_k (c_k−M)²) with respect to ``a``.
    """
    n = a.size

    # ----- self‑convolution -------------------------------------------------
    a_fft = np.fft.rfft(a, n=fft_len_self)
    c = np.fft.irfft(a_fft * a_fft, n=fft_len_self)[: 2 * n - 1]

    # ----- target (average) value -----------------------------------------
    M = (a.sum() ** 2) / (2 * n - 1)          # = np.mean(c)
    resid = c - M

    # ----- gradient = 4·(resid * a_rev) ------------------------------------
    a_rev = a[::-1]
    r_fft = np.fft.rfft(resid, n=fft_len_grad)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len_grad)
    full = np.fft.irfft(r_fft * a_rev_fft, n=fft_len_grad)

    # slice corresponding to the original a
    grad = 4.0 * full[n - 1 : n - 1 + n]
    return c, grad


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_max(a: np.ndarray, k_star: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)_k   at the current peak ``k_star``.
    The i‑th component is 2·a_{k_star‑i} when this index exists, else 0.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, k_star - (n - 1))
    i_end = min(n - 1, k_star)
    i = np.arange(i_start, i_end + 1, dtype=np.intp)
    j = k_star - i
    grad[i] = 2.0 * a[j]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a simplex projection.  Used for “jitter’’ restarts.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise – try to reuse a previously discovered construction
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
    else:
        # default size – 5 000 entries give a good trade‑off between flexibility
        # and computational cost.
        n = 5000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise to the probability simplex (sum = 1, non‑negative)
    a = _project_simplex(a)
    n = a.size

    # Pre‑computed FFT lengths
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)                     # current convolution
    cur_score = 2.0 * n * c.max()                       # current evaluate_sequence score
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 3) Phase division
    # ------------------------------------------------------------------
    # 40 % of the budget → L2‑error descent,
    # 55 % → max‑subgradient descent,
    # 5 %  → occasional jitter / restarts.
    phase1_end = start_time + 0.40 * total_time
    phase2_end = start_time + 0.95 * total_time

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()

        # --------------------------------------------------------------
        # Phase 1 – global L2‑error gradient descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # Linear schedule for the step size (larger early, tiny later)
            frac = (now - start_time) / (phase1_end - start_time)
            eta = 0.20 * (1.0 - frac) + 1e-6 * frac

            # Compute both convolution and L2‑gradient in one pass
            c, grad = _gradient_l2_error(a, fft_len_self, fft_len_grad)

            # Projected gradient step (stay on the simplex)
            a = _project_simplex(a - eta * grad)

        # --------------------------------------------------------------
        # Phase 2 – direct minimisation of the max‑convolution entry
        # --------------------------------------------------------------
        elif now < phase2_end:
            # Re‑compute the convolution (cheap, n≈5 000)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

            # Locate the current peak
            k_star = int(c.argmax())

            # Sub‑gradient of the max‑operator at the peak
            grad = _grad_max(a, k_star)

            # Cooling schedule for the step size
            t_frac = (now - phase1_end) / (phase2_end - phase1_end)
            eta = 0.05 * (1.0 - t_frac) + 1e-6 * t_frac

            # Candidate update
            a_cand = _project_simplex(a - eta * grad)

            # Evaluate candidate
            c_cand = _self_conv(a_cand, fft_len_self)
            cand_score = 2.0 * n * c_cand.max()

            # Accept only improving moves (strict descent)
            if cand_score < cur_score:
                a = a_cand
                c = c_cand
                cur_score = cand_score

        # --------------------------------------------------------------
        # Phase 3 – occasional jitter / restart to escape plateaus
        # --------------------------------------------------------------
        else:
            # Simple jitter – log‑normal perturbation of a random block
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Keep the best solution found so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            # Re‑initialise from the current best, add a modest log‑normal noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart after stagnation (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clip to the limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
