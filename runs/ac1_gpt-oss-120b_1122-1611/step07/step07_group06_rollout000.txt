"""
Projected sub‑gradient optimiser for the self‑convolution L∞ bound.

The returned value of ``evaluate_sequence`` is
    2·n·max(a*a) / (sum(a))² .
Because the expression is scale‑invariant we keep the vector on the
probability simplex (sum = 1).  The algorithm repeatedly:

* computes the self‑convolution of the current vector,
* finds the index where this convolution attains its maximum,
* builds a sub‑gradient of the max‑operator (the correlation of the
  vector with a unit impulse at that index),
* takes an Adam step in the direction of the negative sub‑gradient,
* projects the result back onto the simplex.

Random multiplicative jitter is applied every few hundred iterations
to help the search escape plateaus, and the whole process is
re‑started from a fresh random sparse vector whenever the time budget
permits.  Exact scores are queried from ``evaluate_sequence`` only
periodically; the best sequence seen so far is returned at the end.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x ≥ 0, sum(x) = 1}.
    Returns a new view – the input array is not modified.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries would be projected to zero – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _compute_gradient(a: np.ndarray, idx_max: int) -> np.ndarray:
    """
    Sub‑gradient of  max_i (a*a)[i]  w.r.t. the vector ``a``,
    evaluated at the index ``idx_max`` where the convolution is maximal.
    Implements the exact derivative:
        g_i = 2·a[idx_max‑i]   (if 0 ≤ idx_max‑i < n, i ≠ idx_max‑i)
        g_i =   a[i]           (if i = idx_max‑i)
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if idx_max < n:
        # Overlap region is i = 0 … idx_max
        length = idx_max + 1
        g[:length] = 2.0 * a[idx_max::-1]
        # Diagonal correction when idx_max is even
        if idx_max % 2 == 0:
            mid = idx_max // 2
            g[mid] = a[mid]
    else:
        # Overlap region is i = idx_max‑(n‑1) … n‑1
        start = idx_max - (n - 1)
        g[start:] = 2.0 * a[start:][::-1]
        if idx_max % 2 == 0:
            i = idx_max // 2
            if 0 <= i < n:
                g[i] = a[i]
    return g

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the provided time budget.

    The routine is a self‑contained optimiser based on projected
    sub‑gradient descent with Adam acceleration, occasional random jitter,
    and multiple short restarts from random sparse initialisations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (keep a few seconds as safety margin)
    # ------------------------------------------------------------------
    overall_deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time_all = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Choose a moderate length – 2048 works well for the time budget
    # ------------------------------------------------------------------
    n = 2048

    # ------------------------------------------------------------------
    # 2) Initialise the global best solution (use the provided construction
    #    if it exists, otherwise start from a uniform vector)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a0.ndim != 1 or a0.size == 0 or not np.isfinite(a0).all():
            a0 = None
    else:
        a0 = None

    if a0 is None:
        best = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # Clip, normalise and adapt length if necessary
        a0 = np.clip(a0, 0.0, None)
        if a0.sum() > 0.0:
            a0 = a0 / a0.sum()
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)

        if a0.size > n:
            best = a0[:n].copy()
        elif a0.size < n:
            best = np.pad(a0, (0, n - a0.size), mode="constant")
        else:
            best = a0.copy()

    best_score = evaluate_sequence(best.tolist())
    best_time = time.monotonic()
    print(f"[0.00s] initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT length (large enough for convolution)
    # ------------------------------------------------------------------
    fft_len = _next_pow2(3 * n)  # >= n + (2n‑1) – 1

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters
    # ------------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_start, lr_end = 0.30, 0.001
    jitter_every = 400
    eval_every = 200

    # ------------------------------------------------------------------
    # 5) Main loop – repeatedly launch short optimisation runs
    # ------------------------------------------------------------------
    while time.monotonic() < overall_deadline:
        # Allocate a modest chunk of time for the next run
        remaining = overall_deadline - time.monotonic()
        chunk_budget = min(30.0, remaining)          # ≤30 s per run
        run_start = time.monotonic()

        # ---- Random sparse initialisation ---------------------------------
        sparsity = rng.integers(20, max(21, n // 4))            # 20 … n/4 non‑zeros
        a = np.zeros(n, dtype=np.float64)
        idxs = rng.choice(n, size=sparsity, replace=False)
        a[idxs] = rng.random(sparsity) + 1e-3                     # avoid exact zeros
        a = a / a.sum()

        # ---- Adam state ----------------------------------------------------
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow, beta2_pow = beta1, beta2

        iteration = 0
        while time.monotonic() < run_start + chunk_budget:
            # ---- Self‑convolution (FFT) ------------------------------------
            a_fft = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

            # ---- Identify the active (max) index -------------------------
            idx_max = int(conv.argmax())
            # max_val = conv[idx_max]          # not used directly but could help diagnostics

            # ---- Sub‑gradient of the max‑operator -------------------------
            grad = _compute_gradient(a, idx_max)

            # ---- Linear schedule for the learning rate --------------------
            elapsed = time.monotonic() - run_start
            frac = elapsed / chunk_budget
            lr = lr_start * (1.0 - frac) + lr_end * frac

            # ---- Adam update ---------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- Gradient step + projection to the simplex -----------------
            a = a - step
            a = np.maximum(a, 0.0)
            if a.sum() == 0.0:
                a[:] = 1.0 / n
            else:
                a = _project_onto_simplex(a)

            iteration += 1

            # ---- Periodic random jitter ----------------------------------
            if iteration % jitter_every == 0:
                width = rng.integers(1, max(2, n // 16))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.09))
                a_j = a.copy()
                a_j[start:start + width] *= factor
                a_j = np.clip(a_j, 0.0, None)
                a = _project_onto_simplex(a_j)

            # ---- Exact evaluation now and then ---------------------------
            if iteration % eval_every == 0:
                exact = evaluate_sequence(a.tolist())
                if exact < best_score - 1e-12:
                    best_score = exact
                    best = a.copy()
                    best_time = time.monotonic()
                    total_elapsed = best_time - start_time_all
                    print(f"[{total_elapsed:6.2f}s] new best = {best_score:.6f} "
                          f"(run iter={iteration}, idx_max={idx_max})")

        # End of this chunk – loop back and possibly start a new random run

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to limits accepted by the evaluator)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
