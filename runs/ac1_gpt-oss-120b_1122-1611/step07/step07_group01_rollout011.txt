def refine(a, rng, fft_len_self, steps=2000):
    n = a.size
    for iter in range(steps):
        c = _self_conv(a, fft_len_self)
        max_c = c.max()

        frac = iter / steps

"""
genetic_ga_refine.py
--------------------

A genetic‑algorithm + mirror‑descent hybrid that searches for a non‑negative
step‑function (a sequence ``a``) minimizing the evaluation function

    2·n·max(conv(a,a)) / (sum(a)^2)

which is exactly the value returned by ``evaluate_sequence`` after the
normalisation ``sum(a)=√(2n)``.

The algorithm works in three stages:

1. **Population initialisation** – a diverse set of candidates is built
   from a few simple analytic shapes (uniform, linear, quadratic,
   exponential, cosine, random Dirichlet).  The known construction
   ``height_sequence_1`` (if present) is also inserted.

2. **Genetic optimisation** – a small population (≈30) is evolved with
   selection, crossover (simple averaging) and mutation (log‑normal block
   perturbations).  The fitness is the exact score computed by an FFT based
   self‑convolution.  Elites are kept, the rest is replaced by offspring.
   When progress stalls for >30 s a full random restart (keeping the current
   best) is performed.

3. **Local refinement** – after the bulk of the time budget the current
   best candidate is polished by a soft‑max mirror‑descent (the same idea
   used in the previous state‑of‑the‑art code) for a few hundred iterations.
   This often squeezes a few extra points out of the best solution.

The routine respects the time budget (default 1000 s) and returns the
best sequence found as a plain list of ``float``.  All values are clipped to
the limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Low‑level convolution utilities (FFT based, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenient wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


# ----------------------------------------------------------------------
#  Scoring helpers
# ----------------------------------------------------------------------


def _score_of(a: np.ndarray, fft_len_self: int) -> float:
    """
    Compute the exact objective value for a candidate ``a``.
    ``a`` may be any non‑negative vector; the function follows the same
    formula as ``evaluate_sequence`` (including the normalisation factor).
    """
    n = a.size
    c = _self_conv(a, fft_len_self)
    max_b = c.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
#  Candidate factories
# ----------------------------------------------------------------------


def _make_base_shape(rng: np.random.Generator, n: int, shape_id: int) -> np.ndarray:
    """Return a raw (unnormalised) non‑negative shape."""
    i = np.arange(n, dtype=np.float64)

    if shape_id == 0:          # uniform
        base = np.ones(n, dtype=np.float64)
    elif shape_id == 1:        # decreasing linear
        base = n - i
    elif shape_id == 2:        # increasing linear
        base = i + 1.0
    elif shape_id == 3:        # decreasing quadratic
        base = (n - i) ** 2
    elif shape_id == 4:        # increasing quadratic
        base = (i + 1) ** 2
    elif shape_id == 5:        # exponential decay
        k = rng.uniform(0.0005, 0.005)
        base = np.exp(-k * i)
    elif shape_id == 6:        # exponential rise
        k = rng.uniform(0.0005, 0.005)
        base = np.exp(-k * (n - i - 1))
    elif shape_id == 7:        # cosine bump (smooth bell)
        base = (np.cos(np.pi * (i - (n - 1) / 2) / (n - 1)) + 1.0) / 2.0
    else:                      # random Dirichlet‑like (Gamma) vector
        alpha = rng.uniform(0.1, 5.0)
        base = rng.gamma(alpha, 1.0, size=n)

    # random linear combination with a second shape ~40 % of the time
    if rng.random() < 0.4:
        other_id = rng.integers(0, 8)
        other = _make_base_shape(rng, n, other_id)
        w = rng.random()
        base = w * base + (1 - w) * other

    return np.clip(base, 0.0, None)


def generate_candidate(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Produce a random feasible candidate:
      * pick a random analytic shape,
      * normalise it so that sum(a) = √(2·n).
    """
    shape_id = rng.integers(0, 8)   # 0‑7 are the analytic families
    raw = _make_base_shape(rng, n, shape_id)

    # ensure we have a non‑zero vector
    if raw.sum() == 0.0:
        raw[:] = 1.0

    # normalise to the required L1‑norm √(2n)
    target = np.sqrt(2.0 * n)
    raw = raw * (target / raw.sum())
    return raw


# ----------------------------------------------------------------------
#  Mutation operator
# ----------------------------------------------------------------------


def _mutate(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Log‑normal block perturbation + occasional global jitter.
    Result is re‑normalised to the required L1‑norm.
    """
    n = a.size
    out = a.copy()

    # 1) block perturbation
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.3))          # moderately strong
    out[start:start + width] *= factor

    # 2) global multiplicative jitter (10 % chance)
    if rng.random() < 0.1:
        out *= np.exp(rng.normal(0.0, 0.07, size=n))

    # clip negatives (should not appear) and re‑normalise
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
        total = 1.0
    target = np.sqrt(2.0 * n)
    out = out * (target / total)
    return out


# ----------------------------------------------------------------------
#  Mirror‑descent refinement (soft‑max smoothing)
# ----------------------------------------------------------------------


def _refine_with_mirror_descent(a: np.ndarray,
                                rng: np.random.Generator,
                                fft_len_self: int,
                                fft_len_wa: int,
                                steps: int = 400) -> np.ndarray:
    """
    Perform a short soft‑max mirror‑descent run starting from ``a``.
    The update keeps the vector on the simplex (normalised L1‑norm).
    """
    n = a.size
    target_norm = np.sqrt(2.0 * n)

    for it in range(steps):
        # fraction of the schedule (0 → 1)
        frac = it / steps

        # linear schedules (same philosophy as the reference code)
        eta = 2.0 * (1.0 - frac) + 0.001 * frac
        beta = 0.5 * (1.0 - frac) + 1500.0 * frac

        # self‑convolution and soft‑max weights
        c = _self_conv(a, fft_len_self)
        max_c = c.max()
        w_raw = np.exp(beta * (c - max_c))
        w = w_raw / w_raw.sum()                     # length 2n‑1

        # gradient of the smoothed objective (slice the central part)
        conv_wa = _conv_fft(w, a, fft_len_wa)       # length 3n‑2
        grad = 2.0 * conv_wa[n - 1:n - 1 + n]

        # exponentiated‑gradient / mirror‑descent step
        exponent = -eta * grad
        exponent = np.clip(exponent, -50.0, 50.0)
        factor = np.exp(exponent)
        a = a * factor
        # renormalise to the required L1‑norm
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a = a * (target_norm / s)

    return a
