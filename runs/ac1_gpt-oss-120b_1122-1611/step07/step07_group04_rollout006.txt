    conv_len = 2*n-1
    fft_len_self = _next_pow2(conv_len)
    # function to compute J quickly:
    def compute_J(a):
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        return 2.0 * n * conv.max()

    time_left = deadline - time.monotonic()
    de_time = time_left * 0.5

    population = []

"""
Two‑phase optimizer for the step‑function problem.

Phase 1 – Global search:
    A small Differential‑Evolution (DE) population explores the simplex.
    Each individual is projected onto the probability simplex so that
    Σa_i = 1 and a_i ≥ 0.  The DE steps use mutation, binomial crossover,
    and selection based on the true objective J = 2·n·max(a*a).

Phase 2 – Local refinement:
    Starting from the best DE vector we run a projected‑Adam optimizer.
    The nonsmooth max is smoothed with a temperature‑controlled soft‑max.
    The gradient of the smoothed objective is
        ∇φ = 2·a ＊  w_rev        (conv with the reversed soft‑max weights)
    Adam’s adaptive learning‑rate together with a simplex projection
    yields rapid convergence.  Random block‑perturbations and a
    restart‑after‑stagnation scheme keep the search from getting trapped.

The whole routine respects the 1000 s time budget (5 s safety margin)
and returns the best coefficient list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A small portfolio of diverse starting points, all normalised to sum = 1."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    # half heavy, half zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method first runs a small Differential‑Evolution
    population for global exploration and then refines the best candidate with a
    projected Adam optimiser that works on a soft‑max smoothed version of the
    max‑convolution objective.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (keep a small safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine problem size -------------------------------------------------
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # enforce a modest upper bound so that the FFTs stay cheap
            n = min(max(n, 512), 4096)          # 512 ≤ n ≤ 4096
        except Exception:
            n = 4096
    else:
        n = 4096                                     # default length

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for a * w_rev

    # --------------------------------------------------------------
    # Helper to compute the true objective J (no clipping)
    # --------------------------------------------------------------
    def compute_J(a_vec: np.ndarray) -> float:
        fa = np.fft.rfft(a_vec, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 2) Phase 1 – Differential Evolution on the simplex
    # --------------------------------------------------------------
    time_left = deadline - time.monotonic()
    de_time = time_left * 0.55                     # ~55 % of the remaining budget
    de_deadline = time.monotonic() + de_time

    popsize = 30
    # initialise a small population
    base_cands = _initial_candidates(n, rng)
    if "height_sequence_1" in globals():
        base_cands.append(
            proj_simplex(np.clip(np.asarray(height_sequence_1,
                                          dtype=np.float64), 0.0, None))
        )
    # fill up to popsize
    while len(base_cands) < popsize:
        base_cands.append(rng.dirichlet(np.ones(n)))
    # pick popsize distinct vectors
    indices = rng.choice(len(base_cands), size=popsize, replace=False)
    population = [proj_simplex(base_cands[i].astype(np.float64)) for i in indices]
    scores = [compute_J(v) for v in population]

    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]

    F = 0.8                       # differential weight
    CR = 0.9                      # crossover probability

    start_time = time.monotonic()
    while time.monotonic() < de_deadline:
        for i in range(popsize):
            # pick three distinct donors
            donors = rng.choice([j for j in range(popsize) if j != i],
                                size=3, replace=False)
            r1, r2, r3 = donors
            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = np.clip(mutant, 0.0, None)
            s = mutant.sum()
            if s == 0.0:
                mutant[:] = 1.0 / n
            else:
                mutant /= s

            # binomial crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):                 # guarantee at least one gene
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = proj_simplex(trial)

            trial_score = compute_J(trial)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] DE new best J = {best_score:.6f}")

        # safety break if we have run out of the DE budget
        if time.monotonic() >= de_deadline:
            break

    # --------------------------------------------------------------
    # 3) Phase 2 – Projected Adam refinement
    # --------------------------------------------------------------
    a = best_vec.copy()
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    it = 0
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / total_time          # ∈[0,1]

        # learning‑rate schedule (exponential decay)
        lr_start, lr_end = 0.25, 0.001
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # soft‑max temperature schedule (slow → sharp)
        alpha_start, alpha_end = 2.0, 2000.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ----- self‑convolution (to obtain c) -----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

        # ----- soft‑max weights -----
        c_max = conv.max()
        shifted = conv - c_max                       # for numerical stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        start_idx = w_rev.size - 1                     # = 2n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- Adam update & projection -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # ----- evaluate current objective -----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        cur_score = float(2.0 * n * conv.max())

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Adam new best J = {best_score:.6f}")

        # ----- occasional jitter to escape plateaus -----
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # ----- restart if stagnation > 25 s -----
        if now - last_improve > 25.0:
            a = best_vec.copy()
            # inject a tiny amount of multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart (no improvement) – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
