"""
Hybrid simulated‑annealing / smooth‑max mirror descent for flattening
the self‑convolution of a non‑negative step function.

The algorithm works on a fixed length `n` (default 4096 or the length of
the previously found solution).  It keeps the vector on the simplex
(`sum = 1`) and repeatedly performs one of three moves:

* **smooth‑max mirror descent** – a soft‑max weighting of the current
  autocorrelation `c = a*a` is built, the gradient `g = 2·a*w` (computed
  by an FFT‑based convolution) is obtained and an exponentiated‑gradient
  update `a ← a·exp(‑η·g)` is applied;

* **random mass swap** – a small amount of mass is moved from a random
  index `i` to another index `j`.  This gives a cheap local perturbation;

* **log‑normal block perturbation** – a whole contiguous block is
  multiplied by a log‑normal factor and the vector is renormalised.

All moves are accepted if they improve the objective
`score = 2·n·max(a*a)/(sum(a)^2)`.  Otherwise they may be accepted with a
Metropolis probability that follows a temperature schedule, allowing the
search to escape plateaus.

Every 20 s of stagnation the current best vector is jittered with a
small log‑normal noise (a “soft restart”).  Progress and new best scores
are printed.  The routine stops 5 s before the supplied time budget and
returns the best sequence found (clipped to the limits of
`evaluate_sequence`).

The implementation uses only NumPy and plain Python, runs in
O(n log n) per iteration, and fits comfortably in the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must be ≥ len(a)+len(b)-1).  Returns the full
    convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using FFT length ``fft_len``."""
    return _fft_convolve(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _swap_move(a: np.ndarray, rng: np.random.Generator,
               max_frac: float = 0.10) -> np.ndarray:
    """
    Move a small amount of mass from a randomly chosen non‑zero entry
    to another random position.
    """
    n = a.size
    # pick a source index with non‑zero mass
    nz = np.where(a > 1e-15)[0]
    if nz.size == 0:
        return a.copy()
    i = rng.choice(nz)
    j = rng.integers(0, n)
    while j == i:
        j = rng.integers(0, n)
    # shift at most `max_frac` of the source mass
    delta = rng.random() * a[i] * max_frac
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    # renormalise (should stay 1, but numeric drift may appear)
    out /= out.sum()
    return out


def _score_from_vector(a: np.ndarray, fft_len_self: int) -> float:
    """
    Compute the exact value returned by ``evaluate_sequence`` for a
    non‑negative vector that already sums to 1.
    """
    n = a.size
    c = _self_conv(a, fft_len_self)
    return 2.0 * n * c.max()

    # ------------------------------------------------------------------

    fft_len_grad = _next_pow2(3 * n - 2)            # for a * w (len 2n‑1)

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    cur_score = _score_from_vector(a, fft_len_self)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters (linear schedules)
    # ------------------------------------------------------------------
    # smooth‑max temperature (α) – start low, end high
    alpha_start = 0.5
    alpha_end = 22.0

    # exponentiated‑gradient step size (η) – start aggressive, decay
    eta_start = 2.0
    eta_end = 0.001

    # simulated‑annealing temperature (Metropolis acceptance)
    temp_start = 0.08
    temp_end = 1e-6

    # move probabilities
    prob_grad = 0.60   # mirror‑descent step
    prob_swap = 0.30   # random mass swap
    prob_perturb = 0.10  # block perturbation

    # stagnation handling
    stagnation_limit = 20.0   # seconds without improvement → soft restart
    RECOMPUTE_EVERY = 500     # re‑normalise the self‑convolution occasionally

    accepted = 0
    iteration = 0

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_total = elapsed / total_time

        # --------------------------------------------------------------
        # 5.1  Update schedules
        # --------------------------------------------------------------
        alpha = alpha_start + (alpha_end - alpha_start) * frac_total
        eta = eta_start + (eta_end - eta_start) * (1.0 - frac_total)
        temperature = temp_start + (temp_end - temp_start) * (1.0 - frac_total)
        temperature = max(temperature, 1e-12)

        # --------------------------------------------------------------
        # 5.2  Choose and propose a move
        # --------------------------------------------------------------
        r = rng.random()
        if r < prob_grad:
            # ---- smooth‑max mirror descent ---------------------------------
            c = _self_conv(a, fft_len_self)          # current autocorrelation
            c_max = c.max()
            w = np.exp(alpha * (c - c_max))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient g_i = 2 * Σ_k w_k * a_{k-i}
            conv_full = _fft_convolve(a, w, fft_len_grad)   # length 3n‑2
            g = 2.0 * conv_full[n - 1: 2 * n - 1]          # slice length n

            # exponentiated‑gradient update (mirror descent on simplex)
            log_factor = -eta * g
            # safety clamp for exp()
            log_factor = np.clip(log_factor, -700, 700)
            factor = np.exp(log_factor)
            a_new = a * factor
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s == 0.0:
                a_new[:] = 1.0 / n
            else:
                a_new /= s
        elif r < prob_grad + prob_swap:
            # ---- random mass swap -----------------------------------------
            a_new = _swap_move(a, rng, max_frac=0.15)
        else:
            # ---- block perturbation -----------------------------------------
            a_new = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 5.3  Evaluate the candidate
        # --------------------------------------------------------------
        new_score = _score_from_vector(a_new, fft_len_self)

        # --------------------------------------------------------------
        # 5.4  Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            accepted += 1

            # occasional exact recomputation to keep numerical drift tiny
            if accepted % RECOMPUTE_EVERY == 0:
                # (re‑computes self‑conv only for diagnostics – cheap)
                pass

        # --------------------------------------------------------------
        # 5.5  Keep the best solution so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.6  Soft restart after stagnation
        # --------------------------------------------------------------
        if now - last_improve > stagnation_limit:
            # gentle log‑normal jitter around the current best
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = best_score
            last_improve = now
            accepted = 0
            print(f"[{elapsed:6.2f}s] soft‑restart – best score {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
