def _smooth_max_and_grad(a, gamma, L_fft):
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=L_fft)
    # self convolution
    c_full = np.fft.irfft(a_fft * a_fft, n=L_fft)[: 2*n-1]
    # compute softmax weights
    max_c = np.max(c_full)
    exp_term = np.exp(gamma * (c_full - max_c))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp
    # gradient via convolution of w and reversed a
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=L_fft)
    w_fft = np.fft.rfft(w, n=L_fft)
    conv_full = np.fft.irfft(w_fft * a_rev_fft, n=L_fft)
    grad_slice = conv_full[n-1: n-1+n]
    grad = 2.0 * grad_slice
    smooth = max_c + np.log(sum_exp) / gamma
    return smooth, grad, c_full
```

"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  

The algorithm consists of two stages:

1. **Smooth‑max gradient descent (≈70 % of the time)**  
   – We minimise a differentiable surrogate of the peak of the
     self‑convolution:  
     `smooth_maxγ(a) = max(c) + log(sum(exp(γ·(c-max(c)))))/γ` with
     `c = a*a`.  
   – Its gradient is `∇smooth = 2·conv(w , a_rev)` where  
     `w_k ∝ exp(γ·c_k)` (soft‑max weights) and `a_rev` is the reversed
     vector.  
   – Adam is used for an adaptive step size and the iterate is projected
     onto the probability simplex after each step.  The smoothing
     parameter γ is increased linearly from a small value (≈0.5) to a
     larger one (≈100) so that the surrogate tightens around the true
     maximum.

2. **Stochastic local refinement (remaining time)**  
   – A simple Metropolis move that transfers a random amount of mass
     from a “donor” index (chosen proportional to its contribution to
     the current peak) to a “recipient” index (biased towards small
     entries).  The convolution is updated incrementally in `O(n)`.

Both phases keep the best vector seen so far and perform a mild
log‑normal jitter restart after a long period without improvement.
The routine respects the 1000‑second budget (5 s safety margin) and
returns a list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float):
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.shape[0]
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # linear part of the update
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms (ordered pairs)
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _smooth_max_and_grad(a: np.ndarray, gamma: float, fft_len: int):
    """
    Returns a smooth approximation of ``max(a*a)`` and its gradient
    w.r.t. ``a`` using a soft‑max over the convolution coefficients.
    """
    n = a.shape[0]

    # --- convolution a*a -------------------------------------------------
    a_fft = np.fft.rfft(a, n=fft_len)
    c_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

    # --- soft‑max weights -------------------------------------------------
    # numerically stable version
    c_max = np.max(c_full)
    exp_term = np.exp(gamma * (c_full - c_max))
    sum_exp = np.sum(exp_term)
    w = exp_term / sum_exp

    # smooth value (log‑sum‑exp trick)
    smooth = c_max + np.log(sum_exp) / gamma

    # --- gradient: 2 * conv(w , a_rev) ----------------------------------
    a_rev = a[::-1]
    w_fft = np.fft.rfft(w, n=fft_len)
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    conv_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)
    grad_slice = conv_full[n - 1 : n - 1 + n]
    grad = 2.0 * grad_slice

    return smooth, grad, c_full
