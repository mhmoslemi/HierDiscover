import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _score_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return the evaluate_sequence objective for a (assumes a non‑negative)."""
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)

def _make_u_shape(n: int, p: float = 1.0) -> np.ndarray:
    """U‑shaped distribution – large at the ends, small in the middle."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / (idx + 1) ** p) + (1.0 / (n - idx) ** p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

def _prepare_initial_vectors(n: int, rng: np.random.Generator):

        return _score_convolution(v, fft_len)

    cur_score = score(a)
    best_a = a.copy()

        probs = a / a.sum()

        if new_score < cur_score:

            cur_score = new_score
            # Track the global best.
            if new_score < best_score - 1e-14:
                best_score = new_score
                best_a = a_new.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")
        else:

            # gradually shrinks as the temperature cools.
            if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):

                cur_score = new_score

            a = np.clip(a + jitter, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            cur_score = score(a)

        # Hard restart when we have been stagnant for a while.
        if time.monotonic() - last_improve > 30.0:
            # Re‑initialise around the current best using a modest Gaussian perturbation.
            a = best_a + rng.normal(scale=0.04, size=n)
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= a_sum
            cur_score = score(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] restart – best score {best_score:.6f}")

    return best_a, best_score

"""
Propose a step‑function that minimises ``evaluate_sequence``.

The optimiser works as follows

1. Build a few sensible initial vectors (uniform, U‑shaped, random
   Dirichlet draws and – when available – the previously best construction).
   All vectors are normalised to sum = 1.

2. Pick the start with the smallest exact objective
   (computed with an FFT‑based convolution).

3. Run a fast simulated‑annealing / hill‑climbing loop:
   * at each iteration move a small random amount of mass from one index
     to another,
   * accept the move if it improves the objective,
   * otherwise accept it with a Metropolis probability that decays with a
     temperature schedule,
   * occasionally add Gaussian jitter and restart from the best solution
     after a period of stagnation.

The objective is evaluated directly (identical to ``evaluate_sequence``)
by means of an FFT, so the optimiser works on the true target function.
The best vector found before the time budget expires is returned.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _score_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the exact ``evaluate_sequence`` objective for a non‑negative
    vector ``a`` (no clipping required – the optimiser keeps values in range).
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 1e-12:      # guard – should never happen
        return np.inf
    return 2.0 * n * max_b / (sum_a ** 2)


def _make_u_shape(n: int, p: float = 1.0) -> np.ndarray:
    """U‑shaped distribution – large at the ends, small in the middle."""
    idx = np.arange(n, dtype=np.float64)
    a = (1.0 / (idx + 1) ** p) + (1.0 / (n - idx) ** p)
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _prepare_initial_vectors(n: int, rng: np.random.Generator):
    """Return a modest list of plausible starting vectors (already normalised)."""
    vectors = []

    # 1) uniform
    vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) a couple of U‑shaped profiles
    vectors.append(_make_u_shape(n, p=0.8))
    vectors.append(_make_u_shape(n, p=1.5))

    # 3) a few random Dirichlet draws (moderate concentration)
    for _ in range(3):
        vectors.append(rng.dirichlet(np.full(n, 0.8)))

    # 4) the previously discovered construction, if it exists
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        s = base.sum()
        if s == 0.0:
            base[:] = 1.0 / base.size
        else:
            base /= s
        if base.size != n:                     # linear interpolation to required length
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            base = np.interp(new_x, old_x, base)
            base = np.clip(base, 0.0, None)
            base /= base.sum()
        vectors.append(base)

    return vectors

    Simulated‑annealing / hill‑climbing optimiser.
    Returns the best vector found and its objective value.
    """
