import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _conv_fft(x, y, fft_len):
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]

def _self_conv(a, fft_len):
    return _conv_fft(a, a, fft_len)

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (a probability vector) that
    minimises ``evaluate_sequence``.
    The algorithm is a mirror‑descent optimiser on the simplex with a
    temperature‑controlled soft‑max approximation of the max‑convolution
    (to obtain a smooth gradient).  It uses a long soft‑max schedule,
    momentum, occasional block jitter and a final local hill‑climbing
    phase.  The implementation works for sequences up to several ten‑
    thousands of entries within the 1000 s budget.
    """
    rng = np.random.default_rng(seed)

    # ---- 1) Initialise a -------------------------------------------------
    # Try to reuse the previously discovered best sequence if present.
    if "height_sequence_1" in globals():
        src = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        L = src.size
        # Upsample to a larger length to give the optimiser more freedom.
        n_target = max(6000, L * 2)
        # Linear interpolation on [0,1]
        x_old = np.linspace(0.0, 1.0, L)
        x_new = np.linspace(0.0, 1.0, n_target)
        a = np.interp(x_new, x_old, src)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # Fallback: uniform distribution of moderate size.
        n_target = 6000
        a = np.full(n_target, 1.0 / n_target, dtype=np.float64)

    n = a.size

    # ---- 2) Pre‑compute FFT lengths ---------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a
    fft_len_wa   = _next_pow2(3 * n)              # for w*a

    # ---- 3) Helper for exact score ----------------------------------------
    def exact_score(vec: np.ndarray) -> float:
        """Return 2·n·max(conv(vec,vec))."""
        c = _self_conv(vec, fft_len_self)
        return 2.0 * n * c.max()

    # ---- 4) Optimisation hyper‑parameters ---------------------------------
    # schedules (linear in time)
    init_eta   = 2.0      # initial learning rate
    final_eta  = 0.01     # final learning rate (still non‑zero)
    init_beta  = 0.5      # initial soft‑max concentration
    final_beta = 5000.0   # final concentration – sharp max
    momentum_coeff = 0.9   # exponential moving average coefficient
    jitter_prob = 0.01    # probability of a block perturb in each iteration
    block_width_factor = 0.1   # max block width = n * factor
    block_sigma = 0.15    # log‑normal perturbation std‑dev

    # ---- 5) Book‑keeping ---------------------------------------------------
    cur_score = exact_score(a)
    best = a.copy()
    best_score = cur_score
    grad_mom = np.zeros_like(a)

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start

    # ---- 6) Main loop (mirror descent) ------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        frac = (now - start) / total   # 0 → 1

        # ---- learning‑rate and soft‑max schedule
        eta  = init_eta * (1.0 - frac) + final_eta * frac
        beta = init_beta * (1.0 - frac) + final_beta * frac

        # ---- self‑convolution and soft‑max weights
        c = _self_conv(a, fft_len_self)          # length 2n‑1
        max_c = c.max()
        # soft‑max (shifted by max_c for numerical stability)
        w_raw = np.exp(beta * (c - max_c))
        w = w_raw / w_raw.sum()                  # probability vector, length 2n‑1

        # ---- gradient of the smoothed objective
        conv_wa = _conv_fft(w, a, fft_len_wa)    # length 3n‑2
        grad = 2.0 * conv_wa[n - 1:n - 1 + n]    # slice of length n

        # ---- momentum update
        grad_mom = momentum_coeff * grad_mom + (1.0 - momentum_coeff) * grad

        # ---- exponentiated‑gradient (mirror descent) step
        exponent = -eta * grad_mom
        exponent = np.clip(exponent, -50.0, 50.0)
        factor = np.exp(exponent)
        a *= factor
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- occasional block jitter (exploration)
        if rng.random() < jitter_prob:
            width = rng.integers(1, max(2, int(block_width_factor * n)))
            start_idx = rng.integers(0, n - width + 1)
            factor_j = np.exp(rng.normal(0.0, block_sigma))
            a[start_idx:start_idx + width] *= factor_j
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
            else:
                a /= s

        # ---- exact score to keep the best solution
        cur_score = 2.0 * n * max_c
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ---- restart after long stagnation (optional, but with a softer trigger)
        if now - last_improve > 45.0:
            # Re‑inject a modest jitter around the current best and continue.
            a = best.copy()
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a *= jitter
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            # recompute convolution to keep variables in sync
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

    # ---- 7) Final local refinement (hill‑climbing) -------------------------
    # Use the remaining few seconds (if any) for a cheap coordinate search.
    remaining = deadline - time.monotonic()
    if remaining > 2.0:
        # Pre‑compute current best score once more (in case it changed)
        best_score = exact_score(best)
        a = best.copy()
        # Simple hill‑climb: move a tiny amount of mass from a random donor
        # to a random receiver.  Accept only if the exact score improves.
        delta = 0.001                # absolute amount of probability mass
        while time.monotonic() < deadline:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            if a[i] < delta:
                continue
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            # exact evaluation
            sc = exact_score(a2)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a2.copy()
                a = a2.copy()
                print(f"[{time.monotonic() - start:6.2f}s] hill‑climb improved to {best_score:.6f}")

    # ---- 8) Return the best sequence ---------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

Optimisation of a non‑negative step function (probability vector) to
minimise the evaluation function defined in the challenge.

The algorithm consists of three stages:

1. **Mirror‑descent on the simplex** – a soft‑max approximation of the
   max‑convolution is optimised by exponentiated‑gradient updates.
   A temperature parameter β is increased from 0.5 to 5000,
   while the learning rate η decays from 2 to 0.01.
   Momentum (exponential moving average of the gradient) and occasional
   block‑wise jitter improve exploration.

2. **Soft restart** – after 45 s of stagnation the current best vector is
   perturbed and the optimisation continues, preventing premature
   convergence to a bad local basin.

3. **Local hill‑climbing** – in the last few seconds a cheap coordinate
   search moves a tiny amount of probability mass from a random donor
   to a random receiver, accepting only genuine improvements.
   This refines the solution beyond the smooth‑gradient phase.

The routine works for sequences of several thousand entries within the

functions are top‑level (no closures) and the code does not use any I/O.

    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length len(x)+len(y)-1).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]

            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 7) Final local refinement – simple hill‑climbing
    # ----------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 2.0:
        best_score = exact_score(best)          # ensure exact value
        a = best.copy()
        delta = 0.001                           # absolute probability mass to move
        while time.monotonic() < deadline:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j or a[i] < delta:
                continue
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            sc = exact_score(a2)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a2.copy()
                a = a2.copy()
                print(f"[{time.monotonic() - start:6.2f}s] hill‑climb improved to {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 8) Return the best sequence (clipped to limits of the evaluator)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
