"""
Propose a step‑function that beats the current bound (≈1.5132).

Algorithm
---------
1.  Initialise from the best known construction (`height_sequence_1`) if
    available, otherwise from a uniform distribution.  The sequence is
    up‑sampled (linear interpolation) to a larger length (≈ 5 000) – a
    longer support provides more freedom to flatten the autocorrelation.

2.  Optimise the shape on the probability simplex using **Adam** on the
    log‑softmax (mirrored) parametrisation `y → a = softmax(y)`.  The
    objective is a smooth approximation of the max‑convolution
    (`log‑sum‑exp` with a temperature β).  The gradient
    `∂max/∂a` is obtained analytically via a convolution:
        grad_a = 2·(w * a)[n‑1:n‑1+n]
    where `w` are the soft‑max weights on the self‑convolution `c = a*a`.

3.  Hyper‑parameters (learning‑rate α and soft‑max concentration β) are
    annealed during the run – α decays from 0.5 to 0.01, β grows from 0.5
    to 2000.  Adam’s moment estimates provide a stable, adaptive update.

4.  Stagnation handling:
    * If no improvement is seen for 30 s the current best solution is
      jittered (log‑normal multiplicative noise) and the optimiser is
      re‑initialised.
    * With a small probability (≈0.3 %) a random contiguous block is
      perturbed to escape shallow local minima.

5.  After the main loop a brief hill‑climbing phase explores random
    block perturbations and keeps any improvement.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence found (clipped to the limits of
`evaluate_sequence`).  All heavy work uses real FFTs for speed.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures, no lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of ``x`` and ``y`` using an FFT of length ``fft_len``.
    Returns the full convolution (length ``len(x)+len(y)-1``).
    """
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)
    return z[: x.size + y.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Convenience wrapper for the self‑convolution a*a."""
    return _conv_fft(a, a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    # width at most 10 % of the vector, at least 1 element
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Main optimisation routine
# ----------------------------------------------------------------------

    # ------------------------------------------------------------------
