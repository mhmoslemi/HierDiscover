"""
Hybrid optimiser for the “max‑self‑convolution” problem.

The goal is to find a non‑negative vector a (length n, ∑a = 1) that minimises
    2·n·max(a * a)          (the denominator (∑a)² is fixed to 1).

The optimiser works in three successive phases:

1. **Smooth p‑norm surrogate** – a soft approximation of the infinity norm.
   The surrogate is (∑_k b_k^p)^{1/p} where b = a∗a.  By gradually
   increasing p we steer the optimisation toward the true max.
   Gradients are obtained via FFT‑based convolutions and the update is
   performed on the log‑softmax (mirror‑descent) parametrisation of a using
   Adam.

2. **Exact max sub‑gradient descent** – once p is large enough we switch to
   the true sub‑gradient of max(b).  The sub‑gradient is
   g = 2·(mask ∗ a) where mask is the indicator of the currently maximal
   convolution entries.  Again Adam on the logits is used.

3. **Simulated‑annealing fine‑tuning** – random perturbations of the logits
   are accepted with a temperature schedule.  This helps to escape shallow
   local minima that gradient descent may miss.

A restart is triggered after 25 s without improvement: the logits are
re‑initialised by adding modest Gaussian noise to the best solution seen so
far.  The best sequence encountered before the time budget expires is
returned (clipped to the limits enforced by ``evaluate_sequence``).

The whole routine respects the 1000 s wall‑time budget and works with any
starting sequence stored in ``height_sequence_1``.

from scipy.special import logsumexp  # only used for a fallback soft‑max (not essential)

# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        # degenerate case – fall back to uniform distribution
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s

# ----------------------------------------------------------------------

    eps = 1e-12
    x = np.log(a0 + eps)                     # initialise logits

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # ------------------------------------------------------------------
    # Pre‑computed FFT lengths (constant throughout the run)
    # ------------------------------------------------------------------
    fft_len_conv = _next_pow2(2 * n - 1)      # for a*a
    fft_len_corr = _next_pow2(3 * n - 2)     # for (weight * a)

    # ------------------------------------------------------------------
    # Book‑keeping of the best solution
    # ------------------------------------------------------------------
    def _score_of(a_vec: np.ndarray) -> float:
        """Exact objective value for a (assumes ∑a = 1)."""
        a_fft = np.fft.rfft(a_vec, n=fft_len_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[:2 * n - 1]
        return 2.0 * n * conv.max()

    a = _softmax_logits(x)
    best_a = a.copy()
    best_score = _score_of(best_a)
    last_improve = start
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned for the three phases)
    # ------------------------------------------------------------------
    # Phase 1 – smooth p‑norm surrogate
    phase1_iters = 30000
    p_start = 2.0
    p_end = 200.0

    # Phase 2 – exact max sub‑gradient
    phase2_iters = 30000
    lr_phase2 = 0.5

    # Phase 3 – simulated annealing
    phase3_iters = 50000
    temp_start = 1.0
    temp_end = 0.001

    # Common Adam parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    it = 0  # global iteration counter

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # Current probability vector
        # --------------------------------------------------------------
        a = _softmax_logits(x)                     # shape (n,)

        # --------------------------------------------------------------
        # Linear self‑convolution a * a via FFT (used in *both* phases)
        # --------------------------------------------------------------
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(a_fft * a_fft,
                            n=fft_len_conv)[:2 * n - 1]   # length 2n‑1

        # --------------------------------------------------------------
        # Decide which phase we are in and build the appropriate gradient
        # --------------------------------------------------------------
        if it < phase1_iters:
            # ------- Phase 1 : smooth p‑norm surrogate -----------------
            # p grows quadratically from p_start to p_end
            frac = it / phase1_iters
            p = p_start + (p_end - p_start) * (frac ** 2)

            # weight = b^{p‑1}
            eps_w = 1e-14
            w = np.maximum(conv, eps_w) ** (p - 1.0)

            # convolution w * a  (correlation needed for gradient)
            w_fft = np.fft.rfft(w, n=fft_len_corr)
            a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
            corr = np.fft.irfft(
                w_fft * np.conj(a_fft_corr), n=fft_len_corr
            )  # length 3n‑2

            start_idx = n - 1
            grad_a = 2.0 * p * corr[start_idx:start_idx + n]

            lr = 0.6  # relatively aggressive ADAM learning rate

        elif it < phase1_iters + phase2_iters:
            # ------- Phase 2 : exact max sub‑gradient -------------------
            max_val = conv.max()
            # tolerance to capture all indices that are virtually maximal
            tol = max_val * 1e-12 + 1e-12
            mask = (conv >= max_val - tol).astype(np.float64)

            # convolution mask * a
            mask_fft = np.fft.rfft(mask, n=fft_len_corr)
            a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
            corr = np.fft.irfft(
                mask_fft * np.conj(a_fft_corr), n=fft_len_corr
            )
            start_idx = n - 1
            grad_a = 2.0 * corr[start_idx:start_idx + n]

            lr = lr_phase2

        else:
            # ------- Phase 3 : simulated annealing ----------------------
            # no gradient – we just try a random jitter and accept
            # according to a temperature schedule.
            cur_score = _score_of(a)

            # propose a perturbed logits vector
            proposal_x = x + rng.normal(scale=0.04, size=n)
            proposal_a = _softmax_logits(proposal_x)
            prop_score = _score_of(proposal_a)

            # linear temperature decay
            elapsed = time.monotonic() - start
            total = deadline - start
            temperature = temp_start + (temp_end - temp_start) * (elapsed / total)

            accept = False
            if prop_score < cur_score:
                accept = True
            else:
                # Metropolis criterion
                if rng.random() < np.exp((cur_score - prop_score) / max(temperature, 1e-12)):
                    accept = True

            if accept:
                x = proposal_x
                a = proposal_a
                cur_score = prop_score

            # record improvement
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start:6.2f}s] new best score = {best_score:.6f}")

            # occasional hard restart in the annealing phase
            if time.monotonic() - last_improve > 30.0:
                x = np.log(best_a + eps) + rng.normal(scale=0.07, size=n)
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.monotonic()
                print(f"[{last_improve - start:6.2f}s] restart – best score {best_score:.6f}")

            it += 1
            continue   # skip the Adam update for this iteration

        # --------------------------------------------------------------
        # Convert gradient w.r.t. a into gradient w.r.t. logits x
        # --------------------------------------------------------------
        dot = np.dot(grad_a, a)                     # ⟨grad_a, a⟩
        grad_x = a * (grad_a - dot)                 # shape (n,)

        # --------------------------------------------------------------
        # Adam update on the logits
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # --------------------------------------------------------------
        # Small stochastic jitter every few hundred steps (helps escape plateaus)
        # --------------------------------------------------------------
        if it % 500 == 0:
            x += rng.normal(0.0, 0.02, size=n)

        # --------------------------------------------------------------
        # Exact score & book‑keeping
        # --------------------------------------------------------------
        cur_score = _score_of(_softmax_logits(x))
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = _softmax_logits(x).copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Hard restart if we have been stagnant for a while
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > 30.0:
            # re‑initialise logits around the best solution found so far
            x = np.log(best_a + eps) + rng.normal(0.0, 0.08, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] restart – best score {best_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
