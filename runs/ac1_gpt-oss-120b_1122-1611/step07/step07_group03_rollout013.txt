"""
Propose a step‑function using a *hard‑max* exponentiated‑gradient descent.

Idea
----
The objective returned by ``evaluate_sequence`` is  

    O(a) = 2·n·max_k (a * a)[k] / (∑a)^2

(we keep ∑a = 1, therefore O = 2·n·max_k c_k where
c = a * a is the linear convolution of a with itself).

The current best score (≈1.51) comes from a smooth‑max mirror‑descent.
Here we take the exact (hard) maximum of the convolution,
compute its sub‑gradient analytically (the gradient of
c_k = Σ_{i+j=k} a_i a_j w.r.t. a_i is 2·a_{k‑i}),
and perform an exponentiated‑gradient step on the simplex.
Because only the entries that participate in the worst convolution
position are penalised, the peak drops faster.

The optimisation loop

* computes the full self‑convolution via an FFT (O(n log n));
* finds the index k⁎ where the convolution is maximal;
* builds the sub‑gradient g_i = 2·a_{k⁎‑i} (0 if the index is out of range);
* updates a_i ← a_i·exp(‑η·g_i) and re‑normalises to sum = 1;
* accepts the move if the score improves, otherwise with a tiny
  Metropolis probability (simulated‑annealing flavour);
* periodically injects a small random block perturbation and restarts
  after 30 s of stagnation.

The schedule for the learning rate η decays from 4 → 0.001;
the whole routine fits easily into the 1000 s budget and
keeps the best vector seen so far.

The returned sequence is clipped to the limits enforced by
``evaluate_sequence`` (non‑negative, ≤ 1000).

# ----------------------------------------------------------------------
#   FFT utilities – pure NumPy, no external dependencies
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_convolve(a: np.ndarray, b: np.ndarray, length: int) -> np.ndarray:
    """
    Linear convolution of real vectors ``a`` and ``b`` using an FFT of
    size ``length`` (must satisfy length ≥ len(a)+len(b)-1).
    Returns the full convolution of length len(a)+len(b)-1.
    """
    fa = np.fft.rfft(a, length)
    fb = np.fft.rfft(b, length)
    conv = np.fft.irfft(fa * fb, length)
    return conv[: a.size + b.size - 1]


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using an FFT length ``fft_len``."""
    return _fft_convolve(a, a, fft_len)

    For a fixed index k = max_idx,
        ∂c_k/∂a_i = 2·a_{k‑i}   (if 0 ≤ k‑i < n, else 0).
    Returns a vector of length n.
    """
    n = a.size
    idx = max_idx - np.arange(n)               # vector of candidate partner indices
    mask = (idx >= 0) & (idx < n)              # keep those inside the array bounds
    g = np.zeros_like(a)
    if mask.any():
        g[mask] = 2.0 * a[idx[mask]]
    return g


# ----------------------------------------------------------------------
#   Random block perturbation (helps escape shallow minima)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
#   Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) minimise
    ``evaluate_sequence`` before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (reserve a few seconds for clean‑up)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine length n and initialise a on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
            a = np.clip(cand, 0.0, None)
            s = a.sum()
            a = a / s if s > 0.0 else np.full_like(a, 1.0 / a.size)
        else:
            # fallback – a reasonably large length gives more freedom
            n = 5000
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        # No prior candidate – start from a Dirichlet draw
        n = 5000
        a = rng.dirichlet(np.ones(n) * 0.4).astype(np.float64)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)   # >= 2*n-1

    # ------------------------------------------------------------------
    # 3) Book‑keeping
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4) Hyper‑parameters (simple linear schedules)
    # ------------------------------------------------------------------
    eta_start = 4.0          # aggressive start
    eta_end = 0.001          # final step size
    temp_start = 0.05        # simulated‑annealing temperature at start
    temp_end = 1e-6          # final temperature

    RECOMPUTE_EVERY = 400    # how often we recompute c from scratch

    accepted = 0

    # ------------------------------------------------------------------
    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # --------------------------------------------------------------
        # 5.1  Learning‑rate and temperature schedules
        # --------------------------------------------------------------
        eta = eta_start + (eta_end - eta_start) * frac          # linear decay
        temperature = temp_start + (temp_end - temp_start) * (1.0 - frac)
        temperature = max(temperature, 1e-12)

        # --------------------------------------------------------------
        # 5.2  Self‑convolution and maximal index
        # --------------------------------------------------------------
        c = _self_conv(a, fft_len_self)
        max_idx = int(np.argmax(c))
        cur_score = 2.0 * n * c[max_idx]

        # --------------------------------------------------------------
        # 5.3  Gradient of the hard maximum
        # --------------------------------------------------------------
        g = _hard_max_gradient(a, max_idx)

        # --------------------------------------------------------------
        # 5.4  Exponentiated‑gradient (mirror) update
        # --------------------------------------------------------------
        log_factor = -eta * g
        # Clip to avoid overflow/underflow in exp
        log_factor = np.clip(log_factor, -700, 700)
        factor = np.exp(log_factor)

        a_new = a * factor
        a_new = np.clip(a_new, 0.0, None)

        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s

        # --------------------------------------------------------------
        # 5.5  Evaluate new objective (exact)
        # --------------------------------------------------------------
        c_new = _self_conv(a_new, fft_len_self)
        new_score = 2.0 * n * c_new.max()

        # --------------------------------------------------------------
        # 5.6  Metropolis acceptance (helps to move out of plateaus)
        # --------------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            if rng.random() < np.exp(-(new_score - cur_score) / temperature):
                accept = True

        if accept:
            a = a_new
            c = c_new
            cur_score = new_score
            accepted += 1

            # Occasionally wipe numerical drift
            if accepted % RECOMPUTE_EVERY == 0:
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # 5.7  Small random jitter (≈2 % of iterations)
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # 5.8  Keep the best solution observed so far
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 5.9  Restart after 30 s without improvement
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            print(f"[{elapsed:6.2f}s] stagnation – restarting from best")
            a = best.copy()
            # add a light log‑normal jitter to shake the current optimum
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            accepted = 0

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to evaluate_sequence limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
