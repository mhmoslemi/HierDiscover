        # Determine per-candidate time budget:

        # For each candidate, allocate sub-time if enough.

        for start_vec in candidates:
            # assign a fraction of remaining time (like 0.05 sec per iteration). But we can run while time left > some min.

            a = start_vec.copy()
            # Transform a to logits x = log(a + eps)
            eps = 1e-12
            x = np.log(a + eps)

            # Adam moments
            m = np.zeros_like(x)
            v = np.zeros_like(x)

            # schedule parameters
            max_iter = 5000 # may overshoot time budgets but break early
            t = 0
            n_len = a.size
            fft_len_self = _next_pow2(2 * n_len - 1)
            # Precompute gradient for each step via softmax weight
            while time.monotonic() < deadline:
                t += 1
                # Compute convolution via FFT
                fa = np.fft.rfft(np.exp(x - np.max(x)), n=fft_len_self)
                # Actually better compute a from logits each iteration before conv.
                a = np.exp(x - np.max(x))
                a /= a.sum()
                conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n_len - 1]
                max_c = conv.max()
                # evaluation score (objective)
                score = 2.0 * n_len * max_c   # since sum=1, denominator = 1
                if score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    # print improvement
                    print(f"[{ time.monotonic() - start_time :6.2f}s] n={n_len} new best {best_score:.6f}")

                # Softmax weight for gradient
                # α schedule linearly increasing: α = α_start * (1 - progress) + α_end * progress
                overall_progress = (time.monotonic() - start_time) / (deadline - start_time)
                alpha = 5.0 * (1 - overall_progress) + 200.0 * overall_progress

                # compute w
                shifted = conv - max_c
                w = np.exp(alpha * shifted)
                w_sum = w.sum()
                if w_sum == 0.0:  # fallback
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # Compute gradient of soft surrogate: grad_a = 2 * (a * w_rev) via conv
                w_rev = w[::-1]

                fa = np.fft.rfft(a, n=grad_len)
                fw = np.fft.rfft(w_rev, n=grad_len)
                grad_full = np.fft.irfft(fa * fw, n=grad_len)
                offset = w_rev.size - 1
                grad_a = 2.0 * grad_full[offset: offset + n_len]

                # Convert to gradient wrt logits x:
                # grad_x = a * (grad_a - (a @ grad_a))
                dot = np.dot(a, grad_a)
                grad_x = a * (grad_a - dot)

                # Adam update
                beta1 = 0.9
                beta2 = 0.999
                eps_adam = 1e-8
                lr_start, lr_end = 0.2, 0.001
                lr = lr_start * (1 - overall_progress) + lr_end * overall_progress

                m = beta1 * m + (1 - beta1) * grad_x
                v = beta2 * v + (1 - beta2) * (grad_x ** 2)
                m_hat = m / (1 - beta1 ** t)
                v_hat = v / (1 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # occasional random perturbation

                    noise = rng.normal(0.0, 0.1, size=n_len)
                    x += noise

                # break if iteration count high and when time low
                if t >= max_iter:
                    break

    # after loops, return best sequence as list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]

Define compute_active_gradient(a, conv, epsilon_factor=1e-8):

    max_val = conv.max()
    tol = max_val * epsilon_factor
    active = np.where(conv >= max_val - tol)[0]
    if active.size == 0: active = np.array([conv.argmax()])
    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    # compute grad_a via convolution with reversed w
    w_rev = w[::-1]
    grad_len = _next_pow2(a.size + w_rev.size - 1)
    fa = np.fft.rfft(a, n=grad_len)
    fw = np.fft.rfft(w_rev, n=grad_len)
    grad_full = np.fft.irfft(fa * fw, n=grad_len)
    offset = w_rev.size - 1
    grad = 2.0 * grad_full[offset: offset + a.size]
    return grad

optimiser – Adam on the active‑set surrogate of max‑self‑convolution

We minimise
        J(a) = 2·n·max(a*a) / (∑a)²
subject to a_i ≥ 0.  Because the objective is scale–invariant we fix
∑a = 1 and work on the probability simplex.

Algorithm
---------
* Initialise a from a diverse portfolio (uniform, exponential decays,
  random Dirichlet draws, possibly the previously discovered sequence).
* Parameterise a via logits `x` (soft‑max):  a = softmax(x).  This guarantees
  non‑negativity and the L1‑norm = 1 automatically.
* At each iteration
    1. Build `a` from the current `x`.
    2. Compute the linear self‑convolution `c = a*a` with FFT.
    3. Find the current maximal value `c_max`.  All indices whose value is
       within a small tolerance of `c_max` form the *active set*.
    4. Build a weight vector `w` that is uniform on the active set.
       The surrogate `φ(a) = wᵀc` has the same gradient at the optimum as
       the true non‑smooth max, but is cheap to compute.
    5. Gradient wrt `a` is  ∇φ = 2·(a * rev(w))  (linear convolution).
       Using the chain rule for the soft‑max we obtain the gradient wrt
       the logits `x`:  g = a·(∇φ – (a·∇φ)).
    6. Perform an Adam update on `x` (adaptable learning‑rate, momentum).
    7. Occasionally inject random noise (log‑space) or a multiplicative
       block perturbation to escape plateaux.
    8. Keep the best sequence seen (lowest `J`) across all restarts.

--------------------
The routine receives `budget_s` (default 1000 s).  A small safety margin is
kept for the harness; the loop stops as soon as the deadline is reached.
All helper functions are top‑level, there are no closures, and the program
uses only NumPy (FFT) and the standard library.

The function returns a plain Python list of non‑negative floats ready to be
fed to `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex (>=0, sum=1)."""
    if v.ndim != 1:
        raise ValueError("Input must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Diversified portfolio of feasible starting vectors (sum=1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


def _active_gradient(a: np.ndarray, conv: np.ndarray,
                     eps_factor: float) -> np.ndarray:
    """
    Gradient of the active‑set surrogate.
    `eps_factor` controls the tolerance for the active set:
        w_i = 1/|A| for i in A, where A = {k : conv[k] >= max - eps*max}.
    Returns the gradient wrt `a` (shape n).
    """
    n = a.size
    max_val = conv.max()
    tol = max_val * eps_factor
    active = np.where(conv >= max_val - tol)[0]
    if active.size == 0:
        active = np.array([conv.argmax()], dtype=int)

    w = np.zeros_like(conv)
    w[active] = 1.0 / active.size
    w_rev = w[::-1]

    grad_len = _next_pow2(n + w_rev.size - 1)           # = next_pow2(3n-2)
    fa = np.fft.rfft(a, n=grad_len)
    fw = np.fft.rfft(w_rev, n=grad_len)
    full = np.fft.irfft(fa * fw, n=grad_len)
    offset = w_rev.size - 1
    grad = 2.0 * full[offset: offset + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The optimiser runs several restarts with
    Adam on a soft‑max surrogate of the max self‑convolution, using an
    active set to focus the gradient on the currently largest entries.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (leave a 5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Choice of sequence lengths to explore
    # ------------------------------------------------------------------
    # A handful of lengths spanning the range that performed well historically.
    lengths = [1024, 2048, 3072, 3809, 4096]

    best_seq = None
    best_score = np.inf

    # ------------------------------------------------------------------
    # 2) Main loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        # Check remaining budget
        if time.monotonic() >= deadline:
            break

        # ------------------------------------------------------------------
        # 2a) Build a pool of initial vectors for this length
        # ------------------------------------------------------------------
        candidates = _initial_candidates(n, rng)

        # If the global previous best is available, include it (rescaled).
        if "height_sequence_1" in globals():
            prev = globals()["height_sequence_1"]
            if isinstance(prev, list) and len(prev) == n:
                v = np.asarray(prev, dtype=np.float64)
                v = np.clip(v, 0.0, None)
                if v.sum() > 0:
                    v /= v.sum()
                    candidates.append(v)

        # ------------------------------------------------------------------
        # 2b) Allocate a sub‑budget for this length (roughly equal share)
        # ------------------------------------------------------------------
        remaining = deadline - time.monotonic()
        if remaining <= 0:
            break
        sub_deadline = time.monotonic() + remaining * 0.18   # ~18 % of remaining time

        # ------------------------------------------------------------------
        # 2c) Optimisation from each start vector
        # ------------------------------------------------------------------
        for start_vec in candidates:
            # Stop early if we have exhausted the sub‑budget
            if time.monotonic() >= sub_deadline:
                break

            a = start_vec.copy()
            a = proj_simplex(a)                # safety
            # Logits (soft‑max parametrisation)
            eps = 1e-12
            x = np.log(a + eps)

            # Adam moments
            m = np.zeros_like(x)
            v_adam = np.zeros_like(x)

            # Pre‑computed FFT lengths
            fft_len_self = _next_pow2(2 * n - 1)          # for a*a
            grad_len = _next_pow2(3 * n - 2)             # for a * w_rev

            iteration = 0
            # ----------------------------------------------------------------
            # Iteration loop
            # ----------------------------------------------------------------
            while time.monotonic() < sub_deadline:
                iteration += 1

                # ---- rebuild a from logits (soft‑max) ----
                a = np.exp(x - np.max(x))
                s = a.sum()
                if s == 0.0:
                    a[:] = 1.0 / n
                else:
                    a /= s

                # ---- self‑convolution (FFT) ----
                fa = np.fft.rfft(a, n=fft_len_self)
                conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]

                max_c = conv.max()
                # Objective (∑a = 1 ⇒ denominator = 1)
                cur_score = 2.0 * n * max_c

                # ---- keep the globally best sequence ----
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] length {n:5d} "
                          f"new best score = {best_score:.6f}")

                # ---- active‑set gradient ----
                # tolerance for the active set shrinks as we progress
                overall_progress = (time.monotonic() - start_time) / (deadline - start_time)
                eps_factor = 1e-4 * (1.0 - overall_progress) + 1e-8 * overall_progress
                grad_a = _active_gradient(a, conv, eps_factor)

                # ---- gradient wrt logits (soft‑max chain rule) ----
                dot = np.dot(a, grad_a)
                grad_x = a * (grad_a - dot)

                # ---- Adam learning‑rate schedule ----
                lr_start, lr_end = 0.5, 0.001
                lr = lr_start * (1.0 - overall_progress) + lr_end * overall_progress

                beta1, beta2 = 0.9, 0.999
                eps_adam = 1e-8

                m = beta1 * m + (1.0 - beta1) * grad_x
                v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_x ** 2)

                m_hat = m / (1.0 - beta1 ** iteration)
                v_hat = v_adam / (1.0 - beta2 ** iteration)

                x -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

                # ---- occasional stochastic perturbations ----
                if rng.random() < 0.01:            # log‑space noise
                    x += rng.normal(0.0, 0.05, size=n)

                if rng.random() < 0.02:            # block multiplicative jitter
                    a = _random_block_perturb(a, rng)
                    x = np.log(a + eps)

                # ---- simple stagnation restart (no improvement for ~30 s) ----
                # (handled implicitly by the jitter above; a dedicated restart
                #  is unnecessary for the short inner runs.)

    # ------------------------------------------------------------------
    # 3) Return the best sequence found (clipped to the limits enforced by
    #    evaluate_sequence for safety).
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution of length 1 (never happens in practice)
        best_seq = np.array([1.0], dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
