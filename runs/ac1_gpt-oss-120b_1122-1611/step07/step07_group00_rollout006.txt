def gradient_conv(a, k):
    n = len(a)
    g = np.zeros_like(a)
    # compute contributions where both indices in range
    # let i ranging from max(0, k-(n-1)) to min(n-1, k)
    i_start = max(0, k-(n-1))
    i_end = min(n-1, k)  # inclusive
    # for i in that range, a_i a_{k-i}
    # derivative w.r.t a_i: a_{k-i}
    # derivative w.r.t a_{k-i}: a_i
    # So we can add contributions:
    # for i in range(i_start, i_end+1):
    #   j = k - i
    #   g[i] += a[j]
    #   g[j] += a[i]
    # This double counts when i == j (i=j => k=2i) resulting in adding a[i] twice; that's fine because derivative is 2 a[i] (makes sense).
    idx_i = np.arange(i_start, i_end+1)
    idx_j = k - idx_i
    g[idx_i] += a[idx_j]
    g[idx_j] += a[idx_i]
    return g

def _optimize_instance(rng, n, budget, init_seq=None, global_best_score=inf) -> (np.ndarray, float):
    """
    Perform a subgradient descent optimization on simplex for a given length n.
    Returns best sequence and its score (evaluate_sequence).
    """
    # initialization:
    if init_seq is not None and len(init_seq) == n:
        a = np.asarray(init_seq, dtype=np.float64)
        a = np.clip(a, 0., None)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full(n, 1.0/n)
    else:
        # random Dirichlet
        a = rng.random(n)
        a = np.abs(a)
        a /= a.sum()

    # Pre compute FFT length
    fft_len = _next_pow2(2*n - 1)

    # hyperparameters
    lr0 = 0.5  # initial learning rate for multiplicative updates (choose appropriate)
    lr_end = 0.01  # final
    # schedule: linear decay based on fraction of time elapsed
    start = time.monotonic()
    best_local_seq = a.copy()
    best_local_score = float('inf')
    last_improve = start
    iter_count = 0

    while time.monotonic() < start + budget:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / budget

        # compute conv
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:2*n-1]
        # conv is real due to rounding errors; make sure
        conv = np.real(conv)
        # compute max
        max_val = conv.max()
        cur_score = 2.0 * n * max_val

        # update best local
        if cur_score < best_local_score - 1e-12:
            best_local_score = cur_score
            best_local_seq = a.copy()
            last_improve = now
            # print progress
            print(f"[{now - start_time:.2f}s] (run n={n}) new local best = {best_local_score:.6f}")

        # gradient of max conv index
        kmax = int(conv.argmax())
        # compute gradient
        g = _gradient_conv_at_k(a, kmax)  # function defined

        # learning rate schedule
        lr = lr0 * (1 - frac) + lr_end * frac

        # multiplicative update (mirrored descent)
        # use exponentiated gradient: a_i = a_i * exp(-lr * g_i)
        exp_term = np.exp(-lr * g)
        a = a * exp_term
        # renorm to simplex
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / sum_a

        iter_count += 1

        # stagnation detection: if no improvement for 10 seconds -> jitter
        if now - last_improve > 10.0:
            # jitter: pick random block and multiply by random factor
            block_width = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - block_width + 1)
            factor = np.exp(rng.normal(0., 0.2))
            a[start_idx:start_idx+block_width] *= factor
            # re-normalize again
            a = np.clip(a, 0., None)
            a = a / a.sum()
            # reset last improve timer
            last_improve = now

    # after loop, compute final score for best_local_seq
    final_score = best_local_score
    return best_local_seq, final_score

"""
Propose a step‑function that minimizes `evaluate_sequence`.
The optimizer uses a two‑phase scheme:
 * Phase 1 (smooth‑max) – runs a short Adam optimisation on a
   soft‑max parametrisation of the simplex using a log‑sum‑exp surrogate.
 * Phase 2 (sub‑gradient) – switches to a simple exponentiated‑gradient/
   multiplicative update based on the exact sub‑gradient of the
   max‑convolution objective.
Multiple independent runs with random lengths and random restarts are
executed inside the global time budget (≈ 1000 s).  The best sequence found
is returned.
"""

import time
import numpy as np

# ---------- Helper utilities ----------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length `fft_len`."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise can cause tiny imaginary parts; keep real
    return np.real(conv)

def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate Fγ(a) w.r.t. a.
    ``w`` is a probability vector (length 2·n‑1) of soft‑max weights.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()

def _gradient_conv_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the convolution entry ``conv[k]`` with respect to ``a``.
    ``conv`` is the linear convolution a*a (full length 2·n‑1).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    # indices i where 0 <= i <= n-1 and 0 <= k-i <= n-1
    i = np.arange(i_start, i_end + 1, dtype=np.intp)
    j = k - i
    # each pair (i, j) contributes a[j] to g[i] and a[i] to g[j]
    np.add.at(g, i, a[j])
    np.add.at(g, j, a[i])
    return g

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    # block width at most 5 % of the length, at least 1
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# ---------- Core optimisation routine ----------

def _optimize_instance(rng: np.random.Generator,
                       n: int,
                       budget: float,
                       init_seq: np.ndarray | None,
                       global_start: float) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑``n`` step‑function for at most ``budget`` seconds.
    Returns the best sequence (sum = 1) found and its exact score.
    """
    # initialise a on the simplex
    if init_seq is not None and init_seq.size == n:
        a = np.asarray(init_seq, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        a = a / s if s > 0 else np.full(n, 1.0 / n)
    else:
        a = rng.random(n)
        a = a / a.sum()

    # FFT length for self‑convolution (used in both phases)
    fft_len = _next_pow2(2 * n - 1)

    # ---------- Phase 1 – smooth‑max with Adam ----------
    # The soft‑max parameters live in the log‑weight space `x`.
    x = np.log(a + 1e-12)                      # avoid log(0)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # hyper‑parameters (linear schedules)
    gamma_start = 0.5
    gamma_end = 30.0
    lr_start = 0.5
    lr_end = 0.02

    # Phase timing

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
The optimizer works in two stages for each independent run.

* **Stage 1 – smooth‑max (Adam).**
  The sequence is represented as a soft‑max of unconstrained log‑weights
  ``x`` (so it always lives on the simplex).  The objective
  ``max_k (a*a)_k`` is approximated by a log‑sum‑exp surrogate.
  Gradients of the surrogate are obtained with a single FFT‑based
  correlation and an Adam step is applied.  A small amount of noise is
  injected if no improvement is seen for a while (soft restart).

* **Stage 2 – exact sub‑gradient (exponentiated‑gradient).**
  After a short warm‑up we switch to the true objective.
  The index ``k*`` where the convolution attains its maximum is found,
  the exact gradient of the entry ``(a*a)[k*]`` w.r.t. ``a`` is computed
  (O(n) using ``np.add.at``), centred, normalised and used in a
  multiplicative update
      ``a_i ← a_i·exp(‑η·g_i)``.
  The vector is re‑normalised to the simplex after each step.
  A tiny jitter is applied if no improvement occurs for a few seconds.

Multiple independent runs with random lengths (≈ 2500‑8000) and random
initialisations are executed until the global time budget (≈ 1000 s) is
exhausted.  The best sequence found (clipped to the limits enforced by
``evaluate_sequence``) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return np.real(conv)


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to ``a``.
    ``w`` is a probability vector (length 2·n‑1) obtained from the
    log‑sum‑exp weights.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _gradient_conv_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the convolution entry ``conv[k]`` with respect to
    the vector ``a`` (where ``conv = a * a`` is the full linear convolution).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n - 1, k)
    i = np.arange(i_start, i_end + 1, dtype=np.intp)
    j = k - i
    # each pair (i, j) contributes a[j] to g[i] and a[i] to g[j]
    np.add.at(g, i, a[j])
    np.add.at(g, j, a[i])
    return g


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _optimize_instance(rng: np.random.Generator,

        a = a / s if s > 0 else np.full(n, 1.0 / n)

        best_seq = np.full(n, 1.0 / n)

    # Clip to evaluate_sequence limits and return a plain list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
