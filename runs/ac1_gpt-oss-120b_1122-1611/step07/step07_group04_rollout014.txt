def gradient_step(a, eta):
    n = len(a)
    fft_len_self = next_pow2(2*n-1)
    # compute self-convolution
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:2*n-1]
    M = conv.max()
    # find first index of max
    kmax = int(np.argmax(conv))
    # w indicator
    w = np.zeros(2*n-1)
    w[kmax] = 1.0
    # compute grad_M via convolution a * w_rev
    w_rev = w[::-1]
    fft_len_grad = next_pow2(n + len(w_rev) - 1)  # = next_pow2(3n-2)
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    conv_full = np.fft.irfft(fa * fw, n=fft_len_grad)
    offset = len(w_rev)-1
    grad_M = 2.0 * conv_full[offset: offset + n]
    S = a.sum()
    C = 2.0 * n
    grad_f = C * (grad_M * S - 2.0 * M) / (S**3)  # shape (n,)
    # update
    a_new = a - eta * grad_f
    # clip
    a_new = np.clip(a_new, 0.0, 1000.0)
    # Ensure sum>0.01 else adjust
    if a_new.sum() < 0.01:
        a_new[:] = 1000.0
    return a_new, float(2 * n * M / (S**2))

"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.  

The algorithm works in three stages:

1. **Broad random sampling** – generate many cheap candidates of different
   shapes (exponential decay, linear, Dirichlet draws, 3‑segment piecewise
   constant, etc.) for a fixed length (≈ 4000).  Keep the best one according
   to the exact objective 2·n·max(conv)/sum².

2. **Exact sub‑gradient descent** – for the current best vector a we compute
   the self‑convolution, locate the index where the maximum is attained,
   and obtain the sub‑gradient of the objective:
       grad f = 2·n·(grad_M·S – 2·M) / S³,
   where M is the maximal convolution value, S = sum(a) and
   grad_M = 2·(a * δ_k_rev) (cross‑correlation with a delta at the active
   shift).  A simple back‑tracking line‑search updates a, followed by clipping
   to the box [0, 1000].  Stagnation triggers a small random block perturbation
   and occasional restarts from the best solution with multiplicative noise.

3. **Simple post‑processing** – after the gradient phase we try a few
   multiplicative scalings (which may saturate some entries at the hard cap
   1000) and keep the scaling that yields the lowest objective.

The whole search respects the 1000 s time budget (5 s safety margin) and
returns the best sequence discovered.  All helper routines are top‑level,
no closures or filesystem access are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, 1000.0)
    return out


def _objective(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` (cheap, no security checks)."""
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    fft_len = _next_pow2(2 * n - 1)
    conv = _conv_self(a, fft_len)
    M = conv.max()
    S = a.sum()
    if S < 0.01:
        return np.inf
    return float(2.0 * n * M / (S ** 2))

#  Main optimisation routine
# ----------------------------------------------------------------------

    and returns the best sequence found.
    """

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # 1) Choose a working length and initialise a feasible vector

        if a0.ndim == 1 and a0.size > 0:
            n = a0.size
            a = np.clip(a0, 0.0, 1000.0)
        else:

    else:

    if a0 is None:
        # Fixed length (power‑of‑two for fast FFT)
        n = 4096

        # ---- generate a diversified portfolio of starting vectors ----
        candidates = []
