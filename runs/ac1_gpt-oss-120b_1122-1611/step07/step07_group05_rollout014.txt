"""
Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm
---------
* **Phase 1 – smoothed‑max gradient descent (≈85 % of the budget)**  
  The true objective is  ``M = max_k (a*a)_k``.  We replace the hard max
  by a smooth soft‑max
      w_k = exp(β·(c_k−max(c))) / Σ_j exp(β·(c_j−max(c))) ,
  where ``c = a*a``.  For large β the weights concentrate on the active
  peak, and the gradient of the smooth objective is
      ∇ = 2·(w * a_rev)[n−1 : n−1+n] ,
  i.e. a convolution of the weight vector with the reversed sequence.
  The update is a projected (onto the probability simplex) additive step
  with a step‑size η that shrinks linearly while β grows from 1 → 200.
  The projection guarantees non‑negativity and a fixed ℓ₁‑norm (scale
  invariance of the evaluation metric).

* **Phase 2 – stochastic fine‑tuning (≈15 % of the budget)**  
  After the smooth phase we switch to the move‑based simulated annealing
  used previously (donor → recipient mass transfers).  The annealing
  temperature follows a linear schedule.  Periodic “jitter’’ steps
  (log‑normal multiplicative noise on a random block) keep the search
  from stagnating.

* **Restarts** – if no improvement is observed for 30 s the current best
  solution is perturbed by moderate log‑normal noise and the process
  continues.

All helper functions are top‑level; there are no closures or lambda
expressions.  The routine respects the 1000 s budget (a 5 s safety
margin).  The returned list is clipped to the limits enforced by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of ``a`` using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(a: np.ndarray, c: np.ndarray, i: int, j: int,
                delta: float) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally (O(n)).  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # subtract donor contribution, add recipient contribution
    c_new[i: i + n] -= 2.0 * delta * a
    c_new[j: j + n] += 2.0 * delta * a

    # quadratic correction terms (i,i), (j,j) and (i,j)
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out

        n = 5000

    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------

    eta_end = 2e-5

    temp_start = 0.07
    temp_end = 1e-7

    move_counter = 0
    RECOMPUTE_EVERY = 1500   # re‑evaluate the full convolution occasionally

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:

        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – smoothed‑max gradient descent
        # --------------------------------------------------------------
        if now < phase1_end:

            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start * (1.0 - frac) + beta_end * frac

            # self‑convolution (needed for the score and for the soft‑max)
            c = _self_conv(a, fft_len_self)

            # soft‑max weights centred on the current peak for stability
            max_c = c.max()
            exp_vals = np.exp(beta * (c - max_c))          # max term → 1
            w = exp_vals / exp_vals.sum()                  # shape (2n‑1,)

            # gradient = 2·(w * a_rev)[n‑1 : n‑1+n]
            a_rev = a[::-1]
            N = _next_pow2(len(w) + n - 1)
            w_fft = np.fft.rfft(w, n=N)
            rev_fft = np.fft.rfft(a_rev, n=N)
            conv_full = np.fft.irfft(w_fft * rev_fft, n=N)[: len(w) + n - 1]
            grad = 2.0 * conv_full[n - 1: n - 1 + n]

            # additive projected gradient step
            a_tmp = a - eta * grad
            a = _project_simplex(a_tmp)

        else:
            # ----------------------------------------------------------
            # Phase 2 – stochastic refinement (simulated annealing)
            # ----------------------------------------------------------
            # linear cooling of temperature
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # locate the current peak of the convolution
            k_max = int(c.argmax())

            # contribution of each index to the peak (a_i·a_{k_max‑i})
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # donor: weighted by its contribution to the peak
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                # nothing useful to move – skip this iteration
                continue

            # recipient – bias towards indices with little mass
            recv_weights = 1.0 - a
            recv_weights[donor] = 0.0
            rw_sum = recv_weights.sum()
            if rw_sum > 0.0:
                recv_weights /= rw_sum
                recipient = rng.choice(np.arange(n), p=recv_weights)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # transfer a fraction of the donor mass
            max_delta = 0.5 * a[donor]
            delta = rng.random() * max_delta

            # apply the move (incremental convolution update)
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # periodic full recomputation to combat drift
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len_self)
                    cur_score = 2.0 * n * c.max()

            # occasional global jitter to escape plateaus
            if rng.random() < 0.015:
                a = _random_block_perturb(a, rng)
                c = _self_conv(a, fft_len_self)
                cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Track the best solution seen so far
        # --------------------------------------------------------------
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Soft restart after prolonged stagnation
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # re‑initialise from the current best with moderate log‑normal noise
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = _project_simplex(a)
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart with jitter (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
