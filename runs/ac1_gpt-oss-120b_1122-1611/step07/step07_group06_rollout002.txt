    if move_type == 'swap':

    rng = np.random.default_rng(seed)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    best_seq = None
    best_score = np.inf
    start_time = time.monotonic()

    if 'height_sequence_1' in globals():
        lengths.append(len(globals()['height_sequence_1']))

    # Remove duplicates, sort

    # allocate roughly equal time to each length (if many lengths)
    total_lengths = len(lengths)
    # we will not strictly allocate but loop lengths while time remains

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # Determine FFT length
        fft_len = _next_pow2(2 * n - 1)

        # Initial vector: uniform start
        a = np.full(n, 1.0 / n, dtype=np.float64)

        # If we have pre-existing height sequence of this length, use it
        if 'height_sequence_1' in globals():
            hs = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if hs.size == n:
                a = hs.copy()
                a = np.clip(a, 0.0, None)
                if a.sum() > 0:
                    a = a / a.sum()
                else:
                    a = np.full(n, 1.0 / n)
        # Score
        current_score = _score(a, fft_len)
        best_local_seq = a.copy()
        best_local_score = current_score
        last_improve = time.monotonic()
        # SA schedule
        t_start = time.monotonic()
        # We'll run until we hit deadline or stagnate for many seconds.
        while time.monotonic() < deadline:
            # fraction of elapsed time relative to total budget (for temperature)
            elapsed = time.monotonic() - t_start
            total = deadline - start_time
            frac = elapsed / total if total > 0 else 1.0
            # temperature schedule: start high then cool
            T = 0.05 * (1.0 - frac) + 1e-6 * frac
            # Propose a move
            move_type = rng.choice(['swap', 'transfer', 'jitter'], p=[0.15, 0.7, 0.15])
            a_new = a.copy()
            if move_type == 'swap':
                i, j = rng.integers(0, n, size=2)
                while i == j:
                    j = rng.integers(0, n)
                a_new[i], a_new[j] = a_new[j], a_new[i]
            elif move_type == 'transfer':
                # pick source with some mass
                nonzero = np.nonzero(a)[0]
                if nonzero.size == 0:
                    continue
                i = rng.choice(nonzero)
                j = rng.integers(0, n)
                if i == j:
                    continue
                # delta proportional to source mass and temperature
                max_delta = a[i] * min(0.5, T * 100)  # higher T allows larger moves
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                a_new[i] -= delta
                a_new[j] += delta
            else:  # jitter
                sigma = 0.2 * (T / (T + 0.01) + 0.05)  # adapt sigma schedule
                factor = np.exp(rng.normal(0.0, sigma, size=n))
                a_new = a * factor
                a_new = _project_onto_simplex(a_new)
            # compute new score
            new_score = _score(a_new, fft_len)
            # acceptance
            if new_score < current_score - 1e-15:
                a = a_new
                current_score = new_score
                if new_score < best_local_score - 1e-12:
                    best_local_score = new_score
                    best_local_seq = a.copy()
                    last_improve = time.monotonic()
                    elapsed_total = last_improve - start_time
                    print(f"[{elapsed_total:6.2f}s] length={n} new best local score={new_score:.6f}")
            else:
                # Metropolis
                accept_prob = math.exp(-(new_score - current_score) / max(T, 1e-12))
                if rng.random() < accept_prob:
                    a = a_new
                    current_score = new_score
            # restart if stagnated for a while
            if time.monotonic() - last_improve > 30.0:
                # create jitter around best local
                sigma_restart = 0.5
                factor = np.exp(rng.normal(0.0, sigma_restart, size=n))
                a = best_local_seq * factor
                a = _project_onto_simplex(a)
                current_score = _score(a, fft_len)
                last_improve = time.monotonic()
                elapsed_total = last_improve - start_time
                print(f"[{elapsed_total:6.2f}s] length={n} restart after stagnation")
        # after exploring this length, compare with global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            elapsed_total = time.monotonic() - start_time
            print(f"[{elapsed_total:6.2f}s] *** new global best {best_score:.6f} (len {n}) ***")

    if best_seq is None:

        best_seq = np.full(1024, 1.0/1024)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

"""
Simulated‑Annealing optimiser for the self‑convolution bound.

The routine explores a non‑negative simplex vector a (∑a = 1) by
random “transfer”, “swap” and multiplicative “jitter” moves.
A temperature schedule turns large, exploratory steps into small,
local refinements.  The objective is the exact value returned by
``evaluate_sequence`` (i.e. 2·n·max(a*a)/|a|₁²).  The algorithm runs
until the time budget (≈ 1000 s) expires, keeps the best vector seen,
and finally returns it as a list of floats.

The implementation uses FFT‑based convolution for fast scoring, a
projection onto the probability simplex, and occasional restarts when
stagnation is detected.  Different sequence lengths are tried
(including the length of a previously discovered construction) to
avoid being trapped in a length‑specific local minimum.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _score(a: np.ndarray, fft_len: int) -> float:
    """
    Compute the exact objective 2·n·max(a*a)/(sum(a)²) using an
    FFT‑based convolution (much faster than ``np.convolve`` for long
    vectors).  The vector a is assumed to be non‑negative; clipping
    and sum‑normalisation are handled outside.
    """
    n = a.shape[0]
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = a.sum()
    # sum_a is kept strictly positive by construction
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` within the given time budget.
    A lightweight simulated‑annealing engine with three move types
    (swap, transfer, jitter) is used; the best vector found is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Global time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # 1) Candidate lengths to explore
    # --------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192]
    if "height_sequence_1" in globals():
        lengths.append(len(globals()["height_sequence_1"]))
    lengths = sorted(set(lengths))

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # 2) Main loop over lengths
    # --------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break

        # FFT size that can hold the full a*a convolution
        fft_len = _next_pow2(2 * n - 1)

        # ---- initialise a simplex vector ---------------------------------
        if "height_sequence_1" in globals():
            h = np.asarray(globals()["height_sequence_1"], dtype=np.float64).copy()
            if h.size != n:
                # adapt size – cut or pad with zeros
                if h.size > n:
                    h = h[:n]
                else:
                    h = np.pad(h, (0, n - h.size), "constant")
            a = np.clip(h, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n)
        else:
            a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

        current_score = _score(a, fft_len)

        # keep a local best for this length
        best_local_seq = a.copy()
        best_local_score = current_score
        last_improve = time.monotonic()

        # ---- temperature schedule parameters -----------------------------
        T_start = 0.05          # initial temperature (allows large moves)
        T_end = 1e-6           # final temperature (only downhill moves)

        # ---- SA main loop -----------------------------------------------
        while time.monotonic() < deadline:
            # fraction of total time elapsed – drives temperature
            elapsed_total = time.monotonic() - start_time
            total_allowed = deadline - start_time
            frac = elapsed_total / total_allowed if total_allowed > 0 else 1.0
            T = T_start * (1.0 - frac) + T_end * frac

            # ----- propose a random move ---------------------------------
            move_type = rng.choice(["swap", "transfer", "jitter"],
                                   p=[0.15, 0.70, 0.15])
            a_new = a.copy()

            if move_type == "swap":
                i, j = rng.integers(0, n, size=2)
                while i == j:
                    j = rng.integers(0, n)
                a_new[i], a_new[j] = a_new[j], a_new[i]

            elif move_type == "transfer":
                # move a small amount from a source index to a target index
                nonzero = np.nonzero(a)[0]
                if nonzero.size == 0:
                    continue
                i = rng.choice(nonzero)
                j = rng.integers(0, n)
                if i == j:
                    continue
                # maximum transferable mass shrinks with temperature
                max_delta = a[i] * min(0.5, T * 100.0)   # at high T up to 50 % of source
                if max_delta <= 0:
                    continue
                delta = rng.random() * max_delta
                a_new[i] -= delta
                a_new[j] += delta

            else:   # jitter – multiplicative log‑normal perturbation
                sigma = 0.2 * (T / (T + 0.01) + 0.05)   # larger sigma early, smaller later
                factor = np.exp(rng.normal(0.0, sigma, size=n))
                a_new = a * factor
                a_new = _project_onto_simplex(a_new)

            # ----- evaluate the proposal ---------------------------------
            new_score = _score(a_new, fft_len)

            # ----- Metropolis acceptance ---------------------------------
            if new_score < current_score - 1e-15:
                accept = True
            else:
                # probability = exp(-(Δ)/T)
                accept = rng.random() < math.exp(-(new_score - current_score) / max(T, 1e-12))

            if accept:
                a = a_new
                current_score = new_score

                # record new local best
                if new_score < best_local_score - 1e-12:
                    best_local_score = new_score
                    best_local_seq = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] length={n} new local best = {new_score:.6f}")

            # ----- stagnation handling -----------------------------------
            if time.monotonic() - last_improve > 30.0:
                # re‑inject a moderate amount of noise around the best local solution
                sigma_restart = 0.5
                factor = np.exp(rng.normal(0.0, sigma_restart, size=n))
                a = best_local_seq * factor
                a = _project_onto_simplex(a)
                current_score = _score(a, fft_len)
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] length={n} restart after stagnation")

        # --------------------------------------------------------------
        # 3) Update global best if this length produced improvement
        # --------------------------------------------------------------
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] *** NEW GLOBAL BEST ({n} entries) = {best_score:.6f} ***")

    # ------------------------------------------------------------------
    # 4) Fallback (should never be needed) and final safety clipping
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Return a plain Python list as required
    return [float(x) for x in best_seq]
