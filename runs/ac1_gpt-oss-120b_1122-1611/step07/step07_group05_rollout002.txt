"""
A two‑phase optimizer for the “max‑convolution” objective.

Phase 1 – smooth‑max gradient descent
    * Approximate max(c) by a soft‑max:  fβ(c)= (1/β)·log∑exp(β·c).
    * The gradient of fβ w.r.t. a is the cross‑correlation of the
      weight vector w∝exp(β·c) with the reversed a.
    * Perform a projected gradient step onto the probability simplex
      (Σa=1, a≥0).  β and the step size η are increased / decreased
      linearly during the allotted time.

Phase 2 – stochastic refinement (simulated annealing)
    * Keep the exact convolution c=a*a (updated incrementally).
    * Repeatedly move a small amount of mass from a “donor” that
      contributes a lot to the current peak of c to a “receiver” that
      contributes little.
    * Accept moves that lower the peak; otherwise accept with a
      Metropolis probability using a linearly cooling temperature.

Both phases share a simple restart mechanism: if no improvement is
seen for a while we jitter the current best solution and continue.
The best vector found before the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _apply_move(
    a: np.ndarray, c: np.ndarray, i: int, j: int, delta: float
) -> tuple[np.ndarray, np.ndarray]:
    """
    Transfer ``delta`` mass from index ``i`` to ``j`` and update the
    convolution ``c`` incrementally.  Returns the new ``a`` and ``c``.
    """
    n = a.size
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    c_new = c.copy()
    # subtract donor contribution, add recipient contribution
    c_new[i : i + n] -= 2.0 * delta * a
    c_new[j : j + n] += 2.0 * delta * a

    # quadratic correction terms
    ii = i + i
    jj = j + j
    ij = i + j
    if ii < c_new.size:
        c_new[ii] += delta * delta
    if jj < c_new.size:
        c_new[jj] += delta * delta
    if ij < c_new.size:
        c_new[ij] -= 2.0 * delta * delta

    return a_new, c_new


def _random_dirichlet(rng: np.random.Generator, n: int, alpha: float = 0.5) -> np.ndarray:
    """Sample a Dirichlet distribution with concentration ``alpha``."""
    x = rng.gamma(alpha, 1.0, size=n)
    if x.sum() == 0:
        return np.full(n, 1.0 / n)
    return x / x.sum()


def _jitter(a: np.ndarray, rng: np.random.Generator, sigma: float = 0.08) -> np.ndarray:
    """Multiplicative log‑normal jitter followed by simplex projection."""
    out = a * np.exp(rng.normal(0.0, sigma, size=a.shape))
    out = np.clip(out, 0.0, None)
    return _project_simplex(out)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Determine length and initialise the vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = init.size
        a = np.clip(init, 0.0, None)
        a = _project_simplex(a)
    else:
        n = 5000
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
        a = _project_simplex(a)                    # ensures exact sum=1

    # ------------------------------------------------------------------
    # 1) Pre‑compute FFT length
    # ------------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Book‑keeping
    # ------------------------------------------------------------------
    # initial convolution and score
    c = _self_conv(a, fft_len)
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    total_time = deadline - start_time
    phase1_end = start_time + 0.70 * total_time   # 70 % for gradient descent

    # ------------------------------------------------------------------
    # 3) Hyper‑parameters
    # ------------------------------------------------------------------
    # Phase‑1 schedules
    eta_start = 0.4          # initial step size
    eta_end = 1e-5           # final step size
    beta_start = 0.5         # soft‑max sharpness at start
    beta_end = 25.0          # sharpness at the end of phase‑1

    # Phase‑2 schedules
    temp_start = 0.06
    temp_end = 1e-7
    RECOMPUTE_EVERY = 800    # full convolution recompute interval

    move_counter = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Phase 1 – smooth‑max gradient descent
        # --------------------------------------------------------------
        if now < phase1_end:
            # progress fraction (0 → 1) inside phase‑1
            prog = (now - start_time) / (phase1_end - start_time)

            eta = eta_start * (1.0 - prog) + eta_end * prog
            beta = beta_start * (1.0 - prog) + beta_end * prog

            # ----- convolution -------------------------------------------------
            a_fft = np.fft.rfft(a, n=fft_len)
            c = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

            # ----- soft‑max weights --------------------------------------------
            max_c = c.max()
            # shift to avoid overflow
            w = np.exp(beta * (c - max_c))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # ----- gradient (cross‑correlation w ⋆ a) -------------------------
            w_fft = np.fft.rfft(w, n=fft_len)
            a_rev_fft = np.fft.rfft(a[::-1], n=fft_len)
            grad_full = np.fft.irfft(w_fft * a_rev_fft, n=fft_len)
            grad = grad_full[n - 1 : n - 1 + n]   # slice of length n

            # ----- projected gradient step ------------------------------------
            a_tmp = a - eta * grad
            a = _project_simplex(a_tmp)

            # recompute score (exact)
            cur_score = 2.0 * n * c.max()

        # --------------------------------------------------------------
        # Phase 2 – stochastic refinement (simulated annealing)
        # --------------------------------------------------------------
        else:
            # temperature schedule (linear cooling)
            t_frac = (now - phase1_end) / (deadline - phase1_end)
            temperature = temp_start * (1.0 - t_frac) + temp_end * t_frac
            temperature = max(temperature, 1e-12)

            # ensure we have current convolution c (it is kept up‑to‑date)
            # locate the current peak
            k_max = int(c.argmax())

            # contribution of each index to the peak (a_i * a_{k_max-i})
            partners = k_max - np.arange(n)
            mask = (partners >= 0) & (partners < n)
            contrib = np.zeros_like(a)
            contrib[mask] = a[mask] * a[partners[mask]]
            total_contrib = contrib.sum()

            # donor: proportional to its contribution to the peak
            if total_contrib > 0.0:
                prob = contrib / total_contrib
                donor = rng.choice(np.arange(n), p=prob)
            else:
                donor = rng.integers(0, n)

            if a[donor] < 1e-14:
                # nothing to move – skip this iteration
                continue

            # recipient: bias towards indices that participate little in the peak
            # we use 1/(1+contrib) as a weight – small contribution → larger weight
            recv_weight = 1.0 / (1.0 + contrib)
            recv_weight[donor] = 0.0
            rw_sum = recv_weight.sum()
            if rw_sum > 0.0:
                recv_weight /= rw_sum
                recipient = rng.choice(np.arange(n), p=recv_weight)
            else:
                recipient = (donor + rng.integers(1, n)) % n

            # transfer amount (up to half of donor mass)
            max_delta = 0.5 * a[donor]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            # apply move (incremental update of convolution)
            a_new, c_new = _apply_move(a, c, donor, recipient, delta)
            new_score = 2.0 * n * c_new.max()

            # Metropolis acceptance
            if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / temperature):
                a = a_new
                c = c_new
                cur_score = new_score
                move_counter += 1

                # periodic exact recomputation to avoid drift
                if move_counter % RECOMPUTE_EVERY == 0:
                    c = _self_conv(a, fft_len)
                    cur_score = 2.0 * n * c.max()

            # occasional random jitter to escape plateaus
            if rng.random() < 0.015:
                a = _jitter(a, rng, sigma=0.10)
                c = _self_conv(a, fft_len)
                cur_score = 2.0 * n * c.max()

        # ------------------------------------------------------------------
        # Keep the best solution seen so far
        # ------------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # Restart after prolonged stagnation
        # ------------------------------------------------------------------
        if now - last_improve > 28.0:
            # Re‑initialise from the current best with a modest jitter
            a = _jitter(best, rng, sigma=0.07)
            c = _self_conv(a, fft_len)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            move_counter = 0
            print(f"[{now - start_time:6.2f}s] restart after stagnation (best {best_score:.6f})")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
