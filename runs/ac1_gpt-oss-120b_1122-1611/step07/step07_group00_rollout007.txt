Implementation of projection onto simplex:

def project_simplex(v):
    """Projection onto simplex sum=1, v>=0."""
    # algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights on the simplex) that improves the
previous best bound (~1.5132).  

Algorithm
---------

1. **Initialisation** – start from the previously found sequence if it exists,
   otherwise use a uniform distribution.  The vector is normalised to sum = 1
   (the objective is homogeneous, so any normalisation works).

2. **Phase 1 – Projected sub‑gradient descent**  
   The objective is `F(a) = 2·n·max_k (a*a)_k`.  
   For a given maximal index `p` the sub‑gradient w.r.t. `a` is
   `g_i = 2·a_{p‑i}` (zero when the index is out of range).  
   We perform an additive step `a ← a – η·g` and project the result back onto
   the probability simplex (non‑negative, sum = 1) using the classic
   O(n log n) projection of Duchi et al.  The step size η follows a
   `η = η0 / sqrt(t)` schedule with occasional random jitter and a restart
   after 20 s of stagnation.

3. **Phase 2 – Local linear‑approximation (LP) refinement**  
   After the sub‑gradient phase we linearise each quadratic convolution
   constraint around the current iterate:
   ```
   Σ_i a_i a_{k‑i} ≤ λ
   ⇒ Σ_i a_cur_{k‑i} a_i ≤ (λ + c_k_cur)/2          (c_k_cur = (a_cur*a_cur)_k)
   ```
   For a small active set (the 200 largest convolution values) we solve a
   linear program that minimises λ under the linearised constraints, the
   simplex constraints and `λ ≥ 0`.  The LP is solved with
   ``scipy.optimize.linprog`` (HiGHS).  Whenever the true objective improves,
   the new point becomes the starting point for the next LP; the process stops
   when no improvement is obtained or the time budget is exhausted.

4. **Return** – the best vector found (clipped to the limits enforced by
   ``evaluate_sequence``) is returned as a plain Python list.

The whole routine respects the 1000 s time budget (a 5 s safety margin) and
uses at most two CPU cores (NumPy FFTs + a single LP solve at a time).  All
helpers are defined at module level, no closures or filesystem access are used.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, sum x = 1}.
    O(n log n) implementation (Duchi et al., 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero -> uniform distribution
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1            # zero‑based index
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _proj_simplex(out)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Load the previous construction if it exists; otherwise use uniform.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a = np.asarray(init_seq, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            a = None
    else:
        a = None

    if a is None:
        # fall back to a uniform distribution of moderate length
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a.sum()
    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution.
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Tracking the best solution discovered so far.
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = float(2.0 * n * c.max())
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4) Phase 1 – Sub‑gradient descent on the exact max‑objective.
    # ------------------------------------------------------------------
    # Allocate roughly half of the available time to the sub‑gradient phase.
    phase1_deadline = start_time + 0.55 * total_time
    it = 0
    eta0 = 0.4  # initial step‑size multiplier

    while time.monotonic() < phase1_deadline:
        it += 1
        now = time.monotonic()
        # compute convolution and locate the maximal index
        c = _self_conv(a, fft_len_self)
        p = int(np.argmax(c))
        cur_score = float(2.0 * n * c[p])
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] new best (sub‑grad) = {best_score:.6f}")

        # sub‑gradient of the max‑convolution (only the active shift p)
        g = np.zeros_like(a)
        lo = max(0, p - (n - 1))
        hi = min(n - 1, p)
        idx = np.arange(lo, hi + 1)
        g[idx] = 2.0 * a[p - idx]

        # diminishing step size (standard 1/√t schedule)
        eta = eta0 / np.sqrt(it)

        # take a step and project back onto the simplex
        a_tmp = a - eta * g
        a = _proj_simplex(a_tmp)

        # occasional stochastic jitter to avoid tiny local traps
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart from the best point if we have stagnated for ≥20 s
        if now - last_improve > 20.0:
            noise = rng.normal(0.0, 0.3, size=n)
            a = _proj_simplex(best + noise)
            it = 0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart after stagnation (sub‑grad)")

    # ------------------------------------------------------------------
    # 5) Phase 2 – Linear‑approximation (LP) refinement.
    # ------------------------------------------------------------------
    # Use the remaining time for a sequence of small LP solves.
    # Only the most offending convolution entries are kept as constraints.
    max_active = 200               # size of the active set for the LP
    min_improve = 1e-6             # stop if LP does not improve enough

    while time.monotonic() < deadline - 2.0:
        now = time.monotonic()
        # recompute convolution for the current best candidate
        c = _self_conv(best, fft_len_self)
        lam_cur = c.max()
        score_cur = 2.0 * n * lam_cur
        # pick the top‑K indices (largest convolution values)
        K = min(max_active, c.size)
        top_idx = np.argpartition(-c, K - 1)[:K]

        # Build the LP matrices (variables = a_0..a_{n-1}, λ)
        rows = np.zeros((K, n + 1), dtype=np.float64)
        rhs = np.empty(K, dtype=np.float64)

        for row_i, k in enumerate(top_idx):
            # the linearised constraint: Σ_i a_cur_{k‑i}·a_i ≤ (λ + c_k_cur)/2
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # fill the coefficient vector for a_i
            idxs = np.arange(lo, hi + 1)
            rows[row_i, idxs] = best[k - idxs]          # a_cur_{k‑i}
            rows[row_i, -1] = -0.5                       # coefficient of λ
            rhs[row_i] = 0.5 * c[k]                     # (λ + c_k_cur)/2 with λ on LHS

        # equality: Σ a_i = 1  (λ has zero coefficient)
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([1.0], dtype=np.float64)

        # variable bounds: a_i ≥ 0, λ ≥ 0
        bounds = [(0.0, None)] * n + [(0.0, None)]

        # objective: minimise λ
        c_obj = np.zeros(n + 1, dtype=np.float64)
        c_obj[-1] = 1.0

        # solve LP
        res = scipy.optimize.linprog(c=c_obj,
                                     A_ub=rows,
                                     b_ub=rhs,
                                     A_eq=A_eq,
                                     b_eq=b_eq,
                                     bounds=bounds,
                                     method='highs',
                                     options={'presolve': True})

        if not res.success:
            print(f"[{now - start_time:6.2f}s] LP failed: {res.message}")
            break

        a_new = res.x[:n]
        lam_new = res.x[-1]
        score_new = 2.0 * n * lam_new

        if score_new + min_improve < best_score:
            best_score = score_new
            best = a_new.copy()
            # re‑normalise to avoid tiny numerical drift
            best = _proj_simplex(best)
            print(f"[{now - start_time:6.2f}s] LP improved score = {best_score:.6f}")
            # continue with the improved point
        else:
            # no further progress – exit the refinement loop
            break

    # ------------------------------------------------------------------
    # 6) Return the best sequence found (clipped to the limits of the evaluator)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
