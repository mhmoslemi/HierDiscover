"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the previous bound (~1.5132).  The optimiser works in three stages:

1.  Adam optimisation on a smooth approximation of the max‑convolution.
    The surrogate Fγ(a) = (1/γ)·log∑exp(γ·(a*a)k) is used with an **exponential**
    schedule for γ (γ grows from 0.5 to 200).  This gives a good global
    direction while keeping the landscape smooth early on.

2.  Aggressive restarts / jitter when the optimiser stalls for more than
    15 s.  Restart points are created by adding moderate log‑normal noise to
    the current best solution, which helps escaping shallow local minima.

3.  A cheap “hill‑climbing” phase that directly moves a small amount of mass
    from one index to another (projected onto the simplex).  It runs for the
    remaining budget and often squeezes a few‑hundredths of a point out of the
    current best value.

All operations are performed with FFT‑based convolutions (O(n log n)) so that
even a sequence with several thousand entries can be updated thousands of
times within the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns a real vector of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa                     # element‑wise multiplication
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate Fγ with respect to a.
    ``w`` is a probability vector (soft‑max weights over the 2·n‑1 convolution
    entries).  The result has length n.
    """
    n = a.size
    a_rev = a[::-1]                       # reversed a for correlation
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)     # correlation a_rev * w
    start = n - 1
    end = start + n
    # factor 2 comes from the derivative of a_i a_{k-i} + a_{k-i} a_i
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()
