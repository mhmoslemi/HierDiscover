"""
Propose a step‑function sequence that (approximately) minimizes

        2·n·max( a * a ) / (∑ a)²

where the convolution is the usual NumPy one.  The implementation
uses two phases:

* **Phase 1 – soft‑max Adam optimisation**  
  The ∞‑norm of the convolution is smoothed with a log‑sum‑exp
  (temperature τ) and Adam updates drive the probability vector.
  The temperature is annealed from a relatively large value (0.5)
  down to 1e‑6, allowing a gradual transition from a smooth
  surrogate to the true max‑objective.

* **Phase 2 – local stochastic hill‑climbing**  
  After the Adam run a fast coordinate‑wise hill‑climber makes tiny
  mass‑exchanges between two indices and accepts a move only if the
  exact evaluation improves.  This often squeezes out the last few
  percent of the bound.

The routine respects the evaluator’s safety checks (non‑negative,
finite, ≤ 1000) and works entirely on the simplex, so the denominator
in the objective is always 1.  It prints progress and returns the best
sequence found before the time‑budget (≈ 1000 s) expires.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen – fallback to uniform
        return np.full_like(v, 1.0 / v.shape[0])
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / w.shape[0]
    else:
        w /= w_sum
    return w


def _softmax_weights(c: np.ndarray, tau: float) -> np.ndarray:
    """
    Numerically stable soft‑max probabilities of a vector ``c`` with
    temperature ``tau``.  The result sums to 1.
    """
    shift = c.max()
    exp_shifted = np.exp((c - shift) / tau)
    total = exp_shifted.sum()
    if total == 0.0:                     # safeguard – should never happen
        return np.full_like(c, 1.0 / c.shape[0])
    return exp_shifted / total

# Main optimisation routine
# ----------------------------------------------------------------------

    minimises ``evaluate_sequence`` within the given time budget.
    The method combines a soft‑max Adam optimiser and a simple
    stochastic hill‑climber.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initialise a probability vector on the simplex
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        start = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if start.ndim != 1 or start.size == 0 or not np.isfinite(start).all():
            start = None
    else:
        start = None

    # choose a length that is a power of two – convenient for FFTs
    n = 4096
    if start is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    else:
        a = start.astype(np.float64).copy()
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a[:] = 1.0 / a.size
        if a.shape[0] != n:
            # resize by truncating / padding with zeros
            if a.shape[0] > n:
                a = a[:n]
            else:
                pad = np.zeros(n - a.shape[0], dtype=np.float64)
                a = np.concatenate([a, pad])
            a = _project_onto_simplex(a)

    # --------------------------------------------------------------
    # 2) FFT set‑up (length enough for a*a via FFT)
    # --------------------------------------------------------------
    fft_len = _next_pow2(3 * n)          # ≥ 2·n‑1 + n‑1
    conv_len = 2 * n - 1

    # --------------------------------------------------------------
    # 3) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    # evaluate exactly once for the starting point
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    best_time = time.monotonic()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Optimisation hyper‑parameters
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1_pow, beta2_pow = beta1, beta2

    # temperature and learning‑rate schedules
    tau_start, tau_end = 0.50, 1e-6
    lr_start, lr_end = 0.60, 0.001

    JITTER_EVERY = 800      # how often to inject multiplicative noise
    PRINT_EVERY = 100       # evaluation frequency (exact score)
    start_time = time.monotonic()
    iteration = 0

    # --------------------------------------------------------------
    # 5) Phase 1 – soft‑max Adam optimiser
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)

        # annealed temperature and learning‑rate
        tau = tau_start * (1.0 - frac) + tau_end * frac
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # forward FFT of the current vector
        a_fft = np.fft.rfft(a, n=fft_len)

        # self‑convolution via point‑wise multiplication
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:conv_len]

        # soft‑max weights – smooth approximation of the max
        w = _softmax_weights(conv, tau)          # length conv_len, sum=1

        # correlation = a ⋆ w  (linear correlation, non‑circular)
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        # extract the “valid” part (same length as a)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a -= step
        a = np.maximum(a, 0.0)                 # enforce non‑negativity
        a = _project_onto_simplex(a)           # keep sum = 1

        iteration += 1

        # occasional multiplicative jitter of a random block
        if iteration % JITTER_EVERY == 0:
            width = rng.integers(1, max(2, n // 15))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.15))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a_jit = np.clip(a_jit, 0.0, None)
            a_jit = _project_onto_simplex(a_jit)
            a = a_jit

        # exact evaluation every few iterations
        if iteration % PRINT_EVERY == 0:
            # maximum of the convolution (computed via FFT already)
            max_c = conv.max()
            score = 2.0 * n * max_c / (a.sum() ** 2)   # sum should be 1
            if score < best_score - 1e-12:
                best_score = float(score)
                best = a.copy()
                best_time = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best = {best_score:.6f} "
                      f"(iter={iteration}, τ={tau:.2e})")
            # also keep a copy of the exact evaluate_sequence value
            # (it applies the same clipping as the scorer)
            eval_seq = evaluate_sequence(a.tolist())
            if eval_seq < best_score - 1e-12:
                best_score = float(eval_seq)
                best = a.copy()
                best_time = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] exact best = {best_score:.6f}")

        # modest restart if we have not improved for a while
        if now - best_time > 45.0:
            a = best.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a *= noise
            a = _project_onto_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            beta1_pow, beta2_pow = beta1, beta2
            best_time = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – injected noise")

        # break if we are within a few seconds of the deadline
        if time.monotonic() > deadline - 5.0:
            break

    # --------------------------------------------------------------
    # 6) Phase 2 – cheap stochastic hill‑climbing around the best
    # --------------------------------------------------------------
    # use the remaining time (if any) for a simple coordinate search
    hill_deadline = time.monotonic() + max(0.0,
                                          deadline - time.monotonic())
    a = best.copy()
    conv = np.fft.irfft(np.fft.rfft(a, fft_len) *
                        np.fft.rfft(a, fft_len), n=fft_len)[:conv_len]
    cur_score = 2.0 * n * conv.max()

    while time.monotonic() < hill_deadline:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # pick a third index (hopefully with small mass) to receive flow
        k = rng.choice(n)
        # amount to move – a modest fraction of the smaller of a[i],a[j]
        delta = min(a[i], a[j]) * rng.uniform(0.02, 0.08)
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] -= delta
        a_new[k] += 2.0 * delta
        if np.min(a_new) < 0.0:
            continue
        # evaluate
        conv_new = np.fft.irfft(np.fft.rfft(a_new, fft_len) *
                               np.fft.rfft(a_new, fft_len), n=fft_len)[:conv_len]
        new_score = 2.0 * n * conv_new.max()
        if new_score < cur_score - 1e-12:
            cur_score = new_score
            a = a_new
            if new_score < best_score - 1e-12:
                best_score = float(new_score)
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] hill‑climb improvement → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (safeguarded against the evaluator’s
    #    clipping limits).  The evaluator is scale‑invariant, so we keep
    #    the simplex‑normalised vector as‑is.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
