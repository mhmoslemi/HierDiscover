def _weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    # a length n, w length 2n-1
    n = a.size
    # fft of a
    fa = np.fft.rfft(a, n=fft_len)
    # fft of w
    fw = np.fft.rfft(w, n=fft_len)
    # conv a * w (full)
    conv_aw = np.fft.irfft(fa * fw, n=fft_len)[: n + w.size - 1]
    # grad2 = conv_aw[:n]
    # rev conv: a * w_rev
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(fa * fw_rev, n=fft_len)[: n + w.size - 1]
    # grad1 = conv_aw_rev[(2*n-2) - np.arange(n)]
    grad1 = conv_aw_rev[(2*n-2) - np.arange(n)]
    grad2 = conv_aw[:n]
    return grad1 + grad2

"""Hybrid smooth‑max gradient descent with simplex projection.

The algorithm works in three phases:

1.  Smooth‑max (log‑sum‑exp) approximation of max(a*a).
    For a sequence `a` (∑a=1, a≥0) we minimise
        Fγ(a)=2·n·(1/γ)·log∑_k exp(γ·(a*a)[k]).
    The gradient of Fγ can be written in closed form using two
    FFT‑based convolutions, which makes each iteration O(n log n).

2.  The smoothing parameter `γ` is increased in stages
    (0.5 → … → 160).  Early stages are very smooth and move the
    solution quickly away from the uniform distribution; later stages
    sharpen the approximation and refine the solution.

3.  A cheap stochastic local refinement finishes the search:
    tiny directed mass‑transfer moves from high‑gradient positions
    to low‑gradient positions are accepted when they improve the
    exact objective.

The routine respects the simplex constraint by Euclidean projection
after every update and stops as soon as the target bound 1.5025 is
reached or the time budget (≈1000 s) is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _weighted_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a*a)[k]  w.r.t. a.
    `w` must have length 2·len(a)‑1 (same length as the convolution).
    """
    n = a.size
    # FFT of a (shared)
    fa = np.fft.rfft(a, n=fft_len)

    # convolution a * w  (gives Σ_j a_j w[i‑j])
    fw = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(fa * fw, n=fft_len)[: n + w.size - 1]

    # convolution a * w_rev  (gives Σ_j a_j w[j+i])
    w_rev = w[::-1]
    fw_rev = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(fa * fw_rev, n=fft_len)[: n + w.size - 1]

    # second term: conv_aw[i] for i=0..n‑1
    grad2 = conv_aw[:n]

    # first term: conv_aw_rev[(2n‑2)‑i] for i=0..n‑1
    grad1 = conv_aw_rev[(2 * n - 2) - np.arange(n)]

    return grad1 + grad2


def _gradient_for_single_active(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)[k] for a single active index k.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(n, k + 1)          # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[k - i]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The function runs for up to ``budget_s`` seconds and returns the best
    sequence found (list of floats).  It starts from the global variable
    ``height_sequence_1`` if present, otherwise from a uniform distribution.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a (normalised to sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64).copy()
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # default length – a power of two makes the FFT tidy
        n0 = 4096
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(3 * n)          # enough for linear convolution

    # --------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best = a.copy()
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – smooth‑max gradient descent with increasing γ
    # --------------------------------------------------------------
    # schedule of smoothing parameters (γ).  Starting small, ending large.
    gammas = [0.5, 2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 160.0]

    # allocate roughly equal time to each γ‑stage
    total_phase_time = deadline - start_time
    stage_time = total_phase_time / len(gammas)

    stage_start = start_time
    for stage_idx, gamma in enumerate(gammas):
        stage_end = stage_start + stage_time
        iteration = 0
        while time.monotonic() < stage_end:
            # --- compute convolution and smooth max ---
            conv = _conv_fft(a, fft_len)                     # length 2n‑1
            # log‑sum‑exp trick for numerical stability
            scaled = gamma * conv
            max_scaled = np.max(scaled)
            exp_term = np.exp(scaled - max_scaled)
            sum_exp = exp_term.sum()
            # soft‑max value (smooth approximation of max)
            smooth_max = (np.log(sum_exp) + max_scaled) / gamma

            # --- compute gradient of the smooth objective ---
            w = exp_term / sum_exp                   # weights sum to 1
            grad_weighted = _weighted_gradient(a, w, fft_len)   # ∂/∂a Σ w·(a*a)
            grad_obj = 2.0 * n * grad_weighted                 # d(2n·smooth_max)/da

            # --- step on the simplex (projected gradient) ---
            eta = 0.6 / np.sqrt(iteration + 1.0)       # diminishing step size
            a_new = a - eta * grad_obj
            a_new = _project_onto_simplex(a_new)

            # --- evaluate exact objective (no scaling needed, sum≈1) ---
            conv_new = _conv_fft(a_new, fft_len)
            score_new = 2.0 * n * conv_new.max()

            if score_new < best_score - 1e-12:
                best_score = score_new
                best = a_new.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (γ={gamma}) = {best_score:.6f} (iter={iteration})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in smooth‑max phase")
                    # final clip & evaluation before returning
                    best = np.clip(best, 0.0, 1000.0)
                    final = evaluate_sequence(best.tolist())
                    print(f"Finished early – final score = {final:.6f}")
                    return best.tolist()

            # Accept the move if it does not worsen the true max‑conv value.
            # This simple safeguard prevents drift caused by the smooth surrogate.
            if score_new <= 2.0 * n * conv.max():
                a = a_new

            iteration += 1
            # occasional progress report
            if iteration % 2000 == 0:
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] phase 1 γ={gamma} iter={iteration}, best={best_score:.6f}")

        stage_start = stage_end

    # --------------------------------------------------------------
    # 4) Phase 2 – stochastic local refinement (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    iteration_local = 0
    # Pre‑compute a gradient map for the current best (used repeatedly)
    while time.monotonic() < deadline:
        # recompute only every few steps to save work
        if iteration_local % 30 == 0:
            conv = _conv_fft(a, fft_len)
            idx_max = int(np.argmax(conv))
            # gradient for the single active index (tight spot)
            grad_active = _gradient_for_single_active(a, idx_max)
            src = int(np.argmax(grad_active))
            # choose a low‑gradient target (could be random among the lowest)
            low_candidates = np.where(grad_active <= np.percentile(grad_active, 15))[0]
            if low_candidates.size > 0:
                tgt = int(rng.choice(low_candidates))
            else:
                tgt = int(rng.integers(0, n))

        # propose a tiny directed mass transfer
        if a[src] <= 0.0:
            src = int(rng.integers(0, n))
        if src == tgt:
            tgt = int(rng.integers(0, n))

        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))   # 0.1 %–5 %
        delta = a[src] * frac

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta
        a_candidate = _project_onto_simplex(a_candidate)

        conv_cand = _conv_fft(a_candidate, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (move {iteration_local})")
            if best_score <= 1.5025:
                print("[early stop] target reached in local phase")
                break

        # keep the improvement if it lowers the true max‑conv value
        if score_cand < 2.0 * n * conv.max():
            a = a_candidate

        iteration_local += 1

        # occasional jitter to escape shallow minima
        if iteration_local % 4000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jitter = a.copy()
            a_jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jitter)

    # --------------------------------------------------------------
    # 5) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
