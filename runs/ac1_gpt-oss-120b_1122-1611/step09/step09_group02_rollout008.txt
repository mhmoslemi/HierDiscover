"""
Propose a step‑function sequence minimizing ``evaluate_sequence``.

Approach:
1. Reduce the search space by grouping neighboring entries into a few
   constant blocks (≈12 samples per block).  The block heights are the
   optimisation variables; the full sequence is obtained by repeating
   each height over its block.
2. Optimise the block heights with a projected Adam optimiser on a
   temperature‑controlled soft‑max surrogate of the max‑convolution.
   The projection is a *weighted* simplex projection that respects the
   true sum‑constraint of the original sequence.
3. When the Adam phase stagnates, inject a multiplicative jitter and
   restart from the best known point.
4. Finally perform a short greedy hill‑climbing on the expanded
   (full‑length) sequence to escape the remaining local plateau.

All heavy sub‑routines (FFT‑based convolution, projection) are
implemented at top‑level.  The routine respects the 1000 s budget and
returns the best sequence found.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def proj_weighted_simplex(v: np.ndarray, block_sizes: np.ndarray) -> np.ndarray:
    """Project v onto the weighted simplex {w>=0, sum_i w_i*size_i = 1}."""
    # work in the space u_i = w_i * size_i, where sum u_i = 1.
    u = v * block_sizes
    u_proj = proj_simplex(u)
    # avoid division by zero (block_sizes > 0 by construction)
    w = u_proj / block_sizes
    return w

def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv

def compute_J(a: np.ndarray, n: int, fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) under the assumption sum(a)=1."""
    fa = np.fft.rfft(a, n=fft_len_self)

    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        if rng.random() < 0.01:
            w = _random_block_perturb(w, rng)
            w = proj_weighted_simplex(w, block_sizes_f)

        # restart after prolonged stagnation
        if time.monotonic() - last_improve > 20.0:
            w = best_w * np.exp(rng.normal(0.0, 0.06, size=m))
            w = proj_weighted_simplex(w, block_sizes_f)
            m_adam.fill(0.0)
            v_adam.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] Adam restart – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) optional hill‑climbing on the full‑length sequence
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        # start from the best full vector obtained so far
        a = w_to_a(best_w)
        best_full_score = best_score
        hc_deadline = time.monotonic() + remaining
        while time.monotonic() < hc_deadline:
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps = rng.random() * min(a[i], 0.01)      # move at most 1 % of donor mass
            if eps <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps
            cur = compute_J(a_new, n, fft_len_self, conv_len)
            if cur < best_full_score:
                best_full_score = cur
                a = a_new
                print(f"[{time.monotonic() - start_time:6.2f}s] hill‑climb new best J = {cur:.6f}")
        best_seq = a.tolist()
        final_score = best_full_score
    else:
        best_seq = w_to_a(best_w).tolist()
        final_score = best_score

    # ------------------------------------------------------------------
    # 8) final clipping (evaluate_sequence would do the same)
    # ------------------------------------------------------------------
    best_seq = [float(min(1000.0, max(0.0, x))) for x in best_seq]

    print(f"[{time.monotonic() - start_time:6.2f}s] final J = {final_score:.6f}")

    return best_seq
