Smoothed maximum-convolution Adam optimizer with random restarts.

The objective to minimise is 2·n·max(a⋆a) where a is a non‑negative
vector on the simplex (∑a_i=1).  The hard max is replaced by a
temperature‑controlled soft‑max (τ·log∑exp(b/τ)) which yields a smooth
gradient that touches all entries of the self‑convolution b.  A classic
Adam optimiser with a linear learning‑rate decay is applied, the iterate
is projected onto the simplex after each step and occasional jitter /
restarts keep the search from stagnating.  Multiple short runs from
different random Dirichlet seeds are performed until the 1000 s budget
expires; the best sequence found is returned.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def _selfconv_max(a: np.ndarray, fft_len: int, conv_len: int):
    """Return the full self‑convolution and its maximum value."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv, float(conv.max())

def _grad_from_softmax(a: np.ndarray, w_soft: np.ndarray,
                       fft_len_grad: int, n: int):
    """
    Exact gradient of 2·n·τ·log∑exp(b/τ) w.r.t. a,
    where w_soft = softmax(b/τ) (length 2·n‑1).
    """
    # Reverse a for convolution
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, n=fft_len_grad)
    fw = np.fft.rfft(w_soft, n=fft_len_grad)
    conv = np.fft.irfft(fa_rev * fw, n=fft_len_grad)
    start = n - 1
    grad_slice = conv[start:start + n]
    grad = 4.0 * n * grad_slice
    return grad

    # ------------------------------------------------------------------

        v /= v.sum()
        seeds.append(v)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    seeds.append(v)                                                  # linear
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    seeds.append(v)                                                  # two spikes

    v = np.zeros(n, dtype=np.float64)

    seeds.append(v)                                                  # half‑heavy

    for _ in range(10):

        seeds.append(rng.dirichlet(np.full(n, conc)))

    best_score = np.inf

        _, max_b = _selfconv_max(vec, fft_len_self, conv_len)
        score = 2.0 * n * max_b
        if score < best_score:

            best_vec = vec.copy()

    # ------------------------------------------------------------------

    tau_factor_start, tau_factor_end = 0.30, 0.001   # τ = factor·max(b)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

        # If not enough time left for a full run, break
        if remaining < 0.8:

        # Initialise a new run (random Dirichlet around the current best)
        # --------------------------------------------------------------

        mix = 0.2
        a = (1.0 - mix) * best_vec + mix * rng.dirichlet(np.full(n, 0.5))
        a = proj_simplex(a)

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow = 1.0
        beta2_pow = 1.0

        # Run Adam for a bounded number of iterations or until time out
        # --------------------------------------------------------------
        for it in range(1, max_iters_per_run + 1):
            now = time.monotonic()
            if now >= deadline:
                break

            # --- self‑convolution and max ---
            conv, max_b = _selfconv_max(a, fft_len_self, conv_len)

            # current objective
            cur_score = 2.0 * n * max_b

            # update global best if needed
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_vec = a.copy()
                last_improve_time = now
                print(f"[{now - start_time:6.2f}s] New best J = {best_score:.6f}")

            # --- temperature (soft‑max) schedule ---
            # fraction of this run that has elapsed
            frac_run = it / max_iters_per_run
            tau_factor = tau_factor_start * (1.0 - frac_run) + tau_factor_end * frac_run
            tau = max_b * tau_factor
            # safeguard against degenerate zero temperature
            if tau < 1e-12:
                # hard max – gradient reduces to the exact sub‑gradient
                w_soft = np.zeros_like(conv)
                idx_max = int(conv.argmax())
                w_soft[idx_max] = 1.0
            else:
                # stable soft‑max (shift by the maximum entry)
                exp_term = np.exp((conv - max_b) / tau)
                w_soft = exp_term / exp_term.sum()

            # --- gradient of the smooth objective ---
            grad = _grad_from_softmax(a, w_soft, fft_len_grad, n)

            # --- Adam bookkeeping ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            # --- learning‑rate schedule (global, not per‑run) ---
            overall_frac = (now - start_time) / (deadline - start_time + safety)
            lr = lr_start * (1.0 - overall_frac) + lr_end * overall_frac

            # --- parameter update and projection ---
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = proj_simplex(a)

            # --- occasional jitter to escape flat plateaux ---
            if rng.random() < 0.015:
                a = _random_block_perturb(a, rng)

            # --- restart on stagnation within the run ---
            if now - last_improve_time > stagnation_restart:
                # perturb the current best and continue
                a = best_vec.copy()
                a = _random_block_perturb(a, rng)
                a = proj_simplex(a)
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow = 1.0
                beta2_pow = 1.0
                last_improve_time = now
                print(f"[{now - start_time:6.2f}s] Restart after stagnation (run {it})")

        # end of one Adam run – loop back for a fresh start
        # small chance to completely re‑sample a fresh Dirichlet
        if rng.random() < 0.05:
            a = rng.dirichlet(np.full(n, rng.uniform(0.05, 2.0)))
            a = proj_simplex(a)

    # ------------------------------------------------------------------
    # 4) Return the best vector (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]

    max_iters_per_run = 1500                # ~1‑2 s per run on a 4 k vector
    lr_start, lr_end = 0.30, 0.005          # global linear LR decay
    tau_factor_start, tau_factor_end = 0.30, 0.001   # τ = factor·max(b)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    last_improve_time = time.monotonic()
    stagnation_restart = 30.0               # seconds without improvement → restart

    total_time = deadline - start_time

    while time.monotonic() < deadline:
        # not enough time left for a full run?
        if deadline - time.monotonic() < 0.8:
            break

        # --------------------------------------------------------------
        # a) Initialise a new short run – mix global best with fresh noise
        # --------------------------------------------------------------
        mix = 0.2
        a = (1.0 - mix) * best_vec + mix * rng.dirichlet(np.full(n, 0.5))
        a = proj_simplex(a)

        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow = 1.0
        beta2_pow = 1.0

        # --------------------------------------------------------------
        # b) Run Adam for a bounded number of iterations
        # --------------------------------------------------------------
        for it in range(1, max_iters_per_run + 1):
            now = time.monotonic()
            if now >= deadline:
                break

            # ----- self‑convolution and current objective -----
            conv, max_b = _selfconv_max(a, fft_len_self, conv_len)
            cur_score = 2.0 * n * max_b

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_vec = a.copy()
                last_improve_time = now
                print(f"[{now - start_time:6.2f}s] New best J = {best_score:.6f}")

            # ----- temperature (soft‑max) schedule -----
            frac_run = it / max_iters_per_run
            tau_factor = tau_factor_start * (1.0 - frac_run) + tau_factor_end * frac_run
            tau = max_b * tau_factor
            if tau < 1e-12:
                # hard max – use exact sub‑gradient
                w_soft = np.zeros_like(conv)
                idx_max = int(conv.argmax())
                w_soft[idx_max] = 1.0
            else:
                # numerically stable soft‑max
                exp_term = np.exp((conv - max_b) / tau)   # max_b is the shift
                w_soft = exp_term / exp_term.sum()

            # ----- gradient of the smooth surrogate -----
            grad = _grad_from_softmax(a, w_soft, fft_len_grad, n)

            # ----- Adam update -----
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            # global learning‑rate decay (linear over the whole budget)
            overall_frac = (now - start_time) / total_time
            lr = lr_start * (1.0 - overall_frac) + lr_end * overall_frac

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = proj_simplex(a)

            # ----- occasional jitter to help escape plateaux -----
            if rng.random() < 0.015:
                a = _random_block_perturb(a, rng)

            # ----- restart on prolonged stagnation -----
            if now - last_improve_time > stagnation_restart:
                a = best_vec.copy()
                a = _random_block_perturb(a, rng)
                a = proj_simplex(a)
                m.fill(0.0)
                v.fill(0.0)
                beta1_pow = 1.0
                beta2_pow = 1.0
                last_improve_time = now
                print(f"[{now - start_time:6.2f}s] Restart after stagnation (run it={it})")

        # small chance to completely re‑sample a fresh Dirichlet for the next run
        if rng.random() < 0.05:
            a = rng.dirichlet(np.full(n, rng.uniform(0.05, 2.0)))
            a = proj_simplex(a)

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
