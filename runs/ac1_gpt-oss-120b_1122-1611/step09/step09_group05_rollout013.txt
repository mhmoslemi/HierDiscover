def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 3  # safety

    best_J = np.inf
    best_vec = None

    # choose a set of lengths to try
    lengths = [512, 1024, 2048, 4096]
    # also try length of prior
    if 'height_sequence_1' in globals():
        prev_len = len(height_sequence_1)
        lengths.append(prev_len)
    lengths = [int(l) for l in lengths if l >= 256 and l <= 8192]

    # loop over lengths and random restarts until time ends
    while time.monotonic() < deadline:
        # randomly choose n from lengths
        n = rng.choice(lengths)

        # initialize a:
        if best_vec is not None and n == len(best_vec):
            a = best_vec.copy()
        else:
            # Dirichlet draw with concentration 0.4 (more spiky)
            a = rng.dirichlet(np.full(n, 0.4))

        # ensure sum = 1 (already)
        # subgradient descent
        max_iters = 3000
        # maybe adapt max_iters based on remaining time
        for t in range(1, max_iters+1):
            now = time.monotonic()
            if now >= deadline:
                break

            # compute conv via FFT
            conv_len = 2*n - 1
            fft_len = 1 << ((conv_len-1).bit_length())
            A = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
            conv_max = conv.max()
            J = 2.0 * n * conv_max

            # update best
            if J < best_J:
                best_J = J
                best_vec = a.copy()
                elapsed = now - start
                print(f"[{elapsed:6.2f}s] New best J={best_J:.6f} n={n}")

            # compute gradient (subgradient for max)
            # active indices within a tolerance
            # use absolute tolerance based on conv_max magnitude
            eps = 1e-12
            active = np.where(conv >= conv_max - eps)[0]
            # compute shift sum
            g = np.zeros_like(a)
            for k in active:
                # compute slice positions
                # j from max(0, k-(n-1)) to min(n-1, k)
                start_j = max(0, k - (n - 1))
                end_j = min(n - 1, k)
                j_range = np.arange(start_j, end_j + 1)
                g[j_range] += a[k - j_range]
            g *= 2.0

            # step size
            step0 = 0.3
            step = step0 / np.sqrt(t)
            a = a - step * g
            # project onto simplex
            # use proj_simplex defined earlier
            a = proj_simplex(a)

            # optional early break if gradient small
            if np.linalg.norm(g) * step < 1e-10:
                break

        # after finishing this run, maybe restart with jitter
        # jitter best vector slightly and continue
        if best_vec is not None:
            a = best_vec * np.exp(rng.normal(0.0, 0.02, size=n))
            a = proj_simplex(a)

    # Return best vector as list
    if best_vec is None:
        # fallback uniform
        n = 1024
        best_vec = np.full(n, 1.0 / n)
    # Clip to bounds for evaluate_sequence
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    # Normalize again
    s = best_vec.sum()
    if s > 0:
        best_vec = best_vec / s
    else:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)

    return [float(x) for x in best_vec]
```

"""
Projected‑subgradient optimiser for the step‑function problem.

The objective (with ∑a = 1) is   J(a) = 2·n·max(conv(a,a)).
Because the objective is convex in the non‑negative orthant,
the subgradient of max(conv) is obtained from the active convolution
indices.  The optimiser repeatedly

  – evaluates J and the convolution,
  – builds the subgradient from the active indices,
  – takes a diminishing step opposite to the subgradient,
  – projects the iterate back onto the probability simplex.

Multiple random restarts (different lengths and initial shapes) are
performed until the allocated time budget is exhausted.  When no
improvement is seen for 30 s a small multiplicative jitter is applied
to the current best point to escape plateaus.  The best vector found is
returned as a plain list of floats (it already satisfies the constraints
required by ``evaluate_sequence``).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_vector(n: int, rng: np.random.Generator) -> np.ndarray:
    """Return a random feasible probability vector of length *n*."""
    choice = rng.choice(
        ["dirichlet", "exp", "lin_inc", "lin_dec", "uniform", "two_spike"]
    )
    if choice == "dirichlet":
        # concentration in (0.2, 0.8) gives a variety of shapes
        alpha = rng.uniform(0.2, 0.8)
        return rng.dirichlet(np.full(n, alpha))
    if choice == "exp":
        r = rng.uniform(0.90, 0.99)
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        return v
    if choice == "lin_inc":
        v = np.arange(1, n + 1, dtype=np.float64)
        v /= v.sum()
        return v
    if choice == "lin_dec":
        v = np.arange(n, 0, -1, dtype=np.float64)
        v /= v.sum()
        return v
    if choice == "two_spike":
        v = np.zeros(n, dtype=np.float64)
        v[0] = 0.5
        v[-1] = 0.5
        return v
    # uniform fallback
    return np.full(n, 1.0 / n, dtype=np.float64)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the full self‑convolution of *a* using an FFT of length *fft_len*."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    conv_len = 2 * a.size - 1
    return conv[:conv_len]


def _subgradient_maxconv(a: np.ndarray, conv: np.ndarray, conv_max: float) -> np.ndarray:
    """
    Subgradient of   max(conv(a,a))   w.r.t. *a*.
    It is 2·∑_{k∈Active} shift_k(a) where shift_k reproduces a_{k‑j}.
    """
    n = a.size
    # tolerance to collect all indices that are (numerically) maximal
    tol = max(conv_max * 1e-12, 1e-14)
    active = np.where(conv >= conv_max - tol)[0]

    g = np.zeros_like(a)
    for k in active:
        if k < n:
            # contribution on the left side of the convolution
            g[: k + 1] += a[k::-1]
        else:
            # contribution on the right side
            offset = k - (n - 1)
            # slice a[n‑1 : offset‑1 : -1] works for offset≥1
            g[offset:] += a[n - 1 : offset - 1 : -1]
    g *= 2.0
    return g


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # a small safety margin

    # ------------------------------------------------------------------
    # 0. Candidate lengths (the previous construction is included if present)
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]
    if "height_sequence_1" in globals():
        try:
            prev_len = len(globals()["height_sequence_1"])
            if 256 <= prev_len <= 8192:
                lengths.append(prev_len)
        except Exception:
            pass
    lengths = sorted(set(lengths))

    best_J = np.inf
    best_vec = None
    last_improve = start_time

    # ------------------------------------------------------------------
    # 1. Main optimisation loop – repeated random restarts
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (larger lengths are a bit more likely)
        n = rng.choice(lengths)
        conv_len = 2 * n - 1
        fft_len = _next_pow2(conv_len)

        # initialise a feasible point
        if best_vec is not None and len(best_vec) == n:
            a = best_vec.copy()
        else:
            a = _initial_vector(n, rng)

        # normalise – the projection routine also guarantees this, but keep it exact
        a /= a.sum()

        # ------------------------------------------------------------------
        # 2. Projected sub‑gradient descent on the simplex
        # ------------------------------------------------------------------
        step0 = 0.5                 # initial step magnitude
        max_inner = 4000           # safe upper bound on inner iterations
        inner_last_improve = time.monotonic()

        for it in range(1, max_inner + 1):
            now = time.monotonic()
            if now >= deadline:
                break

            # 2.1 current objective
            conv = _conv_self(a, fft_len)
            conv_max = conv.max()
            J = 2.0 * n * conv_max

            # 2.2 record global improvement
            if J < best_J - 1e-12:
                best_J = J
                best_vec = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New best J={best_J:.6f}  n={n}")
                last_improve = now
                inner_last_improve = now

            # 2.3 sub‑gradient of the max‑convolution term
            g = _subgradient_maxconv(a, conv, conv_max)

            # 2.4 step size (Polyak‑style diminishing schedule)
            step = step0 / np.sqrt(it)

            # 2.5 gradient step + simplex projection
            a = a - step * g
            a = proj_simplex(a)

            # 2.6 stagnation handling – jitter after 30 s without improvement
            if now - inner_last_improve > 30.0:
                if best_vec is not None:
                    a = best_vec * np.exp(rng.normal(0.0, 0.03, size=n))
                    a = proj_simplex(a)
                inner_last_improve = now
                # modestly shrink the initial step to keep later moves gentle
                step0 = max(step0 * 0.9, 0.05)

        # ------------------------------------------------------------------
        # 3. After a run, add a tiny random perturbation to the best point
        #    before the next restart (helps to explore new basins)
        # ------------------------------------------------------------------
        if best_vec is not None:
            a = best_vec * np.exp(rng.normal(0.0, 0.02, size=len(best_vec)))
            a = proj_simplex(a)

    # ------------------------------------------------------------------
    # 4. Fallback in the unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 1024
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 5. Final normalisation and clipping to respect ``evaluate_sequence`` limits
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)

    return [float(x) for x in best_vec]
