            a = _random_block_perturb(a, rng)
    # final return
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]

"""
Propose a step‑function (non‑negative vector) that approximately minimises
``evaluate_sequence``.  The algorithm is deliberately different from the
previous DE → sub‑gradient → Adam pipeline:

1. **Initialisation** – take the best known construction (if any), otherwise
   start from a few diverse simple candidates (uniform, exponential decay,
   Dirichlet draws).

2. **Random‑search warm‑up** – spend a few seconds sampling Dirichlet vectors
   and keep the best; this gives a good starting point for the intensive phase.

3. **Simulated‑Annealing (SA) optimiser** – the main workhorse.
   Each SA step proposes either
   * a multiplicative log‑normal perturbation of a random contiguous block,
   * or a projected gradient move that reduces the currently maximal
     self‑convolution entry.
   The proposal is accepted if it improves the objective or with a
   Metropolis probability that decays with a linear temperature schedule.
   The temperature, block‑size and perturbation scale all shrink
   gradually, allowing large exploratory moves early and fine‑grained
   refinement later.

4. **Optional Adam polish** – after the SA budget is exhausted a very short
   Adam run on a soft‑max surrogate of the convolution sharpens the final
   result.

All helpers are top‑level (no closures) and the routine respects a 1000 s
budget (leaving a 5 s safety margin).  The returned vector is already
projected onto the simplex, clipped to the evaluation limits and converted
to a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Top‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator,
                          sigma: float = 0.15) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = proj_simplex(out)
    return out


def self_convolution(a: np.ndarray, fft_len: int,
                     conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int,
              conv_len: int) -> float:
    """
    Exact objective used by ``evaluate_sequence`` **under the
    simplex normalisation** (sum(a)=1):
        J = 2·n·max(a∗a)
    """
    conv = self_convolution(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The search consists of a short random‑search warm‑up,
    a simulated‑annealing phase, and a brief Adam refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a problem size
    # ------------------------------------------------------------------
    # Use the length of the previous construction if it exists and enlarge it
    # a little – a few extra degrees of freedom often help flatten the
    # convolution.
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
        except Exception:
            n0 = 3809
        # target n: roughly 1.3 × previous length, capped to a comfortable range
        n = max(2048, min(8000, int(1.3 * n0)))
    else:
        n = 5000                                   # a reasonable default

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2) Helper for objective (sum is implicitly 1)
    # ------------------------------------------------------------------
    # (compute_J defined above)

    # ------------------------------------------------------------------
    # 3) Build an initial portfolio of candidates and keep the best one
    # ------------------------------------------------------------------
    candidates: list[np.ndarray] = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays (a few rates)
    for r in [0.99, 0.975, 0.95, 0.9, 0.85]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # Dirichlet draws – moderately concentrated
    for _ in range(5):
        candidates.append(rng.dirichlet(np.ones(n) * 0.3))

    # previous best, interpolated/rescaled to the current length
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s == 0.0:
            prev = np.full_like(prev, 1.0 / n)
        else:
            prev = prev / s
        candidates.append(prev)

    # ------------------------------------------------------------------
    # 4) Evaluate the portfolio and keep the current best
    # ------------------------------------------------------------------
    best_vec = None
    best_val = float("inf")
    for v in candidates:
        val = compute_J(v, n, fft_len, conv_len)
        if val < best_val:
            best_val = val
            best_vec = v.copy()
    elapsed = time.monotonic() - start
    print(f"[{elapsed:6.2f}s] initial best J = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Tiny random‑search warm‑up (≈5 % of remaining budget)
    # ------------------------------------------------------------------
    warmup_deadline = time.monotonic() + 0.05 * (deadline - time.monotonic())
    while time.monotonic() < warmup_deadline:
        # draw a moderately sparse Dirichlet vector
        v = rng.dirichlet(np.ones(n) * 0.1)
        val = compute_J(v, n, fft_len, conv_len)
        if val < best_val:
            best_val = val
            best_vec = v.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] warm‑up new best J = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Simulated‑Annealing optimisation
    # ------------------------------------------------------------------
    # SA parameters – they decay linearly with the fraction of the remaining
    # time, enabling aggressive exploration at the beginning and fine tuning
    # near the end.
    T_start = 0.04          # initial temperature
    T_end = 1e-6            # final temperature
    sigma_start = 0.30      # initial perturbation width (log‑normal)
    sigma_end = 0.015       # final perturbation width
    eta_start = 0.30        # initial step size for gradient moves
    eta_end = 0.001         # final step size for gradient moves

    a = best_vec.copy()
    cur_val = best_val
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        fraction = elapsed / budget_s                       # ∈[0,1]

        # linear schedules
        T = T_start * (1.0 - fraction) + T_end * fraction
        sigma = sigma_start * (1.0 - fraction) + sigma_end * fraction
        eta = eta_start * (1.0 - fraction) + eta_end * fraction

        # --------------------------------------------------------------
        # 6.1) Propose a candidate
        # --------------------------------------------------------------
        if rng.random() < 0.55:
            # ----- block‑scaling move (multiplicative log‑normal) -----
            a_new = _random_block_perturb(a, rng, sigma=sigma)
        else:
            # ----- projected gradient move on the *hard* max -----
            conv = self_convolution(a, fft_len, conv_len)
            k_max = int(np.argmax(conv))

            # sub‑gradient of the maximal entry (hard max)
            idx = k_max - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[idx[mask]]

            a_new = a - eta * grad
            a_new = np.clip(a_new, 0.0, None)
            a_new = proj_simplex(a_new)

        # --------------------------------------------------------------
        # 6.2) Evaluate the proposal
        # --------------------------------------------------------------
        new_val = compute_J(a_new, n, fft_len, conv_len)

        # --------------------------------------------------------------
        # 6.3) Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # uphill move – accept with probability exp(-Δ/T)
            prob = np.exp(-(new_val - cur_val) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_new
            cur_val = new_val
            if cur_val < best_val - 1e-14:
                best_val = cur_val
                best_vec = a.copy()
                last_improve = now
                elapsed = now - start
                print(f"[{elapsed:6.2f}s] SA new best J = {best_val:.6f}")

        # --------------------------------------------------------------
        # 6.4) Small random jitter to keep the search lively
        # --------------------------------------------------------------
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng, sigma=sigma)

        # --------------------------------------------------------------
        # 6.5) Restart after a long stagnation period
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            # perturb the current best and continue from there
            a = _random_block_perturb(best_vec, rng, sigma=0.6)
            cur_val = best_val
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] SA restart – best J = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7) Very short Adam polish (optional – uses a soft‑max surrogate)
    # ------------------------------------------------------------------
    # Run only if a sliver of time remains (≥1 s).  This does not hurt the
    # overall result and may shave off a few extra digits.
    remaining = deadline - time.monotonic()
    if remaining > 1.0:
        # Adam hyper‑parameters
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        m = np.zeros_like(best_vec)
        v = np.zeros_like(best_vec)
        it = 0
        adam_start = time.monotonic()
        adam_deadline = deadline
        total_adam = adam_deadline - adam_start

        while time.monotonic() < adam_deadline:
            it += 1
            now_it = time.monotonic()
            frac = (now_it - adam_start) / total_adam

            # learning‑rate schedule
            lr = 0.25 * (1.0 - frac) + 0.001 * frac
            # temperature for soft‑max (starts low, ends high)
            alpha = 2.0 * (1.0 - frac) + 3000.0 * frac

            # ----- linear convolution (hard) -----
            conv = self_convolution(best_vec, fft_len, conv_len)

            # ----- smooth surrogate (soft‑max) -----
            c_max = conv.max()
            shifted = conv - c_max
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the surrogate: (a ⋆ w_rev)
            w_rev = w[::-1]
            fa = np.fft.rfft(best_vec, n=fft_len)
            fw = np.fft.rfft(w_rev, n=fft_len)
            full = np.fft.irfft(fa * fw, n=fft_len)
            start_idx = w_rev.size - 1          # = 2·n‑2
            grad = 2.0 * full[start_idx:start_idx + n]

            # Adam update + simplex projection
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            best_vec = best_vec - lr * m_hat / (np.sqrt(v_hat) + eps)
            best_vec = proj_simplex(best_vec)

            # evaluate true objective
            cur_obj = compute_J(best_vec, n, fft_len, conv_len)
            if cur_obj < best_val - 1e-14:
                best_val = cur_obj
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] Adam polish new best J = {best_val:.6f}")

            # (optional) occasional jitter
            if rng.random() < 0.01:
                best_vec = _random_block_perturb(best_vec, rng, sigma=0.04)

    # ------------------------------------------------------------------
    # 8) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
