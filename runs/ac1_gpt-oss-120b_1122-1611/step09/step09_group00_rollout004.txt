def _local_refine(a, rng, deadline):
    n = a.size
    fft_len = _next_pow2(3 * n)
    best = a.copy()
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()
    iteration = 0
    # compute gradient for index that yields maximum conv
    while time.monotonic() < deadline:
        conv = _conv_fft(best, fft_len)
        max_b = conv.max()
        idx_max = int(np.argmax(conv))
        # gradient for this idx only
        grad = _gradient_from_active(best, np.array([idx_max]))
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))
        if src == tgt:
            # random fallback
            src = rng.integers(0, n)
            tgt = rng.integers(0, n)
        # try moving some mass
        if best[src] == 0.0:
            src = rng.integers(0, n)
        frac = rng.uniform(0.001, 0.03)  # up to 3%
        delta = best[src] * frac
        cand = best.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)  # ensure sum=1 and non-negative
        cand_score = 2.0 * n * _conv_fft(cand, fft_len).max()
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand
        # occasional random perturbation
        if iteration % 2000 == 0:
            # jitter a random block

"""
Multi‑restart Adam optimizer with focused local refinement.

The search works in three logical layers

1.  **Problem size** – we start from the best known construction
    (the global ``height_sequence_1`` if present) and, if it is shorter
    than a modest target (≈5000), we up‑sample it to a power‑of‑two length
    (helps FFT bookkeeping).  When no prior construction is available we
    fall back to a uniform start.

2.  **Adam phase** – each restart runs a few hundred Adam steps on the
    log‑probabilities (exponentiated‑gradient ≈ Adam).  The stochastic
    gradient is the exact sub‑gradient of the current maximal convolution
    entries (all indices within a tiny tolerance of the max).  The update
    is projected back to the simplex by normalising the exponentiated
    variables.  A small amount of Gaussian jitter is added occasionally
    to shake the iterates out of plateaus.

3.  **Local refinement** – after every Adam run we perform a cheap
    coordinate‑wise mass‑transfer search that repeatedly moves a small
    fraction of probability mass from the index with the largest positive
    gradient (i.e. most harmful for the max‑convolution) towards the
    index with the smallest gradient.  The move is accepted only if the
    exact score improves.  Random “block jitter’’ moves are also tried
    periodically.

The total time budget (``budget_s``) is split evenly across a small
fixed number of independent restarts (default = 4).  The best sequence
found over all restarts is returned (clipped to the evaluator limits)."""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("only 1‑D vectors are supported")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.full_like(v, 1.0 / len(v))
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution vector of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of   max_k (a*a)[k]   for the set ``active`` of
    indices attaining (or nearly attaining) the current maximum.
    For each active index k the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact evaluate_sequence score for a vector whose sum is 1."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam optimiser (log‑space)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   rng: np.random.Generator,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Run Adam on the log‑probabilities of ``a0`` until ``deadline``.
    Returns the best vector seen and its score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)

    log_a = np.log(a + 1e-15)          # keep away from log(0)
    m = np.zeros_like(log_a)
    v = np.zeros_like(log_a)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.20                       # initial learning‑rate (in log‑space)

    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        # active set – all indices within a relative tolerance of the max
        eps_active = max_b * 1e-6
        active = np.where(conv >= max_b - eps_active)[0]

        grad = _gradient_from_active(a, active)

        # Adam update on log‑probabilities
        lr = lr0 / np.sqrt(it)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.exp(log_a)
        a /= a.sum()                     # projection onto the simplex

        # occasional exact evaluation
        if it % 20 == 0:
            cur_score = _score_from_a(a, fft_len)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                # progress print is done by the caller

        # tiny stochastic jitter to escape flat regions
        if rng.random() < 0.001:
            log_a += rng.normal(scale=0.02, size=log_a.shape)

    return best, best_score


# ----------------------------------------------------------------------
# Focused local refinement (mass‑transfer moves)
# ----------------------------------------------------------------------


def _local_refine(a0: np.ndarray,
                  rng: np.random.Generator,
                  deadline: float) -> tuple[np.ndarray, float]:
    """
    Very cheap directed neighbourhood search.
    Repeatedly moves a small fraction of probability from the index with
    the largest positive gradient (w.r.t. the current maximal convolution)
    to the index with the smallest gradient.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)

    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        conv = _conv_fft(best, fft_len)
        max_b = conv.max()
        idx_max = int(np.argmax(conv))

        # gradient corresponding to the *single* active index
        grad = _gradient_from_active(best, np.array([idx_max]))
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))

        if src == tgt:
            src = rng.integers(0, n)
            tgt = rng.integers(0, n)

        if best[src] <= 0.0:
            src = rng.integers(0, n)
        if src == tgt:
            tgt = rng.integers(0, n)

        # transfer a tiny fraction of the source mass
        frac = rng.uniform(0.001, 0.03)            # 0.1 % – 3 %
        delta = best[src] * frac

        cand = best.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        cand_score = _score_from_a(cand, fft_len)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand

        # block‑jitter every few thousand iterations
        if it % 2000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(loc=0.0, scale=0.018))
            jitter = best.copy()
            jitter[start:start + width] *= factor
            jitter = _project_onto_simplex(jitter)
            jitter_score = _score_from_a(jitter, fft_len)
            if jitter_score < best_score:
                best_score = jitter_score
                best = jitter

    return best, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.

    The method performs several independent Adam‑optimisation runs
    (with different random starts), each followed by a fast local
    refinement.  The best solution found overall is returned.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a small safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 0) Determine problem size and initial vector
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base_seq = np.clip(base_seq, 0.0, None)
        if base_seq.sum() > 0.0:
            base_seq /= base_seq.sum()
        else:
            base_seq = np.full_like(base_seq, 1.0 / base_seq.size)
        base_n = base_seq.size
    else:
        base_seq = None
        base_n = 0

    # we would like at least a few thousand entries – larger gives a
    # richer search space but also costs a bit more per iteration.
    target_n = max(base_n, 5000)
    n = _next_pow2(target_n)

    if base_seq is not None and base_seq.size != n:
        # linear interpolation to the new length
        xs_old = np.linspace(0, base_seq.size - 1, base_seq.size)
        xs_new = np.linspace(0, base_seq.size - 1, n)
        a0 = np.interp(xs_new, xs_old, base_seq)
        a0 = np.clip(a0, 0.0, None)
        a0 /= a0.sum()
    elif base_seq is not None:
        a0 = base_seq.copy()
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 1) Split the remaining budget into a few independent restarts
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    n_restarts = 4
    restart_budget = remaining / n_restarts

    best_overall = a0.copy()
    best_score = _score_from_a(best_overall, _next_pow2(3 * n))
    print(f"[0.00s] initial (length={n}) score = {best_score:.6f}")

    for restart_idx in range(n_restarts):
        # random start for all but the first restart
        if restart_idx == 0:
            a_start = best_overall.copy()
        else:
            # sparse Dirichlet (alpha < 1) to generate diverse shapes
            a_start = rng.dirichlet(alpha=np.full(n, 0.5))

        # ------------------------------------------------------------------
        # 2) Adam optimisation phase
        # ------------------------------------------------------------------
        restart_deadline = time.monotonic() + restart_budget * 0.70
        a_opt, score_opt = _adam_optimize(a_start, rng, restart_deadline)

        # ------------------------------------------------------------------
        # 3) Local refinement phase (takes the remaining 30 % of the slot)
        # ------------------------------------------------------------------
        refine_deadline = time.monotonic() + restart_budget * 0.30
        a_ref, score_ref = _local_refine(a_opt, rng, refine_deadline)

        # ------------------------------------------------------------------
        # 4) Update global best if improvement was found
        # ------------------------------------------------------------------
        if score_ref < best_score - 1e-12:
            best_score = score_ref
            best_overall = a_ref.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new overall best (restart {restart_idx}) = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final evaluation with the official checker (protect against clipping)
    # ------------------------------------------------------------------
    best_overall = np.clip(best_overall, 0.0, 1000.0)
    final_score = evaluate_sequence(best_overall.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_overall]
