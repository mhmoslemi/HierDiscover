Propose a step‑function sequence that minimises ``evaluate_sequence``.

Algorithm
---------
1. Global sweep:
   - Uniform baseline.
   - Two–edge families (linear, power‑taper, exponential‑taper).
   - “Min‑power’’ shape  w_i ∝ (min(i,n‑1‑i)+1)^{-s}.
   - Cosine family (as in the reference code).
   - Random Dirichlet draws (various concentrations) and the previous best
     sequence if it exists.
   The best ~8 candidates (small J) are retained.

2. Local refinement for each retained seed:
   * **Adam** on a soft‑max surrogate of the max‑convolution (temperature
     grows to 5000, learning‑rate decays).
   * **Sub‑gradient** descent on the true max (binary mask of the maximal
     convolution entries) with a diminishing step‑size.
   * **Hill‑climb** – tiny mass transfers between random coordinates,
     accepting only improvements (occasional jitter to escape plateaus).

   Time is split proportionally among the seeds; the whole process stops
   5 s before the allotted budget.

3. The best probability vector *p* is scaled to the required L1‑norm
   ``sqrt(2·n)`` and clipped to the limits enforced by ``evaluate_sequence``.

All heavy lifting (convolution, gradient) uses FFT‑based O(n log n) kernels,
so thousands of objective evaluations fit easily into the 1000 s budget.
"""

# Imports & utilities (top‑level – no closures)
# ----------------------------------------------------------------------
import time
import math
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """Two uniform edge blocks of length L, linear taper controlled by gamma."""
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    if L > 1:
        idx = np.arange(L)
        left = 1.0 - gamma * (idx / (L - 1))
    else:
        left = np.ones(1)
    v[:L] = left
    v[-L:] = left[::-1]                     # mirrored right block
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()

def _edge_power_shape(n: int, L: int, power: float) -> np.ndarray:
    """Two edge blocks with power‑law taper (power=1 → linear)."""
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    if L > 1:
        idx = np.arange(L)
        left = (1.0 - idx / (L - 1)) ** power
    else:
        left = np.ones(1)
    v[:L] = left
    v[-L:] = left[::-1]
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()

def _edge_exp_shape(n: int, L: int, lam: float) -> np.ndarray:
    """Two edge blocks with exponential taper exp(-lam * i/(L‑1))."""
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    if L > 1:
        idx = np.arange(L)
        left = np.exp(-lam * idx / (L - 1))
    else:
        left = np.ones(1)
    v[:L] = left
    v[-L:] = left[::-1]
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()

def _min_pow_shape(n: int, s: float) -> np.ndarray:
    """Weight ∝ (min(i,n‑1‑i)+1)^{-s}, i.e. heavy edges, light centre."""
    idx = np.arange(n)
    w = (np.minimum(idx, n - 1 - idx) + 1) ** (-s)
    if w.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return w / w.sum()

def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape (1 + r·cos θ)^power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()

def _objective(p: np.ndarray, n: int, fft_len_self: int) -> float:
    """Exact objective J = 2·n·max(p*p)."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())

def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max (sums to 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum

def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of soft‑max surrogate w.r.t. a."""
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1                     # = 2n‑2
    return 2.0 * full[start:start + a.size]

def _grad_subgrad(a: np.ndarray, mask: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient for the true max‑convolution (binary mask)."""
    w_rev = mask[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1
    return 2.0 * full[start:start + a.size]

def _adam_optimize(p_start: np.ndarray,
                   n: int,
                   fft_len_self: int,
                   fft_len_grad: int,
                   time_limit: float,
                   rng: np.random.Generator) -> np.ndarray:
    """
    Projected Adam on the soft‑max surrogate.
    Returns the best probability vector seen before the time limit.
    """
    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_a = a.copy()
    best_J = _objective(a, n, fft_len_self)
    start = time.monotonic()
    it = 0
    while time.monotonic() < time_limit:
        it += 1
        # schedule
        now = time.monotonic()
        frac = (now - start) / (time_limit - start)
        lr = 0.8 * (0.5 ** frac) + 0.001 * (1 - 0.5 ** frac)  # decaying LR
        alpha = 1.0 * (1.0 - frac) + 5000.0 * frac            # heating
        # soft‑max surrogate
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)
        grad = _grad_softmax(a, w, fft_len_grad)
        # Adam update + projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)
        # evaluate true objective
        cur_J = _objective(a, n, fft_len_self)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
        # occasional jitter to escape plateaus
        if rng.random() < 0.02:
            block = rng.integers(1, max(2, n // 25))
            start_idx = rng.integers(0, n - block + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block] *= factor
            a_pert = proj_simplex(a_pert)
            a = a_pert
    return best_a

def _subgrad_optimize(p_start: np.ndarray,
                      n: int,
                      fft_len_self: int,
                      fft_len_grad: int,
                      time_limit: float,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Sub‑gradient descent on the exact max‑convolution.
    Simple 1/√t step‑size schedule, occasional random jitter.
    """
    a = p_start.copy()
    best_a = a.copy()
    best_J = _objective(a, n, fft_len_self)
    start = time.monotonic()
    it = 0
    while time.monotonic() < time_limit:
        it += 1
        conv = _self_conv(a, fft_len_self)
        max_val = conv.max()
        mask = (conv >= max_val - 1e-12).astype(np.float64)   # binary mask
        grad = _grad_subgrad(a, mask, fft_len_grad)
        lr = 0.3 / math.sqrt(it)                               # diminishing step
        a = a - lr * grad
        a = proj_simplex(a)
        cur_J = _objective(a, n, fft_len_self)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
        # jitter every few hundred iterations
        if it % 500 == 0 and rng.random() < 0.5:
            eps = 1e-3 * rng.normal(size=n)
            a = proj_simplex(a + eps)
    return best_a

def _hillclimb_optimize(p_start: np.ndarray,
                        n: int,
                        fft_len_self: int,
                        budget_steps: int,
                        rng: np.random.Generator) -> np.ndarray:
    """
    Very cheap local hill‑climbing: move a tiny mass ε from one coordinate
    to another; keep the move only if J improves.
    """
    a = p_start.copy()
    best_J = _objective(a, n, fft_len_self)
    best_a = a.copy()
    for _ in range(budget_steps):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = rng.uniform(0.0, 0.01)          # max 1 % of total mass
        eps = min(eps, a[i])                 # cannot go negative
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        cur_J = _objective(a_new, n, fft_len_self)
        if cur_J < best_J:
            best_J = cur_J
            best_a = a_new.copy()
            a = a_new
        else:
            # occasional random acceptance to escape shallow local minima
            if rng.random() < 0.001:
                a = a_new
    return best_a


# ----------------------------------------------------------------------
# Main optimisation routine (exposed)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Global sweep followed by three‑phase local
    optimisation (Adam → sub‑gradient → hill‑climb).  Returns the best sequence
    found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size (use previous length if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep the FFT size reasonable
            n = max(256, min(n, 8192))
        except Exception:
            n = 3809
    else:
        n = 3809

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Global cheap exploration
    # ------------------------------------------------------------------
    best_candidates: list[tuple[float, np.ndarray]] = []   # (J, p)

    def _push(p_vec: np.ndarray):
        """Evaluate a candidate and keep the current five best."""
        J = _objective(p_vec, n, fft_len_self)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > 8:
            best_candidates.pop()                         # keep only top‑8

    # a) uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push(uniform)

    # b) edge‑shape grid (linear taper)
    for frac in np.linspace(0.08, 0.42, 9):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            p = _edge_shape(n, L, gamma)
            _push(p)

    # c) edge‑shape with power‑law taper
    for frac in np.linspace(0.08, 0.42, 7):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for power in [0.3, 0.7, 1.0, 2.0, 4.0]:
            p = _edge_power_shape(n, L, power)
            _push(p)

    # d) edge‑shape with exponential taper
    for frac in np.linspace(0.08, 0.42, 7):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for lam in np.linspace(0.0, 3.0, 8):   # lam=0 → flat block
            p = _edge_exp_shape(n, L, lam)
            _push(p)

    # e) min‑power family (heavy edges, light centre)
    for s in np.linspace(0.2, 3.5, 12):
        p = _min_pow_shape(n, s)
        _push(p)

    # f) cosine family (known good baseline)
    for r in np.linspace(0.0, 1.0, 11):
        for pw in np.linspace(0.5, 3.0, 9):
            p = _cosine_shape(n, r, pw)
            _push(p)

    # g) random Dirichlet draws (moderately concentrated)
    for _ in range(40):
        alpha = rng.uniform(0.05, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # h) include previous best sequence if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = uniform
            else:
                prev = proj_simplex(prev / prev.sum())
            _push(prev)
        except Exception:
            pass

    # keep only the very best seeds
    best_candidates.sort(key=lambda x: x[0])
    top_seeds = best_candidates[:5]   # (J, p)

    print(f"[global] Best J after sweep: {top_seeds[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 3) Local refinement – three‑phase optimisation
    # ------------------------------------------------------------------
    # allocate roughly 85 % of remaining time to refinement
    remaining = deadline - time.monotonic()
    refine_budget = remaining * 0.85
    # split roughly equally among seeds
    per_seed_budget = refine_budget / max(1, len(top_seeds))

    best_J, best_p = top_seeds[0]  # initialise global best

    for seed_idx, (J0, p0) in enumerate(top_seeds):
        if time.monotonic() >= deadline:
            break
        print(f"[seed {seed_idx}] start J={J0:.6f}")

        # ----- Adam phase (≈60 % of per‑seed time) -----
        time_adam = per_seed_budget * 0.60
        p_adam = _adam_optimize(p0, n, fft_len_self, fft_len_grad,
                                min(time_adam, deadline - time.monotonic()), rng)

        cur_J = _objective(p_adam, n, fft_len_self)
        if cur_J < best_J:
            best_J, best_p = cur_J, p_adam.copy()
            print(f"[seed {seed_idx}] Adam improvement → J={best_J:.6f}")

        # ----- Sub‑gradient phase (≈30 % of per‑seed time) -----
        time_sub = per_seed_budget * 0.30
        p_sub = _subgrad_optimize(p_adam, n, fft_len_self, fft_len_grad,
                                  min(time_sub, deadline - time.monotonic()), rng)
        cur_J = _objective(p_sub, n, fft_len_self)
        if cur_J < best_J:
            best_J, best_p = cur_J, p_sub.copy()
            print(f"[seed {seed_idx}] Sub‑grad improvement → J={best_J:.6f}")

        # ----- Hill‑climb phase (remaining time) -----
        time_hc = per_seed_budget - (time_adam + time_sub)
        # allocate a fixed number of cheap steps (≈10 000 per second)
        steps = int(max(5000, min(200000, time_hc * 18000)))  # heuristic
        p_hc = _hillclimb_optimize(p_sub, n, fft_len_self, steps, rng)
        cur_J = _objective(p_hc, n, fft_len_self)
        if cur_J < best_J:
            best_J, best_p = cur_J, p_hc.copy()
            print(f"[seed {seed_idx}] Hill‑climb improvement → J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 4) Final random‑restart walk (use any leftover time)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline - 1.0:    # keep a second buffer
        # start from a fresh Dirichlet draw
        alpha = rng.uniform(0.02, 0.4)
        p0 = rng.dirichlet(np.full(n, alpha))
        # very short Adam run to get into a promising basin
        p_tmp = _adam_optimize(p0, n, fft_len_self, fft_len_grad,
                               min(8.0, deadline - time.monotonic()), rng)
        cur_J = _objective(p_tmp, n, fft_len_self)
        if cur_J < best_J:
            best_J, best_p = cur_J, p_tmp.copy()
            print(f"[restart] New best J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 5) Scale to the required L1‑norm and return
    # ------------------------------------------------------------------
    scale = math.sqrt(2.0 * n)                 # required L1‑norm
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # obey evaluate_sequence limits
    print(f"[final] Best J = {best_J:.6f}  (scaled L1 = {best_seq.sum():.4f})")
    return [float(x) for x in best_seq]
