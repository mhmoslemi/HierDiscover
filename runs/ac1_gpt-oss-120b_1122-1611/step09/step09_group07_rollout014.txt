def propose_candidate(seed=0,budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0,budget_s-5.0)
    
    # dimension
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n = int(base.shape[0])
        # clip negative, etc.
        n = max(256, min(4096, n))
    else:
        n = 2048
    # conv length, FFT length
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len-1).bit_length())
    
    def compute_J(vec):
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())
    
    best_J = float('inf')
    best_vec = None
    
    # baseline uniform
    a = np.full(n, 1.0/n)
    cur_J = compute_J(a)
    if cur_J < best_J:
        best_J, best_vec = cur_J, a.copy()
    
    # function to maybe run SA on given start
    def run_sa(start_vec, max_iters, start_T, end_T):
        nonlocal best_J, best_vec
        a = start_vec.copy()
        cur_J = compute_J(a)
        if cur_J < best_J:
            best_J, best_vec = cur_J, a.copy()
        for it in range(max_iters):
            now = time.monotonic()
            if now >= deadline:
                break
            # linear schedule
            frac = it / max_iters
            T = start_T * (1 - frac) + end_T * frac
            # propose move
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # delta proportion
            delta = min(a[i], rng.random() * 0.05)  # up to 5% of a[i]
            if delta <= 0:
                continue
            a[i] -= delta
            a[j] += delta
            # evaluate
            new_J = compute_J(a)
            if new_J < cur_J or rng.random() < np.exp(-(new_J - cur_J)/max(T,1e-12)):
                cur_J = new_J
                if cur_J < best_J:
                    best_J, best_vec = cur_J, a.copy()
            else:
                # revert
                a[i] += delta
                a[j] -= delta
    # schedule multiple restarts
    total_iters_per_run = 20000  # adjustable based on time
    while time.monotonic() < deadline - 2.0:

"""
Simple block‑search + stochastic refinement for the step‑function minimisation problem.

The routine first explores a low‑dimensional family of two‑block
(step) sequences (constant height on a prefix of the vector and another
height on the suffix).  A coarse grid over the block size and the
prefix height is scanned; the best candidate is retained.  Afterwards a
light simulated‑annealing hill‑climber (random mass transfers) is run
starting from the best block vector and, if available, from the
previously discovered global variable ``height_sequence_1``.  The whole
search respects the 1000 s budget and finally returns the best vector
found as a plain ``list[float]`` – all numbers are non‑negative and
automatically satisfy the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_max(vec: np.ndarray, fft_len: int) -> float:
    """Return max coefficient of the full self‑convolution of *vec*."""
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # length of full convolution is 2*len(vec)-1
    conv_len = 2 * vec.size - 1
    return float(conv[:conv_len].max())

def _objective(vec: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute the value that `evaluate_sequence` would return,
    assuming that ``vec`` already sums to one.
    (The denominator drops out – it is exactly ``2 * n * max(conv)``.)
    """
    return 2.0 * n * _conv_max(vec, fft_len)

def _two_block_candidate(n: int, k: int, a_val: float) -> np.ndarray:
    """
    Return a length‑`n` vector consisting of a constant prefix of length `k`
    with height ``a_val`` and a constant suffix with the height required
    to make the whole vector sum to one.
    """
    if k == 0:
        # only the suffix exists
        return np.full(n, 1.0 / n)
    # compute height of the suffix
    b_val = (1.0 - k * a_val) / (n - k)
    # numerical safety – clip tiny negatives to zero
    if b_val < 0.0:
        b_val = 0.0
    return np.concatenate((np.full(k, a_val), np.full(n - k, b_val)))

    Search for a non‑negative step function that minimises `evaluate_sequence`.

    The algorithm works in two stages:
    1️⃣ a cheap exhaustive search over a two‑block parametrisation;
    2️⃣ a stochastic refinement that repeatedly moves a small amount of mass
       between two random positions (simulated‑annealing style).

    The best sequence found within the time budget is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0 – problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(base.size)

            a_vals = np.linspace(0.0, a_max, num=30)
            for a in a_vals:
                cand = _two_block_candidate(n, k, a) if a > 0 else np.full(n, 1.0 / n)
                obj = evaluate(cand)
                if obj < best_obj:
                    best_obj, best_vec = obj, cand.copy()
                    print(f"[refine] k={k:4d} a={a:.6f} → new best J={best_obj:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # parameters for the annealing schedule
    start_T = 0.05
    end_T = 1e-6
    max_iters = 200000   # an upper bound – the loop will break earlier

    cur_vec = best_vec.copy()
    cur_obj = best_obj

        if delta <= 0.0:

        cur_vec[j] += delta

            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj, best_vec = new_obj, cur_vec.copy()
                print(f"[{time.monotonic():6.2f}s] SA new best J={best_obj:.6f}")
        else:
            # revert the move
            cur_vec[i] += delta
            cur_vec[j] -= delta

        # occasional “restart” to avoid getting stuck
        if it % 30000 == 0:
            # re‑initialise from the current best but add a tiny random jitter
            cur_vec = best_vec.copy()
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec *= jitter
            cur_vec /= cur_vec.sum()
            cur_obj = evaluate(cur_vec)

    # --------------------------------------------------------------
    # 6 – final safety clipping (the evaluator also clips, but we do it
    #     explicitly to stay inside the allowed range)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    print(f"Finished – best J = {best_obj:.6f} after {it} SA steps")
    return [float(x) for x in best_vec]

"""
Simple block‑search + stochastic refinement for the step‑function problem.

The routine first scans a cheap two‑block family (a constant prefix of
length *k* and a constant suffix) on a coarse grid.  The best
candidate is then refined by a simulated‑annealing style hill‑climber
that repeatedly moves a small amount of mass between two random
positions.  If a previous construction is available via the global
`height_sequence_1` variable it is also used as a starting point.
The whole search respects the 1000 s budget (a 5 s safety margin) and
returns the best vector found as a plain ``list[float]`` – all entries
are non‑negative and automatically satisfy the limits imposed by
``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_max(vec: np.ndarray, fft_len: int) -> float:
    """Maximum coefficient of the full self‑convolution of *vec*."""
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    conv_len = 2 * vec.size - 1
    return float(conv[:conv_len].max())


def _objective(vec: np.ndarray, n: int, fft_len: int) -> float:
    """
    Value that ``evaluate_sequence`` would return for *vec* (assuming
    ``sum(vec) == 1``).  The denominator cancels, leaving ``2·n·max``.
    """
    return 2.0 * n * _conv_max(vec, fft_len)


def _make_two_block(n: int, k: int, a_val: float):
    """
    Return a length‑*n* vector consisting of a constant prefix of length
    *k* with height ``a_val`` and a constant suffix whose height makes
    the whole vector sum to one.  If ``a_val`` is zero the prefix is
    empty (all zeros) and the suffix is uniform.  Returns ``None`` for
    infeasible (negative) suffix heights.
    """
    if k == 0:
        return np.full(n, 1.0 / n)

    if a_val <= 0.0:
        b_val = 1.0 / (n - k)
        return np.concatenate((np.zeros(k), np.full(n - k, b_val)))

    b_val = (1.0 - k * a_val) / (n - k)
    if b_val < 0.0:
        return None
    return np.concatenate((np.full(k, a_val), np.full(n - k, b_val)))


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.

    The algorithm is:
    1️⃣ exhaustive search over a two‑block parametric family;
    2️⃣ stochastic refinement (random mass transfers with a tiny
       simulated‑annealing schedule).
    The best sequence encountered before the time limit is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0 – problem dimension
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(base.size)
        except Exception:
            n = 2000
    else:
        n = 2000                     # default size – fast enough for the budget

    # keep n modest for speed (but not too small)
    n = max(512, min(4096, n))

    # ------------------------------------------------------------------
    # 1 – FFT preparation (same length for every evaluation)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # power‑of‑two length for speed

    # ------------------------------------------------------------------
    # Helper that evaluates a vector (assumes it already sums to 1)
    # ------------------------------------------------------------------
    def _eval(vec: np.ndarray) -> float:
        return _objective(vec, n, fft_len)

    # ------------------------------------------------------------------
    # 2 – baseline: uniform vector
    # ------------------------------------------------------------------
    best_vec = np.full(n, 1.0 / n)
    best_obj = _eval(best_vec)
    print(f"[init] uniform length={n} → J={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 3 – coarse grid over two‑block families
    # ------------------------------------------------------------------
    # we try about n // 100 different block sizes
    block_step = max(10, n // 100)
    ks = np.arange(block_step, n, block_step, dtype=int)

    for k in ks:
        a_max = 1.0 / k                     # highest feasible height on the prefix
        # 20 height samples (linear spacing – cheap and enough for a coarse scan)
        a_vals = np.linspace(0.0, a_max, num=20)
        for a in a_vals:
            cand = _make_two_block(n, k, a)
            if cand is None:
                continue
            obj = _eval(cand)
            if obj < best_obj:
                best_obj, best_vec = obj, cand.copy()
                print(f"[grid] k={k:4d} a={a:.6f} → new best J={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 4 – refined grid around the best block size (if any)
    # ------------------------------------------------------------------
    # estimate the block size of the current best vector (first index where
    # the value differs from the tail height)
    tail_height = best_vec[-1]
    diff = np.abs(best_vec - tail_height)
    k_est = int(np.argmax(diff > 1e-12)) + 1 if diff.max() > 1e-12 else 0

    if 0 < k_est < n:
        lo = max(block_step, k_est - block_step)
        hi = min(n - block_step, k_est + block_step)
        ks_fine = np.arange(lo, hi + 1, max(1, block_step // 5))
        for k in ks_fine:
            a_max = 1.0 / k
            a_vals = np.linspace(0.0, a_max, num=30)
            for a in a_vals:
                cand = _make_two_block(n, k, a)
                if cand is None:
                    continue
                obj = _eval(cand)
                if obj < best_obj:
                    best_obj, best_vec = obj, cand.copy()
                    print(f"[refine] k={k:4d} a={a:.6f} → new best J={best_obj:.6f}")

    # ------------------------------------------------------------------
    # 5 – stochastic refinement (simulated‑annealing style)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # SA parameters
    start_T = 0.05
    end_T = 1e-6
    max_sa_iters = 250_000     # upper bound; loop stops earlier if time runs out

    cur_vec = best_vec.copy()
    cur_obj = best_obj
    it = 0

    while time.monotonic() < deadline and it < max_sa_iters:
        it += 1
        frac = it / max_sa_iters
        T = start_T * (1.0 - frac) + end_T * frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a small fraction of the donor mass
        delta = min(cur_vec[i], rng.random() * 0.05)
        if delta <= 0.0:
            continue

        cur_vec[i] -= delta
        cur_vec[j] += delta
        new_obj = _eval(cur_vec)

        # acceptance rule (downhill always, uphill with Boltzmann prob.)
        if new_obj < cur_obj or rng.random() < np.exp(-(new_obj - cur_obj) / max(T, 1e-12)):
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj, best_vec = new_obj, cur_vec.copy()
                now = time.monotonic()
                print(f"[{now:.2f}s] SA new best J={best_obj:.6f}")
        else:
            # revert the move
            cur_vec[i] += delta
            cur_vec[j] -= delta

        # occasional restart from the current best with a tiny jitter
        if it % 30_000 == 0:
            cur_vec = best_vec.copy()
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec = cur_vec * jitter
            cur_vec /= cur_vec.sum()
            cur_obj = _eval(cur_vec)

    # ------------------------------------------------------------------
    # 6 – final clipping (evaluate_sequence applies the same limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    print(f"Finished – best J = {best_obj:.6f} after {it} SA steps")
    return [float(x) for x in best_vec]
