"""
Two‑phase optimiser for the step‑function attack.

Phase 1 – Very broad exploration:
    * Uniform baseline.
    * Exhaustive low‑dimensional families:
        – edge‑heavy blocks (size & taper),
        – symmetric cosine shapes,
        – symmetric exponential decay shapes.
    * Many random Dirichlet draws (various concentrations).
    * The best sequence from the previous run (height_sequence_1).

  All candidates are normalised to the probability simplex
  (∑aᵢ = 1).  The five (or ten) best candidates (lowest
  2·n·max(a*a)) are kept.

Phase 2 – Local refinement of each seed:
    * A projected Adam optimiser on a soft‑max surrogate of
      the max‑convolution.  Temperature grows from 10 to 15000
      (very close to the true max) while the learning‑rate decays.
    * When the allotted time for a seed is almost exhausted,
      the optimiser switches to a pure sub‑gradient descent
      on the exact max‑convolution (active‑index gradient).
    * Occasional jitter and a simple restart on stagnation
      keep the search from getting trapped.

The whole routine respects the supplied time‑budget (≤ budget_s‑5 s safety
margin).  At the end the best probability vector is scaled to
∑aᵢ = √(2 n) (the scaling does not change the objective) and clipped
to the limits enforced by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """Two edge blocks of length L with linear taper (γ∈[0,1])."""
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape (1+r·cosθ)^power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    v = np.clip(v, 0.0, None)
    return v / v.sum()


def _exp_decay_shape(n: int, lam: float) -> np.ndarray:
    """Symmetric exponential decay from both ends, normalised."""
    idx = np.arange(n, dtype=np.float64)
    v = np.exp(-lam * idx) + np.exp(-lam * (n - 1 - idx))
    v = np.clip(v, 0.0, None)
    return v / v.sum()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max weights (sum to 1)."""
    c_max = conv.max()
    shifted = conv - c_max          # improve numerical stability
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate w.r.t. a (length‑n vector)."""
    w_rev = w[::-1]                     # reversed soft‑max weights
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1          # = 2n‑2
    return 2.0 * full[start_idx:start_idx + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """Exact objective J = 2·n·max(p*p) for a probability vector p."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


def _max_conv(p: np.ndarray, fft_len_self: int) -> tuple[np.ndarray, float]:
    """Return (full convolution, maximal entry)."""
    conv = _self_conv(p, fft_len_self)
    return conv, float(conv.max())


def _subgradient(p: np.ndarray, conv: np.ndarray, active_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max (p∗p) at the current active indices.
    For each active convolution index k the derivative is
    2·p_{k‑i}.  The contributions from all active k are summed.
    """
    n = p.size
    grad = np.zeros_like(p)
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1)
        grad[i_range] += 2.0 * p[k - i_range]
    if active_idx.size > 0:
        grad /= float(active_idx.size)      # average to keep step size stable
    return grad


# ----------------------------------------------------------------------
# Optimisation sub‑routines
# ----------------------------------------------------------------------


def _adam_phase(p_start: np.ndarray,
                n: int,
                fft_len_self: int,
                fft_len_grad: int,
                time_limit: float,
                rng: np.random.Generator,
                lr_start: float = 1.0,
                lr_end: float = 0.001,
                alpha_start: float = 10.0,
                alpha_end: float = 15000.0,
                beta1: float = 0.9,
                beta2: float = 0.999,
                eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the soft‑max surrogate of max (p∗p).
    Returns (best_p, best_J) observed during the allotted time.
    """
    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_p = a.copy()
    best_J = _objective(a, fft_len_self, n)

    start = time.monotonic()
    it = 0
    last_improve = start

    while time.monotonic() - start < time_limit:
        it += 1
        now = time.monotonic()
        frac = (now - start) / time_limit          # ∈[0,1]

        # learning‑rate schedule (exponential decay)
        lr = lr_start * (0.5 ** frac) + lr_end * (1.0 - 0.5 ** frac)

        # temperature schedule (slow start → fast increase)
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # convolution & soft‑max weights
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)

        # gradient of the surrogate
        grad = _grad_softmax(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = proj_simplex(a)

        # evaluate true objective (no smoothing)
        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_p = a.copy()
            last_improve = now
            print(f"[{now - start:5.1f}s] Adam new best J = {best_J:.6f}")

        # jitter to escape plateaus (tiny log‑normal noise on a block)
        if rng.random() < 0.02:
            block_width = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - block_width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block_width] *= factor
            a = proj_simplex(a_pert)

        # restart after too long without improvement
        if now - last_improve > 15.0:
            a = best_p.copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[{now - start:5.1f}s] Adam restart (stagnation) – best J = {best_J:.6f}")

    return best_p, best_J


def _subgradient_phase(p_start: np.ndarray,
                       n: int,
                       fft_len_self: int,
                       time_limit: float,
                       rng: np.random.Generator,
                       step0: float = 0.5,
                       jitter_prob: float = 0.01) -> tuple[np.ndarray, float]:
    """
    Pure sub‑gradient descent on the exact max‑convolution.
    Returns (best_p, best_J) observed during the allotted time.
    """
    a = p_start.copy()
    best_p = a.copy()
    best_J = _objective(a, fft_len_self, n)

    start = time.monotonic()
    it = 0
    last_improve = start

    while time.monotonic() - start < time_limit:
        it += 1
        conv, M = _max_conv(a, fft_len_self)
        cur_J = 2.0 * n * M

        # active indices (within a tiny tolerance of the maximum)
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # compute sub‑gradient
        grad = _subgradient(a, conv, active)

        # step size decays like 1/√t
        step = step0 / np.sqrt(it)
        a = proj_simplex(a - step * grad)

        # occasional jitter
        if rng.random() < jitter_prob:
            block_width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - block_width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block_width] *= factor
            a = proj_simplex(a_pert)

        # update best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_p = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:5.1f}s] Sub‑grad new best J = {best_J:.6f}")

    return best_p, best_J


def _refine_candidate(p0: np.ndarray,
                      n: int,
                      fft_len_self: int,
                      fft_len_grad: int,
                      time_budget: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Hybrid refinement: 60 % Adam on the smoothed surrogate,
    40 % exact sub‑gradient descent.
    Returns (best_prob_vector, best_J).
    """
    # split the time budget
    adam_time = time_budget * 0.6
    subgrad_time = max(0.0, time_budget - adam_time)

    # Adam phase
    p_adam, J_adam = _adam_phase(p0, n, fft_len_self, fft_len_grad,
                                 adam_time, rng)

    # Sub‑gradient phase (starting from the best Adam result)
    p_final, J_final = _subgradient_phase(p_adam, n,
                                          fft_len_self,
                                          subgrad_time,
                                          rng)

    return p_final, J_final

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine consists of a cheap global sweep
    over several low‑dimensional families followed by a hybrid Adam + sub‑gradient
    local optimisation for the most promising seeds.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = int(len(globals()["height_sequence_1"]))
            n = min(max(n0, 256), 4096)   # keep FFT cheap, cap at 4096
        except Exception:
            n = 4096
    else:
        n = 4096

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Global cheap exploration (many low‑dimensional families)
    # --------------------------------------------------------------
    K_KEEP = 10                      # keep the best K_KEEP seeds
    best_candidates = []             # list of (J, p) tuples

    def _push(p_vec: np.ndarray):
        """Evaluate candidate and keep only the top K_KEEP."""
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > K_KEEP:
            best_candidates.pop()

    # a) uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push(uniform)

    # b) edge‑shape grid (fine sweep)
    for frac in np.linspace(0.05, 0.45, 21):   # block‑size fraction
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 21):
            p = _edge_shape(n, L, gamma)
            _push(p)

    # c) cosine‑shape grid (fine sweep)
    for r in np.linspace(0.0, 1.0, 21):
        for pw in np.linspace(0.2, 5.0, 25):
            p = _cosine_shape(n, r, pw)
            _push(p)

    # d) exponential‑decay family
    for lam in np.linspace(0.0, 5.0, 31):   # lam=0 yields uniform
        p = _exp_decay_shape(n, lam)
        _push(p)

    # e) a large batch of random Dirichlet draws
    for _ in range(200):
        # draw a concentration parameter that favours both dense and sparse vectors
        alpha = rng.uniform(0.001, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # f) include the previous best sequence if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = uniform
            else:
                prev = proj_simplex(prev / prev.sum())
            _push(prev)
        except Exception:
            pass

    # Keep the best candidate seen so far
    best_candidates.sort(key=lambda x: x[0])
    best_J, best_p = best_candidates[0]

    # --------------------------------------------------------------
    # 3) Local refinement – hybrid Adam + exact sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # time already exhausted – return whatever we have
        scale = np.sqrt(2.0 * n)
        seq = np.clip(best_p * scale, 0.0, 1000.0)
        return [float(x) for x in seq]

    # allocate ~80 % of the remaining time to refinement
    refine_budget = remaining * 0.80
    per_seed_budget = refine_budget / max(1, len(best_candidates))

    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break
        time_allowed = min(per_seed_budget, deadline - time.monotonic())
        if time_allowed <= 0.0:
            break
        p_refined, J_refined = _refine_candidate(p0, n,
                                                 fft_len_self,
                                                 fft_len_grad,
                                                 time_allowed,
                                                 rng)
        if J_refined < best_J - 1e-12:
            best_J, best_p = J_refined, p_refined.copy()
            print(f"[refine] New global best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence (scaled to satisfy the original norm)
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)               # required L1‑norm = sqrt(2 n)
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
