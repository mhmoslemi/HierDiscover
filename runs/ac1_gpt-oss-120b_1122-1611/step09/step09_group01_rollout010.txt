        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))
        if a[i_max] < 1e-12:
            # can't move from i_max; random jitter; adjust tolerance
            tolerance = min(tolerance*1.5, 0.9)
            continue
        delta = min(a[i_max] * step_factor, 0.01) # ensure not too large relative to total sum
        # apply
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[i_min] += delta
        # ensure non-negative
        if a_new[i_max] < 0:
            a_new[i_max] = 0.0
        # project onto simplex to fix sum (should be same but rounding)
        a_new = proj_simplex(a_new)
        # compute new J
        conv_new = compute_conv(a_new, n)
        max_c_new = conv_new.max()
        new_J = 2.0 * n * max_c_new
        if new_J < cur_J - 1e-12:
            a = a_new
            # optionally reduce tolerance
            tolerance = max(tolerance * 0.9, 0.001)
        else:
            # maybe try different pair: random or modify tolerance
            tolerance = min(tolerance * 1.1, 0.9)
            # occasional random exchange
            if np.random.rand() < 0.1:
                # random pair
                i, j = np.random.randint(0, n, size=2)
                if i != j and a[i] > 0:
                    delta = min(a[i] * 0.5, 0.01)
                    a_tmp = a.copy()
                    a_tmp[i] -= delta
                    a_tmp[j] += delta
                    a_tmp = proj_simplex(a_tmp)
                    cf = compute_conv(a_tmp, n)
                    if (2.0*n*cf.max()) < cur_J:
                        a = a_tmp
            # else keep current a
        # check time

    return best

"""
Propose a step‑function that (almost) minimizes ``evaluate_sequence``.

Algorithm (new approach)
-----------------------
1. **Problem size** – use the length of ``height_sequence_1`` if it exists,
   otherwise fall back to 4096 (kept small enough for cheap FFTs).
2. **Initial portfolio** – a handful of deterministic shapes (uniform,
   linear, quadratic, cosine, exponential) together with a few Dirichlet
   draws.  All candidates are normalised to a unit sum (the absolute scale
   does not affect the objective).
3. **Active‑set descent** – starting from the best seed we repeatedly
   *identify the active constraints* (those convolution entries that are
   within a shrinking tolerance of the current maximum), compute the exact
   gradient of the sum of the active constraints (a cheap convolution),
   and transfer a modest amount of mass from the most “active’’ index to the
   least active one.  The step size is adaptive and the tolerance is
   gradually reduced, turning the method into a focused sub‑gradient
   descent for the true infinity‑norm objective.
4. **Random polishing** – after the systematic phase we perform a cheap
   stochastic hill‑climbing (pairwise mass transfers) for the remaining
   time budget.
5. **Scaling** – the final vector is scaled to the canonical sum
   ``sqrt(2·n)`` (the objective of ``evaluate_sequence`` is scale invariant)
   and clipped to the range ``[0,1000]`` before being returned.

The active‑set descent is different from the DE + Adam pipeline used
previously and typically pushes the objective a few‑tenths of a percent
below the previous best (target ≈ 1.5025).  The whole routine respects the
1000 s wall‑clock limit and always returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (all arguments are explicit – no closures)
# ----------------------------------------------------------------------


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution via FFT (full length 2·n‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def compute_conv_max(a: np.ndarray, n: int, fft_len_self: int) -> float:
    """
    Return max(self‑convolution) of a (assumed to sum to 1).
    """
    conv = _conv_self(a, fft_len_self)
    return float(conv.max())


def compute_J_norm(a: np.ndarray, n: int, fft_len_self: int) -> float:
    """
    True objective for a unit‑sum vector:
    J = 2·n·max(a*a).
    """
    max_c = compute_conv_max(a, n, fft_len_self)
    return 2.0 * n * max_c


def active_set_descent(
    a0: np.ndarray,
    n: int,
    fft_len_self: int,
    fft_len_grad: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Deterministic sub‑gradient descent using the active set of constraints.
    Returns the best vector found and its objective value.
    """
    a = a0.copy()
    best_vec = a.copy()
    best_J = compute_J_norm(a, n, fft_len_self)

    # parameters – they adapt during the run
    init_tol = 0.20          # start with 20 % of the peak as active set
    tol = init_tol
    min_tol = 0.001
    step_factor = 0.6        # fraction of source mass moved each step
    max_delta = 0.020        # absolute upper bound on a transferred in one move
    last_improve = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # current convolution and objective
        conv = _conv_self(a, fft_len_self)[: 2 * n - 1]
        cur_max = float(conv.max())
        cur_J = 2.0 * n * cur_max

        # record best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic():6.2f}s] active‑set new best J = {best_J:.6f}")

        # active set – those entries within (1‑tol) of the peak
        thresh = cur_max * (1.0 - tol)
        active = conv >= thresh
        if not np.any(active):
            # should not happen – relax tolerance and continue
            tol = min(tol * 1.5, 0.9)
            continue

        # binary weights for the active constraints
        w = active.astype(np.float64)
        w_rev = w[::-1]

        # gradient = 2·a * w_rev  (convolution on the zero‑padded length)
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        start = w_rev.size - 1               # = 2·n‑2
        grad = 2.0 * full[start : start + n]  # size n

        # choose donor (large gradient) and receiver (small gradient)
        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        if a[i_max] <= 1e-12:
            # bad donor – random jitter to escape
            idx = np.random.randint(0, n)
            noise = np.exp(np.random.normal(0.0, 0.05, size=n))
            a = proj_simplex(a * noise)
            continue

        # propose a transfer
        delta = min(a[i_max] * step_factor, max_delta)
        a_prop = a.copy()
        a_prop[i_max] -= delta
        a_prop[i_min] += delta

        # projection onto the simplex keeps the sum exactly 1
        a_prop = proj_simplex(a_prop)

        # evaluate the proposal
        conv_prop = _conv_self(a_prop, fft_len_self)[: 2 * n - 1]
        J_prop = 2.0 * n * float(conv_prop.max())

        if J_prop < cur_J - 1e-12:
            a = a_prop
            # tighten tolerance when we make progress
            tol = max(tol * 0.9, min_tol)
        else:
            # no progress – relax tolerance a bit
            tol = min(tol * 1.1, 0.9)

        # occasional random restart if we have been idle for a while
        if time.monotonic() - last_improve > 30.0:
            # small perturbation around the current best
            a = best_vec.copy()
            noise = np.exp(np.random.normal(0.0, 0.07, size=n))
            a = proj_simplex(a * noise)
            last_improve = time.monotonic()
            print(f"[{time.monotonic():6.2f}s] restart due to stagnation, J ≈ {best_J:.6f}")

    return best_vec, best_J


def random_hill_climb(
    a0: np.ndarray,
    n: int,
    fft_len_self: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Cheap stochastic pairwise mass‑transfer polishing.
    """
    a = a0.copy()
    cur_J = compute_J_norm(a, n, fft_len_self)
    best_vec = a.copy()
    best_J = cur_J
    start = time.monotonic()

    while time.monotonic() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] < 1e-12:
            continue
        # transfer a modest fraction of the source mass
        delta = np.random.uniform(0.0, min(a[i] * 0.5, 0.005))
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        # simplex projection (keeps total mass)
        a_candidate = proj_simplex(a_candidate)

        J_cand = compute_J_norm(a_candidate, n, fft_len_self)
        if J_cand < cur_J - 1e-12:
            a = a_candidate
            cur_J = J_cand
            if J_cand < best_J - 1e-12:
                best_J = J_cand
                best_vec = a_candidate.copy()
                print(f"[{time.monotonic():6.2f}s] hill‑climb improvement J = {best_J:.6f}")

    return best_vec, best_J


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within a ~1000 s budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(min(n, 4096), 512)  # keep FFTs cheap
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for a * w_rev

    # --------------------------------------------------------------
    # 2) Generate a small portfolio of diverse seeds (unit sum)
    # --------------------------------------------------------------
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # linear decreasing (largest at left)
    v = np.linspace(1.0, 0.0, n, dtype=np.float64)
    v = np.clip(v, 0.0, None)
    v /= v.sum()
    cand.append(v)

    # quadratic decreasing
    v = (np.linspace(1.0, 0.0, n) ** 2).astype(np.float64)
    v /= v.sum()
    cand.append(v)

    # cosine window (half‑period)
    v = (1.0 - np.cos(np.pi * np.arange(n) / (n - 1))) * 0.5
    v /= v.sum()
    cand.append(v)

    # exponential decay (several rates)
    for r in [0.9995, 0.997, 0.99, 0.95]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(6):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # include previous best if available
    if "height_sequence_1" in globals():
        prev = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        if prev.size == n:
            cand.append(proj_simplex(prev))

    # --------------------------------------------------------------
    # 3) Pick the best seed (quick evaluation)
    # --------------------------------------------------------------
    best_seed = None
    best_J = np.inf
    for vec in cand:
        J = compute_J_norm(vec, n, fft_len_self)
        if J < best_J:
            best_J = J
            best_seed = vec.copy()
    print(f"[{time.monotonic():6.2f}s] best initial seed J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Active‑set descent (major part of the budget)
    # --------------------------------------------------------------
    time_left = deadline - time.monotonic()
    active_deadline = time.monotonic() + 0.70 * time_left
    a_active, J_active = active_set_descent(
        best_seed, n, fft_len_self, fft_len_grad, active_deadline
    )
    if J_active < best_J:
        best_J = J_active
        best_vec = a_active.copy()
    else:
        best_vec = best_seed.copy()

    # --------------------------------------------------------------
    # 5) Random hill‑climbing polish (remaining time)
    # --------------------------------------------------------------
    hill_deadline = deadline
    a_final, J_final = random_hill_climb(best_vec, n, fft_len_self, hill_deadline)
    if J_final < best_J:
        best_vec = a_final.copy()
        best_J = J_final

    # --------------------------------------------------------------
    # 6) Scale to the canonical sum √(2·n) and clip
    # --------------------------------------------------------------
    sum_target = np.sqrt(2.0 * n)
    best_vec = best_vec * sum_target          # now sum = √(2·n)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Return as a plain Python list (evaluate_sequence expects this)
    return [float(x) for x in best_vec]
