def grad_softmax(a, w, fft_len_grad): # from code above

    w_rev = w[::-1]

    fa = np.fft.rfft(a, n=fft_len_grad) // compute a's rfft

    fw = np.fft.rfft(w_rev, n=fft_len_grad)

    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start_idx = w_rev.size - 1

    return 2.0 * full[start_idx:start_idx + a.size]

def run_exponential_gradient(p_start, fft_len_self, fft_len_grad, n, time_budget, rng):
    # p_start is probability vector (sum=1)
    start = time.monotonic()
    end = start + time_budget
    p = p_start.copy()
    best_p = p.copy()
    best_J = _objective(p, fft_len_self, n)
    last_improve = start

    # hyperparameters
    lr_start = 0.5
    lr_end = 0.001
    alpha_start = 1.0
    alpha_end = 4000.0

    while time.monotonic() < end:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / time_budget
        # schedule
        lr = lr_start * (1 - frac) + lr_end * frac
        alpha = alpha_start * (1 - frac) + alpha_end * frac

        conv = _self_conv(p, fft_len_self)
        w = _softmax_weights(conv, alpha)
        grad = _grad_softmax(p, w, fft_len_grad)

        # EG update
        p = p * np.exp(-lr * grad)
        # ensure positivity
        p = np.clip(p, 1e-15, None)
        p = p / p.sum()

        if rng.random() < 0.01:
            # jitter: randomly scale small subarray
            block_width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - block_width + 1)
            factor = np.exp(rng.normal(0.0, 0.06))
            p_pert = p.copy()
            p_pert[start_idx:start_idx+block_width] *= factor
            p_pert = np.clip(p_pert, 1e-15, None)
            p_pert = p_pert / p_pert.sum()
            p = p_pert

        # evaluate true objective
        cur_J = _objective(p, fft_len_self, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_p = p.copy()
            last_improve = now
            print(f"[EG] t={now-start:.1f}s best J={best_J:.6f}")

        if now - last_improve > 30.0:
            # stagnation: restart from best with noise
            p = best_p.copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            p = p * noise
            p = np.clip(p, 1e-15, None)
            p = p / p.sum()
            last_improve = now
            print(f"[EG] restart after stagnation at t={now-start:.1f}s J={best_J:.6f}")

    return best_p, best_J

def run_adam(p_start, fft_len_self, fft_len_grad, n, time_budget, rng):

    start= time.monotonic()
    end= start+time_budget

    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_J = _objective(a, fft_len_self, n)
    best_p = a.copy()
    last_improve = start
    it = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # schedule params
    lr_start = 0.25
    lr_end = 0.001
    alpha_start = 100.0
    alpha_end = 5000.0

    while time.monotonic() < end:
        it += 1
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / time_budget

        lr = lr_start * (1 - frac) + lr_end * frac
        alpha = alpha_start * (1 - frac) + alpha_end * frac

        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)
        grad = _grad_softmax(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)

"""
Propose a non‑negative step function that (approximately) minimizes
evaluate_sequence.  The algorithm consists of:

* A cheap global sweep over several low‑dimensional families
  (uniform, edge‑heavy, cosine, triangular, random Dirichlet) to obtain
  a handful of promising seeds.
* For each seed run a projected Exponential‑Gradient descent on a
  temperature‑smoothed max‑convolution surrogate (soft‑max).  The
  temperature is raised linearly from a small value to a large one while
  the learning‑rate decays.  Stagnation triggers a restart with tiny
  noise.
* The best seed is finally polished by a projected Adam optimiser
  (identical to the one used in the reference solution) for the
  remaining time budget.
* The final probability vector is scaled to satisfy the original
  normalization (L1‑norm = sqrt(2 n)) and clipped to the limits enforced
  by ``evaluate_sequence``.

The whole search respects a global time budget (default 1000 s) and
guarantees that a sequence is always returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape: (1 + r*cos(theta))**power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L with linear taper controlled by gamma.
    When gamma==0 the blocks are uniform; gamma==1 makes them drop to
    zero at the inner border.
    """
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _triangular_shape(n: int, width_frac: float) -> np.ndarray:
    """Symmetric triangular bump of given relative width."""
    if width_frac <= 0 or width_frac > 1:
        return np.full(n, 1.0 / n)
    w = int(width_frac * n)
    if w < 2:
        return np.full(n, 1.0 / n)
    half = w // 2
    v = np.zeros(n, dtype=np.float64)
    # rising edge
    if half > 0:
        v[:half] = np.linspace(0, 1, half, endpoint=False)
    # falling edge
    rest = w - half
    if rest > 0:
        v[half:w] = np.linspace(1, 0, rest, endpoint=False)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max weights (sum to 1)."""
    # shift by the maximum for numerical stability (all values ≤ 0)
    shifted = conv - conv.max()
    # when alpha is zero we fall back to uniform weights
    if alpha == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    exps = np.exp(alpha * shifted)
    s = exps.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / s


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate w.r.t. a (length‑n vector)."""
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1
    return 2.0 * full[start_idx:start_idx + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """
    Returns the true objective J = 2·n·max(p*p) for a probability vector p.
    """
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())

# Optimisation primitives
# ----------------------------------------------------------------------


def run_exponential_gradient(p_start: np.ndarray,
                             fft_len_self: int,
                             fft_len_grad: int,
                             n: int,
                             time_budget: float,
                             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Exponential‑Gradient descent on a soft‑max surrogate of the
    max‑convolution.  Returns the best probability vector and its objective.
    """
    start = time.monotonic()
    end = start + time_budget

    p = p_start.copy()
    best_p = p.copy()
    best_J = _objective(p, fft_len_self, n)
    last_improve = start

    # hyper‑parameters (linear schedules)
    lr_start, lr_end = 0.5, 0.001
    alpha_start, alpha_end = 1.0, 4000.0

    while time.monotonic() < end:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / time_budget

        # schedules
        lr = lr_start * (1.0 - frac) + lr_end * frac
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        conv = _self_conv(p, fft_len_self)
        w = _softmax_weights(conv, alpha)
        grad = _grad_softmax(p, w, fft_len_grad)

        # exponential‑gradient step (KL‑projection = renormalisation)
        p = p * np.exp(-lr * grad)
        p = np.clip(p, 1e-15, None)
        p = p / p.sum()

        # occasional jitter to escape shallow plateaus
        if rng.random() < 0.01:
            block_width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - block_width + 1)
            factor = np.exp(rng.normal(0.0, 0.06))
            p_pert = p.copy()
            p_pert[start_idx:start_idx + block_width] *= factor
            p_pert = np.clip(p_pert, 1e-15, None)
            p_pert = p_pert / p_pert.sum()
            p = p_pert

        cur_J = _objective(p, fft_len_self, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_p = p.copy()
            last_improve = now
            print(f"[EG] t={now - start:.1f}s  best J={best_J:.6f}")

        # restart if stagnated for too long
        if now - last_improve > 30.0:
            p = best_p.copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            p = p * noise
            p = np.clip(p, 1e-15, None)
            p = p / p.sum()
            last_improve = now
            print(f"[EG] restart after stagnation at t={now - start:.1f}s  J={best_J:.6f}")

    return best_p, best_J


def run_adam(p_start: np.ndarray,
             fft_len_self: int,
             fft_len_grad: int,
             n: int,
             time_budget: float,
             rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the same soft‑max surrogate.
    Returns the best probability vector and its objective.
    """
    start = time.monotonic()
    end = start + time_budget

    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # schedules
    lr_start, lr_end = 0.25, 0.001
    alpha_start, alpha_end = 100.0, 5000.0

    best_p = a.copy()
    best_J = _objective(a, fft_len_self, n)
    last_improve = start

    while time.monotonic() < end:
        it += 1
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / time_budget

        lr = lr_start * (1.0 - frac) + lr_end * frac
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)
        grad = _grad_softmax(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # occasional jitter
        if rng.random() < 0.02:
            block_width = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - block_width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block_width] *= factor
            a_pert = proj_simplex(a_pert)
            a = a_pert

        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_p = a.copy()
            last_improve = now
            print(f"[Adam] t={now - start:.1f}s  best J={best_J:.6f}")

        if now - last_improve > 25.0:
            a = best_p.copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[Adam] restart after stagnation at t={now - start:.1f}s  J={best_J:.6f}")

    return best_p, best_J

        if time_limit <= 0:
            break
        print(f"EG refinement seed {idx + 1}/{len(best_candidates)} – budget {time_limit:.1f}s")
        p_opt, J_opt = run_exponential_gradient(seed_vec,
                                                 fft_len_self,
                                                 fft_len_grad,
                                                 n,
                                                 time_limit,
                                                 rng)
        if J_opt < best_J_global - 1e-12:
            best_J_global = J_opt
            best_p_global = p_opt.copy()
            print(f"New global best after EG: J = {best_J_global:.6f}")

    # --------------------------------------------------------------
    # 2) Final polishing – projected Adam
    # --------------------------------------------------------------
    remaining2 = deadline - time.monotonic()
    if remaining2 > 1.0:
        print(f"Adam polishing – remaining budget {remaining2:.1f}s")
        p_opt, J_opt = run_adam(best_p_global,
                                fft_len_self,
                                fft_len_grad,
                                n,
                                remaining2,
                                rng)
        if J_opt < best_J_global - 1e-12:
            best_J_global = J_opt
            best_p_global = p_opt.copy()
            print(f"New global best after Adam: J = {best_J_global:.6f}")

    # --------------------------------------------------------------
    # 3) Rescale to the original evaluation norm and return
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)          # required L1‑norm for evaluate_sequence
    best_seq = best_p_global * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
