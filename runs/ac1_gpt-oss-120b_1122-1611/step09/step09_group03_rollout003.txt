def propose_candidate(seed=0,budget_s=1000):
    import time, numpy as np
    rng = np.random.default_rng(seed)

    # get initial a
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0, None)
        # trim leading/trailing zeros
        nonzero = np.where(a > 1e-14)[0]
        if nonzero.size:
            a = a[nonzero[0]:nonzero[-1]+1]
    else:
        n = 2000
        a = rng.dirichlet(np.full(n, 0.5))
    # normalize to sum 1
    if a.sum() == 0:
        a = np.full_like(a, 1.0/a.size)
    a = a / a.sum()

    # compute J
    def compute_J(vec):
        n = vec.size
        L = 1 << ((2*n-2).bit_length())
        fa = np.fft.rfft(vec, n=L)
        conv = np.fft.irfft(fa*fa, n=L)[:2*n-1]
        maxc = conv.max()
        return 2.0 * n * maxc

    best_a = a.copy()
    best_J = compute_J(best_a)
    # time management
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    iter_count = 0
    # parameters
    epsilon_factor = 0.2   # fraction of donor mass to move
    epsilon_min = 1e-6
    epsilon_max = 0.5
    temperature = 0.01
    temp_decay = (budget_s - 5.0) / (budget_s - 5.0) # placeholder; maybe linear to zero
    # We'll update temperature linearly with time left.

    while time.monotonic() < deadline:
        iter_count += 1
        n = a.size
        # choose donor index biased towards larger mass
        donor = rng.choice(n, p=a)  # probability proportional to mass
        # choose recipient uniformly from others
        recipient = rng.integers(0, n-1)
        if recipient >= donor:
            recipient += 1
        # compute epsilon
        max_epsilon = a[donor] * epsilon_factor
        if max_epsilon < epsilon_min:
            # cannot move from this donor, skip
            continue
        # random epsilon up to max_epsilon
        eps = rng.uniform(epsilon_min, max_epsilon)
        # create candidate
        a_cand = a.copy()
        a_cand[donor] -= eps
        a_cand[recipient] += eps
        # ensure nonnegative (should be)
        # compute J
        J_cand = compute_J(a_cand)
        # acceptance
        if J_cand < best_J - 1e-12:
            best_J = J_cand
            best_a = a_cand.copy()
            a = a_cand
            # possibly increase epsilon_factor
            epsilon_factor = min(epsilon_factor * 1.05, epsilon_max)
            # print progress
            elapsed = time.monotonic() - (deadline - budget_s + 5.0)
            print(f"[{elapsed:.2f}s] iter {iter_count} new best J = {best_J:.6f}")
        else:
            # maybe accept with probability

            elapsed_frac = (time.monotonic() - (deadline - budget_s + 5.0)) / (budget_s - 5.0)
            temperature = max(1e-6, 0.02 * (1 - elapsed_frac))
            if rng.random() < np.exp(-(J_cand - best_J) / temperature):
                a = a_cand
                # small degrade maybe adjust epsilon
            else:
                # reject, maybe reduce epsilon_factor
                epsilon_factor = max(epsilon_factor * 0.98, epsilon_min)

        # periodic compress leading/trailing near-zero entries
        if iter_count % 2000 == 0:
            # trim only if there are leading/trailing values below threshold
            small_thr = 1e-12
            nz = np.where(best_a > small_thr)[0]
            if nz.size and (nz[0] > 0 or nz[-1] < best_a.size - 1):
                best_a = best_a[nz[0]: nz[-1]+1]
                # renormalize sum to 1
                best_a = best_a / best_a.sum()
                a = best_a.copy()
                best_J = compute_J(best_a)
                print(f"[compress] new length {best_a.size}, J = {best_J:.6f}")
    # after loop, final best_a may have leading zeros trimmed
    # ensure normalized sum = 1
    best_a = best_a / best_a.sum()
    # optionally round small entries to 0 to avoid noise
    small_thr = 1e-12

    # final trim again
    nz = np.where(best_a > 0)[0]
    if nz.size:
        best_a = best_a[nz[0]: nz[-1]+1]
    # return list
    return [float(x) for x in best_a]

"""
Propose a step‑function (list of non‑negative heights) that improves the bound
given by ``evaluate_sequence``.

Algorithm
---------
* Initialise from the previously‑found construction ``height_sequence_1``
  (if it exists) or from a Dirichlet draw; afterwards normalise to sum=1.
* Repeatedly:
    – Compute the self‑convolution with an FFT (cheap O(n log n)).
    – Locate the index k where the convolution attains its maximum.
    – Build a “contribution” vector w_i = a_i·a_{k‑i} (zero when the index is
      out of range).  This tells how much each entry participates in the
      offending peak.
    – Choose a donor index i with probability proportional to w_i
      (i.e. an entry that strongly contributes to the peak) and a recipient
      index j that contributes little.
    – Transfer a small fraction ε of mass from i to j, staying on the simplex.
    – Accept the move if the true objective (2·n·max(conv)) improves; otherwise
      keep the old vector.  The step size ε adapts automatically.
* If no improvement is seen for a while a random “block jitter’’ (log‑normal
  multiplicative noise on a random contiguous block) is applied to escape
  plateaus.
* Periodically trim leading/trailing zeroes – this shortens the vector and
  reduces the factor ``2·n`` in the objective without affecting the peak,
  giving an immediate improvement.
* The best vector encountered before the time budget expires is returned
  (converted to a plain Python list of floats).

The routine respects the 1000 s budget, works on a single thread and uses only
numpy (no external LP solver).  It is deliberately different from the
projected‑gradient‑soft‑max method that was previously employed.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def compute_J(vec: np.ndarray) -> float:
    """
    Fast evaluation of the objective 2·n·max(conv(vec,vec)) assuming
    ``vec`` is already normalised to sum = 1.
    """
    n = vec.size
    fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine follows the description in the module
    docstring.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (keep a 5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initialise candidate
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            a0 = np.clip(a0, 0.0, None)
            # trim obvious zero padding at the ends
            nz = np.where(a0 > 1e-15)[0]
            if nz.size:
                a0 = a0[nz[0] : nz[-1] + 1]
            else:
                a0 = np.zeros(1, dtype=np.float64)
        except Exception:
            a0 = None
    else:
        a0 = None

    if a0 is None or a0.size == 0:
        # fall back to a moderate‑size Dirichlet draw
        n0 = 2000
        a0 = rng.dirichlet(np.full(n0, 0.5))

    # normalise to sum = 1 (the objective is scale‑invariant)
    if a0.sum() == 0.0:
        a0 = np.full_like(a0, 1.0 / a0.size)
    else:
        a0 = a0 / a0.sum()

    a = a0.copy()
    n = a.size
    best_a = a.copy()
    best_J = compute_J(best_a)

    print(f"[init] length={n}, J={best_J:.6f}")

    # --------------------------------------------------------------
    # 2) Parameters for the local search
    # --------------------------------------------------------------
    epsilon_factor = 0.20          # fraction of donor mass that may be moved
    epsilon_min = 1e-9
    epsilon_max = 0.5
    stagnation_limit = 30.0       # seconds without improvement → jitter
    last_improve = time.monotonic()
    iter_cnt = 0

    # --------------------------------------------------------------
    # 3) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iter_cnt += 1

        # ----- 3.1 compute convolution and locate the peak -----
        fft_len = _next_pow2(2 * n - 1)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
        max_c = conv.max()
        k_max = int(conv.argmax())

        # ----- 3.2 build contribution vector (how much each entry feeds the peak) -----
        contrib = np.zeros_like(a)
        start = max(0, k_max - (n - 1))
        end = min(k_max, n - 1)
        if start <= end:
            idx = np.arange(start, end + 1, dtype=np.int64)
            rev = k_max - idx
            contrib[idx] = a[idx] * a[rev]

        # ----- 3.3 choose donor (high contribution) and recipient (low contribution) -----
        if contrib.sum() > 0 and rng.random() < 0.85:
            # biased donor
            donor = rng.choice(idx, p=contrib[idx] / contrib[idx].sum())
        else:
            donor = rng.choice(n, p=a)                     # otherwise proportional to mass

        # pick a recipient that does NOT contribute much to the current peak
        low_mask = (contrib < (contrib.max() * 0.1))
        if low_mask.any():
            recipient = rng.choice(np.flatnonzero(low_mask))
            # avoid picking the same index
            if recipient == donor:
                recipient = (recipient + 1) % n
        else:
            # fall back to a random different index
            recipient = rng.integers(0, n - 1)
            if recipient >= donor:
                recipient += 1

        # ----- 3.4 propose a move (transfer a little mass) -----
        max_move = a[donor] * epsilon_factor
        if max_move < epsilon_min:
            # donor too small – skip this iteration
            continue
        epsilon = rng.uniform(epsilon_min, max_move)
        a_cand = a.copy()
        a_cand[donor] -= epsilon
        a_cand[recipient] += epsilon

        # ----- 3.5 evaluate the candidate -----
        cand_J = compute_J(a_cand)

        # ----- 3.6 accept if improvement -----
        if cand_J < best_J - 1e-12:
            a = a_cand
            best_a = a_cand.copy()
            best_J = cand_J
            last_improve = time.monotonic()
            # slightly enlarge step size after a success
            epsilon_factor = min(epsilon_factor * 1.03, epsilon_max)
            elapsed = time.monotonic() - (deadline - budget_s + 5.0)
            print(f"[{elapsed:6.2f}s] iter {iter_cnt:5d}  new best J = {best_J:.6f}")
        else:
            # no improvement – shrink the aggressiveness a bit
            epsilon_factor = max(epsilon_factor * 0.97, epsilon_min)

        # ----- 3.7 occasional jitter to escape plateaus -----
        now = time.monotonic()
        if now - last_improve > stagnation_limit:
            a = _random_block_perturb(best_a, rng)
            n = a.size
            epsilon_factor = max(epsilon_factor, 0.05)
            last_improve = now
            elapsed = now - (deadline - budget_s + 5.0)
            print(f"[{elapsed:6.2f}s] jitter (stagnation) – J still {best_J:.6f}")

        # ----- 3.8 occasional trimming of leading/trailing near‑zero entries -----
        if iter_cnt % 4000 == 0:
            tiny = 1e-12
            nz = np.where(best_a > tiny)[0]
            if nz.size and (nz[0] > 0 or nz[-1] < best_a.size - 1):
                best_a = best_a[nz[0] : nz[-1] + 1]
                best_a /= best_a.sum()
                a = best_a.copy()
                n = a.size
                best_J = compute_J(best_a)
                elapsed = time.monotonic() - (deadline - budget_s + 5.0)
                print(f"[{elapsed:6.2f}s] trim – new length {n}, J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Final clean‑up and return
    # --------------------------------------------------------------
    # final tiny‑value cleanup
    best_a[best_a < 1e-14] = 0.0
    nz = np.where(best_a > 0)[0]
    if nz.size:
        best_a = best_a[nz[0] : nz[-1] + 1]

    # normalise once more (just in case rounding introduced a tiny drift)
    if best_a.sum() == 0.0:
        # fallback to a uniform tiny vector
        best_a = np.full(1, 1.0)
    else:
        best_a = best_a / best_a.sum()

    return [float(x) for x in best_a]
