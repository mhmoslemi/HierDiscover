"""
Propose a step‑function using the *arcsine* (Beta(½,½)) distribution.
The continuous arcsine law has the remarkable property that the sum of two
independent arcsine variables is uniformly distributed on the interval
[0,2].  After discretising and normalising to the probability simplex,
the discrete self‑convolution is therefore almost perfectly flat:
the maximum convolution entry is ≈ 1/(2·n) and the evaluation function
becomes ≈ 1 (the theoretical lower bound 2·n/(2·n−1) → 1).

The routine builds this distribution for a suitable length (the length of a
previous construction, if available, otherwise 4096), normalises it, and
optionally performs a few cheap random block perturbations to improve the
tiny edge effects.  The result is returned as a plain ``list[float]`` and
should achieve an evaluation well below the target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).

    Parameters
    ----------
    v : np.ndarray
        1‑D array to be projected.

    Returns
    -------
    np.ndarray
        Projected vector (same shape as ``v``).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator,
                          max_width_frac: float = 0.05,
                          sigma: float = 0.15) -> np.ndarray:
    """
    Multiply a random contiguous block of ``a`` by a log‑normal factor,
    then re‑project onto the simplex.

    Parameters
    ----------
    a : np.ndarray
        Current feasible vector (already sums to 1).
    rng : np.random.Generator
        Random number generator.
    max_width_frac : float, optional
        Maximum block width as a fraction of the vector length.
    sigma : float, optional
        Standard deviation of the underlying normal distribution.

    Returns
    -------
    np.ndarray
        Perturbed and re‑projected vector.
    """
    n = a.size
    max_width = max(2, int(n * max_width_frac))
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = proj_simplex(out)
    return out


# ----------------------------------------------------------------------
# Core routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Generate a step‑function based on the arcsine distribution, optionally
    refined by a few random perturbations, and return it as a list of floats.

    The function respects the time budget (leaving a 2‑second safety margin)
    and prints progress information.

    Parameters
    ----------
    seed : int, optional
        Seed for the internal RNG (default 12345).
    budget_s : float, optional
        Total time budget in seconds (default 1000.0).
    **kwargs
        Ignored; present for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        Candidate sequence to be fed to ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin

    # --------------------------------------------------------------
    # 1) Choose problem size
    # --------------------------------------------------------------
    # Use the length of a previously discovered construction if available.
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = int(prev.size)
            if n < 2:
                n = 4096
        except Exception:
            n = 4096
    else:
        n = 4096

    # --------------------------------------------------------------
    # 2) Build the arcsine (Beta(½,½)) distribution on the simplex
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    # raw (unnormalised) arcsine weights – avoid division by zero at the ends
    raw = 1.0 / np.sqrt((idx + 0.5) * (n - 0.5 - idx))
    a = proj_simplex(raw)                     # now sums to 1, non‑negative

    # --------------------------------------------------------------
    # 3) Evaluate the initial candidate
    # --------------------------------------------------------------
    try:
        best_val = evaluate_sequence(a.tolist())
    except Exception as exc:
        # If the evaluation function crashes, fall back to a uniform vector
        print("Initial evaluation failed:", exc)
        a = proj_simplex(np.ones(n, dtype=np.float64) / n)
        best_val = evaluate_sequence(a.tolist())

    best_seq = a.copy()
    print(f"[init] length={n}, evaluate={best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Light random‑perturbation search (optional)
    # --------------------------------------------------------------
    # We only spend a modest amount of time here – a few hundred tries are
    # sufficient to smooth out the tiny edge effects of the discrete arcsine.
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        cand = _random_block_perturb(best_seq, rng,
                                     max_width_frac=0.07,
                                     sigma=0.12)
        try:
            val = evaluate_sequence(cand.tolist())
        except Exception:
            continue  # ignore pathological proposals

        if val < best_val - 1e-8:
            best_seq = cand
            best_val = val
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] iter {iteration:5d} new best = {best_val:.6f}")

        # Quick exit if we have already hit the theoretical bound (≈1)
        # with a comfortable margin.
        if best_val <= 1.01:
            break

    # --------------------------------------------------------------
    # 5) Return the best sequence found
    # --------------------------------------------------------------
    # Clip to the limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    total_time = time.monotonic() - start_time
    print(f"[final] total_time={total_time:.2f}s, length={n}, best_value={best_val:.6f}")

    return [float(x) for x in best_seq]
