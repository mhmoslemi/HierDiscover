def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s-5.0)
    # Get initial vector
    init = _initial_candidate(rng)
    a = init.copy()
    n = a.size
    # FFT parameters
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    # compute objective
    def obj(vec):
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        return float(2.0*n*conv.max())
    best_a = a.copy()
    best_val = obj(a)
    print(f"[init] objective = {best_val:.6f}")
    # parameters
    # SA parameters
    start_temp = 0.02
    end_temp = 1e-5
    total_iters_est = 200000 # approximate, we will check time
    # step sizes
    sigma_factor_init = 0.4
    sigma_factor_end = 0.02
    max_eps_init = 0.05
    max_eps_end = 0.001
    it = 0
    while time.monotonic() < deadline:
        # compute fraction of runtime
        elapsed = time.monotonic() - (deadline - budget_s + 5.0)
        frac = min(1.0, elapsed / (budget_s - 5.0))
        # temperature schedule exponential
        temp = start_temp * (end_temp/start_temp)**frac
        sigma_factor = sigma_factor_init * (sigma_factor_end/sigma_factor_init)**frac
        max_eps = max_eps_init * (max_eps_end/max_eps_init)**frac
        # propose move
        move_type = rng.integers(0, 3)
        cand = a.copy()
        if move_type == 0:
            # block scaling
            width = rng.integers(1, max(2, n//8))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, sigma_factor))
            cand[start:start+width] *= factor
        elif move_type == 1:
            # transfer epsilon mass
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            eps = min(cand[i], max_eps * rng.random())
            cand[i] -= eps
            cand[j] += eps
        else:
            # global lognormal perturbation
            factor_vec = np.exp(rng.normal(0.0, sigma_factor, size=n))
            cand *= factor_vec
        # clip and renormalize
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s
        # evaluate
        val = obj(cand)
        # acceptance
        if val < best_val - 1e-12:
            best_val = val
            best_a = cand.copy()
            a = cand
            print(f"[{elapsed:6.2f}s] new best = {best_val:.6f}")
        else:
            # SA acceptance
            if rng.random() < np.exp(-(val - best_val) / (temp + 1e-12)):
                a = cand
        # occasional gradient descent step
        if it % 1000 == 0 and it > 0:
            # gradient descent step using projected gradient as earlier
            # compute gradient of softmax surrogate with current alpha
            alpha = 100.0  # moderate temperature
            fa_r = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fa_r*fa_r, n=fft_len)[:conv_len]
            c_max = conv.max()
            w = np.exp(alpha * (conv - c_max))
            w_sum = w.sum()
            w /= w_sum
            w_rev = w[::-1]
            # gradient: 2 * (a * w_rev) (convolution)
            grad = 2.0 * np.fft.irfft(np.fft.rfft(a, n=fft_len) * np.fft.rfft(w_rev, n=fft_len), n=fft_len)[:n]
            # line search (simple)
            step = 0.1
            a_new = proj_simplex(a - step * grad)
            val_new = obj(a_new)
            if val_new < best_val - 1e-12:
                best_val = val_new
                best_a = a_new.copy()
                a = a_new
                print(f"[{elapsed:6.2f}s] gradient step improved = {best_val:.6f}")
        it += 1
    # final output
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]

"""
Simulated‑annealing search for a low‑value step function.

The objective to minimise is
    J(a) = 2·n·max(a∗a) / (sum(a)²) ,

which is invariant under scaling, so we keep the vector on the probability
simplex (sum = 1).  Starting from a previous construction (if any) we perform
a temperature‑controlled random walk:

* three elementary moves are used
      – block scaling,
      – transfer of a small mass between two positions,
      – global log‑normal perturbation;
* after each move the vector is projected back onto the simplex;
* a Metropolis acceptance rule with an exponentially decaying temperature
  allows occasional uphill steps;
* the move amplitudes (log‑normal sigma and transferred mass) shrink with
  the temperature;
* if no improvement has been seen for a while a “jitter’’ (large block
  perturbation) is applied to escape plateaus;
* the best candidate found before the time budget expires is returned
  (clipped to the limits enforced by ``evaluate_sequence``).

The implementation is fully vectorised, uses FFT‑based convolution for a
fast evaluation of ``max(a∗a)`` and respects the 1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    the set {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a multiplicative log‑normal perturbation to a random contiguous
    block of the vector and project back onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidate(rng: np.random.Generator) -> np.ndarray:
    """
    Return a feasible starting vector (non‑negative, sum = 1).

    Preference is given to a previously discovered construction stored in the
    global variable ``height_sequence_1``; otherwise a Dirichlet draw of
    dimension 4096 is used.
    """
    if "height_sequence_1" in globals():
        try:
            raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            max_len = 4096
            if raw.size > max_len:
                raw = raw[:max_len]
            elif raw.size < max_len // 2:
                raw = np.concatenate([raw, np.zeros(max_len - raw.size)])
            raw = np.clip(raw, 0.0, None)
            return proj_simplex(raw)
        except Exception:
            pass
    # fall back to a Dirichlet draw (α = 0.5 gives some variability)
    n = 4096
    return proj_simplex(rng.dirichlet(np.full(n, 0.5)))

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  A simulated‑annealing scheme with three simple
    moves and occasional jitter is employed.  The function respects the given
    time budget and returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size & initial point
    # ------------------------------------------------------------------
    a = _initial_candidate(rng)            # on simplex, sum = 1
    n = a.size

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # length for self‑convolution

    # ------------------------------------------------------------------
    # 2) Fast evaluation of the objective (J = 2·n·max(a∗a))
    # ------------------------------------------------------------------
    def compute_J(vec: np.ndarray) -> float:
        """Return 2·n·max(vec * vec) using FFT‑based convolution."""
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())

    best_a = a.copy()
    best_J = compute_J(a)
    print(f"[init] n = {n}, J = {best_J:.6f}")

    # 3) Simulated‑annealing parameters
    # ------------------------------------------------------------------
    # Temperature schedule (exponential decay)
    temp_start = 0.20
    temp_end = 1e-5

    sigma_start = 0.50
    sigma_end = 0.02

    eps_end = 0.001

    # Stagnation handling
    last_improve = start_time
    stagnation_limit = 30.0   # seconds without improvement → jitter

    iteration = 0

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time
        frac = min(1.0, elapsed / (budget_s - 5.0))

        # temperature and step‑size according to the elapsed proportion
        temperature = temp_start * ((temp_end / temp_start) ** frac)
        sigma = sigma_start * ((sigma_end / sigma_start) ** frac)
        eps = eps_start * ((eps_end / eps_start) ** frac)

        # --------------------------------------------------------------
        # Propose a candidate using one of three simple moves
        # --------------------------------------------------------------
        cand = a.copy()
        move_type = rng.integers(0, 3)

        if move_type == 0:
            # --- block scaling -------------------------------------------------
            width = rng.integers(1, max(2, n // 8))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, sigma))
            cand[start:start + width] *= factor

        elif move_type == 1:
            # --- transfer a small amount of mass -------------------------------
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                # amount transferred is a fraction of a[i] (but not larger than a[i])
                delta = min(cand[i], eps * rng.random())
                cand[i] -= delta
                cand[j] += delta

        else:
            # --- global log‑normal perturbation ---------------------------------
            cand *= np.exp(rng.normal(0.0, sigma, size=n))

        # --------------------------------------------------------------
        # Project back onto the simplex (non‑negative, sum = 1)
        # --------------------------------------------------------------
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            continue
        cand /= s

        # --------------------------------------------------------------
        # Evaluate the new candidate
        # --------------------------------------------------------------
        cand_J = compute_J(cand)

        # --------------------------------------------------------------
        # Acceptance test (Metropolis)
        # --------------------------------------------------------------
        if cand_J < best_J - 1e-12:                      # strict improvement
            best_J = cand_J
            best_a = cand.copy()
            a = cand
            last_improve = now
            print(f"[{elapsed:6.2f}s] iter {iteration:6d} NEW BEST J = {best_J:.6f}")
        else:
            # uphill move accepted with probability exp(-Δ/temperature)
            if rng.random() < np.exp(-(cand_J - best_J) / (temperature + 1e-12)):
                a = cand

        # --------------------------------------------------------------
        # Stagnation handling: apply a larger jitter if we have been stuck
        # --------------------------------------------------------------
        if now - last_improve > stagnation_limit:
            a = _random_block_perturb(best_a, rng)
            # ensure we are on the simplex again
            a = proj_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] jitter (stagnation) – J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits used by the
    #    evaluation function)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
