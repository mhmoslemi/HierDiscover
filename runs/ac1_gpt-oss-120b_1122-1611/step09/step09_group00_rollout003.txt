def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)

    # Determine overall deadline
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # list of candidate lengths to try
    candidate_lengths = [2048, 3072, 3809, 4096, 8192] # we ensure each < 10000.
    # but ensure we don't exceed memory/time.

    # Initialize global best using any existing known sequence
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        if best.sum() > 0:
            best /= best.sum()
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        # default uniform of length 4096
        n0 = 4096
        best = np.full(n0, 1.0 / n0, dtype=np.float64)

    best = np.array(best, dtype=np.float64)
    n_best = best.size
    fft_len_best = _next_pow2(3 * n_best)
    best_score = _compute_score(best, fft_len_best)

    print(f"[0.00s] initial best score = {best_score:.6f} (len={n_best})")

    # Main loop: repeated restarts
    while time.monotonic() < deadline:
        # Choose length
        # If best score is above target use same length, else maybe try other lengths
        # We'll randomly pick length from list
        n = int(rng.choice(candidate_lengths))
        # Initialize starting sequence
        if n == n_best:
            a0 = best.copy()
            # jitter
            jitter = rng.normal(scale=0.01, size=n)
            a0 = np.maximum(a0 + jitter, 0.0)
            a0 = _project_onto_simplex(a0)
        else:
            # random Dirichlet
            alpha = np.full(n, 0.5)  # encourages sparsity
            a0 = rng.dirichlet(alpha).astype(np.float64)
        # run optimizer for a slice of time
        slice_budget = min(30.0, deadline - time.monotonic())
        a_opt, score_opt = _optimize_sequence(a0, slice_budget, rng)
        # Update global best
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            n_best = best.size
            fft_len_best = _next_pow2(3 * n_best)
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new global best = {best_score:.6f} (len={n_best})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # final processing: clip
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start_time
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]

def _optimize_sequence(a0, budget, rng):
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3*n)
    best_local = a.copy()
    score_best = _compute_score(a, fft_len)
    start = time.monotonic()
    deadline = start + budget

    # Phase1 exponentiated gradient (up to 40% of budget)
    # We compute fraction of budget for each phase.
    phase1_end = start + 0.4 * budget
    iteration = 0
    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        eps = max_val * 1e-8
        active = np.where(conv >= max_val - eps)[0]
        grad = _gradient_from_active(a, active)
        eta = 1.2 / np.sqrt(iteration + 1.0)  # larger step
        # exponential update
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)
        # evaluate
        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < score_best - 1e-12:
            score_best = score
            best_local = a.copy()
        iteration += 1
        if iteration % 500 == 0:
            pass  # optionally log

    # Phase2 additive subgradient with line search (up to 30% budget)
    phase2_end = start + 0.7 * budget
    iteration = 0
    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        eps = max_val * 1e-6
        active = np.where(conv >= max_val - eps)[0]
        grad = _gradient_from_active(a, active)
        # compute step size
        cur_score = 2.0 * n * max_val
        diff = cur_score - score_best
        norm2 = np.dot(grad, grad) + 1e-12
        step = diff / norm2 if diff > 0 else 0.05 / np.sqrt(iteration + 1.0)
        # line search
        for factor in [1.0, 0.5, 0.25, 0.125]:
            cand = a - step * factor * grad
            cand = _project_onto_simplex(cand)
            conv_cand = _conv_fft(cand, fft_len)
            cand_score = 2.0 * n * conv_cand.max()
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                if cand_score < score_best - 1e-12:
                    score_best = cand_score
                    best_local = cand.copy()
                break
        iteration += 1

    # Phase3 random local moves (remaining budget)
    iteration = 0
    while time.monotonic() < deadline:
        # compute gradient at max index
        conv = _conv_fft(a, fft_len)
        max_idx = int(np.argmax(conv))
        grad_max = _gradient_from_active(a, np.array([max_idx]))
        src = int(np.argmax(grad_max))
        # pick target with low gradient (maybe min)
        tgt = int(np.argmin(grad_max))
        if src == tgt:
            tgt = (tgt + 1) % n
        # choose transfer fraction (log-uniform)
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.1)))  # 0.1% - 10%
        delta = a[src] * frac
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)
        conv_cand = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        if cand_score < score_best - 1e-12:
            score_best = cand_score
            best_local = cand.copy()
            a = cand
        else:
            # accept with small probability to escape if worse (simulated annealing)
            if rng.random() < 0.001:
                a = cand
        iteration += 1
        # occasional jitter
        if iteration % 2000 == 0:
            width = max(1, n // 100)
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.01))
            jittered = a.copy()
            jittered[start_idx:start_idx+width] *= factor
            a = _project_onto_simplex(jittered)
    return best_local, score_best

"""
Propose a step‐function (non‑negative sequence) that minimises

    2 * n * max(a * a) / (sum(a)**2)

where “*” is linear convolution.  The objective is scale‑invariant, so we
keep all iterates on the probability simplex (sum = 1).  The optimiser
combines three ideas:

1. Exponentiated‑gradient descent on a few “worst” convolution indices –
   the update is a multiplicative move a_i ← a_i·exp(‑η·grad_i) followed
   by Euclidean projection onto the simplex.

2. Additive sub‑gradient descent with an adaptive Polyak‑style step‑size
   and a short back‑tracking line‑search.

3. Stochastic local refinement (mass‑transfer moves, occasional jitter and
   a tiny simulated‑annealing acceptance rule) that explores the neighbourhood
   of the current best point.

The routine performs several short random‑restart runs (different
initialisations and/or different lengths) until the time budget is exhausted.
The best sequence seen is returned.  All heavy work (convolution) is done
with an FFT; the score is evaluated analytically (exactly the same as
``evaluate_sequence`` for sum=1) and only once at the very end the real
``evaluate_sequence`` is called.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection expects 1‑D vector.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns an array of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set ``active`` of indices
    where the current maximum is attained (or nearly attained).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact objective for a simplex‑normalised vector (sum(a)=1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimiser for a *single* length / initial point
# ----------------------------------------------------------------------


def _optimize_one_run(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """
    Run three phases (exp‑grad → additive → stochastic) on a copy of ``a0``
    for at most ``time_budget`` seconds.  Returns the best sequence found
    in this run and its score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)

    # best point seen in this run
    best = a.copy()
    best_score = _score_from_a(best, fft_len)

    start = time.monotonic()
    deadline = start + time_budget

    # --------------------------------------------------------------
    # Phase 1 – exponentiated‑gradient (≈40 % of allotted time)
    # --------------------------------------------------------------
    phase1_end = start + 0.40 * time_budget
    it = 0
    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        # keep the active set tiny – only a few worst indices
        top_k = np.argpartition(-conv, 4)[:4]          # 4 indices with largest values
        grad = _gradient_from_active(a, top_k)

        # learning rate – start relatively large and decay like 1/√t
        eta = 1.2 / math.sqrt(it + 1.0)
        # multiplicative update (mirror‑descent on the simplex)
        a = a * np.exp(-eta * grad)
        a = _project_onto_simplex(a)

        # cheap score check
        cur_score = _score_from_a(a, fft_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
        it += 1
        if it % 500 == 0:
            pass          # no verbose output inside the tight loop

    # --------------------------------------------------------------
    # Phase 2 – additive sub‑gradient with adaptive step‑size (≈30 %)
    # --------------------------------------------------------------
    phase2_end = start + 0.70 * time_budget
    it = 0
    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        eps = max_val * 1e-6                     # a small tolerance
        active = np.where(conv >= max_val - eps)[0]
        if active.size == 0:
            active = np.array([np.argmax(conv)])

        grad = _gradient_from_active(a, active)

        # current score
        cur_score = 2.0 * n * max_val

        # Polyak‑style step size (if we are above the best seen)
        diff = cur_score - best_score
        norm2 = np.dot(grad, grad) + 1e-12
        if diff > 0:
            step = diff / norm2
        else:
            step = 0.05 / math.sqrt(it + 1.0)   # fall‑back base step

        # simple back‑tracking line search
        improved = False
        for factor in (1.0, 0.5, 0.25, 0.125):
            cand = a - step * factor * grad
            cand = _project_onto_simplex(cand)
            cand_score = _score_from_a(cand, fft_len)
            if cand_score < cur_score:
                a = cand
                cur_score = cand_score
                improved = True
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best = cand.copy()
                break
        if not improved:
            # still project the plain step – it may not improve but keeps
            # us moving in the right direction
            cand = a - step * grad
            a = _project_onto_simplex(cand)

        it += 1

    # --------------------------------------------------------------
    # Phase 3 – stochastic local refinement (remaining time)
    # --------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        # recompute convolution only every few steps (saves FFTs)
        if it % 30 == 0:
            conv = _conv_fft(a, fft_len)
            max_idx = int(np.argmax(conv))
            # gradient for the *single* active index – gives a quick “tight‑spot” map
            grad_max = _gradient_from_active(a, np.array([max_idx]))
            src = int(np.argmax(grad_max))
            # pick a low‑gradient target (could be the absolute minimum)
            tgt = int(np.argmin(grad_max))
            if src == tgt:
                tgt = (tgt + 1) % n

        # propose a directed mass transfer (log‑uniform fraction)
        if a[src] <= 0.0:
            src = int(rng.integers(0, n))
        if src == tgt:
            tgt = int(rng.integers(0, n))

        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.15)))   # 0.1 % – 15 %
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)

        cand_score = _score_from_a(cand, fft_len)

        # deterministic improvement
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand.copy()
            a = cand
        else:
            # occasional simulated‑annealing acceptance to escape plateaus
            temperature = max(1e-4, (deadline - time.monotonic()) / (deadline - start))
            prob = math.exp(-(cand_score - cur_score) / temperature) if cand_score > cur_score else 1.0
            if rng.random() < prob:
                a = cand

        # occasional jitter to shake the solution
        if it % 4000 == 0 and it > 0:
            width = max(1, n // 120)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            jitter = a.copy()
            jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(jitter)

        it += 1

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine uses several short random‑restart runs, each consisting of
    exponentiated‑gradient, additive sub‑gradient and stochastic refinement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Initialise the global best – use the previously discovered sequence if
    # available, otherwise a uniform distribution of length 4096.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        best = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        best = np.clip(best, 0.0, None)
        if best.sum() > 0.0:
            best = best / best.sum()
        else:
            best = np.full_like(best, 1.0 / best.size)
    else:
        n0 = 4096
        best = np.full(n0, 1.0 / n0, dtype=np.float64)

    best_score = _score_from_a(best, _next_pow2(3 * best.size))
    print(f"[0.00s] initial global best = {best_score:.6f} (len={best.size})")

    # ------------------------------------------------------------------
    # List of lengths we will explore (powers of two and the current best)
    # ------------------------------------------------------------------
    length_pool = [1024, 2048, 3072, 3809, 4096, 6144]
    # make sure we never exceed the deadline with a single run
    min_run_time = 20.0   # seconds per restart (adjustable)

    # ------------------------------------------------------------------
    # Main restart loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a length (biased towards the length of the current best)
        if rng.random() < 0.6:
            n = best.size
        else:
            n = int(rng.choice(length_pool))

        # ----------------------------------------------------------------
        # Build an initial point for this run
        # ----------------------------------------------------------------
        if n == best.size:
            # jitter the current best to escape a possible basin
            a0 = best.copy()
            noise = rng.normal(scale=0.01, size=n)
            a0 = np.maximum(a0 + noise, 0.0)
            a0 = _project_onto_simplex(a0)
        else:
            # random Dirichlet start – α=0.6 gives a moderately sparse start
            alpha = np.full(n, 0.6)
            a0 = rng.dirichlet(alpha).astype(np.float64)

        # ----------------------------------------------------------------
        # Run a short optimisation on this start
        # ----------------------------------------------------------------
        remaining = deadline - time.monotonic()
        run_budget = min(min_run_time, remaining)
        if run_budget < 1.0:          # not enough time for another run
            break

        a_opt, score_opt = _optimize_one_run(a0, run_budget, rng)

        # ----------------------------------------------------------------
        # Update the global best if we improved
        # ----------------------------------------------------------------
        if score_opt < best_score - 1e-12:
            best = a_opt
            best_score = score_opt
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] NEW GLOBAL BEST = {best_score:.6f} (len={best.size})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # Final processing – clip to evaluator limits and compute the exact score
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)          # evaluator caps values anyway
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    # make the sequence available for a potential next run
    globals()["height_sequence_1"] = best.tolist()
    return [float(x) for x in best]
