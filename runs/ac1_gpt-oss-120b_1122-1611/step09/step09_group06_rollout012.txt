"""
Propose a step‑function using a multi‑start, smoothed‑max gradient descent.
The algorithm
  * builds a diverse set of simple seeds (uniform, exponential, linear,
    Dirichlet draws, and the previous best if available);
  * finds the best seed according to the exact objective J = 2·n·max(a*a);
  * refines this seed with a projected Adam optimizer on a smooth approximation
    of the max‑convolution (soft‑max on the self‑convolution vector).  The
    smoothing parameter λ grows from a small value (≈0.5) to a large value
    (≈10³) so that the method first explores the landscape and then focuses
    on the true ℓ∞‑objective;
  * occasional Dirichlet jumps and random block perturbations help to escape
    plateaus.  All iterates are projected onto the probability simplex, which
    guarantees the scale‑invariance of the original evaluation function.
The routine respects a 1000 s budget (with a 5 s safety margin) and returns
the best sequence found.
"""

# Helper utilities (top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator):
    """Return a list of diverse probability vectors of length ``n``."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponentially decaying tails
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy / half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a handful of Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # previous construction, if present
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        prev /= prev.sum()
        cand.append(prev)

    return cand


def _compute_J_fast(a: np.ndarray, fft_len: int, conv_len: int):
    """
    Exact objective J = 2·n·max(a⋆a) for a already on the simplex.
    Returns (J, full_convolution_vector).
    """
    n = a.size
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    J = float(2.0 * n * conv.max())
    return J, conv


def _softmax_weights(b: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max over ``b`` with temperature 1/lam, returned as a probability vector."""
    # subtract max for numerical stability
    b_shift = b - b.max()
    exp_term = np.exp(lam * b_shift)
    w = exp_term / exp_term.sum()
    return w


def _gradient_via_fft(a: np.ndarray, w: np.ndarray, fft_len: int):
    """
    Compute the exact gradient of the smoothed max:
        g = 2 * corr(w, a)
    using FFTs (linear correlation, not convolution).
    Returns a vector of length len(a).
    """
    # FFT of w (same length as a)
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a, n=fft_len)
    # correlation = ifft( fw * conj(fa) )
    corr_full = np.fft.irfft(fw * np.conj(fa), n=fft_len)
    # central segment corresponding to lags 0 … n‑1
    n = a.size
    start = n - 1
    g = 2.0 * corr_full[start:start + n]
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method uses several seeds, selects the best, then performs a
    projected Adam optimisation on a smooth approximation of the ℓ∞‑objective.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Problem size (use the length of the previous best if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        n = max(256, min(n, 5000))           # keep transforms cheap
    else:
        n = 4000

    conv_len = 2 * n - 1
    # we need a length that can host both the convolution (2n‑1) and the
    # correlation (3n‑2); pick the next power of two for speed.
    fft_len = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # 2) Build a diverse seed pool and pick the best one (exact J)
    # ------------------------------------------------------------------
    seeds = _initial_candidates(n, rng)
    best_vec = None
    best_score = np.inf
    for s in seeds:
        s = proj_simplex(s.astype(np.float64))
        J, _ = _compute_J_fast(s, fft_len, conv_len)
        if J < best_score:
            best_score = J
            best_vec = s.copy()
    print(f"[{time.monotonic() - start_time:6.2f}s] Initial best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Optimisation – projected Adam on a soft‑max approximation
    # ------------------------------------------------------------------
    a = best_vec.copy()          # current iterate
    m = np.zeros_like(a)         # Adam first moment
    v = np.zeros_like(a)         # Adam second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.monotonic()
    stagnation_limit = 20.0      # seconds without improvement → restart

    # learning‑rate schedule (linear decay)
    lr_start, lr_end = 0.5, 0.001

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed_frac = (now - start_time) / (deadline - start_time)

        # --- evaluate convolution and objective ---------------------------------
        J, conv = _compute_J_fast(a, fft_len, conv_len)

        if J < best_score - 1e-12:
            best_score = J
            best_vec = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] New best J = {best_score:.6f}")

        # --- smoothing parameter (λ grows) ------------------------------------
        lam = 0.5 + 1000.0 * (elapsed_frac ** 2)   # start close to 0.5, end ≈1000

        # --- soft‑max weights ----------------------------------------------------
        w = _softmax_weights(conv, lam)

        # --- gradient (exact) ----------------------------------------------------
        g = _gradient_via_fft(a, w, fft_len)

        # --- Adam step -----------------------------------------------------------
        lr = lr_start * (1.0 - elapsed_frac) + lr_end * elapsed_frac
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        # occasional Dirichlet “jump” to explore new regions
        if rng.random() < 0.006:
            mix = 0.12
            a_new = proj_simplex((1.0 - mix) * a_new + mix * rng.dirichlet(np.ones(n) * 0.3))

        a = a_new

        # --- stagnation handling -------------------------------------------------
        if now - last_improve > stagnation_limit:
            # perturb the best point a little and continue
            eps = np.exp(rng.normal(0.0, 0.10))
            a = best_vec.copy()
            block_width = max(1, n // 20)
            start = rng.integers(0, n - block_width + 1)
            a[start:start + block_width] *= eps
            a = proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Restart after stagnation – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to admissible limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
