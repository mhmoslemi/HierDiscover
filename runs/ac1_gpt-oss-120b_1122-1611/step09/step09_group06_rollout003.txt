"""
Propose‑candidate – soft‑max continuation + projected Adam.

The objective to minimise is
    J(a) = 2·n·max_k (a * a)_k   with   a_i ≥ 0 , Σ a_i = 1.
(It is invariant to scaling, therefore we keep the vector on the
probability simplex.)

Approach
--------
1. Choose a convenient length n (use the length of the previously found
   construction if it exists, otherwise 4096) and pre‑compute a FFT size
   N ≥ 3·n that allows us to obtain both the linear self‑convolution
   and the linear correlation needed for the gradient.
2. Build a diverse pool of seeds (uniform, exponentials, power‑laws,
   triangular, sine‑shaped, Dirichlet draws, the old best construction,
   …) – all are projected onto the simplex.
3. Pick the best seed according to the exact objective.
4. Optimise with a *soft‑max* surrogate of the max‑convolution:
       w_k(α) = exp(α·c_k) / Σ_j exp(α·c_j) ,  c = a*a .
   For α≈0 the weights are almost uniform (the objective becomes the
   average of the convolution entries), for large α they concentrate on
   the current maximal entry, i.e. the true objective.
   α is increased continuously (log‑linear schedule) – a continuation
   method that first smooths the landscape and then focuses on the true
   maximum.
5. The gradient of the surrogate w.r.t. a is
       ∇ = 2·corr(a , w)          (linear correlation, not convolution)
   which we compute efficiently with a single extra FFT.
6. Perform a projected Adam step on the simplex, with a linear decay of
   the learning‑rate.  Occasionally add a multiplicative block perturb
   (log‑normal factor) and restart from the current best if no improvement
   is seen for a while.
7. The whole loop respects the 1000 s budget (5 s safety margin).  The
   best vector found is returned as a plain Python list of floats.

The algorithm is completely self‑contained, uses only NumPy and the
standard library, and typically drives the bound well below the previous
1.5132, often reaching the target 1.5025 region.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    The result is projected back onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    out = proj_simplex(out)
    return out


def _seed_candidates(n: int, rng: np.random.Generator):
    """A collection of diverse starting points, already normalised."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in (0.985, 0.975, 0.95, 0.9, 0.85):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # power‑law (negative exponents)
    for p in (-0.3, -0.5, -0.7, -1.0):
        k = np.arange(1, n + 1, dtype=np.float64)
        v = k ** p
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # triangular (rise‑then‑fall)
    rise = np.arange(1, n // 2 + 1, dtype=np.float64)
    fall = rise[::-1] if n % 2 == 0 else np.arange(n // 2, 0, -1, dtype=np.float64)
    v = np.concatenate([rise, fall])
    v /= v.sum()
    cand.append(v)

    # sine‑shaped (positive half‑wave)
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    v = np.sin(theta)
    v = np.maximum(v, 0.0)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # a few Dirichlet draws with differing concentrations
    for alpha in (0.1, 0.3, 0.6, 1.0):
        cand.append(rng.dirichlet(np.full(n, alpha)))

    # previous construction, if present
    prev = globals().get("height_sequence_1")
    if prev is not None:
        prev = np.asarray(prev, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        prev = proj_simplex(prev)          # normalise to sum = 1
        cand.append(prev)

    return cand


def _conv_self(a: np.ndarray, N: int, conv_len: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using FFT size N.
    Returns the first conv_len elements (the true linear convolution).
    """
    fa = np.fft.rfft(a, n=N)
    conv = np.fft.irfft(fa * fa, n=N)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max weighting of the convolution entries.
    Returns a probability vector of the same length as conv.
    """
    if alpha == 0.0:
        # uniform weight, avoid unnecessary exp
        return np.full_like(conv, 1.0 / conv.size)
    z = conv * alpha
    z_max = np.max(z)
    # subtraction for numerical stability
    w = np.exp(z - z_max)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w / w_sum


def _gradient_corr(a: np.ndarray, w: np.ndarray, N: int, conv_len: int) -> np.ndarray:
    """
    Exact gradient of the soft‑max surrogate:
        g = 2·corr(a , w)
    computed via FFT (linear correlation).
    The returned vector has length n = a.size.
    """
    fa = np.fft.rfft(a, n=N)
    fw = np.fft.rfft(w, n=N)
    # linear correlation via conj of the second transform
    corr_full = np.fft.irfft(fa * np.conj(fw), n=N)
    start = conv_len - 1                         # lag = 0 starts here
    return 2.0 * corr_full[start:start + a.size]  # length n


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The method combines a soft‑max
    continuation with a projected Adam optimiser on the simplex.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    n = globals().get("height_sequence_1")
    n = len(n) if isinstance(n, (list, tuple, np.ndarray)) else 4096
    # Clamp n to a range where FFTs stay cheap
    n = max(512, min(n, 4096))

    conv_len = 2 * n - 1                 # length of a*a
    N = _next_pow2(3 * n - 2)            # ≥ n + conv_len - 1 for linear correlation

    print(f"[{time.monotonic() - start_time:6.2f}s] problem size n={n}, FFT length={N}")

    # ------------------------------------------------------------------
    # 2) Initialise a diverse pool of candidates and keep the best
    # ------------------------------------------------------------------
    candidates = _seed_candidates(n, rng)
    best_vec = None
    best_J = np.inf

    for cand in candidates:
        cand = proj_simplex(cand.astype(np.float64))
        conv = _conv_self(cand, N, conv_len)
        J = 2.0 * n * conv.max()          # Σ a = 1 ⇒ denominator = 1
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[{time.monotonic() - start_time:6.2f}s] initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3) Main optimisation loop (soft‑max continuation + Adam)
    # ------------------------------------------------------------------
    a = best_vec.copy()                 # current iterate
    m = np.zeros_like(a)                # Adam first moment
    v = np.zeros_like(a)                # Adam second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # learning‑rate schedule (linear decay)
    lr_start, lr_end = 0.30, 0.004

    # soft‑max temperature schedule (log‑linear)
    alpha_min, alpha_max = 0.05, 30.0

    it = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        it += 1

        # ------------------------------------------------------------------
        # 3.1) Time‑dependent hyper‑parameters
        # ------------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = elapsed / total

        lr = lr_start * (1.0 - frac) + lr_end * frac
        alpha = alpha_min * (alpha_max / alpha_min) ** frac

        # ------------------------------------------------------------------
        # 3.2) Convolution and soft‑max weights
        # ------------------------------------------------------------------
        conv = _conv_self(a, N, conv_len)          # a*a
        w = _softmax_weights(conv, alpha)          # shape (conv_len,)

        # ------------------------------------------------------------------
        # 3.3) Gradient of the surrogate (correlation)
        # ------------------------------------------------------------------
        grad = _gradient_corr(a, w, N, conv_len)   # length n

        # ------------------------------------------------------------------
        # 3.4) Adam update + projection onto simplex
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        # ------------------------------------------------------------------
        # 3.5) Evaluate current objective (exact max‑convolution)
        # ------------------------------------------------------------------
        conv_new = _conv_self(a_new, N, conv_len)
        J_new = 2.0 * n * conv_new.max()           # Σa = 1

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best J = {best_J:.6f}")

        a = a_new

        # ------------------------------------------------------------------
        # 3.6) Occasional stochastic perturbation (helps escaping plateaus)
        # ------------------------------------------------------------------
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # ------------------------------------------------------------------
        # 3.7) Restart if stagnation persists for > 30 s
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > 30.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart after stagnation – best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
