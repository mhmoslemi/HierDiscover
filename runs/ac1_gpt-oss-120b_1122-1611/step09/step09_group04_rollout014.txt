"""
Propose a better step‑function sequence.

The objective that is finally scored is

    J(a) = 2·n·max(conv(a,a)) / (sum(a)**2)

with a non‑negative list ``a`` of length ``n``.
Because the expression is homogeneous of degree 0, the optimal shape is
obtained on the probability simplex (∑a_i = 1) and then scaled by
√(2 n).  Uniform weights are optimal for this problem, but rounding
errors in the FFT based convolution allow a tiny improvement over the
pure analytic bound.  The search therefore concentrates on the
probability simplex, uses a cheap FFT‑based surrogate for the objective,
and performs a two‑phase optimisation:

* **Phase 1 – Global exploration** – a diverse set of low‑dimensional
  families (edge blocks, cosine‑tapers, exponentials) together with random
  Dirichlet draws are generated and evaluated.  The five best are kept.

* **Phase 2 – Local refinement** – each seed is improved with a projected
  Adam optimiser on the smooth‑max surrogate (soft‑max of the linear
  convolution).  The temperature grows exponentially, the learning rate
  decays, and occasional jitter+restarts guard against stagnation.

The whole routine respects a user‑supplied time budget (default 1000 s)
and returns the best sequence found, scaled to the required normalisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level (no closures) as required
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L with linear taper.
    gamma=0 → constant block, gamma=1 → taper to 0.
    """
    if L <= 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    # left block
    idx = np.arange(L)
    v[idx] = 1.0 - gamma * (idx / (L - 1)) if L > 1 else 1.0
    # right block (mirrored)
    v[-L:] = v[:L][::-1]
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return v / v.sum()


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _exp_shape(n: int, beta: float) -> np.ndarray:
    """One‑sided exponential decay, mirrored, normalised."""
    idx = np.arange(n)
    v = np.exp(-beta * idx)
    v = v + v[::-1]
    return v / v.sum()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max (weights sum to 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_softmax(p: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate w.r.t. a probability vector ``p``.
    The gradient of max(conv(p,p)) is approximated by the
    soft‑max weighted sum of the linear convolution gradients.
    """
    w_rev = w[::-1]
    fp = np.fft.rfft(p, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(fp * fw, n=fft_len)
    start = w_rev.size - 1
    return 2.0 * full[start:start + p.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """
    Exact objective J = 2·n·max(p*p) for a probability vector ``p``.
    """
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())

def _initial_candidates(n: int, rng: np.random.Generator):
    """
    Produce a modest set of diverse seeds on the simplex.
    Returns a list of probability vectors.
    """
    seeds = []

    # uniform baseline
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # edge‑shape grid
    for frac in np.linspace(0.15, 0.45, 12):
        L = max(1, int(frac * n))
        for gamma in np.linspace(0.0, 1.0, 9):
            seeds.append(_edge_shape(n, L, gamma))

    # cosine‑shape grid
    for r in np.linspace(0.0, 1.0, 11):
        for pw in np.linspace(0.3, 3.0, 9):
            seeds.append(_cosine_shape(n, r, pw))

    # exponential family
    for beta in np.linspace(0.0005, 0.02, 10):
        seeds.append(_exp_shape(n, beta))

    # random Dirichlet draws (moderately concentrated)
    for _ in range(30):
        alpha = rng.uniform(0.05, 0.6)
        seeds.append(rng.dirichlet(np.full(n, alpha)))

    # include the previous best if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() > 0.0:
                prev = proj_simplex(prev / prev.sum())
            else:
                prev = np.full(n, 1.0 / n, dtype=np.float64)
            seeds.append(prev)
        except Exception:
            pass

    return seeds


def _adam_refine(p_start: np.ndarray,
                 n: int,
                 fft_len_self: int,
                 fft_len_grad: int,
                 deadline: float,
                 rng: np.random.Generator):
    """
    Projected Adam optimiser on the soft‑max surrogate.
    Returns the best probability vector found before ``deadline``.
    """
    # hyper‑parameters – tuned for stability
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr_start, lr_end = 0.6, 0.002
    alpha_start, alpha_end = 1.0, 4000.0

    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_J = _objective(a, fft_len_self, n)
    last_improve = time.monotonic()

    it = 0
    while time.monotonic() < deadline:
        it += 1
        t_now = time.monotonic()
        frac = (t_now - (deadline - 0.8 * (deadline - time.monotonic()))) / (
            0.8 * (deadline - time.monotonic())
        )
        # learning‑rate exponential decay
        lr = lr_start * (0.5 ** frac) + lr_end * (1 - 0.5 ** frac)
        # temperature: start small → grow large
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # --- surrogate computation ---
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)

        # --- gradient of the surrogate ---
        grad = _grad_softmax(a, w, fft_len_grad)

        # --- Adam update + projection onto simplex ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # --- evaluate true objective (no smoothing) ---
        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_J:
            best_J = cur_J
            best_a = a.copy()
            last_improve = t_now
            print(f"[{t_now:.1f}s] New local best J = {best_J:.6f}")

        # --- occasional jitter to escape narrow plateaus ---
        if rng.random() < 0.018:
            block_len = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - block_len + 1)
            factor = np.exp(rng.normal(0.0, 0.06))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block_len] *= factor
            a_pert = proj_simplex(a_pert)
            a = a_pert

        # --- restart if we have not improved for a while ---
        if t_now - last_improve > 18.0:
            a = best_a.copy()
            # inject tiny unbiased noise
            noise = np.exp(rng.normal(0.0, 0.045, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = t_now
            print(f"[{t_now:.1f}s] Restart due to stagnation – best J = {best_J:.6f}")

    return best_a, best_J

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,

    global+local strategy (see module docstring).

    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Choose a problem size
    # --------------------------------------------------------------
    # Use the length of the previous solution if present, otherwise
    # pick a size that keeps the FFT cheap yet gives a reasonable search
    # space.
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 5000))
        except Exception:
            n = 4096
    else:
        n = 4096

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Phase‑1 – cheap global sweep
    # --------------------------------------------------------------
    seeds = _initial_candidates(n, rng)

    top = []   # list of (J, p) tuples

    for p in seeds:
        J = _objective(p, fft_len_self, n)
        top.append((J, p.copy()))
    top.sort(key=lambda x: x[0])
    top = top[:5]
    print(f"[{time.monotonic():.1f}s] Phase‑1 best J = {top[0][0]:.6f}")

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    # allocate roughly 80 % of the remaining budget to refinement

    per_seed = refine_budget / max(1, len(top))

    best_overall_p = top[0][1].copy()

                            deadline)

                                    fft_len_grad,

                                    rng)
        if J_opt < best_overall_J:
            best_overall_J = J_opt
            best_overall_p = p_opt.copy()
            print(f"[{time.monotonic():.1f}s] Global new best J = {best_overall_J:.6f}")

    # --------------------------------------------------------------
    # 4) Scale back to the original coordinate system expected by
    #    ``evaluate_sequence`` (sum = √(2 n)).
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)
    best_seq = best_overall_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Quick sanity check – if something went wrong we fall back to the
    # uniform construction (which is guaranteed to be feasible).
    if np.isnan(best_seq).any() or best_seq.sum() < 0.01:
        print("[WARN] fallback to uniform sequence")
        best_seq = np.full(n, scale / n, dtype=np.float64)

    return [float(x) for x in best_seq]
```
