"""
Gradient‑soft‑max descent + occasional simulated‑annealing tweaks.

The objective is      J(a) = 2·n·max(conv(a,a))                (∑a = 1).
Because the max is non‑smooth we minimise a smooth surrogate

    Sα(a) = (1/α)·log Σ_k exp(α·conv_k(a))

which converges to the true max when α → ∞.  The gradient of Sα can be
written in closed form using a convolution, so each iteration costs only
a few FFT calls (O(n log n)).  The algorithm proceeds as follows:

1. Load the previously discovered construction (``height_sequence_1``) if
   available, otherwise start from a uniform vector; normalise to the
   probability simplex.
2. Repeatedly
      – compute the current convolution and the soft‑max weights,
      – obtain the gradient via a second convolution,
      – propose a multiplicative (exponentiated‑gradient) update,
      – project back onto the simplex,
      – accept the move with a Metropolis rule whose temperature cools
        linearly from ~5 % of the current best value to 1e‑8.
   The step‑size η and the soft‑max temperature α are increased
   gradually (α → 200, η → 0.01) to sharpen the approximation.
3. If no improvement has been seen for 30 s a small Gaussian jitter
   around the best‑so‑far vector is injected to escape plateaus.
4. After the time budget expires the best vector encountered is returned.
   It is clipped to the limits imposed by ``evaluate_sequence`` and
   renormalised (the renormalisation does not affect the objective
   appreciably).

The method is completely deterministic apart from the random seed and
runs on a single core; all heavy work is performed with NumPy FFTs.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, Σw = 1}.  Used as a safety net; the main algorithm employs
    multiplicative updates which already stay on the simplex.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return full linear convolution a*a using an FFT of length fft_len."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * a.size - 1]


def _correlate_fft(w: np.ndarray, a_rev: np.ndarray) -> np.ndarray:
    """
    Correlate w (length 2n‑1) with a_rev (length n) and return the
    slice of length n that corresponds to Σ_j w_{i+j}·a_j.
    """
    n = a_rev.size
    L = _next_pow2(w.size + n - 1)
    W = np.fft.rfft(w, n=L)
    A = np.fft.rfft(a_rev, n=L)
    full = np.fft.irfft(W * A, n=L)
    # the desired slice starts at offset n‑1
    return full[n - 1 : n - 1 + n]


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a step‑function vector that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. Choose problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 1. Initialise a feasible vector on the simplex.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # small random perturbation just to avoid pathological symmetry
    a0 = a0 * np.exp(rng.normal(0.0, 0.015, size=n))
    a0 = a0 / a0.sum()
    cur_a = a0.copy()

    # ------------------------------------------------------------------
    # 2. Pre‑compute a few quantities.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin

    # initial evaluation
    cur_conv = _conv_fft(cur_a, fft_len)
    cur_max = cur_conv.max()
    cur_J = 2.0 * n * cur_max
    best_a = cur_a.copy()
    best_J = cur_J
    last_improve = start_time

    # schedules
    T_start = best_J * 0.05          # 5 % of current best
    T_end = 1e-8
    eta = 0.30                       # initial step‑size for multiplicative update
    alpha = 1.0                      # soft‑max temperature
    max_alpha = 200.0
    min_eta = 0.01

    iteration = 0
    print(f"[0.00s] start J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac_time = (now - start_time) / (deadline - start_time)
        T = T_start * (1.0 - frac_time) + T_end * frac_time

        # ---- gradient of the soft‑max surrogate --------------------------------
        # soft‑max weights; shift for numerical stability
        shift = cur_conv - cur_max
        w = np.exp(alpha * shift)               # w_k ≥ 0, Σw_k = 1 after normalisation
        sum_w = w.sum()

        # correlation to obtain gradient (see analysis)
        a_rev = cur_a[::-1]
        corr = _correlate_fft(w, a_rev)       # length n
        grad = 2.0 * corr / sum_w             # ∂Sα/∂a

        # ---- propose multiplicative (exponentiated‑gradient) move ---------------
        a_prop = cur_a * np.exp(-eta * grad)   # ensures positivity
        a_prop_sum = a_prop.sum()
        if a_prop_sum == 0.0:
            a_prop = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a_prop /= a_prop_sum                  # back onto simplex

        # occasional global jitter to improve exploration
        if iteration % 5000 == 0 and iteration > 0:
            noise = np.exp(rng.normal(0.0, 0.018, size=n))
            a_prop = a_prop * noise
            a_prop /= a_prop.sum()

        # ---- evaluate the proposal ------------------------------------------------
        prop_conv = _conv_fft(a_prop, fft_len)
        prop_max = prop_conv.max()
        prop_J = 2.0 * n * prop_max

        # ---- Metropolis acceptance ------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            # probability of accepting an uphill move
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = a_prop
            cur_conv = prop_conv
            cur_max = prop_max
            cur_J = prop_J

            if prop_J < best_J - 1e-12:
                best_J = prop_J
                best_a = cur_a.copy()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best J = {best_J:.6f}")

        # ---- stagnation handling -------------------------------------------------
        if now - last_improve > 30.0:
            # jitter the best solution to escape plateaus
            cur_a = best_a * np.exp(rng.normal(0.0, 0.04, size=n))
            cur_a /= cur_a.sum()
            cur_conv = _conv_fft(cur_a, fft_len)
            cur_max = cur_conv.max()
            cur_J = 2.0 * n * cur_max
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] jitter after stagnation – best J = {best_J:.6f}")

        # ---- schedule updates ----------------------------------------------------
        iteration += 1
        if iteration % 2000 == 0:
            alpha = min(alpha * 1.18, max_alpha)   # sharpen the surrogate
        if iteration % 1000 == 0:
            eta = max(eta * 0.97, min_eta)         # slow down as we converge

    # ------------------------------------------------------------------
    # 4. Return the best vector (post‑process for evaluate_sequence).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = best_a.sum()
    if s == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= s
    return [float(x) for x in best_a]
