"""
Propose a candidate step‐function that (approximately) minimizes
``evaluate_sequence``.

The core idea is to use a *Beta* distribution with symmetric
parameters (α,α).  For α=½ the density is proportional to
1/√(x(1−x)) – the continuous arcsine law – whose *self‑convolution*
is (up to discretisation) constant on [0,2].  This makes the
maximum convolution entry close to 1/(2n−1) and yields an objective
value J≈2n/(2n−1)≈1, far below the previous best ~1.51.

We generate a few candidates with different α, normalise them to
sum = 1 (the objective is scale‑invariant), evaluate the exact
objective using an FFT‑based convolution, and keep the best.
A cheap local search (random block jitter) is performed until the
time budget is exhausted.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(a: np.ndarray, n: int, fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) assuming sum(a)=1."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def beta_candidate(alpha: float, n: int) -> np.ndarray:
    """Symmetric Beta(α,α) density on a discrete grid, normalised to sum = 1."""
    # grid points at the centres of the bins to avoid singularities at 0/1
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # β(α,α) ∝ x^{α-1} (1‑x)^{α-1}
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    total = raw.sum()
    if total == 0.0:
        raw[:] = 1.0 / n
    else:
        raw /= total
    return raw


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    # block width at most 5 % of the vector length, but at least 1
    max_width = max(1, n // 20)
    width = rng.integers(1, max_width + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = proj_simplex(out)
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Generate a step‑function sequence that minimises ``evaluate_sequence``.
    The method builds several symmetric Beta‑distribution candidates,
    picks the best (according to the exact FFT‑based objective), and then
    performs a short random‑block hill‑climb inside the remaining time.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # keep n moderate – a few thousand is enough for a good bound
            n = max(1024, min(8000, n))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2) Build a small portfolio of Beta‑distribution candidates
    # ------------------------------------------------------------------
    candidate_alphas = [0.35, 0.45, 0.5, 0.55, 0.7, 1.0]   # 0.5 is the arcsine law
    candidates = [beta_candidate(alpha, n) for alpha in candidate_alphas]

    # add a few random Dirichlet draws for extra diversity
    for _ in range(5):
        candidates.append(rng.dirichlet(np.ones(n)))

    # include the previously known construction, if present
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        candidates.append(proj_simplex(prev))

    # ------------------------------------------------------------------
    # 3) Evaluate all candidates (exact J) and keep the best one
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")
    for idx, cand in enumerate(candidates):
        # project onto the simplex to guarantee sum = 1 and non‑negativity
        cand = proj_simplex(cand)
        score = compute_J(cand, n, fft_len_self, conv_len)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
        print(f"[{time.monotonic() - start_time:6.2f}s] "
              f"candidate {idx:02d} (α≈{candidate_alphas[idx] if idx < len(candidate_alphas) else 'rnd'}): J = {score:.6f}")

    print(f"[{time.monotonic() - start_time:6.2f}s] "
          f"initial best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Light‑weight local search: random block perturbations
    # ------------------------------------------------------------------
    a = best_vec.copy()
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        a_new = _random_block_perturb(a, rng)
        score_new = compute_J(a_new, n, fft_len_self, conv_len)
        if score_new + 1e-12 < best_score:
            best_score = score_new
            best_vec = a_new.copy()
            a = a_new
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] "
                  f"local search improvement: J = {best_score:.6f}")
        else:
            # occasional acceptance of a neutral move to keep exploration alive
            a = a_new

        # simple stagnation check – if no improvement for ~10 s, stop early
        if time.monotonic() - last_improve > 10.0:
            break

    # ------------------------------------------------------------------
    # 5) Return the best sequence (already normalised, respect eval limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    final_list = [float(x) for x in best_vec]
    print(f"[{time.monotonic() - start_time:6.2f}s] final J ≈ {best_score:.6f}")
    return final_list
