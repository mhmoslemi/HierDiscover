        data = []

"""
Propose a step‐function minimizing ``evaluate_sequence``.

Algorithm
---------

1️⃣  Initialise a probability vector `f` (sum = 1) from the supplied seed
    ``height_sequence_1`` (down‑sampled to ≤ 2048 entries) or from a random
    Dirichlet draw if no good seed exists.

2️⃣  **Alternating LP phase** – for the current `f` compute
    `M = max(f*f)`.  Build the Toeplitz matrix `T_f` (size
    ``(2·n‑1) × n``) that implements the linear map ``b ↦ f * b``.
    Solve the linear programme

        maximise   Σ b_i
        subject to T_f·b ≤ M ,   b ≥ 0

    (`scipy.optimize.linprog`, method = “highs”).  
    If the optimum sum `Σb` exceeds 1 we obtain slack: we normalise
    `g = b / Σb` (so `Σg = 1`) and move the current iterate towards it

        f ← (1‑t)·f + t·g,   t decreasing from 0.5 to 0.02.

    The iteration is repeated while time permits (≈ 30 s are kept for the
    final refinement).  The best vector seen is stored.

3️⃣  **Local swap search** – with the best vector from phase 2 we perform a
    cheap combinatorial refinement: repeatedly pick two indices, swap their
    heights and keep the swap if it lowers the true objective
    `J(f) = 2·n·max(f*f)`.  Small uphill moves are occasionally accepted
    (temperature = 1e‑4) to escape plateaus.  The loop runs until the
    remaining budget is exhausted.

4️⃣  Clip the result to the limits imposed by ``evaluate_sequence`` and
    return it as a plain ``list[float]``.  All numbers are non‑negative,
    the sum is ≈ 1, and the routine respects the global 1000 s budget.

The combination of a global LP step (which can increase the total mass
while respecting the current peak) and a fast stochastic local search
usually improves the peak beyond the previous Adam‑based method,
bringing the evaluation value below the target 1.5025.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _toeplitz_conv_matrix(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2·n‑1)×n Toeplitz matrix T_f that computes
    (T_f @ b) = f * b (full linear convolution).
    Each diagonal i (offset i) is constant = f[i].
    """
    n = f.size
    conv_len = 2 * n - 1
    # All diagonals have length n (the number of columns)
    data = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)                # 0 … n‑1
    T = sp.diags(data, offsets, shape=(conv_len, n), format="csc")
    return T


def _objective_J(a: np.ndarray, fft_len: int) -> float:
    """Return J = 2·n·max(a*a) for a (assumed to sum to 1)."""
    n = a.size
    conv = _self_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method consists of an LP‑based global step
    followed by a cheap swap‑based local refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Choose problem size
    # --------------------------------------------------------------
    MAX_N = 2048               # keep the LP tractable
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        # down‑sample uniformly if it is longer than MAX_N
        if base.size > MAX_N:
            idx = np.linspace(0, base.size - 1, MAX_N).astype(int)
            base = base[idx]
        elif base.size < MAX_N:
            # pad with zeros to reach MAX_N (extra freedom for the optimiser)
            pad_len = MAX_N - base.size
            base = np.concatenate([base, np.zeros(pad_len)])
        n = base.size
        a = base.copy()
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / a_sum
        print(f"[init] Seeded from height_sequence_1 (len={n})")
    else:
        # No seed – start from a random Dirichlet draw (moderately sparse)
        n = MAX_N
        a = rng.dirichlet(alpha=np.full(n, 0.3))
        print("[init] No seed – random Dirichlet start")

    # --------------------------------------------------------------
    # Prepare FFT length (next power of two for speed)
    # --------------------------------------------------------------
    fft_len = 1 << ((2 * n - 2).bit_length())   # length for a*a convolution

    # --------------------------------------------------------------
    # Keep track of the best vector seen
    # --------------------------------------------------------------
    best_vec = a.copy()
    best_J = _objective_J(best_vec, fft_len)
    print(f"[init] J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 2) Alternating LP phase
    # --------------------------------------------------------------
    # Reserve some time for the final permutation phase
    perm_reserve = 30.0                         # seconds
    lp_deadline = deadline - perm_reserve
    it = 0
    last_improve = time.monotonic()
    while time.monotonic() < lp_deadline:
        it += 1
        # --- current peak and convolution matrix ---
        conv = _self_convolution(a, fft_len)
        M = conv.max()

        # --- build Toeplitz matrix for the linear constraints ---
        T_f = _toeplitz_conv_matrix(a)          # shape (2n‑1, n)

        # --- linear programme: maximise sum(b) s.t. T_f·b ≤ M, b≥0 ---
        c = -np.ones(n)                         # minimise –sum(b)
        ub = np.full(2 * n - 1, M)              # right‑hand side
        bounds = [(0.0, None)] * n
        res = opt.linprog(c, A_ub=T_f, b_ub=ub,
                          bounds=bounds,
                          method="highs",
                          options={"presolve": True})
        if not res.success:
            print(f"[LP] iteration {it}: linprog failed (status={res.status})")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # No slack – cannot improve further
            print(f"[LP] iteration {it}: no slack (Σb={sum_b:.6f}) – stop")
            break

        # --- construct the improved direction g (normalised) ---
        g = b / sum_b                           # Σg = 1

        # --- mixing schedule (t decreasing from 0.5 → 0.02) ---
        progress = min(1.0, (time.monotonic() - (deadline - perm_reserve - 30)) /
                             (lp_deadline - (deadline - perm_reserve - 30) + 1e-12))
        t_start, t_end = 0.5, 0.02
        t = t_start * (1.0 - progress) + t_end * progress

        # Perform the convex combination
        a_new = (1.0 - t) * a + t * g
        a_new = proj_simplex(a_new)             # protect against numerical drift

        # --- evaluate improvement ---
        J_new = _objective_J(a_new, fft_len)
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            print(f"[LP {it:3d}] new best J = {best_J:.6f}")

        # Occasionally inject a random block perturbation to escape plateaus
        if rng.random() < 0.03:
            a_new = _random_block_perturb(a_new, rng)

        # Stagnation check – if no improvement for >20 s, restart from best
        if time.monotonic() - last_improve > 20.0:
            a_new = best_vec.copy()
            # tiny multiplicative noise before restarting
            a_new = a_new * np.exp(rng.normal(0.0, 0.07, size=n))
            a_new = proj_simplex(a_new)
            last_improve = time.monotonic()
            print(f"[LP {it:3d}] restart due to stagnation – best J = {best_J:.6f}")

        a = a_new

    # --------------------------------------------------------------
    # 3) Permutation / swap local search phase
    # --------------------------------------------------------------
    swap_deadline = deadline
    cur_vec = best_vec.copy()
    cur_J = best_J
    attempts = 0
    accepts = 0
    temperature = 1e-4

    while time.monotonic() < swap_deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        cand = cur_vec.copy()
        cand[i], cand[j] = cand[j], cand[i]   # simple swap

        cand_J = _objective_J(cand, fft_len)

        if cand_J < cur_J - 1e-12:
            cur_vec = cand
            cur_J = cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                print(f"[swap {attempts}] new best J = {best_J:.6f}")
        else:
            # tiny probability of accepting a worsening move
            if rng.random() < np.exp(-(cand_J - cur_J) / temperature):
                cur_vec = cand
                cur_J = cand_J
                accepts += 1

        # Every few tens of thousands of attempts apply a stronger perturbation
        if attempts % 30000 == 0:
            cur_vec = _random_block_perturb(cur_vec, rng)

    print(f"[summary] Swap attempts: {attempts}, accepts: {accepts}")
    print(f"[summary] Best J found = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence (clip to evaluation limits)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
