"""
Two‑phase optimizer for the step‑function problem (enhanced).

Phase 1 – Global exploration:
    * Uniform baseline.
    * Edge‑heavy shapes with optional interior plateau (parameterised by
      block length, taper and interior height ratio).
    * Cosine families and a collection of Dirichlet random draws.
    * The best few candidates (by the true objective) are kept.

Phase 2 – Local refinement:
    * Adam on a smoothed surrogate (soft‑max on the convolution) with an
      exponential temperature schedule.
    * When the temperature becomes large we switch to a hard max‑subgradient
      (weights are the near‑maximum indices of the convolution).
    * Learning‑rate decays exponentially, stochastic jitter and restarts
      avoid stagnation.
    * The best vector over all seeds is returned, scaled to satisfy the
      normalisation required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_from_weights(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the (soft‑max or sub‑gradient) surrogate w.r.t. a.
    ``w`` is a probability vector of length 2·n‑1 (weights for each
    convolution index). The gradient equals 2·(a * w_rev) where * is a
    linear convolution.
    """
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1            # = 2·n‑2
    return 2.0 * full[start_idx:start_idx + a.size]


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max weights (sum to 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape, normalised to a probability vector."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """Two edge blocks of length L with linear taper controlled by gamma."""
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _edge_mid_shape(n: int, L: int, gamma: float, mid_ratio: float) -> np.ndarray:
    """
    Two tapered edge blocks (length L, taper gamma) plus an interior plateau.
    The interior height is ``mid_ratio`` times the peak edge height (which is 1).
    """
    if L == 0 or 2 * L >= n:
        # fallback to pure edge shape if interior would be empty or negative
        return _edge_shape(n, L, gamma)

    # Edge block (non‑normalised)
    left = np.arange(L, dtype=np.float64)
    left_vals = 1.0 - gamma * (left / (L - 1)) if L > 1 else np.array([1.0])
    # Mirror for right side
    right_vals = left_vals[::-1]

    interior_len = n - 2 * L
    interior_vals = np.full(interior_len, max(0.0, mid_ratio), dtype=np.float64)

    v = np.concatenate([left_vals, interior_vals, right_vals])
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """
    True objective J = 2·n·max(p*p) for a probability vector ``p``.
    """
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def _adam_refine(p_start: np.ndarray,
                 n: int,
                 fft_len_self: int,
                 fft_len_grad: int,
                 time_limit: float,
                 rng: np.random.Generator) -> tuple[float, np.ndarray]:
    """
    Refine a probability vector using Adam with a two‑stage schedule:
        • 0 %–split : soft‑max surrogate (smooth)
        • split %–100 % : hard max‑subgradient (sharp)
    Returns the best objective value seen and the corresponding vector.
    """
    # Adam hyper‑parameters (fixed)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # Scheduling parameters
    split = 0.6                     # fraction of time allocated to soft‑max phase
    lr_start, lr_end = 0.5, 0.001   # learning‑rate schedule (exponential decay)
    alpha_start, alpha_end = 5.0, 3000.0   # temperature for soft‑max
    near_start, near_end = 0.5, 0.99999   # fraction of max kept in sub‑gradient mask

    a = proj_simplex(p_start.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_J = _objective(a, fft_len_self, n)
    last_improve = time.monotonic()
    start = time.monotonic()
    it = 0

    while time.monotonic() < time_limit:
        it += 1
        now = time.monotonic()
        frac = (now - start) / (time_limit - start)          # ∈[0,1]

        # --------------------------------------------------------------
        # Learning‑rate (exponential decay)
        # --------------------------------------------------------------
        # we decay the base learning‑rate exponentially inside each phase
        if frac < split:
            # soft‑max phase
            phase_frac = frac / split
            lr = lr_start * (0.5 ** phase_frac) + lr_end * (1 - 0.5 ** phase_frac)
            # temperature rises exponentially
            alpha = alpha_start * (1.0 - phase_frac) + alpha_end * phase_frac
            # compute surrogate weights
            conv = _self_conv(a, fft_len_self)
            w = _softmax_weights(conv, alpha)
        else:
            # hard‑max sub‑gradient phase
            phase_frac = (frac - split) / (1.0 - split)
            lr = lr_start * 0.1 * (0.5 ** phase_frac) + lr_end * (1 - 0.5 ** phase_frac)
            conv = _self_conv(a, fft_len_self)
            max_val = conv.max()
            near = near_start + (near_end - near_start) * phase_frac
            eps_thr = max_val * (1.0 - near)
            mask = (conv >= max_val - eps_thr).astype(float)
            if mask.sum() == 0.0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w = mask / mask.sum()

        # --------------------------------------------------------------
        # Gradient of the surrogate (soft‑max or hard‑max)
        # --------------------------------------------------------------
        grad = _grad_from_weights(a, w, fft_len_grad)

        # --------------------------------------------------------------
        # Adam update + projection onto simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # --------------------------------------------------------------
        # Evaluate true objective (no smoothing)
        # --------------------------------------------------------------
        cur_J = _objective(a, fft_len_self, n)

        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{now - start:.1f}s] new best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # Occasional jitter to escape tiny plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            block_width = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - block_width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block_width] *= factor
            a_pert = proj_simplex(a_pert)
            a = a_pert

        # --------------------------------------------------------------
        # Restart on stagnation
        # --------------------------------------------------------------
        if now - last_improve > 22.0:
            a = best_a.copy()
            # inject a little unbiased noise
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[{now - start:.1f}s] restart – best J = {best_J:.6f}")

    return best_J, best_a


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine performs a cheap global sweep over
    low‑dimensional families, keeps the top few seeds and refines each with
    a two‑stage Adam optimiser on a smoothed max‑convolution surrogate.
    The best shape found is returned after scaling to satisfy the sum
    normalisation required by the evaluation function.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep FFT cheap – cap at 4096, lower bound 256
            n = min(max(n, 256), 4096)
        except Exception:
            n = 4096
    else:
        n = 4096

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Global cheap exploration
    # --------------------------------------------------------------
    candidates = []                     # list of (J, p) tuples, J = true objective

    def _push(p_vec: np.ndarray):
        """Evaluate a probability vector and keep the best few (≤7)."""
        J = _objective(p_vec, fft_len_self, n)
        candidates.append((J, p_vec.copy()))
        candidates.sort(key=lambda x: x[0])
        if len(candidates) > 7:
            candidates.pop()

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push(uniform)

    # ---- edge‑mid family -------------------------------------------------
    for frac in np.linspace(0.15, 0.45, 13):          # block‑size fraction
        L = int(frac * n)
        if L == 0 or 2 * L >= n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            for mid in np.linspace(0.0, 1.0, 9):
                p = _edge_mid_shape(n, L, gamma, mid)
                _push(p)

    # ---- cosine family ---------------------------------------------------
    for r in np.linspace(0.0, 1.0, 13):
        for pw in np.linspace(0.3, 3.0, 9):
            p = _cosine_shape(n, r, pw)
            _push(p)

    # ---- random Dirichlet draws -----------------------------------------
    for _ in range(40):
        # concentration shape parameter between 0.05 and 0.6 (bias towards sparse)
        alpha = rng.uniform(0.05, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # ---- include previous best (if any) -----------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = uniform
            else:
                prev = proj_simplex(prev / prev.sum())
            _push(prev)
        except Exception:
            pass

    # keep the top seeds
    candidates.sort(key=lambda x: x[0])
    top_seeds = candidates[:5]  # (J, p) tuples

    best_overall_J = top_seeds[0][0]
    best_overall_p = top_seeds[0][1].copy()
    print(f"global exploration completed – best J = {best_overall_J:.6f}")

    # --------------------------------------------------------------
    # 2) Local refinement – Adam with two‑stage schedule
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # time ran out; return the current best.
        scale = np.sqrt(2.0 * n)
        best_seq = best_overall_p * scale
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return [float(x) for x in best_seq]

    # allocate ~80 % of the remaining budget to refinement
    refine_budget = remaining * 0.80
    per_seed = refine_budget / max(1, len(top_seeds))

    for seed_idx, (J0, p0) in enumerate(top_seeds):
        if time.monotonic() >= deadline:
            break
        time_limit = min(per_seed, deadline - time.monotonic())
        print(f"seed {seed_idx + 1}/{len(top_seeds)} – time budget {time_limit:.1f}s")
        J_refined, p_refined = _adam_refine(p0, n, fft_len_self, fft_len_grad,
                                            time_limit, rng)
        if J_refined < best_overall_J - 1e-12:
            best_overall_J = J_refined
            best_overall_p = p_refined.copy()
            print(f"*** new global best after seed {seed_idx + 1}: J = {best_overall_J:.6f}")

    # --------------------------------------------------------------
    # 3) Final scaling and return
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)                     # required L1‑norm for evaluation
    best_seq = best_overall_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)    # respect evaluator's clipping
    print(f"final best J = {best_overall_J:.6f} (scaled sequence returned)")
    return [float(x) for x in best_seq]
