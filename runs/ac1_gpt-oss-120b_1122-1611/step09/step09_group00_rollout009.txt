def propose_candidate(seed=..., budget_s=1000.0):
    rng = np.random.default_rng(seed)
    # Determine n
    n = 2048
    S = np.sqrt(2*n)
    # Precompute FFT length
    fft_len = _next_pow2(2*n)
    # Initialize population
    pop_size = 60
    # if global height_sequence_1 exists, include it normalized to S sum.
    # else random Dirichlet
    pop = []
    for i in range(pop_size):
        if i == 0 and 'height_sequence_1' in globals():
            a = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            a = np.clip(a, 0, None)
            if a.size != n:
                # adjust size: maybe truncation or interpolation; we can randomly sample portion
                if a.size > n:
                    a = a[:n]
                else:
                    # pad with zeros and random fill.
                    extra = np.zeros(n - a.size)
                    a = np.concatenate([a, extra])
            a = a / a.sum() * S
        else:
            # sample from exponential
            exp = rng.exponential(scale=1.0, size=n)
            a = exp / exp.sum() * S
        pop.append(a)

    # compute scores for population
    scores = np.empty(pop_size)
    for i,a in enumerate(pop):
        conv = _conv_fft(a, fft_len)
        scores[i] = conv.max() # because objective = max_b (since sum normalized)
    best_idx = np.argmin(scores)
    best = pop[best_idx].copy()
    best_score = scores[best_idx]

    print(f"[0.00s] initial best score = {best_score:.6f}")

    # Loop parameters
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    generation = 0
    # Hyperparameters
    elite_frac = 0.2
    mutation_rate = 0.5 # probability each child mutated
    # Might adjust later

    while time.monotonic() < deadline:
        # Selection: sort and keep elite
        elite_count = max(1, int(pop_size * elite_frac))
        # Get indices of sorted scores
        sorted_idx = np.argsort(scores)
        elite = [pop[i] for i in sorted_idx[:elite_count]]
        # Generate new population
        new_pop = elite.copy()
        # Fill rest using crossover + mutation
        while len(new_pop) < pop_size:
            # pick two parents randomly from elite
            p1, p2 = rng.choice(elite, size=2, replace=False)
            # Crossover: weighted average with random weight w in [0,1]
            w = rng.random()
            child = w * p1 + (1-w) * p2
            # ensure sum = S automatically (since both sums are S)
            # mutate child with probability
            if rng.random() < mutation_rate:
                child = mutate_candidate(child, rng, S)
            # enforce non-negativity (mutation may create small negative)
            child = np.clip(child, 0.0, None)
            # reproject to simplex sum S
            child = _project_onto_simplex(child) * S  # _project returns sum=1
            new_pop.append(child)

        # evaluate new population
        scores = np.empty(pop_size)
        for i,a in enumerate(new_pop):
            conv = _conv_fft(a, fft_len)
            scores[i] = conv.max()
        # update best
        gen_best_idx = np.argmin(scores)
        if scores[gen_best_idx] < best_score - 1e-12:
            best_score = scores[gen_best_idx]
            best = new_pop[gen_best_idx].copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (gen {generation}) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

"""
propose_candidate
=================

A three‑stage optimiser for the “max‑convolution” objective used by the
challenge:

1. **Exponentiated‑gradient descent** – updates the log‑weights of a
   probability vector using the exact sub‑gradient of the    max‑convolution
   (all indices that are (almost) tight are used, not only the top‑3).

2. **Projected additive sub‑gradient descent** – a classic sub‑gradient step
   with a diminishing step‑size and Euclidean projection onto the simplex.

3. **Simulated‑annealing local refinement** – tiny directed mass‑transfer
   moves are accepted either if they improve the score or with a
   temperature‑dependent probability, allowing the search to escape local
   minima.  Occasional “jitter” (random scaling of a short block) helps
   re‑stir the distribution.

The routine works on the simplex (∑a_i = 1) so that the evaluation score is
simply `2·n·max(conv)`; the final sequence is returned unchanged (the
evaluation function rescales it automatically).  If a previous construction
named ``height_sequence_1`` exists it is used as a warm‑start; otherwise a
uniform random start of length 1024 is used.

All helper functions are top‑level, there are no closures or filesystem
accesses, and the routine respects the 1000 s time budget (5 s safety
margin)."""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fall back to the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2·len(a)‑1.
    """
    a_fft = rfft(a, n=fft_len)
    conv = irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k, the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _mutate_candidate(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Small stochastic perturbation of a probability vector.
    A log‑space Gaussian perturbation is applied and the result is re‑projected
    onto the simplex.
    """
    log_a = np.log(a + 1e-12)
    sigma = 0.15                     # 15 % noise on log‑scale
    log_a += rng.normal(scale=sigma, size=a.shape)
    a_new = np.exp(log_a)
    a_new = _project_onto_simplex(a_new)   # still sums to 1
    return a_new


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Determine problem size and initialise a feasible vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        n = a0.size
        if a0.sum() == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a0 / a0.sum()
    else:
        # default to a moderate size that is fast to evaluate
        n = 1024
        a = rng.random(n)
        a = a / a.sum()

    # fft parameters (linear convolution length = 2·n‑1)
    fft_len = _next_pow2(2 * n)

    # --------------------------------------------------------------
    # 1) Book‑keeping for best solution seen so far
    # --------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * n * conv.max()            # same as evaluate_sequence for sum=1
    best = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    target_score = 1.5025

    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 1 – exponentiated‑gradient descent (≈20 % of time)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.20 * (deadline - start_time)
    it = 0
    while time.monotonic() < phase1_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-8                       # all near‑max indices are active
        active = np.where(conv >= max_b - eps)[0]
        grad = _gradient_from_active(a, active)

        # diminishing learning rate (larger at the beginning)
        eta = 1.2 / np.sqrt(it + 1.0)
        log_a = np.log(a + 1e-15)
        log_a -= eta * grad
        a = np.exp(log_a)
        a = _project_onto_simplex(a)            # stay on simplex

        # score check
        conv = _conv_fft(a, fft_len)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (exp‑grad) = {best_score:.6f} (it={it})")
            if best_score <= target_score:
                print("[early stop] target reached in phase 1")
                break
        it += 1
        if it % 500 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] phase 1 it={it}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 2 – additive projected sub‑gradient descent (≈30 % of time)
    # --------------------------------------------------------------
    a = best.copy()
    phase2_end = start_time + 0.55 * (deadline - start_time)
    it = 0
    while time.monotonic() < phase2_end:
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        eps = max_b * 1e-10
        active = np.where(conv >= max_b - eps)[0]
        grad = _gradient_from_active(a, active)

        step = 0.45 / np.sqrt(it + 1.0)
        a_candidate = a - step * grad
        a_candidate = _project_onto_simplex(a_candidate)

        conv_cand = _conv_fft(a_candidate, fft_len)
        score_cand = 2.0 * n * conv_cand.max()

        # accept if it does not worsen the objective
        if score_cand <= 2.0 * n * max_b:
            a = a_candidate

        # update global best
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (sub‑grad) = {best_score:.6f} (it={it})")
            if best_score <= target_score:
                print("[early stop] target reached in phase 2")
                break

        it += 1
        if it % 500 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] phase 2 it={it}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 3 – simulated‑annealing local refinement (remaining time)
    # --------------------------------------------------------------
    a = best.copy()
    curr_score = best_score
    T = 0.012                     # starting temperature
    decay = 0.99985               # exponential cooling
    it = 0
    while time.monotonic() < deadline:
        # propose a tiny directed mass transfer
        src = rng.integers(0, n)
        tgt = rng.integers(0, n)
        if src == tgt or a[src] <= 1e-15:
            continue
        # multiplicative random fraction in [0.001, 0.05]
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)
        cand = _project_onto_simplex(cand)

        conv_cand = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()
        Δ = cand_score - curr_score

        # Metropolis acceptance
        if Δ < 0 or rng.random() < np.exp(-Δ / max(T, 1e-12)):
            a = cand
            curr_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (SA) = {best_score:.6f} (it={it})")
                if best_score <= target_score:
                    print("[early stop] target reached in SA phase")
                    break

        # occasional jitter – random scaling of a short block
        if it % 6000 == 0 and it > 0:
            width = rng.integers(1, max(2, n // 80))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jitter = a.copy()
            a_jitter[start:start + width] *= factor
            a = _project_onto_simplex(a_jitter)

        # cooling schedule
        T *= decay
        it += 1
        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SA it={it}, T={T:.3e}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
