        start_idx = w_rev.size - 1
        grad = 2.0 * full[start_idx:start_idx+n]
        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad*grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)
        # occasional jitter
        if rng.random() < 0.005:
            a = _random_block_perturb(a, rng)
        # evaluate
        cur_J = compute_J(a, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:.2f}s] GD new best J = {best_J:.6f}")
        # restart if stagnated > 15 seconds
        if time.monotonic() - last_improve > 15.0:
            a = best.copy()
            # add small noise
            a = a * np.exp(rng.normal(0.0, 0.07, size=n))
            a = proj_simplex(a)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:.2f}s] GD restart")
    return best

        cols = []
        data = []
        for k in range(2*n-1):
            j_start = max(0, k-n+1)
            j_end = min(k, n-1)

                cols.append(j)
                data.append(a[k-j])

        c = -np.ones(n)

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds,

        if not res.success:

        a_new = proj_simplex(a_new)

"""
Propose a step‑function sequence with a low value of
evaluate_sequence().  The optimizer combines:

* a quick random Dirichlet search,
* a projected Adam optimiser on a soft‑max smooth approximation of the
  max‑convolution,
* an optional linear‑programming “mass‑inflation’’ step that tries to
  increase the total mass under the current peak bound,
* an iterative spectral‑flattening step that forces the Fourier
  magnitudes to be (almost) constant,
* a cheap stochastic swap / mass‑transfer local search.

The whole routine respects a 1000 s budget (5 s safety margin) and
always returns a non‑negative simplex‑normalised vector.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Utility functions (all top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block, then project."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def compute_J(vec: np.ndarray, n: int) -> float:
    """Return the exact objective 2·n·max(a*a).  The vector must be summed to 1."""
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def max_convolution(vec: np.ndarray, n: int) -> float:
    """Maximum entry of self‑convolution a*a (used for the LP step)."""
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(conv.max())


# ----------------------------------------------------------------------
# Phase 1 – cheap random Dirichlet sampling
# ----------------------------------------------------------------------


def random_dirichlet_search(n: int, time_budget: float, rng: np.random.Generator):
    """Return the best random Dirichlet vector found within the time budget."""
    best = np.full(n, 1.0 / n)
    best_J = compute_J(best, n)
    start = time.monotonic()
    while time.monotonic() - start < time_budget:
        # concentration parameter drawn uniformly – explores both spiky and smooth
        alpha = rng.uniform(0.3, 2.0)
        x = rng.gamma(alpha, 1.0, size=n)
        s = x.sum()
        if s == 0.0:
            continue
        x /= s
        J = compute_J(x, n)
        if J < best_J:
            best_J = J
            best = x
            print(f"[rand] new best J = {best_J:.6f}")
    return best, best_J


# ----------------------------------------------------------------------
# Phase 2 – Projected Adam on a smoothed max‑convolution
# ----------------------------------------------------------------------


def adam_sgd_phase(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Projected Adam optimisation using a soft‑max approximation of the max."""
    n = a0.size
    a = a0.copy()
    best = a.copy()
    best_J = compute_J(best, n)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    start = time.monotonic()
    last_improve = start
    it = 0

    # pre‑compute FFT lengths
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(3 * n - 2)

    while time.monotonic() - start < time_budget:
        it += 1
        now = time.monotonic()
        frac = (now - start) / time_budget  # in [0,1]

        # learning‑rate & temperature schedules
        lr = 0.5 * (1 - frac) + 0.001 * frac
        alpha = 10.0 * (1 - frac) + 5000.0 * frac

        # self‑convolution
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        cmax = conv.max()
        # soft‑max on the convolution
        shifted = conv - cmax
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed objective
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
        start_idx = w_rev.size - 1
        grad = 2.0 * full[start_idx:start_idx + n]

        # Adam update + simplex projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # occasional jitter
        if rng.random() < 0.005:
            a = _random_block_perturb(a, rng)

        cur_J = compute_J(a, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best = a.copy()
            last_improve = now
            print(f"[{now - start:6.2f}s] Adam new best J = {best_J:.6f}")

        # restart on stagnation
        if now - last_improve > 15.0:
            a = best.copy()
            a = a * np.exp(rng.normal(0.0, 0.07, size=n))
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start:6.2f}s] Adam restart (stagnation)")
    return best, best_J


# ----------------------------------------------------------------------
# Phase 3 – Linear‑programming “mass‑inflation’’ step
# ----------------------------------------------------------------------


def lp_mass_inflation_phase(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Solve a small LP that tries to increase total mass under the current M."""
    n = a0.size
    a = a0.copy()
    best = a.copy()
    best_J = compute_J(best, n)

    start = time.monotonic()
    # Build the Toeplitz (upper‑triangular) constraint matrix once per iteration
    while time.monotonic() - start < time_budget:
        M = max_convolution(a, n)

        # Build sparse matrix A where A[k, j] = a[k‑j] (zero when out of range)
        diag_vals = [np.full(n - d, a[d]) for d in range(n)]
        offsets = np.arange(0, n)
        A = sparse.diags(diag_vals, offsets, shape=(2 * n - 1, n), format="csr")

        # Linear program: maximise sum(b)  <=> minimise -sum(b)
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(
            c,
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break
        b = b / sum_b                     # normalise to a probability vector

        # Mix a little towards b (small step size)
        t = 0.2
        a_new = (1 - t) * a + t * b
        a_new = proj_simplex(a_new)

        cur_J = compute_J(a_new, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best = a_new.copy()
            a = a_new
            print(f"[LP] new best J = {best_J:.6f}")
        else:
            break   # stop if no improvement
    return best, best_J


# ----------------------------------------------------------------------
# Phase 4 – Spectral flattening (constant magnitude in Fourier domain)
# ----------------------------------------------------------------------


def spectral_flatten_phase(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Iteratively force the DFT magnitude to be (almost) constant."""
    n = a0.size
    a = a0.copy()
    best = a.copy()
    best_J = compute_J(best, n)

    start = time.monotonic()
    iter_cnt = 0
    while time.monotonic() - start < time_budget and iter_cnt < 200:
        # forward FFT
        A = np.fft.rfft(a)                     # length n//2+1
        total_power = np.sum(np.abs(A) ** 2)   # = n * sum(a_i^2)
        amp0 = np.abs(A[0])                    # DC component = sum(a) = 1
        num_bins = A.size
        # target amplitude for non‑DC frequencies (average magnitude)
        target_sq = (total_power - amp0 ** 2) / (num_bins - 1)
        target_amp = np.sqrt(max(target_sq, 0.0))

        # random phases for all non‑DC bins
        phases = np.exp(1j * rng.random(num_bins - 1) * 2.0 * np.pi)
        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC exactly
        A_new[1:] = target_amp * phases

        # inverse FFT → new candidate
        a_new = np.fft.irfft(A_new, n=n)
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full(n, 1.0 / n)
        else:
            a_new /= s

        a = a_new
        cur_J = compute_J(a, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best = a.copy()
            print(f"[spec] new best J = {best_J:.6f}")
        iter_cnt += 1
    return best, best_J


# ----------------------------------------------------------------------
# Phase 5 – Simple stochastic local‑swap search
# ----------------------------------------------------------------------


def swap_local_search_phase(a0: np.ndarray, time_budget: float, rng: np.random.Generator):
    """Swap (or small mass‑transfer) moves with a tiny uphill probability."""
    n = a0.size
    cur = a0.copy()
    cur_J = compute_J(cur, n)
    best = cur.copy()
    best_J = cur_J

    start = time.monotonic()
    attempts = 0
    accepts = 0
    temp = 1e-4

    while time.monotonic() - start < time_budget:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose exchanging the two values
        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]

        cand_J = compute_J(cand, n)
        attempts += 1
        if cand_J < cur_J - 1e-12:
            cur, cur_J = cand, cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best, best_J = cand.copy(), cand_J
                print(f"[swap] new best J = {best_J:.6f}")
        else:
            if rng.random() < np.exp(-(cand_J - cur_J) / temp):
                cur, cur_J = cand, cand_J
                accepts += 1

        # occasional larger perturbation to escape deep minima
        if attempts % 20000 == 0:
            cur = _random_block_perturb(cur, rng)
            cur = proj_simplex(cur)

    print(f"Swap phase finished – attempts: {attempts}, accepts: {accepts}")
    return best, best_J


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine uses several complementary heuristics
    and respects the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine problem dimension
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)
            n = base.size
            s = base.sum()
            if s < 1e-12:
                base = np.full(n, 1.0 / n)
            else:
                base = base / s
            init = base.copy()
            print(f"[init] Using provided sequence (len={n})")
        except Exception:
            n = 3809
            init = np.full(n, 1.0 / n)
            print(f"[init] Provided sequence unusable – using uniform (len={n})")
    else:
        n = 3809
        init = np.full(n, 1.0 / n)
        print(f"[init] No seed – using uniform (len={n})")

    # --------------------------------------------------------------
    # 2) Random Dirichlet search (≈5 % of the budget)
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    rand_time = min(remaining * 0.05, 30.0)
    best_vec, best_J = random_dirichlet_search(n, rand_time, rng)

    # --------------------------------------------------------------
    # 3) Projected Adam optimisation (≈60 % of the remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        adam_time = remaining * 0.60
        print(f"[phase] Adam optimisation – budget {adam_time:.2f}s")
        best_vec, best_J = adam_sgd_phase(best_vec, adam_time, rng)

    # --------------------------------------------------------------
    # 4) Linear‑programming mass‑inflation step (if time permits)
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 20.0:
        lp_time = remaining * 0.20
        print(f"[phase] LP mass‑inflation – budget {lp_time:.2f}s")
        best_vec, best_J = lp_mass_inflation_phase(best_vec, lp_time, rng)

    # --------------------------------------------------------------
    # 5) Spectral flattening (≈15 % of the remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 10.0:
        spec_time = remaining * 0.30
        print(f"[phase] Spectral flattening – budget {spec_time:.2f}s")
        best_vec, best_J = spectral_flatten_phase(best_vec, spec_time, rng)

    # --------------------------------------------------------------
    # 6) Final stochastic swap / local‑search phase (use the rest)
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        print(f"[phase] Swap local‑search – budget {remaining:.2f}s")
        best_vec, best_J = swap_local_search_phase(best_vec, remaining, rng)

    print(f"[final] Best J found = {best_J:.6f}")

    # --------------------------------------------------------------
    # 7) Clip to the limits imposed by evaluate_sequence and return
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
