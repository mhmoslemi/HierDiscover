"""
Propose a step‑function (non‑negative sequence) that (approximately) minimises
``evaluate_sequence``.

Algorithm
---------

1. **Problem size** – Use the length of the previously found sequence
   ``height_sequence_1`` when it exists, otherwise a default of 4096.
   The sequence is always normalised to sum = 1 (the objective is scale‑invariant).

2. **Initial portfolio** – Build a diverse set of candidate vectors coming from
   a few simple parametric families (uniform, geometric, quadratic “U‑shape”,
   inverse‑quadratic, symmetric exponentials, power‑laws) together with a few
   Dirichlet draws and the previously known construction (interpolated if needed).
   The best candidate (lowest exact objective) is taken as a warm‑start.

3. **Soft‑max Adam refinement** – The true objective is
   ``J(a)=2·n·max(a*a)``.  We replace the hard max by a temperature‑controlled
   soft‑max:
                     w_k = exp(α·(c_k−c_max))/∑exp(·)
   where ``c = a*a``.
   The gradient of the surrogate is
        ∇ = 2·(a  ⋆  w_rev)                (⋆ = linear convolution)
   and is updated by Adam with projection onto the probability simplex.
   The temperature α grows from 1 to 20000 linearly in the allotted time,
   while the learning rate decays from 0.25 to 0.0005.

   Random block perturbations (multiplicative log‑normal noise on a random
   contiguous block) are injected every ≈2 % of the iterations to avoid
   stagnation.  If no improvement is seen for 15 s the current iterate is
   re‑initialised from the best vector with a small jitter.

4. **Hard‑max sub‑gradient polishing** – After the Adam phase we run a few
   iterations of true sub‑gradient descent on the hard max.  To accelerate
   convergence the gradient is formed from the *top‑K* convolution entries
   (K = 3) rather than only the single maximal one.  The step size decays
   from 0.5 to 1e‑4.  Stagnation triggers a restart from the best point.

5. **Return** – The best vector found (clipped to the limits imposed by
   ``evaluate_sequence``) is converted to a plain Python list.

All heavy work (FFT‑based convolutions) is performed with NumPy’s
``rfft``/``irfft`` which is fast enough for the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative components that sum to one).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block
    followed by simplex projection.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return proj_simplex(out)


def self_convolution(a: np.ndarray, fft_len_self: int, conv_len: int) -> np.ndarray:
    """Linear convolution a * a using an FFT of length ``fft_len_self``."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a*a) (the sum of a is assumed to be 1)."""
    conv = self_convolution(a, fft_len_self, conv_len)
    return float(2.0 * n * conv.max())

        candidates.append(v_sym)

        candidates.append(proj_simplex(prev))

    # --------------------------------------------------------------

    it = 0
    last_improve = time.monotonic()

    while time.monotonic() < adam_deadline:
        it += 1
        now_iter = time.monotonic()
        frac = (now_iter - (deadline - remaining)) / (adam_deadline - (deadline - remaining))

        # learning‑rate schedule (log‑linear decay)
        lr_start, lr_end = 0.25, 0.0005
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # temperature schedule for the soft‑max surrogate
        alpha_start, alpha_end = 1.0, 20000.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ----- convolution of a with itself -----
        conv = self_convolution(a, fft_len_self, conv_len)

        # ----- soft‑max weights over the convolution entries -----
        cmax = conv.max()
        shifted = conv - cmax                       # improves numerical stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate (2·a⋆w_rev) -----
        w_rev = w[::-1]                             # reverse for linear convolution
        fa = np.fft.rfft(a, n=fft_len_self)
        fw = np.fft.rfft(w_rev, n=fft_len_self)
        full = np.fft.irfft(fa * fw, n=fft_len_self)
        start_idx = w_rev.size - 1                 # index aligning the linear convolution
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- Adam update with simplex projection -----
        m = beta1 * m + (1.0 - beta1) * grad
        v_mom = beta2 * v_mom + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_mom / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # ----- evaluate true objective -----
        cur_score = compute_J(a, n, fft_len_self, conv_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = now_iter
            print(f"[{now_iter - now:6.2f}s] Adam new best J = {best_score:.6f}")

        # occasional jitter to shake the iterate
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # restart if we see no improvement for 15 s
        if now_iter - last_improve > 15.0:
            a = best_vec.copy()
            a = a * np.exp(rng.normal(0.0, 0.05, size=n))   # tiny noise
            a = proj_simplex(a)
            last_improve = now_iter
            print(f"[{now_iter - now:6.2f}s] Adam restart – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Phase‑II: Hard‑max sub‑gradient polishing (top‑K smoothing)
    # --------------------------------------------------------------
    # Allocate the rest of the budget.
    sg_deadline = deadline
    a = best_vec.copy()
    last_improve = time.monotonic()
    it = 0
    # hyper‑parameters for the sub‑gradient phase
    eta_start, eta_end = 0.5, 1e-4          # step‑size schedule
    top_k = 3                               # use the three largest convolution entries

    while time.monotonic() < sg_deadline:
        it += 1
        conv = self_convolution(a, fft_len_self, conv_len)

        # indices of the K largest entries (helps to move several peaks at once)
        if top_k == 1:
            ks = [int(conv.argmax())]
        else:
            # argpartition is O(n) and cheap for modest K
            ks = np.argpartition(conv, -top_k)[-top_k:]

        grad = np.zeros_like(a)
        for k in ks:
            idx = k - np.arange(n)
            mask = (idx >= 0) & (idx < n)
            grad[mask] += 2.0 * a[idx[mask]]

        # step‑size schedule (linear decay in absolute time)
        elapsed = time.monotonic() - now
        total = sg_deadline - now
        frac = elapsed / total
        eta = eta_start * (1.0 - frac) + eta_end * frac

        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)
        a_new = proj_simplex(a_new)

        cur_score = compute_J(a_new, n, fft_len_self, conv_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - now:6.2f}s] Sub‑grad new best J = {best_score:.6f}")

        a = a_new

        # occasional jitter
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart if stuck for more than 12 s
        if time.monotonic() - last_improve > 12.0:
            a = best_vec.copy()
            a = a * np.exp(rng.normal(0.0, 0.03, size=n))
            a = proj_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - now:6.2f}s] Sub‑grad restart – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits enforced by evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
