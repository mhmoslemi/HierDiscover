"""
Optimisation of a non‑negative step function (simplex vector) that minimises
evaluate_sequence().  The method combines

* a cheap, diverse set of initial points,
* a short differential‑evolution phase on a modest size (n≈1024) for global
  exploration,
* up‑sampling of the best small‑scale solution,
* a projected gradient descent on a temperature‑controlled soft‑max surrogate
  of the max‑convolution (the exact objective is scale‑invariant, so we keep
  the vector normalised to sum = 1),
* occasional jitter / restart to escape plateaus,
* a final pairwise “mass‑transfer’’ hill‑climb.

The whole routine respects the 1000 s wall‑clock budget (5 s safety margin)
and always returns the best vector seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra / simplex utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑d array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block, then project."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Convolution / objective helpers (exact and surrogate)
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """Exact objective J = 2·n·max(a*a) for a vector already normalised to sum = 1."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    max_b = conv.max()
    return float(2.0 * n * max_b)


def compute_conv_and_J(a: np.ndarray,
                       n: int,
                       fft_len_self: int,
                       conv_len: int):
    """Return full self‑convolution and the exact objective J."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    J = float(2.0 * n * conv.max())
    return conv, J


def softmax_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     n: int,
                     fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the temperature‑controlled soft‑max surrogate:
        G = 2·a ⋆ w   (⋆ = correlation)
    where w_k ∝ exp(alpha·(conv_k – max(conv))).
    """
    # soft‑max weights
    cmax = conv.max()
    shifted = conv - cmax
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # correlation a ⋆ w  == convolution of a with reversed w
    w_rev = w[::-1]

    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start = w_rev.size - 1                # first index that corresponds to shift 0
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The procedure respects a ~995 s wall‑clock budget
    and always returns the best vector encountered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    # Small dimension for the global‐exploration stage
    n_small = 1024
    # Large dimension for the final refinement (up‑sampled)
    n_large = 4096                       # can be increased to 8192 if time permits

    # If a previous construction exists, try to respect its length
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n_prev = len(prev)
            # If previous length is between our two scales we use it as the large size
            if 512 <= n_prev <= 8192:
                n_large = max(min(n_prev, 8192), 512)
        except Exception:
            pass

    # ------------------------------------------------------------------
    # Helper to evaluate J fast (normalised vectors)
    # ------------------------------------------------------------------
    def _eval_J(v: np.ndarray, n_: int) -> float:
        fft_len = _next_pow2(2 * n_ - 1)
        return compute_J(v, n_, fft_len, 2 * n_ - 1)

    # ------------------------------------------------------------------
    # 2) Differential‑Evolution on the small scale (global exploration)
    # ------------------------------------------------------------------
    n = n_small
    conv_len_small = 2 * n - 1
    fft_len_self_small = _next_pow2(conv_len_small)
    popsize = 40
    # ---- initialise a diverse portfolio --------------------------------
    base = []

    # uniform
    base.append(np.full(n, 1.0 / n, dtype=np.float64))

    # several exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        base.append(v)

    # linear ramp
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    base.append(v)

    # Dirichlet draws (moderately concentrated)
    for _ in range(6):
        base.append(rng.dirichlet(np.ones(n) * 0.4))

    # if a previous construction is present, include a trimmed version
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if len(old) >= n:
                trimmed = old[:n]
            else:
                # pad with zeros and renormalise
                trimmed = np.concatenate([old,
                                          np.zeros(n - len(old), dtype=np.float64)])
            trimmed = np.clip(trimmed, 0.0, None)
            trimmed /= trimmed.sum()
            base.append(trimmed)
        except Exception:
            pass

    # fill up to popsize
    while len(base) < popsize:
        base.append(rng.dirichlet(np.ones(n) * 0.3))

    idxs = rng.choice(len(base), size=popsize, replace=False)
    population = [proj_simplex(base[i].astype(np.float64)) for i in idxs]
    scores = [_eval_J(v, n) for v in population]

    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]

    # DE hyper‑parameters
    F = 0.85
    CR = 0.9

    # allocate ~40 % of the total budget to DE on the small problem
    de_deadline = start_time + 0.40 * (deadline - start_time)

    while time.monotonic() < de_deadline:
        for i in range(popsize):
            # three distinct donors
            donors = rng.choice([j for j in range(popsize) if j != i],
                                size=3, replace=False)
            r1, r2, r3 = donors
            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = np.clip(mutant, 0.0, None)
            s = mutant.sum()
            if s == 0.0:
                mutant[:] = 1.0 / n
            else:
                mutant /= s

            # binomial crossover
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(n)] = True
            trial = np.where(mask, mutant, population[i])
            trial = proj_simplex(trial)

            trial_score = _eval_J(trial, n)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] DE new best J (small) = {best_score:.6f}")

        # safety exit
        if time.monotonic() >= de_deadline:
            break

    # --------------------------------------------------------------
    # 3) Upsample to the large scale and initialise large‑scale optimisation
    # --------------------------------------------------------------
    n = n_large
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(3 * n - 2)

    # linear interpolation of the best small vector
    x_small = np.linspace(0.0, 1.0, len(best_vec), endpoint=True)
    x_large = np.linspace(0.0, 1.0, n, endpoint=True)
    a = np.interp(x_large, x_small, best_vec)
    a = np.clip(a, 0.0, None)
    a = proj_simplex(a)

    # evaluate the exact objective on the large scale
    cur_J = compute_J(a, n, fft_len_self, conv_len)
    best_vec = a.copy()
    best_score = cur_J
    last_improve = time.monotonic()

    # --------------------------------------------------------------
    # 4) Projected gradient descent on a smooth surrogate (soft‑max)
    # --------------------------------------------------------------
    it = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # schedule parameters
    alpha_start, alpha_end = 2.0, 2e5          # low → very high temperature
    lr_start, lr_end = 0.45, 0.001

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)   # ∈[0,1]

        # temperature (exponential schedule)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        # current convolution and true J (needed for restart decisions)
        conv, cur_J = compute_conv_and_J(a, n, fft_len_self, conv_len)

        # gradient of the soft‑max surrogate
        grad = softmax_gradient(a, conv, alpha, n, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # projected step
        a_candidate = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_candidate = proj_simplex(a_candidate)

        cand_J = compute_J(a_candidate, n, fft_len_self, conv_len)

        if cand_J < best_score - 1e-13:
            best_score = cand_J
            best_vec = a_candidate.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Adam improvement J = {best_score:.6f}")

        a = a_candidate

        # occasional jitter (helps to escape flat plateaus)
        if rng.random() < 0.018:
            a = _random_block_perturb(a, rng)

        # restart if stagnation for ~25 s
        if now - last_improve > 25.0:
            a = best_vec.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart after stagnation – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Fine‑grained pairwise hill‑climbing (local polishing)
    # --------------------------------------------------------------
    a = best_vec.copy()
    cur_score = best_score
    hill_start = time.monotonic()
    while time.monotonic() < deadline and (time.monotonic() - hill_start) < 9.0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(a[i], 0.004)   # modest step size
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = proj_simplex(cand)

        cand_score = compute_J(cand, n, fft_len_self, conv_len)
        if cand_score < cur_score - 1e-13:
            a = cand
            cur_score = cand_score
            best_vec = cand.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] hill‑climb improvement J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clipped to evaluate_sequence limits)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
