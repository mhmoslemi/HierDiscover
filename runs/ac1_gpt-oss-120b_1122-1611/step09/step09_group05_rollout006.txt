            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # compute subgradient from max indices
        max_idxs = np.where(conv >= max_val - 1e-12)[0]
        # compute gradient
        g = np.zeros_like(cur_vec)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            seg = cur_vec[lo:hi+1][::-1]
            g[lo:hi+1] += 2.0 * seg

        # exponentiated update

        # avoid all zeros

            new_vec = np.full_like(cur_vec, 1.0/n)

        # compute new_J

        new_J = 2.0 * n * conv_new.max()

        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = new_vec
            cur_J = new_J

        # occasional random pair move (explore)
        if iteration % 500 == 0:
            # propose pair move
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            donor = cur_vec[i]
            if donor > 0:
                eps = donor * rng.uniform(0.0, 0.05)
                cand_pair = cur_vec.copy()
                cand_pair[i] -= eps
                cand_pair[j] += eps
                cand_pair = proj_simplex(cand_pair)
                # evaluate
                A_c = np.fft.rfft(cand_pair, n=fft_len)
                conv_c = np.fft.irfft(A_c * A_c, n=fft_len)[:conv_len]
                J_c = 2.0 * n * conv_c.max()
                if J_c < cur_J or rng.random() < prob:
                    cur_vec = cand_pair
                    cur_J = J_c

        # stagnation handling
        if now - last_improve > 30.0:
            # jitter best solution
            cur_vec = best_vec * np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec /= cur_vec.sum()
            # recalc J
            A = np.fft.rfft(cur_vec, n=fft_len)
            conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
            cur_J = 2.0 * n * conv.max()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        iteration += 1

    # final best vector
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec = np.full(n, 1.0/n)
    else:
        best_vec = best_vec / s

    return [float(x) for x in best_vec]

"""
Propose a step‐function that minimizes
    J(a) = 2·n·max(conv(a,a)) / (∑a)²
using a two‑phase strategy:

1.  A cheap parametric scan (symmetric exponential / power‑law families)
    gives a strong starting point.
2.  A mirror‑descent / sub‑gradient annealing loop refines the vector.
    The sub‑gradient of   max(conv)   is known analytically:
        ∂max(conv)/∂a_i = 2·a_{k*‑i}  (k* = any index attaining the maximum).
    An exponentiated‑gradient step (which stays on the simplex) is applied,
    combined with a Metropolis acceptance rule and occasional random “pair‑move”
    perturbations to escape plateaus.

The routine respects the 1 000 s time budget, prints progress
and finally returns a list of non‑negative heights ready for
`evaluate_sequence`.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, n: int, conv_len: int, fft_len: int) -> float:
    """Return max(conv(a,a)) using an FFT of length `fft_len`."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _compute_J(a: np.ndarray, n: int, conv_len: int, fft_len: int) -> float:
    """Exact objective J = 2·n·max(conv) (∑a is assumed = 1)."""
    return float(2.0 * n * _max_convolution(a, n, conv_len, fft_len))


def _sym_exponential(beta: float, n: int) -> np.ndarray:
    """Symmetric exponential weight: w_i = exp(-beta * min(i, n‑1‑i))."""
    d = np.minimum(np.arange(n, dtype=np.float64),
                   n - 1 - np.arange(n, dtype=np.float64))
    w = np.exp(-beta * d)
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


def _sym_power(alpha: float, n: int) -> np.ndarray:
    """Symmetric power‑law weight: w_i = (min(i, n‑1‑i)+1)^{-alpha}."""
    d = np.minimum(np.arange(n, dtype=np.float64),
                   n - 1 - np.arange(n, dtype=np.float64))
    w = (d + 1.0) ** (-alpha)
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``. The method first scans a few parametric families
    (symmetric exponential and power‑law), then refines the best candidate
    with a mirror‑descent / sub‑gradient annealing loop.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. Problem size – keep it manageable for FFT based work.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(1024, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 1. Quick parametric scan (exponential & power‑law families).
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    # ---- symmetric exponential -------------------------------------------------
    betas = np.linspace(0.0, 8.0, 81)                # coarse grid
    for beta in betas:
        vec = _sym_exponential(beta, n)
        J = _compute_J(vec, n, conv_len, fft_len)
        if J < best_J:
            best_J, best_vec = J, vec.copy()

    # refine the best beta with a few golden‑section steps
    if best_vec is not None:
        # locate the beta that gave the current best
        best_beta = betas[np.argmin([_compute_J(_sym_exponential(b, n), n,
                                                conv_len, fft_len) for b in betas])]
        lo, hi = max(0.0, best_beta - 1.0), best_beta + 1.0
        phi = (np.sqrt(5.0) - 1.0) / 2.0
        for _ in range(30):
            m1 = lo + (1.0 - phi) * (hi - lo)
            m2 = lo + phi * (hi - lo)
            v1, v2 = _sym_exponential(m1, n), _sym_exponential(m2, n)
            J1, J2 = _compute_J(v1, n, conv_len, fft_len), _compute_J(v2, n, conv_len, fft_len)
            if J1 < J2:
                hi = m2
                if J1 < best_J:
                    best_J, best_vec = J1, v1.copy()
            else:
                lo = m1
                if J2 < best_J:
                    best_J, best_vec = J2, v2.copy()

    # ---- symmetric power‑law --------------------------------------------------
    alphas = np.linspace(0.0, 4.0, 41)
    for alpha in alphas:
        vec = _sym_power(alpha, n)
        J = _compute_J(vec, n, conv_len, fft_len)
        if J < best_J:
            best_J, best_vec = J, vec.copy()

    # ------------------------------------------------------------------
    # 2. Sub‑gradient / mirror descent loop (annealed).
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # temperature schedule for Metropolis acceptance
    T_start = best_J * 0.05
    T_end = 1e-8

    # step‑size schedule (mirror‑descent, exponentiated update)
    eta0 = 0.5

    last_improve = start_time
    iteration = 0

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # ---- current convolution & max index --------------------------------
        A = np.fft.rfft(cur_vec, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_val = float(conv.max())
        cur_J = float(2.0 * n * max_val)

        # keep global best
        if cur_J < best_J - 1e-12:
            best_J, best_vec = cur_J, cur_vec.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- sub‑gradient of the max‑convolution ---------------------------
        max_idxs = np.where(conv >= max_val - 1e-12)[0]   # all active indices
        g = np.zeros_like(cur_vec)
        for k in max_idxs:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # segment of cur_vec that contributes to derivative for this k
            seg = cur_vec[lo:hi + 1][::-1]               # reversed slice
            g[lo:hi + 1] += 2.0 * seg

        # ---- mirror‑descent exponentiated step ----------------------------
        eta = eta0 / np.sqrt(iteration + 1.0)
        new_vec = cur_vec * np.exp(-eta * g)            # stay positive
        s = new_vec.sum()
        if s == 0.0:
            new_vec = np.full_like(cur_vec, 1.0 / n)
        else:
            new_vec /= s

        # ---- evaluate proposed vector ---------------------------------------
        A_new = np.fft.rfft(new_vec, n=fft_len)
        conv_new = np.fft.irfft(A_new * A_new, n=fft_len)[:conv_len]
        new_J = float(2.0 * n * conv_new.max())

        # ---- Metropolis acceptance -----------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True
        if accept:
            cur_vec = new_vec
            cur_J = new_J

        # ---- occasional random “pair” move (helps escape) -----------------
        if iteration % 500 == 0:
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            donor = cur_vec[i]
            if donor > 0.0:
                eps = donor * rng.uniform(0.0, 0.05)
                cand = cur_vec.copy()
                cand[i] -= eps
                cand[j] += eps
                cand = proj_simplex(cand)
                A_c = np.fft.rfft(cand, n=fft_len)
                conv_c = np.fft.irfft(A_c * A_c, n=fft_len)[:conv_len]
                J_c = float(2.0 * n * conv_c.max())
                if J_c < cur_J or rng.random() < prob:
                    cur_vec = cand
                    cur_J = J_c

        # ---- stagnation handling -------------------------------------------
        if now - last_improve > 30.0:
            # re‑start from the best with a small Gaussian jitter
            cur_vec = best_vec * np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec /= cur_vec.sum()
            # recompute its objective
            A = np.fft.rfft(cur_vec, n=fft_len)
            conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
            cur_J = float(2.0 * n * conv.max())
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Return the best vector (post‑processed for safety)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(x) for x in best_vec]
