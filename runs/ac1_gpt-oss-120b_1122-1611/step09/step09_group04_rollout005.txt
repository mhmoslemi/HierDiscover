"""
Propose a high‑quality step‑function (probability vector) that minimises
```
evaluate_sequence```.
The algorithm consists of three stages:

1. **Broad global sweep** – generate a diverse set of candidate
   probability vectors from several low‑dimensional families
   (edge blocks, tapered blocks, exponential edges, cosine families,
   random Dirichlet draws, sparse spike patterns, and the previous best
   if available).  The five (or ten) best candidates are kept.

2. **Projected Adam on a soft‑max surrogate** – each seed is refined
   using Adam on the smooth approximation of the max‑convolution.
   The temperature is increased exponentially from a smooth regime
   (α≈1) to a very sharp one (α≈5 000) so that the surrogate converges
   to the true max.  Learning‑rate follows an exponential decay.
   The iterates are always projected onto the probability simplex.

3. **Sub‑gradient polishing** – after the Adam phase the exact
   max‑convolution is known.  A sub‑gradient of the true max‑value is
   computed analytically (shift of the current vector) and a
   decreasing‑step‑size projected gradient descent is performed to
   polish the solution and eliminate the remaining smoothing bias.

The whole search respects the supplied time budget (default 1000 s) and
returns the best vector found, scaled to satisfy the original
normalisation `∑ a_i = √(2 n)`.  All numbers are non‑negative and clipped
to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# BASIC linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a pre‑allocated FFT length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max (weights sum to 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate w.r.t. a."""
    # w_rev is the reversed weight vector (convolution reversal)
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(fa * fw, n=fft_len)
    start = w_rev.size - 1                     # = 2n‑2
    return 2.0 * full[start:start + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """Exact objective J = 2·n·max(p⋆p) for a probability vector p."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())

# Shape families used in the global sweep

def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L with linear taper controlled by gamma.
    gamma=0 ⇒ flat blocks, gamma=1 ⇒ tapered to 0 at the inner border.
    """
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return v / s


def _ratio_edge_shape(n: int, L: int, ratio: float) -> np.ndarray:
    """
    Two edge blocks of length L, interior of length n‑2·L.
    Edge height = ratio * interior height.
    """
    interior = n - 2 * L
    if interior < 0:
        return np.full(n, 1.0 / n)
    # Let interior height = h, then edge height = r·h.
    # Total mass = 2·L·r·h + interior·h = 1  ⇒  h = 1 / (interior + 2·L·r)
    h_int = 1.0 / (interior + 2.0 * L * ratio)
    h_edge = ratio * h_int
    v = np.full(n, h_int, dtype=np.float64)
    if L > 0:
        v[:L] = h_edge
        v[-L:] = h_edge
    return v / v.sum()


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape (1+r·cosθ)^power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _exp_edge_shape(n: int, lam: float) -> np.ndarray:
    """Exponential decay from both ends, symmetric."""
    i = np.arange(n, dtype=np.float64)
    v = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
    return v / v.sum()


def _sparse_spike_shape(rng: np.random.Generator, n: int, k: int) -> np.ndarray:
    """k equally‑weighted spikes placed at random positions."""
    if k <= 0 or k > n:
        return np.full(n, 1.0 / n)
    idx = rng.choice(n, size=k, replace=False)
    v = np.zeros(n, dtype=np.float64)
    v[idx] = 1.0 / k
    return v

def _adam_refine(p_start: np.ndarray,
                 fft_len_self: int,
                 fft_len_grad: int,
                 n: int,
                 time_limit: float,
                 rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the soft‑max surrogate.
    Returns (best_vector, best_J) found within *time_limit* seconds.
    """
    a = p_start.copy()
    best_a = a.copy()
    best_J = _objective(a, fft_len_self, n)

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    it = 0
    start = time.monotonic()
    last_improve = start

    while time.monotonic() < time_limit:
        it += 1
        now = time.monotonic()
        frac = (now - start) / (time_limit - start)  # [0,1]

        # Learning‑rate: exponential decay from 0.5 → 0.001
        lr = 0.5 * (0.5 ** frac) + 0.001 * (1 - 0.5 ** frac)

        # Temperature: exponential growth from α=1 → α=5000
        alpha = 1.0 * (5000.0 / 1.0) ** frac

        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)

        grad = _grad_softmax(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            # occasional progress report
            print(f"[{now - start:.1f}s] Adam new best J={best_J:.6f}")

        # Small jitter to help escape plateaus
        if rng.random() < 0.02:
            block = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - block + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block] *= factor
            a_pert = proj_simplex(a_pert)
            a = a_pert

        # Restart on stagnation (no improvement for ~15 s)
        if now - last_improve > 15.0:
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[{now - start:.1f}s] Adam restart – best J={best_J:.6f}")

    return best_a, best_J


def _subgrad_refine(p_start: np.ndarray,
                    fft_len_self: int,
                    n: int,
                    time_limit: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *exact* max‑convolution.
    Returns the best vector and its objective encountered during the
    allocated time slice.
    """
    a = p_start.copy()
    best_a = a.copy()
    best_J = _objective(a, fft_len_self, n)

    start = time.monotonic()
    # initial step size – chosen empirically
    step0 = 0.5

    while time.monotonic() < time_limit:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / (time_limit - start)  # [0,1]

        # step decays roughly as (1‑frac)^5
        step = step0 * (0.5 ** (5 * frac))

        conv = _self_conv(a, fft_len_self)
        max_val = conv.max()
        # active set = all indices within 1e‑12 of the maximum
        active = np.where(conv >= max_val - 1e-12)[0]

        # sub‑gradient: average of the gradients of the active shifts
        grad = np.zeros_like(a)
        for k in active:
            # indices i where 0 ≤ k‑i < n  ⇔ i ∈ [max(0,k‑n+1) , min(k,n‑1)]
            i_start = max(0, k - (n - 1))
            i_end = min(k, n - 1)
            i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
            grad[i_range] += 2.0 * a[k - i_range]
        grad /= max(1, active.size)   # avoid division by zero

        a = a - step * grad
        a = proj_simplex(a)

        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_J:
            best_J = cur_J
            best_a = a.copy()
            # occasional progress report
            print(f"[{elapsed:.1f}s] Sub‑grad new best J={best_J:.6f}")

    return best_a, best_J

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine runs a cheap global
    sweep, followed by aggressive Adam refinement and a final
    sub‑gradient polishing stage, all under the provided time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 5000))   # keep FFT cheap but allow sizable n
        except Exception:
            n = 3809
    else:
        n = 3809                     # default used by the reference solution

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Global cheap exploration
    # ------------------------------------------------------------------
    best_candidates = []   # list of (J, p) tuples, keep the top K

    def _push_candidate(p_vec: np.ndarray):
        """Evaluate a candidate and keep the best K."""
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        # keep at most 12 best vectors
        if len(best_candidates) > 12:
            best_candidates.pop()

    # a) pure uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push_candidate(uniform)

    # b) systematic edge‑shape sweep
    for frac in np.linspace(0.10, 0.40, 13):          # block‑size fraction
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            p = _edge_shape(n, L, gamma)
            _push_candidate(p)

    # c) ratio‑edge families (different edge‑to‑interior heights)
    for frac in np.linspace(0.08, 0.30, 11):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for r in np.linspace(0.5, 5.0, 10):
            p = _ratio_edge_shape(n, L, r)
            _push_candidate(p)

    # d) cosine families
    for r in np.linspace(0.0, 1.0, 13):
        for pw in np.linspace(0.2, 4.0, 13):
            p = _cosine_shape(n, r, pw)
            _push_candidate(p)

    # e) exponential edge shapes
    for lam in np.logspace(-4, -1, 12):
        p = _exp_edge_shape(n, lam)
        _push_candidate(p)

    # f) random Dirichlet draws (moderately concentrated)
    for _ in range(80):
        alpha = rng.uniform(0.02, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push_candidate(p)

    # g) sparse random spikes (k = sqrt(n) ≈ 60)
    k_spike = max(2, int(np.sqrt(n)))
    for _ in range(30):
        p = _sparse_spike_shape(rng, n, k_spike)
        _push_candidate(p)

    # h) include the previous best sequence if it exists
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0:
                prev = uniform
            else:
                prev = proj_simplex(prev / prev.sum())
            _push_candidate(prev)
        except Exception:
            pass

    # keep the current global best
    best_candidates.sort(key=lambda x: x[0])
    best_J, best_p = best_candidates[0]

    # ------------------------------------------------------------------
    # 3) Local refinement – Adam + sub‑gradient polishing
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # budget exhausted – return the best we have
        scale = np.sqrt(2.0 * n)
        seq = np.clip(best_p * scale, 0.0, 1000.0)
        return [float(x) for x in seq]

    # allocate 70 % of remaining time to Adam, 30 % to sub‑gradient
    adam_budget = remaining * 0.70
    subgrad_budget = remaining - adam_budget

    per_seed_adam = adam_budget / max(1, len(best_candidates))
    per_seed_sub = subgrad_budget / max(1, len(best_candidates))

    # Run refinement on each seed
    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break

        # ---- Adam phase ----
        t_adam = min(per_seed_adam, deadline - time.monotonic())
        p_adam, J_adam = _adam_refine(p0, fft_len_self, fft_len_grad,
                                      n, t_adam, rng)

        # ---- Sub‑gradient polishing ----
        t_sub = min(per_seed_sub, deadline - time.monotonic())
        p_fin, J_fin = _subgrad_refine(p_adam, fft_len_self, n, t_sub)

        # Update global best if improved
        if J_fin < best_J:
            best_J, best_p = J_fin, p_fin.copy()
            print(f"*** New global best J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 5) Scale to the original normalisation and return
    # ------------------------------------------------------------------
    scale = np.sqrt(2.0 * n)          # required L1‑norm before evaluate_sequence
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
