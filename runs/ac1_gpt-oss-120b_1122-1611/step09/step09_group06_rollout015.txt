        # gradient via convolution a * w
        fw = np.fft.rfft(w, n=fft_len)
        conv_corr = np.fft.irfft(fa * fw, n=fft_len)
        offset = w_len - 1
        grad = 2.0 * conv_corr[offset:offset + n]   # gradient of smoothmax
        # Scale gradient by factor 2n to match objective
        grad = grad * (2 * n)  # because J = 2n * smoothmax(conv)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        # learning rate schedule: start high and decay
        lr_start = 0.5
        lr_end = 0.001
        # compute fraction of phase2 time used
        # We'll just use linear schedule based on iteration
        # but also compute time based fraction
        # We can maintain a count of iteration and use at most predetermined max it (maybe 50000)
        # We'll compute fraction based on it relative to estimated max iterations (maybe 20000)
        # Instead we use linear schedule relative to time left
        t_left = deadline - time.monotonic()

        # fraction of phase2 used:

        lr = lr_start * (1.0 - frac_phase2) + lr_end * frac_phase2
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # projection onto simplex
        a_new = proj_simplex(a_new)
        a = a_new
        # evaluate if better
        cur_score = _compute_J(a, n, _next_pow2(2*n-1), conv_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_score:.6f}")
        # occasional jitter
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)
    # return best_vec clipped
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]

"""
Smooth‑max Adam optimizer for the non‑negative step‑function problem.

The objective (after normalising the sum to 1) is
    J(a) = 2·n·max_k (a * a)[k]                     (1)

where * denotes (linear) convolution.  Equation (1) is exactly the value
returned by ``evaluate_sequence`` for a normalised vector.
The bottleneck is the non‑smooth “max”.  This implementation replaces
max by a smooth log‑sum‑exp (soft‑max) with an annealing parameter α.
The gradient of the smooth objective can be written as a *weighted*
convolution and is obtained efficiently by FFTs.  An Adam optimiser
with a linearly decreasing learning‑rate is run on the simplex using
the Euclidean projection ``proj_simplex``.  A short simulated‑annealing
phase provides a diverse starting point; stagnation triggers a tiny
random perturbation to escape shallow plateaus.

The routine respects the 1000 s budget (5 s safety margin) and returns
the best coefficient list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator):
    """Diverse seeds, all normalised to sum = 1."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in (0.99, 0.975, 0.95, 0.90, 0.85, 0.80):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # Dirichlet draws (moderately concentrated)
    for _ in range(8):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        if prev.sum() == 0.0:
            prev[:] = 1.0 / n
        else:
            prev /= prev.sum()
        cand.append(prev)

    return cand


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 12))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm consists of:

    1.  A short simulated‑annealing global phase to obtain a decent start.
    2.  A smooth‑max Adam refinement that uses the exact gradient of a
        log‑sum‑exp approximation of the max‑convolution.
    3.  Periodic tiny random perturbations to escape stagnation.

    The best vector (projected onto the simplex) found before the time
    budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) problem size
    # --------------------------------------------------------------
    # Keep a size comparable to the reference construction but allow a
    # modest increase (up to about 8000) which still keeps FFT cheap.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 4096
    n = max(512, min(n, 8192))

    conv_len = 2 * n - 1                     # length of a*a
    fft_len = _next_pow2(3 * n - 2)          # enough for a * (a*a) convolutions

    # --------------------------------------------------------------
    # 2) initialise a diverse pool of candidates and keep the best
    # --------------------------------------------------------------
    candidates = _initial_candidates(n, rng)
    best_vec = None
    best_score = np.inf

    for cand in candidates:
        cand = proj_simplex(cand.astype(np.float64))
        fa = np.fft.rfft(cand, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        score = float(2.0 * n * conv.max())
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
    print(f"[{time.monotonic() - start_time:6.2f}s] Initial best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – short simulated annealing (global search)
    # --------------------------------------------------------------
    # Allocate roughly 15 % of the remaining budget.
    time_now = time.monotonic()
    remaining = deadline - time_now
    sa_deadline = time_now + 0.15 * remaining

    a = best_vec.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    it = 0

    # linear temperature schedule
    T_start = 0.13
    T_end = 1e-4

    while time.monotonic() < sa_deadline:
        it += 1
        frac = (time.monotonic() - time_now) / (sa_deadline - time_now)
        T = T_start * (1.0 - frac) + T_end * frac

        # – simple mass‑transfer move –
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        eps = min(a[i], rng.exponential(scale=0.018))
        if eps <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s

        fa = np.fft.rfft(a_new, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        new_score = float(2.0 * n * conv.max())
        delta = new_score - cur_score

        accept = False
        if delta < 0.0:
            accept = True
        elif rng.random() < np.exp(-delta / max(T, 1e-12)):
            accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_vec = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start_time:6.2f}s] SA new best J = {best_score:.6f}")

        # – occasional Dirichlet jump –
        if rng.random() < 0.007:
            p = 0.14
            dirich = rng.dirichlet(np.ones(n) * 0.3)
            a_trial = (1.0 - p) * a + p * dirich
            a_trial = proj_simplex(a_trial)
            fa = np.fft.rfft(a_trial, n=fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
            trial_score = float(2.0 * n * conv.max())
            if trial_score < cur_score:
                a = a_trial
                cur_score = trial_score
                if trial_score < best_score - 1e-12:
                    best_score = trial_score
                    best_vec = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve - start_time:6.2f}s] SA Dirichlet jump improved J = {best_score:.6f}")

        # – stagnation restart –
        if time.monotonic() - last_improve > 14.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            fa = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
            cur_score = float(2.0 * n * conv.max())
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] SA restart (stagnation) – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – smooth‑max Adam optimisation (local refinement)
    # --------------------------------------------------------------
    time_now = time.monotonic()
    remaining = deadline - time_now
    if remaining <= 0.0:
        # budget exhausted
        best_vec = np.clip(best_vec, 0.0, 1000.0)
        return [float(x) for x in best_vec]

    a = best_vec.copy()
    m = np.zeros_like(a)          # Adam first moment
    v = np.zeros_like(a)          # Adam second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # schedules
    lr_start, lr_end = 0.46, 0.001
    alpha_start, alpha_end = 0.8, 36.0          # smooth‑max temperature
    it_adam = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        it_adam += 1

        # fraction of phase‑2 time already spent
        elapsed_phase2 = time.monotonic() - time_now
        frac_phase2 = elapsed_phase2 / remaining if remaining > 0 else 1.0

        # learning‑rate (linear decay)
        lr = lr_start * (1.0 - frac_phase2) + lr_end * frac_phase2

        # smooth‑max temperature (α grows linearly)
        alpha = alpha_start + (alpha_end - alpha_start) * frac_phase2

        # ----- convolution & smooth‑max weights -----
        fa = np.fft.rfft(a, n=fft_len)

        # self‑convolution a*a
        conv_self = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        # smooth‑max (log‑sum‑exp) weights
        conv_max = conv_self.max()
        # numeric stability: subtract the max before exponentiation
        w = np.exp(alpha * (conv_self - conv_max))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smooth‑max -----
        fw = np.fft.rfft(w, n=fft_len)
        # a * w (full convolution)
        conv_aw = np.fft.irfft(fa * fw, n=fft_len)
        offset = conv_len - 1          # index where a⋆w aligns with a[0]
        grad = 2.0 * conv_aw[offset:offset + n]   # 2·Σ_j a_j w_{i+j}
        grad = grad * (2.0 * n)                     # chain rule for J = 2n·smoothmax

        # ----- Adam update on the simplex (via Euclidean projection) -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it_adam)
        v_hat = v / (1.0 - beta2 ** it_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        # evaluate exact objective (no smoothing)
        fa_new = np.fft.rfft(a_new, n=fft_len)
        conv_exact = np.fft.irfft(fa_new * fa_new, n=fft_len)[:conv_len]
        cur_score = float(2.0 * n * conv_exact.max())

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Adam new best J = {best_score:.6f}")

        a = a_new

        # occasional tiny jitter to help leave flat regions
        if rng.random() < 0.018:
            a = _random_block_perturb(a, rng)

        # restart if stuck for a while
        if time.monotonic() - last_improve > 22.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Adam restart (stagnation) – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
