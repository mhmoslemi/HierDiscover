def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)

    # candidate lengths:
    lengths = [256, 512, 1024, 2048, 4096]
    if 'height_sequence_1' in globals():
        try:
            n0 = len(globals()['height_sequence_1'])
            if n0 >= 256 and n0 <= 4096:
                lengths.append(n0)
        except Exception:
            pass
    lengths = sorted(set(lengths))

    best_score = float('inf')
    best_vec = None
    best_n = None

    # allocate some time for shape sweep: e.g., 20% of total budget
    sweep_deadline = start + (deadline - start) * 0.2

    # shape sweep loop
    while time.monotonic() < sweep_deadline:
        # choose random n from lengths
        n = rng.choice(lengths)
        # compute fft lengths for later use (store maybe)
        conv_len = 2*n-1
        fft_len_self = 1 << ((conv_len - 1).bit_length())  # next power of 2
        # pick shape family
        shape_type = rng.integers(0, 5)
        # generate vector a (sum 1)
        if shape_type == 0:
            # uniform
            a = np.full(n, 1.0/n)
        elif shape_type == 1:
            # beta shape
            # sample logs for alpha,beta
            log_low, log_high = math.log(0.05), math.log(10.0)
            alpha = math.exp(rng.uniform(log_low, log_high))
            beta = math.exp(rng.uniform(log_low, log_high))
            x = (np.arange(n)+0.5)/n
            eps = 1e-12
            x = np.clip(x, eps, 1-eps)
            a = (x**(alpha-1)) * ((1-x)**(beta-1))
        elif shape_type == 2:
            # double exponential
            log_low, log_high = math.log(1e-4), math.log(5.0)
            gamma = math.exp(rng.uniform(log_low, log_high))
            i = np.arange(n)
            a = np.exp(-gamma*i) + np.exp(-gamma*(n-1-i))
        elif shape_type == 3:
            # power law
            p = rng.uniform(0.1, 5.0)
            i = np.arange(1, n+1)
            a = (i**(-p) + (n+1-i)**(-p))
        else:
            # Dirichlet draw
            conc = rng.uniform(0.05, 5.0)
            a = rng.dirichlet(np.ones(n) * conc)

        # ensure non-negative, clip small epsilon
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        # maybe also clip large values to avoid overflow but not needed

        # compute J
        J = compute_J(a, n, fft_len_self, conv_len)
        if J < best_score:
            best_score = J
            best_vec = a.copy()
            best_n = n
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] New best from sweep: n={n} J={best_score:.6f}")

    # If no candidate found (unlikely), fall back to uniform with n=4096
    if best_vec is None:

    # final hill climbing: small random mass exchanges but larger amplitude
    a = best_vec.copy()
    cur_score = best_score
    hill_deadline = deadline
    while time.monotonic() < hill_deadline:
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        # choose transfer amount up to maybe 5% of a[i] or 0.01
        max_transfer = min(a[i], 0.01)
        if max_transfer <= 0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta
        cand_J = compute_J(a_candidate, n, fft_len_self, conv_len)
        if cand_J < cur_score - 1e-12:
            a = a_candidate
            cur_score = cand_J
            best_vec = a_candidate.copy()
            best_score = cand_J
            print(f"[{time.monotonic() - start:6.2f}s] hillclimb improved J={best_score:.6f}")

    # Ensure clipping
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    # Convert to list of floats
    return [float(x) for x in best_vec]

"""
Optimiser for ``evaluate_sequence``.

Algorithm
---------
1. **Exploratory parametric sweep** –  Generate many candidate
   vectors from a handful of flexible families (Beta‑shaped,
   double‑exponential, power‑law, Dirichlet, uniform, linear ramp).
   All vectors are projected onto the probability simplex
   (non‑negative, sum = 1).  The true objective
   ``J = 2·n·max(a*a)`` is evaluated with a fast FFT‑based routine.
   The best candidate seen is retained.

2. **Exact sub‑gradient Adam refinement** –  For the current best
   vector compute the self‑convolution, locate the index where the
   maximum is attained and form the exact sub‑gradient of the
   objective (∂J/∂a_i = 4·n·a_{k_max‑i}).  Adam with a decaying
   learning‑rate updates the vector, each step being projected back
   onto the simplex.

3. **Local stochastic polishing** –  Random pairwise mass‑transfer
   moves (shifting a tiny amount of probability from one coordinate
   to another) are accepted when they improve the objective.
   The step size slowly decays and the process runs until the time
   budget expires.

The routine respects a 1000 s wall‑clock limit (5 s safety margin)
and always returns the best sequence found.  It works for any
previously discovered sequence ``height_sequence_1`` if present.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    then project onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Objective helpers (true objective, not the surrogate)
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """Return J = 2·n·max(a*a) for a normalised vector."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def compute_conv_and_J(a: np.ndarray,
                       n: int,
                       fft_len_self: int,
                       conv_len: int):
    """Return full self‑convolution (length 2·n‑1) and J."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    J = float(2.0 * n * conv.max())
    return conv, J

# ----------------------------------------------------------------------

                     budget_s: float = 1000.0,

    Search for a non‑negative step‑function (list of heights) that minimises

    margin) and always returns the best sequence encountered.
    """

    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    lengths = [256, 512, 1024, 2048, 4096]
    if "height_sequence_1" in globals():
        try:

    sweep_deadline = start_time + (deadline - start_time) * 0.20

    best_n = None

            alpha = math.exp(rng.uniform(log_low, log_high))
            beta = math.exp(rng.uniform(log_low, log_high))
            x = (np.arange(n) + 0.5) / n

            a = (x ** (alpha - 1)) * ((1.0 - x) ** (beta - 1))

            i = np.arange(n, dtype=np.float64)

        # Adam update

    # --------------------------------------------------------------
