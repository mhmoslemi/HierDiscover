def propose_candidate(seed=0, budget_s=1000.0):

    rng = np.random.default_rng(seed)

    # time planning
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5

    # dimension selection
    if 'height_sequence_1' in globals():
        base = np.asarray(height_sequence_1, dtype=float)
        n = len(base)
    else:
        n = 4096

    n = max(512, min(n, 5000))  # restrict

    # allocate budgets
    subgrad_budget = (budget_s - 5) * 0.7   # 70%
    local_budget = (budget_s - 5) - subgrad_budget

    # initialize a
    if 'height_sequence_1' in globals() and len(base) == n:
        a = base.copy()
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0/n)
        else:
            a = a / s
    else:
        a = np.full(n, 1.0/n)

    best = a.copy()
    best_J = compute_J(best)

    # precompute FFT lengths
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # Subgradient phase
    iter = 0
    η0 = 0.2
    last_improve = start_time

    while time.monotonic() < start_time + subgrad_budget:
        iter += 1
        # compute conv and max
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        # gather max indices
        # tolerance relative: maybe 1e-12 relative
        idxs = np.flatnonzero(conv >= max_val - 1e-12)
        k = rng.choice(idxs)
        # compute gradient
        g = np.zeros_like(a)
        # compute slice start-end
        # valid i: 0 <= i < n, 0 <= k - i < n => i in [max(0, k - (n-1)), min(k, n-1)]

        i_end = min(k, n - 1)  # inclusive; for slicing exclusive, i_end+1
        if i_start <= i_end:
            # g[i_start:i_end+1] = a[k - np.arange(i_start, i_end+1)]
            # compute reversed slice of a
            # indices j = k - i
            # we can compute j_arr = k - np.arange(i_start,i_end+1)
            # then assign

            g[i_start:i_end+1] = a[js]
        # gradient is 2*g
        g = 2.0 * g

        # step size schedule (inverse sqrt)
        elapsed = time.monotonic() - start_time
        frac = elapsed / subgrad_budget
        η = η0 / (1.0 + 10.0 * frac)  # decays to η0/11 ~ 0.018

        a = a - η * g
        # project onto simplex
        a = proj_simplex(a)

            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:.2f}s] subgrad new best J={best_J:.6f}")

    cur_J = best_J

"""
Hybrid optimizer for the step‑function max‑convolution problem.

Algorithm outline
-----------------
1️⃣  **Parametric seeding** – Randomly sample a few low‑dimensional
    families (power‑law shapes a_i ∝ (i+1)^α·(n‑i)^β) and a few Dirichlet
    draws.  The best of these seeds (few seconds) initializes the main
    search.

2️⃣  **Projected sub‑gradient descent** – The objective is
    f(a)=max_k (a*a)[k].  For a maximal index *k* the sub‑gradient is
    g_i = 2·a_{k‑i} (zero when k‑i is out of range).  We perform an
    exponentiated‑gradient update
        a_i ← a_i·exp(‑η·g_i)  and renormalise to the simplex.
    The step‑size η decays as η₀/√t.  Restarts with small multiplicative
    noise are triggered after 30 s of stagnation.

3️⃣  **Fine‑grained local search** – With the remaining time we run a
    cheap combinatorial refinement:
    * **Transfer move** – move a random fraction δ from index *i* to *j*
      (δ≤a_i).  The self‑convolution can be updated in O(n) by
      adding/subtracting the shifted copy of a; the change of the peak
      is evaluated without a full FFT.
    * **Swap move** – occasional full swaps evaluated by FFT (rare).

The routine respects the 1000 s budget (5 s safety margin), works on
any length (default capped at 4096 for speed), and returns a non‑negative
list whose evaluation value is printed during the run.  All helper
functions are defined at top level – no closures or filesystem access.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out

                    best_J = Jc
                    best_vec = cand.copy()
                    print(f"[seed] new best (dir) J={best_J:.6f}")

    # --------------------------------------------------------------
    # 1️⃣  Projected sub‑gradient descent (≈ 70 % of remaining budget)
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    subgrad_deadline = time.monotonic() + 0.7 * remaining

    a = best_vec.copy()
    best_J = compute_J(a)
    best_vec = a.copy()
    it = 0
    last_improve = time.monotonic()
    eta0 = 0.12                               # initial step size

    while time.monotonic() < subgrad_deadline:
        it += 1

        # self‑convolution and maximal index
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        max_inds = np.flatnonzero(conv >= max_val - 1e-12)
        k = rng.choice(max_inds)

        # sub‑gradient of the active max (2·a_{k‑i})
        if k < n:
            g = np.empty_like(a)
            g[:k + 1] = a[k::-1]
            g[k + 1:] = 0.0
        else:
            i_start = k - (n - 1)
            g = np.empty_like(a)
            g[:i_start] = 0.0
            g[i_start:] = a[i_start:][::-1]
        grad = 2.0 * g

        # decreasing step size
        eta = eta0 / np.sqrt(it)

        # exponentiated‑gradient update + renormalisation
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        cur_J = compute_J(a)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:.2f}s] sub‑grad new best J={best_J:.6f}")

        # stagnation handling
        if time.monotonic() - last_improve > 30.0:
            a = best_vec.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = a / a.sum()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:.2f}s] restart after stagnation")

        # occasional block perturbation to shake the iterate
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

    # --------------------------------------------------------------
    # 2️⃣  Local refinement – transfer + swap moves (remaining budget)
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    fa = np.fft.rfft(cur_vec, n=fft_len)
    conv_cur = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    cur_J = float(2.0 * n * conv_cur.max())

    best_seq = cur_vec.copy()
    best_seq_J = cur_J

    attempts = 0
    accepts = 0
    while time.monotonic() < deadline:
        attempts += 1

        if rng.random() < 0.7:
            # ----- transfer a fraction from i → j -----
            i = rng.integers(0, n)
            if cur_vec[i] <= 1e-14:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = cur_vec[i] * rng.random()          # 0 ≤ δ ≤ a_i

            # incremental convolution update (O(n))
            new_conv = conv_cur.copy()
            new_conv[i:i + n] -= 2.0 * delta * cur_vec
            new_conv[j:j + n] += 2.0 * delta * cur_vec
            new_conv[2 * i] += delta * delta
            new_conv[2 * j] += delta * delta
            new_conv[i + j] -= 2.0 * delta * delta

            new_J = 2.0 * n * new_conv.max()
            if new_J < cur_J - 1e-12:
                # accept improvement
                cur_vec[i] -= delta
                cur_vec[j] += delta
                conv_cur = new_conv
                cur_J = new_J
                accepts += 1
                if new_J < best_seq_J - 1e-12:
                    best_seq_J = new_J
                    best_seq = cur_vec.copy()
                    print(f"[{time.monotonic() - start_time:.2f}s] transfer new best J={best_seq_J:.6f}")
            else:
                # tiny uphill probability
                if rng.random() < np.exp(-(new_J - cur_J) / 1e-4):
                    cur_vec[i] -= delta
                    cur_vec[j] += delta
                    conv_cur = new_conv
                    cur_J = new_J
                    accepts += 1
        else:
            # ----- swap two entries (rare, evaluated via FFT) -----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur_vec.copy()
            cand[i], cand[j] = cand[j], cand[i]
            fa = np.fft.rfft(cand, n=fft_len)
            conv_cand = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
            cand_J = float(2.0 * n * conv_cand.max())
            if cand_J < cur_J - 1e-12:
                cur_vec = cand
                conv_cur = conv_cand
                cur_J = cand_J
                accepts += 1
                if cand_J < best_seq_J - 1e-12:
                    best_seq_J = cand_J
                    best_seq = cand.copy()
                    print(f"[{time.monotonic() - start_time:.2f}s] swap new best J={best_seq_J:.6f}")
            else:
                if rng.random() < np.exp(-(cand_J - cur_J) / 1e-4):
                    cur_vec = cand
                    conv_cur = conv_cand
                    cur_J = cand_J
                    accepts += 1

        # occasional heavy perturbation to escape deep plateaus
        if attempts % 20000 == 0:
            cur_vec = _random_block_perturb(cur_vec, rng)
            fa = np.fft.rfft(cur_vec, n=fft_len)
            conv_cur = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
            cur_J = 2.0 * n * conv_cur.max()

    print(f"Finished – best J={best_seq_J:.6f} (attempts={attempts}, accepts={accepts})")
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
