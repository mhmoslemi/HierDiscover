"""
Propose a step‑function minimising `evaluate_sequence` using
smooth‑max gradient descent (Adam) on the autocorrelation peak.
The algorithm:
  1. Initialise a probability vector a (sum = 1) – use the previous best
     if available, otherwise a uniform seed.
  2. At each iteration compute the self‑convolution c = a⋆a.
  3. Approximate max(c) by a smooth soft‑max:
        w_k ∝ exp(β·c_k)   (β grows from 0.5 to ≈50)
     The gradient of this smooth max is
        ∇ = 2· (w * a_rev)   (restricted to n entries),
     efficiently obtained by FFT.
  4. Perform an Adam step with a linearly decaying learning‑rate,
     project back onto the probability simplex.
  5. Keep the best vector seen (measured by the exact objective
     J = 2·n·max(a⋆a)).  If stagnation lasts >20 s a small random
     perturbation is injected to escape plateaus.
The routine respects a 1000 s budget (leaving a 5 s safety margin) and
returns the best coefficient list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable soft‑max with temperature 1/β."""
    y = beta * x
    m = np.max(y)
    e = np.exp(y - m)
    return e / e.sum()


def _conv_self_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Return the linear self‑convolution a⋆a (length = conv_len)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _grad_softmax(a: np.ndarray,
                  w: np.ndarray,
                  fft_len: int) -> np.ndarray:
    """
    Exact gradient of the smooth‑max objective.
    Computes 2·(w ⋆ a_rev) on the indices that correspond to shape length n.
    """
    a_rev = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fa_rev = np.fft.rfft(a_rev, n=fft_len)
    full = np.fft.irfft(fw * fa_rev, n=fft_len)
    n = a.shape[0]
    # slice n entries starting at offset (len(w)-1)//2 = n-1
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` using smooth‑max Adam optimisation.
    """
    rng = np.random.default_rng(seed)

    # ----- 0) time budget (5 s safety margin) -----
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ----- 1) problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        # reuse the length of the best known construction
        n = len(height_sequence_1)
    else:
        n = 2048                      # a reasonable default

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)                  # for a⋆a
    fft_len_grad = _next_pow2(conv_len + n - 1)          # for w⋆a_rev

    # ----- 2) initialise a -------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        if a.size != n:
            if a.size < n:
                a = np.concatenate([a, np.zeros(n - a.size)])
            else:
                a = a[:n]
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    a = proj_simplex(a)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # schedules
    lr_start, lr_end = 0.40, 5e-4           # learning‑rate
    beta_start, beta_end = 0.5, 50.0       # smoothing temperature

    best_a = a.copy()
    conv_self = _conv_self_fft(a, fft_len_self, conv_len)
    best_J = float(2.0 * n * conv_self.max())

    last_improve = start_time
    it = 0

    print(f"[{0.0:6.2f}s] initial J = {best_J:.6f}")

    # ----- 3) optimisation loop -------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)

        # learning‑rate and smoothing temperature
        lr = lr_start * (1.0 - frac) + lr_end * frac
        beta = beta_start * (1.0 - frac) + beta_end * frac

        # ---- evaluate exact objective (for logging) ----
        conv_self = _conv_self_fft(a, fft_len_self, conv_len)
        M = conv_self.max()
        J = float(2.0 * n * M)

        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] new best J = {best_J:.6f} (it {it})")

        # ---- smooth‑max weights ----
        w = _softmax(conv_self, beta)

        # ---- gradient of smooth‑max ----
        grad = _grad_softmax(a, w, fft_len_grad)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        it_float = float(it)
        m_hat = m / (1.0 - beta1 ** it_float)
        v_hat = v / (1.0 - beta2 ** it_float)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        a = a_new

        # ---- occasional stochastic jitter to escape plateaus ----
        if rng.random() < 0.015:
            # multiplicative log‑normal perturb on a random block
            width = rng.integers(1, max(2, n // 12))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a[start:start + width] *= factor
            a = np.maximum(a, 0.0)
            a = proj_simplex(a)

        # ---- stagnation restart ----
        if now - last_improve > 25.0:
            # inject a modest random Dirichlet perturbation around best
            a = best_a.copy()
            dirich = rng.dirichlet(np.ones(n) * 0.25)
            a = 0.85 * a + 0.15 * dirich
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart after stagnation (it {it})")

    # ----------------------------------------------------------------------
    # 4) Return the best sequence (clipped to evaluation limits)
    # ----------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
