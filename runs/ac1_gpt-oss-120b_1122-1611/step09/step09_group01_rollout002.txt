"""
Sub‑gradient descent from a portfolio of diverse seeds.

The objective of ``evaluate_sequence`` (with sum = 1) is

    J(a) = 2·n·max_k (a*a)[k] ,

i.e. we want to flatten the self‑convolution of a non‑negative
probability vector.  The exact sub‑gradient of the max‑convolution at
the current maximising index k∗ is simply the reversed vector a shifted
by k∗ (factor 2).  We therefore perform a projected sub‑gradient descent
with decreasing step size, starting from many different initial shapes
(uniform, linear, sinusoidal, Dirichlet draws, the previous best, …).
Occasional random perturbations and a final pair‑wise hill‑climb polish
ensure we escape plateaus.  The routine respects a 1000 s wall‑clock
budget (5 s safety margin) and always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers (FFT‑based, fully vectorised)
# ----------------------------------------------------------------------


def compute_conv_and_J(a: np.ndarray,
                       n: int,
                       fft_len: int,
                       conv_len: int):
    """
    Return full self‑convolution (first `conv_len` entries) and the
    objective J = 2·n·max(conv) for a probability vector `a`.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    maxv = conv.max()
    J = float(2.0 * n * maxv)
    return conv, J


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _initial_seeds(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A diversified portfolio of starting points, already normalised."""
    seeds = []

    # 1) uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    seeds.append(v / v.sum())

    # 3) sinusoidal (positive half‑wave)
    x = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (x + 0.5) / n)  # strictly positive
    seeds.append(s / s.sum())

    # 4) cosine‑squared (bell‑shaped)
    c = np.cos(np.pi * (x - n / 2) / n) ** 2
    seeds.append(c / c.sum())

    # 5) two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    seeds.append(v)

    # 6) half‑heavy, half‑zero (sparse)
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    seeds.append(v)

    # 7) a few Dirichlet draws (moderately sparse)
    for _ in range(12):
        seeds.append(rng.dirichlet(np.ones(n) * 0.2))

    # 8) previous best if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            prev = proj_simplex(prev)
            seeds.append(prev)
        except Exception:
            pass

    return seeds


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on max‑convolution
# ----------------------------------------------------------------------


def _sgd_maxconv(a0: np.ndarray,
                 n: int,
                 fft_len: int,
                 conv_len: int,
                 deadline: float,
                 rng: np.random.Generator,
                 best_J: float,
                 best_vec: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Run a simple sub‑gradient method starting from ``a0``.
    Returns the best vector found and its J‑value.
    """
    a = a0.copy()
    eta0 = 0.4                     # initial learning‑rate
    t = 0
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        t += 1
        # ---------- evaluate convolution ----------
        conv, J = compute_conv_and_J(a, n, fft_len, conv_len)
        if J < best_J - 1e-12:
            best_J = J
            best_vec = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SGD new best J = {best_J:.6f}")

        # ---------- sub‑gradient for the active index ----------
        k = int(np.argmax(conv))          # active shift
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        idx_i = np.arange(lo, hi + 1, dtype=np.intp)
        idx_j = k - idx_i
        grad = np.zeros_like(a)
        grad[idx_i] = 2.0 * a[idx_j]      # exact sub‑gradient of max(conv)

        # ---------- step and projection ----------
        eta = eta0 / np.sqrt(t)
        a = a - eta * grad
        a = proj_simplex(a)

        # ---------- occasional stochastic perturbation ----------
        if rng.random() < 0.012:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = proj_simplex(a * noise)

        # ---------- stagnation restart ----------
        if time.monotonic() - last_improve > 12.0:
            # re‑initialise from the current best but add a tiny jitter
            a = best_vec.copy()
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            a = proj_simplex(a * jitter)
            last_improve = time.monotonic()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] SGD restart (stagnation) – J = {best_J:.6f}")

    return best_vec, best_J


# ----------------------------------------------------------------------
# Final polishing – cheap pairwise mass transfers
# ----------------------------------------------------------------------


def _pairwise_hill_climb(a: np.ndarray,
                         n: int,
                         fft_len: int,
                         conv_len: int,
                         deadline: float,
                         rng: np.random.Generator,
                         cur_best_J: float,
                         cur_best_vec: np.ndarray) -> tuple[np.ndarray, float]:
    """Small random transfers for a few seconds."""
    cur = a.copy()
    cur_J = cur_best_J
    start = time.monotonic()
    while time.monotonic() < deadline and (time.monotonic() - start) < 12.0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(cur[i], 0.004)
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        cand = cur.copy()
        cand[i] -= delta
        cand[j] += delta
        _, cand_J = compute_conv_and_J(cand, n, fft_len, conv_len)
        if cand_J < cur_J - 1e-12:
            cur = cand
            cur_J = cand_J
            cur_best_vec = cand.copy()
            cur_best_J = cand_J
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] pairwise improve J = {cur_best_J:.6f}")
    return cur_best_vec, cur_best_J


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  Uses a portfolio of seeds,
    projected sub‑gradient descent on the exact max‑convolution,
    and a short hill‑climb polish.  The routine respects a 1000 s
    wall‑clock budget (5 s safety margin) and always returns the best
    sequence seen.
    """
    global start_time
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) time budget (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = min(max(n, 512), 4096)     # keep FFT cheap
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # length for a*a

    # --------------------------------------------------------------
    # 2) generate a diversified seed pool
    # --------------------------------------------------------------
    seeds = _initial_seeds(n, rng)

    # evaluate seeds (cheap – only one FFT each)
    best_J = float("inf")
    best_vec = None
    for s in seeds:
        _, J = compute_conv_and_J(s, n, fft_len_self, conv_len)
        if J < best_J:
            best_J = J
            best_vec = s.copy()
    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3) keep the most promising seeds (top‑k) and run SGD
    # --------------------------------------------------------------
    # sort seeds by their initial J and keep the top few
    seed_scores = [(compute_conv_and_J(s, n, fft_len_self, conv_len)[0], s) for s in seeds]
    seed_scores.sort(key=lambda x: x[0])
    top_seeds = [s for _, s in seed_scores[:5]]

    # allocate most of the remaining budget to SGD
    while time.monotonic() < deadline:
        for idx, s0 in enumerate(top_seeds):
            # run a short SGD session starting from this seed
            best_vec, best_J = _sgd_maxconv(
                a0=s0,
                n=n,
                fft_len=fft_len_self,
                conv_len=conv_len,
                deadline=deadline,
                rng=rng,
                best_J=best_J,
                best_vec=best_vec,
            )
            if time.monotonic() >= deadline:
                break
        # after a full pass we break – the remaining time is reserved for polishing
        break

    # --------------------------------------------------------------
    # 4) final cheap pairwise hill‑climbing
    # --------------------------------------------------------------
    best_vec, best_J = _pairwise_hill_climb(
        a=best_vec,
        n=n,
        fft_len=fft_len_self,
        conv_len=conv_len,
        deadline=deadline,
        rng=rng,
        cur_best_J=best_J,
        cur_best_vec=best_vec,
    )

    # --------------------------------------------------------------
    # 5) enforce limits of ``evaluate_sequence`` and return
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]
