"""
Hybrid gradient‑descent / simulated‑annealing optimiser.

The objective for a non‑negative step function a (length n) is  
    J(a) = 2·n·max(conv(a,a)) / (sum a)^2 .
Because the problem is scale‑invariant we normalise the vector to
sum = 1 and minimise the simpler quantity

    F(a) = 2·n·max(conv(a,a))   ( = J for a normalised vector).

The optimiser works on the probability simplex:
* a set of diverse initial probability vectors is generated
  (uniform, geometric decays, Dirichlet draws and the previously
  known construction if available).
* A long simulated‑annealing loop is run.  In each iteration one of
  four move types is proposed:
      – **grad**  : a projected sub‑gradient step that reduces the
        current maximal convolution entry;
      – **pair_small / pair_large** : transfer a random fraction of mass
        between two coordinates (as in the reference code);
      – **global** : a multiplicative log‑normal perturbation of all
        coordinates.
* After a move the new vector is projected back onto the simplex
  (Euclidean projection) and the true objective F(a) is evaluated
  with an FFT‑based convolution (O(n log n)).
* Acceptance follows a Metropolis rule with a linearly cooling
  temperature.  If no improvement is seen for 30 s the current state
  is re‑initialised by adding a tiny Gaussian jitter to the best
  solution seen so far (escape from plateaus).
* The loop runs until the allotted time budget (default 1000 s) is
  exhausted; the best vector encountered is returned as a plain list
  of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a using real FFT (zero‑padded to fft_len)."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _compute_F(a: np.ndarray, n: int, conv_len: int, fft_len: int) -> float:
    """
    Compute the true objective F = 2·n·max(conv(a,a)).
    The vector a is assumed to lie on the probability simplex.
    """
    conv = _conv_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of conv(a,a)[k] with respect to a.
    For a linear convolution conv_k = Σ_i a_i a_{k‑i},
    the gradient is (a_{k‑i} + a_{i‑k}) with out‑of‑range terms ignored.
    """
    n = a.shape[0]
    idx = np.arange(n)
    # term a_{k-i}
    idx1 = k - idx
    mask1 = (idx1 >= 0) & (idx1 < n)
    g = np.zeros_like(a)
    g[mask1] = a[idx1[mask1]]
    # term a_{i-k}
    idx2 = idx - k
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] += a[idx2[mask2]]
    return g

        F = _compute_F(cand, n, conv_len, fft_len)
        if F < best_F:
            best_F = F
            best_vec = cand.copy()
    print(f"[0.00s] Initial best F = {best_F:.6f}")

    # --------------------------------------------------------------
    # Simulated‑annealing / gradient descent loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # temperature schedule (linear cooling)
    T_start = best_F * 0.05   # 5 % of current best
    T_end = 1e-8

    # current state
    cur_vec = best_vec.copy()
    cur_F = best_F
    last_improve = start_time

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # ----------------------------------------------------------
        # Propose a move
        # ----------------------------------------------------------
        move_type = rng.choice(
            ["grad", "pair_small", "pair_large", "global"],
            p=[0.5, 0.2, 0.1, 0.2]
        )

        if move_type == "grad":
            # sub‑gradient step at the current maximal convolution entry
            conv = _conv_fft(cur_vec, fft_len, conv_len)
            max_val = conv.max()
            max_idxs = np.where(conv == max_val)[0]
            k = rng.choice(max_idxs)
            g = _gradient_at_max(cur_vec, k)

            # step size η is a small fraction of 1/n; adaptively scaled
            eta = (0.5 + rng.random()) / n   # randomise a little
            new_vec = proj_simplex(cur_vec - eta * g)

        elif move_type.startswith("pair"):
            # pick donor and receiver (distinct)
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

            donor_val = cur_vec[i]
            if donor_val == 0.0:
                # fallback to a simple global perturb if donor empty
                move_type = "global"
                # (fall through to global case)
            else:
                if move_type == "pair_small":
                    frac_transfer = rng.uniform(0.0, 0.05)   # ≤ 5 %
                else:  # pair_large
                    frac_transfer = rng.uniform(0.05, 0.5)   # up to 50 %
                epsilon = donor_val * frac_transfer
                new_vec = cur_vec.copy()
                new_vec[i] -= epsilon
                new_vec[j] += epsilon
                new_vec = proj_simplex(new_vec)

        if move_type == "global":
            # multiplicative log‑normal perturbation of the whole vector
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            new_vec = cur_vec * noise
            new_vec = proj_simplex(new_vec)

        # ----------------------------------------------------------
        # Evaluate the proposal
        # ----------------------------------------------------------
        new_F = _compute_F(new_vec, n, conv_len, fft_len)

        # ----------------------------------------------------------
        # Acceptance (Metropolis rule)
        # ----------------------------------------------------------
        accept = False
        if new_F < cur_F:
            accept = True
        else:
            prob = np.exp(-(new_F - cur_F) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = new_vec
            cur_F = new_F
            if new_F < best_F - 1e-12:
                best_F = new_F
                best_vec = new_vec.copy()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New best F = {best_F:.6f}")

        # ----------------------------------------------------------
        # Stagnation handling
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # add a tiny Gaussian jitter to the best solution and continue
            cur_vec = best_vec * np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = proj_simplex(cur_vec)
            cur_F = _compute_F(cur_vec, n, conv_len, fft_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best F = {best_F:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # Return the best vector (clip to the limits imposed by evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s
    return [float(x) for x in best_vec]
