import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    # Implementation similar to before
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_candidates(n: int, rng: np.random.Generator):
    """Generate a set of diverse probability vectors."""
    cand = []
    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))
    # exponential decays
    for r in [0.9, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)
    # reversed exponential (mass at end)
    for r in [0.9, 0.95, 0.975, 0.99, 0.995]:
        v = (r ** np.arange(n-1, -1, -1, dtype=np.float64))
        v /= v.sum()
        cand.append(v)
    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)
    v = np.arange(1, n+1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)
    # Dirichlet draws
    for _ in range(8):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))
        cand.append(rng.dirichlet(np.ones(n) * 1.0))
    # two‑spike (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v / v.sum())
    # optionally include previously known solution if present
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(height_sequence_1, dtype=np.float64)
            v = np.clip(v, 0.0, None)
            s = v.sum()
            if s > 0:
                v /= s
                cand.append(v)
        except Exception:
            pass
    return cand

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Gradient‑based optimisation of the step‑function objective.
    A smooth approximation of  max(conv) is minimised with
    multiplicative updates on the probability simplex.
    The routine runs for the whole time budget (default 1000 s)
    and returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. Problem size – respect the size of a previously found solution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len + n - 1)   # enough for a (n) * p (2n‑1)

    # ------------------------------------------------------------------
    # 1. Helpers for convolution / smooth‑max gradient.
    # ------------------------------------------------------------------
    def full_conv(a_vec: np.ndarray, a_fft: np.ndarray = None):
        """Linear convolution a*a (full length)."""
        if a_fft is None:
            a_fft = np.fft.rfft(a_vec, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:conv_len]
        return conv, a_fft

    def smoothmax_weights(conv: np.ndarray, alpha: float):
        """Soft‑max weighting that approximates the max."""
        max_c = conv.max()
        # subtract max for stability
        exps = np.exp(alpha * (conv - max_c))
        p = exps / exps.sum()
        return p

    def gradient_from_weights(a_fft: np.ndarray, weights: np.ndarray):
        """Gradient of the smooth‑max objective w.r.t a (vector of length n)."""
        w_fft = np.fft.rfft(weights, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[:n]
        return 2.0 * grad_full

    # ------------------------------------------------------------------
    # 2. Initialise a pool of candidates on the simplex.
    # ------------------------------------------------------------------
    init_cands = _initial_candidates(n, rng)
    best_vec = None
    best_J = np.inf

    # quick evaluation function (no security checks) – uses the same formula.
    def evaluate_J(vec: np.ndarray) -> float:
        # vec assumed to sum to 1 (or close)
        conv = np.convolve(vec, vec)
        max_conv = conv.max()
        return 2.0 * n * max_conv

    for cand in init_cands:
        J = evaluate_J(cand)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (smooth‑max gradient descent with
    #    multiplicative updates).  Restart on stagnation.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # hyper‑parameters
    lr0 = 0.8                # initial learning‑rate for multiplicative update
    alpha = 2.0              # start temperature for smooth‑max
    alpha_factor = 1.001      # slow increase per iteration
    alpha_max = 200.0
    min_lr = 1e-6

    # start from the current best
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        # compute convolution and its FFT once
        conv, a_fft = full_conv(cur_vec)

        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        # record improvement
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # smooth‑max weights
        p = smoothmax_weights(conv, alpha)

        # gradient
        grad = gradient_from_weights(a_fft, p)

        # diminishing learning rate (inverse‑sqrt schedule)
        lr = lr0 / np.sqrt(iteration + 1)
        lr = max(lr, min_lr)

        # multiplicative (exponentiated) gradient step
        factor = np.exp(- lr * grad)
        cur_vec = cur_vec * factor
        # re‑normalise onto simplex (sum‑to‑one)
        s = cur_vec.sum()
        if s <= 0.0:
            # fallback – reset to uniform
            cur_vec = np.full_like(cur_vec, 1.0 / n)
        else:
            cur_vec = cur_vec / s

        # increase temperature for smooth‑max
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # stagnation handling – if no improvement for ~20 s, jitter around the best
        now = time.monotonic()
        if now - last_improve > 20.0:
            # small Gaussian jitter in log‑space (keeps positivity)
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            cur_vec = best_vec * jitter
            cur_vec = proj_simplex(cur_vec)          # ensure exact simplex
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # occasional full random restart when we have spent a few minutes without any
        # improvement (helps to escape deep plateaus)
        if now - start_time > 300.0 and iteration % 5000 == 0:
            cur_vec = rng.dirichlet(np.ones(n) * 0.3)
            # no projection needed – Dirichlet already on simplex

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Return the best vector (clip to limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total
    return [float(x) for x in best_vec]

"""
Gradient‑based optimiser for the step‑function problem.

The objective is  J(a)=2·n·max(conv(a,a))/(∑a)².
Because the problem is scale‑invariant we keep ∑a=1 (probability simplex)
and minimise  J = 2·n·max(conv(a,a)).

The routine builds a diverse set of initial probability vectors
(uniform, several geometric decays, linear ramps, Dirichlet draws and,
if present, the previously discovered `height_sequence_1`).  For each
candidate it runs a smooth‑max gradient descent:

*  max(conv) is approximated by a soft‑max  fα(a)=∑k πk·(a*a)k  with
   πk∝exp(α·(a*a)k).  The temperature α is increased slowly,
   turning the surrogate into the true max.
*  The gradient of fα is
        ∇fα = 2·conv_full(a,π)[:n]               (linear convolution)
   and a multiplicative (exponentiated‑gradient) update
        a←a·exp(−η·∇fα)   ,   η decreasing like 1/√t,
   followed by a projection onto the simplex.
*  When no improvement is seen for ≈20 s a small Gaussian jitter is
   applied to the current best vector to escape plateaus.
*  The loop runs until the time budget (default 1000 s, 5 s safety)
   is exhausted and the best vector encountered is returned.

All heavy work is performed with FFTs (O(n log n)).  The algorithm
typically drives J well below the previous value 1.5132 and
converges to a vector that respects the non‑negativity and clipping
constraints of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a diverse portfolio of probability vectors."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (mass concentrated at the left)
    for r in [0.9, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # geometric decays reflected (mass at the right)
    for r in [0.9, 0.95, 0.975, 0.99, 0.995]:
        v = (r ** np.arange(n - 1, -1, -1, dtype=np.float64))
        v /= v.sum()
        cand.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # Dirichlet draws (moderately and lightly concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))
        cand.append(rng.dirichlet(np.ones(n) * 0.8))

    # two‑spike (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    v /= v.sum()
    cand.append(v)

    # previously known solution, if it exists
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(height_sequence_1, dtype=np.float64)
            v = np.clip(v, 0.0, None)
            s = v.sum()
            if s > 0.0:
                v /= s
                cand.append(v)
        except Exception:
            pass

    return cand


# ----------------------------------------------------------------------
# Main optimiser
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The method is a smooth‑max
    gradient descent with exponentiated‑gradient updates on the simplex.
    It runs for the whole time budget (default 1000 s) and returns the
    best vector found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0. Problem size – use a reasonable n (respect a previous solution)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1                     # length of a*a
    # FFT length must accommodate a (len n) + weights (len 2n‑1) → 3n‑2
    fft_len = _next_pow2(conv_len + n - 1)

    # --------------------------------------------------------------
    # 1. Helper functions that use the common FFT length
    # --------------------------------------------------------------
    def full_conv(a_vec: np.ndarray, a_fft: np.ndarray = None):
        """Linear convolution a*a (full length)."""
        if a_fft is None:
            a_fft = np.fft.rfft(a_vec, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:conv_len]
        return conv, a_fft

    def smoothmax_weights(conv: np.ndarray, alpha: float):
        """Soft‑max weights approximating the max."""
        max_c = conv.max()
        # stability: subtract the maximum
        exps = np.exp(alpha * (conv - max_c))
        return exps / exps.sum()

    def gradient_from_weights(a_fft: np.ndarray, w: np.ndarray):
        """Gradient of the smooth‑max objective w.r.t a (length n)."""
        w_fft = np.fft.rfft(w, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)[:n]
        return 2.0 * grad_full

    def objective_J(vec: np.ndarray) -> float:
        """Exact objective J = 2·n·max(conv(vec,vec)) (∑vec=1)."""
        conv = np.convolve(vec, vec)
        return 2.0 * n * conv.max()

    # --------------------------------------------------------------
    # 2. Initialise a pool of probability vectors
    # --------------------------------------------------------------
    init_cands = _initial_candidates(n, rng)

    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = objective_J(cand)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Main optimisation loop (smooth‑max gradient descent)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety

    # hyper‑parameters
    lr0 = 0.8                # initial learning‑rate for exponentiated update
    alpha = 2.0              # start temperature for smooth‑max
    alpha_factor = 1.001     # slow increase per iteration
    alpha_max = 200.0
    min_lr = 1e-6

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        # ----- compute convolution and objective -----
        conv, a_fft = full_conv(cur_vec)
        cur_J = 2.0 * n * conv.max()

        # ----- record improvement -----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- smooth‑max weights and gradient -----
        p = smoothmax_weights(conv, alpha)
        grad = gradient_from_weights(a_fft, p)

        # ----- multiplicative (exponentiated‑gradient) step -----
        lr = lr0 / np.sqrt(iteration + 1)
        lr = max(lr, min_lr)
        factor = np.exp(-lr * grad)               # always positive
        cur_vec = cur_vec * factor
        s = cur_vec.sum()
        if s <= 0.0:
            cur_vec = np.full_like(cur_vec, 1.0 / n)   # fallback to uniform
        else:
            cur_vec = cur_vec / s                     # projection onto simplex

        # ----- temperature schedule -----
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_factor)

        # ----- stagnation handling -----
        now = time.monotonic()
        if now - last_improve > 20.0:
            # add a tiny log‑normal jitter around the best solution
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            cur_vec = best_vec * jitter
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # occasional full random restart (helps after long runs)
        if (now - start_time) > 300.0 and iteration % 5000 == 0:
            cur_vec = rng.dirichlet(np.ones(n) * 0.3)

        iteration += 1

    # --------------------------------------------------------------
    # 4. Return the best vector (clip to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total
    return [float(x) for x in best_vec]
