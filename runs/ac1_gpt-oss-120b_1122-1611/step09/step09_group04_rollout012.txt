"""
Monotone decreasing shapes search with subgradient refinement.

Algorithm:
- Determine sequence length n (from previous best if available, else 4096).
- Enumerate a family of simple analytic probability vectors:
   * linear decreasing (triangular)
   * quadratic decreasing
   * exponential decay with various rates
   * uniform baseline
   * a few random Dirichlet draws.
- Keep the best few candidates according to the true objective (max
  self‑convolution).
- For each seed run a fast sub‑gradient descent that directly minimises
  the L∞‑norm of the convolution (i.e. the quantity returned by
  ``evaluate_sequence``).  The gradient of the maximal convolution entry
  is obtained analytically, then the iterate is projected onto the
  probability simplex.
- Occasionally inject jitter and restart on stagnation.
- The best probability vector is scaled to have ℓ₁‑norm √(2·n) and
  returned as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """True objective  J = 2·n·max(p∗p) for a probability vector p."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


def _linear_shape(n: int) -> np.ndarray:
    """Monotonically decreasing linear shape, normalised to sum = 1."""
    w = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    return w / w.sum()


def _quadratic_shape(n: int) -> np.ndarray:
    """Quadratic decreasing shape (weights (n‑i)²)."""
    i = np.arange(n, dtype=np.float64)
    w = (n - i) ** 2
    return w / w.sum()


def _exp_shape(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i (0<r<1), normalised to sum = 1."""
    if r <= 0.0 or r >= 1.0:
        raise ValueError("r must be in (0,1)")
    powers = r ** np.arange(n, dtype=np.float64)
    return powers / powers.sum()


def _grad_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the (non‑smooth) objective wrt ``a`` at the
    index where the convolution attains its maximum.
    The derivative of Σ_i a_i a_{k‑i} is 2·a_{k‑i}, zero outside the overlap.
    """
    n = a.size
    rev = a[::-1]
    offset = n - 1 - k_max
    grad = np.zeros_like(a)
    if offset >= 0:
        # overlap region: positions 0 … n‑offset‑1
        grad[: n - offset] = rev[offset:]
    else:
        # offset < 0  →  k_max > n‑1
        off = -offset
        grad[off:] = rev[: n - off]
    return 2.0 * grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.
    The routine combines a cheap global sweep over simple analytic
    families with a fast sub‑gradient descent operating on the true
    ℓ∞‑objective.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep FFT size modest – 4096 is a safe default upper bound
            n = max(256, min(n, 4096))
        except Exception:
            n = 4096
    else:
        n = 4096

    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 2) Generate a pool of inexpensive candidates
    # --------------------------------------------------------------
    best_candidates = []                     # list of (J, p) tuples

    def _push(p_vec: np.ndarray):
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        # keep only a small elite set
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > 10:
            best_candidates.pop()

    # a) uniform baseline
    _push(np.full(n, 1.0 / n, dtype=np.float64))

    # b) analytic families
    _push(_linear_shape(n))
    _push(_quadratic_shape(n))

    # c) exponential decay with a fine grid on r
    for r in np.linspace(0.85, 0.999, 12):
        _push(_exp_shape(n, r))

    # d) a handful of random Dirichlet draws (concentrated)
    for _ in range(20):
        alpha = rng.uniform(0.02, 0.3)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # e) include the previous best if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = np.full(n, 1.0 / n)
            else:
                prev = proj_simplex(prev / prev.sum())
            _push(prev)
        except Exception:
            pass

    # keep elite set sorted
    best_candidates.sort(key=lambda x: x[0])
    best_J, best_p = best_candidates[0]

    # --------------------------------------------------------------
    # 3) Sub‑gradient descent from each elite seed
    # --------------------------------------------------------------
    # Simple schedule – start with a relatively large step size and
    # decay exponentially.
    lr_start, lr_end = 0.5, 0.001
    stagnation_time = 12.0      # seconds without improvement → restart

    def _subgrad_opt(p_start: np.ndarray, time_limit: float):
        nonlocal best_J, best_p
        a = p_start.copy()
        it = 0
        last_improve = time.monotonic()
        start = time.monotonic()
        while time.monotonic() < time_limit:
            it += 1
            now = time.monotonic()
            # linear‑in‑time decay of the learning rate
            frac = (now - start) / (time_limit - start + 1e-12)
            lr = lr_start * (1.0 - frac) + lr_end * frac

            # convolution and index of the maximal entry
            conv = _self_conv(a, fft_len_self)
            k_max = int(np.argmax(conv))

            # gradient of that entry (exact)
            g = _grad_max_conv(a, k_max)

            # projected gradient step
            a -= lr * g
            a = proj_simplex(a)

            cur_J = _objective(a, fft_len_self, n)
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_p = a.copy()
                last_improve = now
                print(f"[{now - start:.1f}s] new best J={best_J:.6f}")

            # jitter if we are stuck for a while
            if now - last_improve > stagnation_time:
                # inject tiny log‑normal noise on a random block
                block = rng.integers(1, max(2, n // 30))
                start_idx = rng.integers(0, n - block + 1)
                factor = np.exp(rng.normal(0.0, 0.06))
                a_pert = a.copy()
                a_pert[start_idx:start_idx + block] *= factor
                a = proj_simplex(a_pert)
                last_improve = now
                print(f"[{now - start:.1f}s] restart (stagnation) – best J={best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Allocate time to refinement and run the optimizer
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    refine_budget = remaining * 0.85          # 85 % for refinement
    per_seed = refine_budget / max(1, len(best_candidates))

    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break
        t_limit = min(per_seed, deadline - time.monotonic())
        _subgrad_opt(p0, t_limit)

    # --------------------------------------------------------------
    # 5) Return the best sequence (scaled to the required ℓ₁‑norm)
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

"""
Monotone decreasing shapes search with sub‑gradient refinement.

Algorithm:
- Determine sequence length n (from a previous best if present, otherwise 4096).
- Enumerate a few analytic families of probability vectors:
    * uniform,
    * linear decreasing,
    * quadratic decreasing,
    * geometric (exponential) decay with several rates,
    * a handful of random Dirichlet draws.
- Keep the best few candidates according to the *true* objective
  J = 2·n·max(convolution(p,p)).
- From each elite seed run a fast sub‑gradient descent that directly
  minimises the ℓ∞‑norm of the convolution (the quantity returned by
  ``evaluate_sequence``).  The gradient of the maximal convolution entry
  is obtained analytically; after each step we project back onto the
  probability simplex.
- Occasionally inject jitter and restart on stagnation.
- The best probability vector is finally scaled to have ℓ₁‑norm √(2·n)
  (as required by the evaluator) and returned as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """True objective J = 2·n·max(p∗p) for a probability vector p."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


def _linear_shape(n: int) -> np.ndarray:
    """Monotonically decreasing linear shape, normalised to sum = 1."""
    w = np.arange(n, 0, -1, dtype=np.float64)   # n, n‑1, …, 1
    return w / w.sum()


def _quadratic_shape(n: int) -> np.ndarray:
    """Quadratic decreasing shape (weights (n‑i)²)."""
    i = np.arange(n, dtype=np.float64)
    w = (n - i) ** 2
    return w / w.sum()


def _exp_shape(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r**i for 0 < r < 1, normalised to sum = 1."""
    if not (0.0 < r < 1.0):
        raise ValueError("r must be in (0,1)")
    powers = r ** np.arange(n, dtype=np.float64)
    return powers / powers.sum()


def _grad_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Gradient of the maximal convolution entry w.r.t. ``a``.
    For conv(k) = Σ_i a_i a_{k-i} the derivative is 2·a_{k-i} on the
    overlapping region, zero elsewhere.
    """
    n = a.size
    rev = a[::-1]
    offset = n - 1 - k_max
    grad = np.zeros_like(a)
    if offset >= 0:
        # overlap region is indices 0 … n‑offset‑1
        grad[: n - offset] = rev[offset:]
    else:
        # k_max > n‑1  →  shift in the opposite direction
        off = -offset
        grad[off:] = rev[: n - off]
    return 2.0 * grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  Two phases are used:
    1) cheap enumeration of simple analytic families,
    2) fast sub‑gradient descent from the elite seeds.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 4096))   # keep FFT cheap
        except Exception:
            n = 4096
    else:
        n = 4096

    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 2) Generate a pool of inexpensive candidates
    # --------------------------------------------------------------
    best_candidates = []            # list of (J, p) tuples

    def _push(p_vec: np.ndarray):
        """Evaluate a candidate and keep only the best few."""
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > 10:
            best_candidates.pop()   # keep elite set of size 10

    # a) uniform baseline
    _push(np.full(n, 1.0 / n, dtype=np.float64))

    # b) analytic families
    _push(_linear_shape(n))
    _push(_quadratic_shape(n))

    # c) exponential decays with a grid of rates
    for r in np.linspace(0.85, 0.999, 12):
        _push(_exp_shape(n, r))

    # d) random Dirichlet draws (moderately concentrated)
    for _ in range(20):
        alpha = rng.uniform(0.02, 0.30)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # e) previous best sequence, if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = np.full(n, 1.0 / n)
            else:
                prev = proj_simplex(prev / prev.sum())
            _push(prev)
        except Exception:
            pass

    # keep the elite set sorted
    best_candidates.sort(key=lambda x: x[0])
    best_J, best_p = best_candidates[0]

    # --------------------------------------------------------------
    # 3) Sub‑gradient descent from each elite seed
    # --------------------------------------------------------------
    lr_start, lr_end = 0.5, 0.001
    stagnation_seconds = 12.0

    def _subgrad_opt(p_start: np.ndarray, time_limit: float):
        """Projected sub‑gradient descent on the true ℓ∞‑objective."""
        nonlocal best_J, best_p
        a = p_start.copy()
        last_improve = time.monotonic()
        start = time.monotonic()
        while time.monotonic() < time_limit:
            now = time.monotonic()
            frac = (now - start) / (time_limit - start + 1e-12)
            lr = lr_start * (1.0 - frac) + lr_end * frac

            conv = _self_conv(a, fft_len_self)
            k_max = int(np.argmax(conv))

            grad = _grad_max_conv(a, k_max)

            a -= lr * grad
            a = proj_simplex(a)

            cur_J = _objective(a, fft_len_self, n)
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_p = a.copy()
                last_improve = now
                print(f"[{now - start:.1f}s] new best J={best_J:.6f}")

            # restart on stagnation
            if now - last_improve > stagnation_seconds:
                block = rng.integers(1, max(2, n // 30))
                start_idx = rng.integers(0, n - block + 1)
                factor = np.exp(rng.normal(0.0, 0.06))
                a_pert = a.copy()
                a_pert[start_idx:start_idx + block] *= factor
                a = proj_simplex(a_pert)
                last_improve = now
                print(f"[{now - start:.1f}s] restart (stagnation) – best J={best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Allocate time for refinement and run the optimizer
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    refine_budget = remaining * 0.85          # ~85 % for refinement
    per_seed = refine_budget / max(1, len(best_candidates))

    for _, seed_p in best_candidates:
        if time.monotonic() >= deadline:
            break
        limit = min(per_seed, deadline - time.monotonic())
        _subgrad_opt(seed_p, limit)

    # --------------------------------------------------------------
    # 5) Scale to the ℓ₁‑norm required by ``evaluate_sequence`` and return
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)      # required ℓ₁‑norm
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
