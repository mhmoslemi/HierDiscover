"""
Hybrid optimization for the “max‑self‑convolution” objective.

Algorithm
---------
1. Problem size – Use the length of the previously found sequence
   (``height_sequence_1``) if it exists, otherwise default to 4096.
2. Diverse initial pool – uniform, several exponentials, an inverse‑sqrt
   shape, a “central‑binomial” (≈ arcsine) shape and a handful of Dirichlet
   draws.  All vectors are projected onto the probability simplex.
3. Quick screening – Evaluate every candidate with an FFT based exact
   objective and keep the best one.
4. Sub‑gradient/Adam phase – Repeatedly compute the exact max‑convolution,
   form the active‑set (all indices attaining the current maximum) and
   obtain the true sub‑gradient:
        g = 2·a * (active_mask)ᵗ   (implemented via an FFT convolution)
   An Adam‑style update with a linearly decreasing learning rate is used,
   followed by projection onto the simplex.
   Small multiplicative block perturbations and a restart on stagnation
   keep the search from getting trapped.
5. Pair‑wise hill‑climbing – A cheap local search swaps a tiny amount of
   mass between two random positions whenever it improves the true
   objective.
6. Time‑budget – The routine respects a 1000 s wall‑clock budget (leaving a
   5 s safety margin) and always returns the best sequence seen.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block, then simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return proj_simplex(out)


# ----------------------------------------------------------------------
# Objective helpers (exact, no surrogates)
# ----------------------------------------------------------------------


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Self‑convolution of a using a pre‑computed FFT length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def compute_J_fast(a: np.ndarray,
                   n: int,
                   fft_len: int,
                   conv_len: int) -> float:
    """Exact objective J = 2·n·max(a*a)/(sum(a)**2)."""
    conv = _conv_fft(a, fft_len, conv_len)
    maxv = conv.max()
    s = a.sum()
    if s == 0.0:
        return np.inf
    return float(2.0 * n * maxv / (s * s))


def active_subgradient(a: np.ndarray,
                       active_mask: np.ndarray,
                       fft_len_grad: int,
                       conv_len: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)_k for the given active set.
    Returns a vector of length n.
    """
    # reversed mask
    mask_rev = active_mask[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(mask_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    # slice that corresponds to indices 0..n‑1
    grad = 2.0 * full[conv_len - 1:conv_len - 1 + a.size]
    return grad

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` under a 1000 s wall‑clock budget.
    Returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(512, min(n, 8192))        # keep FFTs cheap
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for gradient

    # --------------------------------------------------------------
    # 2) Generate diverse initial candidates (already on the simplex)
    # --------------------------------------------------------------
    candidates = []

    # uniform

    # exponential decays
    for r in [0.999, 0.997, 0.995, 0.99, 0.975, 0.95, 0.9, 0.85]:

    # --------------------------------------------------------------
    a = best_vec.copy()

    v_acc = np.zeros_like(a)        # second moment

    it = 0

        active_mask[active_idx] = 1.0

        grad = active_subgradient(a, active_mask, fft_len_grad, conv_len)

        v_acc = beta2 * v_acc + (1.0 - beta2) * (grad * grad)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
