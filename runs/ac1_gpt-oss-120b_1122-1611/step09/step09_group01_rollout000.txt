def beta_distribution(alpha: float, n: int) -> np.ndarray:
    # symmetric beta(alpha, alpha) on (0,1)
    # x_i are midpoints of bins
    i = np.arange(n, dtype=np.float64)

        return np.full(n, 1.0 / n, dtype=np.float64)

"""
Propose a non‑negative step function that attains a very low value of
``evaluate_sequence``.

The construction is based on the symmetric Beta(½,½) (arcsine) density.
Its continuous self‑convolution is uniform, which makes the discrete
self‑convolution almost flat; consequently the objective
   J = 2·n·max(a*a) / (sum a)^2
approaches the theoretical lower bound J≈1.  The routine

* sweeps a few shape parameters α in the family Beta(α,α) (including the
  optimal α=0.5),
* evaluates each candidate with a fast FFT‑based convolution,
* applies a few cheap random‑block perturbations,
* finishes with a short Adam‑style optimisation on a soft‑max surrogate,
* always respects the 1000 s wall‑clock budget.

The returned list is clipped to the limits imposed by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    then project onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = math.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Objective helpers (FFT based, no closures)
# ----------------------------------------------------------------------


def compute_conv(a: np.ndarray,
                 fft_len: int,
                 conv_len: int) -> np.ndarray:
    """Full self‑convolution (first conv_len entries)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray,
              n: int,
              fft_len: int,
              conv_len: int) -> float:
    """Return the true objective J = 2·n·max(a*a)."""
    conv = compute_conv(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def softmax_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     alpha_tmp: float,
                     n_tmp: int,
                     fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the temperature‑controlled soft‑max surrogate:
    g = 2·a * w_rev,   where w is the soft‑max over the convolution values.
    """
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha_tmp * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    w_rev = w[::-1]

    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start = w_rev.size - 1                 # = 2·n‑2
    grad = 2.0 * full[start:start + n_tmp]
    return grad

# Parametric family – symmetric Beta(α,α)
# ----------------------------------------------------------------------


def beta_distribution(alpha: float, n: int) -> np.ndarray:
    """
    Return a discretised symmetric Beta(α,α) density on n points,
    normalised to sum = 1.  Mid‑point rule is used (x_i = (i+0.5)/n).
    """
    if alpha <= 0.0:
        raise ValueError("alpha must be positive")
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n                     # points in (0,1)

    # log‑normalisation constant to avoid overflow:
    #   C = Gamma(2α) / Gamma(α)^2
    logC = math.lgamma(2.0 * alpha) - 2.0 * math.lgamma(alpha)

    # log‑density (finite because x∈(0,1))
    logdens = logC + (alpha - 1.0) * (np.log(x) + np.log(1.0 - x))
    dens = np.exp(logdens)

    total = dens.sum()
    if total == 0.0 or not np.isfinite(total):
        # fallback – uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    a = dens / total
    return a

    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a 5 s safety margin as required
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size – honour any previously found length
    # --------------------------------------------------------------
    try:
        n = len(globals()["height_sequence_1"])
        n = max(512, min(n, 4096))          # keep FFTs cheap
    except Exception:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for a * w_rev

    best_score = float('inf')
    best_vec = None

    # --------------------------------------------------------------
    # 2) Sweep a few α‑values in the Beta(α,α) family
    #    (α=0.5 is the continuous arcsine distribution)
    # --------------------------------------------------------------
    alphas = np.concatenate((
        np.linspace(0.30, 0.70, 9),   # fine grain around the optimum
        np.linspace(0.80, 3.00, 12),  # broader exploration
        np.array([0.5])               # guarantee the exact arcsine case
    ))
    alphas = np.unique(alphas)
    alphas = alphas[(alphas > 0) & (alphas <= 5)]

    for alpha in alphas:
        if time.monotonic() >= deadline:
            break

        a = beta_distribution(alpha, n)
        score = compute_J(a, n, fft_len_self, conv_len)

        if score < best_score:
            best_score = score
            best_vec = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] α={alpha:.3f} new best J={best_score:.6f}")

        # ----------------------------------------------------------
        # 3) Small random block perturbations around the current shape
        # ----------------------------------------------------------
        for _ in range(4):
            if time.monotonic() >= deadline:
                break
            cand = _random_block_perturb(a, rng)
            cand_score = compute_J(cand, n, fft_len_self, conv_len)
            if cand_score < best_score:
                best_score = cand_score
                best_vec = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] pert α={alpha:.3f} J={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) If a previous best exists, give it a quick chance as well
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.size != n:
                # truncate or pad with zeros to match n
                if prev.size > n:
                    prev = prev[:n]
                else:
                    prev = np.concatenate([prev,
                                           np.zeros(n - prev.size, dtype=np.float64)])
            prev = proj_simplex(prev)
            prev_score = compute_J(prev, n, fft_len_self, conv_len)
            if prev_score < best_score:
                best_score = prev_score
                best_vec = prev.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] previous best J={best_score:.6f}")
        except Exception:
            pass  # ignore malformed previous data

    # --------------------------------------------------------------
    # 5) Short Adam‑style refinement on the true objective
    # --------------------------------------------------------------
    if best_vec is None:
        # fallback uniform – should almost never happen
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_J(best_vec, n, fft_len_self, conv_len)

    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)   # ∈[0,1]

        # learning‑rate schedule (fast → slow)
        lr = 0.25 * (1.0 - frac) + 0.001 * frac
        # temperature schedule – slowly sharpen soft‑max
        alpha_tmp = 5.0 * (1.0 - frac) + 5000.0 * frac

        # current convolution (full length) and true objective
        conv_full = compute_conv(a, fft_len_self, conv_len)
        cur_J = float(2.0 * n * conv_full.max())

        # gradient of the smooth surrogate
        grad = softmax_gradient(a, conv_full, alpha_tmp,
                                n, fft_len_grad)

        # Adam update with projection onto the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # evaluate true (non‑smoothed) objective
        conv_full = compute_conv(a, fft_len_self, conv_len)
        cur_J = float(2.0 * n * conv_full.max())

        if cur_J < best_score - 1e-12:
            best_score = cur_J
            best_vec = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Adam refined J={best_score:.6f}")

        # occasional jitter to avoid tiny plateaus
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart if we have been stagnant for a while
        if now - last_improve > 30.0:
            a = best_vec.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best J={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best vector, clipped to the limits of evaluate_sequence
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]
