def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    import numpy as np, time
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 0.5   # safety margin
    # choose n based on earlier sequence length if available else default
    if "height_sequence_1" in globals():
        n0 = len(height_sequence_1)
        n = max(256, min(n0, 8192))
    else:
        n = 2048
    n = int(n)
    # make FFT length power of 2 for conv length (2n-1) and correlation length (3n-2)
    conv_len = 2*n - 1
    corr_len = 3*n - 2
    fft_len = 1 << ((corr_len - 1).bit_length())
    # Initialize a
    # Use mixture: uniform + Dirichlet + prior solution if available
    candidates = []
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # Dirichlet with small concentration -> spiky
    for conc in [0.05,0.2,0.5,1.0]:
        candidates.append(rng.dirichlet(np.full(n, conc)))
    # two-spike (ends)
    v = np.zeros(n)
    v[0]=0.5; v[-1]=0.5
    candidates.append(v)
    # prior
    if "height_sequence_1" in globals():
        v = np.array(height_sequence_1[:n], dtype=np.float64)
        v = np.clip(v, 0, None)
        s=v.sum()
        if s>0: v/=s
        else: v[:]=1.0/n
        candidates.append(v)
    # pick best among candidates
    best_J = np.inf
    best_a = None
    def compute_J_from_a(a):
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())
    for cand in candidates:
        J=compute_J_from_a(cand)
        if J<best_J:
            best_J=J
            best_a=cand.copy()
    print(f"[0.00s] Init best J={best_J:.6f} n={n}")
    # Gradient descent with softmax smoothing
    a = best_a.copy()
    # hyperparameters
    beta = 0.5   # start smoothing
    beta_max = 20.0
    beta_mul = 1.005  # increase per iteration
    lr0 = 4.0   # initial learning rate
    # track best overall across runs maybe different lengths - but we use single.
    best_overall_J = best_J
    best_overall_a = a.copy()
    iter_count=0
    # small epsilon for numeric stability
    while time.monotonic() < deadline:
        # compute FFT of a
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        # compute softmax weights
        # to avoid overflow use max
        temp = beta * conv
        maxtemp = np.max(temp)
        expw = np.exp(temp - maxtemp)
        w = expw / np.sum(expw)
        # compute gradient via correlation
        W = np.fft.rfft(w, n=fft_len)
        corr = np.fft.irfft(W * np.conj(A), n=fft_len)
        grad = 2.0 * corr[n-1:n-1+n]
        # learning rate schedule
        lr = lr0 / np.sqrt(iter_count + 1)
        # update
        a_new = a - lr * grad
        # project onto simplex
        a_new = proj_simplex(a_new)
        # evaluate new J
        J_new = float(2.0 * n * np.fft.irfft(np.fft.rfft(a_new, n=fft_len) * np.fft.rfft(a_new, n=fft_len), n=fft_len)[:conv_len].max())
        if J_new < best_overall_J:
            best_overall_J = J_new
            best_overall_a = a_new.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:.2f}s] New best J={best_overall_J:.6f}")
        # acceptance: we always accept new a (deterministic gradient descent). But incorporate if J not improve? Might still move.
        # We can just replace a = a_new regardless (gradient descent on smoothed objective).
        a = a_new
        # increase beta slowly every iter
        if beta < beta_max:
            beta *= beta_mul
        iter_count+=1
        # occasional random perturbation to escape local minima
        if iter_count % 2000 == 0:
            # add small noise
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            a = proj_simplex(a * noise)
    # after loop, return best_overall_a scaled to allow sum not needed? evaluate_sequence will rescale anyway.
    return [float(x) for x in best_overall_a]

Implementation for gradient step:

```
def grad_step(a, L_fft, conv_len, n, beta):
    A = np.fft.rfft(a, n=L_fft)
    conv = np.fft.irfft(A * A, n=L_fft)[:conv_len]
    # softmax
    scaled = beta * conv
    max_scaled = np.max(scaled)
    expw = np.exp(scaled - max_scaled)
    w = expw / expw.sum()
    # gradient
    W = np.fft.rfft(w, n=L_fft)
    corr = np.fft.irfft(W * np.conj(A), n=L_fft)
    grad = 2.0 * np.real(corr[n-1:n-1+n])
    return grad, np.max(conv)
```

def refine_with_adam(a, n, L_fft, conv_len, max_iters, deadline):
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    alpha=0.1
    beta = 0.5
    best_j = float('inf')
    best_a = a.copy()
    for t in range(1, max_iters+1):
        if time.monotonic() >= deadline: break
        grad, max_conv = grad_step(a, L_fft, conv_len, n, beta)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a - step)
        J = 2 * n * max_conv
        if J < best_j:
            best_j = J
            best_a = a.copy()
        # schedule beta increase
        if beta < 20.0:
            beta *= 1.001
        # maybe reduce alpha gradually
        if t % 1000 == 0:
            alpha *= 0.9
    return best_a, best_j
```

"""
Gradient‑descent / Adam optimiser for the “step‑function” problem.

The objective is  J(a)=2·n·max(conv(a,a))   (we keep Σa=1, i.e. a lies on the
probability simplex).  The algorithm works directly on a smooth approximation
of the max using a soft‑max with a temperature β that is slowly increased.
For each β we compute the gradient of the smoothed objective by a single
FFT‑based convolution, then perform an Adam step and project back onto the
simplex.  Random restarts (Dirichlet draws with several concentration
parameters) and occasional multiplicative jitter are used to escape plateaus.
The best sequence found before the time limit is returned.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute J = 2·n·max(conv(a,a)) for a vector that already sums to 1.
    Uses an FFT of length `fft_len`.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def _grad_step(a: np.ndarray,
               n: int,
               fft_len: int,
               conv_len: int,
               beta: float):
    """
    One gradient computation of the smoothed objective.
    Returns (gradient, max_convolution_value).
    """
    # forward FFT of the current a
    A = np.fft.rfft(a, n=fft_len)

    # full linear convolution a*a (length conv_len)
    conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]

    # soft‑max weights over the convolution entries
    scaled = beta * conv_full
    m = np.max(scaled)                         # numerical stability
    expw = np.exp(scaled - m)
    w = expw / expw.sum()                     # Σ w = 1

    # gradient = 2 * correlation(w, a)
    W = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(W * np.conj(A), n=fft_len)
    # the correlation slice that corresponds to indices 0 … n‑1
    grad = 2.0 * np.real(corr_full[n - 1:n - 1 + n])

    return grad, conv_full.max()


def _adam_refine(init_a: np.ndarray,
                 n: int,
                 fft_len: int,
                 conv_len: int,
                 beta_start: float,
                 beta_max: float,
                 max_iters: int,
                 deadline: float):
    """
    Adam optimiser starting from `init_a`.  The temperature β is increased
    slowly from `beta_start` toward `beta_max`.  Returns the best vector seen
    and its J‑value.
    """
    a = proj_simplex(init_a.copy())
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta = beta_start
    beta_inc = 1.001  # geometric increase
    # Adam hyper‑parameters
    alpha = 0.10
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    best_J = float('inf')
    best_a = a.copy()

    for t in range(1, max_iters + 1):
        if time.monotonic() >= deadline:
            break

        grad, max_conv = _grad_step(a, n, fft_len, conv_len, beta)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a - step)

        J = 2.0 * n * max_conv
        if J < best_J:
            best_J = J
            best_a = a.copy()

        # temperature schedule (slow increase)
        if beta < beta_max:
            beta *= beta_inc

        # occasional jitter to help escape flat regions
        if t % 2000 == 0:
            noise = np.exp(np.random.default_rng().normal(0.0, 0.02, size=n))
            a = proj_simplex(a * noise)

    return best_a, best_J


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  A fast ADAM optimiser operating on a smooth surrogate
    of the max‑convolution is used, together with several random restarts.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # 0. Choose a problem size.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    n = int(n)
    conv_len = 2 * n - 1
    corr_len = 3 * n - 2                     # length needed for correlation
    fft_len = _next_pow2(corr_len)           # power‑of‑two length for speed

    # ----------------------------------------------------------------------
    # 1. Build an initial pool of candidates on the simplex.
    # ----------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (hand‑crafted)
    for r in [0.95, 0.97, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Dirichlet draws with various concentrations
    for conc in [0.01, 0.05, 0.2, 0.5, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.full(n, conc)))

    # two‑spike at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    init_cands.append(v)

    # include a truncated version of the previously known construction,
    # if it exists.
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size > n:
            prev = prev[:n]
        else:
            # pad with zeros if shorter
            prev = np.pad(prev, (0, n - prev.size), constant_values=0.0)
        if prev.sum() == 0.0:
            prev[:] = 1.0 / n
        else:
            prev /= prev.sum()
        init_cands.append(prev)

    # ----------------------------------------------------------------------
    # 2. Evaluate the pool and keep the best starter.
    # ----------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = _objective_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # 3. Main optimisation loop (Adam + random restarts).
    # ----------------------------------------------------------------------
    overall_best_J = best_J
    overall_best_vec = best_vec.copy()
    last_improve = start_time

    # hyper‑parameters for the Adam refinements
    beta_start = 0.5
    beta_max = 30.0
    max_iters_per_restart = 5000

    while time.monotonic() < deadline:
        now = time.monotonic()
        time_left = deadline - now
        # decide whether to start from the current best (plus jitter) or from
        # a fresh random Dirichlet vector.
        if rng.random() < 0.6:
            # jittered copy of the current overall best
            init = overall_best_vec * np.exp(rng.normal(0.0, 0.03, size=n))
        else:
            # fresh random start – different concentrations are tried
            conc = rng.choice([0.01, 0.05, 0.2, 0.5, 1.0])
            init = rng.dirichlet(np.full(n, conc))

        # run a short Adam refinement; the deadline ensures we never exceed
        # the time budget.
        a_refined, J_refined = _adam_refine(
            init,
            n,
            fft_len,
            conv_len,
            beta_start,
            beta_max,
            max_iters_per_restart,
            deadline,
        )

        if J_refined < overall_best_J - 1e-12:
            overall_best_J = J_refined
            overall_best_vec = a_refined.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] New overall best J = {overall_best_J:.6f}")

        # if we have been idle for a while, inject a stronger random perturbation
        if now - last_improve > 30.0:
            # create a fresh random start to escape a possible plateau
            init = rng.dirichlet(np.full(n, 0.2))
            a_refined, J_refined = _adam_refine(
                init,
                n,
                fft_len,
                conv_len,
                beta_start,
                beta_max,
                max_iters_per_restart,
                deadline,
            )
            if J_refined < overall_best_J - 1e-12:
                overall_best_J = J_refined
                overall_best_vec = a_refined.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {overall_best_J:.6f}")

        # early exit if we already beat the target bound
        if overall_best_J < 1.5025:
            break

    # ----------------------------------------------------------------------
    # 4. Return the best vector (already on the simplex, safe for evaluate_sequence).
    # ----------------------------------------------------------------------
    overall_best_vec = np.clip(overall_best_vec, 0.0, 1000.0)
    s = overall_best_vec.sum()
    if s == 0.0:
        overall_best_vec[:] = 1.0 / n
    else:
        overall_best_vec /= s
    return [float(x) for x in overall_best_vec]
