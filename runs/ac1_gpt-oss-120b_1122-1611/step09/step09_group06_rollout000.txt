    n = a.shape[0]
    conv_len = 2*n-1
    if fft_len_self is None:

    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

"""
Propose a step‑function sequence with a lower value of
``evaluate_sequence`` by combining:

* a continuation (homotopy) on a smooth approximation of the
  ``max`` in the objective (soft‑max with increasing temperature);
* projected Adam optimisation using the exact gradient of the soft‑max;
* a final polishing phase that uses the exact sub‑gradient of the
  true max‑convolution (as in the reference implementation).

The algorithm works on the simplex (non‑negative entries with unit sum),
which leaves the original objective invariant because
``evaluate_sequence`` is homogeneous of degree 0.
A simple Euclidean projection onto the simplex is used after every
gradient step.  Random “block‑perturb” jumps and restarts are performed
when progress stalls.  The whole search respects the 1000 s budget
(with a 5 s safety margin).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    # block width ≈ up to 10 % of the vector
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear (non‑circular) self‑convolution of a with itself."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _conv_and_max(a: np.ndarray, fft_len_self: int) -> tuple[np.ndarray, int, float]:
    """Return self‑convolution, index of its maximum and the max value."""
    conv = _conv_self(a, fft_len_self)
    idx = int(conv.argmax())
    return conv, idx, float(conv.max())


def _subgradient_max(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(a⋆a) at an arg‑max position ``idx``.
    For every i we have ∂b_idx/∂a_i = 2·a_{idx‑i} (zero when index out of range).
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(idx, n - 1)  # inclusive
    i_range = np.arange(i_start, i_end + 1)
    j_range = idx - i_range
    g[i_range] = 2.0 * a[j_range]
    return g


def _smooth_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     fft_len_corr: int) -> np.ndarray:
    """
    Gradient of the smooth objective
        f(a) = logsumexp(alpha * (a⋆a)) / alpha
    with respect to a (a lives on the simplex).
    """
    # soft‑max weights, numerically stable
    max_alpha = np.max(alpha * conv)
    exp_vals = np.exp(alpha * conv - max_alpha)
    w = exp_vals / exp_vals.sum()                # shape (2n‑1,)

    # correlation r_i = Σ_j a_j w_{i+j}
    fa = np.fft.rfft(a, n=fft_len_corr)
    fw = np.fft.rfft(w, n=fft_len_corr)
    corr = np.fft.irfft(np.conj(fa) * fw, n=fft_len_corr)[: a.shape[0]]
    grad = 2.0 * corr
    return grad


def _objective_J(a: np.ndarray, fft_len_self: int) -> float:
    """Exact objective J = 2·n·max(a⋆a) (sum(a)=1 ⇒ denominator=1)."""
    n = a.shape[0]
    conv = _conv_self(a, fft_len_self)
    return float(2.0 * n * conv.max())
