def incremental_move(a, conv, i, j, delta):
    n = a.shape[0]
    delta_i = -delta
    delta_j = delta
    # keep copy of conv for update
    new_conv = conv.copy()
    new_conv[i:i+n] += 2.0 * delta_i * a
    new_conv[j:j+n] += 2.0 * delta_j * a
    new_conv[2*i] += delta_i * delta_i
    new_conv[2*j] += delta_j * delta_j
    new_conv[i+j] += 2.0 * delta_i * delta_j
    new_a = a.copy()
    new_a[i] += delta_i
    new_a[j] += delta_j
    return new_a, new_conv

def grad_of_max(a, max_idx):
    n = a.shape[0]
    idx = np.arange(n)
    j = max_idx - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    return grad

"""
Propose a candidate step‑function sequence.
The optimizer works in three stages:

1️⃣  Soft‑max Adam on the logits of the simplex.
    – a is represented by logits z (a = softmax(z)).
    – A temperature‑controlled soft‑max (parameter α) approximates the
      non‑smooth max of the self‑convolution.
    – Adam updates are applied to the logits; the learning‑rate and α are
      annealed (lr: 0.5→0.001, α: 10→20000).
    – Restarts with multiplicative noise prevent long stagnation.

2️⃣  Directed simulated‑annealing moves.
    – The current maximal convolution index K gives the exact gradient
      of the peak (∇a_i = 2·a_{K‑i} when defined).
    – Moves transfer a random fraction of mass from a high‑gradient
      coordinate to a low‑gradient one; the move is evaluated by an
      O(n) incremental update of the convolution.
    – Acceptance follows a temperature schedule (0.02→1e‑6) with a
      Metropolis rule; full recomputation of the convolution every
      few thousand steps removes accumulated numerical drift.

3️⃣  Fast permutation‑swap refinement.
    – Starting from the best vector found, random swaps are proposed.
    – The objective for a swap is recomputed with a single FFT
      (full convolution); improvements are kept, occasional uphill
      swaps are allowed with a tiny temperature.

The whole routine respects a 1000 s budget (5 s safety margin) and
returns the best sequence discovered.  All numbers are non‑negative
and clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def grad_of_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Gradient of the peak convolution entry with respect to the vector a.
    For convolution index k = max_idx we have
        ∂conv_k/∂a_i = 2·a_{k‑i}  (if 0≤k‑i<n, else 0).
    """
    n = a.shape[0]
    idx = np.arange(n)
    j = max_idx - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm combines a
    soft‑max Adam optimiser on logits, a directed simulated‑annealing
    phase with O(n) incremental convolution updates, and a final
    permutation‑swap refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start_time + total_budget

    # --------------------------------------------------------------
    # Problem dimension and initial sequence
    # --------------------------------------------------------------
    base_seq = globals().get('height_sequence_1')
    if isinstance(base_seq, list) and len(base_seq) >= 512:
        a0 = np.asarray(base_seq, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
    else:
        a0 = None

    # we keep the problem size modest for speed (≤4096)
    max_n = 4096
    min_n = 512

    if a0 is not None:
        n = a0.size
        n = max(min_n, min(n, max_n))
        a = a0[:n].copy()
    else:
        n = max_n
        # sparse random start (Dirichlet‑like)
        a = rng.gamma(0.3, 1.0, size=n)

    # normalise to sum 1 (objective is scale‑invariant)
    s = a.sum()
    if s <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s

    # FFT helpers
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)         # for a * w_rev

    # --------------------------------------------------------------
    # Phase 1 – soft‑max Adam on logits
    # --------------------------------------------------------------
    eps = 1e-12
    z = np.log(a + eps)          # logits (soft‑max ensures simplex)
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999

    phase1_budget = total_budget * 0.45
    phase1_end = start_time + phase1_budget
    it = 0
    best_a = a.copy()
    # initial objective
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    best_J = 2.0 * n * conv.max()
    last_improve = start_time
    print(f"[init] n={n}, J={best_J:.6f}")

    while time.monotonic() < phase1_end:
        it += 1
        # ----- decode logits -----
        max_z = np.max(z)
        a = np.exp(z - max_z)
        a /= a.sum()

        # ----- convolution and soft‑max weights -----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        cmax = conv.max()

        # anneal α (soft‑max temperature)
        elapsed = time.monotonic() - start_time
        frac = elapsed / phase1_budget
        alpha = 10.0 * (1.0 - frac) + 20000.0 * frac   # 10 → 20000

        shifted = conv - cmax
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient wrt a (smooth max) -----
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
        start_idx = len(w_rev) - 1
        grad_a = 2.0 * full[start_idx:start_idx + n]

        # ----- chain rule through soft‑max -----
        dot = np.dot(grad_a, a)
        grad_z = a * (grad_a - dot)

        # ----- Adam step on logits -----
        lr = 0.5 * (1.0 - frac) + 0.001 * frac   # 0.5 → 0.001
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z * grad_z)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- record best -----
        cur_J = 2.0 * n * cmax
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Phase‑1 new best J = {best_J:.6f}")

        # ----- restart on stagnation -----
        if time.monotonic() - last_improve > 20.0:
            # re‑initialise around the current best
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = best_a * noise
            a /= a.sum()
            z = np.log(a + eps)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Phase‑1 restart (stagnation)")

    # --------------------------------------------------------------
    # Phase 2 – directed simulated annealing with incremental updates
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    # exact convolution for the starting point
    fa = np.fft.rfft(cur_a, n=fft_len_self)
    cur_conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    cur_J = 2.0 * n * cur_conv.max()
    best_J = cur_J
    best_a = cur_a.copy()
    last_improve = time.monotonic()

    phase2_budget = total_budget * 0.40
    phase2_end = phase1_end + phase2_budget
    move_counter = 0
    recalc_every = max(1000, n // 4)

    while time.monotonic() < phase2_end:
        now = time.monotonic()
        # gradient of the current peak
        max_idx = cur_conv.argmax()
        grad = grad_of_max(cur_a, max_idx)

        # source probabilities (high gradient) and destination (low gradient)
        src_sum = grad.sum()
        if src_sum > 0:
            src_probs = grad / src_sum
        else:
            src_probs = np.full(n, 1.0 / n)

        inv = grad.max() - grad + 1e-12
        dst_sum = inv.sum()
        if dst_sum > 0:
            dst_probs = inv / dst_sum
        else:
            dst_probs = np.full(n, 1.0 / n)

        i = rng.choice(n, p=src_probs)
        j = rng.choice(n, p=dst_probs)
        if i == j:
            continue

        # fraction of mass to move – shrinks as time progresses
        elapsed = now - phase1_end
        frac = elapsed / (phase2_end - phase1_end)
        max_frac = 0.2 * (1.0 - frac) + 0.02   # between ~0.22 and 0.02
        delta = cur_a[i] * rng.uniform(0.01, max_frac)
        if delta <= 0.0:
            continue

        # ----- tentative incremental update -----
        new_a = cur_a.copy()
        new_conv = cur_conv.copy()
        delta_i = -delta
        delta_j = delta

        new_conv[i:i+n] += 2.0 * delta_i * cur_a
        new_conv[j:j+n] += 2.0 * delta_j * cur_a
        new_conv[2*i] += delta_i * delta_i
        new_conv[2*j] += delta_j * delta_j
        new_conv[i + j] += 2.0 * delta_i * delta_j

        new_a[i] += delta_i
        new_a[j] += delta_j

        new_J = 2.0 * n * new_conv.max()

        # temperature schedule
        T_start, T_end = 0.02, 1e-6
        T = T_start * (1.0 - frac) + T_end * frac

        # Metropolis acceptance
        if new_J < cur_J - 1e-12 or rng.random() < np.exp(-(new_J - cur_J) / T):
            cur_a = new_a
            cur_conv = new_conv
            cur_J = new_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = cur_a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] Phase‑2 new best J = {best_J:.6f}")

        move_counter += 1
        # periodic full recomputation to avoid drift
        if move_counter % recalc_every == 0:
            fa = np.fft.rfft(cur_a, n=fft_len_self)
            cur_conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
            cur_J = 2.0 * n * cur_conv.max()
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = cur_a.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] Recalc phase‑2 best J = {best_J:.6f}")

        # restart if stuck for a while
        if now - last_improve > 30.0:
            cur_a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            cur_a *= noise
            cur_a /= cur_a.sum()
            fa = np.fft.rfft(cur_a, n=fft_len_self)
            cur_conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
            cur_J = 2.0 * n * cur_conv.max()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Phase‑2 restart (stagnation)")

    # --------------------------------------------------------------
    # Phase 3 – cheap permutation‑swap refinement
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_J = best_J
    attempts = 0
    accepts = 0

    while time.monotonic() < deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand_a = cur_a.copy()
        cand_a[i], cand_a[j] = cand_a[j], cand_a[i]

        # full recompute (FFT) for the candidate
        fa = np.fft.rfft(cand_a, n=fft_len_self)
        conv_cand = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        cand_J = 2.0 * n * conv_cand.max()

        if cand_J < cur_J - 1e-12:
            cur_a = cand_a
            cur_J = cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_a = cand_a.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] swap new best J = {best_J:.6f}")
        else:
            # tiny uphill probability to escape plateaus
            if rng.random() < np.exp(-(cand_J - cur_J) / 1e-4):
                cur_a = cand_a
                cur_J = cand_J
                accepts += 1

        # occasional heavy perturbation to re‑seed the local search
        if attempts % 20000 == 0:
            cur_a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.2, size=n))
            cur_a *= noise
            cur_a /= cur_a.sum()
            fa = np.fft.rfft(cur_a, n=fft_len_self)
            cur_conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
            cur_J = 2.0 * n * cur_conv.max()

    print(f"Swap phase finished – attempts: {attempts}, accepts: {accepts}")
    print(f"Best J found = {best_J:.6f}")

    # --------------------------------------------------------------
    # Return the best sequence (clip to evaluation limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
