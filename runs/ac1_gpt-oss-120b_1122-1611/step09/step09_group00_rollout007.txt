def init_edge_power_law(n, alpha=0.8, eps=1e-9):
    idx = np.arange(n)
    left = (idx + 1) ** (-alpha)
    right = (n - idx) ** (-alpha)  # indexes from 1 to n
    a = left + right
    a = np.clip(a, eps, None)
    a /= a.sum()

"""
Propose a step‑function that minimizes `evaluate_sequence`.

The algorithm is based on a smooth surrogate of the max‑convolution
objective:

* Initialise a large vector (default 16384) with a power‑law edge‑heavy
  shape a_i ∝ (i+1)^(-α) + (N-i)^(-α) (α≈0.8).  This already makes the
  self‑convolution fairly flat.

* Repeatedly minimise a soft‑max version of the convolution.
  For the current a, the convolution c = a∗a is computed via an FFT.
  A soft‑max weight w_k = exp(β·c_k)/∑exp(β·c) approximates the
  “active‑set’’ of indices attaining the maximum.
  The gradient of the surrogate is
      ∂/∂a_i  Σ_k w_k (a∗a)_k = 2·∑_k w_k a_{k‑i}
  which can be obtained by one more FFT (a convolved with the reversed
  weight vector).

* An exponentiated‑gradient (mirror‑descent on the simplex) update is
  performed on the log‑probabilities log a_i with a diminishing step‑size.
  After each update a is projected back onto the probability simplex.

* The temperature β is slowly increased (β←β·1.015 up to β_max) so that
  the surrogate gradually focuses on the true maximum.

* Every few iterations the exact objective 2·N·max(c) is evaluated.
  The best vector seen so far is stored and returned at the end.

The whole routine respects the time budget (≈ 1000 s) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # uniform vector if nothing is positive
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise if needed
    w_sum = w.sum()
    if w_sum == 0.0:
        w.fill(1.0 / len(v))
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft_general(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two arbitrary 1‑D arrays using an FFT of length ``fft_len``.
    Returns the full convolution (len(x)+len(y)-1 elements, may be longer).
    """
    xf = np.fft.rfft(x, n=fft_len)
    yf = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(xf * yf, n=fft_len)
    return conv


def _smooth_gradient(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth (soft‑max) surrogate with respect to a.
    Implements   g_i = 2 * Σ_k w_k * a_{k-i} .
    Computed efficiently via FFT.
    """
    n = a.size
    # reverse w
    w_rev = w[::-1]
    # full convolution a * w_rev
    conv_full = _conv_fft_general(a, w_rev, fft_len)
    # slice the part that corresponds to indices n‑1 … 2n‑2 (inclusive)
    # conv_full length ≥ 3n‑2, the needed slice has length n
    start = n - 1
    end = 2 * n - 1
    slice_part = conv_full[start:end]      # shape (n,)
    # reverse to obtain gradient ordering
    grad = 2.0 * slice_part[::-1]
    return grad


def _initial_edge_power_law(n: int, alpha: float = 0.8) -> np.ndarray:
    """
    Edge‑heavy initial distribution:
        a_i ∝ (i+1)^(-α) + (n-i)^(-α)   (i = 0 … n‑1)
    Normalised to sum = 1.
    """
    idx = np.arange(n, dtype=np.float64)
    left = (idx + 1.0) ** (-alpha)
    right = (n - idx) ** (-alpha)
    a = left + right
    # avoid exact zeros – keep a tiny epsilon for numerical stability
    a = np.maximum(a, 1e-15)
    a /= a.sum()
    return a

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Choose the length N.
    #    Larger N gives more degrees of freedom at the cost of CPU time.
    # --------------------------------------------------------------
    N = 16384                     # power of two → tidy FFT bookkeeping
    # If a previous construction exists, optionally up‑sample it.
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if old.size < N:
            # pad with zeros and renormalise – this gives a warm start
            a = np.zeros(N, dtype=np.float64)
            a[: old.size] = old
            a = np.clip(a, 1e-15, None)   # avoid exact zeros for log‑steps
            a /= a.sum()
        else:
            a = old[:N].copy()
            a = np.clip(a, 1e-15, None)
            a /= a.sum()
    else:
        # default: edge‑heavy power‑law shape
        a = _initial_edge_power_law(N, alpha=0.8)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    fft_len = _next_pow2(3 * N)   # safe for a*a and a*w_rev

    # --------------------------------------------------------------

    conv = _conv_fft(a, fft_len)
    best_score = 2.0 * N * conv.max()   # exact score because Σa = 1
    best = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep 5 s safety margin

    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Parameters for the smooth surrogate optimisation
    # --------------------------------------------------------------
    beta = 2.0                # initial inverse temperature
    beta_factor = 1.015       # gradual increase per iteration
    beta_max = 200.0

    # --------------------------------------------------------------
    # 4) Main optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    eval_every = 15           # evaluate exact score only every few steps
    while time.monotonic() < deadline:

        conv = _conv_fft(a, fft_len)          # length 2N‑1
        max_conv = conv.max()

        w_exp = np.exp(shifted)
        w = w_exp / w_exp.sum()                # shape (2N‑1,)

        # ----- 4.2) gradient of the surrogate -----
        grad = _smooth_gradient(a, w, fft_len)

        # ----- 4.3) exponentiated‑gradient update -----

        log_a -= eta * grad
        a = np.exp(log_a)
        a = _project_onto_simplex(a)

            conv_full = _conv_fft(a, fft_len)
            score = 2.0 * N * conv_full.max()
            if score < best_score - 1e-12:
                best_score = score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (smooth) = {best_score:.6f} (it={iteration})")
                if best_score <= 1.5025:
                    print("[early stop] target reached")
                    break

        # ----- 4.5) temperature annealing -----
        beta = min(beta_max, beta * beta_factor)

        # ----- 4.6) occasional jitter to escape plateaus -----

            a_jit = a.copy()

    # Clip to evaluator limits before returning (just in case)
