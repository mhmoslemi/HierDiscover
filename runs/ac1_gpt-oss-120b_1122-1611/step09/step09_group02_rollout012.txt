"""
propose_candidate
=================

A three‑phase method that (a) builds a principled start‑point from the
inverse‑FFT of a constant target autocorrelation (the “square‑root‑FFT”
initialiser), (b) enriches the portfolio with a few simple analytic
candidates (uniform and exponential decays), and (c) refines the
best vector with a projected *exponentiated‑gradient* optimiser that
optimises a soft‑max surrogate of the true max‑convolution.  All steps
respect the simplex constraint (non‑negative entries that sum to 1) and
the overall 1000 s time‑budget.

The core of the refinement is

    w_k = softmax_α ( (a*a)_k )
    grad_i = 2·(a ⋆ w_rev)_i

and the update is the mirror‑descent step

    a_i ← a_i·exp(−η·grad_i) ,   a←proj_simplex(a).

Both the learning‑rate η and the temperature α are annealed
linearly to zero‑temperature, which drives the surrogate to the true
hard maximum while keeping the landscape smooth at the beginning.
Random block jitter and occasional restarts from the current best
prevent stagnation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block, followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) (the sum of a is assumed to be 1)."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def self_convolution(a: np.ndarray,
                     fft_len_self: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


def _sqrt_fft_initialiser(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Build a start‑vector from the inverse FFT of the square‑root
    of a constant target autocorrelation (length 2·n−1, value 1/(2·n−1)).
    The result is clipped to non‑negative values and projected onto the simplex.
    """
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # constant target autocorrelation (same mass at every lag)
    target = np.full(conv_len, 1.0 / conv_len, dtype=np.float64)

    # FFT of the target (real → complex spectrum)
    C = np.fft.rfft(target, n=fft_len)

    # Take the complex square‑root of the spectrum
    sqrtC = np.sqrt(C)

    # Inverse FFT gives a candidate sequence (real‑valued up to numerical error)
    cand_full = np.fft.irfft(sqrtC, n=fft_len)
    cand = np.real(cand_full[:n])

    # Remove negatives, renormalise to the simplex
    cand = np.clip(cand, 0.0, None)
    if cand.sum() == 0.0:
        cand[:] = 1.0 / n
    else:
        cand /= cand.sum()
    return cand


def _exponential_candidate(n: int, r: float) -> np.ndarray:
    """Geometric decay a_i ∝ r^i, normalised to the simplex."""
    v = r ** np.arange(n, dtype=np.float64)
    if v.sum() == 0.0:
        v[:] = 1.0 / n
    else:
        v /= v.sum()
    return v

    square‑root‑FFT construction, enriches it with a few analytic candidates,
    and then runs a projected exponentiated‑gradient optimiser that

    """

    # c) several geometric decays (r in [0.80, 0.99])
    for r in np.linspace(0.80, 0.99, 10):
        candidates.append(_exponential_candidate(n, r))

    # d) mirrored exponential (large values at both ends)
    for r in np.linspace(0.80, 0.99, 5):
        left = _exponential_candidate(n // 2, r)
        right = left[::-1]
        v = np.concatenate([left, right]) if n % 2 == 0 else np.concatenate([left, [0.5], right])
        v = np.clip(v, 0.0, None)
        v /= v.sum()
        candidates.append(v)

    # e) if a previous construction exists, interpolate it to length n
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        if old.size != n:
            x_old = np.linspace(0.0, 1.0, old.size)
            x_new = np.linspace(0.0, 1.0, n)
            old = np.interp(x_new, x_old, old)
        candidates.append(proj_simplex(old))

    # f) a handful of Dirichlet draws (moderate concentration)
    for _ in range(5):
        candidates.append(rng.dirichlet(np.ones(n) * 0.3))

    # ------------------------------------------------------------------
    # 3) Evaluate all portfolios and keep the best
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        cand = proj_simplex(cand)          # safety projection
        score = compute_J(cand, n, fft_len_self, conv_len)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
        # occasional logging
        elapsed = time.monotonic() - start
        print(f"[{elapsed:6.2f}s] init #{idx:02d} – J = {score:.6f}")

    # ------------------------------------------------------------------
    # 4) Local refinement – projected exponentiated‑gradient (soft‑max)
    # ------------------------------------------------------------------
    a = best_vec.copy()
    last_improve = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # ---------- convolution ----------
        conv = self_convolution(a, fft_len_self, conv_len)
        max_val = conv.max()

        # ---------- soft‑max (temperature α) ----------
        # linear annealing of α from 2 → 3000
        total_time = deadline - start
        elapsed = time.monotonic() - start
        frac = elapsed / total_time if total_time > 0 else 0.0
        alpha = 2.0 * (1.0 - frac) + 3000.0 * frac

        shifted = conv - max_val          # improve numerical stability
        # clip exponent to avoid overflow in np.exp
        exp_arg = np.clip(alpha * shifted, -700.0, 700.0)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        start_idx = w_rev.size - 1          # = 2·n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ---------- exponentiated‑gradient update ----------
        # learning‑rate η annealed from 0.5 → 1e‑4
        eta = 0.5 * (1.0 - frac) + 1e-4 * frac
        a = a * np.exp(-eta * grad)

        # projection onto simplex (simple normalisation is enough)
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---------- evaluate true objective ----------
        cur_score = compute_J(a, n, fft_len_self, conv_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:6.2f}s] new best J = {best_score:.6f}")

        # ---------- occasional stochastic jitter ----------
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # ---------- stagnation restart ----------
        if time.monotonic() - last_improve > 30.0:
            # re‑initialise from the current best with a small perturbation
            a = best_vec * np.exp(rng.normal(0.0, 0.02, size=n))
            a = proj_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:6.2f}s] restart after stagnation – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
