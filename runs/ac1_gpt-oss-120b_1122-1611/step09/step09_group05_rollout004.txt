def compute_conv(a, fft_len=None):
    n = len(a)
    if fft_len is None:
        conv_len = 2*n-1
        fft_len = 1 << ((conv_len-1).bit_length())
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:2*n-1]
    return conv

    while time.time() < deadline:

        # continue outer while loop for new start

    if best_a.sum() < 0.01:

    # return list
    return [float(x) for x in best_a]

"""
Gradient‑softmax optimiser for the non‑negative step‑function problem.

We minimise the scale‑invariant objective
    J(a) = 2·n·max(conv(a,a))          (∑a = 1)

by iterating on the probability simplex.
The max is approximated with a soft‑max (parameter β) so that the
objective becomes differentiable.  An Adam optimiser with a decreasing
learning rate is applied to the gradient, after which the iterate is
projected back onto the simplex (Euclidean projection).  β is increased
gradually so that the optimisation first smooths the whole convolution
profile and later concentrates on the true maximum.

Multiple restarts (jittered copies of the current best and fresh Dirichlet
samples) are performed until the time budget expires.  The best vector
found is returned as a plain list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are equal (or numerical noise); return uniform.
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  Uses a soft‑maxed gradient descent
    with Adam updates on the probability simplex.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. Problem size – honour a previously known construction if present.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(8192, n))          # keep FFT sizes reasonable
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    conv_fft_len = _next_pow2(conv_len)               # for a*a
    grad_len = 3 * n - 2
    grad_fft_len = _next_pow2(grad_len)               # for p * rev(a)

    # ------------------------------------------------------------------
    # 1. Helper routines (FFT‑based convolution, J‑value, gradient)
    # ------------------------------------------------------------------
    def conv_self(a: np.ndarray) -> np.ndarray:
        """Linear convolution a * a (length 2n‑1) via FFT."""
        A = np.fft.rfft(a, n=conv_fft_len)
        c = np.fft.irfft(A * A, n=conv_fft_len)[:conv_len]
        return c

    def obj_J(a: np.ndarray, c: np.ndarray | None = None) -> float:
        """Return the true objective J for a (assumed Σa = 1)."""
        if c is None:
            c = conv_self(a)
        return float(2.0 * n * c.max())

    def grad_softmax(a: np.ndarray, c: np.ndarray, beta: float) -> np.ndarray:
        """
        Gradient of the soft‑max approximation of max(conv) w.r.t. a.
        Uses FFT for the convolution p * rev(a).
        """
        max_c = c.max()
        # soft‑max weights (avoid overflow)
        exp_term = np.exp(beta * (c - max_c))
        p = exp_term / exp_term.sum()
        # convolution with reversed a
        rev = a[::-1]
        P = np.fft.rfft(p, n=grad_fft_len)
        R = np.fft.rfft(rev, n=grad_fft_len)
        full = np.fft.irfft(P * R, n=grad_fft_len)
        # slice corresponds to indices n‑1 … n‑1+n‑1
        g = 2.0 * full[n - 1 : n - 1 + n]
        return g

    # ------------------------------------------------------------------
    # 2. Generate a diverse set of initial probability vectors.
    # ------------------------------------------------------------------
    init_vectors = []

    # uniform
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few geometric decays
    for r in (0.9, 0.95, 0.97, 0.99):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_vectors.append(v)

    # Dirichlet draws (moderately spread)
    for _ in range(8):
        init_vectors.append(rng.dirichlet(np.ones(n) * 0.4))

    # previous best if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = np.full(n, 1.0 / n)
        init_vectors.append(v)

    best_a = None
    best_J = np.inf

    for cand in init_vectors:
        if cand.shape[0] != n:
            continue
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s == 0.0:
            cand = np.full(n, 1.0 / n)
        else:
            cand /= s
        c = conv_self(cand)
        J = obj_J(cand, c)
        if J < best_J:
            best_J = J
            best_a = cand.copy()
    print(f"[0.00s] initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – Adam + soft‑max + simplex projection.
    # ------------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)      # safety margin

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    base_lr = 0.001                     # works well for the simplex scale
    max_inner = 2000                    # max iterations per restart

    while time.time() < deadline:
        # ---- initialise a fresh starting point ---------------------------------
        if rng.random() < 0.15:
            cur = rng.dirichlet(np.ones(n) * 0.4)
        else:
            # jitter around the current best
            cur = best_a * np.exp(rng.normal(0.0, 0.02, size=n))
            cur = proj_simplex(cur)

        m = np.zeros(n, dtype=np.float64)   # first moment
        v2 = np.zeros(n, dtype=np.float64)  # second moment
        inner_start = time.time()

        for it in range(1, max_inner + 1):
            # ---- evaluate current objective ------------------------------------
            c = conv_self(cur)
            J_cur = obj_J(cur, c)

            # keep global best
            if J_cur + 1e-12 < best_J:
                best_J = J_cur
                best_a = cur.copy()
                elapsed = time.time() - start_time
                print(f"[{elapsed:6.2f}s] new best J = {best_J:.6f}")

            # ---- soft‑max temperature (β) schedule -----------------------------
            # start smooth (β≈1) and ramp to ~200 over the first 400 steps
            beta = 1.0 + 199.0 * min(1.0, it / 400.0)

            # ---- gradient of the smoothed objective ---------------------------
            grad = grad_softmax(cur, c, beta)

            # ---- Adam update --------------------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad
            v2 = beta2 * v2 + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v2 / (1.0 - beta2 ** it)

            step = base_lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- gradient step + projection onto the simplex ------------------
            cur = cur - step
            cur = proj_simplex(cur)

            # ---- time checks -------------------------------------------------
            if time.time() >= deadline:
                break

            # occasional early‑exit from inner loop if no progress for a while
            if time.time() - inner_start > 30.0:
                # give the outer loop a chance to restart from a new seed
                break

    # ------------------------------------------------------------------
    # 4. Return the best vector (already normalised, safe for evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = best_a.sum()
    if s == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s
    return [float(x) for x in best_a]
