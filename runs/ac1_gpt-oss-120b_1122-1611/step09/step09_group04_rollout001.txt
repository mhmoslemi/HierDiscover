"""
Propose a step‑function (list of non‑negative heights) that minimizes
evaluate_sequence().  The optimizer is a two‑stage heuristic:

1️⃣  Global cheap sweep – generate many low‑dimensional families
    (uniform, edge blocks, cosine, polynomial, Dirichlet draws,
    previous best).  Keep the top‑K candidates w.r.t. the true
    objective J = 2·n·max(conv(p,p)) where p is the probability vector.

2️⃣  Local refinement – for every seed run
    • a short Adam optimisation on a log‑sum‑exp surrogate of the max,
      with an aggressive temperature schedule,
    • then an exponentiated‑gradient sub‑gradient descent on the exact
      max‑convolution (hard max).  Both phases respect the simplex
      constraint (non‑negative, sum = 1) via either projection or simple
      renormalisation.  Periodic jitter and stagnation restarts improve
      exploration.

The best probability vector is finally scaled to have L1‑norm √(2 n),
clipped to the limits enforced inside evaluate_sequence(),
and returned as a Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities (no closures)
# --------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """Two edge blocks of length L with linear taper controlled by gamma."""
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return v / s


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape: (1 + r*cos(theta))**power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    v = v.clip(min=0.0)
    s = v.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return v / s


def _poly_shape(n: int, beta: float) -> np.ndarray:
    """Polynomial shape: (i+1)^beta, normalised."""
    i = np.arange(n, dtype=np.float64) + 1.0
    if beta < 0:
        # decreasing shape, larger values on the left
        v = (n - i + 1) ** (-beta)
    else:
        v = i ** beta
    v = v.clip(min=0.0)
    s = v.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return v / s


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max weights (sum to 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate w.r.t. a (length‑n vector)."""
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1
    return 2.0 * full[start_idx:start_idx + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """
    Returns the true objective J = 2·n·max(p*p) for a probability vector p.
    """
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


# --------------------------------------------------------------
# Optimisation sub‑routines
# --------------------------------------------------------------


def _adam_refine(p_start: np.ndarray,
                 time_limit: float,
                 n: int,
                 fft_len_self: int,
                 fft_len_grad: int,
                 rng: np.random.Generator,
                 best_state):
    """
    Projected Adam on the log‑sum‑exp surrogate of the max‑convolution.
    ``best_state`` is a dict with keys ``best_J`` and ``best_p`` that is
    updated in‑place when a better candidate is found.
    """
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    # aggressive schedule (temperature grows fast)
    lr_start, lr_end = 1.0, 0.001
    alpha_start, alpha_end = 500.0, 5000.0

    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    last_improve = time.monotonic()
    start = time.monotonic()

    while time.monotonic() < time_limit:
        it += 1
        now = time.monotonic()
        frac = (now - start) / (time_limit - start)  # ∈[0,1]

        # learning‑rate and temperature schedules
        lr = lr_start * (0.5 ** frac) + lr_end * (1 - 0.5 ** frac)
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # surrogate objective
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)

        # gradient of the surrogate
        grad = _grad_softmax(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # true objective (hard max)
        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_state["best_J"] - 1e-12:
            best_state["best_J"] = cur_J
            best_state["best_p"] = a.copy()
            last_improve = now
            print(f"[{now - start:.1f}s] Adam improved J = {cur_J:.6f}")

        # occasional jitter to escape plateaus
        if rng.random() < 0.02:
            block_w = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - block_w + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            pert = a.copy()
            pert[start_idx:start_idx + block_w] *= factor
            pert = proj_simplex(pert)
            a = pert

        # restart on stagnation
        if now - last_improve > 20.0:
            a = best_state["best_p"].copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start:.1f}s] Adam restart due to stagnation")


def _exp_gradient_refine(p_start: np.ndarray,
                         time_limit: float,
                         n: int,
                         fft_len_self: int,
                         rng: np.random.Generator,
                         best_state):
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Uses the hard max (no smoothing) and multiplicative updates that keep
    the iterate on the simplex automatically.
    """
    p = p_start.copy()
    iteration = 0
    last_improve = time.monotonic()
    start = time.monotonic()
    eta0 = 0.5  # initial step‑size (temperature for EG)

    while time.monotonic() < time_limit:
        iteration += 1
        conv = _self_conv(p, fft_len_self)
        max_val = conv.max()
        k_max = int(conv.argmax())
        cur_J = 2.0 * n * max_val
        if cur_J < best_state["best_J"] - 1e-12:
            best_state["best_J"] = cur_J
            best_state["best_p"] = p.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:.1f}s] EG improved J = {cur_J:.6f}")

        # sub‑gradient of the hard max entry
        grad = np.zeros_like(p)
        idx = np.arange(n)
        j = k_max - idx
        mask = (j >= 0) & (j < n)
        grad[mask] = 2.0 * p[j[mask]]

        # step‑size schedule (decreasing)
        eta = eta0 / np.sqrt(iteration)

        # multiplicative update (exponentiated gradient)
        p = p * np.exp(-eta * grad)
        s = p.sum()
        if s == 0.0 or not np.isfinite(s):
            p = np.full_like(p, 1.0 / n)
        else:
            p = p / s

        # jitter to diversify search
        if rng.random() < 0.02:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            p = p * noise
            p = p / p.sum()

        # restart on stagnation
        if time.monotonic() - last_improve > 25.0:
            p = best_state["best_p"].copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            p = p * noise
            p = p / p.sum()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:.1f}s] EG restart due to stagnation")

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Two‑phase technique:
        1. Global cheap sweep over a diverse family of shapes.
        2. Local refinement using (a) Adam on a smooth‑max surrogate,
           (b) exponentiated‑gradient sub‑gradient descent on the exact max.
    Returns a list of heights (scaled to satisfy the L1‑norm √(2 n)).
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # -----------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -----------------------------------------------------------------
    # 1) Problem size (try to keep the FFT cheap)
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = min(max(n, 256), 8192)   # cap a bit higher than before
        except Exception:
            n = 4096
    else:
        n = 4096

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # -----------------------------------------------------------------
    # 2) Global cheap exploration – gather a pool of promising seeds
    # -----------------------------------------------------------------
    K_KEEP = 15  # keep this many best seeds for refinement
    best_candidates = []  # list of (J, p) tuples

    def _push(p_vec: np.ndarray):
        """Evaluate candidate and keep the best K_KEEP."""
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > K_KEEP:
            best_candidates.pop()

    # a) uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push(uniform)

    # b) edge shapes (grid over block size & taper)
    for frac in np.linspace(0.12, 0.45, 12):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            p = _edge_shape(n, L, gamma)
            _push(p)

    # c) cosine family
    for r in np.linspace(0.0, 1.0, 11):
        for pw in np.linspace(0.3, 3.0, 9):
            p = _cosine_shape(n, r, pw)
            _push(p)

    # d) polynomial (power) family
    for beta in np.linspace(-2.0, 2.0, 13):
        p = _poly_shape(n, beta)
        _push(p)

    # e) random Dirichlet draws with varied concentration
    for _ in range(40):
        alpha = rng.uniform(0.01, 0.7)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # f) include previous best if available
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() > 0.0:
                prev = proj_simplex(prev / prev.sum())
                _push(prev)
        except Exception:
            pass

    # After global sweep
    best_candidates.sort(key=lambda x: x[0])
    best_J, best_p = best_candidates[0]
    print(f"Global sweep finished. best J = {best_J:.6f}")

    # -----------------------------------------------------------------
    # 3) Local refinement – allocate remaining time
    # -----------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # time over – just return the best we have
        scale = np.sqrt(2.0 * n)
        out_seq = np.clip(best_p * scale, 0.0, 1000.0)
        return [float(x) for x in out_seq]

    # Give 80 % of the remaining budget to refinement
    refine_budget = remaining * 0.80
    per_seed = max(1.0, refine_budget / max(1, len(best_candidates)))

    # Shared state for best solution
    best_state = {"best_J": best_J, "best_p": best_p.copy()}

    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break

        # allocate time for this seed; keep a small safety margin
        seed_time = min(per_seed, deadline - time.monotonic() - 2.0)
        if seed_time <= 0.0:
            break

        # split seed budget: 40 % Adam, 60 % exponentiated‑gradient
        adam_time = seed_time * 0.40
        eg_time = seed_time - adam_time

        # Adam phase (soft‑max surrogate)
        _adam_refine(p0, time.monotonic() + adam_time,
                     n, fft_len_self, fft_len_grad, rng, best_state)

        # Exponentiated‑gradient phase (hard max)
        _exp_gradient_refine(best_state["best_p"], time.monotonic() + eg_time,
                             n, fft_len_self, rng, best_state)

    # -----------------------------------------------------------------
    # 4) Return best sequence (scale to L1‑norm √(2 n) as required)
    # -----------------------------------------------------------------
    best_seq = best_state["best_p"] * np.sqrt(2.0 * n)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
