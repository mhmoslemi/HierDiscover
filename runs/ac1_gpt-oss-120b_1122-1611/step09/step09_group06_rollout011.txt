"""
Propose a high‑quality step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.  The optimizer consists of two stages:

1️⃣  *Soft‑max continuation* – minimise a smooth approximation of the
    max‑convolution using Adam.  The temperature α of the soft‑max is
    increased from a small value to a large one, so the optimiser first
    sees a smooth landscape and then focusses on the true maximum.

2️⃣  *Exact max‑subgradient* – once α is large we switch to the exact
    sub‑gradient of the max entry (the same idea as the reference code)
    but with a slightly different learning‑rate schedule and more aggressive
    restarts.

Both stages respect the 1000 s budget (5 s safety margin) and the best
vector found is returned.  The implementation works on the probability
simplex (∑a_i = 1) – the objective `evaluate_sequence` is scale‑invariant.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator):
    """Generate a diverse set of initial points, already normalised to sum = 1."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a handful of Dirichlet draws
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        if prev.sum() == 0.0:
            prev[:] = 1.0 / n
        else:
            prev /= prev.sum()
        cand.append(prev)

    return cand


def _conv_self(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full self‑convolution a⋆a using an FFT of length ``fft_len``."""
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _conv_pair(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution of two real vectors a and b (FFT length ``fft_len``)."""
    Fa = np.fft.rfft(a, n=fft_len)
    Fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(Fa * Fb, n=fft_len)
    return conv


def _smooth_max_gradient(a: np.ndarray,
                         conv_self: np.ndarray,
                         alpha: float,
                         fft_len_grad: int,
                         conv_len: int) -> np.ndarray:
    """
    Gradient of the smooth max M_α = (1/α)log Σ exp(α·(a⋆a))
    w.r.t. a.  The result has shape (len(a),).
    """
    # soft‑max weights over the convolution entries
    # stability: subtract the maximum before exponentiation
    z = alpha * conv_self
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    w = exp_z / exp_z.sum()            # shape (conv_len,)

    # gradient = 2 * (a ★ w) where ★ denotes the convolution that matches
    # g_i = Σ_j a_j w_{i+j}.  This is obtained by convolving a with the
    # reversed weight vector and picking the appropriate slice.
    w_rev = w[::-1]
    full = _conv_pair(a, w_rev, fft_len_grad)          # length = n + conv_len - 1
    # pick indices conv_len-1 .. conv_len-1-(n-1)
    start = conv_len - 1
    g = 2.0 * full[start - np.arange(a.size)]
    return g


def _max_subgradient(a: np.ndarray,
                     conv_self: np.ndarray,
                     conv_len: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(a⋆a) at an arbitrary arg‑max position.
    """
    idx = int(conv_self.argmax())
    n = a.size
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(idx, n - 1)
    i_range = np.arange(i_start, i_end + 1)
    j_range = idx - i_range
    g[i_range] = 2.0 * a[j_range]
    return g


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

        it += 1
        # --- compute current objective ---
        conv_self = _conv_self(a, fft_len_conv, conv_len)
        max_b = conv_self.max()
        J = 2.0 * n * max_b

        if J < best_score - 1e-12:
            best_score = J
            best_vec = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Phase 1 new best J = {best_score:.6f}")

        # --- schedules (based on total elapsed fraction) ---
        total_elapsed = time.monotonic() - start_time
        total_budget = deadline - start_time
        frac = total_elapsed / total_budget
        alpha = alpha_start + (alpha_end - alpha_start) * (frac ** 1.5)
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # --- smooth‑max gradient ---
        grad = _smooth_max_gradient(a, conv_self, alpha,
                                    fft_len_grad, conv_len)

        # --- Adam update + projection ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        # occasional jitter to help escape shallow plateaus
        if rng.random() < 0.01:
            a_new = _random_block_perturb(a_new, rng)

        a = a_new

        # restart if no improvement for a while
        if time.monotonic() - last_improve > 20.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Phase 1 restart (stagnation) – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – Exact max‑subgradient Adam refinement
    # ------------------------------------------------------------------
    # allocate the remaining time
    time_now = time.monotonic()
    if time_now >= deadline:
        # out of time – return the current best
        best_vec = np.clip(best_vec, 0.0, 1000.0)
        return [float(x) for x in best_vec]

    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    last_improve = time.monotonic()

    # learning‑rate schedule for this phase (a bit more aggressive)
    lr_start2, lr_end2 = 0.30, 0.0005

    while time.monotonic() < deadline:
        it += 1
        # forward pass
        conv_self = _conv_self(a, fft_len_conv, conv_len)
        max_b = conv_self.max()
        J = 2.0 * n * max_b

        if J < best_score - 1e-12:
            best_score = J
            best_vec = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Phase 2 new best J = {best_score:.6f}")

        # schedules
        total_elapsed = time.monotonic() - start_time
        total_budget = deadline - start_time
        frac = total_elapsed / total_budget
        lr = lr_start2 * (1.0 - frac) + lr_end2 * frac

        # exact max‑subgradient
        grad = _max_subgradient(a, conv_self, conv_len)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        # occasional random block perturbation
        if rng.random() < 0.015:
            a_new = _random_block_perturb(a_new, rng)

        a = a_new

        # restart if stuck
        if time.monotonic() - last_improve > 25.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Phase 2 restart (stagnation) – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
