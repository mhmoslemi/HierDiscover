def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape: (1 + r*cos(theta))**power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L with linear taper controlled by gamma.
    When gamma==0 the blocks are uniform; gamma==1 makes them drop to
    zero at the inner border.
    """
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max weights (sum to 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate w.r.t. a (length‑n vector)."""
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1
    return 2.0 * full[start_idx:start_idx + a.size]


def _grad_hardmax(a: np.ndarray, mask: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Sub‑gradient of the *hard* max (max convolution) w.r.t. a.
    ``mask`` has the same length as the full self‑convolution (2n‑1)
    and contains non‑negative weights (typically 0/1 for the peak set).
    """
    mask_rev = mask[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(mask_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = mask_rev.size - 1
    return 2.0 * full[start_idx:start_idx + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """True objective = max self‑convolution (when sum(p)=1)."""
    conv = _self_conv(p, fft_len_self)
    return float(conv.max())

# Optimisation phases
# ----------------------------------------------------------------------


def _adam_refine(p_start: np.ndarray,
                 fft_len_self: int,
                 fft_len_grad: int,
                 time_limit: float,
                 rng: np.random.Generator,
                 best_J_ref: list,
                 best_p_ref: list) -> None:
    """
    Projected Adam optimiser on the soft‑max surrogate.
    ``best_J_ref`` and ``best_p_ref`` are single‑element lists that are
    mutated in‑place to keep track of the global best.
    """
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    start = time.monotonic()
    it = 0
    last_improve = start

    # schedules
    lr_start, lr_end = 0.5, 0.001
    alpha_start, alpha_end = 5.0, 1e5

    while time.monotonic() < time_limit:
        it += 1
        now = time.monotonic()
        frac = (now - start) / (time_limit - start)  # ∈[0,1]

        # learning‑rate (exponential decay)
        lr = lr_start * (0.5 ** frac) + lr_end * (1.0 - 0.5 ** frac)

        # temperature (linear increase)
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # convolution & soft‑max weights
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)

        # gradient of the surrogate
        grad = _grad_softmax(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto simplex
        a = proj_simplex(a)

        # evaluate true objective
        cur_J = _objective(a, fft_len_self, a.size)
        if cur_J < best_J_ref[0] - 1e-12:
            best_J_ref[0] = cur_J
            best_p_ref[0] = a.copy()
            last_improve = now
            print(f"[Adam] t={now - start:.1f}s  new best J={cur_J:.6f}")

        # occasional jitter to escape plateaus
        if rng.random() < 0.02:
            block_width = rng.integers(1, max(2, a.size // 20))
            start_idx = rng.integers(0, a.size - block_width + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block_width] *= factor
            a = proj_simplex(a_pert)

        # restart on stagnation
        if now - last_improve > 20.0:
            a = best_p_ref[0].copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=a.size))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[Adam] restart after stagnation – best J={best_J_ref[0]:.6f}")


def _hardmax_refine(p_start: np.ndarray,
                    fft_len_self: int,
                    fft_len_grad: int,
                    time_limit: float,
                    rng: np.random.Generator,
                    best_J_ref: list,
                    best_p_ref: list) -> None:
    """
    Sub‑gradient descent on the true max‑convolution.
    ``best_J_ref`` and ``best_p_ref`` are single‑element lists that are
    mutated in‑place.
    """
    a = p_start.copy()
    start = time.monotonic()
    last_improve = start

    # step‑size schedule (exponential decay from 0.4 → 0.001)
    step_start, step_end = 0.4, 0.001

    while time.monotonic() < time_limit:
        now = time.monotonic()
        frac = (now - start) / (time_limit - start)
        step = step_start * (step_end / step_start) ** frac

        # current convolution and maximal value
        conv = _self_conv(a, fft_len_self)
        M = conv.max()

        # tolerance for the peak set – small relative band
        eps = max(1e-9, 1e-4 * M)
        mask = (conv >= M - eps).astype(np.float64)

        # sub‑gradient (hard‑max)
        grad = _grad_hardmax(a, mask, fft_len_grad)

        # gradient step + projection
        a = a - step * grad
        a = proj_simplex(a)

        # evaluate objective
        cur_J = M  # same as _objective(a, ...) but cheaper
        if cur_J < best_J_ref[0] - 1e-12:
            best_J_ref[0] = cur_J
            best_p_ref[0] = a.copy()
            last_improve = now
            print(f"[Hard] t={now - start:.1f}s  new best J={cur_J:.6f}")

        # jitter every few iterations
        if rng.random() < 0.015:
            block = rng.integers(0, a.size)
            delta = np.exp(rng.normal(0.0, 0.08))
            a_pert = a.copy()
            a_pert[block] = max(0.0, a_pert[block] * delta)
            a = proj_simplex(a_pert)

        # restart on stagnation
        if now - last_improve > 25.0:
            a = best_p_ref[0].copy()
            noise = np.exp(rng.normal(0.0, 0.04, size=a.size))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[Hard] restart – best J={best_J_ref[0]:.6f}")

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm combines a cheap
    global sweep, Adam optimisation on a soft‑max surrogate and a final
    hard‑max sub‑gradient descent.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine problem size
    # --------------------------------------------------------------
    # Use the length of the previous best if available, otherwise 4000.
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = min(max(n, 256), 6000)      # keep FFT manageable
        except Exception:
            n = 4000
    else:
        n = 4000

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Global cheap exploration (seed generation)
    # --------------------------------------------------------------
    best_candidates = []          # list of (J, p) tuples, J = max conv
    def _push(p_vec: np.ndarray):
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > 8:
            best_candidates.pop()

    # a) uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push(uniform)

    # b) edge‑shape grid
    for frac in np.linspace(0.15, 0.45, 13):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            _push(_edge_shape(n, L, gamma))

    # c) cosine‑family grid
    for r in np.linspace(0.0, 1.0, 11):
        for pw in np.linspace(0.2, 3.5, 11):
            _push(_cosine_shape(n, r, pw))

    # d) Dirichlet draws (moderately concentrated)
    for _ in range(25):
        alpha = rng.uniform(0.05, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # e) previous best (if any)
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = uniform
            else:
                prev = proj_simplex(prev / prev.sum())
            _push(prev)
        except Exception:
            pass

    # Retain the current global best
    best_candidates.sort(key=lambda x: x[0])
    best_J, best_p = best_candidates[0][0], best_candidates[0][1].copy()

    # --------------------------------------------------------------
    # 3) Allocate time for the two optimisation phases
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # fallback – just return the current best
        scale = np.sqrt(2.0 * n)
        seq = np.clip(best_p * scale, 0.0, 1000.0)
        return [float(x) for x in seq]

    # 30 % for Adam, 60 % for hard‑max, 10 % left as buffer
    adam_budget = remaining * 0.30
    hard_budget = remaining * 0.60

    # --------------------------------------------------------------
    # 4) Phase A – Adam on the soft‑max surrogate
    # --------------------------------------------------------------
    per_seed_adam = adam_budget / max(1, len(best_candidates))
    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break
        time_limit = min(per_seed_adam,
                         deadline - time.monotonic())
        # mutable containers to share best across seeds
        best_J_ref = [best_J]
        best_p_ref = [best_p]
        _adam_refine(p0, fft_len_self, fft_len_grad,
                     time_limit, rng,
                     best_J_ref, best_p_ref)
        # update global best if improved
        if best_J_ref[0] < best_J - 1e-12:
            best_J, best_p = best_J_ref[0], best_p_ref[0].copy()
            print(f"[PhaseA] Global best improved to J={best_J:.6f}")

    # --------------------------------------------------------------
    # 5) Phase B – Hard‑max sub‑gradient refinement
    # --------------------------------------------------------------
    per_seed_hard = hard_budget / max(1, len(best_candidates))
    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break
        time_limit = min(per_seed_hard,
                         deadline - time.monotonic())
        best_J_ref = [best_J]
        best_p_ref = [best_p]
        _hardmax_refine(p0, fft_len_self, fft_len_grad,
                        time_limit, rng,
                        best_J_ref, best_p_ref)
        if best_J_ref[0] < best_J - 1e-12:
            best_J, best_p = best_J_ref[0], best_p_ref[0].copy()
            print(f"[PhaseB] Global best improved to J={best_J:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (scaled to √(2 n))
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)                # required L1‑norm
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
