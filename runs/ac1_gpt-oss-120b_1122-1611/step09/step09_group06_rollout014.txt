"""
Propose a candidate step‑function using
* a smooth approximation of the max‑convolution (log‑sum‑exp),
* Adam with Euclidean projection onto the probability simplex,
* multiple random restarts and occasional perturbations.

The smoother (beta) starts low (very smooth) and is increased
linearly so that the optimisation first sees a “soft’’ maximum,
then gradually focuses on the true ℓ∞‑norm.  The learning‑rate
decays linearly as well.  All heavy work (convolution, gradient)
is done with real FFTs (O(n log n)).  The routine respects a
1000 s budget (5 s safety margin) and returns the best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator):
    """Diverse seeds, all already normalised to sum = 1."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(7):
        cand.append(rng.dirichlet(np.ones(n) * 0.4))

    # previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        prev /= prev.sum()
        cand.append(prev)

    return cand


def _compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a⋆a) for a already on the simplex."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Return soft‑max weights w_i = exp(beta·(b_i−max))/∑exp … (numerically stable)."""
    b_max = b.max()
    exp_shift = np.exp(beta * (b - b_max))
    s = exp_shift.sum()
    if s == 0.0:
        # shouldn't happen, but fall back to a uniform weight
        return np.full_like(b, 1.0 / b.size)
    return exp_shift / s


def _gradient_softmax(a: np.ndarray,
                      fa: np.ndarray,
                      w: np.ndarray,
                      fft_len: int) -> np.ndarray:
    """
    Compute the gradient of the soft‑max smoothed max‑convolution.
    g = 2 * (w * a) where * denotes linear convolution.
    Both w and a are zero‑padded to ``fft_len``; the result is sliced to length n.
    """
    fw = np.fft.rfft(w, n=fft_len)
    # convolution w ⋆ a in frequency domain = fw * fa
    grad_full = np.fft.irfft(fw * fa, n=fft_len)
    # first n entries correspond to the required gradient (linear convolution)
    return 2.0 * grad_full[:a.size]


def _random_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Small random log‑normal perturbation of a random block – keeps sum=1."""
    n = a.size
    # choose block width (1 … n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.13))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method combines a smoothed max‑convolution objective with Adam
    optimisation on the probability simplex, and restarts with random
    perturbations when progress stalls.
    """
    rng = np.random.default_rng(seed)

    # ---- time handling (5 s safety margin) ----
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ---- problem size -------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # keep a size that is comfortable for FFT
        n = max(1024, min(n, 8192))
    else:
        n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ---- initialise a diverse pool of seeds -------------------------
    candidates = _initial_candidates(n, rng)
    best_vec = None
    best_score = np.inf
    for idx, cand in enumerate(candidates):
        cand = proj_simplex(cand.astype(np.float64))
        sc = _compute_J(cand, n, fft_len, conv_len)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
        if idx == 0:
            print(f"[{time.monotonic() - start_time:6.2f}s] "
                  f"uniform J = {sc:.6f}")

    print(f"[{time.monotonic() - start_time:6.2f}s] Initial best J = {best_score:.6f}")

    # ---- optimisation hyper‑parameters --------------------------------
    # Adam
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    # learning‑rate schedule (linear decay)
    lr_start, lr_end = 0.45, 0.006
    # smoothing (beta) schedule – start very smooth, end fairly sharp
    beta_start, beta_end = 0.25, 15.0
    # restart policy
    stagnation_limit = 22.0            # seconds without improvement → restart

    # ---- main optimisation loop (many short Adam runs) --------------
    next_restart = time.monotonic()
    it_global = 0
    while time.monotonic() < deadline:
        it_global += 1

        # Choose a starting point for this run
        if rng.random() < 0.30 and best_vec is not None:
            a = _random_perturb(best_vec, rng)
        else:
            # Dirichlet draw with modest concentration
            a = rng.dirichlet(np.ones(n) * 0.45)
        a = proj_simplex(a)

        # Adam moments
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        # Local time handling
        local_start = time.monotonic()
        last_improve = local_start

        # Run Adam until we hit a stagnation timeout or overall deadline
        while True:
            now = time.monotonic()
            if now >= deadline:
                break
            if now - last_improve > stagnation_limit:
                # local descent stalled – break to start a fresh restart
                break

            # fraction of total budget elapsed – drives schedules
            frac = (now - start_time) / total_time
            lr = lr_start * (1.0 - frac) + lr_end * frac
            beta_smooth = beta_start * (1.0 - frac) + beta_end * frac

            # ---- FFT of the current iterate (shared) ----
            fa = np.fft.rfft(a, n=fft_len)

            # convolution a⋆a (full linear)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

            # soft‑max weights over all shifts
            w = _softmax_weights(conv, beta_smooth)

            # gradient of the smoothed objective
            grad = _gradient_softmax(a, fa, w, fft_len)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            it = it_global  # bias‑correction uses global iteration count
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = proj_simplex(a_new)

            # evaluate (fast) objective
            cur_score = _compute_J(a_new, n, fft_len, conv_len)

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_vec = a_new.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] New global best J = {best_score:.6f}")

            a = a_new

            # occasional jitter to help escape shallow plateaus
            if rng.random() < 0.015:
                a = _random_perturb(a, rng)

        # after a run we report the time used – useful for debugging
        run_time = time.monotonic() - local_start
        if time.monotonic() < deadline:
            print(f"[{time.monotonic() - start_time:6.2f}s] Restart #{it_global} "
                  f"finished ({run_time:.2f}s), best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (clipped to the evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
