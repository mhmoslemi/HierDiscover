"""
Smooth‑max gradient descent for the step‑function problem.

* Initialise a small pool of diverse seeds (uniform, exponential,
  linear, two spikes, Dirichlet draws and the previous best if available).
* Keep the best seed (lowest J) and run a projected gradient descent on a
  smooth approximation of the max‑convolution (log‑sum‑exp).  The gradient
  is computed analytically via two FFT‑based convolutions, which is O(n log n).
* Learning‑rate and the smoothness parameter β are increased linearly with
  time, starting with a very smooth landscape (β≈5) and ending with a sharp
  approximation of the true max (β≈80).  After each step the iterate is
  projected back onto the probability simplex (non‑negative, sum = 1).
* Occasional random jitter and a stagnation restart help escape plateaus.
* The routine respects a 1000 s budget (5 s safety margin) and returns the
  best coefficient list found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (non‑negative, sum = 1)."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates(n: int, rng: np.random.Generator):
    """Diverse seed vectors, already normalised to sum = 1."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays (r<1) – left‑heavy
    for r in (0.999, 0.995, 0.99, 0.975, 0.95):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        prev /= prev.sum()
        cand.append(prev)

    return cand


def _fft_convolve(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution via FFT (result length = len(a)+len(b)-1)."""
    fa = np.fft.rfft(a, n=fft_len)
    fb = np.fft.rfft(b, n=fft_len)
    conv = np.fft.irfft(fa * fb, n=fft_len)
    return conv[: a.size + b.size - 1]


def _objective_J(a: np.ndarray, n: int, fft_len_full: int, conv_len_full: int) -> float:
    """
    Exact objective J = 2·n·max(a⋆a) for a already on the simplex.
    (Matches evaluate_sequence up to the clipping limits.)
    """
    conv = _fft_convolve(a, a, fft_len_full)[:conv_len_full]
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a ** 2))


def _smooth_max_gradient(a: np.ndarray,
                         n: int,
                         fft_len_full: int,
                         conv_len_full: int,
                         fft_len_wa: int,
                         conv_len_wa: int,
                         beta: float) -> np.ndarray:
    """
    Gradient of the smoothed max‑convolution
    (log‑sum‑exp with parameter beta), projected onto the simplex is
    performed outside this routine.
    """
    # --- self‑convolution ---
    conv_full = _fft_convolve(a, a, fft_len_full)[:conv_len_full]

    # --- soft‑max weights (stable) ---
    max_c = conv_full.max()
    exp_w = np.exp(beta * (conv_full - max_c))
    w = exp_w / exp_w.sum()                     # shape (2n‑1,)

    # --- gradients via two convolutions with w and reversed w ---
    # conv1 = w * a
    conv1 = _fft_convolve(w, a, fft_len_wa)[:conv_len_wa]
    # conv2 = w[::-1] * a
    conv2 = _fft_convolve(w[::-1], a, fft_len_wa)[:conv_len_wa]

    # slice the part that corresponds to the gradient entries
    start = n - 1
    end = start + n
    g = 2.0 * (conv1[start:end] + conv2[start:end])
    return g


def _descent_from_seed(seed_vec: np.ndarray,
                       n: int,
                       deadline: float,
                       rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the smoothed max‑convolution.
    Returns the best vector found and its J‑value.
    """
    # FFT sizes (pre‑computed for speed)
    conv_len_full = 2 * n - 1
    fft_len_full = _next_pow2(conv_len_full)

    conv_len_wa = n + conv_len_full - 1           # length of w⋆a
    fft_len_wa = _next_pow2(conv_len_wa)

    # start from a simplex‑projected seed
    a = proj_simplex(seed_vec.astype(np.float64))

    best_vec = a.copy()
    best_score = _objective_J(a, n, fft_len_full, conv_len_full)
    last_improve = time.monotonic()

    # schedules (linear in wall‑clock progress)
    beta_start, beta_end = 5.0, 80.0
    lr_start, lr_end = 0.8, 0.001

    # main loop
    while time.monotonic() < deadline:
        now = time.monotonic()
        progress = (now - deadline + (deadline - start_time)) / (deadline - start_time + 1e-12)

        beta = beta_start * (1.0 - progress) + beta_end * progress
        lr = lr_start * (1.0 - progress) + lr_end * progress

        # gradient of smooth max
        g = _smooth_max_gradient(a, n,
                                 fft_len_full, conv_len_full,
                                 fft_len_wa, conv_len_wa,
                                 beta)

        # projected gradient step
        a_new = a - lr * g
        a_new = proj_simplex(a_new)

        # evaluate exact objective (fast FFT version)
        score = _objective_J(a_new, n, fft_len_full, conv_len_full)

        if score < best_score - 1e-12:
            best_score = score
            best_vec = a_new.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] descent improved J = {best_score:.6f}")

        a = a_new

        # occasional jitter to help escape flat regions
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # restart if stagnation > 15 s
        if now - last_improve > 15.0:
            a = _random_block_perturb(best_vec, rng)
            last_improve = now
            # no need to recompute best_score here – it stays unchanged

    return best_vec, best_score
