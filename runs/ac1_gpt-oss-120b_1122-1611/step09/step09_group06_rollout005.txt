"""
Propose a step‑function (non‑negative sequence) that minimises

    J(a) = 2·n·max(a∗a) / (sum(a))²

with the invariant that scaling a does not change the objective.  We therefore
work on the probability simplex (sum = 1) and optimise the *soft‑max* of the
self‑convolution:

    f_α(a) = (1/α)·log∑_k exp(α·(a∗a)_k).

For small α the objective is smooth (close to the L₂‑norm of the
convolution) which gives a good global search direction; α is increased
linearly so that the final iterations concentrate on the true
∞‑norm.  An Adam optimiser supplies adaptive step‑sizes and a Euclidean
projection keeps the iterate on the simplex.  Periodic random
perturbations and stagnation restarts help escape plateaus.  When the
time budget is almost exhausted a few cheap simulated‑annealing moves
fine‑tune the best-found vector.

The routine respects the 1000 s limit (5 s safety margin) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator):
    """A handful of diverse seeds (all sum = 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing (largest at start, smallest at end)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends (tiny mass, just for diversity)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # include previous best if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64).copy()
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        prev /= prev.sum()
        cand.append(prev)

    return cand


def _conv_and_max(a: np.ndarray, fft_len: int, conv_len: int):
    """Full self‑convolution and its maximum (value and index)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    idx = int(conv.argmax())
    return conv, idx, float(conv.max())


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method combines a smooth‑max Adam descent with occasional random
    perturbations and a brief simulated‑annealing fine‑tuning phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) time bookkeeping (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 1) problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # keep size moderate – large enough for a good bound but cheap enough
        n = max(512, min(n, 4096))
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # for a∗a
    corr_len = _next_pow2(3 * n - 2)        # for a∗w (linear convolution)

    # --------------------------------------------------------------
    # 2) initialise a small pool of candidates, pick the best
    # --------------------------------------------------------------
    candidates = _initial_candidates(n, rng)
    best_vec = None
    best_score = np.inf

    for cand in candidates:
        cand = proj_simplex(cand.astype(np.float64))
        conv, _, max_val = _conv_and_max(cand, fft_len, conv_len)
        score = 2.0 * n * max_val
        if score < best_score:
            best_score = score
            best_vec = cand.copy()

    print(f"[{time.monotonic() - start_time:6.2f}s] "
          f"Initial best J = {best_score:.6f} (n={n})")

    # --------------------------------------------------------------
    # 3) Adam optimisation on the smoothed max‑convolution objective
    # --------------------------------------------------------------
    # hyper‑parameters (linear schedules)
    lr_start, lr_end = 0.6, 0.001          # learning‑rate
    alpha_start, alpha_end = 0.5, 80.0    # smoothing parameter

    a = best_vec.copy()
    m = np.zeros_like(a)      # first moment
    v = np.zeros_like(a)      # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    it = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time

        # schedules
        lr = lr_start * (1.0 - frac) + lr_end * frac
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ----------------------------------------------------------
        # a) self‑convolution and soft‑max weights
        # ----------------------------------------------------------
        conv, _, max_conv = _conv_and_max(a, fft_len, conv_len)

        # soft‑max over the convolution values (always numerically stable)
        shifted = conv - max_conv                     # ≤ 0
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # ----------------------------------------------------------
        # b) gradient of the smoothed objective:
        #    ∂f/∂a = 2·(a * w)   (linear convolution, not reversed)
        # ----------------------------------------------------------
        Fa = np.fft.rfft(a, n=corr_len)
        Fw = np.fft.rfft(w, n=corr_len)
        conv_aw = np.fft.irfft(Fa * Fw, n=corr_len)[:n]
        grad = 2.0 * conv_aw

        # ----------------------------------------------------------
        # c) Adam update + projection onto the simplex
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        # exact objective (no smoothing) for book‑keeping
        _, _, max_conv_new = _conv_and_max(a_new, fft_len, conv_len)
        score_new = 2.0 * n * max_conv_new

        if score_new < best_score - 1e-12:
            best_score = score_new
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] "
                  f"Adam improvement – J = {best_score:.6f}")

        a = a_new

        # ----------------------------------------------------------
        # d) occasional random jitter to shake out of plateaus
        # ----------------------------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # ----------------------------------------------------------
        # e) stagnation restart
        # ----------------------------------------------------------
        if time.monotonic() - last_improve > 30.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] "
                  f"Restart after stagnation – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final cheap simulated‑annealing polishing (≈5 s)
    # --------------------------------------------------------------
    time_left = deadline - time.monotonic()
    if time_left > 5.0:
        a = best_vec.copy()
        cur_score = best_score
        # simple temperature schedule – we keep it low because we are almost done
        T_start, T_end = 0.02, 0.0001
        sa_deadline = time.monotonic() + time_left
        while time.monotonic() < sa_deadline:
            T = T_start  # essentially greedy now
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            eps = rng.exponential(scale=0.005)
            eps = min(eps, a[i])          # cannot make a[i] negative
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()          # stay on simplex
            _, _, max_conv_new = _conv_and_max(a_new, fft_len, conv_len)
            new_score = 2.0 * n * max_conv_new
            delta = new_score - cur_score
            accept = False
            if delta < 0.0:
                accept = True
            elif rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True
            if accept:
                a = a_new
                cur_score = new_score
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_vec = a.copy()
                    print(f"[{time.monotonic() - start_time:6.2f}s] "
                          f"SA polishing – J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluation limits)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
