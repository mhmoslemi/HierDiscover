def subgradient_maxconv(a):
    conv = np.fft.irfft(np.fft.rfft(a, n=fft_len_self) * np.fft.rfft(a, n=fft_len_self), n=fft_len_self)[:conv_len]
    maxv = conv.max()
    active = (conv >= maxv - 1e-12)   # bool vector length conv_len
    w = active.astype(np.float64)  # weight each active index equally
    if w.sum() == 0:
        w[...] = 1.0/len(w)
    else:
        w/= w.sum()
    # compute gradient: sum_k w_k * grad_k = avg of grads.
    # grad_k_i = 2 * a_{k-i} if valid else 0.
    # Equivalent to gradient = 2 * (a convolve reverse(w)) padded to length n.
    # Use FFT: note a length n, w length conv_len (2n-1)
    # pad a to fft_len_self (>=conv_len)
    w_rev = w[::-1]
    # convolution length: len(a)+len(w_rev)-1 = n + conv_len -1 = n + (2n-1) -1 = 3n -2
    fft_len = _next_pow2(3*n - 2)
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(fa*fw, n=fft_len)
    # choose indices start = w_rev.size - 1 (2n-2) to start + n
    start = w_rev.size - 1
    grad = 2*full[start:start+n]
    return grad

The optimizer works on the *simplex* (non‑negative entries that sum to 1) so that the denominator of `evaluate_sequence` is constant.  
For a vector `a` on the simplex the score reduces to

```
J(a) = 2·n·max_k (a * a)[k]          (n = len(a))
```

Hence we only have to minimise the maximum entry of the self‑convolution.

The algorithm is built from three complementary phases, each of which respects the remaining time budget:

| Phase | Goal | Method |
|------|------|--------|
| **1 – Global sampling** | produce a diverse set of promising shapes for several lengths | uniform / exponential / linear ramps + many Dirichlet draws (different concentrations). |
| **2 – Sub‑gradient descent** | quickly move every candidate towards a local optimum of the *exact* max‑convolution (no smoothing) | At each step the sub‑gradient of `max_k (a*a)[k]` is computed analytically (average of the gradients of the active ties). A simple diminishing step‑size together with Euclidean projection onto the simplex drives the search. |
| **3 – Simulated annealing (SA)** | escape from shallow basins and polish the best candidate | Repeatedly transfer a small amount of mass from one index to another; the move is accepted according to a temperature schedule that slowly cools to zero. |
| **4 – Final pair‑wise hill‑climb** | squeeze out any remaining tiny improvements | Very small mass transfers are tried greedily. |

All heavy operations are performed with FFTs (`O(n log n)`) and the code checks the wall‑clock time after every batch, aborting gracefully before the 5 s safety margin.

The implementation is completely self‑contained, uses only `numpy` and `time`, and follows the required signature `propose_candidate()`.

```python
"""Search for a non‑negative step function that minimises `evaluate_sequence`.

The algorithm proceeds through
  1) a diverse global sampling for several problem sizes,
  2) a projected sub‑gradient descent on the exact max‑convolution,
  3) a simulated‑annealing refinement,
  4) a short greedy pair‑wise hill‑climb.

The whole search respects a 1000 s wall‑clock budget (5 s safety margin) and
always returns the best vector seen.
"""
import time
import numpy as np


# ----------------------------------------------------------------------
# Linear‑algebra utilities (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution helpers
# ----------------------------------------------------------------------
def compute_conv(a: np.ndarray, n: int, fft_len: int) -> np.ndarray:
    """Self‑convolution of length 2·n‑1 using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fa * fa, n=fft_len)
    return conv_full[: 2 * n - 1]


def compute_J(a: np.ndarray, n: int, fft_len_self: int) -> float:
    """Exact objective J = 2·n·max(a*a) for a vector that already sums to 1."""
    conv = compute_conv(a, n, fft_len_self)
    maxv = conv.max()
    return float(2.0 * n * maxv)


def compute_conv_and_J(a: np.ndarray, n: int, fft_len_self: int):
    """Return full self‑convolution and the exact objective J."""
    conv = compute_conv(a, n, fft_len_self)
    maxv = conv.max()
    J = float(2.0 * n * maxv)
    return conv, J


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution
# ----------------------------------------------------------------------
def subgrad_maxconv(a: np.ndarray, n: int,
                    fft_len_self: int, conv_len: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)[k]  (the inner max of J).
    The active set is taken as all k whose value is within 1e‑12·max.
    """
    # 1) compute the full convolution
    conv = compute_conv(a, n, fft_len_self)

    maxv = conv.max()
    eps = 1e-12 * maxv
    active = conv >= maxv - eps          # bool mask, length 2·n‑1

    # 2) build a uniform weight vector over the active indices
    w = active.astype(np.float64)
    if w.sum() == 0.0:
        w[:] = 1.0 / w.size
    else:
        w /= w.sum()

    # 3) gradient of a single entry k is 2·a shifted by k (reversed)
    #    The averaged sub‑gradient is the convolution of a with w_rev.
    w_rev = w[::-1]                      # reverse the weight vector
    fft_len = _next_pow2(3 * n - 2)       # length needed for a * w_rev
    fa = np.fft.rfft(a, n=fft_len)
    fw = np.fft.rfft(w_rev, n=fft_len)
    full = np.fft.irfft(fa * fw, n=fft_len)

    start = w_rev.size - 1                # = 2·n‑2
    grad = 2.0 * full[start:start + n]    # length n
    return grad


# ----------------------------------------------------------------------
# Core search routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The routine respects a wall‑clock budget of ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time budget (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_score = np.inf
    best_vec = None
    best_n = None

    # ------------------------------------------------------------------
    # 1) problem sizes to try (smaller sizes are cheap, large sizes give
    #    the best asymptotics).  All vectors are projected onto the simplex,
    #    i.e. they sum to 1, so J = 2·n·maxconv.
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]

    for n in lengths:
        if time.monotonic() > deadline:
            break
        conv_len = 2 * n - 1
        fft_len_self = _next_pow2(conv_len)

        # ------------------------------------------------------------------
        # 2) generate a portfolio of diverse seeds (already normalised)
        # ------------------------------------------------------------------
        seeds = []

        # uniform
        seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

        # several exponential decays with random rates
        for r in rng.uniform(0.98, 0.9999, size=5):
            v = r ** np.arange(n, dtype=np.float64)
            v /= v.sum()
            seeds.append(v)

        # linear decreasing ramp
        v = np.linspace(1.0, 0.0, n, dtype=np.float64)
        v /= v.sum()
        seeds.append(v)

        # two spikes at the ends
        v = np.zeros(n, dtype=np.float64)
        v[0] = v[-1] = 0.5
        seeds.append(v)

        # half‑heavy, half‑zero
        half = n // 2
        v = np.zeros(n, dtype=np.float64)
        v[:half] = 2.0 / n
        seeds.append(v)

        # several Dirichlet draws (different concentrations)
        for concentration in [0.05, 0.2, 0.5, 1.0]:
            for _ in range(4):
                seeds.append(rng.dirichlet(np.ones(n) * concentration))

        # include the previous global best if it exists (trim or pad)
        if best_vec is not None:
            old = np.asarray(best_vec, dtype=np.float64)
            if old.size >= n:
                cand = old[:n]
            else:
                cand = np.concatenate([old, np.zeros(n - old.size)])
            cand = proj_simplex(cand)
            seeds.append(cand)

        # ------------------------------------------------------------------
        # 3) sub‑gradient descent for each seed
        # ------------------------------------------------------------------
        for seed_vec in seeds:
            if time.monotonic() > deadline:
                break
            a = seed_vec.copy()
            # simple diminishing step size schedule
            step0 = 0.7
            max_iters = 800  # enough for a quick local improvement
            prev_J = compute_J(a, n, fft_len_self)

            for it in range(1, max_iters + 1):
                grad = subgrad_maxconv(a, n, fft_len_self, conv_len)
                step = step0 / np.sqrt(it)          # 1/√t decay
                a_new = a - step * grad
                a_new = proj_simplex(a_new)

                cur_J = compute_J(a_new, n, fft_len_self)

                # accept if better; otherwise keep the old point
                if cur_J < prev_J:
                    a = a_new
                    prev_J = cur_J
                    if cur_J < best_score - 1e-12:
                        best_score = cur_J
                        best_vec = a_new.copy()
                        best_n = n
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] New best (sub‑grad) J={best_score:.6f} n={n}")

                # early termination of this seed if no progress for a while
                if it % 200 == 0 and cur_J >= prev_J - 1e-12:
                    break

        # ------------------------------------------------------------------
        # 4) Simulated annealing from the best vector of this length
        # ------------------------------------------------------------------
        if best_vec is not None and best_n == n:
            a = best_vec.copy()
            cur_J = best_score

            # temperature schedule (geometric cooling)
            T_start = cur_J * 0.4 + 1e-6
            T_end = cur_J * 1e-4 + 1e-12
            total_sa_iters = 200_000

            for k in range(total_sa_iters):
                if time.monotonic() > deadline:
                    break
                # linear interpolation of temperature
                frac = k / total_sa_iters
                T = T_start * (1.0 - frac) + T_end * frac

                # propose a mass transfer
                i = rng.integers(0, n)
                if a[i] <= 0.0:
                    continue
                j = rng.integers(0, n)
                if i == j:
                    continue

                # move at most 10 % of the source mass
                delta = rng.random() * 0.1 * a[i]

                a_cand = a.copy()
                a_cand[i] -= delta
                a_cand[j] += delta

                # compute new objective
                _, cand_J = compute_conv_and_J(a_cand, n, fft_len_self)

                dJ = cand_J - cur_J
                if dJ < 0 or rng.random() < np.exp(-dJ / (T + 1e-12)):
                    a = a_cand
                    cur_J = cand_J
                    if cur_J < best_score - 1e-12:
                        best_score = cur_J
                        best_vec = a_cand.copy()
                        best_n = n
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] New best (SA)    J={best_score:.6f} n={n}")

                # occasional random restart to avoid stagnation
                if (k + 1) % 30_000 == 0:
                    a = best_vec.copy()
                    cur_J = best_score

        # ------------------------------------------------------------------
        # 5) tiny pair‑wise polishing for this length (max 5 s)
        # ------------------------------------------------------------------
        if best_vec is not None and best_n == n:
            a = best_vec.copy()
            cur_J = best_score
            hill_start = time.monotonic()
            while time.monotonic() < deadline and (time.monotonic() - hill_start) < 5.0:
                i, j = rng.integers(0, n, size=2)
                if i == j or a[i] <= 0.0:
                    continue
                # tiny transfer (≤ 0.003 of source)
                delta = rng.random() * min(a[i], 0.003)
                a_cand = a.copy()
                a_cand[i] -= delta
                a_cand[j] += delta
                cand_J = compute_J(a_cand, n, fft_len_self)
                if cand_J < cur_J - 1e-12:
                    a = a_cand
                    cur_J = cand_J
                    best_vec = a_cand.copy()
                    best_score = cand_J
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] Pairwise improv J={best_score:.6f} n={n}")

    # ------------------------------------------------------------------
    # 6) fallback in the unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_vec is None:
        n = 1024
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_J(best_vec, n, _next_pow2(2 * n - 1))

    # ------------------------------------------------------------------
    # 7) final clipping to respect the limits of evaluate_sequence
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # return a plain Python list as required by the evaluation harness
    return [float(v) for v in best_vec]
```
