"""
Propose a step‑function (non‑negative heights) that approximately minimises
``evaluate_sequence``.  The optimiser consists of three stages:

0️⃣  Fast parametric sweep over several families of symmetric shapes
    (exponential, power‑law, block‑edge, triangular, sinusoidal, Dirichlet)
    on a handful of modest lengths (≈1 k – 8 k).  The best candidate is kept.

1️⃣  Projected sub‑gradient descent on the *hard* max‑convolution
    (the true objective).  A linearly decaying stepsize and occasional
    random block perturbations keep the search out of plateaus.

2️⃣  Projected Adam optimiser that minimises a temperature‑controlled
    soft‑max surrogate of the max‑convolution – this refines the solution
    to high precision.

All helpers are top‑level, the algorithm respects a 1000 s time budget
(with a 5 s safety margin) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) (assumes sum(a)=1)."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def self_convolution(a: np.ndarray,
                     fft_len_self: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv

# Main optimisation routine
# ----------------------------------------------------------------------

        # ------------------------------------------------------------------
        for lam in np.linspace(0.0, 5.0, 51):
            if time.monotonic() >= sweep_deadline:
                break
            i = np.arange(n, dtype=np.float64)
            a = np.exp(-lam * i) + np.exp(-lam * (n - 1 - i))
            a /= a.sum()
            score = compute_J(a, n, fft_len_self, conv_len)
            if score < best_score:
                best_score, best_vec, best_n = score, a.copy(), n

        # ------------------------------------------------------------------
        # 0b) power‑law decay from both ends
        # ------------------------------------------------------------------
        for p in np.linspace(0.0, 4.0, 41):
            if time.monotonic() >= sweep_deadline:
                break
            i = np.arange(1, n + 1, dtype=np.float64)
            a = (i ** (-p)) + ((n - i + 1) ** (-p))
            a /= a.sum()
            score = compute_J(a, n, fft_len_self, conv_len)
            if score < best_score:
                best_score, best_vec, best_n = score, a.copy(), n

        # ------------------------------------------------------------------
        # 0c) block‑edge families (two heavy edge blocks + interior)
        # ------------------------------------------------------------------
        for w_frac in np.linspace(0.02, 0.45, 9):
            if time.monotonic() >= sweep_deadline:
                break
            w = int(round(w_frac * n))
            if w < 1 or 2 * w >= n:
                continue
            interior_len = n - 2 * w
            for m_edge in np.linspace(0.05, 0.45, 9):
                if time.monotonic() >= sweep_deadline:
                    break
                if interior_len == 0 and not np.isclose(2 * m_edge, 1.0):
                    # no interior – edge masses must sum to 1
                    continue
                edge_val = m_edge / w
                interior_val = (1.0 - 2 * m_edge) / interior_len if interior_len > 0 else 0.0
                if interior_val < 0:
                    continue
                a = np.full(n, interior_val, dtype=np.float64)
                a[:w] = edge_val
                a[-w:] = edge_val
                a /= a.sum()                     # safety – eliminates rounding errors
                score = compute_J(a, n, fft_len_self, conv_len)
                if score < best_score:
                    best_score, best_vec, best_n = score, a.copy(), n

        # ------------------------------------------------------------------
        # 0d) triangular (peak in centre)
        # ------------------------------------------------------------------
        i = np.arange(n, dtype=np.float64)
        a = (i + 1) * (n - i)
        a /= a.sum()
        score = compute_J(a, n, fft_len_self, conv_len)
        if score < best_score:
            best_score, best_vec, best_n = score, a.copy(), n

        # ------------------------------------------------------------------
        # 0e) sinusoidal families (different powers)
        # ------------------------------------------------------------------
        theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
        base = np.sin(theta)
        for alpha in [1, 2, 3, 4]:
            a = base ** alpha
            a /= a.sum()
            score = compute_J(a, n, fft_len_self, conv_len)
            if score < best_score:
                best_score, best_vec, best_n = score, a.copy(), n

        # ------------------------------------------------------------------
        # 0f) random Dirichlet draws (various concentrations)
        # ------------------------------------------------------------------
        for conc in [0.02, 0.05, 0.1, 0.3, 0.7, 1.5]:
            if time.monotonic() >= sweep_deadline:
                break
            for _ in range(4):
                a = rng.dirichlet(np.ones(n) * conc)
                score = compute_J(a, n, fft_len_self, conv_len)
                if score < best_score:
                    best_score, best_vec, best_n = score, a.copy(), n

        # ------------------------------------------------------------------
        # 0g) interpolate previous best construction (if supplied)
        # ------------------------------------------------------------------
        if prev_global is not None:
            if prev_global.size != n:
                x_old = np.linspace(0.0, 1.0, prev_global.size)
                x_new = np.linspace(0.0, 1.0, n)
                a = np.interp(x_new, x_old, prev_global)
            else:
                a = prev_global.copy()
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            score = compute_J(a, n, fft_len_self, conv_len)
            if score < best_score:
                best_score, best_vec, best_n = score, a.copy(), n

        # time‑check after each n
        if time.monotonic() >= sweep_deadline:
            break

    elapsed = time.monotonic() - start
    print(f"[{elapsed:6.2f}s] param sweep – best J={best_score:.6f} (n={best_n})")

    # --------------------------------------------------------------
    # 2) Projected sub‑gradient descent on the hard max‑convolution
    # --------------------------------------------------------------
    # allocate the next ~35 % of the remaining time to this phase
    subgrad_deadline = start + (deadline - start) * 0.65
    n = best_n
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)

    a = best_vec.copy()
    best_score_sg = best_score
    best_vec_sg = a.copy()
    last_improve = time.monotonic()
    eta_start, eta_end = 0.8, 1e-4

    while time.monotonic() < subgrad_deadline:
        conv = self_convolution(a, fft_len_self, conv_len)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the maximal entry (hard max)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # linear stepsize schedule
        frac = (time.monotonic() - start) / (subgrad_deadline - start)
        eta = eta_start * (1.0 - frac) + eta_end * frac

        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s

        cur_score = compute_J(a_new, n, fft_len_self, conv_len)
        if cur_score < best_score_sg - 1e-12:
            best_score_sg = cur_score
            best_vec_sg = a_new.copy()
            a = a_new
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:6.2f}s] sub‑grad new best J={best_score_sg:.6f}")
        else:
            a = a_new

        # occasional random block jitter (helps escape plateaus)
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # restart if stagnation lasts > 18 s
        if time.monotonic() - last_improve > 18.0:
            a = best_vec_sg.copy()
            a = a * np.exp(rng.normal(0.0, 0.04, size=n))
            a = proj_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:6.2f}s] sub‑grad restart")

    # --------------------------------------------------------------
    # 3) Adam refinement with a soft‑max surrogate (≈35 % of time)
    # --------------------------------------------------------------
    adam_deadline = deadline
    a = best_vec_sg.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.monotonic()
    start_adam = time.monotonic()
    total_adam = adam_deadline - start_adam
    fft_len_grad = _next_pow2(3 * n - 2)   # length needed for a ⋆ w_rev convolution

    while time.monotonic() < adam_deadline:
        it += 1
        now = time.monotonic()
        prog = (now - start_adam) / total_adam   # ∈[0,1]

        # learning‑rate schedule (exponential decay)
        lr = 0.25 * (1.0 - prog) + 0.001

        # temperature schedule (slowly harden the soft‑max)
        alpha = 2.0 * (1.0 - prog) + 3000.0

        # ----- convolution of current iterate -----
        conv = self_convolution(a, fft_len_self, conv_len)

        # ----- soft‑max weights over convolution entries -----
        c_max = conv.max()
        shifted = conv - c_max            # improve numerical stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        start_idx = w_rev.size - 1          # aligns the linear‑convolution result
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- Adam update + projection onto the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        cur_score = compute_J(a, n, fft_len_self, conv_len)
        if cur_score < best_score_sg - 1e-12:
            best_score_sg = cur_score
            best_vec_sg = a.copy()
            last_improve = now
            print(f"[{now - start:6.2f}s] Adam new best J={best_score_sg:.6f}")

        # occasional jitter to avoid getting stuck in a flat region
        if rng.random() < 0.015:
            a = _random_block_perturb(a, rng)

        # restart after 30 s without improvement
        if now - last_improve > 30.0:
            a = best_vec_sg.copy()
            a = a * np.exp(rng.normal(0.0, 0.06, size=n))
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start:6.2f}s] Adam restart")

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clip to evaluation limits)
    # --------------------------------------------------------------
    best_vec_sg = np.clip(best_vec_sg, 0.0, 1000.0)
    return [float(x) for x in best_vec_sg]
