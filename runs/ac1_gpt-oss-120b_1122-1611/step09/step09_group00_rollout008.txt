"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.  

The algorithm proceeds in three stages:

1. **Smart initialisation** – try a few parametric families (Gaussian,
   power‑law, uniform) and keep the best one.
2. **Smooth‑max gradient descent** – minimise the p‑norm
   ``(∑ (a∗a)_k^p)^{1/p}`` with an Adam‑style optimiser while the vector
   stays on the probability simplex (sum = 1, a_i ≥ 0).  The exponent ``p``
   is increased (4 → 16 → 64 → 256) so that the surrogate becomes tighter
   to the true ``max_k (a∗a)_k``.
3. **Stochastic local refinement** – directed tiny mass‑transfer moves
   (source = index with largest contribution to the current maximum,
   target = index with smallest contribution) are accepted if they
   improve the exact objective.  Occasional jitter prevents stagnation.

All work is timed; the routine stops a few seconds before the given
budget and returns the best sequence seen so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level only)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT of length ``fft_len``.
    Returns the (2*len(a)-1)‑vector of convolution values.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a∗a)[k]`` for the set ``active`` of
    indices attaining the current maximum.  For each active lag k,
    the contribution to ∂max/∂a_i is a_{k-i} (counted twice because the
    convolution is symmetric).
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    grad *= 2.0                         # each pair appears twice
    return grad


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """Exact score for a simplex‑normalised vector (sum = 1)."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return float(2.0 * n * conv.max())


def _smooth_max_gradient(a: np.ndarray, p: int, fft_len: int) -> np.ndarray:
    """
    Gradient of the smooth approximation
        F_p(a) = ( Σ_k (a∗a)[k]^p )^{1/p}
    w.r.t. a (ignoring the overall 2 n factor, which is constant).
    The formula reduces to
        grad = 2 * corr( w , a )
    where w_k = (a∗a)[k]^{p‑1}.
    """
    n = a.size
    conv = _conv_fft(a, fft_len)               # length 2n‑1
    # avoid under‑flow / zero‑powers
    eps = 1e-12
    w = np.maximum(conv, eps) ** (p - 1)       # weight vector
    # linear correlation w ⋆ a (size 3n‑2) via FFT
    L = fft_len                                 # already ≥ 3n   (safe)
    w_fft = np.fft.rfft(w, n=L)
    a_fft = np.fft.rfft(a, n=L)
    corr = np.fft.irfft(w_fft * np.conj(a_fft), n=L)   # length L
    offset = len(w) - 1                         # 2n‑2
    grad = 2.0 * corr[offset : offset + n]     # exactly the needed slice
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function (list of heights) that

    """

    # ------------------------------------------------------------------

        if s > 0.0:

        else:

    else:

    start_candidates = []

    n_default = 4096
    if a0 is not None:
