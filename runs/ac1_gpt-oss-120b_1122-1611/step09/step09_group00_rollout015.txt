def propose_candidate(seed=..., budget_s=1000.0):
   rng = np.random.default_rng(seed)

   # determine n: use existing global or default 4096
   if "height_sequence_1" in globals():
       n = len(globals()["height_sequence_1"])
   else: n = 4096

   # Determine bin count
   K = 128 if n >= 4096 else max(8, n//32)
   block = n // K

   fft_len = _next_pow2(3 * n)

   a = np.full(n, 1.0/n) # initial uniform

   best = a.copy()
   best_score = 2.0 * n * _conv_fft(best, fft_len).max()
   print(f"initial uniform score = {best_score:.6f}")

   start_time = time.monotonic()
   deadline = start_time + max(0.0, float(budget_s) - 5.0)

   # Phase 0: random bin search
   iteration = 0
   while time.monotonic() < deadline * 0.4: # allocate 40% time
       bins = rng.dirichlet(np.ones(K))
       seq = np.repeat(bins, block)
       # if n % K != 0, fill leftover with random extra part
       if seq.size < n:
           seq = np.concatenate([seq, bins[:n - seq.size]])
       # evaluate
       conv = _conv_fft(seq, fft_len)
       score = 2.0 * n * conv.max()
       if score < best_score:
           best_score = score
           best = seq.copy()
           elapsed = time.monotonic() - start_time
           print(f"[{elapsed:6.2f}s] new best (rand) = {best_score:.6f}")
       iteration += 1
       if iteration % 1000 == 0:
           elapsed = time.monotonic() - start_time
           print(f"[{elapsed:6.2f}s] random search iter={iteration}, best={best_score:.6f}")

"""
A multi‑phase optimizer for the max‑convolution objective.
Phases
-------
0. Initialise – use the previous best sequence if available, otherwise a uniform vector.
1. Random exploration – generate many candidates using
   • Dirichlet draws on a coarse binning (with optional monotone ordering)
   • Power‑law shapes a_i ∝ (i+1)**‑p.
   Keep the best candidate.
2. Exponentiated‑gradient descent – sub‑gradient computed from the (tiny) active set
   of indices attaining the current maximum of a∗a.
3. Additive sub‑gradient descent – standard projected sub‑gradient updates with a
   diminishing step size.
4. Stochastic local refinement – directed mass‑transfer moves guided by the
   gradient of the currently most‑violated convolution index, plus occasional
   jitter.
The routine respects the simplex constraint (∑a_i=1) and the non‑negativity
requirement.  The total running time is limited to ``budget_s`` seconds
(the harness reserves a 5 s safety margin).  The best sequence discovered
is returned as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum).
    For each active index k, the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    The method combines a random global search on a low‑dimensional
    parametrisation with exponentiated‑gradient, additive sub‑gradient,
    and stochastic local refinement phases.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` and normalise to sum = 1
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
    else:
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    fft_len = _next_pow2(3 * n)           # enough for linear convolution

    # --------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best = a0.copy()
    best_score = 2.0 * n * _conv_fft(best, fft_len).max()
    print(f"[0.00s] initial score = {best_score:.6f}")

            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (move={it_loc})")
            if best_score <= 1.5025:
                print("[early stop] target reached in local phase")
                break
        # keep improvement (or at least no worsening)
        cur_max = conv.max()
        if score_c <= 2.0 * n * cur_max:
            a = a_cand

        # occasional jitter to shake the solution
        if it_loc % 4000 == 0 and it_loc > 0:
            width = max(1, n // 80)
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jit = a.copy()
            a_jit[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jit)

        it_loc += 1
        if it_loc % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local iter={it_loc}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 7) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
