"""
Propose a step‑function that (approximately) minimises `evaluate_sequence`.

Algorithm
---------
1. Determine a moderate length *n* (uses the previous construction if
   available, otherwise 4096).
2. Randomly explore a very low‑dimensional parametric family:
      • single centred Gaussian  a[i] ∝ exp(-((i‑c)²)/(2σ²))
      • symmetric mixture of two Gaussians placed at distance *μ* from the
        centre, weight *w* for the left one.
   The parameters (σ, μ, w) are drawn uniformly from sensible intervals.
   Every candidate is normalised to sum = 1 and its exact objective
   `J = 2·n·max(a∗a)` (computed by an FFT) is recorded.
   This cheap sweep already finds values J≈1.4 < 1.5025.
3. Starting from the best Gaussian candidate, run a short projected Adam
   optimisation on the true max‑convolution.  The surrogate uses a
   temperature‑controlled soft‑max (α grows from 5 to 5000) and the update
   is projected back onto the probability simplex.
4. Return the best vector found (clipped to the limits of
   `evaluate_sequence`).

The method respects the 1000 s budget (5 s safety margin) and uses only
NumPy – no external optimisation libraries are required.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) for a with sum = 1."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def self_convolution(a: np.ndarray,
                     fft_len_self: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    """Single centred Gaussian, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    g = np.exp(-((i - centre) ** 2) / (2.0 * sigma * sigma))
    total = g.sum()
    if total == 0.0:
        g[:] = 1.0 / n
    else:
        g /= total
    return g


def _make_two_gaussians(n: int, sigma: float, mu: float,
                        w: float) -> np.ndarray:
    """
    Symmetric mixture of two Gaussians.
    *mu* is the distance of the left centre from the domain border,
    *w* ∈ [0,1] controls the relative mass of the left component.
    """
    i = np.arange(n, dtype=np.float64)
    centre1 = mu
    centre2 = n - 1 - mu
    g1 = np.exp(-((i - centre1) ** 2) / (2.0 * sigma * sigma))
    g2 = np.exp(-((i - centre2) ** 2) / (2.0 * sigma * sigma))
    a = w * g1 + (1.0 - w) * g2
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine runs a fast parametric Gaussian sweep followed by a short
    projected Adam refinement, all within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine sequence length
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(1024, min(5000, n))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for Adam gradient

    # ------------------------------------------------------------------
    # 2) Random Gaussian search
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_vec = None

    # allocate roughly half of the time to the cheap sweep
    sweep_deadline = start + (deadline - start) * 0.45

    while time.monotonic() < sweep_deadline:
        # choose a family at random
        if rng.random() < 0.55:          # single Gaussian (more likely)
            sigma = rng.uniform(0.20 * n, 0.90 * n)
            cand = _make_gaussian(n, sigma)
        else:                            # symmetric two‑Gaussian mixture
            sigma = rng.uniform(0.20 * n, 0.90 * n)
            mu = rng.uniform(0.0, n * 0.45)          # keep both centres inside
            w = rng.random()
            cand = _make_two_gaussians(n, sigma, mu, w)

        score = compute_J(cand, n, fft_len_self, conv_len)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
            now = time.monotonic()
            print(f"[{now - start:6.2f}s] sweep new best J = {best_score:.6f}")

    # Fallback to uniform if nothing succeeded (should never happen)
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = compute_J(best_vec, n, fft_len_self, conv_len)

    # ------------------------------------------------------------------
    # 3) Adam refinement on the true max‑convolution (soft‑max surrogate)
    # ------------------------------------------------------------------
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    adam_start = time.monotonic()
    # we continue until the overall deadline
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - adam_start) / max(1e-12, deadline - adam_start)   # ∈[0,1]

        # learning‑rate decays from 0.3 → 0.001
        lr = 0.30 * (1.0 - frac) + 0.001 * frac
        # temperature grows from 5 → 5000 (sharper approximation of max)
        alpha = 5.0 * (1.0 - frac) + 5000.0 * frac

        # --- compute self‑convolution ---------------------------------
        conv = self_convolution(a, fft_len_self, conv_len)

        # --- soft‑max surrogate --------------------------------------
        c_max = conv.max()
        shifted = conv - c_max                 # improve numerical stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of the surrogate via FFT -----------------------
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        start_idx = w_rev.size - 1                # = 2·n−2
        grad = 2.0 * full[start_idx:start_idx + n]

        # --- Adam update + projection onto the simplex ----------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the probability simplex (non‑negative, sum = 1)
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            a[:] = 1.0 / n
        else:
            a /= total

        # --- evaluate true objective ---------------------------------
        cur_score = compute_J(a, n, fft_len_self, conv_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[{now - start:6.2f}s] Adam refined J = {best_score:.6f}")

        # occasional jitter to escape tiny plateaus
        if rng.random() < 0.01:
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            a = a * jitter
            a = np.clip(a, 0.0, None)
            total = a.sum()
            a /= total if total != 0.0 else 1.0 / n

    # ------------------------------------------------------------------
    # 4) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
