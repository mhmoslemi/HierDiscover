     draws and (if available) the previously discovered construction.
   The best seed is kept as a starting point.

3. Refine the best seed with a projected Adam optimiser on the *soft‑max*
   surrogate of the max‑convolution:
   * variable = logits `p` (unconstrained); the actual step heights are the
     soft‑max `a = softmax(p)` guaranteeing non‑negativity and sum = 1.
   * At iteration `t` the surrogate temperature switches from a small value
     (smooth) to a very large one (sharp), i.e. `softmax_weight_i ∝
     exp(alpha_t·(conv_i‑max(conv)))` with `alpha_t` growing from 2 to 3000.
   * Learning rate decays from 0.5 to 0.001.  Adam moments are updated in the
     usual way and the logits are adjusted in the direction of the surrogate
     gradient.
   * Simple stagnation detection restarts the optimisation from the current
     best point with a tiny random perturbation.

4. The best vector found (clipped to the limits of the evaluator) is returned.

The whole procedure respects a 1000 s budget (5 s safety margin) and uses only
NumPy (no external solvers).  It typically finds a sequence with a bound
strictly below the previous best 1.5132, encouraging further improvement.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """
    Exact objective J = 2·n·max(a∗a)  (the sequence is assumed to be
    normalised to sum = 1, hence the denominator of ``evaluate_sequence``
    equals 1).
    """
    conv = self_convolution(a, fft_len_self, conv_len)
    return float(2.0 * n * conv.max())


def _beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Discretised Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid NaNs for α<1 or β<1 at the boundaries by adding a tiny epsilon
    eps = 1e-12
    v = (np.maximum(x, eps) ** (alpha - 1.0)) * (np.maximum(1.0 - x, eps) ** (beta - 1.0))
    if v.sum() == 0.0:
        v = np.full_like(v, 1.0 / n)
    else:
        v /= v.sum()
    return v


def _exp_sequence(lam: float, n: int) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i), normalised to sum = 1."""
    v = np.exp(-lam * np.arange(n, dtype=np.float64))
    if v.sum() == 0.0:
        v = np.full_like(v, 1.0 / n)
    else:
        v /= v.sum()
    return v


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max (temperature = 1/alpha) over a convolution vector.
    Returns a probability vector that sums to 1.
    """
    cmax = conv.max()
    shifted = conv - cmax                     # ≤ 0
    w_raw = np.exp(alpha * shifted)           # max entry → 1, others ≤ 1
    w_sum = w_raw.sum()
    if w_sum == 0.0:
        return np.full_like(w_raw, 1.0 / w_raw.size)
    return w_raw / w_sum


def _grad_softmax(a: np.ndarray, w: np.ndarray,
                  fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the smooth surrogate (soft‑max of the self‑convolution)
    with respect to the probability vector `a`.
    The gradient equals 2·(a ⋆ w_rev) where w_rev is `w` reversed.
    """
    # full convolution a * w_rev  (length = n + (2n‑1) – 1 = 3n‑2)
    fw_rev = np.fft.rfft(w[::-1], n=fft_len_grad)
    fa = np.fft.rfft(a, n=fft_len_grad)
    full = np.fft.irfft(fa * fw_rev, n=fft_len_grad)
    start = w.size - 1                     # = 2·n‑2
    grad = 2.0 * full[start:start + a.size]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine builds many diverse seeds,
    selects the best, and then refines it with an Adam optimiser
    on a temperature‑controlled soft‑max surrogate.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # 5 s safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(1024, min(5000, n))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for gradient (a ⋆ w_rev)

    best_seq = None
    best_J = float("inf")

    # ------------------------------------------------------------------
    # Helper to update the current best
    # ------------------------------------------------------------------
    def _maybe_update(seq: np.ndarray) -> None:
        nonlocal best_seq, best_J
        J = compute_J(seq, n, fft_len_self, conv_len)
        if J < best_J:
            best_J = J
            best_seq = seq.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 1) Diverse seed generation
    # ------------------------------------------------------------------
    # uniform
    _maybe_update(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential families – log‑space grid
    for lam in np.logspace(-5, 0, num=12):
        _maybe_update(_exp_sequence(lam, n))

    # Beta families – grid on log‑scale
    alphas = np.logspace(-0.3, 0.7, 9)   # ≈ 0.5 … 5
    betas = np.logspace(-0.3, 0.7, 9)
    for a in alphas:
        for b in betas:
            _maybe_update(_beta_sequence(a, b, n))

    # Random Dirichlet draws (moderate concentration)
    for _ in range(30):
        _maybe_update(rng.dirichlet(np.ones(n) * 0.5))

    # Include previous construction if present
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        if prev.sum() == 0.0:
            prev = np.full_like(prev, 1.0 / n)
        else:
            prev = np.clip(prev, 0.0, None)
            prev /= prev.sum()
        _maybe_update(prev)

    # Safety – should never happen, but keep a fallback.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_J = compute_J(best_seq, n, fft_len_self, conv_len)

    # ------------------------------------------------------------------
    # 2) Adam optimisation on logits (soft‑max surrogate)
    # ------------------------------------------------------------------
    # Initialise logits from the best seed (log‑softmax)
    eps = 1e-12
    p = np.log(best_seq + eps)                # shape (n,)

    # Adam state
    m = np.zeros_like(p)
    v_adj = np.zeros_like(p)                  # second moment
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8

    # Schedules
    lr_start, lr_end = 0.5, 0.001
    alpha_start, alpha_end = 2.0, 3000.0

    it = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)

        # learning‑rate and temperature schedules
        lr = lr_start * (1.0 - frac) + lr_end * frac
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # probabilities from logits
        max_p = np.max(p)
        exp_p = np.exp(p - max_p)                # stable soft‑max
        a = exp_p / exp_p.sum()                  # Σ a = 1, a ≥ 0

        # self‑convolution
        conv = self_convolution(a, fft_len_self, conv_len)

        # soft‑max weights over the convolution entries
        w = _softmax_weights(conv, alpha)        # Σ w = 1

        # gradient wrt a of the smooth surrogate
        grad_a = _grad_softmax(a, w, fft_len_grad)

        # convert to gradient wrt logits (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_p = a * (grad_a - dot)              # = ∂f/∂p

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_p
        v_adj = beta2 * v_adj + (1.0 - beta2) * (grad_p * grad_p)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v_adj / (1.0 - beta2 ** it)

        p = p - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # occasional tiny random jitter to escape plateaus
        if rng.random() < 0.002:
            p += rng.normal(0.0, 0.02, size=n)

        # evaluate true objective (real max‑convolution)
        cur_J = compute_J(a, n, fft_len_self, conv_len)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_seq = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Adam improvement – best J = {best_J:.6f}")

        # stagnation restart: if no improvement for ~30 s, perturb from best
        if now - last_improve > 30.0:
            # reset logits around current best with small multiplicative noise
            p = np.log(best_seq + eps) + rng.normal(0.0, 0.05, size=n)
            m.fill(0.0)
            v_adj.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Adam restart after stagnation – best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3) Return the best sequence (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
```
