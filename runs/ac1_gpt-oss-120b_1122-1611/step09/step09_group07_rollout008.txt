"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.  The optimizer has two stages.

Stage 1 –  *soft‑max gradient descent* on the simplex.
    •  The objective  J = 2·n·max(a∗a)  (the denominator of
       ``evaluate_sequence`` cancels after normalisation).
    •  A temperature‑controlled soft‑max approximates the max, giving a
       smooth gradient that we compute by a single FFT‐convolution.
    •  Adam with learning‑rate decay drives the iterates; the temperature
       (α) grows from 10 to 2·10⁴ so that the soft‑max becomes sharply
       focused on the true peak.
    •  Restarts are triggered after 15 s of stagnation – a fresh
       Dirichlet draw (α=0.5) injects diversity.

Stage 2 –  *stochastic mass‑transfer local search*.
    •  Randomly move a small amount of mass from one index to another.
    •  Keep the move if it lowers J, otherwise accept with a tiny
       Metropolis probability (temperature 1e‑5) to escape plateaus.
    •  Heavy block‑perturbations are performed every 5·10⁴ attempts.

The whole routine respects the 1000 s budget (5 s safety margin)
and returns the best vector found.  All numbers are clipped to the
limits required by ``evaluate_sequence`` before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.3))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return proj_simplex(out)


def compute_J(vec: np.ndarray) -> float:
    """
    Return the value of the objective J = 2·n·max(conv) for a vector
    that already satisfies sum(vec)=1.
    """
    n = vec.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())

    start_time = time.monotonic()

    # allocate about 70 % of the budget to the gradient phase
    gd_deadline = start_time + 0.70 * (deadline - start_time)

    last_improve = start_time
    it = 0
    while time.monotonic() < gd_deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (gd_deadline - start_time)   # ∈[0,1]

        # ----- schedules -------------------------------------------------
        lr = 0.5 * (1.0 - frac) + 0.001 * frac          # 0.5 → 0.001
        alpha = 10.0 * (1.0 - frac) + 20000.0 * frac    # 10 → 20000

        # ----- self‑convolution & soft‑max weights -----------------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

        c_max = conv.max()
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----------------------
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
        start_idx = w_rev.size - 1                # = 2n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- Adam update + projection ---------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # ----- keep best -------------------------------------------------
        cur_J = compute_J(a)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Adam new best J = {best_J:.6f}")

        # ----- occasional jitter ----------------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # ----- restart after stagnation ---------------------------------
        if now - last_improve > 15.0:
            # fresh random start (Dirichlet) to explore a new basin
            a = rng.gamma(0.5, 1.0, size=n)
            a = proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart (stagnation) – new random start")

    # --------------------------------------------------------------
    # 5) Phase 2 – Stochastic mass‑transfer local search
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    attempts = 0
    accepts = 0
    temp = 1e-5                       # tiny temperature for uphill moves

    while time.monotonic() < deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a random fraction of mass from i to j
        if cur_vec[i] <= 0.0:
            continue
        delta = cur_vec[i] * rng.uniform(0.0, 0.2)   # up to 20 % of source mass
        cand = cur_vec.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_J = compute_J(cand)

        if cand_J < cur_J - 1e-12:
            cur_vec = cand
            cur_J = cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] move new best J = {best_J:.6f}")
        else:
            # occasional uphill acceptance
            if rng.random() < np.exp(-(cand_J - cur_J) / temp):
                cur_vec = cand
                cur_J = cand_J
                accepts += 1

        # heavy perturbation every 50 k attempts
        if attempts % 50000 == 0:
            cur_vec = _random_block_perturb(cur_vec, rng)

    print(f"Finished – best J = {best_J:.6f} (attempts {attempts}, accepts {accepts})")
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
