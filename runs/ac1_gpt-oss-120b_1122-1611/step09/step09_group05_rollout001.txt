Hybrid sub‑gradient / simulated‑annealing optimiser.

* Work on the simplex (∑a = 1, a_i ≥ 0) – the objective
  J(a) = 2·n·max(conv(a,a)) is scale‑invariant.
* For a given vector a we compute the full auto‑convolution with a real FFT.
* The active set (indices where the convolution reaches its maximum) gives a
  sub‑gradient of max(conv).  The sub‑gradient is a sum of shifted copies of a.
* A multiplicative (exponentiated‑gradient) step moves mass away from the active
  positions; the step size decays like 1/√t.
* Acceptance follows a Metropolis rule with a linear cooling schedule (5 % of
  the current best objective down to 1e‑8).  This lets the search escape
  shallow plateaus.
* Whenever no improvement has been seen for 30 s a small random jitter is
  applied to the best solution to restart the walk.
* Several diverse initial vectors (uniform, geometric, linear ramps,
  Gaussian bells, Dirichlet draws and the previously found `height_sequence_1`
  if present) are evaluated, the best of them is taken as the start, and the
  optimiser runs until the supplied time budget expires.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # 1. Choose a problem size.
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # keep it reasonable for FFT speed
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ----------------------------------------------------------------------
    # 2. Build a diverse portfolio of initial probability vectors.
    # ----------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.95, 0.975, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps – decreasing / increasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells with a few widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 10.0, n / 8.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – moderately and strongly concentrated
    for alpha in [0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # ----------------------------------------------------------------------
    # 3. Pick the best of the initial candidates.
    # ----------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full(cand, fft_len, conv_len)
        max_conv = conv.max()
        J = float(2.0 * n * max_conv)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best  J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # 4. Main optimisation loop (sub‑gradient + annealing).
    # ----------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # temperature schedule (linear cooling)
    T_start = best_J * 0.05          # 5 % of the current best
    T_end = 1e-8

    # start from the best candidate
    cur_vec = best_vec.copy()
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)
    cur_J = float(2.0 * n * cur_conv.max())

    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # linear cooling of the Metropolis temperature
        T = T_start * (1.0 - elapsed / (deadline - start_time)) + T_end * (elapsed / (deadline - start_time))

        # ------------------------------------------------------------------
        # 4.1  Evaluate current state
        # ------------------------------------------------------------------
        cur_conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = cur_conv.max()
        cur_J = float(2.0 * n * max_conv)

        # update global best if needed
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.2  Stagnation handling: jitter after 30 s without improvement
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            # small multiplicative Gaussian perturbation of the best solution
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            # recompute its objective (used as the new current state)
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            cur_J = float(2.0 * n * cur_conv.max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.3  Compute a sub‑gradient of max(conv) at the active set.
        # ------------------------------------------------------------------
        # tolerance to collect all indices that are essentially maximal
        tol = max(1e-12, max_conv * 1e-12)
        active = np.where(cur_conv >= max_conv - tol)[0]

        grad = np.zeros_like(cur_vec)
        for p in active:
            # indices j for which a_{p‑j} is defined
            start = p - (n - 1)
            if start < 0:
                start = 0
            end = p if p < n else n - 1
            idx = np.arange(start, end + 1)
            grad[idx] += 2.0 * cur_vec[p - idx]   # sub‑gradient contribution

        # ------------------------------------------------------------------
        # 4.4  Propose a new point using a multiplicative (exponentiated) step.
        # ------------------------------------------------------------------
        # step size decays like 1/√t  (t = iteration+1)
        step = 0.5 / np.sqrt(iteration + 1.0)
        # multiplicative update (always stays non‑negative)
        factor = np.exp(-step * grad)
        proposal = cur_vec * factor
        # projection back onto the simplex (just renormalise, all entries ≥0)
        s = proposal.sum()
        if s == 0.0:
            proposal[:] = 1.0 / n
        else:
            proposal /= s

        # ------------------------------------------------------------------
        # 4.5  Evaluate the proposal.
        # ------------------------------------------------------------------
        prop_conv = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ------------------------------------------------------------------
        # 4.6  Metropolis acceptance.
        # ------------------------------------------------------------------
        accept = False
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = prop_J

        iteration += 1

    # ----------------------------------------------------------------------
    # 5. Return the best solution found (after safe clipping and re‑normalisation).
    # ----------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
