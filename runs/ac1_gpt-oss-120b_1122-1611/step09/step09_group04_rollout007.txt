    c_vec = np.concatenate([p, np.zeros(n-1)])

"""
Propose a step‑function sequence that minimizes ``evaluate_sequence``.
Algorithm (two‑phase):

Phase 1 – Generate a small pool of diverse seeds (uniform, edge blocks,
cosine‑shapes, Dirichlet draws and the previously discovered sequence).
All seeds are normalised to the probability simplex (sum = 1).

Phase 2 – For the best few seeds run an *LP‑mixing* refinement.
Given a current probability vector ``p`` we compute its self‑convolution
peak ``M = max(p * p)``.  We then solve the linear program

    maximise    ∑_i g_i
    subject to   (p * g)_k ≤ M   for all k = 0 … 2n‑2
                 g_i ≥ 0

which finds a distribution ``g`` that can be added without raising the
peak.  After normalising ``g`` to sum = 1 we move a little towards it

    p ← (1‑t)·p + t·g    (t is decreased when progress stalls)

and project back onto the simplex.  Repeating this step (while respecting
the time budget) steadily lowers the maximal self‑convolution.
The best probability vector found is finally rescaled to have
ℓ₁‑norm = √(2 n) as required by the evaluator.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """True objective J = 2·n·max(p * p) for a probability vector p."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L; gamma=0 → uniform block,
    gamma=1 → linearly taper to zero.
    """
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)

    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0

    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0

    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape: (1 + r·cosθ)^power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.

    The routine respects the given time budget (seconds) and returns a
    sequence scaled to satisfy the evaluator's L1‑norm requirement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep FFT cheap, cap to a few thousand entries
            n = min(max(n, 256), 4096)
        except Exception:
            n = 2048
    else:
        n = 2048

    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Global cheap exploration – build a small pool of seeds
    # ------------------------------------------------------------------
    seed_pool = []                     # list of (J, prob_vec)

    def _push(p_vec: np.ndarray):
        J = _objective(p_vec, fft_len_self, n)
        seed_pool.append((J, p_vec.copy()))

    # uniform baseline
    _push(np.full(n, 1.0 / n))

    # include previous construction if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0:
                prev = np.full_like(prev, 1.0 / prev.size)
            else:
                prev = proj_simplex(prev / prev.sum())
            _push(prev)
        except Exception:
            pass

    # a handful of edge‑shape candidates
    for frac in np.linspace(0.12, 0.40, 8):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            _push(_edge_shape(n, L, gamma))

    # cosine‑shape grid
    for r in np.linspace(0.0, 1.0, 7):
        for pw in np.linspace(0.3, 2.5, 7):
            _push(_cosine_shape(n, r, pw))

    # random Dirichlet draws (moderately concentrated)
    for _ in range(15):
        alpha = rng.uniform(0.05, 0.6)
        _push(rng.dirichlet(np.full(n, alpha)))

    # keep only the top‑3 seeds (lowest J)
    seed_pool.sort(key=lambda x: x[0])
    seed_pool = seed_pool[:3]

    # ------------------------------------------------------------------
    # 3) LP‑mixing local refinement
    # ------------------------------------------------------------------
    best_J, best_p = seed_pool[0]

    # mixing schedule – start fairly aggressive, shrink on stagnation
    t_init = 0.35
    t_min = 0.02

    # helper: solve the LP for a given p and bound M
    def _solve_lp(p_vec: np.ndarray, M: float):
        """Return a feasible g (normalised to sum = 1) or None."""
        # Build the full convolution matrix A where (A @ g) = p * g
        c_vec = np.concatenate([p_vec, np.zeros(n - 1)])
        r_vec = np.concatenate([p_vec[:1], np.zeros(n - 1)])
        A = toeplitz(c_vec, r_vec)                     # shape (2n‑1, n)

        # maximise sum(g)  → minimise -sum(g)
        res = linprog(-np.ones(n),
                      A_ub=A,
                      b_ub=np.full(2 * n - 1, M, dtype=np.float64),
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        # free memory early
        del A
        if not res.success:
            return None
        g = res.x
        s = g.sum()
        if s <= 0.0:
            return None
        return g / s

    # allocate roughly 80 % of the remaining time to refinement
    remaining = deadline - time.monotonic()
    refine_budget = remaining * 0.80
    per_seed = refine_budget / max(1, len(seed_pool))

    for seed_idx, (seed_J, seed_p) in enumerate(seed_pool):
        if time.monotonic() >= deadline:
            break
        p = seed_p.copy()
        local_best_J = seed_J
        t = t_init
        last_improve = time.monotonic()
        seed_deadline = time.monotonic() + per_seed

        while time.monotonic() < seed_deadline:
            # compute current peak
            conv_pp = _self_conv(p, fft_len_self)
            M = conv_pp.max()

            g = _solve_lp(p, M)
            if g is None:
                # LP failed – fall back to a tiny random jitter
                jitter = np.exp(rng.normal(0.0, 0.07, size=n))
                p = proj_simplex(p * jitter)
                continue

            # convex combination
            new_p = (1.0 - t) * p + t * g
            new_p = proj_simplex(new_p)

            new_J = _objective(new_p, fft_len_self, n)

            if new_J < local_best_J - 1e-12:
                local_best_J = new_J
                p = new_p
                last_improve = time.monotonic()
                # keep mixing aggressive after an improvement
                t = min(t_init, t * 1.1)
                print(f"[seed {seed_idx}] iter improvement: J={local_best_J:.6f}")
            else:
                # no progress – damp the step size
                t = max(t_min, t * 0.7)
                # occasional jitter to escape flat regions
                if rng.random() < 0.03:
                    blk = rng.integers(1, max(2, n // 30))
                    start_idx = rng.integers(0, n - blk + 1)
                    factor = np.exp(rng.normal(0.0, 0.09))
                    pj = p.copy()
                    pj[start_idx:start_idx + blk] *= factor
                    p = proj_simplex(pj)

                # restart if stuck for a while
                if time.monotonic() - last_improve > 25.0:
                    # reset to the best point seen for this seed
                    p = new_p.copy()
                    last_improve = time.monotonic()
                    t = t_init
                    print(f"[seed {seed_idx}] restart after stagnation – J={local_best_J:.6f}")

            # safety exit if mixing becomes too tiny
            if t <= t_min and time.monotonic() - last_improve > 40.0:
                break

        # keep global best
        if local_best_J < best_J:
            best_J, best_p = local_best_J, p.copy()
            print(f"*** new global best after seed {seed_idx}: J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 4) Return the best sequence (scaled to satisfy the evaluator)
    # ------------------------------------------------------------------
    scale = np.sqrt(2.0 * n)           # required ℓ₁‑norm
    best_seq = best_p * scale
    # Clip to the limits imposed inside ``evaluate_sequence``
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
