* Sample a wide variety of analytically‑parameterised step‑functions
  (uniform, exponential, linear, symmetric‑exponential, beta‑pdf,
  Gaussian, Dirichlet draws).
* Keep the best candidate seen (objective = 2·n·max(a∗a) since the

  convolution entry.  The gradient of the max‑convolution is obtained
  by a single FFT convolution with a one‑hot mask.
* Occasionally perform cheap pairwise mass‑transfer hill‑climbing
  moves from the current best.
* After the time budget expires a final polishing phase runs a few
  hundred sub‑gradient steps on the best vector.

The routine respects the 1000 s wall‑clock limit (5 s safety margin)
and always returns the best sequence found.

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (all arguments are explicit – no closures)
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """True objective J = 2·n·max(a∗a) for a vector already on the simplex."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def compute_conv_and_J(a: np.ndarray,
                       n: int,
                       fft_len_self: int,
                       conv_len: int):
    """Return the full self‑convolution (length 2·n‑1) and objective J."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    J = float(2.0 * n * conv.max())
    return conv, J


def gradient_at_max(a: np.ndarray,
                    max_idx: int,
                    n: int,
                    fft_len_grad: int,
                    conv_len: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    For a given index ``max_idx`` (where conv is maximal) the gradient is
    ``∇ conv[max_idx] = 2·a * w_rev`` where ``w`` is the one‑hot mask.
    """
    w = np.zeros(conv_len, dtype=np.float64)
    w[max_idx] = 1.0
    w_rev = w[::-1]

    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start = conv_len - 1               # = 2·n‑2
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine respects a 1000 s wall‑clock budget (5 s safety margin) and
    always returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = min(max(n, 256), 4096)          # keep FFTs cheap
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for a * w_rev

    # --------------------------------------------------------------
    # 2) Initialise best candidate (uniform, plus previous best)
    # --------------------------------------------------------------
    best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    best_score = compute_J(best_vec, n, fft_len_self, conv_len)

    if "height_sequence_1" in globals():
        try:
            cand = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
            cand = proj_simplex(cand)
            if cand.size == n:
                cand_score = compute_J(cand, n, fft_len_self, conv_len)
                if cand_score < best_score:
                    best_vec = cand.copy()
                    best_score = cand_score
        except Exception:
            pass

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] initial best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Main stochastic search
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ---- choose a generation method ---------------------------------
        method = rng.integers(0, 7)
        if method == 0:                          # uniform
            a = np.full(n, 1.0 / n, dtype=np.float64)

        elif method == 1:                        # exponential decay (start → end)
            r = rng.uniform(0.85, 0.9999)
            a = r ** np.arange(n, dtype=np.float64)

        elif method == 2:                        # symmetric exponential about centre
            r = rng.uniform(0.85, 0.9999)
            dist = np.abs(np.arange(n) - (n - 1) / 2)
            a = r ** dist

        elif method == 3:                        # linearly decreasing
            a = np.arange(n, 0, -1, dtype=np.float64)

        elif method == 4:                        # beta‑pdf shape
            alpha = rng.uniform(0.1, 5.0)
            beta = rng.uniform(0.1, 5.0)
            x = (np.arange(n) + 0.5) / n
            a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
            if a.sum() == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)

        elif method == 5:                        # centred Gaussian
            sigma = rng.uniform(0.05, 0.3)          # relative width
            x = np.arange(n) / (n - 1)
            a = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)

        else:                                    # Dirichlet draw
            concentration = rng.uniform(0.05, 5.0)
            a = rng.dirichlet(np.full(n, concentration))

        # ---- normalise and (occasionally) smooth -----------------------
        a = np.asarray(a, dtype=np.float64)
        if rng.random() < 0.2:                     # cheap moving‑average smoothing
            wlen = rng.integers(3, 7)
            ker = np.ones(wlen) / wlen
            a = np.convolve(a, ker, mode='same')
        a = proj_simplex(a)

        # ---- evaluate --------------------------------------------------
        cur_J = compute_J(a, n, fft_len_self, conv_len)

        if cur_J < best_score:
            best_score = cur_J
            best_vec = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best J = {best_score:.6f} (method {method})")

            # ----- short sub‑gradient refinement on the new record -------
            a_ref = a.copy()
            last_improve = time.monotonic()
            for t in range(1, 151):
                conv, _ = compute_conv_and_J(a_ref, n, fft_len_self, conv_len)
                max_idx = int(conv.argmax())
                grad = gradient_at_max(a_ref, max_idx, n, fft_len_grad, conv_len)
                eta = 0.4 / np.sqrt(t)                # diminishing step size
                a_ref = a_ref - eta * grad
                a_ref = proj_simplex(a_ref)

                new_J = compute_J(a_ref, n, fft_len_self, conv_len)
                if new_J < best_score - 1e-12:
                    best_score = new_J
                    best_vec = a_ref.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] refined best J = {best_score:.6f} (iter {t})")
                    last_improve = time.monotonic()
                # stop early if stuck
                if time.monotonic() - last_improve > 5.0:
                    break

        # ---- occasional hill‑climbing from the global best -------------
        if best_vec is not None and rng.random() < 0.05:
            a_hc = best_vec.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j:
                max_transfer = min(a_hc[i], 0.004)
                if max_transfer > 0.0:
                    delta = rng.uniform(0.0, max_transfer)
                    a_hc[i] -= delta
                    a_hc[j] += delta
                    a_hc = proj_simplex(a_hc)
                    new_J = compute_J(a_hc, n, fft_len_self, conv_len)
                    if new_J < best_score - 1e-12:
                        best_score = new_J
                        best_vec = a_hc.copy()
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] hill‑climb improved J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final polishing (a few hundred sub‑gradient steps)
    # --------------------------------------------------------------
    if best_vec is not None:
        a_fin = best_vec.copy()
        for t in range(1, 401):
            conv, _ = compute_conv_and_J(a_fin, n, fft_len_self, conv_len)
            max_idx = int(conv.argmax())
            grad = gradient_at_max(a_fin, max_idx, n, fft_len_grad, conv_len)
            eta = 0.2 / np.sqrt(t)
            a_fin = a_fin - eta * grad
            a_fin = proj_simplex(a_fin)

        final_J = compute_J(a_fin, n, fft_len_self, conv_len)
        if final_J < best_score:
            best_score = final_J
            best_vec = a_fin.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] final polish J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
