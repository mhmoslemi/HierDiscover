def propose_candidate(seed=..., budget_s=1000.0):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    # Initialize a
    if "height_sequence_1" in globals():
        raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # maybe pad/truncate to max_len=4096 or keep original length
        a = proj_simplex(np.clip(raw, 0, None))
    else:
        n = 4096
        a = proj_simplex(rng.dirichlet(np.full(n, 0.5)))
    n = a.size
    # precompute fft_len
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len - 1).bit_length())
    best_a = a.copy()
    # compute initial J
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
    best_J = 2.0 * n * conv.max()
    print(f"[init] J={best_J:.6f}")

    # Main loop: combined subgradient and random perturbations
    step = 0.1
    step_min = 1e-6
    step_max = 10.0
    stagnation_time = 30.0
    last_improve = time.monotonic()
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # compute conv and max idx
        fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        max_idx = np.argmax(conv_full)
        max_val = conv_full[max_idx]
        # compute gradient g
        g = np.zeros_like(a)
        start_i = max(0, max_idx - (n - 1))
        end_i = min(n - 1, max_idx)
        idx_range = np.arange(start_i, end_i + 1)
        g[idx_range] = 2.0 * a[max_idx - idx_range]  # works for diagonal case too
        # line search
        improved = False
        step_try = step
        while step_try >= step_min:
            a_candidate = proj_simplex(a - step_try * g)
            # compute J candidate
            fa_c = np.fft.rfft(a_candidate, n=fft_len)
            conv_c = np.fft.irfft(fa_c*fa_c, n=fft_len)[:conv_len]
            J_c = 2.0 * n * conv_c.max()
            if J_c < best_J - 1e-12:
                improved = True
                break
            step_try *= 0.5
        if improved:
            a = a_candidate
            best_a = a.copy()
            best_J = J_c
            last_improve = time.monotonic()
            step = min(step_try * 1.2, step_max)
            elapsed = time.monotonic() - (deadline - budget_s + 5.0)
            print(f"[{elapsed:6.2f}s] iter {iteration:5d} new best J={best_J:.6f}")
        else:
            # reduce step
            step = max(step * 0.5, step_min)

        # periodic random block perturb if stagnating
        if time.monotonic() - last_improve > stagnation_time:
            # jitter: pick random block and multiply
            width = rng.integers(1, max(2, n // 20))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.1))
            a_jit = a.copy()
            a_jit[start:start+width] *= factor
            a_jit = np.clip(a_jit, 0, None)
            a_jit = proj_simplex(a_jit)
            # compute J of jitter
            fa_j = np.fft.rfft(a_jit, n=fft_len)
            conv_j = np.fft.irfft(fa_j*fa_j, n=fft_len)[:conv_len]

            else:

            # reset step
            step = max(step, 0.01)

    # final return best sequence clipped
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]

"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence.  

The algorithm is a simple projected sub‑gradient method for the
L∞‑norm of the self‑convolution, combined with occasional random
“jitter” moves to escape plateaus.

* The vector is kept on the probability simplex (sum = 1).  Because
  evaluate_sequence is scale‑invariant this is sufficient – the final
  result is simply projected back onto the allowed range [0,1000].

* At each iteration we compute the full convolution a*a via FFT, find
  the index `k_max` where the convolution attains its maximum, and use
  the exact sub‑gradient of the max‑convolution (the vector whose i‑th
  entry is 2·a_{k_max‑i} whenever this index exists).  A back‑tracking
  line‑search is performed and the step is projected onto the simplex.

* If no improvement is observed for a while a small multiplicative
  log‑normal perturbation of a random contiguous block is applied; the
  perturbed vector is again projected onto the simplex.

* The best vector seen during the whole run is returned.

The routine respects the 1000 s time budget (keeping a 5 s safety
margin) and works for any length up to a few ten‑thousands of entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.10))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = proj_simplex(out)
    return out

# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.

    The routine keeps the vector on the simplex (sum = 1) and performs
    projected sub‑gradient descent on the L∞‑norm of the self‑convolution.
    Random block perturbations are injected when progress stalls.
    The best vector found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (keep a small safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise a feasible vector (non‑negative, sum = 1)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            raw = np.clip(raw, 0.0, None)
            a = proj_simplex(raw)
        except Exception:
            a = None
    else:
        a = None

    if a is None:
        # fall back to a Dirichlet draw (α≈1 gives a fairly uniform start)
        n = 4096
        a = proj_simplex(rng.dirichlet(np.full(n, 1.0)))
    else:
        n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for the convolution (full length = 2n‑1)
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 3) Helper: compute the objective J = 2·n·max(a*a)  (sum(a)=1)
    # ------------------------------------------------------------------
    def compute_J(vec: np.ndarray) -> float:
        """Return the value that evaluate_sequence would produce for a simplex vector."""
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())

    best_a = a.copy()
    best_J = compute_J(a)
    print(f"[init] n={n}  J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 4) Optimisation parameters
    # ------------------------------------------------------------------
    step = 0.2               # initial trial step size
    step_min = 1e-7
    step_max = 5.0
    stagnation_limit = 25.0  # seconds without improvement → jitter
    last_improve = time.monotonic()
    iteration = 0

    # ------------------------------------------------------------------
    # 5) Main optimisation loop (projected sub‑gradient descent)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ---- compute full convolution and locate its maximum ----
        fa = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        k_max = int(np.argmax(conv_full))
        max_val = float(conv_full[k_max])

        # ---- sub‑gradient of the max‑convolution (exact) ----
        # gradient_i = 2 * a_{k_max - i}  if the index exists, else 0
        g = np.zeros_like(a)
        i_start = max(0, k_max - (n - 1))
        i_end = min(n - 1, k_max)
        idx = np.arange(i_start, i_end + 1, dtype=int)
        g[idx] = 2.0 * a[k_max - idx]          # the factor 2 is exact for the sub‑gradient

        # ---- back‑tracking line‑search on the true objective ----
        improved = False
        step_try = step
        while step_try >= step_min:
            a_cand = proj_simplex(a - step_try * g)
            J_cand = compute_J(a_cand)

            # Accept only strict improvement (helps avoid wandering)
            if J_cand < best_J - 1e-12:
                improved = True
                break

            step_try *= 0.5

        if improved:
            a = a_cand
            best_a = a.copy()
            best_J = J_cand
            last_improve = time.monotonic()
            step = min(step_try * 1.3, step_max)

            elapsed = time.monotonic() - (deadline - budget_s + 5.0)
            print(f"[{elapsed:6.2f}s] iter {iteration:5d}  new best J = {best_J:.6f}")
        else:
            # no progress – shrink the step size
            step = max(step * 0.5, step_min)

        # ----------------------------------------------------------------
        # 6) Random block perturbation if we have been stuck for a while
        # ----------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > stagnation_limit:
            a = _random_block_perturb(best_a, rng)
            # give the jitter a chance to move the iterate
            step = max(step, 0.1)
            last_improve = now
            elapsed = now - (deadline - budget_s + 5.0)
            print(f"[{elapsed:6.2f}s] jitter (stagnation) – continue search")

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
