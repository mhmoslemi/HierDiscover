def propose_candidate(seed=12345, budget_s=1000., **kwargs):
    rng = np.random.default_rng(seed)
    now = time.monotonic()
    deadline = now + budget_s - 5.0

    # Determine n
    n = len(globals().get('height_sequence_1', []))
    if n == 0: n = 4096
    else: n = max(1024, min(5000, n))
    conv_len = 2*n-1
    fft_len = _next_pow2(conv_len)

    # compute J function
    def J_of(a):
        conv = self_convolution(a, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # initialization
    best = np.full(n, 1.0/n, dtype=np.float64)
    best_J = J_of(best)
    # maybe also include previous best if available
    if 'height_sequence_1' in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            x_old = np.linspace(0,1,prev.size)
            x_new = np.linspace(0,1,n)
            prev = np.interp(x_new, x_old, prev)
        prev = proj_simplex(prev)  # ensure sum=1
        J_prev = J_of(prev)
        if J_prev < best_J:
            best = prev.copy()
            best_J = J_prev

    # generate few random dirichlet seeds and keep best
    for alpha in [0.05, 0.2, 0.5, 1.0, 5.0]:
        a0 = rng.dirichlet(np.full(n, alpha))
        J0 = J_of(a0)
        if J0 < best_J:
            best = a0.copy()
            best_J = J0

    # Global search via random block perturb + gradient descent
    a = best.copy()
    last_improve = now
    # parameters
    temp0 = 1.0   # initial temperature for annealing
    temp_end = 1e-3
    max_iters = 1000000  # fallback
    iter_count = 0

    while time.monotonic() < deadline:
        # compute conv and max shift
        conv = self_convolution(a, fft_len, conv_len)
        k_max = int(conv.argmax())
        max_val = conv[k_max]

        # gradient for hard max: for each i where 0<=k_max-i<n
        grad = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # pick donor i (largest grad) and receiver j (smallest grad)
        # we use top/bottom quantiles to avoid ties
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))

        # choose step size
        progress = (time.monotonic() - now) / (deadline - now)
        eps = (0.2 * (1 - progress) + 0.001) * a[i]  # decreasing fraction
        # ensure not moving all mass
        eps = min(eps, a[i])

        # propose move
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # evaluate
        J_new = J_of(a_new)

        # temperature schedule for Metropolis acceptance (to escape)
        T = temp0 * (1 - progress) + temp_end * progress
        if J_new < best_J - 1e-12:
            best = a_new.copy()
            best_J = J_new
            a = a_new
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - now:6.2f}s] new best J={best_J:.6f}")
        else:
            # accept with probability
            if rng.random() < np.exp(-(J_new - best_J) / T):
                a = a_new

        # occasional random block perturb to add diversity
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # periodic restart if no improvement for 30 sec
        if time.monotonic() - last_improve > 30.0:
            # jitter around best
            a = best.copy()
            a = a * np.exp(rng.normal(0.0, 0.05, size=n))
            a = proj_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - now:6.2f}s] restart – best J = {best_J:.6f}")

        iter_count += 1

    # final return best, clipped to [0,1000]
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Hybrid hill‑climbing / simulated‑annealing optimizer for the
self‑convolution upper‑bound.

The algorithm works on a probability simplex (sum = 1, a_i ≥ 0).
It starts from a collection of diverse seeds (uniform, Dirichlet
draws and the previously‑known construction) and then performs
many cheap “mass‑transfer” steps:

* For the current maximal convolution index k* we compute the exact
  sub‑gradient (2·a_{k*–i}) and move a small amount of probability
  from the index with the largest gradient to the one with the
  smallest gradient.  This is equivalent to a projected sub‑gradient
  step on the hard max‑convolution.

* The step size is annealed (large at the start, gradually shrinking)
  and the move is always accepted if it improves the objective.
  Otherwise a Metropolis‑type acceptance with a cooling temperature
  allows occasional uphill moves to escape local plateaus.

* Every ~1 % of the iterations a random log‑normal block perturbation
  is applied to inject diversity; after 30 s without improvement the
  current iterate is re‑initialized by a small jitter around the best
  solution seen so far.

The routine respects the 1000 s budget (leaving a small safety margin)
and returns the best sequence found.  All helper functions are top‑level
and no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out


def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of the given length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective  J = 2·n·max(a∗a) for a simplex‑normalised vector."""
    conv = self_convolution(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The method combines:
      • a diverse pool of initial candidates,
      • a fast sub‑gradient hill‑climber on the hard max‑convolution,
      • occasional simulated‑annealing moves and random block jitter.
    It respects the time budget (5 s safety margin) and returns the best
    sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    now = time.monotonic()
    deadline = now + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size (use previous construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(1024, min(5000, n))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # length for a*a convolution

    # ------------------------------------------------------------------
    # 2) Helper to evaluate the true objective
    # ------------------------------------------------------------------
    def objective(v: np.ndarray) -> float:
        return _objective(v, n, fft_len, conv_len)

    # ------------------------------------------------------------------
    # 3) Build a diverse initial portfolio (all normalised to sum=1)
    # ------------------------------------------------------------------
    candidates: list[np.ndarray] = []

    # uniform start
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few exponential decays (r close to 1 are interesting)
    for r in [0.99, 0.995, 0.998, 0.999, 0.9995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # linear decay
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    candidates.append(v)

    # Dirichlet draws with various concentration parameters
    for conc in [0.02, 0.05, 0.1, 0.3, 0.7, 1.5]:
        for _ in range(2):
            v = rng.dirichlet(np.full(n, conc))
            candidates.append(v.astype(np.float64))

    # include the previously known construction if present
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        prev = proj_simplex(prev)
        candidates.append(prev)

    # ------------------------------------------------------------------
    # 4) Pick the best start among the portfolio
    # ------------------------------------------------------------------
    best_vec = None
    best_score = np.inf
    for cand in candidates:
        s = objective(cand)
        if s < best_score:
            best_score = s
            best_vec = cand.copy()
    print(f"[{0.0:6.2f}s] Initial best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Main optimisation loop – mass‑transfer hill‑climb + SA
    # ------------------------------------------------------------------
    a = best_vec.copy()          # current iterate
    last_improve = now
    iteration = 0

    # temperature schedule (Metropolis acceptance)
    T_start = 1.0
    T_end = 1e-4

    while time.monotonic() < deadline:
        iteration += 1

        # ---------- compute convolution and locate the worst shift ----------
        conv = self_convolution(a, fft_len, conv_len)
        k_max = int(conv.argmax())
        max_val = conv[k_max]

        # ---------- exact sub‑gradient of the hard max ----------
        grad = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad[mask] = 2.0 * a[idx[mask]]

        # ---------- choose donor (largest grad) and receiver (smallest) ----------
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        # Guard against degenerate case where gradient is flat
        if i == j:
            # pick random distinct pair
            i, j = rng.choice(n, size=2, replace=False)

        # ---------- step‑size (decreases with time) ----------
        elapsed = time.monotonic() - now
        frac = elapsed / (deadline - now)          # ∈[0,1]
        # Start with up to 20 % of the donor mass, shrink to 0.1 % as we near the deadline
        eps = a[i] * (0.20 * (1.0 - frac) + 0.001)
        eps = min(eps, a[i])                       # keep non‑negative

        # ---------- propose move ----------
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps   # sum stays exactly 1

        # ---------- evaluate ----------
        new_score = objective(a_new)

        # ---------- acceptance ----------
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_vec = a_new.copy()
            a = a_new
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - now:6.2f}s] New best J = {best_score:.6f}")
            continue

        # Metropolis acceptance with a cooling temperature
        T = T_start * (1.0 - frac) + T_end * frac
        if rng.random() < np.exp(-(new_score - best_score) / max(T, 1e-12)):
            a = a_new
        # else keep current a

        # ---------- occasional random block jitter ----------
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # ---------- stagnation restart ----------
        if time.monotonic() - last_improve > 30.0:
            # jitter around the best known vector
            a = best_vec.copy()
            a = a * np.exp(rng.normal(0.0, 0.06, size=n))
            a = proj_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - now:6.2f}s] Restart – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
