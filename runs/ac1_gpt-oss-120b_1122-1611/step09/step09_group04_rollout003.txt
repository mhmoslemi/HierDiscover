"""
Propose a step‑function sequence with a low value of
evaluate_sequence(sequence).

The algorithm consists of three stages:

1️⃣ **Global cheap sweep** – generate a diverse pool of candidates:
   * uniform, edge‑heavy, cosine‑shaped families (as in the reference)
   * a few symmetric Dirichlet draws (shape is forced to be mirror‑symmetric)
   * the previous best sequence (if available).

2️⃣ **Projected Adam on a smooth‑max surrogate** – each seed is refined
   with Adam on the soft‑max approximation of the max‑convolution.
   After every Adam update the vector is symmetrised
   (p ← (p + p[::-1])/2) and projected back onto the probability simplex.
   Learning‑rate and temperature follow exponential schedules.
   Jitter, restarts and a small “stagnation” logic are kept from the reference.

3️⃣ **Deterministic sub‑gradient phase** – starting from the best Adam result
   we repeatedly minimise the true max‑convolution:
   * locate the shift k where the auto‑correlation is maximal,
   * compute the sub‑gradient of this entry (2·p shifted),
   * take a projected gradient step, re‑symmetrise and re‑project.
   The step‑size is adapted and a simple restart on stagnation is performed.

The final vector is scaled to satisfy the normalisation used in the
evaluation function (∑a = √(2 n)).  All values are clipped to the limits
enforced inside ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape, normalised to sum = 1."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """Two edge blocks of length L with linear taper controlled by gamma."""
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[0] = 1.0
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max weights (sum = 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate w.r.t. a.
    Computes 2·(a * w_rev) where * is linear convolution.
    """
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1          # aligns convolution result
    return 2.0 * full[start_idx:start_idx + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """Exact objective  J = 2·n·max(p∗p) for a probability vector p."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force exact mirror symmetry: v ← (v+v[::-1])/2."""
    return (v + v[::-1]) * 0.5


def _subgradient_step(p: np.ndarray, step: float, fft_len_self: int, n: int):
    """
    One sub‑gradient iteration on the true max‑convolution.
    Returns the updated probability vector.
    """
    conv = _self_conv(p, fft_len_self)
    k = int(np.argmax(conv))                     # shift where max occurs
    # gradient of conv_k w.r.t. p is 2 * shifted version of p
    grad = np.zeros_like(p)
    i_min = max(0, k - (n - 1))
    i_max = min(k, n - 1)
    idx_i = np.arange(i_min, i_max + 1)
    idx_j = k - idx_i
    grad[idx_i] = 2.0 * p[idx_j]
    # projected gradient step + symmetrisation
    new_p = proj_simplex(p - step * grad)
    new_p = _symmetrize(new_p)
    new_p = proj_simplex(new_p)
    return new_p

                best_J_loc = cur_J
                last_improve = now
                print(f"[{now - start:.1f}s] Adam improvement: J = {cur_J:.6f}")

            # jitter to escape plateaus
            if rng.random() < 0.015:
                block_width = rng.integers(1, max(2, n // 30))
                start_idx = rng.integers(0, n - block_width + 1)
                factor = np.exp(rng.normal(0.0, 0.07))
                a_pert = a.copy()
                a_pert[start_idx:start_idx + block_width] *= factor
                a_pert = proj_simplex(a_pert)
                a_pert = _symmetrize(a_pert)
                a = proj_simplex(a_pert)

            # restart on stagnation
            if now - last_improve > 18.0:
                a = proj_simplex(a)                 # ensure feasibility
                a = _symmetrize(a)
                a = proj_simplex(a)
                # small unbiased noise
                noise = np.exp(rng.normal(0.0, 0.045, size=n))
                a = proj_simplex(a * noise)
                a = _symmetrize(a)
                a = proj_simplex(a)
                last_improve = now
                print(f"[{now - start:.1f}s] Adam restart – best J = {best_J_loc:.6f}")

        return a, best_J_loc

    # --------------------------------------------------------------
    # 3) Run Adam on the top seeds respecting the remaining budget
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    refine_budget = remaining * 0.78                     # leave a little for sub‑grad
    per_seed = refine_budget / max(1, len(best_candidates))

    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break
        tlim = min(per_seed, deadline - time.monotonic())
        refined_p, refined_J = _adam_refine(p0, tlim)
        # keep the best result from this seed
        best_candidates.append((refined_J, refined_p.copy()))
        best_candidates.sort(key=lambda x: x[0])
        best_candidates = best_candidates[:7]   # prune again

    # --------------------------------------------------------------
    # 4) Deterministic sub‑gradient phase on the overall best seed
    # --------------------------------------------------------------
    best_J, best_p = best_candidates[0]
    subgrad_time = max(0.0, deadline - time.monotonic() - 2.0)   # 2 s guard
    if subgrad_time > 5.0:                                      # need at least a few secs
        step = 0.12
        last_improve = time.monotonic()
        start_sg = time.monotonic()
        while time.monotonic() < deadline - 2.0:
            # one sub‑gradient iteration
            new_p = _subgradient_step(best_p, step, fft_len_self, n)
            cur_J = _objective(new_p, fft_len_self, n)
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_p = new_p.copy()
                last_improve = time.monotonic()
                step = step * 1.07
                print(f"[{time.monotonic() - start_sg:.1f}s] Sub‑grad better: J = {best_J:.6f}")
            else:
                step = step * 0.73

            # stagnation restart
            if time.monotonic() - last_improve > 12.0:
                best_p = proj_simplex(best_p)            # ensure feasibility
                best_p = _symmetrize(best_p)
                best_p = proj_simplex(best_p)
                step = 0.12
                last_improve = time.monotonic()
                print(f"[{time.monotonic() - start_sg:.1f}s] Sub‑grad restart – J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5) Return the scaled sequence
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)                # required L1‑norm for the original evaluator
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"[{time.monotonic() - start_time:.1f}s] Finished – final J ≈ {best_J:.6f}")
    return [float(x) for x in best_seq]
