    # Softmax weights
    max_conv = conv.max()
    # for stability, shift by max
    exp_vals = np.exp(alpha * (conv - max_conv))
    w = exp_vals / exp_vals.sum()
    a_rev = a[::-1]
    # conv w * a_rev
    W_fft = np.fft.rfft(w, n=fft_len)
    A_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    # Since w length = conv_len, a_rev length = n, conv length = conv_len + n -1 = 3n-2
    # We'll compute full convolution and slice
    tmp1 = np.fft.irfft(W_fft * A_rev_fft, n=fft_len)

    start = len(a) - 1

    # reverse w
    w_rev = w[::-1]

    tmp2 = np.fft.irfft(W_rev_fft * A_rev_fft, n=fft_len)
    grad2 = tmp2[start:start+len(a)]

    grad = 2.0 * len(a) * (grad1 + grad2)
    return grad

def _optimize_for_n(n, rng, global_deadline, start_time, target_J=1.5025):
    fft_len = _next_pow2(3 * n)  # for both conv and gradient
    conv_len = 2 * n - 1

    # Initial candidates
    candidates = []
    candidates.append(np.full(n, 1.0/n, dtype=np.float64))
    candidates.append(rng.dirichlet(np.ones(n) * 0.5))
    candidates.append(rng.dirichlet(np.ones(n) * 0.1))
    # include previous solution if available
    if "height_sequence_1" in globals():
        src = np.asarray(height_sequence_1, dtype=np.float64)
        src = np.clip(src, 0.0, None)
        # resample or truncate/pad to length n
        if src.shape[0] != n:
            # linear interpolation to new size
            x_old = np.linspace(0, 1, src.shape[0])
            x_new = np.linspace(0, 1, n)
            src = np.interp(x_new, x_old, src)
        src_sum = src.sum()
        if src_sum == 0.0:
            src = np.full(n, 1.0/n)
        else:
            src /= src_sum
        candidates.append(src)

    best_vec = None
    best_J = np.inf

    for cand in candidates:
        conv = compute_self_conv(cand, fft_len)
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()

    # Main iterative refinement
    a = best_vec.copy()
    cur_J = best_J
    last_improve = time.time()
    iteration = 0
    max_iters = 200000  # large enough
    # hyperparameters
    eta0 = 0.3
    alpha_start = 20.0
    alpha_end = 5000.0
    # schedule alpha increase linearly with iteration
    while time.time() < global_deadline and iteration < max_iters:
        # compute convolution and J
        conv = compute_self_conv(a, fft_len)
        max_conv = conv.max()
        J = 2.0 * n * max_conv

        if J < best_J - 1e-12:
            best_J = J
            best_vec = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] n={n} new best J={best_J:.6f}")

            if best_J <= target_J:
                # reached target, exit early
                break

        # schedule alpha
        frac = min(1.0, iteration / (max_iters - 1))
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # compute gradient
        grad, w = compute_gradient(a, conv, alpha, fft_len)

        # step size decreasing
        eta = eta0 / np.sqrt(iteration + 1)

        # gradient descent step
        a_new = a - eta * grad
        a_new = proj_simplex(a_new)

        # occasional stochastic perturbation to escape flat regions
        if rng.random() < 0.005:
            a_new = a_new * np.exp(rng.normal(0.0, 0.02, size=n))
            a_new = proj_simplex(a_new)

        a = a_new
        iteration += 1

        # stagnation handling
        now = time.time()
        if now - last_improve > 30.0:
            # re‑initialize around best solution
            a = best_vec * np.exp(rng.normal(0.0, 0.04, size=n))

            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] n={n} restart after stagnation, best J={best_J:.6f}")

    return best_vec, best_J

"""
Simple‐gradient optimiser for the step‑function problem.

The objective (after fixing ∑a = 1) is
        J(a) = 2·n·max(conv(a,a)) .
It is non‑smooth because of the max, so we minimise a smooth surrogate

        Lα(a) = 2·n·log_sum_exp(α·conv) / α ,

whose gradient is easy to obtain via FFTs.  The algorithm

* chooses a problem size n (by default the length of a previously found
  construction, bounded to [256,8192]),
* initialises a few probability vectors (uniform, Dirichlet draws,
  optionally the known solution interpolated to length n),
* runs a projected gradient descent on Lα, linearly increasing α from a
  modest value (≈20) up to a large one (≈5 000) so that the surrogate
  gradually approaches the true max‑objective,
* uses a diminishing step size ηk≈η0/√k, projects each iterate onto the
  simplex, and occasionally adds a tiny multiplicative jitter to escape
  plateaus,
* restarts from the current best after 30 s without improvement,
* returns the best vector found (already normalised, clipped to the limits
  required by ``evaluate_sequence``).

All heavy work (self‑convolution and the gradient) is performed with
FFT‑based convolution, so the optimiser can handle n up to 8192 comfortably
within the 1000 s budget.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, Σ w = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Return the (full) auto‑convolution a*a of length 2·n‑1."""
    A = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(A * A, n=fft_len)
    n = a.shape[0]
    return conv_full[: 2 * n - 1]


def _gradient(a: np.ndarray, conv: np.ndarray, alpha: float, fft_len: int):
    """
    Gradient of the smooth surrogate Lα(a) = 2·n·log_sum_exp(α·conv)/α.
    Returns the gradient vector (shape n) and the soft‑max weights w.
    """
    n = a.shape[0]
    # soft‑max over convolution values (shifted for numerical stability)
    max_c = conv.max()
    exp_vals = np.exp(alpha * (conv - max_c))
    w = exp_vals / exp_vals.sum()  # shape (2·n‑1,)

    # convolution w * a_rev  (full length = 3·n‑2)
    a_rev = a[::-1]
    W_fft = np.fft.rfft(w, n=fft_len)
    A_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    conv_wa = np.fft.irfft(W_fft * A_rev_fft, n=fft_len)

    # central slice gives Σ_i w_i a_{i‑j}
    start = n - 1
    grad_center = conv_wa[start : start + n]

    # dL/da = 2·n·Σ_i w_i·∂conv_i/∂a = 2·n·Σ_i w_i·2·a_{i‑j}
    grad = 4.0 * n * grad_center
    return grad, w


def _optimize_for_n(n: int, rng: np.random.Generator,
                    global_deadline: float, start_time: float,
                    target_J: float = 1.5025):
    """
    Gradient‑descent optimisation for a fixed length n.
    Returns the best vector found and its J‑value.
    """
    fft_len = _next_pow2(3 * n)          # enough for all convolutions
    conv_len = 2 * n - 1

    # ---- initial pool -------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws with different concentrations
    for conc in (0.3, 0.7, 1.0):
        init_candidates.append(rng.dirichlet(np.ones(n) * conc))

    # if a previous construction exists, interpolate it to length n
    if "height_sequence_1" in globals():
        src = np.asarray(height_sequence_1, dtype=np.float64)
        src = np.clip(src, 0.0, None)
        if src.shape[0] != n:
            x_old = np.linspace(0.0, 1.0, src.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            src = np.interp(x_new, x_old, src)
        s = src.sum()
        if s == 0.0:
            src = np.full(n, 1.0 / n)
        else:
            src /= s
        init_candidates.append(src)

    # pick the best of the pool as the starting point
    best_vec = None
    best_J = np.inf
    for cand in init_candidates:
        conv = _self_convolution(cand, fft_len)
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()

    a = best_vec.copy()
    last_improve = time.time()
    iteration = 0
    max_iters = 200_000                     # generous upper bound

    # hyper‑parameters
    eta0 = 0.30                              # initial step‑size factor
    alpha0 = 20.0
    alpha_target = 5_000.0

    while time.time() < global_deadline and iteration < max_iters:
        conv = _self_convolution(a, fft_len)
        max_conv = conv.max()
        J = 2.0 * n * max_conv

        if J < best_J - 1e-12:
            best_J = J
            best_vec = a.copy()
            last_improve = time.time()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] n={n} new best J = {best_J:.6f}")

            if best_J <= target_J:
                # good enough – stop early
                break

        # continuation on α (linear schedule)
        frac = iteration / max_iters
        alpha = alpha0 + (alpha_target - alpha0) * frac

        # gradient of the smooth surrogate
        grad, _ = _gradient(a, conv, alpha, fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(iteration + 1.0)

        # projected gradient step
        a_new = a - eta * grad
        a_new = proj_simplex(a_new)

        # occasional tiny stochastic bump to escape tiny plateaus
        if rng.random() < 0.005:
            a_new = a_new * np.exp(rng.normal(0.0, 0.02, size=n))
            a_new = proj_simplex(a_new)

        a = a_new
        iteration += 1

        # restart if we have been stagnant for a while
        if time.time() - last_improve > 30.0:
            a = best_vec * np.exp(rng.normal(0.0, 0.04, size=n))
            a = proj_simplex(a)
            last_improve = time.time()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] n={n} restart after stagnation – best J = {best_J:.6f}")

    return best_vec, best_J


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs a gradient‑descent method on a
    smooth approximation of the max‑convolution objective, uses a simple
    continuation on the smoothing parameter and returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    # leave a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ---- decide the problem size ---------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    # run the optimiser for the chosen length
    best_vec, best_J = _optimize_for_n(n, rng, deadline, start_time)

    # final sanitisation (evaluate_sequence will also clip, but we do it here)
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s

    print(f"Final best J = {best_J:.6f} with length = {len(best_vec)}")
    return [float(v) for v in best_vec]
