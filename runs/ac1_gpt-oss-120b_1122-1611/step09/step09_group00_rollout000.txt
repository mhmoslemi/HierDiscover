def propose_candidate(seed=..., budget_s=1000.0):
    rng = np.random.default_rng(seed)

    # Get existing sequence height_sequence_1 if defined, else uniform length 4096

        a = np.clip(a, 0.0, None)
    else:

        return 2.0 * n * conv.max()

    # baseline

    deadline = start + (budget_s - 5.0) # safety

The algorithm searches for a probability vector `a` (non‑negative, sum = 1) that minimizes  

\[
\text{score}(a)=\frac{2\,n\;\max(a*a)}{(\sum a)^2}=2\,n\;\max(a*a)
\]

where `*` denotes the discrete convolution.  
The search consists of three phases:

1. **Smoothed exponentiated‑gradient** – a smooth (log‑sum‑exp) surrogate of the
   max‑convolution is minimised.  The gradient of the surrogate can be obtained
   by a single FFT‑based convolution of the current distribution with a weight
   vector derived from the soft‑max of the convolution values.

2. **Targeted pairwise mass transfers** – the index that contributes most to the
   current maximal convolution is identified, a donor (`src`) with the largest
   gradient and a receiver (`tgt`) with the smallest gradient are chosen, and a
   small amount of probability mass is moved from `src` to `tgt`.  Moves are
   accepted if they do not increase the exact score; improvements update the
   global best.

3. **Simulated annealing** – occasional random moves are accepted according to a
   slowly decreasing temperature, allowing the search to escape shallow local
   minima.

All phases respect the simplex constraint by normalising after every update.
The implementation uses only NumPy and the provided `evaluate_sequence`
function, runs entirely in‑memory and respects the 1000 s time budget.

"""Propose a step‑function sequence with a low max‑convolution score.

The search works in three stages (all stay on the probability simplex):
1. Smoothed exponentiated‑gradient descent using a log‑sum‑exp surrogate of
   max(conv).  The gradient is obtained by one FFT‑based convolution of the
   current distribution with a soft‑max weight vector.
2. Targeted pairwise mass transfers that explicitly reduce the maximal
   convolution entry.
3. Simulated‑annealing‑style random moves to escape local minima.

The routine stops a few seconds before the given budget, prints progress

def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to *x*."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (FFT, length 2*len(a)-1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_weighted(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Convolution of a (length n) with a weight vector w (length 2n‑1).
    Returns the full linear convolution (length n+len(w)-1 = 3n‑2).
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    full = np.fft.irfft(a_fft * w_fft, n=fft_len)[: a.size + w.size - 1]
    return full


def _gradient_for_index(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of the convolution entry (a*a)[idx].
    For each i the component is a[idx‑i] (zero when out of range).
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(n, idx + 1)          # exclusive
    i = np.arange(i_start, i_end, dtype=int)
    grad[i] = a[idx - i]
    return grad


def _simplex_normalise(v: np.ndarray) -> np.ndarray:
    """Renormalise a non‑negative vector to unit sum (protect against round‑off)."""
    s = v.sum()
    if s <= 0.0:
        # avoid division by zero – fall back to uniform distribution
        return np.full_like(v, 1.0 / v.size)
    return v / s

        # exact evaluation
        conv_cand = _conv_self(cand, fft_len)
        score_cand = _score_from_conv(conv_cand)

        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (pair) = {best_score:.6f} (it={it2})")
            if best_score <= 1.5025:
                print("[early stop] target reached in pairwise phase")
                break

        # accept non‑worsening moves to keep the walk downhill
        if score_cand <= _score_from_conv(conv):
            a = cand

        it2 += 1

        # occasional random jitter to avoid stagnation
        if it2 % 2500 == 0:
            jitter = rng.normal(scale=0.002, size=n)
            a = _simplex_normalise(np.clip(a + jitter, 0.0, None))

        if it2 % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] pairwise iter={it2}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 3 – Simulated‑annealing refinement (remaining time)
    # --------------------------------------------------------------
    a = best_a.copy()
    it3 = 0
    # temperature schedule: start modest, decay to near zero
    temp_start = 0.02 * best_score
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start
        fraction = elapsed / (deadline - start)
        temperature = temp_start * (1.0 - fraction)

        # current convolution and its maximiser
        conv = _conv_self(a, fft_len)
        max_idx = int(np.argmax(conv))

        # gradient for the maximal entry (used to bias source selection)
        grad_max = _gradient_for_index(a, max_idx)

        # source: probability ∝ (grad_max)² (stronger bias than linear)
        prob_src = grad_max ** 2
        prob_sum = prob_src.sum()
        if prob_sum > 0.0:
            prob_src /= prob_sum
            src = int(rng.choice(n, p=prob_src))
        else:
            src = rng.integers(0, n)

        # target: uniform random (different from src)
        tgt = rng.integers(0, n)
        if tgt == src:
            tgt = (tgt + 1) % n

        # delta = small fraction of source mass
        delta = a[src] * rng.uniform(0.001, 0.025)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _simplex_normalise(cand)

        # evaluate candidate
        conv_cand = _conv_self(cand, fft_len)
        score_cand = _score_from_conv(conv_cand)

        # metropolis acceptance
        curr_score = _score_from_conv(conv)
        if score_cand < best_score - 1e-12:
            best_score = score_cand
            best_a = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (SA)   = {best_score:.6f} (it={it3})")
            if best_score <= 1.5025:
                print("[early stop] target reached in SA phase")
                break

        if score_cand < curr_score or rng.random() < math.exp((curr_score - score_cand) / max(1e-12, temperature)):
            a = cand

        it3 += 1
        if it3 % 5000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] SA iter={it3}, best={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_time = time.monotonic() - start
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
