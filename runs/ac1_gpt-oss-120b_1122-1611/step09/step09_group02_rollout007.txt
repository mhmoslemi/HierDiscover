def conv_matrix(f):
    n = len(f)
    row = []
    col = []
    data = []
    for k in range(2*n-1):
        i_min = max(0, k-(n-1))

            row.append(k)
            col.append(j)
            data.append(f[i])
    return csr_matrix((data, (row, col)), shape=(2*n-1, n))

def proj_simplex_sum(v, S):
    # Project onto {x>=0, sum x = S}
    if S == 0:
        return np.zeros_like(v)
    # Use algorithm slight adaptation:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, S/len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Light‑weight LP / sub‑gradient optimiser for the step‑function problem.

The algorithm works in three stages:

1.  Build a small portfolio of diverse initial vectors for each chosen
    length `n`.  All vectors are projected onto the simplex and scaled to
    the canonical sum `S = sqrt(2·n)`.  This makes the true objective
    `J = 2·n·max(a∗a)/S²` equal to the maximal self‑convolution value
    `max(a∗a)`, which is cheaper to evaluate.

2.  **LP refinement** – For a fixed `a` we solve the linear programme

        maximise   1ᵀ g
        subject to  a ∗ g ≤ M          (M = max(a∗a))
                    g ≥ 0

   The convolution `a∗g` is linear in `g`, therefore the constraints are
   represented by a sparse Toeplitz matrix built from `a`.  The LP solution
   `g` is scaled to the target sum `S` and mixed with the current vector
   `a ← (1‑t)·a + t·g`.  With a modest step size `t` the maximal convolution
   value never increases; we repeat this step until the time budget expires
   (or no further improvement is seen).

3.  **Projected sub‑gradient descent** – After the LP phase we perform a
    few iterations of hard‑max sub‑gradient descent on `max(a∗a)` with a
    simplex projection.  This removes any remaining plateaus.

All helpers are top‑level (no closures).  The routine respects the supplied
time budget (`budget_s`) and returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# 1️⃣  Utility functions (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex (∑=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def proj_simplex_sum(v: np.ndarray, S: float) -> np.ndarray:
    """Project onto the simplex with prescribed sum `S`."""
    if S <= 0.0:
        return np.zeros_like(v)
    w = proj_simplex(v)
    return w * S


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A small, diverse portfolio of vectors (unnormalised, sum = 1)."""
    cand: list[np.ndarray] = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # 3) linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # 4) two spikes on the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # 5) half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # 6) a handful of Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    return cand


def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using a power‑of‑two FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def conv_matrix_from_f(f: np.ndarray) -> csr_matrix:
    """
    Build the (2·n‑1)×n matrix `A` such that (A @ g) = f ∗ g.
    `f` and `g` are 0‑based vectors of length `n`.
    """
    n = f.size
    # row indices = i + j  (i from f, j from g)
    i = np.arange(n, dtype=np.int32)
    j = np.arange(n, dtype=np.int32)

    row = np.repeat(i, n) + np.tile(j, n)          # shape (n*n,)
    col = np.tile(j, n)                           # shape (n*n,)
    data = np.repeat(f, n)                        # shape (n*n,)

    # CSR matrix; rows run from 0 to 2·n‑2 inclusive
    A = csr_matrix((data, (row, col)), shape=(2 * n - 1, n))
    return A


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise 1ᵀ g   subject to   f ∗ g ≤ M , g ≥ 0 .
    Returns the optimal `g` (may be all zeros if the LP fails).
    """
    n = f.size
    A = conv_matrix_from_f(f)                         # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog minimises; we maximise sum(g) ⇔ minimise –sum(g)
    c = -np.ones(n, dtype=np.float64)

    # bounds: g_j ≥ 0
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fall back to a zero vector (feasible)
        return np.zeros_like(f)
    return res.x


# ----------------------------------------------------------------------
# 2️⃣  Refinement phases
# ----------------------------------------------------------------------


def lp_refine(initial_a: np.ndarray,
              S_target: float,
              fft_len: int,
              conv_len: int,
              time_deadline: float,
              max_iters: int = 30) -> tuple[np.ndarray, float]:
    """
    Repeated LP‐based improvement of `initial_a`.

    Returns the best vector found and its maximal self‑convolution value.
    """
    a = proj_simplex_sum(initial_a, S_target)
    conv = self_convolution(a, fft_len, conv_len)
    best_M = conv.max()
    best_a = a.copy()

    # mixing factor (start relatively small to guarantee monotonicity)
    t = 0.2
    last_improve = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() >= time_deadline:
            break

        # 1) current bound
        M = best_M

        # 2) solve LP to maximise sum(g) under the current bound
        g = solve_lp_max_sum(a, M)

        # 3) scale `g` to the target sum S_target
        sum_g = g.sum()
        if sum_g == 0.0:
            # nothing useful was found – abort the LP phase
            break
        g = (g / sum_g) * S_target

        # 4) convex combination
        a_new = (1.0 - t) * a + t * g
        # (sum is preserved automatically)

        # 5) evaluate the new maximal self‑convolution
        conv_new = self_convolution(a_new, fft_len, conv_len)
        M_new = conv_new.max()

        # 6) accept only if we made progress
        if M_new < best_M - 1e-12:
            best_M = M_new
            best_a = a_new.copy()
            a = a_new
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - (time_deadline - (time_deadline - time.monotonic())):6.2f}s] "
                  f"LP iter {it:02d} – new best max = {best_M:.6f}")
            # slowly decrease the step size to fine‑tune later on
            t = max(0.01, t * 0.9)
        else:
            # no improvement – shrink step size and try again
            t = max(0.01, t * 0.7)

        # restart if we have been stagnant for >15 s
        if time.monotonic() - last_improve > 15.0:
            a = best_a.copy()
            t = 0.2
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - (time_deadline - (time_deadline - time.monotonic())):6.2f}s] "
                  f"LP stagnation – restart (it={it})")

    return best_a, best_M


def subgrad_refine(initial_a: np.ndarray,
                   S_target: float,
                   fft_len: int,
                   conv_len: int,
                   time_deadline: float,
                   max_iters: int = 200) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on `max(a∗a)` (hard‑max).
    """
    a = proj_simplex_sum(initial_a, S_target)
    conv = self_convolution(a, fft_len, conv_len)
    best_M = conv.max()
    best_a = a.copy()

    start = time.monotonic()
    last_improve = start

    for it in range(max_iters):
        if time.monotonic() >= time_deadline:
            break

        conv = self_convolution(a, fft_len, conv_len)
        k_max = int(conv.argmax())

        # gradient of the tight convolution entry
        idx = k_max - np.arange(a.size)
        mask = (idx >= 0) & (idx < a.size)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # simple diminishing step size schedule
        elapsed = time.monotonic() - start
        total = time_deadline - start
        eta = 0.8 * (1.0 - elapsed / total) + 1e-4

        a = a - eta * grad
        a = proj_simplex_sum(a, S_target)

        M = self_convolution(a, fft_len, conv_len).max()
        if M < best_M - 1e-12:
            best_M = M
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:6.2f}s] Sub‑grad it {it:03d} – new best max = {best_M:.6f}")

        if time.monotonic() - last_improve > 12.0:
            # re‑initialise from the current best to escape shallow plateaus
            a = best_a.copy()
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:6.2f}s] Sub‑grad stagnation – restart (it={it})")

    return best_a, best_M

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine tries several problem sizes, runs a fast LP‑based refinement
    followed by a short sub‑gradient descent, and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # Reserve a small safety margin
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # 0️⃣ Choose a set of candidate lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096]

    # If a previous construction is available, also explore a neighbourhood
    if "height_sequence_1" in globals():
        try:
            prev_len = len(globals()["height_sequence_1"])
            for delta in (-200, -100, -50, 0, 50, 100, 200):
                n = max(128, min(prev_len + delta, 5000))
                lengths.append(n)
        except Exception:
            pass

    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # Global best bookkeeping
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_sequence = None

    # ------------------------------------------------------------------
    # Loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        now = time.monotonic()
        if now >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        S_target = np.sqrt(2.0 * n)               # canonical sum
        conv_len = 2 * n - 1
        fft_len = _next_pow2(conv_len)

        # ---- 1️⃣  Generate a few diverse starters (sum = 1, later scaled) ----
        starters = _initial_candidates(n, rng)
        # add a couple of extra Dirichlet draws for extra diversity
        for _ in range(3):
            starters.append(rng.dirichlet(np.ones(n) * 0.5))

        # ---- 1️⃣‑a  Evaluate each starter (cheap, using max‑conv) ----
        local_best_a = None
        local_best_M = float("inf")
        for v in starters:
            a = proj_simplex_sum(v, S_target)
            M = self_convolution(a, fft_len, conv_len).max()
            if M < local_best_M:
                local_best_M = M
                local_best_a = a

        # ---- 2️⃣  LP refinement (time‑boxed) ----
        remaining = deadline - time.monotonic()
        # allocate roughly 40 % of the remaining budget to the LP phase
        lp_deadline = time.monotonic() + remaining * 0.4
        a_lp, M_lp = lp_refine(local_best_a, S_target,
                               fft_len, conv_len, lp_deadline, max_iters=25)

        # ---- 3️⃣  Sub‑gradient refinement (use whatever time is left) ----
        sub_deadline = deadline
        a_sub, M_sub = subgrad_refine(a_lp, S_target,
                                      fft_len, conv_len, sub_deadline,
                                      max_iters=150)

        # ---- 4️⃣  Final evaluation with the provided routine ----
        candidate = np.clip(a_sub, 0.0, 1000.0).tolist()
        try:
            score = evaluate_sequence(candidate)     # supplied externally
        except Exception:
            score = float("inf")

        print(f"[{time.monotonic() - start_time:6.2f}s] Length {n} – final J = {score:.6f}")

        if score < best_score:
            best_score = score
            best_sequence = candidate
            print(f"[{time.monotonic() - start_time:6.2f}s] *** New global best J = {best_score:.6f} ***")

    # ------------------------------------------------------------------
    # Fallback (should never happen, but guarantees a return value)
    # ------------------------------------------------------------------
    if best_sequence is None:
        n = 1024
        S_target = np.sqrt(2.0 * n)
        best_sequence = (np.full(n, S_target / n)).tolist()
        print("Fallback to uniform vector.")

    # Ensure we respect the clipping that ``evaluate_sequence`` performs
    best_sequence = [float(min(1000.0, max(0.0, x))) for x in best_sequence]
    return best_sequence
