def lp_gradient(a, conv, p, n, fft_len_grad):
    w = np.power(conv, p-1)
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1
    grad = 2.0 * full[start:start+n]
    return grad
```

"""
Propose a non‑negative step function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – Use the length of a previously found sequence
   (``height_sequence_1``) if it exists, otherwise fall back to 4096.
2. **Initial guess** – Take the previous best (if available) or a uniform
   distribution and project it onto the probability simplex (sum = 1).
3. **Large‑scale optimisation** – Run a projected Adam optimiser that
   minimises a *p‑norm* surrogate of the convolution (p grows from 2 to
   ≈64).  This smoothly flattens the self‑convolution and converges to the
   true max‑convolution gradient for large p.
4. **Sub‑gradient polishing** – After the p‑norm phase, switch to a pure
   sub‑gradient descent on the exact max‑convolution (soft‑max with a very
   high temperature).  A decreasing learning‑rate schedule and occasional
   random block perturbations avoid stagnation.
5. **Local hill‑climbing** – A cheap pair‑wise mass‑transfer / swap /
   block‑shuffle procedure explores the remaining neighbourhood and can
   still improve the objective.
6. **Time‑budget handling** – The routine respects the 1000 s wall‑clock
   budget (5 s safety margin) and always returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    then project onto the simplex."""
    n = a.size
    # block width between 1% and 10% of n
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Objective helpers (no closures)
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) for a vector that already sums to 1."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def compute_conv_and_J(a: np.ndarray,
                       n: int,
                       fft_len_self: int,
                       conv_len: int):
    """
    Return the full self‑convolution (length 2·n‑1) and the exact J.
    """
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    J = float(2.0 * n * conv.max())
    return conv, J


def lp_gradient(a: np.ndarray,
                conv: np.ndarray,
                p: int,
                n: int,
                fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the Lp‑surrogate  Σ conv[k]^p .
    Gives  g = 2·a * w_rev   with w = conv^{p‑1}  (element‑wise).
    """
    # avoid overflow/underflow – clip to a reasonable range
    w = np.power(conv, p - 1, dtype=np.float64)
    w_rev = w[::-1]

    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    # the gradient lives in the slice [conv_len‑1 : conv_len‑1 + n]
    start = w_rev.size - 1          # = 2·n‑2
    grad = 2.0 * full[start:start + n]
    return grad


def softmax_gradient(a: np.ndarray,
                     conv: np.ndarray,
                     alpha: float,
                     n: int,
                     fft_len_grad: int) -> np.ndarray:
    """
    Gradient of a temperature‑controlled soft‑max surrogate.
    g = 2·a * w_rev   with w = softmax(alpha·conv).
    """
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    w_rev = w[::-1]

    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start = w_rev.size - 1
    grad = 2.0 * full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock budget
    (5 s safety margin) and always returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = min(max(n, 512), 8192)          # keep everything tractable
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for gradients

    # --------------------------------------------------------------
    # 2) Initialisation
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        a0 = proj_simplex(a0)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)    # uniform start

    # keep a copy of the best vector found so far
    best_vec = a0.copy()
    best_score = compute_J(best_vec, n, fft_len_self, conv_len)

    # --------------------------------------------------------------
    # 3) Large‑scale Lp‑norm Adam optimisation
    # --------------------------------------------------------------
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = start_time

    # allocate ~55 % of the remaining budget to the Lp‑optimisation phase
    time_left = deadline - time.monotonic()
    lp_deadline = time.monotonic() + 0.55 * time_left

    while time.monotonic() < lp_deadline:
        it += 1
        now = time.monotonic()
        # progress fraction in [0,1] for the whole optimisation horizon
        frac = (now - start_time) / (deadline - start_time)

        # ----- learning‑rate schedule -----
        lr_start, lr_end = 0.60, 0.001
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # ----- p‑schedule: start quadratic, finish with high‑order -----
        p_start, p_end = 2, 64          # integer powers
        p = int(p_start + (p_end - p_start) * frac)

        # ----- compute convolution and surrogate gradient -----
        conv, cur_J = compute_conv_and_J(a, n, fft_len_self, conv_len)

        # exact objective update
        if cur_J < best_score - 1e-12:
            best_score = cur_J
            best_vec = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Lp‑Adam new best J = {best_score:.6f}")

        # gradient of Σ conv^p  (smooth surrogate for max)
        grad = lp_gradient(a, conv, p, n, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # occasional jitter to escape flat regions
        if rng.random() < 0.015:
            a = _random_block_perturb(a, rng)

        # restart if no progress for a while
        if now - last_improve > 25.0:
            a = best_vec.copy()
            # add a tiny multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Lp‑Adam restart – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    # reuse the best vector as a fresh start
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    last_improve = time.monotonic()

    # allocate the majority of the remaining time to this phase
    while time.monotonic() < deadline - 10.0:   # keep 10 s for final polishing
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)

        # learning‑rate schedule (slower than before)
        lr_start, lr_end = 0.25, 0.0005
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # temperature for soft‑max (approaches true max)
        alpha_start, alpha_end = 5.0, 20000.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        conv, cur_J = compute_conv_and_J(a, n, fft_len_self, conv_len)

        if cur_J < best_score - 1e-12:
            best_score = cur_J
            best_vec = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Soft‑max Adam new best J = {best_score:.6f}")

        # gradient of the softened max (soft‑max surrogate)
        grad = softmax_gradient(a, conv, alpha, n, fft_len_grad)

        # Adam step + projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # occasional random block perturbation
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart on stagnation
        if now - last_improve > 20.0:
            a = best_vec.copy()
            # add a modest multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Soft‑max restart – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Fine‑grained local polishing (pair‑wise transfers / swaps)
    # --------------------------------------------------------------
    a = best_vec.copy()
    cur_score = best_score
    hill_start = time.monotonic()
    # keep a tiny reserve for final safety margin
    while time.monotonic() < deadline - 2.0 and (time.monotonic() - hill_start) < 12.0:
        # randomly decide which elementary move to try
        move = rng.choice(["transfer", "swap", "shuffle"], p=[0.6, 0.3, 0.1])
        cand = a.copy()

        if move == "transfer":
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            max_transfer = min(cand[i], 0.008)   # a modest step size
            if max_transfer <= 0.0:
                continue
            delta = rng.uniform(0.0, max_transfer)
            cand[i] -= delta
            cand[j] += delta
        elif move == "swap":
            i, j = rng.integers(0, n, size=2)
            cand[i], cand[j] = cand[j], cand[i]
        else:  # shuffle a whole block of contiguous indices
            width = rng.integers(2, max(3, n // 20))
            start = rng.integers(0, n - width + 1)
            block = cand[start:start + width]
            rng.shuffle(block)
            cand[start:start + width] = block

        # re‑project to ensure feasibility (sum stays 1)
        cand = proj_simplex(cand)

        cand_score = compute_J(cand, n, fft_len_self, conv_len)
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            best_vec = cand.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Local move improvement J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
