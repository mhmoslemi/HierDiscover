    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 1e-6

    if 'height_sequence_1' in globals():
        raw = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        n = raw.size
    else:
        n = 4096

    # Ensure n within allowed range (maybe 4096)
    n = max(256, n)  # at least 256

    # Helper functions compute max convolution fast using FFT; uses zeros padding for linear conv length 2n-1.

    def max_conv(vec):
        # compute convolution via np.fft.rfft
        fa = np.fft.rfft(vec, n=_next_pow2(2*n - 1))
        conv = np.fft.irfft(fa * fa, n=_next_pow2(2*n - 1))
        # conv length at least 2n-1; take first 2n-1 elements
        return conv[:2*n-1].max()

    def compute_J(vec):
        return 2.0 * n * max_conv(vec)

    # initial a: Dirichlet with alpha=0.5 or use previous.
    if 'height_sequence_1' in globals():
        raw = np.asarray(globals()['height_sequence_1'], dtype=np.float64)

        if raw.size != n:

        a = proj_simplex(raw)

"""
Propose a step‑function sequence that lowers
``evaluate_sequence`` (i.e. 2·n·max(conv) with sum(a)=1).

Strategy
--------
1. **Initialisation** – reuse the previous construction if present,
   otherwise draw a Dirichlet vector (≈ uniform) and project onto the
   probability simplex (sum=1, non‑negative).

2. **Phase 1 – Smoothed projected‑gradient descent**  
   * A soft‑max (log‑sum‑exp) approximates the max‑convolution.
   * The gradient of the smooth surrogate is obtained analytically via
     two FFTs per iteration.
   * An adaptive back‑tracking line‑search updates the iterate,
     followed by Euclidean projection onto the simplex.
   * Temperature (soft‑max sharpness) is increased slowly from 2 to a
     large value (≈ 2·10⁴) so the surrogate becomes the true max.
   * When progress stalls a multiplicative log‑normal jitter on a random
     block is applied to escape plateaus.

3. **Phase 2 – Simulated‑annealing fine‑tuning**  
   * Starting from the best vector of Phase 1, repeatedly move a small
     amount of mass between two random coordinates.
   * Accept moves that improve the hard objective, otherwise accept with
     Metropolis probability `exp(-(ΔJ)/T)`.  Temperature `T` is
     exponentially cooled.
   * Track the best ever‑seen vector.

The whole optimisation respects the 1000 s budget (a small safety
margin is kept).  The final vector is clipped to the limits imposed by
``evaluate_sequence`` (≥ 0, ≤ 1000) and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    # block width between 1 and at most 10 % of the vector
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.25))   # a bit larger than before
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = proj_simplex(out)
    return out


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine combines a smoothed projected‑gradient
    descent with a short simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling – keep a 3 s safety margin
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)

    # --------------------------------------------------------------
    # 1) Problem size & initial point
    # --------------------------------------------------------------
    # Use the length of the previously found construction if available,
    # otherwise fall back to a default size of 4096.
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = int(prev.size)
        prev = np.clip(prev, 0.0, None)
        a0 = proj_simplex(prev)
    else:
        n = 4096
        a0 = proj_simplex(rng.dirichlet(np.full(n, 0.5)))

    a = a0.copy()

    # --------------------------------------------------------------
    # Helper: hard objective J = 2·n·max(conv) (sum(a)=1)
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for a * w_rev
    start_idx = (fft_len_grad - conv_len) // 2   # index where the relevant part starts

    def compute_J(vec: np.ndarray) -> float:
        fa = np.fft.rfft(vec, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        return float(2.0 * n * conv.max())

    best_a = a.copy()
    best_J = compute_J(a)
    print(f"[init] length={n}, J={best_J:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 1 – smoothed projected gradient descent
    # --------------------------------------------------------------
    step = 0.8           # initial step size
    step_min = 1e-9
    step_max = 5.0
    alpha = 2.0           # soft‑max temperature (sharpness)
    alpha_mul = 1.003     # geometric increase per iteration
    alpha_max = 20000.0
    last_improve = time.monotonic()
    stagnation_limit = 25.0   # seconds without improvement → jitter

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        remaining = deadline - now
        # Reserve ~200 s for the SA phase
        if remaining < 200.0:
            break

        iteration += 1

        # ----- compute convolution and soft‑max weights -----
        fa_self = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa_self * fa_self, n=fft_len_self)[:conv_len]

        c_max = conv.max()
        shifted = conv - c_max               # for numerical stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- back‑tracking line‑search on the *hard* objective -----
        improved = False
        step_try = step
        while step_try >= step_min:
            cand = proj_simplex(a - step_try * grad)
            Jcand = compute_J(cand)
            if Jcand < best_J - 1e-12:
                improved = True
                break
            step_try *= 0.5

        if improved:
            a = cand
            best_a = a.copy()
            best_J = Jcand
            last_improve = now
            step = min(step_try * 1.3, step_max)
            print(f"[{now - (deadline - budget_s + 3):6.2f}s] iter {iteration:5d}  "
                  f"new best J={best_J:.6f}  step={step:.4f}  α={alpha:.1f}")
        else:
            step = max(step * 0.5, step_min)

        # ----- temperature schedule -----
        alpha = min(alpha * alpha_mul, alpha_max)

        # ----- stagnation handling -----
        if now - last_improve > stagnation_limit:
            a = _random_block_perturb(best_a, rng)
            step = max(step, 0.2)
            last_improve = now
            print(f"[{now - (deadline - budget_s + 3):6.2f}s] jitter – J stays {best_J:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 2 – simulated annealing fine‑tuning
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_J = best_J
    temperature = 0.2                      # initial SA temperature
    cooling = 0.9995                       # exponential cooling factor
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # propose a simple mass‑transfer move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a small fraction of a[i] to a[j]
        delta = rng.random() * a[i] * 0.25
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = proj_simplex(cand)          # re‑project to keep sum=1

        Jcand = compute_J(cand)

        if Jcand < cur_J:
            a = cand
            cur_J = Jcand
            if Jcand < best_J:
                best_J = Jcand
                best_a = cand.copy()
                print(f"[SA it {it}] new best J={best_J:.6f}")
        else:
            # Metropolis acceptance
            prob = np.exp(-(Jcand - cur_J) / temperature)
            if rng.random() < prob:
                a = cand
                cur_J = Jcand

        temperature = max(temperature * cooling, 1e-6)

    # --------------------------------------------------------------
    # 4) Return the best sequence (respect evaluate_sequence's clipping)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
