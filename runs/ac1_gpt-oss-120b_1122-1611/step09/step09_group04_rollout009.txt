"""
Propose a step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm
---------

1. **Problem size** – infer the length ``n`` from the previous
   solution (``height_sequence_1``) or fall back to 3809.

2. **Cheap global exploration** – generate a diverse set of
   probability vectors (sum = 1) from several low‑dimensional families:
   * uniform baseline
   * two‑edge blocks with linear taper (the “edge shape” used before)
   * symmetric cosine shapes
   * symmetric exponential decays from the ends
   * symmetric power‑law decays from the ends
   * a handful of Dirichlet draws (different concentration)
   * random Fourier‑phase vectors (real, clipped to non‑negative)
   * the previous best sequence (if any)

   For each candidate we compute the true objective
   ``J = 2·n·max(conv(p,p))`` (with ``p`` on the simplex) using an
   FFT based convolution.  The best ``K`` candidates are kept (default
   ``K = 15``).

3. **Local refinement** – from each seed we run a projected Adam optimiser
   on a *smooth‑max* surrogate of the convolution.
   The surrogate uses a temperature‑controlled soft‑max over the
   convolution entries.  The temperature grows exponentially from
   ``α_start = 1`` to ``α_end = 10000``; the learning‑rate decays
   exponentially from ``lr_start = 0.8`` to ``lr_end = 0.001``.
   After 70 % of the allocated time the optimizer switches to the
   exact max‑convex sub‑gradient (the gradient of the active convolution
   index) for fine tuning.

   Projection onto the probability simplex is performed after each
   update (Euclidean projection).  Small jitter and a restart on
   stagnation keep the search from getting trapped.

4. **Result** – the best probability vector found is scaled by
   ``√(2·n)`` (the normalisation required by ``evaluate_sequence``) and
   clipped to ``[0,1000]`` before returning.

The whole routine respects the global time‑budget (default 1000 s) and
uses at most two CPUs (NumPy’s FFT is multithreaded).  All helper
functions are top‑level, no closures or lambdas are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape: (1+r·cosθ)^power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L with linear taper controlled by gamma.
    When gamma==0 the blocks are uniform; gamma==1 makes them drop to
    zero at the inner border.
    """
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _exp_shape(n: int, beta: float) -> np.ndarray:
    """Symmetric exponential decay from both ends, normalised."""
    i = np.arange(n, dtype=np.float64)
    v = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
    return v / v.sum()


def _power_shape(n: int, gamma: float) -> np.ndarray:
    """Symmetric power‑law decay from the ends, normalised."""
    i = np.arange(n, dtype=np.float64) + 1.0
    v = (i ** -gamma) + ((n + 1 - i) ** -gamma)
    return v / v.sum()


def _fourier_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Random Fourier‑phase shape:
    draw random phases, build a symmetric real signal via iFFT,
    clip to non‑negative and renormalise.
    """
    # rfft length for real signal of size n
    half = n // 2 + 1
    phases = rng.random(half) * 2.0 * np.pi
    # unit magnitude spectrum (except DC which we keep real)
    spectrum = np.exp(1j * phases)
    # Enforce Hermitian symmetry implicitly via irfft
    a = np.fft.irfft(spectrum, n=n)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()
    return a


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max weights (sum = 1)."""
    # numerical stability
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate with respect to ``a``."""
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1          # = 2·n‑2
    return 2.0 * full[start:start + a.size]


def _grad_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact sub‑gradient of ``conv_k_max`` w.r.t. ``a``.
    For any index i where 0 ≤ k_max‑i < n we have ∂/∂a_i = 2·a_{k_max‑i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    # i ranges from lo..hi inclusive
    i = np.arange(lo, hi + 1)
    j = k_max - i
    grad[i] = 2.0 * a[j]
    return grad


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """True objective J = 2·n·max(p*p) for a probability vector p."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  Returns a list of floats (scaled to the
    required L¹‑norm √(2·n)).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
        except Exception:
            n = 3809
    else:
        n = 3809

    # We keep FFT lengths as powers of two for speed.
    fft_len_self = _next_pow2(2 * n - 1)     # for self‑convolution
    fft_len_grad = _next_pow2(3 * n - 2)     # for gradient (soft‑max)

    # --------------------------------------------------------------
    # 2) Global cheap exploration (enumerate families)
    # --------------------------------------------------------------
    top_k = 15                     # number of seeds to keep
    best_candidates = []           # (J, p) tuples, p on simplex

    def _push(p_vec: np.ndarray):
        """Evaluate a candidate and keep the best ``top_k``."""
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > top_k:
            best_candidates.pop()

    # a) uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push(uniform)

    # b) edge‑shape grid (block‑size fraction & taper)
    for frac in np.linspace(0.12, 0.45, 17):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 13):
            p = _edge_shape(n, L, gamma)
            _push(p)

    # c) cosine‑family grid
    for r in np.linspace(0.0, 1.0, 13):
        for pw in np.linspace(0.3, 3.0, 9):
            p = _cosine_shape(n, r, pw)
            _push(p)

    # d) exponential decay grid
    for beta in np.logspace(-3, -0.1, 12):
        p = _exp_shape(n, beta)
        _push(p)

    # e) power‑law decay grid
    for gam in np.linspace(0.5, 3.0, 12):
        p = _power_shape(n, gam)
        _push(p)

    # f) random Dirichlet draws (moderately concentrated)
    for _ in range(40):
        alpha = rng.uniform(0.05, 0.7)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # g) random Fourier‑phase draws
    for _ in range(20):
        p = _fourier_shape(n, rng)
        _push(p)

    # h) include the previous best sequence if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = uniform
            else:
                prev = proj_simplex(prev / prev.sum())
            _push(prev)
        except Exception:
            pass

    # Keep the best seed found so far
    best_candidates.sort(key=lambda x: x[0])
    best_J, best_p = best_candidates[0]

    # --------------------------------------------------------------
    # 3) Local refinement – projected Adam on soft‑max surrogate,
    #    then switch to exact max gradient.
    # --------------------------------------------------------------
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # Temperature schedule (exponential growth)
    alpha_start, alpha_end = 1.0, 10000.0

    # Learning‑rate schedule (exponential decay)
    lr_start, lr_end = 0.8, 0.001

    def _adam_refine(p_start: np.ndarray, time_limit: float):
        """Run Adam from ``p_start`` for at most ``time_limit`` seconds."""
        nonlocal best_J, best_p
        a = p_start.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        it = 0
        start = time.monotonic()
        last_improve = start

        # Decide the switch point (soft‑max → exact max)
        switch_frac = 0.70          # fraction of the allocated time

        while time.monotonic() < time_limit:
            it += 1
            now = time.monotonic()
            elapsed = now - start
            total_alloc = time_limit - start
            frac = elapsed / total_alloc if total_alloc > 0 else 1.0

            # learning‑rate schedule (exponential decay)
            lr = lr_start * (0.5 ** frac) + lr_end * (1 - 0.5 ** frac)

            # temperature schedule (exponential growth)
            alpha = alpha_start * np.exp(frac * np.log(alpha_end / alpha_start))

            # ---------------------------------------------------------
            # Compute convolution and gradient
            # ---------------------------------------------------------
            conv = _self_conv(a, fft_len_self)

            if frac < switch_frac:
                # soft‑max surrogate
                w = _softmax_weights(conv, alpha)
                grad = _grad_softmax(a, w, fft_len_grad)
            else:
                # exact max (sub‑gradient of the active index)
                k_max = int(np.argmax(conv))
                grad = _grad_max_conv(a, k_max)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # Projection onto the simplex
            a = proj_simplex(a)

            # ---------------------------------------------------------
            # Evaluate true objective (no smoothing)
            # ---------------------------------------------------------
            cur_J = _objective(a, fft_len_self, n)

            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_p = a.copy()
                last_improve = now
                prog = now - start
                print(f"[{prog:.1f}s] New best J = {best_J:.6f}")

            # ---------------------------------------------------------
            # Occasional jitter to escape plateaus
            # ---------------------------------------------------------
            if rng.random() < 0.015:
                block_width = rng.integers(1, max(2, n // 30))
                start_idx = rng.integers(0, n - block_width + 1)
                factor = np.exp(rng.normal(0.0, 0.06))
                a_pert = a.copy()
                a_pert[start_idx:start_idx + block_width] *= factor
                a_pert = proj_simplex(a_pert)
                a = a_pert

            # ---------------------------------------------------------
            # Restart on stagnation
            # ---------------------------------------------------------
            if now - last_improve > 22.0:
                a = best_p.copy()
                noise = np.exp(rng.normal(0.0, 0.05, size=n))
                a = proj_simplex(a * noise)
                last_improve = now
                prog = now - start
                print(f"[{prog:.1f}s] Restart (stagnation) – best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Run Adam from the top seeds while respecting the remaining budget
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # No time left – just return the current best
        scale = np.sqrt(2.0 * n)
        seq = np.clip(best_p * scale, 0.0, 1000.0)
        return [float(x) for x in seq]

    # Allocate 85 % of what is left to refinement, the rest stays as safety.
    refine_budget = remaining * 0.85
    per_seed = refine_budget / max(1, len(best_candidates))

    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break
        time_limit = min(per_seed, deadline - time.monotonic())
        _adam_refine(p0, time_limit)

    # --------------------------------------------------------------
    # 5) Return the best sequence (scaled to √(2·n) as required by the evaluator)
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
