    start = time.monotonic()
    deadline = start + budget_s - 5

    # initialization
    if 'height_sequence_1' in globals():

    else:
        n = 4096
        a = proj_simplex(np.random.default_rng(seed).dirichlet(np.full(n, 0.5)))

    n = a.size
    best_a = a.copy()
    best_J = compute_J(a)  # using 2*n*conv.max()
    print(f"Initial J {best_J}")

    # Phase 1: gradient descent (optional)
    # Use same code but maybe fewer iterations to warm up

    # Phase 2: LP improvement loop
    while time.monotonic() < deadline:
        # compute current convolution
        conv = conv_self(a)
        M = conv.max()
        # Build matrix
        data = np.tile(a, (n, 1))  # shape (n, n)
        offsets = np.arange(n)
        A_ub = sp.diags(data, offsets, shape=(2*n-1, n), format='csr')
        b_ub = np.full(2*n-1, M)
        # Solve LP
        res = opt.linprog(c=np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
        if not res.success:
            print("LP solve failed", res.message)
            break
        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()
        if sum_g <= sum_a + 1e-12:
            print("No improvement in sum")
            break
        # scaling
        g_scaled = g * (sum_a / sum_g)
        # line search on tau
        tau = 0.5
        improved = False
        while tau > 1e-3:
            new_a = (1 - tau) * a + tau * g_scaled
            new_J = compute_J(new_a)
            if new_J < best_J - 1e-12:
                a = new_a
                best_J = new_J
                best_a = a.copy()
                improved = True
                print(f"LP iter improved J={best_J:.6f}")
                break
            tau *= 0.5
        if not improved:
            print("LP line search failed to improve")
            break
    # return best
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]

"""
Step‑function optimizer.

The goal is to minimise
    J(a) = 2·n·max_k (a * a)[k]          (sum(a)=1, a_i ≥ 0)

where “*” is the discrete (full) convolution.  The problem is
non‑convex, so we combine a fast exponentiated‑gradient descent on a
soft‑max approximation of the max‑convolution with occasional
random “jitter’’ moves to escape plateaus.  The method works directly
on the full‑size vector (≈ 3800 entries) and stays well inside the
1000 s time‑budget.

Algorithm
---------
0. Initialise a probability vector on the simplex.
   – If a previous construction (``height_sequence_1``) exists we reuse it.
   – Otherwise a Dirichlet draw (≈ uniform) is used.
1. Repeatedly:
   * Compute the full convolution a∗a with an FFT.
   * Identify the indices attaining the current maximum.
   * Build a soft‑max weight vector w that is uniform on the active set.
   * The (sub)gradient of the max‑convolution is
         g = 2·a ⊛ rev(w)               (⊛ = convolution)
     which we obtain with another FFT.
   * Perform an exponentiated‑gradient step
         a ← a·exp(−η·g) ,   η decreasing from 2 → 0.02,
     followed by a renormalisation onto the simplex.
   * If the objective improves we keep the new point, otherwise we
     keep the old one.
   * When no improvement has been seen for a while we apply a tiny
     random block perturbation (log‑normal multiplicative noise) to
     the current best vector – this helps to jump out of flat plateaus.
2. The best vector encountered before the deadline is returned
   (clipped to the limits of ``evaluate_sequence``).

The code is fully self‑contained; only ``numpy`` and ``time`` are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    # block width = 1 … max(2, n//30)
    width = rng.integers(1, max(2, n // 30))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine uses an
    exponentiated‑gradient descent on a soft‑max surrogate of the
    max‑convolution, with an adaptive learning‑rate schedule and
    occasional stochastic jitter moves.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) problem size & initial point
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # enforce a reasonable length (original code used ≤4096)
        max_len = 4096
        if raw.size > max_len:
            raw = raw[:max_len]
        elif raw.size < max_len // 2:      # pad with zeros if too short
            raw = np.concatenate([raw, np.zeros(max_len - raw.size)])
        raw = np.clip(raw, 0.0, None)
        a = proj_simplex(raw)
    else:
        # fall back to a Dirichlet draw (α=0.5 gives some variability)
        n = 4096
        a = proj_simplex(rng.dirichlet(np.full(n, 0.5)))
    a = a.astype(np.float64)
    n = a.size

    # --------------------------------------------------------------
    # 2) FFT helpers
    # --------------------------------------------------------------
    conv_len = 2 * n - 1                     # length of full convolution
    fft_len_self = _next_pow2(conv_len)      # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)      # for a ⊛ w_rev
    start_idx = (fft_len_grad - conv_len) // 2   # slice to obtain length‑n result

    # --------------------------------------------------------------
    # 3) objective helpers
    # --------------------------------------------------------------
    def _max_convolution(v: np.ndarray) -> float:
        """Return max_k (v∗v)[k] via FFT."""
        fa = np.fft.rfft(v, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        return float(conv.max())

    def _objective(v: np.ndarray) -> float:
        """Return the evaluation value (2·n·max_conv) for a vector already on the simplex."""
        return 2.0 * n * _max_convolution(v)

    # --------------------------------------------------------------
    # 4) initialise bookkeeping
    # --------------------------------------------------------------
    best_a = a.copy()
    best_J = _objective(a)
    cur_a = a.copy()
    cur_J = best_J
    last_improve = start_time
    stagnation_limit = 30.0       # seconds without improvement → jitter
    # learning‑rate schedule (η goes from η_start down to η_end)
    eta_start = 2.0
    eta_end = 0.02

    iteration = 0
    print(f"[init] n = {n}, J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5) main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        elapsed = now - start_time
        total_allowed = deadline - start_time

        # ---- 5.1 compute convolution and identify active indices ----
        fa_self = np.fft.rfft(cur_a, n=fft_len_self)
        conv = np.fft.irfft(fa_self * fa_self, n=fft_len_self)[:conv_len]
        max_val = conv.max()
        # active set: all positions within a tiny tolerance of the maximum
        tol = 1e-14
        active = (np.abs(conv - max_val) <= tol).astype(np.float64)
        if active.sum() == 0.0:
            # fall back to a uniform weight (should never happen)
            active = np.full_like(conv, 1.0 / conv_len)
        else:
            # turn into a probability vector over the active indices
            active = active / active.sum()

        # ---- 5.2 sub‑gradient of the max‑convolution (soft‑max = active) ----
        w_rev = active[::-1]                     # reversed weight vector
        fa_a = np.fft.rfft(cur_a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fw, n=fft_len_grad)
        grad = 2.0 * full[start_idx:start_idx + n]   # shape (n,)

        # ---- 5.3 learning‑rate (linear annealing) ----
        frac = min(1.0, elapsed / total_allowed)   # ∈[0,1]
        eta = (1.0 - frac) * eta_start + frac * eta_end

        # ---- 5.4 exponentiated‑gradient update ----
        # a_i ← a_i·exp(−η·grad_i), then renormalise onto the simplex
        upd = cur_a * np.exp(-eta * grad)
        s = upd.sum()
        if s == 0.0 or np.isnan(s):
            # numerical safeguard – fall back to a uniform step
            upd = proj_simplex(cur_a)
        else:
            upd = upd / s

        # ---- 5.5 accept only if the true objective does not increase ----
        new_J = _objective(upd)
        if new_J < cur_J - 1e-12:          # strict improvement
            cur_a, cur_J = upd, new_J
            if new_J < best_J - 1e-12:
                best_a, best_J = upd.copy(), new_J
                last_improve = now
                print(f"[{elapsed:6.2f}s] iter {iteration:5d} new best J = {best_J:.6f}")
        else:
            # no improvement → keep the old point (but keep searching)
            pass

        # ---- 5.6 stagnation handling (random jitter) ----
        if now - last_improve > stagnation_limit:
            cur_a = _random_block_perturb(best_a, rng)
            cur_J = _objective(cur_a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] jitter – restarting from best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clip to evaluation limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
