Simulated‑annealing optimizer for the self‑convolution bound.
The algorithm works directly on the probability simplex and moves a
fraction of mass from one coordinate to another while maintaining an
exact self‑convolution (linear plus quadratic terms).  Acceptance follows
a temperature schedule, occasional smoothing/perturbations are applied
to escape local minima.  The method is fast (O(n) per move) and fits
comfortably into the 1000 s budget.
"""

import time
import numpy as np

def _initial_vector(seed: int, n: int) -> np.ndarray:
    rng = np.random.default_rng(seed)
    return rng.dirichlet(np.ones(n))

def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step function that minimizes ``evaluate_sequence``.

    The routine tries to improve the best known vector (``height_sequence_1`` if
    present) using a custom simulated‑annealing scheme that moves continuous
    mass between coordinates while keeping the sum equal to one.  All updates
    keep the exact self‑convolution (including the quadratic ``Δ*Δ`` term) so
    that the objective can be evaluated cheaply in O(n) time.

    Parameters
    ----------
    seed: int
        Random seed for reproducibility.
    budget_s: float
        Total wall‑clock budget in seconds.  Five seconds are reserved for the
        caller, so the internal deadline is ``budget_s‑5``.
    kwargs: dict
        Ignored – kept for compatibility with the platform.

    Returns
    -------
    list[float]
        The best vector found (non‑negative, each entry ≤ 1000).  The vector is
        already normalised to sum to one, which makes the value returned by
        ``evaluate_sequence`` equal to ``2·n·max(a*a)``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine problem size ------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)
            n = base.size
            n = max(512, min(n, 4096))
            a = base[:n].copy()
            total = a.sum()
            if total < 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a /= total
            init_source = "height_sequence_1"
        except Exception:
            n = 4096
            a = _initial_vector(seed, n)
            init_source = "random"
    else:
        n = 4096
        a = _initial_vector(seed, n)
        init_source = "random"

    # ------------------------------------------------------------------
    def _compute_J(conv_arr: np.ndarray) -> float:
        return float(2.0 * n * conv_arr.max())

    # ------------------------------------------------------------------
    # Initialise convolution and objective
    # ------------------------------------------------------------------
    conv = np.convolve(a, a)                 # full self‑convolution, length 2·n‑1
    cur_J = _compute_J(conv)
    best_vec = a.copy()

    # ------------------------------------------------------------------
    start = time.monotonic()
    attempts = 0
    accepts = 0

        attempts += 1
        elapsed = time.monotonic() - start
        # fraction of the total allotted time (0 … 1)

def _initial_vector(seed: int, n: int) -> np.ndarray:

    rng = np.random.default_rng(seed)
    return rng.dirichlet(np.ones(n))
