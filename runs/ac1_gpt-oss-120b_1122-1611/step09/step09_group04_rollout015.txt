"""
Propose a better step‑function sequence.

Algorithm
---------
1.  Determine the problem size ``n`` (use the length of the previous
    construction if present, otherwise fall back to a modest default).
2.  Generate a diverse pool of seed vectors on the probability simplex:
    * uniform,
    * two‑edge blocks with optional taper,
    * symmetric cosine families,
    * a handful of Dirichlet draws,
    * the previously found best sequence (if any).
3.  Keep the five seeds with the smallest true objective
    ``J = 2·n·max(conv(p))``.
4.  For each seed run a *projected sub‑gradient* descent on the true
    nonsmooth objective.  The sub‑gradient of ``max(conv(p))`` is obtained
    by a single FFT‑based convolution with a one‑hot weight vector.
    The update is projected onto the probability simplex.
    * Learning‑rate follows an exponential decay.
    * Small random “jitter” steps are injected occasionally to escape
      plateaus.
    * If no improvement occurs for a while the iterate is restarted from
      the best point seen so far with a tiny unbiased perturbation.
5.  The best probability vector found is finally scaled to satisfy the
    original normalisation (sum = √(2·n)) and clipped to the limits
    enforced inside ``evaluate_sequence``.
6.  The routine respects a global time‑budget (default 1000 s) and
    prints progress information.

The method is deliberately different from the original two‑phase Adam
approach: it works directly with the true max‑convolution (no soft‑max
approximation) and uses a simple projected sub‑gradient scheme together
    with extensive global seeding.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L with linear taper.
    gamma=0 → uniform blocks, gamma=1 → taper to zero.
    """
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)

    if L > 1:
        idx = np.arange(L)
        left = 1.0 - gamma * (idx / (L - 1))
        v[idx] = left
        v[-L + idx] = left[::-1]
    else:
        v[0] = 1.0
        v[-1] = 1.0

    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape: (1 + r·cos θ)^power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights (temperature ``alpha``) that sum to 1."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the weighted‑sum surrogate  Σ_k w_k·(a*a)[k]  w.r.t. a.
    ``w`` must have length 2·n‑1 (the linear convolution length).
    """
    w_rev = w[::-1]                       # reverse for convolution
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start_idx = w_rev.size - 1               # = 2n‑2
    # Only the part that aligns with the original vector is needed.
    return 2.0 * full[start_idx:start_idx + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """True objective J = 2·n·max(p * p) for a probability vector ``p``."""
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


def _subgradient_refine(p_start: np.ndarray,
                        n: int,
                        fft_len_self: int,
                        fft_len_grad: int,
                        rng: np.random.Generator,
                        time_limit: float) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best probability vector found and its objective value.
    """
    a = p_start.copy()
    best_a = a.copy()
    best_J = _objective(a, fft_len_self, n)

    start = time.monotonic()
    last_improve = start
    it = 0

    # learning‑rate schedule (exponential decay)
    lr_start, lr_end = 0.7, 0.001

    while time.monotonic() < time_limit:
        it += 1
        now = time.monotonic()
        frac = (now - start) / (time_limit - start)   # ∈[0,1]

        lr = lr_start * (0.5 ** frac) + lr_end * (1 - 0.5 ** frac)

        # ---- convolution and sub‑gradient at the current max ----
        conv = _self_conv(a, fft_len_self)
        max_val = conv.max()
        mask = conv == max_val
        # one‑hot (or averaged) weight vector for the current max index set
        w = mask.astype(np.float64)
        w /= w.sum()
        grad = _grad_softmax(a, w, fft_len_grad)

        # ---- projected sub‑gradient step ----
        a = proj_simplex(a - lr * grad)

        # ---- evaluate ----
        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{now - start:.1f}s] sub‑grad new best J = {best_J:.6f}")

        # ---- occasional jitter to leave plateaus ----
        if rng.random() < 0.015:
            block_w = max(1, n // 80)
            i0 = rng.integers(0, n - block_w + 1)
            factor = np.exp(rng.normal(0.0, 0.07))
            a_pert = a.copy()
            a_pert[i0:i0 + block_w] *= factor
            a = proj_simplex(a_pert)

        # ---- restart on long stagnation ----
        if now - last_improve > 22.0:
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[{now - start:.1f}s] restart (stagnation) – best J = {best_J:.6f}")

    return best_a, best_J

    for frac in np.linspace(0.12, 0.48, 12):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            p = _edge_shape(n, L, gamma)
            push_candidate(p)

    # c) cosine‑family grid
    for r in np.linspace(0.0, 1.0, 11):
        for pw in np.linspace(0.2, 3.5, 9):
            p = _cosine_shape(n, r, pw)
            push_candidate(p)

    # d) random Dirichlet draws with varying concentration
    for _ in range(40):
        alpha = rng.uniform(0.02, 0.9)
        p = rng.dirichlet(np.full(n, alpha))
        push_candidate(p)

    # e) include previous construction if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = uniform
            else:
                prev = proj_simplex(prev / prev.sum())
            push_candidate(prev)
        except Exception:
            pass

    # keep the best few seeds
    candidate_list.sort(key=lambda x: x[0])
    seed_candidates = candidate_list[:5]  # list of (J, p)

    best_J, best_p = seed_candidates[0]

    # --------------------------------------------------------------
    # 3) Local refinement – projected sub‑gradient descent
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # time ran out – just return the best we have
        scale = np.sqrt(2.0 * n)
        best_seq = best_p * scale
        return [float(v) for v in np.clip(best_seq, 0.0, 1000.0)]

    # allocate roughly 80 % of the remaining budget to refinement
    refine_budget = remaining * 0.80
    per_seed = refine_budget / max(1, len(seed_candidates))

    for _, p0 in seed_candidates:
        if time.monotonic() >= deadline:
            break
        limit = min(per_seed, deadline - time.monotonic())
        p_opt, J_opt = _subgradient_refine(p0,
                                            n,
                                            fft_len_self,
                                            fft_len_grad,
                                            rng,
                                            limit)
        if J_opt < best_J - 1e-12:
            best_J, best_p = J_opt, p_opt.copy()
            print(f"*** new global best after seed refinement: J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Final scaling to the form expected by ``evaluate_sequence``
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)                 # required L1‑norm before clipping
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Convert to Python list of floats as required by the external grader
    return [float(x) for x in best_seq]
