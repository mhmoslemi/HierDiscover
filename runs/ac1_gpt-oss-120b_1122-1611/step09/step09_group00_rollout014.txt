    a = a_start.copy()
    log_a = np.log(a + 1e-15)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    best_local_a = a.copy()
    best_local_score = _score(a)
    for t in range(1, max_iters+1):
        if time.monotonic() >= time_limit:
            break
        conv = _conv_fft(a, fft_len)
        max_b = conv.max()
        idx_max = int(np.argmax(conv))
        grad = _gradient_from_active(a, np.array([idx_max]))
        # gradient w.r.t log a
        g_log = grad * a
        # Adam updates
        m = beta1*m + (1-beta1)*g_log
        v = beta2*v + (1-beta2)*(g_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr = 0.5 / np.sqrt(t)   # adapt lr
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

            best_local_a = a.copy()
            # also update global best:
            update_best(a, score)
    return best_local_a, best_local_score

"""
Hybrid parametric + Adam + Simulated‑Annealing optimizer for the
max‑convolution objective.

The algorithm proceeds in three phases:

1. Quick exploration of a few analytically‑motivated families
   (symmetric exponential, random spectral phase) to obtain a good
   starting point.
2. Adam optimisation on the log‑probabilities (equivalently,
   exponentiated‑gradient with adaptive moments) using the exact
   sub‑gradient of the active convolution index.
3. A temperature‑controlled simulated‑annealing stage that performs
   tiny directed mass‑transfer moves together with occasional jitter
   to escape remaining local minima.

All updates respect the simplex constraint ∑a_i=1 by normalising after
each step.  The internal score is 2·n·max(a∗a) which, for a normalised
vector, equals the value returned by ``evaluate_sequence``.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.full_like(v, 1.0 / n)
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical fix if all zero
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w

def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]

def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max_k (a*a)[k] for the active set."""
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad

def _score(a: np.ndarray, fft_len: int) -> float:
    """Return the objective 2·n·max(a*a) assuming sum(a)=1."""
    n = a.size
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # ----------------------------------------------------------------------
    # 0) Problem size
    # ----------------------------------------------------------------------
    n = 4096                                            # a power of two is convenient
    fft_len = _next_pow2(2 * n)                         # enough for linear convolution

    # ----------------------------------------------------------------------
    # 1) Helper to keep the current global best
    # ----------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    def _maybe_update(a_vec: np.ndarray, sc: float) -> None:
        nonlocal best_seq, best_score
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = a_vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new global best = {best_score:.6f}")

    # ----------------------------------------------------------------------
    # 2) Warm‑up: analytically motivated families
    # ----------------------------------------------------------------------
    # 2a) Symmetric exponential decay (mass concentrated at the ends)
    lam_vals = np.logspace(-5, -0.3, num=30)
    for lam in lam_vals:
        if time.monotonic() >= deadline:
            break
        idx = np.arange(n)
        dist = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        dist /= dist.sum()
        sc = _score(dist, fft_len)
        _maybe_update(dist, sc)

    # 2b) Random spectral phases (flat magnitude in Fourier domain)
    for _ in range(30):
        if time.monotonic() >= deadline:
            break
        phase = rng.uniform(0.0, 2.0 * np.pi, size=fft_len // 2 + 1)
        G = np.exp(1j * phase)
        a = np.fft.irfft(G, n=fft_len)[:n]
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            continue
        a /= s
        sc = _score(a, fft_len)
        _maybe_update(a, sc)

    # 2c) Uniform distribution (useful as a sanity baseline)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _maybe_update(uniform, _score(uniform, fft_len))

    # If we have a previous construction, feed it into the pool as well
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if prev.size != n:
            if prev.size < n:
                prev = np.pad(prev, (0, n - prev.size))
            else:
                prev = prev[:n]
        prev = np.clip(prev, 0.0, None)
        if prev.sum() > 0.0:
            prev /= prev.sum()
            _maybe_update(prev, _score(prev, fft_len))

    # ----------------------------------------------------------------------
    # 3) Adam optimisation on the log‑probabilities (gradient descent)
    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – start from uniform
        a = uniform.copy()
    else:
        a = best_seq.copy()

    log_a = np.log(a + 1e-15)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # Reserve ~70 % of the remaining budget for Adam
    adam_deadline = start_time + 0.70 * (deadline - start_time)

    it = 0
    while time.monotonic() < adam_deadline:
        # linear convolution and active index
        conv = _conv_fft(a, fft_len)
        max_val = conv.max()
        idx_max = int(np.argmax(conv))
        grad = _gradient_from_active(a, np.array([idx_max]))
        # gradient wrt log‑a (since a = exp(log_a))
        g_log = grad * a

        it += 1
        # Adam bookkeeping
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # learning‑rate schedule – a gentle decay
        lr = 0.6 / np.sqrt(it)

        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.exp(log_a)
        a /= a.sum()                       # projection onto simplex

        # evaluate (fast)
        cur_score = 2.0 * n * np.max(_conv_fft(a, fft_len))
        _maybe_update(a, cur_score)

        # occasional progress report
        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Adam iter={it}, best={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning
    # ----------------------------------------------------------------------
    # Use the best sequence obtained so far as the starting point
    if best_seq is None:
        current = uniform.copy()
        current_score = _score(current, fft_len)
    else:
        current = best_seq.copy()
        current_score = best_score

    # Temperature schedule – start relatively hot and cool exponentially
    T = 0.05 * current_score
    cooling = 0.9998

    iter_local = 0
    while time.monotonic() < deadline:
        # pick a source index – proportional to current mass (larger mass → larger impact)
        src = int(rng.choice(np.arange(n), p=current))

        # pick a target index – preferably a low‑mass position
        low_mask = current < np.percentile(current, 20)
        low_idxs = np.where(low_mask)[0]
        if low_idxs.size:
            tgt = int(rng.choice(low_idxs))
        else:
            tgt = int(rng.integers(0, n))

        if src == tgt:
            continue

        # fraction of source mass to move; schedule it with temperature so larger moves early
        max_frac = min(0.20, T / current_score + 0.001)   # shrink as we cool
        frac = rng.uniform(0.001, max_frac)
        delta = current[src] * frac
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # small numerical correction – keep on the simplex
        cand = _project_onto_simplex(cand)

        cand_score = _score(cand, fft_len)

        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (anneal) = {best_score:.6f}")

        # temperature decay
        T *= cooling
        iter_local += 1

        # occasional jitter – a tiny random scaling of a short window
        if iter_local % 4000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            jitter = current.copy()
            jitter[start_idx:start_idx + width] *= factor
            jitter = _project_onto_simplex(jitter)
            jitter_score = _score(jitter, fft_len)
            if jitter_score < best_score:
                best_score = jitter_score
                best_seq = jitter.copy()
                current = jitter
                current_score = jitter_score
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (jitter) = {best_score:.6f}")

        # early stop if we already hit the target bound
        if best_score <= 1.5025:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] early stop – target achieved")
            break

    # ----------------------------------------------------------------------
    # 5) Final evaluation and return
    # ----------------------------------------------------------------------
    # Clip to evaluator limits (0 ≤ x ≤ 1000) – the optimizer already respects
    # them but we apply a final safeguard.
    final = np.clip(best_seq, 0.0, 1000.0)

    # The official scorer accepts any list; use the provided evaluator for the
    # final (exact) score so that the harness can see the true value.
    final_score = evaluate_sequence(final.tolist())
    total_time = time.monotonic() - start_time
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in final]

"""
Hybrid optimiser for   2·n·max(a*a)   (sum(a)=1).

The algorithm combines three complementary ideas that were not all used
by the reference code:

1. **Analytic warm‑starts**
   – a symmetric exponential decay,
   – a power‑law decay a_i ∝ (dist(i)+1)^‑p,
   – a random‑phase spectrum (flat magnitude in the Fourier domain).
   All candidates are evaluated fast via an FFT‑based convolution and the
   best among them is kept.

2. **Adam on the log‑probabilities**  
   Optimise the simplex‑constrained vector by gradient descent on
   ``log(a)``.  The gradient of the exact max‑convolution is obtained by
   the active‑set routine ``_gradient_from_active``; the Adam moments give
   a robust, adaptive step size.

3. **Simulated‑annealing fine tune**  
   Starting from the Adam result we perform cheap directed mass‑transfer
   moves (source → target) accepted with a Metropolis rule.  The move
   size shrinks together with the temperature; occasional “jitter”
   (random scaling of a short window) helps to escape remaining local
   minima.

The whole search respects the 1000 s budget (a 5 s safety margin).  The
final sequence is clipped to the limits required by ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, sum(x)=1}."""
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        w = np.full_like(v, 1.0 / n)
        return w
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    if w.sum() == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w.sum()
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear a*a using FFT of length ``fft_len`` (returns length 2·len(a)‑1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max_k (a*a)[k] for the set ``active`` (indices attaining the max).
    For each active k we add a_{k-i} to the i‑th component.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast objective for a normalised vector (sum(a)=1):
    2·n·max(a*a).  ``fft_len`` must be at least 2·len(a).
    """
    n = a.size
    conv = _conv_fft(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    start = time.monotonic()
    # reserve a few seconds for finalisation / safety
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Problem size (a power of two makes the FFT neat)
    # --------------------------------------------------------------
    n = 4096
    fft_len = _next_pow2(2 * n)          # ≥ 2·n, enough for linear convolution

    # --------------------------------------------------------------
    # Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best_seq = None
    best_score = float("inf")

    def _maybe_update(a_vec: np.ndarray, sc: float) -> None:
        nonlocal best_seq, best_score
        if sc < best_score - 1e-12:
            best_score = sc
            best_seq = a_vec.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new global best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Analytic warm‑starts
    # --------------------------------------------------------------

    # 1a) Symmetric exponential decay (mass concentrates at the ends)
    for lam in np.logspace(-5, -0.3, num=30):
        if time.monotonic() >= deadline:
            break
        idx = np.arange(n)
        a = np.exp(-lam * np.minimum(idx, n - 1 - idx))
        a /= a.sum()
        _maybe_update(a, _score(a, fft_len))

    # 1b) Power‑law decay a_i ∝ (dist+1)^‑p   (p=0 is uniform)
    dist = np.minimum(np.arange(n), n - 1 - np.arange(n)) + 1.0
    for p in np.linspace(0.0, 2.0, num=21):
        if time.monotonic() >= deadline:
            break
        a = dist ** (-p)
        a /= a.sum()
        _maybe_update(a, _score(a, fft_len))

    # 1c) Random flat‑magnitude spectrum (random phases)
    for _ in range(30):
        if time.monotonic() >= deadline:
            break
        phase = rng.uniform(0.0, 2.0 * np.pi, size=fft_len // 2 + 1)
        G = np.exp(1j * phase)
        a = np.fft.irfft(G, n=fft_len)[:n]
        a = np.clip(a, 0.0, None)
        total = a.sum()
        if total == 0.0:
            continue
        a /= total
        _maybe_update(a, _score(a, fft_len))

    # 1d) Uniform (simple baseline)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _maybe_update(uniform, _score(uniform, fft_len))

    # 1e) If a previous construction is available, seed from it
    if "height_sequence_1" in globals():

        if prev.size != n:

            else:
                prev = prev[:n]
        prev = np.clip(prev, 0.0, None)
        if prev.sum() > 0.0:
            prev /= prev.sum()
            _maybe_update(prev, _score(prev, fft_len))

    # --------------------------------------------------------------
    # 2) Adam optimisation on the log‑probabilities
    # --------------------------------------------------------------
    # Use the best candidate found so far as the start point
    if best_seq is None:
        a = uniform.copy()
    else:
        a = best_seq.copy()

    log_a = np.log(a + 1e-15)          # avoid log(0)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8

    # Allocate roughly 70 % of the remaining budget to Adam
    adam_deadline = start + 0.70 * (deadline - start)

    it = 0
    while time.monotonic() < adam_deadline:
        # current active index
        conv = _conv_fft(a, fft_len)
        k_max = int(conv.argmax())
        grad = _gradient_from_active(a, np.array([k_max]))
        # gradient w.r.t. log‑a (a = exp(log_a))
        g_log = grad * a

        it += 1
        # Adam updates
        m = beta1 * m + (1.0 - beta1) * g_log
        v = beta2 * v + (1.0 - beta2) * (g_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = 0.6 / np.sqrt(it)          # diminishing learning rate
        log_a -= lr * m_hat / (np.sqrt(v_hat) + eps)

        a = np.exp(log_a)
        a /= a.sum()                    # simplex projection (exact)

        # fast objective value
        cur_score = _score(a, fft_len)
        _maybe_update(a, cur_score)

        if it % 2000 == 0:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Adam iter={it}, best={best_score:.6f}")

        # early exit if the target is already met
        if best_score <= 1.5025:
            print("[early stop] target reached during Adam phase")
            break

    # --------------------------------------------------------------
    # 3) Simulated‑annealing fine tuning
    # --------------------------------------------------------------
    # Start from the best vector obtained so far
    if best_seq is None:
        current = uniform.copy()
        current_score = _score(current, fft_len)
    else:
        current = best_seq.copy()
        current_score = best_score

    # Temperature schedule
    T = 0.05 * current_score          # initial temperature
    cooling = 0.9998

    it_local = 0
    while time.monotonic() < deadline:
        # --- pick source (mass‑rich) and target (mass‑poor) indices ---
        src = int(rng.choice(np.arange(n), p=current))          # probability ∝ current mass

        low_mask = current < np.percentile(current, 20)
        low_idxs = np.where(low_mask)[0]
        if low_idxs.size:
            tgt = int(rng.choice(low_idxs))
        else:
            tgt = int(rng.integers(0, n))

        if src == tgt:
            continue

        # step size shrinks with the temperature
        max_frac = min(0.20, T / current_score + 0.001)
        frac = rng.uniform(0.001, max_frac)
        delta = current[src] * frac
        if delta <= 0.0:
            continue

        cand = current.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = _project_onto_simplex(cand)          # keep on simplex

        cand_score = _score(cand, fft_len)

        # Metropolis acceptance
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - current_score) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] new best (anneal) = {best_score:.6f}")

        # temperature decay
        T *= cooling
        it_local += 1

        # occasional jitter to shake the solution
        if it_local % 4000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))

            jitter = current.copy()
            jitter[start_idx:start_idx + width] *= factor
            jitter = _project_onto_simplex(jitter)
            jitter_score = _score(jitter, fft_len)
            if jitter_score < best_score:
                best_score = jitter_score
                best_seq = jitter.copy()
                current = jitter
                current_score = jitter_score
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] new best (jitter) = {best_score:.6f}")

        # early stop if we have reached the published target
        if best_score <= 1.5025:
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] early stop – target achieved")
            break

    # ------------------------------------------------------------------
    # 5) Final evaluation and return
    # ------------------------------------------------------------------
    final = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(final.tolist())
    total = time.monotonic() - start
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in final]
