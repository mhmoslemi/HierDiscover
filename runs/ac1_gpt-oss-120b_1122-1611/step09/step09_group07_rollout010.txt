def generate_exponential(length, lam, reverse=False):
    idx = np.arange(length, dtype=np.float64)
    if reverse:
        idx = length - 1 - idx
    a = np.exp(-lam * idx)
    a_sum = a.sum()
    if a_sum == 0:

    else:
        a = a / a_sum
    return a

"""
Propose a low‑value step‑function.

The algorithm works in three independent stages:

1️⃣  **Parametric sweep** – cheap families (exponential, power‑law,
    mixed exponential + uniform) are generated for the chosen length
    (default ≤ 4096).  Each candidate is evaluated with an FFT‑based
    exact convolution and the best is kept.

2️⃣  **Projected Adam fine‑tuning** – starting from the best candidate,
    we minimise a smooth approximation of the max‑convolution
    (soft‑max temperature α) while keeping the vector on the simplex
    (Euclidean projection).  Adam moments are used, an occasional
    log‑normal block perturbation helps escape plateaux and a simple
    restart after 20 s of stagnation is applied.

3️⃣  **Local combinatorial search** – the current best vector is further
    refined by random swaps and by moving a tiny amount of mass from one
    coordinate to another (mass‑transfer moves).  An almost‑zero
    temperature permits occasional uphill steps for escaping shallow
    local minima.

The whole routine keeps a strict time budget (≈ 1000 s, 5 s safety
margin) and prints progress information.  The final sequence is clipped
to the limits imposed by ``evaluate_sequence`` before being returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


# ----------------------------------------------------------------------
# Parameterised families of candidates
# ----------------------------------------------------------------------


def generate_exponential(length: int, lam: float, reverse: bool = False) -> np.ndarray:
    """a_i ∝ exp(-lam·i).  If reverse=True the decay runs from the right edge."""
    idx = np.arange(length, dtype=np.float64)
    if reverse:
        idx = length - 1 - idx
    a = np.exp(-lam * idx)
    total = a.sum()
    if total == 0.0:
        a = np.full(length, 1.0 / length)
    else:
        a /= total
    return a


def generate_powerlaw(length: int, exponent: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-exponent).  exponent=0 yields the uniform vector."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    a = idx ** (-exponent)
    total = a.sum()
    if total == 0.0:
        a = np.full(length, 1.0 / length)
    else:
        a /= total
    return a


def generate_mixed_exponential_uniform(length: int, lam: float, weight: float) -> np.ndarray:
    """Convex mix of an exponential decay and the uniform distribution."""
    exp_part = np.exp(-lam * np.arange(length, dtype=np.float64))
    uni_part = np.ones(length, dtype=np.float64)
    a = weight * exp_part + (1.0 - weight) * uni_part
    total = a.sum()
    if total == 0.0:
        a = np.full(length, 1.0 / length)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Exact objective (no rounding, no clipping)
# ----------------------------------------------------------------------


def compute_J(vec: np.ndarray, n: int, conv_len: int, fft_len_self: int) -> float:
    """
    Exact value of the objective 2·n·max(a*a) for a vector that already
    satisfies sum(a)=1 and a_i≥0.
    """
    fa = np.fft.rfft(vec, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The search combines a cheap
    parametric sweep, a projected‑Adam fine‑tuning phase and a final
    swap / mass‑transfer local search.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem dimensions
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)          # protect against negatives
            n = base.size
        except Exception:
            base = None
            n = 4096
    else:
        base = None
        n = 4096                     # default length

    # enforce a modest size (avoid >5000 for speed)
    n = max(512, min(n, 4096))

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for self‑convolution
    fft_len_grad = _next_pow2(3 * n - 2)               # for gradient convolution

    # --------------------------------------------------------------
    # 2) Helper to evaluate the true objective
    # --------------------------------------------------------------
    # (the helper is a thin wrapper around compute_J)
    def eval_J(vec: np.ndarray) -> float:
        return compute_J(vec, n, conv_len, fft_len_self)

    # --------------------------------------------------------------
    # 3) Initialise with the provided seed or uniform
    # --------------------------------------------------------------
    best_vec = None
    best_J = float('inf')
    if base is not None and base.size == n:
        a0 = base.copy()
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
        best_vec = a0
        best_J = eval_J(best_vec)
        print(f"[init] Using supplied sequence (len={n}), J={best_J:.6f}")
    else:
        best_vec = np.full(n, 1.0 / n)
        best_J = eval_J(best_vec)
        print(f"[init] No seed – starting from uniform (len={n}), J={best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Phase A – cheap parametric sweep
    # --------------------------------------------------------------
    # Number of evaluations we are willing to spend before the time runs out
    max_sweep_evals = 2500
    evals_done = 0

    # Exponential decays (both directions)
    for lam in np.linspace(0.0, 0.02, 101):
        if time.monotonic() > deadline or evals_done >= max_sweep_evals:
            break
        cand = generate_exponential(n, lam, reverse=False)
        Jcand = eval_J(cand)
        evals_done += 1
        if Jcand < best_J:
            best_J, best_vec = Jcand, cand.copy()
            print(f"[sweep] lam={lam:.5f} left exp → J={best_J:.6f}")
        cand_rev = generate_exponential(n, lam, reverse=True)
        Jcand = eval_J(cand_rev)
        evals_done += 1
        if Jcand < best_J:
            best_J, best_vec = Jcand, cand_rev.copy()
            print(f"[sweep] lam={lam:.5f} right exp → J={best_J:.6f}")

    # Power‑law family
    if time.monotonic() < deadline and evals_done < max_sweep_evals:
        for expnt in np.linspace(0.0, 5.0, 51):
            if time.monotonic() > deadline or evals_done >= max_sweep_evals:
                break
            cand = generate_powerlaw(n, expnt)
            Jcand = eval_J(cand)
            evals_done += 1
            if Jcand < best_J:
                best_J, best_vec = Jcand, cand.copy()
                print(f"[sweep] power‑law α={expnt:.3f} → J={best_J:.6f}")

    # Mixed exponential + uniform
    if time.monotonic() < deadline and evals_done < max_sweep_evals:
        for lam in np.linspace(0.0, 0.02, 11):
            for w in np.linspace(0.0, 1.0, 11):
                if time.monotonic() > deadline or evals_done >= max_sweep_evals:
                    break
                cand = generate_mixed_exponential_uniform(n, lam, w)
                Jcand = eval_J(cand)
                evals_done += 1
                if Jcand < best_J:
                    best_J, best_vec = Jcand, cand.copy()
                    print(f"[sweep] mixed lam={lam:.4f} w={w:.2f} → J={best_J:.6f}")

    print(f"[sweep] Completed – best J after sweep = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5) Phase B – projected Adam fine‑tuning
    # --------------------------------------------------------------
    # Allocate roughly 30 % of the remaining budget for Adam
    time_left = deadline - time.monotonic()
    if time_left > 10.0:
        adam_budget = time_left * 0.35
    else:
        adam_budget = time_left * 0.5
    adam_deadline = time.monotonic() + adam_budget

    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.monotonic()

    while time.monotonic() < adam_deadline:
        it += 1
        now = time.monotonic()
        frac = (now - (adam_deadline - adam_budget)) / adam_budget  # ∈[0,1]

        # Learning‑rate and temperature schedules (linear annealing)
        lr = 0.3 * (1.0 - frac) + 0.001 * frac
        alpha = 20.0 * (1.0 - frac) + 5000.0 * frac

        # ----- self‑convolution (exact) -----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

        # ----- soft‑max over convolution values -----
        cmax = conv.max()
        shifted = conv - cmax          # for numerical stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed max (linear conv a * w_rev) -----
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)

        start = conv_len - 1                     # = 2·n‑2
        grad = 2.0 * full[start:start + n]       # ∂/∂a of the soft‑max objective

        # ----- Adam update + projection onto the simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # occasional jitter to shake out of small plateaux
        if rng.random() < 0.005:
            a = _random_block_perturb(a, rng)

        cur_J = eval_J(a)
        if cur_J < best_J - 1e-12:
            best_J, best_vec = cur_J, a.copy()
            last_improve = now
            print(f"[adam] it={it:5d}, new best J={best_J:.6f}")

        # restart after stagnation
        if now - last_improve > 20.0:
            a = best_vec.copy()
            a = a * np.exp(rng.normal(0.0, 0.07, size=n))
            a = proj_simplex(a)
            last_improve = now
            print(f"[adam] restart (no progress >20 s), best J={best_J:.6f}")

    # --------------------------------------------------------------
    # 6) Phase C – permutation / mass‑transfer local search
    # --------------------------------------------------------------
    swap_deadline = deadline
    cur_vec = best_vec.copy()
    cur_J = best_J
    attempts = 0
    accepts = 0
    temp = 1e-4                      # tiny temperature for occasional uphill moves

    while time.monotonic() < swap_deadline:
        attempts += 1
        if rng.random() < 0.7:
            # ----- swap two positions -----
            i, j = rng.integers(0, n, size=2)
            cand = cur_vec.copy()
            cand[i], cand[j] = cand[j], cand[i]
        else:
            # ----- move a tiny amount of mass from i to j -----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps_move = min(cur_vec[i], rng.random() * 0.02)  # ≤2 % of source mass
            if eps_move <= 0.0:
                continue
            cand = cur_vec.copy()
            cand[i] -= eps_move
            cand[j] += eps_move

        cand_J = eval_J(cand)
        if cand_J < cur_J - 1e-12:
            cur_vec, cur_J = cand, cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best_J, best_vec = cand_J, cand.copy()
                print(f"[swap] new global best J={best_J:.6f} after {attempts} attempts")
        else:
            # occasional uphill acceptance (tiny temperature)
            if rng.random() < np.exp(-(cand_J - cur_J) / temp):
                cur_vec, cur_J = cand, cand_J
                accepts += 1

        # every so often apply a heavier perturbation to escape deep minima
        if attempts % 50000 == 0:
            cur_vec = _random_block_perturb(cur_vec, rng)

    print(f"[finished] swap attempts={attempts}, accepts={accepts}, best J={best_J:.6f}")

    # --------------------------------------------------------------
    # 7) Return the best sequence (clip to evaluation limits)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
