Simple subgradient descent optimizer for the step‑function problem.
Starts from a diverse set of seeds (uniform, edge‑heavy, cosine, random
Dirichlet), then runs a projected sub‑gradient method on the true
objective J=2·n·max(a*a)/(∑a)^2.

the simplex (scaled to the required L1‑norm) and a diminishing step‑size.
Occasional random jitter and restart on stagnation are applied.

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[:2*a.size - 1]

def proj_simplex(v):
    # Euclidean projection onto the probability simplex (sum=1)
    if v.ndim != 1:
        raise ValueError("vector must be 1‑D")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _project_scaled(v, S):
    # project onto simplex of sum S
    w = proj_simplex(v)
    return w * S

def _shift_vec(a, shift):
    if shift == 0:
        return a
    n = a.size
    if shift > 0:
        out = np.empty_like(a)
        out[:shift] = 0.0
        out[shift:] = a[:n-shift]
        return out
    else:

        out = np.empty_like(a)

"""
Sub‑gradient descent optimizer for the step‑function problem.

The objective to minimise is

    J(a) = 2·n·max(a*a) / (∑a)² ,

where “a*a’’ denotes the linear self‑convolution of the non‑negative vector
a (length n).  After a projection onto the simplex with L¹‑norm
S = √(2 n) the objective reduces to the maximum entry of the convolution,
so the algorithm works directly on the true objective (no surrogate).

* A diverse set of seeds (uniform, two‑edge blocks, cosine families,
  random Dirichlet draws and the previously best sequence, padded to the
  chosen length) is built.
* From each seed a projected sub‑gradient method is run.
  The sub‑gradient of max(a*a) is obtained by averaging the exact
  gradients of all active convolution entries.
* A diminishing step‑size (a₀/√t) and Euclidean projection onto the
  simplex of sum S keep the iterates feasible.
* Small multiplicative jitter and a restart mechanism avoid stagnation.
* After the budget for the sub‑gradient phase a short Adam‑based
  refinement on the smooth‑max (soft‑max) surrogate is performed.
* The best vector found is clipped to the limits imposed inside
  ``evaluate_sequence`` and returned as a plain list of floats.

The routine respects the global time budget (default 1000 s) and prints
progress information.  No closures, file‑IO or network access are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).  Duchi‑et‑al algorithm.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if np.any(cond):
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    else:
        theta = 0.0
    w = np.maximum(v - theta, 0.0)
    # safeguard against the pathological all‑zero case
    if w.sum() == 0.0:
        w = np.full_like(v, 1.0 / n)
    else:
        w = w / w.sum()
    return w


def _project_scaled(v: np.ndarray, S: float) -> np.ndarray:
    """Project onto the simplex of sum =S (non‑negative, ∑=S)."""
    return proj_simplex(v) * S


def _shift_vec(a: np.ndarray, shift: int) -> np.ndarray:
    """Return a shifted copy of ``a`` with zero fill (no wrap‑around)."""
    n = a.size
    if shift == 0:
        return a
    if shift > 0:
        out = np.empty_like(a)
        out[:shift] = 0.0
        out[shift:] = a[: n - shift]
        return out
    # shift < 0
    s = -shift
    out = np.empty_like(a)
    out[-s:] = 0.0
    out[: n - s] = a[s:]
    return out


def _subgrad_max_conv(a: np.ndarray, conv: np.ndarray, n: int):
    """
    Sub‑gradient of the map a ↦ max(conv) where conv = a*a.
    Returns (grad, max_val).  The gradient is an average over all active
    (max‑attaining) shifts.
    """
    max_val = conv.max()
    # active set – use a tiny absolute/relative tolerance to catch ties
    tol = max(1e-12, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]
    shifts = active - (n - 1)                     # map index → shift
    grad = np.zeros_like(a)
    for s in shifts:
        grad += _shift_vec(a, s) + _shift_vec(a, -s)
    if len(shifts) > 0:
        grad /= len(shifts)
    return grad, max_val


def _objective_from_max(max_val: float, n: int, S: float) -> float:
    """Exact objective J = 2·n·max_val / S² (S = ∑a)."""
    return 2.0 * n * max_val / (S * S)


# ----------------------------------------------------------------------
# Seed generation
# ----------------------------------------------------------------------


def _generate_seeds(rng: np.random.Generator, n: int, S: float):
    """Create a list of diverse starting vectors (all have ∑=S)."""
    seeds = []

    # uniform
    seeds.append(np.full(n, S / n))

    # two edge blocks, sweep size and taper
    for frac in np.linspace(0.15, 0.45, 9):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 5):
            v = np.zeros(n, dtype=np.float64)
            if L > 1:
                idx = np.arange(L)
                v[idx] = 1.0 - gamma * (idx / (L - 1))
                v[-L:] = v[:L][::-1]
            else:
                v[0] = 1.0
                v[-1] = 1.0
            v = _project_scaled(v, S)
            seeds.append(v)

    # symmetric cosine family
    for r in np.linspace(0.0, 0.9, 7):
        for pw in np.linspace(0.4, 2.5, 5):
            theta = np.pi * np.arange(n) / (n - 1)
            v = (1.0 + r * np.cos(theta)) ** pw
            v = _project_scaled(v, S)
            seeds.append(v)

    # random Dirichlet draws (moderately concentrated)
    for _ in range(30):
        alpha = rng.uniform(0.04, 0.6)
        v = rng.dirichlet(np.full(n, alpha))
        v = v * S
        seeds.append(v)

    # previous best sequence (padded / truncated to length n)
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size < n:
                tmp = np.zeros(n, dtype=np.float64)
                tmp[: prev.size] = prev
                prev = tmp
            elif prev.size > n:
                prev = prev[:n]
            prev = _project_scaled(prev, S)
            seeds.append(prev)
        except Exception:
            pass

    return seeds


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns a list of heights (float) ready for evaluation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Problem size – we always work with the maximal cheap size (≤4096)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = int(len(height_sequence_1))
        except Exception:
            n0 = 0
    else:
        n0 = 0
    n = max(256, min(4096, max(n0, 4096)))   # end up with 4096 when possible
    S = np.sqrt(2.0 * n)                     # required L¹‑norm

    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 0) Build a modest pool of seeds and keep the best few
    # ------------------------------------------------------------------
    all_seeds = _generate_seeds(rng, n, S)

    seed_records = []
    for v in all_seeds:
        conv = _self_conv(v, fft_len)
        J = _objective_from_max(conv.max(), n, S)
        seed_records.append((J, v.copy()))
    seed_records.sort(key=lambda x: x[0])
    top_k = 8
    top_seeds = [p for _, p in seed_records[:top_k]]
    best_J, best_seq = seed_records[0]

    print(f"[init] n={n}, target L1={S:.3f}, best initial J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 1) Projected sub‑gradient descent from each seed
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    descent_budget = remaining * 0.80
    per_seed = descent_budget / max(1, len(top_seeds))

    for seed_idx, start_vec in enumerate(top_seeds):
        if time.monotonic() >= deadline:
            break
        seed_deadline = min(time.monotonic() + per_seed, deadline)

        a = start_vec.copy()
        it = 0
        last_improve = time.monotonic()
        a0 = 0.8 * S                     # initial step‑size magnitude

        while time.monotonic() < seed_deadline:
            it += 1
            # --- compute current convolution and sub‑gradient ---
            conv = _self_conv(a, fft_len)
            grad, max_val = _subgrad_max_conv(a, conv, n)

            # --- diminishing stepsize (a₀/√t) ---
            lr = a0 / np.sqrt(it)

            # --- gradient step + projection onto simplex of sum S ---
            a = a - lr * grad
            a = _project_scaled(a, S)

            # --- evaluate exact objective after the step ---
            conv_new = _self_conv(a, fft_len)
            J_new = _objective_from_max(conv_new.max(), n, S)

            if J_new < best_J - 1e-12:
                best_J = J_new
                best_seq = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - (deadline - budget_s + 5.0)
                print(f"[seed {seed_idx}] new best J={best_J:.6f} "
                      f"after {it} iters (t={elapsed:.1f}s)")

            # --- occasional jitter to escape flat regions ---
            if rng.random() < 0.015:
                block = rng.integers(1, max(2, n // 30))
                start_idx = rng.integers(0, n - block + 1)
                factor = np.exp(rng.normal(0.0, 0.06))
                a_pert = a.copy()
                a_pert[start_idx:start_idx + block] *= factor
                a_pert = _project_scaled(a_pert, S)
                a = a_pert

            # --- restart if no improvement for a while ---
            if time.monotonic() - last_improve > 18.0:
                a = best_seq.copy()
                noise = np.exp(rng.normal(0.0, 0.04, size=n))
                a = a * noise
                a = _project_scaled(a, S)
                last_improve = time.monotonic()
                print(f"[seed {seed_idx}] restart (stagnation) – best J={best_J:.6f}")

    # ------------------------------------------------------------------
    # 2) Short Adam refinement on the smooth‑max surrogate (if time left)
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 5.0:
        a = best_seq.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr_start, lr_end = 0.4, 0.001
        alpha_start, alpha_end = 6.0, 2500.0
        start = time.monotonic()
        it = 0
        while time.monotonic() < deadline:
            it += 1
            now = time.monotonic()
            frac = (now - start) / remaining
            lr = lr_start * (0.5 ** frac) + lr_end * (1 - 0.5 ** frac)
            alpha = alpha_start * (1.0 - frac) + alpha_end * frac

            # --- smooth‑max (soft‑max) weights ---
            conv = _self_conv(a, fft_len)
            cmax = conv.max()
            shifted = conv - cmax
            w = np.exp(alpha * shifted)
            w = w / w.sum()

            # --- gradient of the surrogate (see original code) ---
            w_rev = w[::-1]
            fa = np.fft.rfft(a, n=fft_len)
            fw = np.fft.rfft(w_rev, n=fft_len)
            full = np.fft.irfft(fa * fw, n=fft_len)
            start_idx = w_rev.size - 1
            grad = 2.0 * full[start_idx:start_idx + n]

            # --- Adam update + projection ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_scaled(a, S)

            # --- true objective check ---
            conv_check = _self_conv(a, fft_len)
            J_check = _objective_from_max(conv_check.max(), n, S)
            if J_check < best_J - 1e-12:
                best_J = J_check
                best_seq = a.copy()
                print(f"[adam] new best J={best_J:.6f} after {it} iters")

    # ------------------------------------------------------------------
    # 3) Return the best vector (clipped to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
```

This implementation respects the time budget, explores a richer set of initial shapes, uses a genuine sub‑gradient method on the true objective (which typically yields tighter progress than the soft‑max surrogate), and adds a final Adam polishing phase.  It has been crafted to improve upon the previous best bound of 1.513 → target 1.5025.
