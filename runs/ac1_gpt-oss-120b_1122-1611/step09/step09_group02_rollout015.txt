def _initial_candidates_custom(n,rng):
    cand = []
    # Uniform
    cand.append(np.full(n,1.0/n))
    # Exponential with many r values
    for r in np.linspace(0.5,0.99,10):
        v = r ** np.arange(n,dtype=np.float64)
        v /= v.sum()
        cand.append(v)
    # Reverse exponential (increase)
    for r in np.linspace(0.5,0.99,5):
        v = r ** np.arange(n-1,-1,-1,dtype=np.float64)
        v /= v.sum()
        cand.append(v)
    # Gaussian
    x = np.arange(n)
    for sigma in [n/10, n/6, n/4]:
        mu = n/2
        v = np.exp(-0.5*((x - mu)/sigma)**2)
        v[ v < 1e-12 ] = 0.0
        v /= v.sum()
        cand.append(v)
    # Linear (descending, ascending)
    v = np.linspace(1,0,n)
    v = np.maximum(v,0)
    v /= v.sum()
    cand.append(v)

    v = np.linspace(0,1,n)
    v /= v.sum()
    cand.append(v)

    # Random Dirichlet draws with varied concentration
    for alpha in [0.1,0.3,0.5,0.7,1.0,2.0]:

    return cand

and returns the best vector found.  The implementation avoids any
closures, uses only NumPy, and works for any length (the previous best
construction is taken into account if it exists in the global namespace).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.10))      # a bit stronger than before
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _project_simplex(out)
    return out


def _conv_max(v: np.ndarray, fft_len: int, conv_len: int, n: int) -> float:
    """Exact max‑convolution value max(v * v) (∑v = 1 is assumed)."""
    fa = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The routine builds a small, well‑chosen
    portfolio of analytic candidates (exponential decays with a finely
    tuned decay factor), then spends the remaining time on cheap random
    Dirichlet draws and tiny block perturbations of the current best.
    It respects the time budget (5 s safety margin) and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
        except Exception:
            n = 4096
    else:
        n = 4096
    # keep n modest (the original construction used ≈3800)
    n = max(512, min(5000, n))

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2) Helper to evaluate the exact objective (sum = 1 assumed)
    # ------------------------------------------------------------------
    def compute_J(v: np.ndarray) -> float:
        """Return 2·n·max(v∗v) – the exact value of evaluate_sequence for sum(v)=1."""
        return 2.0 * n * _conv_max(v, fft_len, conv_len, n)

    # ------------------------------------------------------------------
    # 3) Initialise a diverse portfolio of deterministic candidates
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")

    def _register(v: np.ndarray, label: str = ""):
        nonlocal best_vec, best_score
        v = _project_simplex(v)          # guarantee ∑ = 1, v ≥ 0
        score = compute_J(v)
        if score < best_score:
            best_score = score
            best_vec = v.copy()
            print(f"[{time.monotonic() - start:6.2f}s] {label} – new best J = {best_score:.6f}")

    # uniform
    _register(np.full(n, 1.0 / n), "uniform")

    # linear decreasing / increasing
    lin_dec = np.arange(n, 0, -1, dtype=np.float64)
    _register(lin_dec, "linear‑dec")
    _register(lin_dec[::-1], "linear‑inc")

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        _register(rng.dirichlet(np.ones(n) * 0.3), "dirichlet0.3")

    # the previously discovered construction, if any
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            prev = np.interp(x_new, x_old, prev)
        _register(prev, "previous")

    # ------------------------------------------------------------------
    # 4) Scan a fine grid of exponential decays (theoretical optimum)
    # ------------------------------------------------------------------
    # Theory suggests r_opt = n**(-1/n)  (equivalently λ = ln n)
    # We explore λ = -n·log(r) in a neighbourhood of ln n.
    lam_opt = np.log(n)
    lam_low = max(0.1, lam_opt * 0.3)   # avoid λ = 0 (uniform)
    lam_high = lam_opt * 3.0 + 2.0      # generous upper bound
    n_grid = 300                       # enough points, still cheap

    for lam in np.linspace(lam_low, lam_high, n_grid):
        r = np.exp(-lam / n)                     # r = e^{-λ/n}
        # decreasing exponential
        v = r ** np.arange(n, dtype=np.float64)
        _register(v, f"exp‑dec λ={lam:.3f}")
        # increasing exponential (mirror)
        v_rev = r ** np.arange(n - 1, -1, -1, dtype=np.float64)
        _register(v_rev, f"exp‑inc λ={lam:.3f}")

    # ------------------------------------------------------------------
    # 5) Random Dirichlet draws & tiny perturbations (the bulk of time)
    # ------------------------------------------------------------------
    # Allocate the remaining budget to cheap stochastic exploration.
    while time.monotonic() < deadline:
        # --- Dirichlet draw with random concentration ---
        alpha = rng.uniform(0.05, 5.0)               # from very spiky to very flat
        v = rng.dirichlet(np.full(n, alpha))
        _register(v, "dirichlet‑rand")

        # --- Occasionally jitter the current best ---
        if rng.random() < 0.03:
            v = _random_block_perturb(best_vec, rng)
            _register(v, "block‑perturb")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
