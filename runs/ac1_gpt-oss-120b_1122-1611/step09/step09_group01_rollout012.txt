def compute_J(a, n, fft_len_self, conv_len):
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    max_b = conv.max()

```

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – Use the length of the previously found sequence
   (``height_sequence_1``) if it exists, otherwise 4096.  The length is
   bounded to the interval [512,4096] to keep FFTs cheap.
2. **Rich initial portfolio** – Generate a small set of deterministic
   candidates (uniform, exponential, Gaussian, classic windows,
   Tukey, DPSS if SciPy is available) together with a few random
   Dirichlet draws and the previous best.  All vectors are projected onto
   the probability simplex (non‑negative and sum = 1) and the exact objective
   ``J = 2·n·max(a*a)`` is evaluated via FFT.  The best of these is kept.
3. **Deterministic sub‑gradient descent** – Repeatedly
   * compute the self‑convolution,
   * identify the active index/indices attaining the maximum,
   * form the (averaged) sub‑gradient of the ``max`` functional,
   * take a projected gradient step with a simple 1/√t learning‑rate decay,
   * keep the best vector seen.
   The gradient is obtained efficiently by another FFT‐convolution.
4. **Local polishing** – A cheap pairwise “mass‑transfer’’ hill‑climber
   (move a tiny amount of probability mass from one index to another)
   is run for a few seconds to squeeze out any remaining gain.
5. **Fallback** – If none of the new candidates beats the previous best
   (available through the global ``height_sequence_1`` variable) the
   routine returns that sequence unchanged.

The whole routine respects the 1000 s wall‑clock budget (5 s safety
margin) and always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (all arguments are explicit – no closures)
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """Exact objective J = 2·n·max(a*a) for a vector whose sum = 1."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def compute_conv(a: np.ndarray,
                 n: int,
                 fft_len_self: int,
                 conv_len: int) -> np.ndarray:
    """Full self‑convolution a*a (first 2·n‑1 entries)."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


def subgradient(a: np.ndarray,
                active: np.ndarray,
                n: int,
                fft_len_grad: int,
                conv_len: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution functional.
    ``active`` is the array of indices where the convolution attains its maximum.
    The gradient equals 2·a convolved with the (averaged) weight mask.
    """
    if active.size == 0:
        return np.zeros_like(a)

    w = np.zeros(conv_len, dtype=np.float64)
    w[active] = 1.0 / active.size          # averaging over all active constraints
    w_rev = w[::-1]

    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start = w_rev.size - 1                # = 2·n‑2
    grad = 2.0 * full[start:start + n]
    return grad

# Main entry point

    cur_score = best_score
    hill_start = time.monotonic()
    while time.monotonic() < deadline and (time.monotonic() - hill_start) < 12.0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # move a tiny quantum of probability mass from i to j
        max_transfer = min(a[i], 0.006)          # keep steps modest
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = compute_J(cand, n, fft_len_self, conv_len)
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            best_vec = cand.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] hill‑climb improvement J={best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Fallback to the previous best if we failed to improve it
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev_vec = proj_simplex(np.clip(np.asarray(height_sequence_1, dtype=np.float64),
                                       0.0, None))
        prev_score = compute_J(prev_vec, n, fft_len_self, conv_len)
        if prev_score < best_score:
            best_vec = prev_vec
            best_score = prev_score
            print(f"fallback to previous best J={best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
