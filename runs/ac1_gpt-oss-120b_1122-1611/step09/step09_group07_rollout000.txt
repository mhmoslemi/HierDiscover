Propose a sequence of non‑negative heights minimizing ``evaluate_sequence``.

Algorithm
---------
1. **Candidate lengths** – Use the length of the provided global
   ``height_sequence_1`` (interpolated) and also try length 4096.
2. **Exponentiated‑gradient phase** – For each length run an
   exponentiated‑gradient descent on the simplex.
   * The objective ``max(a*a)`` is smoothed with a soft‑max (temperature
     ``alpha``) allowing a differentiable gradient.
   * Update rule: a ← a·exp(‑lr·∇)  followed by normalisation.
   * Learning‑rate ``lr`` decays from 0.5 → 0.001, the soft‑max temperature
     ``alpha`` climbs from 10 → 4000.
   * Small random block perturbations and restarts keep the search
     from stagnating.
3. **Local hill‑climbing phase** – With the remaining time, repeatedly
   move a random fraction of mass from one index to another (mass‑transfer
   move).  Accept improvements; occasionally accept uphill moves with a
   tiny temperature that decays to zero.
   * Every 10 000 attempts a heavier block‑re‑initialisation is performed.
4. **Best‑ever tracking** – The best vector over all lengths and all phases
   is kept.  If the target bound ``1.5025`` is reached the routine exits
   early.
5. **Time‑budget handling** – The total budget (default 1000 s) is split
   evenly among the candidate lengths; a 5‑second safety margin is kept.

The search stays on the probability simplex (sum = 1) so the evaluation
ratio equals ``2·n·max(a*a)``.  All values are finally clipped to the limits

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        fft_len_grad = _next_pow2(3 * n - 2)          # a * w_rev

        # --------------------------------------------------------------
        # Helper to compute the true objective  J = 2·n·max(a*a)
        # --------------------------------------------------------------
        def compute_J(vec: np.ndarray) -> float:
            fa = np.fft.rfft(vec, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
            return float(2.0 * n * conv.max())

        cur_J = compute_J(a)
        best_vec = a.copy()
        best_J = cur_J
        print(f"[cand {cand_idx + 1}/{n_cands}] n={n}, init J={cur_J:.6f}")

        # ------------------------------------------------------------------
        # Phase 1 – Exponentiated‑gradient descent on smoothed max‑convolution
        # ------------------------------------------------------------------
        # reserve ~60 % of the candidate budget for EG, the rest for hill‑climb
        eg_deadline = cand_deadline - (cand_deadline - cand_start) * 0.4
        it = 0
        last_improve = cand_start

        while time.monotonic() < eg_deadline:
            it += 1
            elapsed = time.monotonic() - cand_start
            frac = elapsed / (eg_deadline - cand_start)   # ∈[0,1]

            # schedules (linear)
            lr = 0.5 * (1.0 - frac) + 0.001 * frac
            alpha = 10.0 * (1.0 - frac) + 4000.0 * frac

            # ----- self‑convolution (full) -----
            fa = np.fft.rfft(a, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

            c_max = conv.max()
            # soft‑max weights on the convolution entries
            shifted = conv - c_max
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w[:] = 1.0 / w.size
            else:
                w /= w_sum

            # ----- gradient of the smoothed objective -----
            w_rev = w[::-1]
            fa_a = np.fft.rfft(a, n=fft_len_grad)
            fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
            full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
            start_idx = w_rev.size - 1
            grad = 2.0 * full[start_idx:start_idx + n]

            # ----- exponentiated‑gradient update -----
            # keep numbers bounded to avoid overflow/underflow
            exp_factor = np.exp(-lr * grad)
            a = a * exp_factor
            total = a.sum()
            if total <= 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= total

            # ----- evaluate -----
            cur_J = compute_J(a)
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = a.copy()
                last_improve = time.monotonic()
                print(f"[cand {cand_idx + 1}] EG iter {it} new best J={best_J:.6f}")
                if best_J <= target_J:
                    print("[early exit] target reached in EG")
                    best_overall_J = best_J
                    best_overall_vec = best_vec.copy()
                    return [float(x) for x in best_overall_vec]

            # occasional jitter to shake the state
            if rng.random() < 0.02:
                width = rng.integers(1, max(2, n // 10))
                start = rng.integers(0, n - width + 1)
                factor = np.exp(rng.normal(0.0, 0.4))
                a[start:start + width] *= factor
                a = np.clip(a, 0.0, None)
                a /= a.sum()

            # restart if stalled for >15 s
            if time.monotonic() - last_improve > 15.0:
                a = best_vec.copy()
                noise = np.exp(rng.normal(0.0, 0.1, size=n))
                a = a * noise
                a = np.clip(a, 0.0, None)
                a /= a.sum()
                last_improve = time.monotonic()
                print(f"[cand {cand_idx + 1}] EG restart after stagnation")

        # ------------------------------------------------------------------
        # Phase 2 – Simple hill‑climbing (mass‑transfer moves)
        # ------------------------------------------------------------------
        cur_vec = best_vec.copy()
        cur_J = best_J
        attempts = 0
        accepts = 0
        hc_start = time.monotonic()
        total_hc_time = cand_deadline - hc_start

        # temperature schedule for occasional uphill moves
        init_temp = 1e-3
        final_temp = 1e-7

        while time.monotonic() < cand_deadline:
            attempts += 1
            # pick source index with some mass
            i = rng.integers(0, n)
            if cur_vec[i] <= 0.0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue

            # move a random fraction (≤30 %) of the source mass to the target
            delta_frac = rng.random() * 0.30
            delta = cur_vec[i] * delta_frac
            if delta <= 0.0:
                continue

            # apply move
            cur_vec[i] -= delta
            cur_vec[j] += delta

            cand_J = compute_J(cur_vec)

            # linearly decaying temperature
            elapsed_hc = time.monotonic() - hc_start
            temp = init_temp * (1.0 - elapsed_hc / total_hc_time) + final_temp * (elapsed_hc / total_hc_time)

            if cand_J < cur_J - 1e-12 or rng.random() < np.exp(-(cand_J - cur_J) / temp):
                # accept
                cur_J = cand_J
                accepts += 1
                if cand_J < best_J - 1e-12:
                    best_J = cand_J
                    best_vec = cur_vec.copy()
                    print(f"[cand {cand_idx + 1}] HC new best J={best_J:.6f} (attempt {attempts})")
                    if best_J <= target_J:
                        print("[early exit] target reached in HC")
                        best_overall_J = best_J
                        best_overall_vec = best_vec.copy()
                        return [float(x) for x in best_overall_vec]
            else:
                # revert move
                cur_vec[i] += delta
                cur_vec[j] -= delta

            # every 10 000 attempts perform a heavier perturbation
            if attempts % 10000 == 0:
                width = rng.integers(1, max(2, n // 10))
                start = rng.integers(0, n - width + 1)
                block = rng.dirichlet(np.ones(width))
                cur_vec[start:start + width] = block
                cur_vec = np.clip(cur_vec, 0.0, None)
                cur_vec /= cur_vec.sum()
                cur_J = compute_J(cur_vec)

        print(f"[cand {cand_idx + 1}] finished – best J={best_J:.6f}")
        # update overall best
        if best_J < best_overall_J:
            best_overall_J = best_J
            best_overall_vec = best_vec.copy()

        # stop early if we are out of time
        if time.monotonic() > deadline - 2.0:
            break

    # ----------------------------------------------------------------------
    # Fallback random sampling if nothing was found (unlikely)
    # ----------------------------------------------------------------------
    if best_overall_vec is None:
        n = 1024
        best_overall_vec = np.full(n, 1.0 / n)

    # Clip to the limits enforced by ``evaluate_sequence``
    best_overall_vec = np.clip(best_overall_vec, 0.0, 1000.0)

    print(f"Overall best J = {best_overall_J:.6f}")
    return [float(x) for x in best_overall_vec]
