"""
Gradient‑guided simulated annealing for the “step‑function” problem.

The objective is   J(a) = 2·n·max(a*a) / (∑a)² .
Because the problem is scale‑invariant we keep the vector on the
probability simplex (∑a = 1) and minimise   J = 2·n·max(conv(a,a)).

Algorithm
---------
0. Choose a moderate problem size n (≈ length of the best known
   construction, bounded between 256 and 8192).  All FFT lengths are
   powers of two for speed.
1. Initialise a small portfolio of diverse probability vectors
   (uniform, geometric, linear ramps, a few Dirichlet draws and, if
   available, the previous best solution).  Keep the best of them.
2. Main loop (until the time budget expires):
   • Compute the exact convolution a*a via FFT.
   • Evaluate the true objective J = 2·n·max(conv) and store the best.
   • Form a *smooth* approximation of the max using a soft‑max
     with temperature 1/α:   w_i ∝ exp(α·(conv_i – max(conv))).
   • The gradient of this smooth objective is
        g = 2·corr(w, a)  (cross‑correlation of w with a),
     which we compute with a second FFT.
   • Perform an exponentiated‑gradient update
        a ← a · exp( –η·g ),   then renormalise onto the simplex.
   • The step size η linearly cools from 0.5 to 0.001;
     the smoothing parameter α linearly heats from 5 to 100.
   • Occasionally (probability 0.001) perturb the whole vector with
     log‑normal noise to improve exploration.
   • If no improvement has been seen for 30 s, restart from the best
     vector with a small Gaussian jitter.
3. Return the best vector found (clipped to the limits of
   `evaluate_sequence`).

The method combines a principled gradient direction for the
non‑smooth objective with the robustness of stochastic restarts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (full length 2·n‑1) using an FFT of length `fft_len`.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def gradient_smooth(a: np.ndarray,
                    conv: np.ndarray,
                    alpha: float,
                    fft_len2: int) -> np.ndarray:
    """
    Gradient of the smooth (soft‑max) approximation of max(conv).

    Parameters
    ----------
    a      : current probability vector (length n, sum=1)
    conv   : linear convolution a*a (length 2·n‑1)
    alpha  : soft‑max sharpness parameter (larger ⇒ closer to true max)
    fft_len2 : FFT length for the second convolution (≥ 3·n‑2)

    Returns
    -------
    grad   : ndarray of length n (the sub‑gradient of the smooth objective)
    """
    n = a.shape[0]
    # soft‑max weights (stable by subtracting the maximum)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))

    # cross‑correlation of w with a  (equivalently conv(w, a[::-1]))
    rev_a = a[::-1]
    W = np.fft.rfft(w, n=fft_len2)
    RevA = np.fft.rfft(rev_a, n=fft_len2)
    full = np.fft.irfft(W * RevA, n=fft_len2)

    # slice the valid part: indices n‑1 … n‑1+n‑1
    start = n - 1
    grad = 2.0 * full[start:start + n]
    return grad

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The optimiser uses a smooth‑max
    surrogate together with exponentiated‑gradient updates and
    stochastic restarts.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n_prev = len(globals()["height_sequence_1"])
            n = max(256, min(n_prev, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048                     # default length

    conv_len = 2 * n - 1
    fft_len1 = _next_pow2(conv_len)                # for a*a
    fft_len2 = _next_pow2(conv_len + n - 1)        # for gradient (w * a_rev)

    # ------------------------------------------------------------------
    # 1. Initialise a few diverse candidate vectors (simplex‑projected)
    # ------------------------------------------------------------------
    init_cands: list[np.ndarray] = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.9, 0.95, 0.975, 0.99):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # a handful of Dirichlet draws (moderately concentrated)
    for _ in range(8):
        init_cands.append(rng.dirichlet(np.ones(n) * 0.4))

    # two‑spike (ends) – just for completeness
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    v /= v.sum()
    init_cands.append(v)

    # add the previously known construction if it matches the size
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        old = np.clip(old, 0.0, None)
        s = old.sum()
        if s > 0.0:
            # simple nearest‑size handling: truncate or pad with zeros
            if old.shape[0] >= n:
                old = old[:n]
            else:
                old = np.pad(old, (0, n - old.shape[0]), constant_values=0.0)
            old /= old.sum()
            init_cands.append(old)

    # ------------------------------------------------------------------
    # 2. Pick the best initial candidate
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = compute_full_convolution(cand, fft_len1)
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (gradient‑guided simulated annealing)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a safety margin

    # schedule parameters
    step_start = 0.5            # initial step size for exponentiated update
    step_end = 0.001            # final step size
    alpha_start = 5.0           # start of soft‑max sharpness
    alpha_end = 100.0           # final sharpness (close to true max)

    cur_vec = best_vec.copy()
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)

        # linearly anneal step and heat the soft‑max
        step = step_start * (1.0 - frac) + step_end * frac
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # ----- evaluate current vector -----
        conv = compute_full_convolution(cur_vec, fft_len1)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- gradient of the smooth max -----
        grad = gradient_smooth(cur_vec, conv, alpha, fft_len2)

        # ----- exponentiated‑gradient update -----
        # protect against overflow/underflow
        cur_vec = cur_vec * np.exp(-step * grad)
        # avoid exact zeros that could stall the gradient
        cur_vec = np.clip(cur_vec, 1e-15, None)
        cur_vec /= cur_vec.sum()

        # ----- occasional global perturbation (log‑normal noise) -----
        if rng.random() < 0.001:
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            cur_vec = cur_vec * noise
            cur_vec = proj_simplex(cur_vec)

        # ----- stagnation handling -----
        if now - last_improve > 30.0:
            # jitter around the best solution and continue
            cur_vec = best_vec * np.exp(rng.normal(0.0, 0.05, size=n))
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        iteration += 1

    # 4. Return the best vector (clip to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / s

    return [float(x) for x in best_vec]
