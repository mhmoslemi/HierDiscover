    a = np.clip(a, 0, 1000)
    n = len(a)

    conv_len = 2*n -1

  n≈4000.
* Simulated annealing with a simple “move ε of mass from i to j”.  The step
  size ε decays with time, and the temperature is lowered linearly.
* Random restarts (Dirichlet draws) are performed when the search stagnates.
* The best candidate encountered (including the optional ``height_sequence_1``
  start) is returned as a plain ``list[float]``.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper for fast evaluation (exactly the same as evaluate_sequence)
# ----------------------------------------------------------------------


def _fast_obj(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a)) / (sum(a)²) with the same clipping rules as
    ``evaluate_sequence``.  The implementation uses FFTs and therefore is fast.
    """
    # clip to the limits of evaluate_sequence
    a = np.clip(a, 0.0, 1000.0)
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf

    n = a.size
    conv_len = 2 * n - 1
    # next power of two for efficient FFT
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = rfft(a, n=fft_len)
    conv = irfft(fa * fa, n=fft_len)[:conv_len]
    max_b = conv.max()
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock budget
    (5 s safety margin) and always returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Choose problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # keep n in a comfortable range for FFTs
            n = max(256, min(4096, n))
        except Exception:
            n = 2048
    else:
        n = 2048                     # a reasonable default

    # --------------------------------------------------------------
    # 2) Initialise a population of seed vectors (all L1‑normalised)
    # --------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        """Project onto the probability simplex (sum = 1)."""
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s == 0.0:
            return np.full_like(v, 1.0 / v.size)
        return v / s

    # start from the previous best if available
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = _normalise(base)
    else:
        base = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = base.copy()
    best_score = _fast_obj(best_vec)
    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated‑annealing / mass‑transfer hill‑climbing
    # --------------------------------------------------------------
    # parameters
    max_temp = 0.5                     # initial temperature (objective units)
    min_temp = 1e-6                    # final temperature
    max_step_frac = 0.1                # max fraction of mass moved in a single step
    stagnation_limit = 30.0            # seconds without improvement → restart

    last_improve = time.monotonic()
    cur_vec = best_vec.copy()
    cur_score = best_score

    while time.monotonic() < deadline:
        now = time.monotonic()
        # linear temperature schedule
        frac = (now - start_time) / (deadline - start_time)
        temp = max_temp * (1.0 - frac) + min_temp * frac

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        if cur_vec[i] <= 0.0:
            continue

        # amount of mass to transfer (scaled by current value)
        delta = rng.uniform(0.0, max_step_frac) * cur_vec[i]
        if delta <= 0.0:
            continue

        cand = cur_vec.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = _fast_obj(cand)

        # Metropolis acceptance
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / temp):
            cur_vec = cand
            cur_score = cand_score

            # track global best
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_vec = cand.copy()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best J = {best_score:.6f}")

        # restart if we have been stuck for a while
        if now - last_improve > stagnation_limit:
            # random Dirichlet restart (moderately concentrated)
            cur_vec = _normalise(rng.dirichlet(np.ones(n) * 0.3))
            cur_score = _fast_obj(cur_vec)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart – current J = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4) Final polishing – cheap pairwise mass transfer
    # --------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_score = best_score
    hill_start = time.monotonic()
    while time.monotonic() < deadline and (time.monotonic() - hill_start) < 10.0:
        i, j = rng.integers(0, n, size=2)
        if i == j or cur_vec[i] == 0.0:
            continue
        # very small deterministic step
        delta = min(cur_vec[i], 0.001)
        cand = cur_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = _fast_obj(cand)
        if cand_score < cur_score - 1e-12:
            cur_vec = cand
            cur_score = cand_score
            best_vec = cand.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] pairwise improvement J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
