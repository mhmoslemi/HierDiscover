def subgradient(a):
    conv = np.convolve(a, a)
    max_val = conv.max()
    # tolerance maybe 1e-12 relative
    idxs = np.where(np.abs(conv - max_val) < 1e-12)[0]
    grad = np.zeros_like(a)
    n = len(a)
    for k in idxs:
        # compute contributions
        start = max(0, k-(n-1))
        end = min(k, n-1)
        # for j from start to end:
        # gradient_j += 2 * a[k - j]
        # can vectorize: grad[start:end+1] += 2 * a[k - np.arange(start, end+1)]
        # Use slicing.
        js = np.arange(start, end+1)
        grad[js] += 2 * a[k - js]
    return grad

This is O(|idxs| * n). With |idxs| usually 1 or a few, O(n) per iteration. Acceptable.

"""
Sub‑gradient descent / simulated annealing for the max‑convolution problem.

The algorithm works on the simplex (non‑negative entries that sum to 1) and
optimises directly the true objective
    J(a) = 2·n·max(conv(a,a)),
using the exact sub‑gradient of the max‑convolution (instead of a soft‑max
surrogate).  A small portfolio of diverse seeds (uniform, exponentials,
sine, Dirichlet draws and the previous best) is explored, each seed being
run for a short sub‑gradient phase followed by a cheap simulated‑annealing
random walk.  The best vector seen is returned (clipped to the limits of
``evaluate_sequence``).

The routine respects a 1000 s wall‑clock budget (5 s safety margin) and
always returns the best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block, then project."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Diverse starting points (already normalised to sum = 1)."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays (coarse sweep)
    for r in [0.999, 0.997, 0.995, 0.99, 0.975, 0.95, 0.9]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # sine‑shaped (always non‑negative)
    v = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    v = np.clip(v, 0.0, None)
    v /= v.sum()
    cand.append(v)

    # half‑cosine (Fejér‑like)
    v = 1.0 - np.cos(np.pi * np.arange(n, dtype=np.float64) / (n - 1))
    v = np.clip(v, 0.0, None)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # several Dirichlet draws (moderately concentrated)
    for _ in range(8):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    return cand


# ----------------------------------------------------------------------
# Objective and sub‑gradient helpers
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """True objective J = 2·n·max(a*a) for a normalised vector."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def compute_conv_fft(a: np.ndarray,
                     n: int,
                     fft_len_self: int,
                     conv_len: int) -> np.ndarray:
    """Full self‑convolution (length 2·n‑1) using FFT."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


def subgradient_from_conv(a: np.ndarray,
                          conv: np.ndarray,
                          max_val: float,
                          n: int) -> np.ndarray:
    """
    Exact sub‑gradient of J at ``a``.
    For every index k attaining the maximal convolution value,
    the contribution to the gradient is
        g_j ← g_j + 2·a_{k‑j}
    (where a_{·}=0 out of range).  The sum over all maximal k's is returned.
    """
    # tolerance to collect all active indices (avoid missing due to round‑off)
    tol = max(1e-12, max_val * 1e-12)
    idxs = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for k in idxs:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # vector of row indices that actually appear in the sum
        js = np.arange(i_start, i_end + 1, dtype=int)
        # a[k - js] is always in‑range by construction
        grad[js] += 2.0 * a[k - js]
    return grad

# ----------------------------------------------------------------------

            elapsed = last_improve - start_time

    # --------------------------------------------------------------
    a = best_vec.copy()

        a_candidate[i] -= delta
        a_candidate[j] += delta

        cand_score = compute_J(a_candidate, n, fft_len_self, conv_len)

        # accept if improvement or with Metropolis probability
        if cand_score < cur_score or rng.random() < np.exp(-(cand_score - cur_score) / max(temperature, 1e-12)):
            a = a_candidate
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_vec = a_candidate.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] SA new best J = {best_score:.6f}")

        # cool down
        temperature *= cooling_rate

        # occasional larger perturbation
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

    # --------------------------------------------------------------
    # 6) Final cheap pairwise hill‑climbing (10 s max)
    # --------------------------------------------------------------
    a = best_vec.copy()
    cur_score = best_score
    hill_start = time.monotonic()
    while time.monotonic() < deadline and (time.monotonic() - hill_start) < 10.0:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        max_transfer = min(a[i], 0.005)     # modest move
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand_score = compute_J(cand, n, fft_len_self, conv_len)
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            best_vec = cand.copy()
            best_score = cand_score
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Pairwise improvement J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 7) Return the best sequence (clipped to the limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
