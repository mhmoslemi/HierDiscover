* Approximate the “max” in the objective 2·n·max(a∗a) by a
  smooth log‑sum‑exp (soft‑max).  The smoothing parameter α is ramped
  from a small value (smooth landscape) to a larger one (sharper
  approximation) during the run.
* The exact gradient of the smooth‑max is a weighted correlation
  between the soft‑max weights w (a distribution over the convolution
  indices) and the current sequence a:
        ∇_a ϕ(a) = 2·corr(w , a) .
  The correlation is obtained efficiently with a single FFT.
* A multiplicative (exponentiated‑gradient) update
        a_i ← a_i·exp(−η·∇_i)
  guarantees non‑negativity; afterwards the vector is re‑normalised.
* Random jitter is injected every few thousand iterations to escape
  plateaus, and a cheap local “mass‑transfer” move is tried periodically.
* The routine runs until the time budget is exhausted (≈1000 s) and
  returns the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2·len(a)−1 (real‑valued).
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    # tiny negative values can appear because of round‑off – clip them
    return np.maximum(conv, 0.0)


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear correlation r[k] = Σ_i w[i+k]·a[i]   (valid for k = 0…len(a)−1)
    using the same ``fft_len`` that is large enough for a linear
    operation (≥ len(w)+len(a)−1).  Returns the first len(a) entries.
    """
    fw = np.fft.rfft(w, n=fft_len)
    fa = np.fft.rfft(a, n=fft_len)
    corr_full = np.fft.irfft(fw * np.conj(fa), n=fft_len)
    return corr_full[: a.size]  # only the needed part


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` under the given time budget.
    Returns the best sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (leave a few seconds safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose problem size -------------------------------------------------
    # Larger n gives more flexibility but also larger FFTs.
    # Empirically n = 8192 works well on the typical execution platform.
    # ------------------------------------------------------------------
    n = 8192

    # ------------------------------------------------------------------
    # 2) Initialise the probability vector a (sum = 1)
    #    – use the previously discovered construction if available,
    #      otherwise start from a Dirichlet draw.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        init = np.clip(init, 0.0, None)
        s = init.sum()
        if s > 0.0:
            init = init / s
        else:
            init = None
    else:
        init = None

    if init is not None:
        # Pad / truncate to the chosen length and jitter a little
        a = np.zeros(n, dtype=np.float64)
        length = min(init.size, n)
        a[:length] = init[:length]
        # add a modest log‑normal jitter to break symmetry
        jitter = np.exp(rng.normal(0.0, 0.1, size=n))
        a = a * jitter
        a_sum = a.sum()
        a = a / (a_sum + 1e-15)
    else:
        # Dirichlet with a small concentration → many near‑zero entries
        a = rng.dirichlet(np.full(n, 0.1))

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT length (enough for both convolution and correlation)
    # ------------------------------------------------------------------
    fft_len = _next_pow2(3 * n)          # certainly ≥ (2n‑1) + n‑1

    # ------------------------------------------------------------------
    # 4) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    conv = _conv_fft(a, fft_len)
    best_score = float(2.0 * n * conv.max())
    best_seq = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Optimisation loop
    # ------------------------------------------------------------------
    it = 0
    # schedule for the soft‑max smoothing
    ramp_iters = 50000          # iterations over which α grows
    min_alpha = 0.5
    max_alpha = 15.0

    # Adam‑style moments for the log‑space (optional – gives a little inertia)
    beta1 = 0.9
    beta2 = 0.999
    m_log = np.zeros_like(a)
    v_log = np.zeros_like(a)

    while time.monotonic() < deadline:
        # ---- a) current convolution and score ---------------------------------
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        conv = np.maximum(conv, 0.0)               # numerical safety
        max_b = conv.max()
        score = 2.0 * n * max_b

        if score < best_score - 1e-12:
            best_score = score
            best_seq = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}  iter={it}")

        # ---- b) smooth‑max weights --------------------------------------------
        if it < ramp_iters:
            alpha = min_alpha + (max_alpha - min_alpha) * (it / ramp_iters)
        else:
            alpha = max_alpha

        scaled = alpha * conv
        # stable soft‑max (subtract the max)
        shift = scaled.max()
        w = np.exp(scaled - shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- c) gradient of the smooth‑max (linear correlation) ---------------
        # correlation w ⊛ a (valid part) – first n entries
        grad = 2.0 * _corr_fft(w, a, fft_len)      # shape (n,)

        # ---- d) exponentiated‑gradient update on log‑probabilities -------------
        # Adam‑style moments on the log‑space gradient
        g_log = -grad                               # we minimise → opposite sign
        m_log = beta1 * m_log + (1.0 - beta1) * g_log
        v_log = beta2 * v_log + (1.0 - beta2) * (g_log * g_log)
        it1 = it + 1
        m_hat = m_log / (1.0 - beta1 ** it1)
        v_hat = v_log / (1.0 - beta2 ** it1)

        # step size (decreases like 1/√t)
        eta = 0.12 / np.sqrt(it1)

        # multiplicative (exponentiated) update performed in log‑space
        log_a = np.log(a + 1e-15) + eta * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # ---- e) occasional random jitter (helps escape flat plateaus) ---------
        if it % 5000 == 0 and it > 0:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a *= jitter
            a /= a.sum()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] jitter applied (iter={it})")

        # ---- f) cheap local mass‑transfer move every few hundred iterations --
        if it % 800 == 0 and it > 0:
            src = np.argmax(grad)                     # moving mass away from
            tgt = np.argmin(grad)                     # the most “problematic’’ spots
            if a[src] > 1e-12 and src != tgt:
                delta = a[src] * 0.02                   # 2 % of the source mass
                cand = a.copy()
                cand[src] -= delta
                cand[tgt] += delta
                cand = np.clip(cand, 0.0, None)
                cand /= cand.sum()
                cand_conv = _conv_fft(cand, fft_len)
                cand_score = 2.0 * n * cand_conv.max()
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_seq = cand.copy()
                    a = cand
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] local move improved to {best_score:.6f}")

        # ---- g) termination checks --------------------------------------------
        if best_score <= 1.5025:
            print("[early stop] target reached")
            break

        it += 1
        # safety: break if we are within a couple of seconds of the deadline
        if time.monotonic() + 0.1 >= deadline:
            break

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clip to evaluator limits)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    total = time.monotonic() - start_time
    print(f"Finished in {total:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_seq]
```
