monotonically decreasing sequences.
--------------------------------------------------------------------
Theory (brief)
-------------
For a non‑negative sequence a of length n with sum 1 the evaluation
function is

    J(a) = 2·n·max_k (a * a)[k]          (the sum‑normalisation cancels)

If a_i ∝ (n‑i)^p (p ≥ 0) then the dominant convolution entry is the
central one (k = n‑1) and

    J(p) = 2·(p+1)²·B(p+1, p+1)          where B is the beta function.

For p = 0 (uniform) J = 2.  Already p = 1 gives J = 4/3 ≈ 1.333,
p = 2 gives J = 0.6 and larger p drives J down rapidly.
Hence a simple parametric family already beats the target bound

---------------
* Choose a problem size n (reuse the length of an existing construction

* Scan a small set of exponents p (including integer values up to 6).
* For each p generate a_i = (n‑i)^p, normalise to sum 1 and compute the

import time
import numpy as np
from scipy.special import beta  # for reference only; not needed for J computation


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _compute_J(vec: np.ndarray) -> float:
    """
    Compute the exact evaluation value J = 2·n·max(conv) for a vector that
    already sums to 1.  Uses FFT for speed.
    """
    n = vec.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())

        print(f"p = {p:<4} → J = {J:.6f}")

    # --------------------------------------------------------------
    # 2) Optional tiny random perturbation to avoid exact ties
    # --------------------------------------------------------------
    if best_vec is not None:
        # Apply a log‑normal multiplicative jitter with very small variance
        jitter = np.exp(rng.normal(0.0, 0.02, size=best_vec.shape))
        perturbed = best_vec * jitter
        perturbed = np.clip(perturbed, 0.0, None)
        perturbed_sum = perturbed.sum()
        if perturbed_sum > 0.0:
            perturbed /= perturbed_sum
            J_pert = _compute_J(perturbed)
            if J_pert < best_J:
                best_J = J_pert
                best_vec = perturbed
                print(f"After jitter → J = {J_pert:.6f}")

    # --------------------------------------------------------------
    # 3) Final sanity checks & return
    # --------------------------------------------------------------
    if best_vec is None:
        # Fallback: a simple uniform distribution (should never happen)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_J = _compute_J(best_vec)

    # Clip to the limits imposed by evaluate_sequence (harmless here)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    print(f"Selected exponent p = {best_p}, final J = {best_J:.6f}, length = {n}")
    return [float(x) for x in best_vec]
