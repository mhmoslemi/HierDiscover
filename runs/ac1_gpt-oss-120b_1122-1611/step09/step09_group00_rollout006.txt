def propose_candidate(...):
    set seed = seed if given else random

    n = 4096
    a = uniform
    if height_sequence_1 available:
      a = that and normalize

    # compute conv and score
    conv = _conv_fft(a, fft_len)

    best = a.copy()
    best_score = cur_score

    # iterative annealing
    start = time.monotonic()
    deadline = start + budget_s - 5

    # schedule parameters
    total_iters = 0
    T0 = 0.01 * cur_score  # initial temperature
    Tf = 0.000001 * cur_score
    # map iteration to temperature linearly in log-space.

    while time.monotonic() < deadline:
        # compute grad from max conv index
        idx_max = int(np.argmax(conv))
        grad = _gradient_from_active(a, np.array([idx_max]))

        # source and target selection
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))

        # propose delta
        # choose random factor r uniform in log-space range
        r = np.exp(np.random.uniform(np.log(0.001), np.log(0.1)))  # up to 10%
        delta = a[src] * r

        # candidate
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        # project onto simplex (though sum stays =1) but ensure nonneg
        # due to floating rounding, clip negative to zero and renorm
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum > 0:
            cand /= cand_sum

        # compute score
        conv_cand = _conv_fft(cand, fft_len)
        cand_score = 2.0 * n * conv_cand.max()

        # acceptance
        iter_num = total_iters
        # temperature schedule
        t_frac = (iter_num / 200000)  # assume maybe 200k iterations? but adapt
        # linear decay: temp = T0 * (1 - t_frac) + Tf * t_frac
        temp = T0 * (1 - t_frac) + Tf * t_frac
        # if cand_score < cur_score accept
        accept = False
        if cand_score < cur_score - 1e-15:
            accept = True
        else:
            # accept with probability exp(-(cand_score - cur_score)/temp)
            # avoid overflow
            if temp > 0:
                prob = np.exp(-(cand_score - cur_score) / temp)
                accept = np.random.rand() < prob

        if accept:
            a = cand
            conv = conv_cand
            cur_score = cand_score

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best = cand.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (anneal) = {best_score:.6f} (iter={total_iters})")
            if best_score <= 1.5025:
                break

        total_iters += 1

        # periodic random jitter or restart to escape stagnation
        if total_iters % 5000 == 0:

            block_len = int(np.random.randint(1, max(2, n // 50)))
            start_idx = np.random.randint(0, n - block_len + 1)
            factor = np.exp(np.random.normal(0, 0.02))
            a_jitter = a.copy()
            a_jitter[start_idx:start_idx+block_len] *= factor
            a_jitter = np.clip(a_jitter, 0.0, None)
            a_jitter /= a_jitter.sum()
            conv_j = _conv_fft(a_jitter, fft_len)
            score_j = 2.0 * n * conv_j.max()
            if score_j < best_score - 1e-12:
                best_score, best = score_j, a_jitter.copy()
                print(f"[{time.monotonic() - start:6.2f}s] new best (jitter) = {best_score:.6f}")
            # optionally accept jitter if improves or with small prob.

        # Occasionally replace a with best + small random perturbation to diversify
        if total_iters % 20000 == 0:
            # bounce
            a = best.copy()
            # add random noise:
            noise = np.random.normal(scale=0.001, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            conv = _conv_fft(a, fft_len)
            cur_score = 2.0 * n * conv.max()

    # after loop, final clamp and evaluate full bound using evaluate_sequence for safety
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return best.tolist()

"""
Propose a step‑function (non‑negative sequence) that minimises
evaluate_sequence() using a simulated‑annealing style search.

The method starts from the best known construction (if available) and
iteratively improves it by directed mass‑transfer moves that try to
reduce the current maximal auto‑correlation.  The move is accepted
unconditionally if it improves the score, otherwise with a probability
that decays linearly with the remaining time (temperature schedule).
Occasional random jitter and global reshuffles help to escape local
minima.  All updates keep the vector on the probability simplex
(∑a_i = 1), which makes the objective scale‑invariant.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the set of active
    convolution indices ``active`` (those attaining the current maximum
    or a selected top‑k set).  For each active index k, the contribution
    to ∂max/∂a_i is a_{k-i} wherever the index is defined.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    # Adapted from Duchi et al. (2008)
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(v, 1.0 / len(v))
    else:
        w /= w_sum
    return w


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Return the value 2 * n * max(conv).  The sum of a is assumed = 1."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    ``evaluate_sequence`` under the given time budget.

    The algorithm:
      * initialise from the previous best (global ``height_sequence_1``) or
        a uniform distribution,
      * iteratively perform directed mass‑transfer moves guided by the
        gradient of the current largest convolution entries,
      * accept worsening moves with a temperature that decays linearly
        with time (simulated annealing),
      * occasionally apply random jitter or a global reshuffle to
        diversify the search.

    The best sequence found before the time limit is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise ``a`` and normalise to sum = 1
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a = a / s
        else:
            a = np.full_like(a, 1.0 / a.size)
    else:
        # default length – a power of two is convenient for FFT
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(3 * n)          # length sufficient for linear conv

    # --------------------------------------------------------------
    # 1) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    cur_conv = _conv_fft(a, fft_len)
    cur_score = _objective_from_conv(cur_conv, n)

    best = a.copy()
    best_score = cur_score
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Simulated‑annealing loop
    # --------------------------------------------------------------
    iteration = 0
    top_k = 5                     # how many large convolution entries are used
    T0 = best_score * 0.05        # initial temperature (5 % of the score)
    Tf = 1e-8 * best_score        # final temperature (practically zero)

    while time.monotonic() < deadline:
        # --- compute a gradient that focuses on the largest conv entries ---
        # Fetch the indices of the current top‑k values
        if top_k >= cur_conv.size:
            active_idx = np.arange(cur_conv.size)
        else:
            active_idx = np.argpartition(-cur_conv, top_k)[:top_k]

        grad = _gradient_from_active(a, active_idx)

        # source = index that most contributes to the current peaks,
        # target = index that contributes the least.
        src = int(np.argmax(grad))
        tgt = int(np.argmin(grad))

        # propose a directed mass transfer
        # draw a fraction in log‑uniform range [0.001, 0.1]
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.1)))
        delta = a[src] * frac

        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta

        # Numerical safety – enforce non‑negativity and renormalise
        # (the subtraction cannot go negative because delta ≤ a[src])
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum != 0.0:
            cand /= cand_sum

        # evaluate the candidate (fast convolution, same normalisation)
        cand_conv = _conv_fft(cand, fft_len)
        cand_score = _objective_from_conv(cand_conv, n)

        # ----------------------------------------------------------
        # Acceptance rule (simulated annealing)
        # ----------------------------------------------------------
        elapsed = time.monotonic() - start_time
        time_frac = (elapsed) / (deadline - start_time)   # in [0,1]
        temp = T0 * (1.0 - time_frac) + Tf * time_frac   # linear cooling

        accepted = False
        if cand_score < cur_score - 1e-15:
            accepted = True
        else:
            # probability of accepting a worse move
            if temp > 0.0:
                prob = np.exp(-(cand_score - cur_score) / temp)
                if rng.random() < prob:
                    accepted = True

        if accepted:
            a = cand
            cur_conv = cand_conv
            cur_score = cand_score

        # keep the globally best one
        if cand_score < best_score - 1e-12:
            best = cand.copy()
            best_score = cand_score
            print(f"[{elapsed:6.2f}s] new best (anneal) = {best_score:.6f} (iter={iteration})")
            if best_score <= 1.5025:
                print("[early stop] target reached")
                break

        # ----------------------------------------------------------
        # Occasional diversification moves
        # ----------------------------------------------------------
        # 1) random jitter on a small block (every 3000 iterations)
        if iteration % 3000 == 0 and iteration > 0:
            block_len = int(rng.integers(1, max(2, n // 60)))
            start_idx = int(rng.integers(0, n - block_len + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jitter = a.copy()
            a_jitter[start_idx:start_idx + block_len] *= factor
            a_jitter = np.clip(a_jitter, 0.0, None)
            a_jitter /= a_jitter.sum()
            jitter_conv = _conv_fft(a_jitter, fft_len)
            jitter_score = _objective_from_conv(jitter_conv, n)
            if jitter_score < best_score - 1e-12:
                best, best_score = a_jitter.copy(), jitter_score
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (jitter) = {best_score:.6f}")
            # Accept jitter as the current state if it is not much worse
            if jitter_score < cur_score + 1e-8:
                a, cur_conv, cur_score = a_jitter, jitter_conv, jitter_score

        # 2) global reshuffle with a tiny probability (exploratory)
        if rng.random() < 0.0005:
            perm = rng.permutation(n)
            a_perm = a[perm]
            # keep the simplex invariant (permutation doesn't change sum)
            perm_conv = _conv_fft(a_perm, fft_len)
            perm_score = _objective_from_conv(perm_conv, n)
            if perm_score < best_score - 1e-12:
                best, best_score = a_perm.copy(), perm_score
                print(f"[{time.monotonic() - start_time:6.2f}s] new best (shuffle) = {best_score:.6f}")
            if perm_score < cur_score:
                a, cur_conv, cur_score = a_perm, perm_conv, perm_score

        iteration += 1

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
