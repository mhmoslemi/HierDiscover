"""
Multi‑scale optimisation for the “step‑function” minimisation problem.

The routine works in three stages:

1. **Coarse scale (n≈1024)** – a quick sweep over a few simple families
   (exponential, power‑law, Dirichlet draws) followed by a short projected
   sub‑gradient descent that minimises the exact objective  
   J = 2·n·max(a∗a)   (the denominator of ``evaluate_sequence`` cancels out
   after normalising the sum to 1).

2. **Upscaling** – the best coarse‑scale vector is linearly interpolated to a
   larger size (default 4096).  This gives a good starting point for the final
   optimisation while keeping the computational cost low.

3. **Fine optimisation** – a longer projected sub‑gradient descent with a
   decreasing step‑size, occasional random restarts and a brief pair‑wise
   “mass‑transfer’’ hill‑climb.  The algorithm respects the 1000 s wall‑clock
   budget and always returns the best vector seen.

All vectors are projected onto the probability simplex (non‑negative,
unit‑sum) after every modification, guaranteeing that the final sequence is
valid for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (no closures)
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective J = 2·n·max(a∗a) for a vector a that already satisfies
    sum(a)=1.  ``fft_len`` must be a power of two ≥ 2·n‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * n - 1]
    return float(2.0 * n * conv.max()), conv


def grad_J_at_max(a: np.ndarray, n: int, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of J = 2·n·max(a∗a) at the current maximising shift ``k_max``.
    The gradient w.r.t. a_j is 4·n·a_{k_max-j} whenever 0 ≤ k_max‑j < n,
    otherwise 0.  This is the exact sub‑gradient of the piecewise‑linear
    objective (the denominator is constant because sum(a)=1).
    """
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(k_max, n - 1)
    # j runs from lo … hi inclusive
    js = np.arange(lo, hi + 1)
    idx = k_max - js
    grad[js] = 4.0 * n * a[idx]
    return grad


def upscale_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from the old length to ``new_n`` followed by a
    projection onto the simplex.
    """
    old_n = v.size
    if old_n == new_n:
        return v.copy()
    x_old = np.linspace(0.0, 1.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_n, endpoint=False)
    v_new = np.interp(x_new, x_old, v)
    return proj_simplex(v_new)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects a 1000 s wall‑clock budget
    (5 s safety margin) and always returns the best sequence encountered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # ------------------------------------------------------------------
    # 1) Choose problem sizes (coarse → fine)
    # ------------------------------------------------------------------
    # Coarse scale – small enough for rapid exploration
    n_coarse = 1024
    # Fine scale – the target size (bounded to keep FFT cheap)
    if "height_sequence_1" in globals():
        try:
            n_fine = min(max(len(globals()["height_sequence_1"]), 512), 4096)
        except Exception:
            n_fine = 4096
    else:
        n_fine = 4096

    # ------------------------------------------------------------------
    # 2) Helper to evaluate a candidate (sum is forced to 1)
    # ------------------------------------------------------------------
    def evaluate_and_store(v: np.ndarray, n: int, best_sofar: tuple[float, np.ndarray]):
        """Return (J, conv) and update the global best if improvement occurs."""
        J, conv = compute_J(v, n, _next_pow2(2 * n - 1))
        best_J, best_vec = best_sofar
        if J < best_J - 1e-12:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best (n={n}) J = {J:.6f}")
            best_J = J
            best_vec = v.copy()
        return J, conv, (best_J, best_vec)

    # ------------------------------------------------------------------
    # 3) ----------------------------------------------------------------
    #    Coarse‑scale global search (parameter sweeps + few random draws)
    # ------------------------------------------------------------------
    best_J_coarse = float("inf")
    best_vec_coarse = None

    # --- exponential family -------------------------------------------------
    for lam in np.linspace(0.0005, 0.0035, 20):
        if time.monotonic() >= deadline:
            break
        v = np.exp(-lam * np.arange(n_coarse, dtype=np.float64))
        v = proj_simplex(v)
        J, _, (best_J_coarse, best_vec_coarse) = evaluate_and_store(
            v, n_coarse, (best_J_coarse, best_vec_coarse)
        )

    # --- power‑law family ---------------------------------------------------
    for p in np.linspace(0.2, 2.0, 20):
        if time.monotonic() >= deadline:
            break
        v = 1.0 / (np.arange(1, n_coarse + 1, dtype=np.float64) ** p)
        v = proj_simplex(v)
        J, _, (best_J_coarse, best_vec_coarse) = evaluate_and_store(
            v, n_coarse, (best_J_coarse, best_vec_coarse)
        )

    # --- random Dirichlet draws --------------------------------------------
    for _ in range(200):
        if time.monotonic() >= deadline:
            break
        v = rng.dirichlet(np.ones(n_coarse) * 0.5)
        J, _, (best_J_coarse, best_vec_coarse) = evaluate_and_store(
            v, n_coarse, (best_J_coarse, best_vec_coarse)
        )

    # ------------------------------------------------------------------
    # 4) Coarse‑scale projected sub‑gradient descent
    # ------------------------------------------------------------------
    if best_vec_coarse is None:
        best_vec_coarse = np.full(n_coarse, 1.0 / n_coarse)
        best_J_coarse, _ = compute_J(best_vec_coarse, n_coarse,
                                     _next_pow2(2 * n_coarse - 1))

    a = best_vec_coarse.copy()
    it = 0
    init_step = 0.8 / n_coarse          # initial step size (will shrink)
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        it += 1
        J, conv = compute_J(a, n_coarse, _next_pow2(2 * n_coarse - 1))
        k_max = int(conv.argmax())

        # sub‑gradient at the active shift
        g = grad_J_at_max(a, n_coarse, k_max)

        # diminishing step size
        eta = init_step / np.sqrt(it)
        a_new = proj_simplex(a - eta * g)

        # evaluate the new point
        J_new, _ = compute_J(a_new, n_coarse, _next_pow2(2 * n_coarse - 1))
        if J_new < best_J_coarse - 1e-12:
            best_J_coarse = J_new
            best_vec_coarse = a_new.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Sub‑grad (coarse) new best J={best_J_coarse:.6f}")

        # stagnation handling – occasional random jitter
        if time.monotonic() - last_improve > 8.0:
            # small multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.05, size=n_coarse))
            a = proj_simplex(a * noise)
            last_improve = time.monotonic()
            continue

        a = a_new

    # ------------------------------------------------------------------
    # 5) Upscale to the fine resolution
    # ------------------------------------------------------------------
    a_fine = upscale_vector(best_vec_coarse, n_fine)

    # If a previous global best (height_sequence_1) exists, incorporate it:
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.sum() == 0.0:
            prev = np.full_like(prev, 1.0 / prev.size)
        else:
            prev = prev / prev.sum()
        # adapt length if necessary
        if prev.size != n_fine:
            prev = upscale_vector(prev, n_fine)
        # pick the better of the two as starting point
        J_prev, _ = compute_J(prev, n_fine, _next_pow2(2 * n_fine - 1))
        J_fine, _ = compute_J(a_fine, n_fine, _next_pow2(2 * n_fine - 1))
        if J_prev < J_fine:
            a_fine = prev.copy()
            best_J_fine = J_prev
        else:
            best_J_fine = J_fine
    else:
        best_J_fine, _ = compute_J(a_fine, n_fine,
                                   _next_pow2(2 * n_fine - 1))

    best_vec_fine = a_fine.copy()

    # ------------------------------------------------------------------
    # 6) Fine‑scale projected sub‑gradient descent (most of the remaining budget)
    # ------------------------------------------------------------------
    it = 0
    init_step = 0.5 / n_fine
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        it += 1
        J, conv = compute_J(a_fine, n_fine, _next_pow2(2 * n_fine - 1))
        k_max = int(conv.argmax())
        g = grad_J_at_max(a_fine, n_fine, k_max)

        eta = init_step / np.sqrt(it)
        a_new = proj_simplex(a_fine - eta * g)

        J_new, _ = compute_J(a_new, n_fine,
                             _next_pow2(2 * n_fine - 1))
        if J_new < best_J_fine - 1e-12:
            best_J_fine = J_new
            best_vec_fine = a_new.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Fine sub‑grad iter {it} new best J={best_J_fine:.6f}")

        # Stagnation → random restart around the best point
        if time.monotonic() - last_improve > 15.0:
            noise = np.exp(rng.normal(0.0, 0.07, size=n_fine))
            a_fine = proj_simplex(best_vec_fine * noise)
            last_improve = time.monotonic()
            continue

        a_fine = a_new

    # ------------------------------------------------------------------
    # 7) Final pairwise hill‑climbing (at most 10 s)
    # ------------------------------------------------------------------
    hill_deadline = min(deadline, time.monotonic() + 10.0)
    a = best_vec_fine.copy()
    cur_J = best_J_fine
    while time.monotonic() < hill_deadline:
        i, j = rng.integers(0, n_fine, size=2)
        if i == j:
            continue
        # move a small amount from i to j
        max_transfer = min(a[i], 0.006)   # modest step size
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        Jc, _ = compute_J(cand, n_fine,
                          _next_pow2(2 * n_fine - 1))
        if Jc < cur_J - 1e-12:
            a = cand
            cur_J = Jc
            best_vec_fine = cand.copy()
            best_J_fine = Jc
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Hill‑climb improvement J={best_J_fine:.6f}")

    # ------------------------------------------------------------------
    # 8) Return the best vector (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec_fine = np.clip(best_vec_fine, 0.0, 1000.0)
    return [float(x) for x in best_vec_fine]
