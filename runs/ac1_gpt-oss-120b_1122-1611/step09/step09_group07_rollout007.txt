"""
Simulated‑annealing search for a non‑negative step function that minimises
``evaluate_sequence``.
--------------------------------------------------------------------
Idea
----
*   The objective after normalising the sum (∑a_i = √(2n)) is simply the
    peak of the aperiodic self‑convolution:
        J = max_k (a * a)_k .
*   Starting from an “arcsine’’ shape – large near the ends and small in the
    middle – the convolution is already rather flat.  This gives a good
    initial bound (often < 1.6).
*   A custom Metropolis–Hastings scheme moves mass between two positions
    while keeping the total sum constant.  The convolution update is performed
    in O(n) time by using the exact analytic formula for the change of a*a
    under a transfer of δ mass from index j to index i.
*   Temperature and step‑size are annealed, occasional block‑re‑randomisation
    is performed and the best vector seen is stored.
*   The whole routine respects the 1000 s budget (5 s safety margin).

The function ``propose_candidate`` returns the best sequence found, clipped to
the limits of ``evaluate_sequence``.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _initial_arcsine_shape(n: int) -> np.ndarray:
    """
    Return a non‑negative arcsine‑like distribution on n points,
    normalised to sum √(2n).
    """
    # mid‑point rule on (0,1)
    x = (np.arange(n) + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x) + 1e-14)   # avoid divergence at exact 0/1
    # Clip extreme values to the limits of evaluate_sequence (1000) – they are
    # already far smaller for the allowed n, but we keep the step for safety.
    a = np.clip(a, 0.0, 1000.0)
    # Normalise the sum to √(2n) (the scale does not affect the objective,
    # but it makes the later formula J = max(b) exact).
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / a.sum())
    return a


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Compute the aperiodic full self‑convolution a*a using FFT.
    Result length is 2·len(a)‑1.
    """
    n = a.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(a, n=fft_len)
    b = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return b


def _transfer_update(
    a: np.ndarray,
    b: np.ndarray,
    i: int,
    j: int,
    delta: float,
) -> None:
    """
    Apply the exact update of the full self‑convolution when δ mass is moved
    from index j to index i (i ≠ j).  ``a`` and ``b`` are modified in place.
    The formula derived from (a+δe_i‑δe_j)*(a+δe_i‑δe_j) is used.
    """
    n = a.size
    conv_len = b.size

    # linear part: 2·δ·(a_{k‑i} – a_{k‑j})
    factor = 2.0 * delta
    # slice updates – NumPy broadcasting does the multiplication internally
    b[i : i + n] += factor * a
    b[j : j + n] -= factor * a

    # quadratic correction terms
    idx2i = 2 * i
    idx2j = 2 * j
    idxij = i + j
    d2 = delta * delta
    if idx2i < conv_len:
        b[idx2i] += d2
    if idx2j < conv_len:
        b[idx2j] += d2
    if idxij < conv_len:
        b[idxij] -= 2.0 * d2

    # finally change the vector a itself
    a[i] += delta
    a[j] -= delta

        _transfer_update(a, b, i, j, delta)
        new_max = float(b.max())

        # ------------------------------------------------------------------
        # Metropolis acceptance test
        # ------------------------------------------------------------------
        accept = False
        if new_max < cur_max - 1e-12:
            accept = True
        else:
            # uphill moves are allowed with a temperature‑scaled probability
            prob = np.exp(-(new_max - cur_max) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                best_iter = it
                print("[{:.1f}s] New best peak = {:.6f} (iter {})".format(
                    elapsed, best_max, it))
        else:
            # revert both a and b
            a[i] = a_i_old
            a[j] = a_j_old
            b = b_old

        # ------------------------------------------------------------------
        # Periodic global re‑randomisation to avoid stagnation
        # ------------------------------------------------------------------
        if it % RE_RANDOMISE_EVERY == 0:
            # pick a random contiguous block and reshuffle its mass
            blk_len = rng.integers(1, max(2, n // 15))
            start = rng.integers(0, n - blk_len + 1)
            block_sum = a[start : start + blk_len].sum()
            # new random weights (Dirichlet with all parameters = 1)
            new_weights = rng.random(blk_len)
            new_weights = new_weights / new_weights.sum() * block_sum
            a[start : start + blk_len] = new_weights
            # recompute the whole convolution from scratch (removes numerical drift)
            b = _full_convolution(a)
            cur_max = float(b.max())
            print("[{:.1f}s] Re‑randomised block (iter {}), peak = {:.6f}".format(
                time.monotonic() - start_time, it, cur_max))

    # --------------------------------------------------------------
    # Return the best vector seen (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    # renormalise exactly to √(2·len) – scale does not affect the score
    target_sum = np.sqrt(2.0 * best_a.size)
    best_a = best_a * (target_sum / best_a.sum())
    print("[final] Best peak = {:.6f} after {} iterations".format(best_max, it))
    return [float(v) for v in best_a]
