"""
Propose a non‑negative step‑function (sequence of heights) that minimises
evaluate_sequence().  The optimiser works in three stages:

1.  Adam optimiser on a smooth log‑sum‑exp surrogate of the peak of the
    self‑convolution.  The variables are the logits of a soft‑max,
    guaranteeing a simplex constraint (non‑negative, sum = 1) without
    explicit projection.

2.  Simulated‑annealing style random mass‑transfer moves that are able
    to escape the basin of attraction of the Adam solution while still
    preserving the simplex constraint.

3.  A cheap directed local‑search that moves mass from the index with
    the largest gradient (i.e. most responsible for the current peak)
    toward indices with the smallest gradient.  Very small steps are
    accepted only when they improve the exact objective.

The routine respects the supplied time‑budget (default 1000 s) and
returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a * a using an FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.size
    if fft_len is None:
        fft_len = _next_pow2(2 * n)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv


def _gradient_logsumexp(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Gradient of   L(a)=1/α·log∑_k exp(α·(a*a)[k])   w.r.t. a.

    Returns a vector of length len(a).  The derivation uses the fact that
    ∂L/∂a_i = 2 * Σ_k w_k·a_{k-i},
    where w_k = softmax(α·conv_k).
    """
    n = a.size
    # softmax weights (numerically stable)
    max_c = conv.max()
    w = np.exp(alpha * (conv - max_c))
    w /= w.sum()

    # convolution of reversed a with weights
    a_rev = a[::-1]
    L = _next_pow2(3 * n)                     # length ≥ 3n‑2
    a_rev_fft = np.fft.rfft(a_rev, n=L)
    w_fft = np.fft.rfft(w, n=L)
    rev_conv = np.fft.irfft(a_rev_fft * w_fft, n=L)[: 3 * n - 1]

    # slice gives the needed n‑dimensional gradient (no factor 2 yet)
    grad = 2.0 * rev_conv[n - 1 : 2 * n - 1]
    return grad


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    Used only for occasional jitter steps where a soft‑max re‑normalisation
    is not appropriate.
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a sequence that minimises evaluate_sequence().
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Determine problem size and initialise globals
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 4096                # default length – a power of two is convenient

    fft_len_conv = _next_pow2(2 * n)   # for a*a
    best_a = None
    best_max = np.inf                 # maximal entry of a*a
    start = time.monotonic()
    # leave a small safety margin for the harness
    deadline = start + max(0.0, float(budget_s) - 2.0)

    # --------------------------------------------------------------
    # 1) Helper to evaluate the true objective (max of the convolution)
    # --------------------------------------------------------------
    def _objective(a_vec: np.ndarray) -> float:
        conv = _conv_fft(a_vec, fft_len_conv)
        return conv.max()

    # --------------------------------------------------------------
    # 2) Run a few restarts of Adam on the log‑sum‑exp surrogate
    # --------------------------------------------------------------
    restart_modes = ["uniform", "sqrt", "random"]
    for mode in restart_modes:
        if time.monotonic() > deadline:
            break

        # ---- initialise a -------------------------------------------------
        if mode == "uniform":
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        elif mode == "sqrt":
            i = np.arange(1, n + 1, dtype=np.float64)
            a_raw = 1.0 / np.sqrt(i)
            a0 = a_raw / a_raw.sum()
        else:  # random Dirichlet, favouring sparsity a bit
            a0 = rng.dirichlet(np.full(n, 0.5))

        # logits for a soft‑max representation (guarantees the simplex)
        logits = np.log(a0 + 1e-15)

        # Adam state
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        lr = 0.06

        # schedule: (alpha, number of iterations)
        schedule = [(20.0, 250), (60.0, 300), (200.0, 400), (500.0, 400)]
        it_counter = 0

        for alpha, steps in schedule:
            for _ in range(steps):
                if time.monotonic() > deadline:
                    break
                it_counter += 1

                # ---- current a (soft‑max) ------------------------------------
                a = np.exp(logits)
                a /= a.sum()                               # stays on simplex

                # ---- convolution and surrogate weights ----------------------
                conv = _conv_fft(a, fft_len_conv)          # length 2n‑1
                max_conv = conv.max()
                # surrogate gradient (log‑sum‑exp)
                grad_a = _gradient_logsumexp(a, conv, alpha)

                # ---- convert gradient to logits (soft‑max chain rule) -------
                dot = np.dot(a, grad_a)
                grad_logits = a * (grad_a - dot)

                # ---- Adam update -------------------------------------------
                m = beta1 * m + (1.0 - beta1) * grad_logits
                v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)

                m_hat = m / (1.0 - beta1 ** it_counter)
                v_hat = v / (1.0 - beta2 ** it_counter)

                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---- keep best solution --------------------------------------
                if max_conv < best_max - 1e-12:
                    best_max = max_conv
                    best_a = a.copy()
                    elapsed = time.monotonic() - start
                    print(f"[{elapsed:6.2f}s] restart {mode} new best max={best_max:.8f}")

                # early exit if we have already reached the target region
                if best_max * 2 * n <= 1.5025:
                    break
            if best_max * 2 * n <= 1.5025 or time.monotonic() > deadline:
                break

    # --------------------------------------------------------------
    # 3) Simulated‑annealing style random mass‑transfer search
    # --------------------------------------------------------------
    if best_a is None:
        # fall back to a uniform start
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_max = _objective(best_a)

    cur_a = best_a.copy()
    cur_max = best_max

    # simple exponential temperature schedule
    T0 = 1e-3
    Tf = 1e-6
    total_sa_steps = 20000
    decay = (Tf / T0) ** (1.0 / total_sa_steps)

    T = T0
    for step in range(total_sa_steps):
        if time.monotonic() > deadline:
            break

        # propose a tiny mass transfer between two distinct indices
        src = rng.integers(0, n)
        tgt = rng.integers(0, n - 1)
        if tgt >= src:
            tgt += 1
        if cur_a[src] == 0.0:
            continue
        frac = rng.uniform(0.001, 0.05)          # 0.1 %–5 % of src mass
        delta = cur_a[src] * frac

        cand = cur_a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()                       # restore simplex

        cand_max = _objective(cand)

        # acceptance rule (down‑hill always, up‑hill with probability)
        if cand_max < cur_max - 1e-12:
            cur_a, cur_max = cand, cand_max
            if cand_max < best_max - 1e-12:
                best_a, best_max = cand.copy(), cand_max
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] SA improvement max={best_max:.8f} step={step}")
        else:
            prob = math.exp(-(cand_max - cur_max) / T)
            if rng.random() < prob:
                cur_a, cur_max = cand, cand_max

        T *= decay  # geometric cooling

        # early stop if target reached
        if best_max * 2 * n <= 1.5025:
            break

    # --------------------------------------------------------------
    # 4) Deterministic local refinement (tiny directed moves)
    # --------------------------------------------------------------
    a = best_a.copy()
    while time.monotonic() < deadline:
        conv = _conv_fft(a, fft_len_conv)
        cur_max = conv.max()
        idx_max = int(np.argmax(conv))

        # gradient for the single active index gives a map of “responsibility”
        # for that peak; we move mass from the biggest contributor to a weak one.
        # The helper below is a lightweight version of _gradient_logsumexp
        # with a single active index.
        def _grad_one_active(arr: np.ndarray, k: int) -> np.ndarray:
            n_ = arr.size
            rev = arr[::-1]
            L_ = _next_pow2(3 * n_)
            rev_fft = np.fft.rfft(rev, n=L_)
            # weight vector is a Kronecker delta at position k
            w = np.zeros(2 * n_ - 1, dtype=np.float64)
            w[k] = 1.0
            w_fft = np.fft.rfft(w, n=L_)
            rev_conv = np.fft.irfft(rev_fft * w_fft, n=L_)[: 3 * n_ - 1]
            grad = 2.0 * rev_conv[n_ - 1 : 2 * n_ - 1]
            return grad

        grad_active = _grad_one_active(a, idx_max)
        src = int(np.argmax(grad_active))
        low_candidates = np.where(grad_active <= np.percentile(grad_active, 12))[0]
        if low_candidates.size == 0:
            tgt = int(rng.integers(0, n))
        else:
            tgt = int(rng.choice(low_candidates))

        if src == tgt:
            tgt = (tgt + 1) % n

        delta = a[src] * rng.uniform(0.001, 0.03)
        cand = a.copy()
        cand[src] -= delta
        cand[tgt] += delta
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()

        cand_max = _conv_fft(cand, fft_len_conv).max()
        if cand_max < best_max - 1e-12:
            best_a, best_max = cand.copy(), cand_max
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] local refinement new best max={best_max:.8f}")

        if cand_max < cur_max:
            a = cand

        # occasional jitter to shake out of plateaus
        if rng.random() < 0.001:
            width = int(rng.integers(1, max(2, n // 150)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.02))
            jitter = a.copy()
            jitter[start_idx:start_idx + width] *= factor
            jitter = np.clip(jitter, 0.0, None)
            jitter /= jitter.sum()
            a = jitter

        # early termination if we already beat the target
        if best_max * 2 * n <= 1.5025:
            break

    # --------------------------------------------------------------
    # 5) Finalise and return
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best_a.tolist())
    elapsed = time.monotonic() - start
    print(f"Finished in {elapsed:.2f}s – best max={best_max:.8f}, final score={final_score:.6f}")

    return [float(x) for x in best_a]
