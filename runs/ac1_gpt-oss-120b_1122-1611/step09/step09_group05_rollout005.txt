def propose_candidate(seed=..., budget_s=1000, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n: maybe length of height_sequence_1 if present else 3000.
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        n = max(1024, min(n, 8192))
    else:
        n = 3000

    # conv lengths etc.
    conv_len = 2*n -1
    fft_len_conv = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(3*n) # for gradient conv

    # Helper compute J from vector a (sum to 1)
    def compute_J(a):
        A = np.fft.rfft(a, fft_len_conv)
        conv = np.fft.irfft(A*A, fft_len_conv)[:conv_len]
        return float(2.0*n*conv.max())

    # Initialize candidate vectors
    candidates = []
    candidates.append(np.full(n, 1.0/n))
    # Dirichlet draws
    for _ in range(4):
        candidates.append(rng.dirichlet(np.ones(n)*0.5))
    # Use previous best if exists
    if 'height_sequence_1' in globals():
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0, None)
        s = h.sum()
        if s == 0:
            h = np.full(n, 1.0/n)
        else:
            h = h / s
        candidates.append(h)

    # Evaluate J and keep best as starting vector
    best_vec = None
    best_J = np.inf
    for cand in candidates:
        Jc = compute_J(cand)
        if Jc < best_J:
            best_J = Jc
            best_vec = cand.copy()
    print(f"Initial best J = {best_J}")

    # Gradient descent parameters
    max_iters = 20000   # high enough
    lr = 0.05
    beta1=0.9
    beta2=0.999
    epsilon0 = None # set later based on initial conv

    # Adam state
    m = np.zeros_like(best_vec)
    v = np.zeros_like(best_vec)

    # start time
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 10.0) # keep margin

It builds on the previous simulated‑annealing code but adds a **smooth‑max gradient descent** stage (Adam optimizer) followed by a short simulated‑annealing refinement.  
The smooth‑max surrogate allows us to compute a usable gradient of the non‑smooth objective `J(a)=2·n·max(conv(a,a))/(∑a)²` and perform fast projected‑gradient steps on the probability simplex.

"""
Gradient‑descent + simulated‑annealing optimizer for the step‑function problem.

We minimise
        J(a) = 2·n·max(conv(a,a)) / (∑a)²
which is scale‑invariant, so we fix ∑a = 1 (probability simplex).

The algorithm works in two phases:

1. **Smooth‑max gradient descent (Adam)**  
   The non‑smooth max is replaced by a smoothed log‑sum‑exp:
        f_ε(a) = ε·log(∑_k exp(conv_k/ε)).
   The gradient of f_ε can be written in closed form using a convolution
   with the (reversed) softmax weights, which we evaluate with FFTs.
   ε is decreased gradually, yielding a continuation method that first
   explores a smooth landscape and then focuses on the true maximum.

2. **Short simulated‑annealing polish**  
   After the gradient phase we perform a few thousand cheap SA moves
   (mass transfers between two positions) to escape possible shallow
   minima.

Both phases keep the best vector seen, project every proposal onto the
simplex, and respect the 1000 s time budget (a small safety margin is
kept).  The returned sequence is clipped to the limits enforced by
`evaluate_sequence`.
"""

import time
import numpy as np
from math import log, exp, sqrt

# ------------------------------------------------------------
# Helper utilities
# ------------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a 1‑D array onto the probability simplex
    { w ≥ 0, Σ w = 1 }.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _compute_J(a: np.ndarray, n: int, conv_len: int, fft_len: int) -> float:
    """
    Exact objective J = 2·n·max(conv(a,a)).
    `a` is assumed to sum to 1 (probability simplex).
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def _smooth_max_weights(conv: np.ndarray, eps: float) -> np.ndarray:
    """
    Compute the soft‑max weights w_k = exp(conv_k/eps) / Σ exp(conv/eps)
    in a numerically stable way.
    """
    # shift by max to avoid overflow
    shift = conv.max()
    exp_args = np.exp((conv - shift) / eps)
    sum_exp = exp_args.sum()
    if sum_exp == 0.0:
        # degenerate case – fall back to uniform weights
        return np.full_like(conv, 1.0 / conv.size)
    return exp_args / sum_exp


def _gradient_smooth_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of f_ε(a)=ε·log Σ exp(conv_k/ε) with respect to a.
    Uses the identity
        ∇f = 2·(a * w_rev)[L‑n:L][::-1]
    where * denotes linear convolution, w_rev is w reversed,
    L = 2·n‑1.
    """
    n = a.size
    L = 2 * n - 1
    w_rev = w[::-1]
    # linear convolution a * w_rev via FFT
    conv_len = n + L - 1           # = 3·n‑2
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(a, fft_len)
    fw = np.fft.rfft(w_rev, fft_len)
    conv_rev = np.fft.irfft(fa * fw, n=fft_len)[:conv_len]
    # slice L‑n … L‑1 (inclusive) and reverse
    slice_ = conv_rev[L - n : L]
    grad = 2.0 * slice_[::-1]      # shape (n,)
    return grad

# Main optimisation routine
# ------------------------------------------------------------

                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.
    The routine combines a smooth‑max Adam optimiser with a short
    simulated‑annealing polishing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------
    init_cands = []

        v /= v.sum()

    best_J = np.inf
    for cand in init_cands:
        Jcand = _compute_J(cand, n, conv_len, fft_len_conv)
        if Jcand < best_J:

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # 2. Smooth‑max gradient descent (Adam)

    # Adam hyper‑parameters
    lr = 0.04
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    v_adam = np.zeros_like(best_vec)
    beta1_pow = 1.0
    beta2_pow = 1.0

    cur_J = best_J

    deadline = start_time + max(0.0, float(budget_s) - 10.0)   # 10 s safety margin

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline and iteration < 25000:
        # ----- convolution and smooth‑max weight computation -----
        # a*a via FFT
        A = np.fft.rfft(cur_vec, n=fft_len_conv)
        conv = np.fft.irfft(A * A, n=fft_len_conv)[:conv_len]

        # adapt smoothing epsilon – start relatively large, shrink over time
        maxc = conv.max()
        # keep ε at least a few percent of the current max to stay stable
        eps = maxc * max(0.015, 0.2 * np.exp(-iteration / 4000.0))

        w = _smooth_max_weights(conv, eps)

        # ----- gradient of the smooth max -----
        grad_smooth = _gradient_smooth_max(cur_vec, w)      # ∇f_ε
        grad_J = 2.0 * n * grad_smooth                     # ∇J = 2·n·∇f_ε

        # ----- Adam update -----
        iteration += 1
        beta1_pow *= beta1
        beta2_pow *= beta2
        m = beta1 * m + (1.0 - beta1) * grad_J
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad_J * grad_J)
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v_adam / (1.0 - beta2_pow)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        new_vec = cur_vec - step
        new_vec = proj_simplex(new_vec)

        new_J = _compute_J(new_vec, n, conv_len, fft_len_conv)

        if new_J < cur_J:
            cur_vec = new_vec
            cur_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_vec = new_vec.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] GD new best J = {best_J:.6f}")

        # occasional stochastic escape if stuck for a while
        if (time.monotonic() - last_improve) > 40.0:
            # jitter the best solution and resume GD
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            cur_vec = best_vec * jitter
            cur_vec = proj_simplex(cur_vec)
            cur_J = _compute_J(cur_vec, n, conv_len, fft_len_conv)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] GD restart after stagnation – J={cur_J:.6f}")

    # --------------------------------------------------------
    # 3. Short simulated‑annealing polishing phase
    # --------------------------------------------------------
    # Re‑initialise SA state from the best GD solution
    cur_vec = best_vec.copy()
    cur_J = best_J
    T_start = best_J * 0.08
    T_end = 1e-7
    last_improve_sa = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        T = T_start * (1.0 - frac) + T_end * frac

        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:

        donor = cur_vec[i]
        if donor <= 0.0:

        # transfer up to 12 % of donor’s mass
        tr = donor * rng.uniform(0.0, 0.12)
        cand = cur_vec.copy()
        cand[i] -= tr
        cand[j] += tr
        cand = proj_simplex(cand)

        cand_J = _compute_J(cand, n, conv_len, fft_len_conv)

        # ----- Metropolis acceptance -----
        accept = False
        if cand_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = cand
            cur_J = cand_J
            if cand_J < best_J:
                best_J = cand_J
                best_vec = cand.copy()
                last_improve_sa = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] SA improved best J = {best_J:.6f}")

        # ----- stagnation handling -----

            cur_J = _compute_J(cur_vec, n, conv_len, fft_len_conv)
            last_improve_sa = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] SA restart after stagnation – J={cur_J:.6f}")

    # --------------------------------------------------------
    # 4. Final normalisation & return
    # --------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s

    print(f"Finished – final best J = {best_J:.6f} (n={n})")
    return [float(x) for x in best_vec]
