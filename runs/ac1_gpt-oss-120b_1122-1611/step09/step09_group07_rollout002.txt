assistantfinal**Solution Overview**

The optimizer works in three stages:

1. **Initialisation** – start from the provided `height_sequence_1` (if any) or from a random point on the simplex.  
2. **Simulated‑Annealing on the simplex** – a temperature controlled walk that
   * moves mass between two entries,
   * swaps two entries,
   * shuffles a short block,
   * or performs a *peak‑reduction* move that explicitly attacks the current
     maximal convolution entry.  
   The objective is `J = 2·n·max(a*a)` (the sum of the vector is kept equal to 1,
   therefore `evaluate_sequence(a) = J`).  
   The annealing schedule slowly drives the temperature from `T₀=5·10⁻²` to
   `T_end=1·10⁻⁶`.  A simple Metropolis rule accepts improving moves always and
   uphill moves with probability `exp(-(ΔJ)/T)`.  The best solution seen so far is
   stored.
3. **Fast permutation refinement** – after the annealing phase the best vector
   is polished by a pure swap‑only local search (30 s).  This often removes the
   last little bumps in the convolution peak.

All operations are performed with `numpy` and FFTs, so a single evaluation of
`J` costs *O(n log n)* and fits comfortably into the 1000 s budget.  The
algorithm is completely self‑contained, uses no file or network I/O and
conforms to the required function signature.

```python

Simulated‑annealing optimizer for the step‑function problem.

The algorithm explores the simplex (non‑negative entries that sum to 1)
using a temperature‑controlled Metropolis walk.  Four move types are used:

* mass transfer between two positions,
* swap of two entries,
* random block shuffle,
* *peak‑reduction* move that directly attacks the current maximal
  convolution entry.

The objective for a vector a is  J = 2·n·max(a*a)   (the denominator of
evaluate_sequence cancels after normalising sum(a)=1).  After a long annealing
phase a short permutation‑only local search is run to polish the result.

The routine respects the 1000 s time budget (with a 5 s safety margin) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(vec: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Compute the full (linear) convolution vec * vec using FFT.
    Returns the first conv_len entries.
    """
    fv = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fv * fv, n=fft_len)[:conv_len]
    # Numerical noise can produce tiny negatives – clip to zero.
    np.maximum(conv, 0.0, out=conv)
    return conv


def _compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv) for a vector that already satisfies sum(vec)=1."""
    conv = _full_convolution(vec, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _peak_index(conv: np.ndarray) -> int:
    """Index of the maximal entry of the convolution."""
    return int(conv.argmax())


def _contrib_to_peak(vec: np.ndarray, k: int) -> np.ndarray:
    """
    For a given convolution index k, return the contribution
    a_i * a_{k-i} for each i (zero when the partner index is out of range).
    """
    n = vec.size
    i = np.arange(n)
    j = k - i
    mask = (j >= 0) & (j < n)
    contrib = np.zeros_like(vec)
    contrib[mask] = vec[i[mask]] * vec[j[mask]]
    return contrib


def _random_block_shuffle(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Randomly shuffle a short contiguous block (length 2…10)."""
    n = vec.size
    width = rng.integers(2, min(11, n + 1))
    start = rng.integers(0, n - width + 1)
    block = vec[start:start + width].copy()
    rng.shuffle(block)
    out = vec.copy()
    out[start:start + width] = block
    return out


def _mass_transfer(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Transfer a random fraction of mass from a non‑zero entry to another entry."""
    n = vec.size
    i = rng.choice(np.where(vec > 0)[0])
    # ensure a different target
    j = rng.integers(0, n - 1)
    if j >= i:
        j += 1
    # epsilon up to 30 % of the source mass
    eps = vec[i] * rng.uniform(0.05, 0.30)
    out = vec.copy()
    out[i] -= eps
    out[j] += eps
    return out


def _swap_move(vec: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Exchange two positions."""
    n = vec.size
    i, j = rng.integers(0, n, size=2)
    while i == j:
        j = rng.integers(0, n)
    out = vec.copy()
    out[i], out[j] = out[j], out[i]
    return out


def _peak_reduce_move(vec: np.ndarray, conv: np.ndarray,
                      rng: np.random.Generator) -> np.ndarray:
    """
    Identify the current peak k*, find the pair (i, k*-i) that contributes
    the most, and move a fraction of the larger partner to a location that
    barely participates in that peak.
    """
    n = vec.size
    k = _peak_index(conv)
    contrib = _contrib_to_peak(vec, k)

    # index with maximal product a_i * a_{k-i}
    i_max = int(contrib.argmax())
    # partner index for that maximal product
    j_max = k - i_max
    # decide which of the two to donate from (the larger one)
    donor = i_max if vec[i_max] >= vec[j_max] else j_max

    # pick a target with a very small value (quantile 10 %)
    low_vals = np.where(vec < np.quantile(vec, 0.12))[0]
    if low_vals.size == 0:
        # fallback to a random index different from donor
        target = rng.integers(0, n - 1)
        if target >= donor:
            target += 1
    else:
        target = rng.choice(low_vals)
        if target == donor:
            # very unlikely, choose another
            target = (target + 1) % n

    eps = vec[donor] * rng.uniform(0.10, 0.35)
    out = vec.copy()
    out[donor] -= eps
    out[target] += eps
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine implements a simulated‑annealing walk on the probability simplex,
    followed by a short permutation‑only refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem dimension
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)          # enforce non‑negativity
            n = base.size
        except Exception:
            base = None
            n = 2048
    else:
        base = None
        n = 2048                     # default size – large enough yet fast

    # Clamp size to a comfortable range
    n = max(512, min(n, 4096))

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (reused many times)
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)              # length for a*a

    # --------------------------------------------------------------
    # 3) Initialise a feasible point (sum = 1)
    # --------------------------------------------------------------
    if base is not None and base.size == n:
        a = base.copy()
        s = a.sum()
        if s < 1e-12:
            a[:] = 1.0 / n
        else:
            a /= s
        print(f"[init] Using provided seed (len={n})")
    else:
        a = np.full(n, 1.0 / n)
        print(f"[init] No seed – starting from uniform (len={n})")

    cur_vec = a.copy()
    cur_conv = _full_convolution(cur_vec, fft_len, conv_len)
    cur_J = _compute_J(cur_vec, n, fft_len, conv_len)

    best_vec = cur_vec.copy()
    best_J = cur_J
    best_conv = cur_conv.copy()

    print(f"[init] J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 4) Simulated‑Annealing phase (≈ 85 % of remaining time)
    # --------------------------------------------------------------
    start = time.monotonic()
    total_sa = (deadline - start) * 0.85
    T0 = 5e-2                      # initial temperature
    T_end = 1e-6                  # final temperature
    it = 0
    last_improve = start

    while True:
        now = time.monotonic()
        if now >= deadline - (deadline - start) * 0.15:   # leave 15 % for swap phase
            break
        elapsed = now - start
        frac = elapsed / total_sa
        T = T0 * (T_end / T0) ** frac   # exponential cooling

        # ----------------------------------------------------------
        # Choose a move type (weighted)
        # ----------------------------------------------------------
        move_type = rng.choice(
            ["transfer", "swap", "shuffle", "peak_reduce"],
            p=[0.40, 0.30, 0.15, 0.15])

        if move_type == "transfer":
            cand_vec = _mass_transfer(cur_vec, rng)
        elif move_type == "swap":
            cand_vec = _swap_move(cur_vec, rng)
        elif move_type == "shuffle":
            cand_vec = _random_block_shuffle(cur_vec, rng)
        else:  # peak_reduce
            cand_vec = _peak_reduce_move(cur_vec, cur_conv, rng)

        # ----------------------------------------------------------
        # Evaluate the candidate
        # ----------------------------------------------------------
        cand_conv = _full_convolution(cand_vec, fft_len, conv_len)
        cand_J = _compute_J(cand_vec, n, fft_len, conv_len)

        delta = cand_J - cur_J
        accept = False
        if delta < 0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True

        if accept:
            cur_vec = cand_vec
            cur_conv = cand_conv
            cur_J = cand_J
            it += 1

            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                best_conv = cur_conv.copy()
                last_improve = now
                print(f"[{now - start:6.2f}s] SA new best J = {best_J:.6f}")

        # occasional full re‑randomisation if stagnated > 30 s
        if now - last_improve > 30.0:
            # restart from the current best with a small perturbation
            cur_vec = best_vec.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec *= noise
            cur_vec /= cur_vec.sum()
            cur_conv = _full_convolution(cur_vec, fft_len, conv_len)
            cur_J = _compute_J(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{now - start:6.2f}s] SA restart (stagnation) – best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5) Permutation‑only local search (≈ 15 % of total budget)
    # --------------------------------------------------------------
    swap_deadline = deadline
    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = best_conv.copy()
    attempts = 0
    accepts = 0
    temp_swap = 1e-5    # tiny temperature to allow occasional uphill swaps

    while time.monotonic() < swap_deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        cand = cur_vec.copy()
        cand[i], cand[j] = cand[j], cand[i]

        cand_conv = _full_convolution(cand, fft_len, conv_len)
        cand_J = _compute_J(cand, n, fft_len, conv_len)

        if cand_J < cur_J - 1e-12:
            cur_vec = cand
            cur_conv = cand_conv
            cur_J = cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                best_conv = cand_conv.copy()
                print(f"[{time.monotonic() - start:6.2f}s] swap new best J = {best_J:.6f}")
        else:
            # tiny uphill probability
            if rng.random() < np.exp(-(cand_J - cur_J) / temp_swap):
                cur_vec = cand
                cur_conv = cand_conv
                cur_J = cand_J
                accepts += 1

        # every 30 000 attempts apply a mild random perturbation to escape plateaus
        if attempts % 30000 == 0:
            cur_vec = _random_block_shuffle(cur_vec, rng)
            cur_conv = _full_convolution(cur_vec, fft_len, conv_len)
            cur_J = _compute_J(cur_vec, n, fft_len, conv_len)

    print(f"Swap phase finished – attempts: {attempts}, accepts: {accepts}")
    print(f"Best J found = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6) Return the best sequence (clip to limits of evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
```
