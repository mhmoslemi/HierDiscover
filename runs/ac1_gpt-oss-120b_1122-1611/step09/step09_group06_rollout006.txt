def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 4096

    n = max(512, min(n, 8192)) # bound length

    conv_len = 2*n - 1
    fft_len = _next_pow2(3*n)   # ensure len >= n + conv_len - 1

    # Helper functions defined above.

    # Phase A: generate random seeds, keep best
    best_vec = None
    best_J = np.inf

    # Sample seeds while time < seed_deadline (e.g., 5% of total)
    seed_deadline = start + 0.05 * (deadline - start)
    while time.monotonic() < seed_deadline:
        # sample Dirichlet concentration; pick random alpha from set
        alpha = rng.choice([0.05, 0.1, 0.3, 0.8, 3.0])
        a0 = rng.dirichlet(np.full(n, alpha))
        # evaluate
        conv = np.fft.irfft(np.fft.rfft(a0, fft_len) * np.fft.rfft(a0, fft_len), n=fft_len)[:conv_len]
        cur_J = 2.0 * n * conv.max()
        if cur_J < best_J:
            best_J = cur_J
            best_vec = a0.copy()
            print(f"[{time.monotonic() - start:6.2f}s] Random seed best J = {best_J:.6f}")

    # If no best found, fallback to uniform
    if best_vec is None:
        best_vec = np.full(n, 1.0/n)

    # Phase B: gradient descent with smoothing
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta_start = 0.5
    beta_end = 40.0
    lr_start = 0.25
    lr_end = 0.001
    beta_power = 0.9   # maybe for moving average?

    iter_count = 0
    last_improve_time = time.monotonic()
    while time.monotonic() < deadline:
        # compute fraction of time relative to smoothing stage
        elapsed = time.monotonic() - start
        total_smooth = deadline - start
        frac = elapsed / total_smooth
        beta = beta_start * (beta_end / beta_start) ** frac   # exponential schedule
        lr = lr_start * (1 - frac) + lr_end * frac

        # compute conv
        a_fft = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:conv_len]
        max_val = float(conv.max())
        cur_J = 2.0 * n * max_val

        # update best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve_time = time.monotonic()
            print(f"[{last_improve_time - start:6.2f}s] New best J={best_J:.6f}")

        # compute softmax weights
        maxc = conv.max()
        exp_args = beta * (conv - maxc)
        # for numerical stability, clip exp_args maybe large?
        # use np.exp directly, hope not overflow.
        w = np.exp(exp_args)
        w_sum = w.sum()
        if w_sum == 0.0:
            # fallback to uniform over conv indices (rare)
            w = np.full_like(conv, 1.0/conv_len)
        else:
            w /= w_sum

        # gradient: correlation a with w
        w_fft = np.fft.rfft(w, fft_len)
        corr = np.fft.irfft(a_fft * np.conj(w_fft), n=fft_len)
        offset = conv_len - 1
        g = 2.0 * corr[offset: offset + n]

        # Adam update
        m = 0.9 * m + 0.1 * g
        v = 0.999 * v + 0.001 * (g * g)
        m_hat = m / (1.0 - 0.9 ** (iter_count+1))
        v_hat = v / (1.0 - 0.999 ** (iter_count+1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # projection onto simplex
        a_new = proj_simplex(a_new)

        a = a_new
        iter_count += 1

        # occasional random perturbation for escaping
        if rng.random() < 0.005:
            a = _random_block_perturb(a, rng)

        # restart on stagnation
        if time.monotonic() - last_improve_time > 30.0:
            # reinitialize around best with small noise
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve_time = time.monotonic()
            print(f"[{last_improve_time - start:6.2f}s] Restart due to stagnation")

    # end while
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]

"""
Propose‑candidate – Smooth‑max gradient descent on the simplex.

The objective to minimise is 2·n·max(a⋆a) (the evaluation function is
scale‑invariant, so we keep the vector on the probability simplex, i.e.
non‑negative and summing to 1).  Direct optimisation of the non‑smooth
max is difficult; instead we minimise the smooth approximation

    φ_β(a) = (1/β)·log(∑_k exp(β·(a⋆a)_k))

which converges to the true max as β → ∞.
For a given β the gradient is the correlation of a with the soft‑max
weights w_k ∝ exp(β·(a⋆a)_k):

    ∇φ_β(a) = 2·(a ⋆ w)      (linear correlation, not circular)

We perform an Adam update on this gradient, project back onto the
simplex, and slowly increase β (exponential schedule) while decaying
the learning‑rate.  The search starts from a diverse pool of Dirichlet
draws (including the previous best construction if available) and
periodically injects random block perturbations and restarts after
stagnation.  The whole routine respects the 1000 s budget (5 s safety
margin) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (x > 0)."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    The perturbed vector is re‑normalised onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear self‑convolution a⋆a using FFT.
    Returns the first ``conv_len`` entries (the exact linear result).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray, fft_len: int, conv_len: int):
    """Return full convolution, index of its maximum and the max value."""
    conv = _conv_full(a, fft_len, conv_len)
    idx = int(conv.argmax())
    return conv, idx, float(conv.max())


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable soft‑max of β·conv (returns a probability vector)."""
    maxc = conv.max()
    # exp(β·(conv - maxc)) ≤ 1, safe from overflow
    exps = np.exp(beta * (conv - maxc))
    sum_exps = exps.sum()
    if sum_exps == 0.0:          # should never happen, guard nevertheless
        return np.full_like(conv, 1.0 / conv.size)
    return exps / sum_exps


def _gradient_softmax(a: np.ndarray,
                     w: np.ndarray,
                     fft_len: int,
                     conv_len: int) -> np.ndarray:
    """
    Exact gradient of the smooth (soft‑max) objective.
    g = 2·correlation(a, w)  (linear correlation, not circular).
    """
    # FFTs (a and w already zero‑padded to ``fft_len``)
    a_fft = np.fft.rfft(a, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)

    # Linear correlation via inverse FFT of a_fft * conj(w_fft)
    corr = np.fft.irfft(a_fft * np.conj(w_fft), n=fft_len)

    # In linear correlation the lag 0 element sits at index (len(w)-1)
    offset = conv_len - 1
    g = 2.0 * corr[offset: offset + a.size]
    return g


# ----------------------------------------------------------------------
# Main entry point – must be named exactly ``propose_candidate``
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The algorithm consists of:

      1. Random Dirichlet seeding (diverse initial points, including the
         previous best construction if present).

      2. Smooth‑max Adam optimisation on the simplex, with an
         exponential schedule for the smoothing parameter β (0.5 → 40)
         and a linear decay for the learning‑rate (0.25 → 0.001).

      3. Periodic random block perturbations and restarts after
         stagnation to escape shallow basins.

    The routine respects a 1000 s budget (5 s safety margin) and
    returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # keep the FFT lengths reasonable
        n = max(512, min(n, 8192))
        prev_seq = np.asarray(height_sequence_1, dtype=np.float64)
        prev_seq = np.clip(prev_seq, 0.0, None)
        if prev_seq.size != n:
            if prev_seq.size < n:
                prev_seq = np.concatenate([prev_seq,
                                           np.zeros(n - prev_seq.size)])
            else:
                prev_seq = prev_seq[:n]
        prev_seq /= prev_seq.sum()
    else:
        n = 4096
        prev_seq = None

    conv_len = 2 * n - 1
    # Length sufficient for linear convolution and correlation:
    fft_len = _next_pow2(3 * n)   # ≥ n + (2n‑1)

    # ------------------------------------------------------------------
    # 2) Initialise a pool of seeds and keep the best so far
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    # uniform seed
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    candidates = [uniform]

    # a few exponential decays
    for r in (0.97, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # two‑spike extreme seed
    spikes = np.zeros(n, dtype=np.float64)
    spikes[0] = spikes[-1] = 0.5
    candidates.append(spikes)

    # Dirichlet draws with various concentration parameters
    for alpha in (0.05, 0.1, 0.3, 0.8, 3.0):
        for _ in range(4):
            candidates.append(rng.dirichlet(np.full(n, alpha)))

    # previous construction (if any)
    if prev_seq is not None:
        candidates.append(prev_seq)

    # evaluate all seeds (fast FFT‑based J)
    for cand in candidates:
        conv = _conv_full(cand, fft_len, conv_len)
        J = 2.0 * n * conv.max()
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[{time.monotonic() - start_time:6.2f}s] "
          f"Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase – Smooth‑max Adam optimisation on the simplex
    # ------------------------------------------------------------------
    a = best_vec.copy()                 # current iterate
    m = np.zeros_like(a)                # 1st moment
    v = np.zeros_like(a)                # 2nd moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # schedule parameters
    beta_start, beta_end = 0.5, 40.0               # smoothing
    lr_start, lr_end = 0.25, 0.001                # learning‑rate
    iter_cnt = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        #   a)  time‑dependent hyper‑parameters
        # ------------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = elapsed / total                     # ∈[0,1]

        # exponential increase of β (smooth‑max becomes sharper)
        beta = beta_start * (beta_end / beta_start) ** frac

        # linear decay of the learning‑rate
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # ------------------------------------------------------------------
        #   b)  evaluate current point (exact J) and keep the best
        # ------------------------------------------------------------------
        conv, _, max_val = _conv_max(a, fft_len, conv_len)
        J = 2.0 * n * max_val
        if J < best_J - 1e-12:
            best_J = J
            best_vec = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] "
                  f"New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        #   c)  gradient of the smooth‑max surrogate
        # ------------------------------------------------------------------
        w = _softmax_weights(conv, beta)                     # soft‑max over conv
        g = _gradient_softmax(a, w, fft_len, conv_len)       # 2·correlation

        # ------------------------------------------------------------------
        #   d)  Adam update on the simplex
        # ------------------------------------------------------------------
        iter_cnt += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        a = a_new

        # ------------------------------------------------------------------
        #   e)  occasional stochastic perturbations (helps escape plateaus)
        # ------------------------------------------------------------------
        if rng.random() < 0.004:
            a = _random_block_perturb(a, rng)

        # ------------------------------------------------------------------
        #   f)  restart if no improvement for a while
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > 30.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Restart after "
                  f"stagnation – best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4) Final local refinement – tiny mass‑transfer SA moves
    # ------------------------------------------------------------------
    # The optimiser above already spent almost the whole budget.
    # If there is a tiny leftover slice we perform a few cheap
    # Metropolis‑style moves to possibly shave off the last bits.
    while time.monotonic() < deadline:
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1
        eps = min(best_vec[i], rng.exponential(scale=0.005))
        if eps <= 0.0:
            continue
        cand = best_vec.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        conv = _conv_full(cand, fft_len, conv_len)
        J = 2.0 * n * conv.max()
        if J < best_J - 1e-12:
            best_J = J
            best_vec = cand
            print(f"[{time.monotonic() - start_time:6.2f}s] SA tweak "
                  f"improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]
