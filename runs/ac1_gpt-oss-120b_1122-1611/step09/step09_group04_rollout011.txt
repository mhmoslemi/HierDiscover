def _grad_softmax(a, w, fft_len_grad):
    # gradient of softmax-weighted sum
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa*fw, n=fft_len_grad)
    start_idx = w_rev.size - 1
    # self note: length of conv is 2n-1; w_rev length = 2n-1
    # the gradient length extraction as before.
    return 2.0 * full[start_idx:start_idx+a.size]

def _objective(p, fft_len_self, n):
    # p already normalized to sum=1
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())

def _adam_optimize(p_start, time_limit, fft_len_self, fft_len_grad, n, rng):
    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    # schedule parameters
    lr_start, lr_end = 0.8, 0.001
    alpha_start, alpha_end = 5.0, 4000.0
    best_a = a.copy()
    best_J = _objective(a, fft_len_self, n)
    start = time.monotonic()
    last_improve = start
    it = 0
    while time.monotonic() < time_limit:
        now = time.monotonic()
        elapsed = now - start
        total = time_limit - start
        frac = elapsed / total if total > 0 else 0.0
        # learning rate schedule exponential decay
        lr = lr_start * (0.5 ** frac) + lr_end * (1 - 0.5 ** frac)
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)
        grad = _grad_softmax(a, w, fft_len_grad)
        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        it += 1
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)
        # evaluate true
        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{now-start:.1f}s] Adam improvement: J={best_J:.6f}")
        # jitter
        if rng.random() < 0.01:
            block = rng.integers(1, max(2, n//20))
            pos = rng.integers(0, n - block + 1)
            factor = np.exp(rng.normal(0.0, 0.05))
            pert = best_a.copy()
            pert[pos:pos+block] *= factor
            pert = proj_simplex(pert)
            a = pert
        # restart if stagnation
        if now - last_improve > 15.0:
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now-start:.1f}s] Adam restart")
    return best_a

def _subgrad_optimize(p_start, time_limit, fft_len_self, n, rng):
    p = p_start.copy()
    best_p = p.copy()
    best_J = _objective(p, fft_len_self, n)
    start = time.monotonic()
    it = 0
    step0 = 0.5
    while time.monotonic() < time_limit:
        it += 1
        conv = _self_conv(p, fft_len_self)
        max_idx = int(conv.argmax())
        # compute subgradient for active shift
        g = np.zeros_like(p)
        k = max_idx
        # vectorized loops:
        for j in range(n):
            idx1 = k - j
            if 0 <= idx1 < n:
                g[j] += p[idx1]
            idx2 = j - k
            if 0 <= idx2 < n:
                g[j] += p[idx2]
        # scaling to objective J = 2*n * conv.max(); gradient = 2*n * (p_{k-j}+p_{j-k})
        grad = 2.0 * n * g
        # step size decreasing
        step = step0 / math.sqrt(it)
        p = p - step * grad
        p = proj_simplex(p)
        cur_J = _objective(p, fft_len_self, n)
        if cur_J < best_J:
            best_J = cur_J
            best_p = p.copy()
            print(f"[{time.monotonic()-start:.1f}s] Subgrad new best J={best_J:.6f}")
        # occasional small jitter
        if rng.random() < 0.02:
            idx = rng.integers(0, n)
            delta = rng.uniform(-0.01, 0.01)
            p2 = p.copy()
            p2[idx] = max(p2[idx] + delta, 0)
            p2 = proj_simplex(p2)
            p = p2
    return best_p

def _push_candidate(p_vec, n, fft_len_self, cand_list, max_keep=5):
    # p_vec sum may not be 1; first normalise onto simplex
    p = np.asarray(p_vec, dtype=np.float64)
    p = np.clip(p, 0.0, None)
    s = p.sum()
    if s == 0:
        p = np.full_like(p, 1.0 / p.size)
    else:
        p = proj_simplex(p / s)
    J = _objective(p, fft_len_self, n)
    cand_list.append((J, p.copy()))
    cand_list.sort(key=lambda x: x[0])
    if len(cand_list) > max_keep:
        cand_list.pop()

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Two‑phase optimizer for the step‑function problem.
    Phase 1 performs an extensive grid‑search over several low‑dimensional
    shape families (edge decay, asymmetric decay, inverse‑power, cosine etc.)
    and random Dirichlet draws.
    Phase 2 refines the best few candidates with a projected Adam optimiser
    on a soft‑max surrogate, followed by a short projected sub‑gradient
    descent on the true max‑convolution objective.
    The final probability vector is scaled to have ℓ₁‑norm √(2 n) before
    returning a list of non‑negative heights.
    """

    # Determine problem size n (length of sequence to produce)
    if 'height_sequence_1' in globals():
        try:
            n = int(len(globals()['height_sequence_1']))
        except Exception:
            n = 3809
    else:
        n = 3809
    # ensure n at least 256 and not too large for FFT speed
    n = max(256, min(int(n), 8192))
    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # time‑budget handling – keep a safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Phase 1 – global exploration
    # ------------------------------------------------------------------
    best_cands = []   # list of (J, p) tuples, p on simplex

    # baseline uniform
    _push_candidate(np.full(n, 1.0 / n), n, fft_len_self, best_cands, max_keep=10)

    # edge shape (symmetric blocks)
    for frac in np.linspace(0.1, 0.45, 12):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            p = _edge_shape(n, L, gamma)
            _push_candidate(p, n, fft_len_self, best_cands, max_keep=10)

    # cosine family
    for r in np.linspace(0.0, 1.0, 13):
        for pw in np.linspace(0.3, 3.0, 9):
            p = _cosine_shape(n, r, pw)
            _push_candidate(p, n, fft_len_self, best_cands, max_keep=10)

    # exponential decay symmetric
    for alpha in np.linspace(0.0, 0.006, 9):
        for power in np.linspace(0.5, 3.0, 7):
            for baseline in [0.0, 0.05, 0.1, 0.2]:
                p = _exp_decay_shape(n, alpha, power, baseline)
                _push_candidate(p, n, fft_len_self, best_cands, max_keep=10)

    # asymmetric decay
    for la in np.linspace(0.0, 0.006, 5):
        for ra in np.linspace(0.0, 0.006, 5):
            for power in np.linspace(0.5, 3.0, 5):
                for baseline in [0.0, 0.05]:
                    p = _asym_decay_shape(n, la, ra, power, baseline)
                    _push_candidate(p, n, fft_len_self, best_cands, max_keep=10)

    # inverse power law shape
    for beta in np.linspace(0.5, 4.0, 8):
        for baseline in [0.0, 0.05, 0.1]:
            p = _inverse_power_shape(n, beta, baseline)
            _push_candidate(p, n, fft_len_self, best_cands, max_keep=10)

    # random Dirichlet draws (various concentration)
    for _ in range(50):
        alpha = rng.uniform(0.02, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push_candidate(p, n, fft_len_self, best_cands, max_keep=10)

    # include previous best if available
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0:
                prev = np.full(n, 1.0/n)
            else:
                prev = proj_simplex(prev / prev.sum())
            _push_candidate(prev, n, fft_len_self, best_cands, max_keep=10)
        except Exception:
            pass

    # keep only top K (say 12)
    best_cands.sort(key=lambda x: x[0])
    top_k = 12
    best_cands = best_cands[:top_k]
    best_J, best_p = best_cands[0]
    print(f"Initial best J after global sweep: {best_J:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – local refinement
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0:
        # time is already out – just return current best
        scale = math.sqrt(2.0 * n)
        return [float(x * scale) for x in best_p]

    # allocate 80% of remaining time to refinement, split across seeds
    refine_budget = remaining * 0.80
    per_seed = refine_budget / max(1, len(best_cands))

    for idx, (J0, p0) in enumerate(best_cands):
        if time.monotonic() >= deadline:
            break
        time_limit = min(per_seed, deadline - time.monotonic())
        # Adam refinement (majority of allocated time)
        adam_time = time_limit * 0.70
        subgrad_time = time_limit - adam_time
        refined = _adam_optimize(p0, time.monotonic() + adam_time, fft_len_self,
                                 fft_len_grad, n, rng)
        # sub‑gradient polishing
        refined = _subgrad_optimize(refined, time.monotonic() + subgrad_time,
                                    fft_len_self, n, rng)
        cur_J = _objective(refined, fft_len_self, n)
        if cur_J < best_J:
            best_J, best_p = cur_J, refined.copy()
            print(f"[seed {idx}] New global best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # Return best sequence scaled to satisfy sum = sqrt(2n)
    # ------------------------------------------------------------------
    scale = math.sqrt(2.0 * n)
    best_seq = best_p * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]

"""
Two‑phase optimizer for the step‑function problem.

Phase 1 – Very broad, cheap global sweep:
    * Enumerates several low‑dimensional shape families:
        – symmetric edge blocks (the original family)
        – cosine‑shaped bumps
        – symmetric exponential / power‑law decays with an optional flat baseline
        – asymmetric exponential decays (different rates on each side)
        – inverse‑power‑law shapes
    * Adds a batch of random Dirichlet draws with various concentrations.
    * All candidates are projected onto the probability simplex and the best
      few (by the true objective) are retained.

Phase 2 – Local refinement of the top seeds:
    * A projected Adam optimiser works on a soft‑max surrogate of the
      max‑convolution (temperature rises from a small value to a large one).
    * After Adam a short projected sub‑gradient descent on the *exact*
      max‑convolution objective polishes the result.
    * Restarts, jitter and a modest learning‑rate schedule keep the search
      from stalling.

The final probability vector `p` (sum = 1) is scaled to have ℓ₁‑norm √(2 n)
as required by ``evaluate_sequence`` and clipped to the range [0, 1000].
"""
import time
import math
import numpy as np


# ----------------------------------------------------------------------
# 0️⃣  Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape (1+r·cos)ᵖ, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """Two edge blocks of length L with linear taper (γ∈[0,1])."""
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _exp_decay_shape(
    n: int, alpha: float, power: float, baseline: float
) -> np.ndarray:
    """Symmetric exponential / power‑law decay (plus optional flat baseline)."""
    idx = np.arange(n)
    dist = np.minimum(idx, n - 1 - idx)
    v = (np.exp(-alpha * dist) ** power) + baseline
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _asym_decay_shape(
    n: int, left_alpha: float, right_alpha: float, power: float, baseline: float
) -> np.ndarray:
    """Independent exponential decays from the two ends (asymmetric)."""
    idx = np.arange(n)
    left_dist = idx
    right_dist = n - 1 - idx
    v = (
        np.exp(-left_alpha * left_dist) ** power
        + np.exp(-right_alpha * right_dist) ** power
        + baseline
    )
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _inverse_power_shape(n: int, beta: float, baseline: float) -> np.ndarray:
    """Weights proportional to (i+1)⁻ᵝ + (n‑i)⁻ᵝ plus a flat baseline."""
    i = np.arange(1, n + 1, dtype=np.float64)
    v = (1.0 / i) ** beta + (1.0 / (n - i + 1)) ** beta + baseline
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max (sums to 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate w.r.t. ``a`` (length‑n)."""
    w_rev = w[::-1]                               # reversed weights
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1                     # = 2·n‑2
    return 2.0 * full[start_idx : start_idx + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """
    True objective J = 2·n·max(p * p) for a probability vector ``p``.
    ``p`` must sum to 1 (or will be normalised by the caller).
    """
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


def _adam_optimize(
    p_start: np.ndarray,
    time_limit: float,
    fft_len_self: int,
    fft_len_grad: int,
    n: int,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Projected Adam on the soft‑max surrogate.
    Returns the best probability vector found before ``time_limit``.
    """
    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # schedule parameters
    lr_start, lr_end = 0.8, 0.001
    alpha_start, alpha_end = 5.0, 4000.0

    best_a = a.copy()
    best_J = _objective(a, fft_len_self, n)
    start = time.monotonic()
    last_improve = start
    it = 0

    while time.monotonic() < time_limit:
        now = time.monotonic()
        frac = (now - start) / max(time_limit - start, 1e-12)   # ∈[0,1]

        # learning‑rate (exponential decay) and temperature schedule
        lr = lr_start * (0.5 ** frac) + lr_end * (1 - 0.5 ** frac)
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # surrogate value & gradient
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)
        grad = _grad_softmax(a, w, fft_len_grad)

        # Adam update
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # true objective
        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{now - start:.1f}s] Adam improve: J={best_J:.6f}")

        # occasional jitter to help escape plateaus
        if rng.random() < 0.01:
            block = rng.integers(1, max(2, n // 20))
            start_idx = rng.integers(0, n - block + 1)
            factor = np.exp(rng.normal(0.0, 0.05))
            pert = best_a.copy()
            pert[start_idx : start_idx + block] *= factor
            pert = proj_simplex(pert)
            a = pert

        # restart on stagnation
        if now - last_improve > 15.0:
            a = best_a.copy()
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start:.1f}s] Adam restart (stagnation)")

    return best_a


def _subgrad_optimize(
    p_start: np.ndarray,
    time_limit: float,
    fft_len_self: int,
    n: int,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Simple projected sub‑gradient descent on the exact max‑convolution objective.
    """
    p = p_start.copy()
    best_p = p.copy()
    best_J = _objective(p, fft_len_self, n)
    start = time.monotonic()
    it = 0
    step0 = 0.5

    while time.monotonic() < time_limit:
        it += 1
        conv = _self_conv(p, fft_len_self)
        k = int(conv.argmax())                       # active shift

        # sub‑gradient of max‑convolution w.r.t. p (see analysis)
        g = np.zeros_like(p)
        for j in range(n):
            i1 = k - j
            if 0 <= i1 < n:
                g[j] += p[i1]
            i2 = j - k
            if 0 <= i2 < n:
                g[j] += p[i2]

        grad = 2.0 * n * g                          # because J = 2·n·max
        step = step0 / math.sqrt(it)                # diminishing steps
        p = p - step * grad
        p = proj_simplex(p)

        cur_J = _objective(p, fft_len_self, n)
        if cur_J < best_J:
            best_J = cur_J
            best_p = p.copy()
            print(f"[{time.monotonic() - start:.1f}s] Subgrad improve: J={best_J:.6f}")

        # tiny random jitter
        if rng.random() < 0.02:
            idx = rng.integers(0, n)
            delta = rng.uniform(-0.01, 0.01)
            p2 = p.copy()
            p2[idx] = max(p2[idx] + delta, 0.0)
            p2 = proj_simplex(p2)
            p = p2

    return best_p


def _push_candidate(
    cand_vec: np.ndarray,
    n: int,
    fft_len_self: int,
    cand_list: list,
) -> None:
    """Normalise ``cand_vec`` onto the simplex, evaluate, and store."""
    v = np.asarray(cand_vec, dtype=np.float64)
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        p = np.full_like(v, 1.0 / n)
    else:
        p = proj_simplex(v / s)
    J = _objective(p, fft_len_self, n)
    cand_list.append((J, p))

# 1️⃣  Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Two‑phase method:
        – extensive global sweep over low‑dimensional families,
        – projected Adam + sub‑gradient polishing of the best seeds.
    The returned list is already scaled so that its ℓ₁‑norm equals √(2 n).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size (use the previous construction if present)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
        except Exception:
            n = 3809
    else:
        n = 3809
    n = max(256, min(int(n), 8192))               # keep FFT cheap

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Global cheap exploration (many low‑dimensional families)
    # --------------------------------------------------------------
    all_cands = []  # will hold (J, p) for every generated shape

    # a) Uniform baseline
    _push_candidate(np.full(n, 1.0 / n), n, fft_len_self, all_cands)

    # b) Symmetric edge‑block family (original)
    for frac in np.linspace(0.1, 0.45, 12):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 9):
            p = _edge_shape(n, L, gamma)
            _push_candidate(p, n, fft_len_self, all_cands)

    # c) Cosine‑shape family
    for r in np.linspace(0.0, 1.0, 13):
        for pw in np.linspace(0.3, 3.0, 9):
            p = _cosine_shape(n, r, pw)
            _push_candidate(p, n, fft_len_self, all_cands)

    # d) Symmetric exponential / power‑law decay (with flat baseline)
    for alpha in np.linspace(0.0, 0.006, 9):
        for power in np.linspace(0.5, 3.0, 7):
            for baseline in (0.0, 0.05, 0.1, 0.2):
                p = _exp_decay_shape(n, alpha, power, baseline)
                _push_candidate(p, n, fft_len_self, all_cands)

    # e) Asymmetric decay (different rates on each side)
    for la in np.linspace(0.0, 0.006, 5):
        for ra in np.linspace(0.0, 0.006, 5):
            for power in np.linspace(0.5, 3.0, 5):
                for baseline in (0.0, 0.05):
                    p = _asym_decay_shape(n, la, ra, power, baseline)
                    _push_candidate(p, n, fft_len_self, all_cands)

    # f) Inverse‑power‑law shape
    for beta in np.linspace(0.5, 4.0, 8):
        for baseline in (0.0, 0.05, 0.1):
            p = _inverse_power_shape(n, beta, baseline)
            _push_candidate(p, n, fft_len_self, all_cands)

    # g) Random Dirichlet draws (various concentrations)
    for _ in range(50):
        alpha = rng.uniform(0.02, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push_candidate(p, n, fft_len_self, all_cands)

    # h) Include the previous best construction if it exists
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0.0:
                prev = np.full(n, 1.0 / n)
            else:
                prev = proj_simplex(prev / prev.sum())
            _push_candidate(prev, n, fft_len_self, all_cands)
        except Exception:
            pass

    # keep only the top few seeds for refinement
    all_cands.sort(key=lambda x: x[0])
    top_k = 12
    seeds = all_cands[:top_k]
    best_J, best_p = seeds[0]
    print(f"Best J after global sweep: {best_J:.6f}")

    # --------------------------------------------------------------
    # 2) Local refinement – Adam → sub‑gradient per seed
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # out of time – return the current best
        scale = math.sqrt(2.0 * n)
        return [float(x * scale) for x in best_p]

    # allocate ~80 % of remaining time to refinement
    refine_budget = remaining * 0.80
    per_seed = refine_budget / max(1, len(seeds))

    for idx, (J0, p0) in enumerate(seeds):
        if time.monotonic() >= deadline:
            break
        # split the per‑seed budget: the majority to Adam, the rest to sub‑grad
        adam_cut = per_seed * 0.70
        subgrad_cut = per_seed - adam_cut

        t_limit_adam = time.monotonic() + adam_cut
        refined = _adam_optimize(p0, t_limit_adam, fft_len_self, fft_len_grad, n, rng)

        t_limit_sub = time.monotonic() + subgrad_cut
        refined = _subgrad_optimize(refined, t_limit_sub, fft_len_self, n, rng)

        cur_J = _objective(refined, fft_len_self, n)
        if cur_J < best_J:
            best_J, best_p = cur_J, refined.copy()
            print(f"[seed {idx}] New global best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3) Return the best sequence scaled to ℓ₁‑norm √(2 n)
    # --------------------------------------------------------------
    scale = math.sqrt(2.0 * n)
    best_seq = best_p * scale
    # respect the clipping performed inside ``evaluate_sequence``
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
