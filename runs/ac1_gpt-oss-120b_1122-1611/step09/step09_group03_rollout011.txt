"""
Propose a step‑function sequence using a greedy Sidon‑set construction.

The evaluation metric is
    J = 2 * n * max(conv(a, a)) / (sum(a) ** 2)

For non‑negative a with sum = 1 this simplifies to  J = 2 * n * max(conv).
The convolution at index k is a sum of products a[i] * a[k‑i]; the
largest entry is attained when many pairs (i, j) share the same sum i+j.
If we choose the support of a to be a Sidon set (all pairwise sums are
different) then each index receives at most one unordered pair and the
maximum value is 2·w² where w = 1/|S| is the uniform weight on the set.
Thus J ≈ 2 n·2/|S|² = 4 n/|S|².  Maximising the size |S| (≈ √(2 n)) drives
the bound towards the theoretical limit of 1.

The algorithm repeatedly tries random permutations of the index set
{0,…,n‑1} and builds a Sidon set greedily:
    – maintain the set `S` of chosen indices,
    – maintain `sum_set = {i+j | i,j∈S}`,
    – add a new index only if none of its pair‑sums already appear.
Many restarts are performed until the time budget is exhausted; the
largest set found is turned into a feasible sequence by assigning the
uniform weight 1/|S| to the selected positions and zero elsewhere.

The routine respects the 1000 s budget, prints progress, and finally
returns the best sequence discovered.
"""

import time
import numpy as np


def _greedy_sidon(n: int, order: np.ndarray) -> list[int]:
    """
    Build a Sidon set using the supplied order of indices.
    Returns the list of selected positions.
    """
    S: list[int] = []
    sum_set: set[int] = set()
    for pos in order:
        pos = int(pos)
        # check self‑sum collision
        if (pos + pos) in sum_set:
            continue
        # check collisions with already chosen positions
        collision = False
        for s in S:
            if (pos + s) in sum_set:
                collision = True
                break
        if collision:
            continue
        # add the new position
        for s in S:
            sum_set.add(pos + s)
        sum_set.add(pos + pos)
        S.append(pos)
    return S


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The search builds large Sidon
    sets on the index line by greedy randomised construction and keeps
    the best (largest) set found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine a reasonable problem size.
    # If a previous construction exists we keep its length, otherwise we
    # use a default of 3809 (the length from the reference run).
    # ------------------------------------------------------------------
    default_len = 3809
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            default_len = max(default_len, prev.size)
        except Exception:
            pass
    n = int(default_len)

    # ------------------------------------------------------------------
    # Time budget handling – stop a few seconds before the hard limit.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_set: list[int] = []
    best_m: int = 0
    best_n: int = n   # length of the final vector (will be trimmed later)
    iteration: int = 0

    # Pre‑generated array of all candidate indices
    all_idx = np.arange(n, dtype=np.int64)

    # ------------------------------------------------------------------
    # Main search loop – many random restarts of the greedy construction.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        # Random order of trial positions
        order = rng.permutation(all_idx)

        S = _greedy_sidon(n, order)
        m = len(S)

        if m > best_m:
            best_m = m
            best_set = S.copy()
            best_n = max(best_set) + 1            # trim trailing zeros
            # Approximate bound (assuming the sum of the sequence is 1)
            approx_J = 2 * best_n * (2.0 / (best_m * best_m))   # 2·n·max_b, max_b≈2/m²
            print(f"[{time.monotonic():.2f}s] iter {iteration:5d} "
                  f"new best size {best_m:3d}, approx J={approx_J:.6f}")

        # Optional early stop when we reach the theoretical optimum |S|≈√(2n)
        # (this is rarely hit, but helps in very fast runs)
        if best_m >= int(np.floor(np.sqrt(2 * n))) and best_m >= 80:
            # size close to the known upper bound – no real benefit in waiting longer
            break

    # ------------------------------------------------------------------
    # Construct the final sequence from the best Sidon set.
    # Uniform weight 1/|S| guarantees sum = 1, which is scale‑invariant.
    # ------------------------------------------------------------------
    if best_m == 0:                                 # fallback – uniform vector
        seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        seq = np.zeros(best_n, dtype=np.float64)
        weight = 1.0 / best_m
        seq[best_set] = weight

    # Apply the same clipping the evaluator would apply (harmless for our values)
    seq = np.clip(seq, 0.0, 1000.0)

    # Return as a plain Python list of floats
    return [float(x) for x in seq]
