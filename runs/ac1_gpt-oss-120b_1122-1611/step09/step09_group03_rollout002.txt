def compute_J(vec):
    n = vec.size
    conv_len = 2*n-1
    fft_len = 1 << ((conv_len-1).bit_length())
    fa = np.fft.rfft(vec, n=fft_len)
    conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
    max_b = conv.max()
    sum_a = vec.sum()
    if sum_a < 1e-9:
        return np.inf
    return float(2.0 * n * max_b / (sum_a*sum_a))
```

    import time

    # Helper functions defined at top: _next_pow2, proj_simplex, compute_J, compute_conv, gradient_step ... (these are top-level)

    # 0. Setup RNG, time.
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)

    # 1. Determine initial length and candidate.
    try:
        prev = globals()["height_sequence_1"]
        init_vec = np.asarray(prev, dtype=np.float64)
        # ensure nonnegative, clip.
        init_vec = np.clip(init_vec, 0.0, None)
        # If too long, cut to max length 5000.
        max_len = 5000
        if init_vec.size > max_len:
            init_vec = init_vec[:max_len]
        # If too short, pad with zeros to length at least min_len = 2000 maybe.
        min_len = 2000
        if init_vec.size < min_len:
            init_vec = np.concatenate([init_vec, np.zeros(min_len - init_vec.size)])
        n = init_vec.size
        a = init_vec.copy()
        # Normalize to sum 1
        s = a.sum()
        if s == 0:
            a[:] = 1.0 / n
        else:
            a /= s
    except Exception:
        # fallback: pick length 3809
        n = 3809
        a = rng.dirichlet(np.full(n, 0.5))
    # Ensure simplex (projected)
    a = proj_simplex(a)

    # track best
    best_a = a.copy()
    best_J = compute_J(a)
    cur_a = a.copy()
    cur_J = best_J

    # Parameters for SA
    # Set temperature schedule: start high relative to J.
    T0 = max(best_J * 0.5, 0.001) # start temperature maybe 0.5*J
    T_end = best_J * 0.0001 + 1e-6
    # sigma for perturbation size
    sigma0 = 0.5
    sigma_end = 0.02

    # Counters
    it = 0
    last_improve = start

    # Loop
    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start
        frac = elapsed / (budget_s - 5.0)
        # Temperature descending
        T = T0 * ((T_end / T0) ** frac)

        sigma = sigma0 * ((sigma_end / sigma0) ** frac)

        # Propose
        width = rng.integers(1, max(2, n // 30))
        start_idx = rng.integers(0, n - width + 1)
        factors = np.exp(rng.normal(0.0, sigma, size=width))
        cand = cur_a.copy()
        cand[start_idx:start_idx+width] *= factors
        # Clip & renorm
        cand = np.clip(cand, 0.0, None)

def _gradient_descent_refine(a, cur_J, rng, max_iters=1000):
    n = a.size
    fft_len_self = _next_pow2(2*n-1)
    fft_len_grad = _next_pow2(3*n-2)
    start_idx = (fft_len_grad - (2*n-1)) // 2
    step = 0.5
    step_min = 1e-8
    step_max = 2.0
    # schedule for alpha: start small, end large
    alpha_start = 5.0
    alpha_end = 5000.0
    for t in range(max_iters):
        # compute conv
        fa_self = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa_self * fa_self, n=fft_len_self)[:2*n-1]
        c_max = conv.max()
        # softmax weight
        alpha = alpha_start + (alpha_end - alpha_start) * (t / max_iters)
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        grad = 2.0 * full[start_idx:start_idx + n]
        # line search
        best_step = step
        found = False
        for _ in range(20):
            a_new = proj_simplex(a - best_step * grad)
            # compute objective
            fa_self_new = np.fft.rfft(a_new, n=fft_len_self)
            conv_new = np.fft.irfft(fa_self_new * fa_self_new, n=fft_len_self)[:2*n-1]
            J_new = float(2.0 * n * conv_new.max())
            if J_new < cur_J - 1e-12:
                found = True
                break
            else:
                best_step *= 0.5
                if best_step < step_min:
                    break
        if found:
            a = a_new
            cur_J = J_new
            step = min(best_step * 1.3, step_max)
        else:
            step = max(step * 0.5, step_min)
    return a, cur_J

"""
Propose a step‑function (non‑negative sequence) that minimises the
evaluation function defined by ``evaluate_sequence``.

Algorithm
---------
1.  Initialise a feasible point on the probability simplex.
    • If a previous construction (``height_sequence_1``) is available it is
      reused (clipped, padded/truncated and projected onto the simplex);
      otherwise a Dirichlet draw (≈ uniform) of length ≈3800 is used.

2.  Simulated‑annealing (SA) phase.
    • Repeatedly propose a multiplicative log‑normal perturbation of a
      randomly‑chosen contiguous block, re‑normalise to the simplex and
      evaluate the true objective (via FFT‑based auto‑convolution).
    • Accept the move if it improves the objective or with a probability
      ``exp(-(ΔJ)/T)`` where the temperature ``T`` decays exponentially.
    • If no improvement has been seen for a while a large random restart
      (uniform simplex point) is performed to escape plateaus.

3.  Interleaved short gradient‑descent refinements.
    • Every few thousand SA steps a few projected‑gradient steps are
      performed on a smooth soft‑max surrogate of the max‑convolution.
      The surrogate’s temperature (soft‑max sharpness) is increased
      gradually, turning the smooth problem into the original ``max``.
    • This provides a fast local optimiser that can polish the current
      SA iterate.

4.  Final polishing.
    • After the time budget is exhausted (leaving a 5‑second safety
      margin) a longer gradient‑descent run is executed starting from the
      best vector seen so far.

All operations stay on the simplex (non‑negative entries that sum to 1),
which thanks to the scale‑invariance of the objective is sufficient.
The returned list is clipped to the limits enforced by ``evaluate_sequence``
(0 ≤ x ≤ 1000).  The whole routine respects the 1000 s time budget
and prints progress information.

import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidate(rng: np.random.Generator) -> np.ndarray:
    """
    Return an initial feasible vector (non‑negative, sum = 1).
    Preference is given to a previously discovered construction if it is
    present; otherwise a Dirichlet draw (≈ uniform) is used.
    """
    # use a previous construction if it exists
    prev = globals().get("height_sequence_1")
    if prev is not None:
        try:
            arr = np.asarray(prev, dtype=np.float64)
            arr = np.clip(arr, 0.0, None)
            # enforce a reasonable size (original runs used ≈3800)
            target_len = max(2000, min(6000, arr.size))
            if arr.size > target_len:
                arr = arr[:target_len]
            elif arr.size < target_len:
                arr = np.concatenate([arr, np.zeros(target_len - arr.size)])
            # project onto simplex
            return proj_simplex(arr)
        except Exception:
            pass

    # fallback: Dirichlet draw of length ≈3800
    n = 3809
    return proj_simplex(rng.dirichlet(np.full(n, 0.5)))


def _conv_max(a: np.ndarray, fft_len_self: int) -> float:
    """
    Compute 2·n·max(a∗a) assuming a is normalised to sum = 1.
    The returned value equals the exact output of ``evaluate_sequence`` for
    the same vector (because the normalisation makes the denominator 1).
    """
    n = a.size
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
    return float(2.0 * n * conv.max())


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max (with temperature 1/α) over the convolution values."""
    cmax = conv.max()
    shifted = conv - cmax          # improve numerical stability
    w = np.exp(alpha * shifted)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def _gradient_step(a: np.ndarray,
                   step: float,
                   alpha: float,
                   fft_len_self: int,
                   fft_len_grad: int,
                   start_idx: int) -> np.ndarray:
    """
    One projected gradient step on the smooth surrogate (soft‑max of the
    convolution) with step size ``step`` and soft‑max temperature ``alpha``.
    """
    n = a.size
    # ---- convolution (needed for soft‑max weights) ----
    fa_self = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa_self * fa_self, n=fft_len_self)[: 2 * n - 1]

    # ---- soft‑max weights ----
    w = _softmax_weights(conv, alpha)

    # ---- gradient of the surrogate ----
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    grad = 2.0 * full[start_idx : start_idx + n]

    # ---- descent + projection ----
    a_new = proj_simplex(a - step * grad)
    return a_new


def _gradient_descent_refine(a: np.ndarray,
                             cur_J: float,
                             rng: np.random.Generator,
                             max_iters: int = 500,
                             step0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Short runs of projected gradient descent on the soft‑max surrogate.
    The temperature ``alpha`` grows from a modest value to a large one,
    gradually turning the surrogate into the true max.  Returns the
    (possibly) improved vector and its exact objective value.
    """
    n = a.size
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(3 * n - 2)
    start_idx = (fft_len_grad - conv_len) // 2

    step = step0
    step_min = 1e-8
    step_max = 2.0
    # soft‑max temperature schedule (linear in iteration count)
    alpha_start = 2.0
    alpha_end = 4000.0

    for it in range(max_iters):
        alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iters)

        a_candidate = _gradient_step(a, step, alpha,
                                     fft_len_self, fft_len_grad, start_idx)
        J_candidate = _conv_max(a_candidate, fft_len_self)

        # simple back‑tracking line search
        if J_candidate < cur_J - 1e-12:
            a = a_candidate
            cur_J = J_candidate
            step = min(step * 1.3, step_max)
        else:
            # shrink step size
            step = max(step * 0.5, step_min)

    return a, cur_J


# ----------------------------------------------------------------------
# Core optimisation routine required by the grader
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine consists of a simulated‑annealing
    phase with occasional gradient‑descent polishing.  It respects a
    1000‑second time budget (leaving a 5‑second safety margin) and prints
    progress information.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time budget handling
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # keep a few seconds buffer for the final polish
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) initialise a feasible point (simplex, sum = 1)
    # ------------------------------------------------------------------
    a = _initial_candidate(rng)            # length fixed by the helper
    n = a.size

    # pre‑compute FFT lengths that stay constant throughout the run
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)    # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)   # for a * w_rev
    start_idx = (fft_len_grad - conv_len) // 2

    # ------------------------------------------------------------------
    # 2) bookkeeping
    # ------------------------------------------------------------------
    cur_a = a.copy()
    cur_J = _conv_max(cur_a, fft_len_self)

    best_a = cur_a.copy()
    best_J = cur_J

    print(f"[init] length={n}, J={best_J:.6f}")

    # SA hyper‑parameters
    T_start = max(best_J * 0.5, 5e-3)       # initial temperature
    T_end = best_J * 1e-4 + 1e-6           # final temperature
    sigma_start = 0.5                     # std‑dev of log‑normal block factor
    sigma_end = 0.02

    stagnation_limit = 30.0                # seconds without improvement → restart
    last_improve = start_time

    it = 0
    # ------------------------------------------------------------------
    # 3) main simulated‑annealing loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (budget_s - 5.0)          # ∈[0,1]

        # temperature & perturbation size (exponential decay)
        T = T_start * ((T_end / T_start) ** frac)
        sigma = sigma_start * ((sigma_end / sigma_start) ** frac)

        # ---- propose a neighbour ------------------------------------
        # pick a random contiguous block
        block_width = rng.integers(1, max(2, n // 30))
        block_start = rng.integers(0, n - block_width + 1)
        # multiplicative log‑normal perturbation
        factor = np.exp(rng.normal(0.0, sigma, size=block_width))
        cand = cur_a.copy()
        cand[block_start : block_start + block_width] *= factor
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand[:] = 1.0 / n
        else:
            cand /= s

        cand_J = _conv_max(cand, fft_len_self)

        # ---- Metropolis acceptance ---------------------------------
        if cand_J < cur_J or rng.random() < np.exp(-(cand_J - cur_J) / max(T, 1e-12)):
            cur_a = cand
            cur_J = cand_J
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_a = cand.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] iter {it:6d} new best J = {best_J:.6f}")

        # ---- stagnation handling -----------------------------------
        if now - last_improve > stagnation_limit:
            # large random restart to shake out of a flat region
            cur_a = proj_simplex(rng.random(n))
            cur_J = _conv_max(cur_a, fft_len_self)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – J = {cur_J:.6f}")

        # ---- occasional gradient polishing -------------------------
        if it % 5000 == 0:
            cur_a, cur_J = _gradient_descent_refine(cur_a, cur_J, rng,
                                                    max_iters=200, step0=0.5)
            if cur_J < best_J:
                best_J = cur_J
                best_a = cur_a.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] GD polish – best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4) final gradient‑descent polishing (more iterations)
    # ------------------------------------------------------------------
    best_a, best_J = _gradient_descent_refine(best_a, best_J, rng,
                                             max_iters=2000, step0=0.5)

    # ------------------------------------------------------------------
    # 5) final safety clipping (evaluate_sequence also clips)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"[final] length={n}, best J={best_J:.6f}")

    # return plain Python list as required by the grader
    return [float(x) for x in best_a]
