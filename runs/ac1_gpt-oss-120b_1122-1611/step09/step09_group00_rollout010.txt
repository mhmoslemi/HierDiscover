```
def smooth_max_grad(a, beta):
    conv = _conv_fft(a, fft_len)   # conv size 2n-1
    max_conv = conv.max()
    # For smoothing, compute weight vector p = exp(beta*(conv - max_conv))
    # Use shift to avoid huge numbers.
    exps = np.exp(beta * (conv - max_conv))
    p = exps / exps.sum()
    # Compute convolution of a and p
    L = fft_len  # precomputed
    A_fft = np.fft.rfft(a, n=L)
    P_fft = np.fft.rfft(p, n=L)
    conv_ap = np.fft.irfft(A_fft * P_fft, n=L)[: n + len(p) - 1]   # full conv length
    # Compute convolution with reversed p
    p_rev = p[::-1]
    P_rev_fft = np.fft.rfft(p_rev, n=L)
    conv_ap_rev = np.fft.irfft(A_fft * P_rev_fft, n=L)[: n + len(p_rev) - 1]
    # term2 = conv_ap[0:n]
    term2 = conv_ap[:n]
    # term1 = conv_ap_rev[len(p)-1 - np.arange(n)]  # because m-1 - k indexing
    m = len(p)
    term1 = conv_ap_rev[m-1 - np.arange(n)]
    grad = term1 + term2
    return grad, conv, max_conv
```

assistantfinal**Solution Overview**

The evaluation function  

\[
F(a)=\frac{2n\max (a*a)}{(\sum a)^2}
\]

is scale‑invariant, therefore we always keep the sequence on the simplex
\(\sum a_i=1,\;a_i\ge 0\).  
The goal is to make the self‑convolution \(c=a*a\) as flat as possible,
i.e. to minimise its supremum.

The new search routine consists of three phases

1. **Smooth‑max minimisation** –  a differentiable surrogate  
   \(\operatorname{softmax}_\beta(c)=\frac1\beta\log\!\sum_i e^{\beta c_i}\)  
   is used.  Its gradient is a weighted sum of the true gradients of the
   individual convolution entries.  An exponentiated‑gradient (mirror‑descent)
   update on the log‑probabilities keeps the iterate non‑negative and
   automatically enforces the simplex constraint.

2. **Hard‑max sub‑gradient descent** – after the surrogate has
   flattened the convolution we switch to the exact sub‑gradient of the
   max‑operator.  The full active set (all indices within a tiny tolerance
   of the current maximum) is used and a projected additive step
   performs the final fine tuning.

3. **Stochastic local refinement** – tiny directed mass‑transfer moves
   (source ≈ largest gradient entry, target ≈ smallest) are accepted when
   they improve the true score.  Random jitter is added occasionally to
   escape shallow plateaus.

All three phases respect the simplex constraint via the Euclidean
projection onto the probability simplex.  FFT‑based convolution
(\(O(n\log n)\)) makes each iteration cheap enough to run many thousands
of steps within the 1000 s budget.

The routine starts from the best sequence found so far
(`height_sequence_1`) if it exists, otherwise from a uniform distribution.
The final sequence is clipped to the limits imposed by the evaluator and
returned.

```python
"""
Fast three‑phase optimiser for the max‑convolution objective.

Phase 1 – smooth‑max (exponentiated‑gradient) minimisation.
Phase 2 – exact hard‑max sub‑gradient descent.
Phase 3 – stochastic directed mass‑transfer refinement.

All iterates stay on the simplex (∑a_i = 1, a_i ≥ 0) using a Euclidean
projection.  FFT‑based convolutions give O(n log n) per iteration,
allowing many thousands of updates within the 1000 s budget.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length ``fft_len``.
    Returns the full convolution of length 2*len(a)-1.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_active(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of ``max_k (a*a)[k]`` for the active set
    (indices attaining the current maximum).  For each active index k
    the contribution to ∂max/∂a_i is a_{k-i}.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in active:
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        i = np.arange(i_start, i_end, dtype=int)
        grad[i] += a[k - i]
    return grad


# ----------------------------------------------------------------------
# Phase‑1 – smooth‑max gradient (exponentiated‑gradient)
# ----------------------------------------------------------------------


def _smooth_max_step(a: np.ndarray,
                     fft_len: int,
                     beta: float,
                     eta: float) -> np.ndarray:
    """
    One exponentiated‑gradient step on the smoothed max objective.
    Returns the updated probability vector (still on the simplex).
    """
    n = a.size
    # convolution a*a
    conv = _conv_fft(a, fft_len)                # length 2n‑1
    max_conv = conv.max()
    # smooth weights (log‑sum‑exp trick)
    w = np.exp(beta * (conv - max_conv))
    w /= w.sum()                               # now ∑ w_i = 1

    # ----- gradient of the smoothed max -----
    # term2 = (a * w)[k]  for k = 0 … n‑1
    A_fft = np.fft.rfft(a, n=fft_len)
    W_fft = np.fft.rfft(w, n=fft_len)
    conv_aw = np.fft.irfft(A_fft * W_fft, n=fft_len)
    conv_aw = conv_aw[: n + len(w) - 1]        # full linear conv
    term2 = conv_aw[:n]

    # term1 = (a * w_rev)[m‑1‑k] ,  m = len(w)
    w_rev = w[::-1]
    W_rev_fft = np.fft.rfft(w_rev, n=fft_len)
    conv_aw_rev = np.fft.irfft(A_fft * W_rev_fft, n=fft_len)
    conv_aw_rev = conv_aw_rev[: n + len(w_rev) - 1]
    m = len(w)
    term1 = conv_aw_rev[m - 1 - np.arange(n)]

    grad = term1 + term2                         # ∂/∂a of weighted sum

    # exponentiated‑gradient update on log‑probabilities
    log_a = np.log(a + 1e-15) - eta * grad
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Phase‑2 – exact hard‑max sub‑gradient descent
# ----------------------------------------------------------------------


def _hard_max_step(a: np.ndarray,
                   fft_len: int,
                   step_factor: float,
                   iteration: int) -> np.ndarray:
    """
    Projected additive sub‑gradient step for the true max objective.
    """
    conv = _conv_fft(a, fft_len)
    max_b = conv.max()
    eps = max_b * 1e-6
    active = np.where(conv >= max_b - eps)[0]
    if active.size == 0:
        active = np.array([np.argmax(conv)])

    grad = _gradient_from_active(a, active)
    step = step_factor / np.sqrt(iteration + 1.0)
    a_candidate = a - step * grad
    a_candidate = _project_onto_simplex(a_candidate)
    return a_candidate


# ----------------------------------------------------------------------
# Phase‑3 – stochastic local refinement
# ----------------------------------------------------------------------


def _local_refinement_step(a: np.ndarray,
                           fft_len: int,
                           rng: np.random.Generator,
                           src: int,
                           tgt: int) -> tuple[np.ndarray, int, int]:
    """
    Tiny directed mass‑transfer move.
    Returns (new_sequence, new_src, new_tgt)
    """
    n = a.size
    if a[src] <= 0.0:
        src = int(rng.integers(0, n))
    if src == tgt:
        tgt = int(rng.integers(0, n))

    # transfer fraction in [0.1 % , 5 %] (log‑uniform)
    frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
    delta = a[src] * frac

    a_new = a.copy()
    a_new[src] -= delta
    a_new[tgt] += delta
    a_new = _project_onto_simplex(a_new)
    return a_new, src, tgt


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

        a = np.clip(a, 0.0, None)

        else:

            best = a_candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached in local phase")
                break

        # keep the improvement, otherwise stay where we are
        if score_cand < 2.0 * n * max_b:
            a = a_candidate

        it += 1

        # occasional jitter to shake the solution out of tiny plateaus
        if it % 4000 == 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jitter = a.copy()
            a_jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jitter)

        if it % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local iter={it}, best={best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence found (clip to evaluator limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
```
