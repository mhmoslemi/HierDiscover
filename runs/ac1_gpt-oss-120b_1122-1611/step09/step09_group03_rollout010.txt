def compute_J(a):
    n = a.size
    conv_len = 2*n - 1
    # compute convolution via FFT using next power of 2
    fft_len = 1 << ((conv_len - 1).bit_length())
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_b = conv.max()
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))

"""
Propose a step‑function sequence that lowers the evaluation score.

Algorithm (hill‑climbing with gradient‑guided mass transfers)
-----------------------------------------------------------
1. Initialise a probability vector `a` (non‑negative, sum = 1).  If a previous
   construction (``height_sequence_1``) exists it is re‑used, otherwise a
   Dirichlet draw is taken.
2. In each iteration we:
   * Compute the self‑convolution `c = a * a` (full) via a single FFT.
   * Locate all indices where `c` attains its maximum (`M`); form the mask
     `w` (1 on the maximal positions, 0 elsewhere).
   * Compute the sub‑gradient of the max‑convolution with respect to `a`
     using another FFT:
         grad = 2·(a * w[::-1])
   * Pick the index `i_high` with the largest gradient entry (the entry that
     contributes most to the current peak) and the index `i_low` with the
     smallest gradient entry.  Transfer a small amount of probability mass
     ε from `i_high` to `i_low`.
   * The new candidate `a'` is evaluated (again by a single FFT).  If the
     objective `J = 2·n·max(a'*a')` improves, the move is kept; otherwise it
     is rejected (or accepted with a tiny annealing probability).
3. When no improvement has been seen for a while we perform a random
   “block perturbation’’ (multiplicative log‑normal factor on a random
   contiguous block) to escape plateaus.
4. The loop runs until the time budget (≈ 1000 s) is exhausted.  The best
   sequence observed is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _initial_candidate(rng: np.random.Generator) -> np.ndarray:
    """
    Return a feasible probability vector (non‑negative, sum = 1).  Preference
    is given to a previously discovered construction (``height_sequence_1``);
    otherwise a Dirichlet draw is used.
    """
    if "height_sequence_1" in globals():
        try:
            raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            raw = np.clip(raw, 0.0, None)
            if raw.size == 0:
                raise ValueError
            raw = raw / raw.sum()
            return raw
        except Exception:
            pass
    # fallback: Dirichlet draw with modest concentration (α=0.5)
    n = 4096
    raw = rng.dirichlet(np.full(n, 0.5))
    return raw / raw.sum()


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Uses a fast hill‑climbing method guided by the exact sub‑gradient of the
    max‑convolution, together with occasional stochastic block perturbations.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # 1) Initialise a probability vector (sum = 1)
    # --------------------------------------------------------------
    a = _initial_candidate(rng)                     # shape (n,)
    n = a.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(n + conv_len - 1)           # length for a * w convolutions

    # --------------------------------------------------------------
    # Helper: evaluate objective J = 2·n·max(a*a) (sum = 1 ⇒ scaling invariant)
    # --------------------------------------------------------------
    def compute_J(vec: np.ndarray) -> float:
        """Return the exact objective value for a probability vector."""
        fa = np.fft.rfft(vec, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())

    best_a = a.copy()
    best_J = compute_J(a)
    print(f"[init] n={n:,}  J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 2) Optimisation loop
    # --------------------------------------------------------------
    iteration = 0
    last_improve = time.monotonic()
    stagnation_limit = 20.0          # seconds without improvement → jitter
    temperature_start = 0.01
    temperature_end = 1e-6

    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        # linear cooling schedule for the occasional uphill‑move acceptance
        frac_time = (now - start_time) / (deadline - start_time)
        temperature = temperature_start * (1 - frac_time) + temperature_end * frac_time

        # -------- compute self‑convolution and maximal indices --------
        a_fft = np.fft.rfft(a, n=fft_len)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[:conv_len]
        M = conv_full.max()
        # mask of positions attaining the maximum
        w = np.zeros(conv_len, dtype=np.float64)
        w[conv_full == M] = 1.0

        # -------- sub‑gradient of the max‑convolution -----------------
        w_rev = w[::-1]
        w_fft = np.fft.rfft(w_rev, n=fft_len)
        grad_full = np.fft.irfft(a_fft * w_fft, n=fft_len)
        # extract the part that corresponds to the n coordinates of a
        start_idx = conv_len - 1
        grad = 2.0 * grad_full[start_idx:start_idx + n]

        # -------- choose a transfer direction -------------------------
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))

        # avoid degenerate moves
        if i_high == i_low or a[i_high] < 1e-12:
            # fallback random jitter
            a = _random_block_perturb(best_a, rng)
            continue

        # magnitude of the transfer (randomised fraction of the source mass)
        eps_frac = rng.uniform(0.05, 0.20)
        eps = eps_frac * a[i_high]
        eps = min(eps, 0.1)                     # hard ceiling to keep steps modest

        # ------- construct candidate after the transfer ---------------
        a_candidate = a.copy()
        a_candidate[i_high] -= eps
        a_candidate[i_low] += eps

        # ------- evaluate candidate ------------------------------------
        J_candidate = compute_J(a_candidate)

        # ------- acceptance test ---------------------------------------
        if J_candidate < best_J - 1e-12:
            a = a_candidate
            best_J = J_candidate
            best_a = a_candidate.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] iter {iteration:5d} new best J = {best_J:.6f}")
        else:
            # small probability of accepting a worsening move (simulated annealing)
            if rng.random() < np.exp(-(J_candidate - best_J) / max(temperature, 1e-12)):
                a = a_candidate
            # stagnation handling
            if now - last_improve > stagnation_limit:
                a = _random_block_perturb(best_a, rng)
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] jitter (stagnation) – J remains {best_J:.6f}")

        # (optional) periodic re‑normalisation to counteract numerical drift
        if iteration % 500 == 0:
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s

    # --------------------------------------------------------------
    # 3) Return the best sequence (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
