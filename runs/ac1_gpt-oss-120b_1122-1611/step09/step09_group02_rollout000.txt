"""
Optimization of a non‑negative step function for the objective

    J(a) = 2·n·max(a∗a) / (sum a)²   (the denominator is scale‑invariant)

The code uses three phases inside the 1000 s budget:

1. **Initial portfolio** – a handful of analytically‑motivated vectors
   (uniform, exponential decay, linear decay, two spikes, Dirichlet draws,
   and the previously known best construction).  The best of them is kept.

2. **Active‑set linear‑programming update** –  
   Given the current vector `f` (normalised to sum = 1) we compute
   `M = max(f∗f)`.  We then solve a linear program

        maximise   Σ g_j
        subject to g ≥ 0
                   (f∗g)[k] ≤ M   for the *active* shifts k
       
   where only the `max_active` largest entries of `f∗f` are kept as
   constraints (typically ≤ 200).  The solution `g` points in a direction
   that keeps every important convolution entry ≤ M while spreading the
   mass of `g`.  A convex combination `f←(1‑t)f + t·g` (followed by a
   projection onto the probability simplex) is accepted whenever it
   strictly improves the true objective.  The step size `t` is adapted
   and the process repeats until the time budget is exhausted.

3. **Soft‑max Adam refinement** –  A final short Adam optimisation on the
   smooth surrogate `softmax_α(f∗f)` (temperature `α` ramps up) cleans up
   any remaining plateaux.

All helpers are top‑level, the algorithm respects the 5 s safety margin
and returns the best sequence found.
"""

import time
import numpy as np
import cvxpy as cp

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Small set of diverse starting vectors, each already normalised."""
    cand: list[np.ndarray] = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes on the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of a given length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a)  (the vector is assumed to sum to 1)."""
    conv = self_convolution(a, fft_len_self, conv_len)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _active_constraints(conv: np.ndarray, max_k: int = 200) -> np.ndarray:
    """Indices of the `max_k` largest entries of a convolution array."""
    if conv.size <= max_k:
        return np.arange(conv.size)
    idx = np.argpartition(-conv, max_k - 1)[:max_k]
    # optional ordering (not required for correctness)
    idx = idx[np.argsort(-conv[idx])]
    return idx.astype(int)


def solve_lp_for_g(f: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray | None:
    """
    Linear programme:
        maximise   Σ g_j
        subject to g ≥ 0
                   (f∗g)[k] ≤ M    for k ∈ active_idx

    Returns the optimal vector `g` (may be all‑zero if the problem is
    infeasible) or ``None`` on solver failure.
    """
    n = f.size
    g = cp.Variable(n, nonneg=True)
    constraints = []

    # each active shift produces a linear inequality
    for k in active_idx:
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        # coefficients for g[lo:hi+1] are f[k‑j]
        coeffs = f[k - np.arange(lo, hi + 1, dtype=int)]
        constraints.append(cp.sum(cp.multiply(coeffs, g[lo:hi + 1])) <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    except Exception:          # any solver failure – fall back to None
        return None

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)   # numerical safety


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The method combines an active‑set LP improvement
    with a final Adam refinement.  The function respects the 1000 s time
    budget (5 s safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    now = time.monotonic()
    deadline = now + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(1024, min(5000, n))          # keep n moderate
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a

    # ------------------------------------------------------------------
    # 2) initialise from a small portfolio (including the previous best)
    # ------------------------------------------------------------------
    init_cands = _initial_candidates(n, rng)

    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        prev = np.clip(prev, 0.0, None)
        if prev.sum() == 0.0:
            prev[:] = 1.0 / n
        else:
            prev /= prev.sum()
        init_cands.append(prev)

    # pick the best among the initial portfolio
    best_vec = None
    best_score = np.inf
    for cand in init_cands:
        vec = proj_simplex(cand)          # guarantee sum = 1, non‑negative
        scr = compute_J(vec, n, fft_len_self, conv_len)
        if scr < best_score:
            best_score = scr
            best_vec = vec.copy()
    print(f"[0.00s] Initial best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Active‑set LP phase
    # ------------------------------------------------------------------
    f = best_vec.copy()
    best_score = compute_J(f, n, fft_len_self, conv_len)
    best_vec = f.copy()

    max_active = 200          # number of constraints handed to the LP
    step_size = 0.2           # initial convex‑combination weight
    min_step = 0.02
    decay_when_success = 0.9
    stagnation_counter = 0

    while time.monotonic() < deadline:
        # current convolution and worst value
        conv_f = self_convolution(f, fft_len_self, conv_len)
        M = conv_f.max()
        J_cur = 2.0 * n * M

        # active constraints (largest entries of f∗f)
        active_idx = _active_constraints(conv_f, max_active)

        # solve the LP – may return None on failure
        g = solve_lp_for_g(f, M, active_idx)
        if g is None or g.sum() == 0.0:
            # as a cheap rescue, apply a random block jitter
            f = _random_block_perturb(f, rng)
            continue

        # normalise the LP solution to a probability vector
        g = g / g.sum()

        # try a few decreasing step sizes until an improvement is seen
        improved = False
        t = step_size
        while t >= min_step:
            f_new = (1.0 - t) * f + t * g
            f_new = np.clip(f_new, 0.0, None)
            s = f_new.sum()
            if s == 0.0:
                f_new = np.full_like(f_new, 1.0 / n)
            else:
                f_new /= s

            J_new = compute_J(f_new, n, fft_len_self, conv_len)
            if J_new < best_score - 1e-12:
                best_score = J_new
                best_vec = f_new.copy()
                f = f_new
                improved = True
                print(f"[{time.monotonic() - now:6.2f}s] LP iter – new best J = {best_score:.6f}")
                break
            else:
                t *= 0.5

        if improved:
            # successful step – a little more aggressive next time
            step_size = min(0.5, step_size / decay_when_success)
            stagnation_counter = 0
            continue

        # No improvement → shrink step size and occasionally jitter
        step_size = max(min_step, step_size * 0.9)
        stagnation_counter += 1
        if rng.random() < 0.3:
            f = _random_block_perturb(f, rng)

        # If we have been stagnant for a while, leave the LP phase
        if stagnation_counter > 30:
            break

    # ------------------------------------------------------------------
    # 4) Adam refinement on a soft‑max surrogate
    # ------------------------------------------------------------------
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it_adam = 0
    last_improve = time.monotonic()
    start_adam = time.monotonic()
    total_adam = deadline - start_adam

    while time.monotonic() < deadline:
        it_adam += 1
        now_iter = time.monotonic()
        frac = (now_iter - start_adam) / total_adam if total_adam > 0 else 1.0

        # learning‑rate schedule
        lr_start, lr_end = 0.25, 0.001
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # temperature schedule for the smooth max
        alpha_start, alpha_end = 2.0, 2000.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # self‑convolution of the current iterate
        conv = self_convolution(a, fft_len_self, conv_len)

        # soft‑max over the convolution entries
        c_max = conv.max()
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the soft‑max surrogate (via FFT)
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_self)
        fw = np.fft.rfft(w_rev, n=fft_len_self)
        full = np.fft.irfft(fa * fw, n=fft_len_self)
        start_idx = w_rev.size - 1                     # = 2·n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # Adam update + projection onto the simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it_adam)
        v_hat = v / (1.0 - beta2 ** it_adam)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # true objective evaluation
        cur_score = compute_J(a, n, fft_len_self, conv_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = now_iter
            print(f"[{now_iter - now:6.2f}s] Adam new best J = {best_score:.6f}")

        # occasional random jitter
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart if no improvement for a while
        if now_iter - last_improve > 15.0:
            a = best_vec.copy()
            a = a * np.exp(rng.normal(0.0, 0.07, size=n))
            a = proj_simplex(a)
            last_improve = now_iter
            print(f"[{now_iter - now:6.2f}s] Adam restart")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
