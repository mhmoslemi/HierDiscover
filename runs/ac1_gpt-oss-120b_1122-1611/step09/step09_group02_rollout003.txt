"""
Simulated‑annealing + short projected sub‑gradient refinement.

The routine builds a probability‑simplex vector (non‑negative, sum = 1)
and repeatedly moves a tiny amount of mass from one index to another.
The annealing temperature and maximal transfer size are linearly
decreased, which lets the search explore widely at first and then
fine‑tune.  The objective (2·n·max(a∗a) when the sum is 1) is evaluated
using an FFT‑based convolution – fast enough to afford many thousands of
iterations.  After the annealing stage a few sub‑gradient steps are run
to polish the best vector found.  The best sequence seen is returned
(automatically clipped to the limits enforced by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a)   (assumes sum(a)=1)."""
    conv = self_convolution(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The algorithm consists of two stages:

    1.  Simulated‑annealing style mass‑transfer moves (the bulk of the
        time budget).  Temperature and maximal transfer magnitude are
        linearly cooled.
    2.  A short projected sub‑gradient descent (hard‑max surrogate) that
        polishes the best vector found by stage 1.

    The best sequence seen is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (1 s safety margin)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # 1) Choose problem size
    # ------------------------------------------------------------------
    # Use the length of the previously known construction if present.
    # Clamp to a moderate interval to keep the runtime reasonable.
    if "height_sequence_1" in globals():
        try:
            prev_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = prev_seq.size
        except Exception:
            n = 4000
    else:
        n = 4000
    n = max(2000, min(5000, n))          # keep n moderate

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2) Initial vector (projected to simplex)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        # Interpolate if length differs
        if a0.size != n:
            x_old = np.linspace(0.0, 1.0, a0.size)
            x_new = np.linspace(0.0, 1.0, n)
            a0 = np.interp(x_new, x_old, a0)
        a = proj_simplex(a0)
    else:
        # a uniform start is a good baseline
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3) Pre‑compute the current objective
    # ------------------------------------------------------------------
    conv = self_convolution(a, fft_len, conv_len)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    print(f"[{0.0:6.2f}s] start – J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing stage
    # ------------------------------------------------------------------
    # Annealing schedules (linear in elapsed fraction)
    T_start, T_end = 1e-2, 1e-6          # temperature for uphill moves
    d_start, d_end = 0.10, 1e-4          # maximal mass transfer per move

    # Jitter parameters (occasional “big shake” to escape plateaux)
    jitter_every = 5000                 # iterations
    jitter_sigma = 0.12                 # std‑dev of multiplicative log‑normal noise

    it = 0
    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        elapsed = now - start
        frac = elapsed / (deadline - start)

        # annealed parameters
        T = T_start * (1.0 - frac) + T_end * frac
        d_max = d_start * (1.0 - frac) + d_end * frac

        # ---- propose a mass‑transfer move --------------------------------
        # pick source index weighted by current mass (more likely to move from
        # heavy locations) and a uniform destination.
        src = rng.choice(n, p=a)                # source index
        if a[src] <= 0.0:
            continue
        # amount to move – at most d_max and at most what the source holds
        delta = rng.random() * min(d_max, a[src])
        if delta <= 0.0:
            continue
        dst = rng.integers(0, n)                # destination index
        if dst == src:
            continue

        # apply move in‑place
        a[src] -= delta
        a[dst] += delta

        # recompute convolution and new objective
        conv = self_convolution(a, fft_len, conv_len)
        new_score = 2.0 * n * conv.max()
        Δ = new_score - cur_score

        # accept if improvement, otherwise with Metropolis probability
        accept = False
        if Δ <= 0.0:
            accept = True
        else:
            # avoid overflow for very small T
            if T > 0.0:
                accept = rng.random() < np.exp(-Δ / T)

        if accept:
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                print(f"[{elapsed:6.2f}s] SA new best J = {best_score:.6f}")
        else:
            # revert the move
            a[dst] -= delta
            a[src] += delta

        # occasional jitter to inject diversity
        it += 1
        if it % jitter_every == 0:
            # re‑seed from the best vector with multiplicative noise
            a = best_a.copy()
            a *= np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = proj_simplex(a)
            conv = self_convolution(a, fft_len, conv_len)
            cur_score = 2.0 * n * conv.max()
            # no improvement check here – jitter is a restart
            print(f"[{time.monotonic() - start:6.2f}s] SA jitter – current J = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Short projected sub‑gradient descent (hard‑max surrogate)
    # ------------------------------------------------------------------
    # Allocate a few seconds for polishing (≈5 % of total budget)
    polish_deadline = start + 0.95 * (deadline - start)
    a = best_a.copy()
    # recompute convolution for the polishing start
    conv = self_convolution(a, fft_len, conv_len)
    cur_score = 2.0 * n * conv.max()

    # step‑size schedule (linear decay)
    eta_start, eta_end = 0.3, 1e-4

    while time.monotonic() < polish_deadline:
        # locate the index of the maximal convolution entry
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max entry (hard max)
        # grad[i] = 2 * a[j] where j = k_max - i (if 0≤j<n)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # learning‑rate based on elapsed fraction of the polishing phase
        now = time.monotonic()
        elapsed = now - start
        ft = (elapsed) / (polish_deadline - start)
        eta = eta_start * (1.0 - ft) + eta_end * ft

        # projected gradient step (simple Euclidean projection onto simplex)
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)          # non‑negativity
        a_new = proj_simplex(a_new)

        # evaluate
        conv_new = self_convolution(a_new, fft_len, conv_len)
        new_score = 2.0 * n * conv_new.max()

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            print(f"[{now - start:6.2f}s] Sub‑grad new best J = {best_score:.6f}")

        a = a_new
        conv = conv_new

    # ------------------------------------------------------------------
    # 6) Return the best vector (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
