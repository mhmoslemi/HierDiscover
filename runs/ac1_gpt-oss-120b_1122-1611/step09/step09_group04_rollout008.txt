def adam_refine(p_start, n, fft_len_self, fft_len_grad, deadline, best_record, rng):
    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    last_improve = time.monotonic()
    start = time.monotonic()
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start
        total = deadline - start
        if total <= 0:
            break
        frac = elapsed / total
        # schedule
        lr = 0.8 * (0.5 ** frac) + 0.001 * (1 - 0.5 ** frac)
        alpha = 2.0 * (1.0 - frac) + 5000.0 * frac
        # compute conv and softmax weights
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)
        # gradient
        grad = _grad_softmax(a, w, fft_len_grad)
        # Adam
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)
        cur_J = _objective(a, fft_len_self, n)
        if cur_J < best_record['J'] - 1e-12:
            best_record['J'] = cur_J
            best_record['p'] = a.copy()
            last_improve = now
            print(f"[Adam] t={elapsed:.1f}s best J={cur_J:.6f}")
        # jitter
        if rng.random() < 0.01:
            block = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - block + 1)
            factor = np.exp(rng.normal(0.0, 0.08))
            a_pert = a.copy()
            a_pert[start_idx:start_idx+block] *= factor
            a_pert = proj_simplex(a_pert)
            a = a_pert
        # restart on stagnation
        if now - last_improve > 15.0:
            a = best_record['p'].copy()
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[Adam] restart at {elapsed:.1f}s best J={best_record['J']:.6f}")

def lp_refine(p_start, n, fft_len_self, deadline, best_record, rng, K_active=30, t_mix=0.5):
    p = p_start.copy()
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        now = time.monotonic()
        # compute conv
        conv = _self_conv(p, fft_len_self)
        M = conv.max()
        # select top K constraints
        K = min(K_active, conv.size)
        top_idx = np.argpartition(conv, -K)[-K:]
        # Build constraint rows matrix (K x n)
        rows = np.zeros((K, n), dtype=np.float64)
        for row_i, k in enumerate(top_idx):
            j_start = max(0, k - (n - 1))
            j_end = min(k, n - 1) + 1
            js = np.arange(j_start, j_end)
            rows[row_i, j_start:j_end] = p[k - js]
        # Linear program
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g) -> minimize -sum(g)
        b_ub = np.full(K, M, dtype=np.float64)
        bounds = [(0, None)] * n
        try:
            res = linprog(c, A_ub=rows, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            print("LP error:", e)
            break
        if not res.success:
            # if fails, break
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 0:
            break
        g = g / sum_g
        # Combine
        p_new = (1 - t_mix) * p + t_mix * g
        p_new = proj_simplex(p_new)
        cur_J = _objective(p_new, fft_len_self, n)
        if cur_J < best_record['J'] - 1e-12:
            best_record['J'] = cur_J
            best_record['p'] = p_new.copy()
            last_improve = now
            print(f"[LP] t={now - (deadline - (deadline - now)):.1f}s new best J={cur_J:.6f}")
        # check stagnation
        if now - last_improve > 20.0:
            # jitter
            idx = rng.integers(0, n)
            p_new = p_new.copy()
            p_new[idx] = max(0.0, p_new[idx] + rng.normal(0.0, 0.02))
            p_new = proj_simplex(p_new)
            last_improve = now
        p = p_new

"""
Hybrid optimizer for the step‑function max‑convolution problem.

Strategy
--------
1. Global exploration – cheap families (edge blocks, cosine, Dirichlet) and
   the previous best sequence.  The top‑few candidates (probability vectors)
   are kept.

2. Local refinement – each seed undergoes two complementary upgrades:
   * Projected Adam on a temperature‑controlled soft‑max surrogate of the
     max‑convolution (smooth, gradient‑based).
   * An active‑set linear program that maximises the ℓ¹‑norm of a
     partner vector g under the constraints (p⋆g) ≤ M, where M is the
     current maximum convolution value.  Only the K largest convolution
     constraints are kept, making the LP tiny and cheap.

   The two phases are run sequentially (≈50 % time each) for every seed,
   with jitter / restart mechanisms to escape plateaus.

3. The best probability vector found is finally scaled to the
   √(2 n) ‑norm required by ``evaluate_sequence`` and clipped to the
   allowed range.

All helper functions are defined at top‑level (no closures) and the
search respects the supplied time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families used for cheap global search
# ----------------------------------------------------------------------


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape: (1 + r*cos(theta))**power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L with linear taper.
    gamma=0 → uniform blocks; gamma=1 → linearly decay to zero.
    """
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


# ----------------------------------------------------------------------
# Soft‑max surrogate utilities (used by Adam)
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max (weights sum to 1)."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate w.r.t. a."""
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1
    return 2.0 * full[start_idx:start_idx + a.size]


# ----------------------------------------------------------------------
# Exact objective (what ``evaluate_sequence`` returns after scaling)
# ----------------------------------------------------------------------


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """
    J(p) = 2·n·max(p∗p)  – the exact value returned by evaluate_sequence
    for the scaled sequence.
    """
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Projected Adam refinement (smooth max surrogate)
# ----------------------------------------------------------------------


def adam_refine(p_start: np.ndarray,
                n: int,
                fft_len_self: int,
                fft_len_grad: int,
                deadline: float,
                start_time: float,
                best_record: dict,
                rng: np.random.Generator) -> None:
    """
    Adam optimiser on the soft‑max surrogate.
    Updates ``best_record`` (keys: 'J' and 'p') in place.
    """
    a = p_start.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    it = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        if total <= 0:
            break
        frac = elapsed / total

        # learning‑rate schedule (slow decay)
        lr = 0.8 * (0.5 ** frac) + 0.001 * (1 - 0.5 ** frac)

        # temperature schedule – from small to very large
        alpha = 2.0 * (1.0 - frac) + 5000.0 * frac

        # convolution & soft‑max weights
        conv = _self_conv(a, fft_len_self)
        w = _softmax_weights(conv, alpha)

        # gradient of the smoothed surrogate
        grad = _grad_softmax(a, w, fft_len_grad)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto the simplex
        a = proj_simplex(a)

        # true objective (no smoothing)
        cur_J = _objective(a, fft_len_self, n)

        if cur_J < best_record['J'] - 1e-12:
            best_record['J'] = cur_J
            best_record['p'] = a.copy()
            last_improve = now
            print(f"[Adam] {elapsed:.1f}s – new best J = {cur_J:.6f}")

        # occasional jitter to escape plateaus
        if rng.random() < 0.01:
            block = rng.integers(1, max(2, n // 30))
            start_idx = rng.integers(0, n - block + 1)
            factor = np.exp(rng.normal(0.0, 0.08))
            a_pert = a.copy()
            a_pert[start_idx:start_idx + block] *= factor
            a_pert = proj_simplex(a_pert)
            a = a_pert

        # restart on stagnation
        if now - last_improve > 15.0:
            a = best_record['p'].copy()
            noise = np.exp(rng.normal(0.0, 0.04, size=n))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[Adam] {elapsed:.1f}s – restart (stagnation)")


# ----------------------------------------------------------------------
# Active‑set linear‑program refinement
# ----------------------------------------------------------------------


def _build_constraint_matrix(p: np.ndarray, idxs: np.ndarray) -> np.ndarray:
    """
    Build the (|idxs| × n) matrix A where   A[k, j] = p[idxs[k] - j]
    for indices j that keep the argument inside [0, n‑1]; elsewhere zero.
    """
    n = p.size
    K = idxs.size
    rows = np.zeros((K, n), dtype=np.float64)
    for r, k in enumerate(idxs):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1) + 1
        js = np.arange(j_start, j_end)
        rows[r, j_start:j_end] = p[k - js]
    return rows


def lp_refine(p_start: np.ndarray,
              n: int,
              fft_len_self: int,
              deadline: float,
              start_time: float,
              best_record: dict,
              rng: np.random.Generator,
              K_active: int = 30,
              t_mix: float = 0.5) -> None:
    """
    Active‑set linear programme:
        maximise   1ᵀg
        subject to g ≥ 0,
                   (p⋆g)[k] ≤ M  for the K largest k.
    The solution g is normalised to sum‑to‑1 and mixed with p.
    Updates ``best_record`` in place.
    """
    p = p_start.copy()
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        conv = _self_conv(p, fft_len_self)
        M = conv.max()

        # pick the K_active largest constraints (indices)
        K = min(K_active, conv.size)
        top_idx = np.argpartition(conv, -K)[-K:]

        # build the (K × n) constraint matrix
        A = _build_constraint_matrix(p, top_idx)

        # Linear programme (maximise sum(g) -> minimise -sum(g))
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(K, M, dtype=np.float64)
        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception as e:
            print("[LP] linprog exception:", e)
            break

        if not res.success:
            # If the LP fails we abort this refinement branch.
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break
        g /= sum_g  # normalise to a probability vector

        # convex combination with the current point
        p_new = (1.0 - t_mix) * p + t_mix * g
        p_new = proj_simplex(p_new)

        cur_J = _objective(p_new, fft_len_self, n)
        if cur_J < best_record['J'] - 1e-12:
            best_record['J'] = cur_J
            best_record['p'] = p_new.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[LP] {elapsed:.1f}s – new best J = {cur_J:.6f}")

        # stagnation handling (tiny jitter)
        if now - last_improve > 20.0:
            # inject a small random perturbation on a random coordinate
            idx = rng.integers(0, n)
            pert = p_new.copy()
            pert[idx] = max(0.0, pert[idx] + rng.normal(0.0, 0.02))
            pert = proj_simplex(pert)
            p_new = pert
            last_improve = now

        p = p_new


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Uses a two‑phase scheme:
      * cheap global sweep over low‑dimensional families;
      * local refinement via projected Adam and an active‑set LP.
    The best sequence found within the time budget is returned.
    """
    rng = np.random.default_rng(seed)

    # ----------- time handling ------------------------------------------------
    # a short safety margin is kept (≈5 s) to avoid forced termination.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ----------- problem size -------------------------------------------------
    try:
        n = int(len(globals()["height_sequence_1"]))
        n = min(max(n, 256), 4096)  # keep FFT sizes reasonable
    except Exception:
        n = 4096

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # ----------- cheap global exploration ------------------------------------
    best_candidates: list[tuple[float, np.ndarray]] = []

    def _push(p_vec: np.ndarray) -> None:
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > 7:
            best_candidates.pop()

    # uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push(uniform)

    # edge‑shape grid (finer than the previous run)
    for frac in np.linspace(0.10, 0.50, 20):
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 21):
            p = _edge_shape(n, L, gamma)
            _push(p)

    # cosine‑shape grid
    for r in np.linspace(0.0, 1.0, 15):
        for pw in np.linspace(0.3, 3.0, 13):
            p = _cosine_shape(n, r, pw)
            _push(p)

    # random Dirichlet draws (moderately concentrated)
    for _ in range(50):
        alpha = rng.uniform(0.05, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push(p)

    # previous best (if available)
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.sum() > 0.0:
                p_prev = proj_simplex(prev / prev.sum())
                _push(p_prev)
        except Exception:
            pass

    # keep only the top seeds
    best_candidates = sorted(best_candidates, key=lambda x: x[0])[:5]

    # record of the globally best point
    best_record = {
        'J': best_candidates[0][0],
        'p': best_candidates[0][1].copy()
    }
    print(f"[Init] best J after global sweep = {best_record['J']:.6f}")

    # ----------- local refinement (Adam + LP) -------------------------------
    remaining = deadline - time.monotonic()
    if remaining <= 0.0:
        # time ran out before refinement – return what we have
        scale = np.sqrt(2.0 * n)
        return [float(x) for x in np.clip(best_record['p'] * scale, 0.0, 1000.0)]

    per_seed = remaining / max(1, len(best_candidates))

    for seed_J, seed_p in best_candidates:
        if time.monotonic() >= deadline:
            break

        # allocate half of the per‑seed budget to Adam, half to LP
        seed_start = time.monotonic()
        seed_end = seed_start + per_seed
        mid_point = seed_start + per_seed * 0.5

        # -------------------------------------------------------------
        # 1) Adam on the soft‑max surrogate
        # -------------------------------------------------------------
        adam_refine(seed_p, n, fft_len_self, fft_len_grad,
                    deadline=mid_point,
                    start_time=start_time,
                    best_record=best_record,
                    rng=rng)

        # -------------------------------------------------------------
        # 2) Active‑set LP improvement
        # -------------------------------------------------------------
        lp_refine(best_record['p'], n, fft_len_self,
                  deadline=seed_end,
                  start_time=start_time,
                  best_record=best_record,
                  rng=rng,
                  K_active=30,
                  t_mix=0.5)

    # ----------- final scaling ------------------------------------------------
    scale = np.sqrt(2.0 * n)                     # required ℓ¹‑norm
    best_seq = best_record['p'] * scale
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
