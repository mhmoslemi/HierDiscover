def _softmax_max_convolution(a, alpha, fft_len, conv_len):

  Random block perturbations and stagnation restarts avoid local traps.

* **Phase 2 – Exact max‑convolution Adam**  
  Starting from the best vector of Phase 1 we switch to the exact
  sub‑gradient of the true max (`∂max(a⋆a)/∂a`) and continue a projected
  Adam run with a smaller learning‑rate.  This fine‑tunes the solution
  to the original objective.

* Throughout both phases the exact objective
        J(a) = 2·n·max(a⋆a)   (a is already normalised to sum = 1)
  is monitored via FFT (≈ O(n log n)).  The best vector found before the
  time budget expires is returned.  All numbers are clipped to the
  evaluation limits before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator):
    """Diverse seeds, all normalised to sum = 1."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # previous construction if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        if prev.sum() == 0.0:
            prev = np.full_like(prev, 1.0 / n)
        else:
            prev = prev / prev.sum()
        cand.append(prev)

    return cand


def _compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a⋆a) for a already on the simplex."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray, fft_len: int, conv_len: int):
    """Return full self‑convolution, its arg‑max index and the max value."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    idx = int(conv.argmax())
    return conv, idx, float(conv.max())


def _subgradient_max(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(a⋆a) at the (any) arg‑max position `idx`.
    For all i with 0 ≤ i < n and 0 ≤ idx‑i < n we have
        ∂b_idx/∂a_i = 2·a_{idx‑i},
    otherwise the derivative is zero.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(idx, n - 1)            # inclusive
    i_range = np.arange(i_start, i_end + 1)
    j_range = idx - i_range
    g[i_range] = 2.0 * a[j_range]
    return g


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute soft‑max weights p_k = exp(α·c_k)/Σ exp(α·c_k) in a numerically
    stable way (c = conv).
    """
    max_c = conv.max()
    exp_term = np.exp(alpha * (conv - max_c))
    total = exp_term.sum()
    if total == 0.0:
        # should not happen, fallback to uniform
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / total


def _gradient_softmax(a: np.ndarray, p: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Exact gradient of the soft‑max surrogate Lα(a) = Σ_k p_k·(a⋆a)_k.
    Returns a vector of length n.
    """
    n = a.size
    # reverse the probability vector
    p_rev = p[::-1]

    # FFTs (both of length fft_len)
    fa = np.fft.rfft(a, n=fft_len)
    fp = np.fft.rfft(p_rev, n=fft_len)

    # convolution a ⋆ p_rev  (full length = n + conv_len - 1 = 3n-2)
    grad_full = np.fft.irfft(fa * fp, n=fft_len)[: n + conv_len - 1]

    # extract the needed slice and reverse
    seg = grad_full[n - 1 : 2 * n - 1]          # length n
    grad = 2.0 * seg[::-1]                     # sub‑gradient of Lα
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The method consists of:

    1. Global warm‑start with a small pool of hand‑crafted candidates.
    2. Phase 1 – smooth‑max Adam optimisation (soft‑max temperature
       annealing) with symmetrisation and simplex projection.
    3. Phase 2 – exact max‑convolution Adam refinement (same machinery as
       the reference implementation but starting from the improved
       candidate).

    The routine respects the 1000 s budget (5 s safety margin) and
    returns the best sequence it found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Timing
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # 1) Problem size (power of two for fast FFT)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n0 = len(height_sequence_1)
        n = max(512, min(n0, 4096))
    else:
        n = 4096
    n = _next_pow2(n)                # force power of two (≤ 4096)
    conv_len = 2 * n - 1
    # FFT length large enough for a⋆a (2n‑1) and for a⋆p_rev (3n‑2)
    fft_len = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    # 2) Initialise diverse candidates, keep the best on the exact J
    # ------------------------------------------------------------------
    candidates = _initial_candidates(n, rng)
    best_vec = None
    best_score = np.inf
    for cand in candidates:
        cand = proj_simplex(cand.astype(np.float64))
        sc = _compute_J(cand, n, fft_len, conv_len)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    print(f"[{time.monotonic() - start_time:6.2f}s] Initial best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – Smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.70 * (deadline - start_time)
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # schedule parameters
    alpha_start, alpha_end = 0.5, 80.0          # temperature (soft‑max)
    lr_start, lr_end = 0.5, 0.001               # learning‑rate

    it = 0
    last_improve = start_time

    while time.monotonic() < phase1_deadline:
        it += 1
        now = time.monotonic()
        # fraction of Phase 1 elapsed
        frac = (now - start_time) / (phase1_deadline - start_time)

        # exponential schedule for α, linear for lr
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # ----- compute convolution and soft‑max weights -----
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]

        p = _softmax_weights(conv, alpha)                     # soft‑max probs

        # ----- gradient of smooth surrogate -----
        grad = _gradient_softmax(a, p, fft_len, conv_len)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # symmetrise (a ← (a + reverse(a))/2) – reduces unnecessary asymmetry
        a_new = (a_new + a_new[::-1]) * 0.5

        # projection onto simplex
        a_new = proj_simplex(a_new)

        # ----- evaluate exact objective (cheap via FFT) -----
        cur_J = _compute_J(a_new, n, fft_len, conv_len)
        if cur_J < best_score - 1e-12:
            best_score = cur_J
            best_vec = a_new.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Phase1 new best J = {best_score:.6f}")

        a = a_new

        # occasional random block jitter (helps escape plateaus)
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # restart if stagnated for 30 s
        if now - last_improve > 30.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Phase1 restart – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – Exact max‑convolution Adam refinement
    # ------------------------------------------------------------------
    now = time.monotonic()
    remaining = deadline - now
    if remaining <= 0.0:
        # out of time – return the best we have
        best_vec = np.clip(best_vec, 0.0, 1000.0)
        return [float(x) for x in best_vec]

    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # learning‑rate schedule for the fine‑tuning phase
    lr_start_f, lr_end_f = 0.28, 0.001
    it_adam = 0
    last_improve = now

    # reuse the same fft_len for consistency
    while time.monotonic() < deadline:
        it_adam += 1
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        lr = lr_start_f * (1.0 - frac) + lr_end_f * frac

        # ----- exact max‑convolution and its sub‑gradient -----
        conv, idx_max, _ = _conv_and_max(a, fft_len, conv_len)
        grad = _subgradient_max(a, idx_max)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it_adam)
        v_hat = v / (1.0 - beta2 ** it_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # symmetrisation (kept for consistency) and projection
        a_new = (a_new + a_new[::-1]) * 0.5
        a_new = proj_simplex(a_new)

        cur_J = _compute_J(a_new, n, fft_len, conv_len)
        if cur_J < best_score - 1e-12:
            best_score = cur_J
            best_vec = a_new.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Phase2 new best J = {best_score:.6f}")

        a = a_new

        # occasional jitter
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # stagnation restart (20 s)
        if now - last_improve > 20.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Phase2 restart – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
