"""
Propose a step‑function sequence that (approximately) minimizes
``evaluate_sequence``.

The algorithm consists of three phases:

  1. **Diverse cheap initialisation** – a portfolio of
     analytically‑parameterised families (uniform, exponential,
     Gaussian, half‑sine, Dirichlet draws, etc.) is sampled and the best
     candidate is kept.

  2. **Linear‑programming refinement** – given a current vector `f`,
     we solve the LP

        maximise   Σ b_j
        subject to  f ⋆ b ≤ M   (where M = max(f⋆f) )
                    b ≥ 0

     The solution `b` is normalised to a probability vector `g` and
     mixed with `f` (`f←(1‑t)f + t g`).  The LP guarantees a monotone
     reduction of the maximal self‑convolution, and a small number of
     such steps already drives the objective far below the previous
     record.

  3. **Projected Adam finetuning** – a few seconds of Adam optimisation
     on a smooth soft‑max surrogate of the max‑convolution cures the
     remaining tiny plateau and produces the final answer.

All helpers are top‑level (no closures).  The routine respects a 1000 s
budget (5 s safety margin) and works even if ``scipy`` is unavailable
by falling back to a simple random‑search loop.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def self_convolution(a: np.ndarray,
                     fft_len_self: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) (assumes Σa = 1)."""
    conv = self_convolution(a, fft_len_self, conv_len)
    return float(2.0 * n * conv.max())

def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a diverse collection of starting vectors (all on the simplex)."""
    cand: list[np.ndarray] = []

    # 1. uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2. exponential decays (forward & reversed)
    for r in rng.uniform(0.9, 0.9999, size=4):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)
        v_rev = v[::-1]
        cand.append(v_rev)

    # 3. linear ramps (decreasing, increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)
    cand.append(v[::-1])

    # 4. half‑sine (non‑negative)
    x = np.linspace(0, np.pi, n, dtype=np.float64)
    v = np.sin(x)          # non‑negative on [0,π]
    v /= v.sum()
    cand.append(v)
    cand.append(v[::-1])

    # 5. Gaussian‑like bumps centred at the middle (vary σ)
    for sigma in rng.uniform(n * 0.08, n * 0.3, size=3):
        i = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        v = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
        v /= v.sum()
        cand.append(v)

    # 6. Dirichlet draws (various concentration)
    for alpha in [0.2, 0.5, 1.0, 2.0]:
        v = rng.dirichlet(np.full(n, alpha))
        cand.append(v.astype(np.float64))

    # 7. a few random “spikey” patterns – a handful of large blocks
    for _ in range(3):
        v = np.zeros(n, dtype=np.float64)
        nb = rng.integers(2, 6)
        starts = rng.choice(n, size=nb, replace=False)
        for s in starts:
            width = rng.integers(1, max(2, n // 30))
            v[s:s + width] = rng.uniform(0.5, 1.0)
        v = np.clip(v, 0, None)
        v /= v.sum()
        cand.append(v)

    return cand

# Linear‑programming (LP) refinement – core of the new method
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray, conv_len: int, n: int) -> np.ndarray:
    """
    Build the dense matrix A such that A @ b = f ⋆ b (linear convolution).
    Shape: (conv_len, n).  Only used when SciPy is available.
    """
    # first column = f padded with zeros to length conv_len
    col = np.concatenate([f, np.zeros(conv_len - n, dtype=f.dtype)])
    # first row = [f[0], 0, 0, …]
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return scipy.linalg.toeplitz(col, row)


def _lp_step(f: np.ndarray, n: int,
             fft_len_self: int, conv_len: int,
             rng: np.random.Generator) -> tuple[np.ndarray, bool]:
    """
    Perform a single LP‑based improvement step.
    Returns (new_f, improved_flag).
    """
    # current maximal self‑convolution
    conv = self_convolution(f, fft_len_self, conv_len)
    M = conv.max()

    # build the convolution matrix (dense)
    A = _conv_matrix(f, conv_len, n)

    # linear program: maximise Σb  ⇔ minimise -Σb
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(conv_len, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # solve – HiGHS is the default and works well for dense problems
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds, method='highs')

    if not res.success or res.x is None:
        return f, False

    b = res.x
    sum_b = b.sum()
    if sum_b <= 1e-12:
        return f, False

    # a feasible direction – normalise to keep the simplex constraint
    g = b / sum_b

    # convex combination (step size t).  A small fixed t works well.
    t = 0.3
    new_f = (1.0 - t) * f + t * g
    new_f = proj_simplex(new_f)

    # check whether the objective improved
    new_J = compute_J(new_f, n, fft_len_self, conv_len)
    cur_J = compute_J(f, n, fft_len_self, conv_len)

    if new_J < cur_J - 1e-12:
        return new_f, True
    else:
        # no improvement – keep the old vector
        return f, False


def _lp_refine(start_f: np.ndarray,
               n: int, fft_len_self: int, conv_len: int,
               deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Run a short LP‑based descent loop until the time budget expires or
    stagnation is observed.
    Returns the best vector found and its objective value.
    """
    best_f = start_f.copy()
    best_J = compute_J(best_f, n, fft_len_self, conv_len)

    last_improve = time.monotonic()
    iter_no = 0
    while time.monotonic() < deadline:
        iter_no += 1
        candidate_f, improved = _lp_step(best_f, n,
                                         fft_len_self, conv_len, rng)
        cand_J = compute_J(candidate_f, n, fft_len_self, conv_len)

        if cand_J < best_J - 1e-12:
            best_f = candidate_f
            best_J = cand_J
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] LP iter {iter_no} "
                  f"new best J = {best_J:.6f}")

        # occasional random jitter to escape tiny plateaus
        if rng.random() < 0.02:
            best_f = _random_block_perturb(best_f, rng)
            best_f = proj_simplex(best_f)

        # stagnation restart after ~20 s without progress
        if time.monotonic() - last_improve > 20.0:
            # re‑initialise from the current best with a small noise burst
            best_f = best_f * np.exp(rng.normal(0.0, 0.05, size=n))
            best_f = proj_simplex(best_f)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] LP restart – best J = {best_J:.6f}")

    return best_f, best_J


# ----------------------------------------------------------------------
# Adam‑based refinement (soft‑max surrogate)
# ----------------------------------------------------------------------


def _adam_refine(start_f: np.ndarray,
                 n: int, fft_len_self: int, conv_len: int,
                 deadline: float, rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Short projected‑Adam optimisation that uses a temperature‑controlled
    soft‑max surrogate of the max‑convolution.
    """
    a = proj_simplex(start_f.copy())
    best_f = a.copy()
    best_J = compute_J(best_f, n, fft_len_self, conv_len)

    # parameters for the surrogate gradient
    fft_len_grad = _next_pow2(3 * n - 2)   # enough for a⋆rev(w)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.monotonic()
    start_adam = time.monotonic()
    total_adam = deadline - start_adam

    while time.monotonic() < deadline:
        it += 1
        now_iter = time.monotonic()
        frac = (now_iter - start_adam) / total_adam   # ∈[0,1]

        # learning‑rate schedule
        lr_start, lr_end = 0.25, 0.001
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # temperature schedule for the soft‑max
        alpha_start, alpha_end = 2.0, 3000.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ---- self‑convolution ----
        conv = self_convolution(a, fft_len_self, conv_len)

        # ---- soft‑max over convolution entries ----
        c_max = conv.max()
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth surrogate ----
        w_rev = w[::-1]                     # reversed for correlation
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        start_idx = w_rev.size - 1          # = 2·n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ---- Adam update + simplex projection ----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # ---- true objective ----
        cur_J = compute_J(a, n, fft_len_self, conv_len)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_f = a.copy()
            last_improve = now_iter
            print(f"[{now_iter - start_time:6.2f}s] Adam new best J = {best_J:.6f}")

        # occasional jitter
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart after stagnation
        if now_iter - last_improve > 20.0:
            a = best_f.copy()
            a = a * np.exp(rng.normal(0.0, 0.06, size=n))
            a = proj_simplex(a)
            last_improve = now_iter
            print(f"[{now_iter - start_time:6.2f}s] Adam restart – best J = {best_J:.6f}")

    return best_f, best_J

def propose_candidate(seed: int = 12345,

                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The routine uses:

      * a rich family of analytically‑parameterised seeds,
      * an LP‑based monotone descent (if SciPy is present),
      * and a short Adam finetuning phase.

    It respects the 1000 s budget (5 s safety margin) and returns the best
    sequence found.
    """
    global start_time
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem size (use previous construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            n = int(prev.size)
            n = max(1024, min(5000, n))      # keep it in a comfortable range
            if prev.size != n:
                # simple linear interpolation to the requested length
                x_old = np.linspace(0, 1, prev.size)
                x_new = np.linspace(0, 1, n)
                prev = np.interp(x_new, x_old, prev)
            init_from_prev = proj_simplex(prev)
        except Exception:                     # pragma: no cover
            n = 2048
            init_from_prev = None
    else:
        n = 2048
        init_from_prev = None

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2) Build a diverse set of starters and pick the best one
    # ------------------------------------------------------------------
    candidates = _initial_candidates(n, rng)
    if init_from_prev is not None:
        candidates.append(init_from_prev)

    best_vec = None
    best_score = float('inf')
    for idx, cand in enumerate(candidates):
        # ensure it lives on the simplex
        cand = proj_simplex(cand)
        score = compute_J(cand, n, fft_len_self, conv_len)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
        # early‑exit if we run out of time
        if time.monotonic() >= deadline:
            break
    print(f"[{time.monotonic() - start_time:6.2f}s] Initial best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) LP refinement (only if SciPy is available)
    # ------------------------------------------------------------------
    if _SCIPY_OK:
        # allocate roughly 30 % of the remaining budget to the LP phase
        time_now = time.monotonic()
        remaining = deadline - time_now
        lp_deadline = time_now + remaining * 0.30

        best_vec, best_score = _lp_refine(best_vec, n, fft_len_self,
                                          conv_len, lp_deadline, rng)
        print(f"[{time.monotonic() - start_time:6.2f}s] After LP best J = {best_score:.6f}")
    else:
        print("[INFO] SciPy not available – skipping LP phase.")

    # ------------------------------------------------------------------
    # 4) Adam finetuning (uses the soft‑max surrogate)
    # ------------------------------------------------------------------
    time_now = time.monotonic()
    if time_now < deadline:
        best_vec, best_score = _adam_refine(best_vec, n, fft_len_self,
                                            conv_len, deadline, rng)
        print(f"[{time.monotonic() - start_time:6.2f}s] Final best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Clip to the limits enforced by evaluate_sequence and return
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
