import time, numpy as np, scipy.optimize

def propose_candidate(seed=0, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # decide n
    default_n = 4096
    if 'height_sequence_1' in globals():
        n0 = len(height_sequence_1)
        n = max(1024, min(n0*2, 8192))  # maybe use double length for better
    else:
        n = default_n

    # we want n to be power of two for FFT speed
    n = 1 << ((n-1).bit_length()) # next power of two

    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len-1).bit_length())  # next pow2

    # Pre-allocated arrays for FFT maybe reuse ?

    # define function to compute J for a vector a (already normalized)
    def compute_J(a):
        # Ensure shape (n,)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        maxb = conv.max()
        return float(2.0 * n * maxb)

    # Parameterized shape generator
    K = 6
    def construct_a(params):
        # params length 3*K
        mus = params[:K]
        sigmas = params[K:2*K]
        alphas = params[2*K:3*K]
        # enforce bounds: mus in [0,n-1], sigmas >0
        # Already ensured by bounds in DE
        i = np.arange(n)
        a = np.zeros(n)
        weights = np.exp(alphas - np.max(alphas))
        weights /= weights.sum()
        for k in range(K):
            a += weights[k] * np.exp(-0.5 * ((i - mus[k]) / sigmas[k])**2)
        total = a.sum()
        if total == 0:
            return np.full(n, 1.0/n)
        return a / total

    # Objective for DE: compute J for parameters
    def obj_params(params):
        a = construct_a(params)
        return compute_J(a)

    # DE bounds
    bounds = []
    for _ in range(K):
        bounds.append((0, n-1))          # mu
    for _ in range(K):
        bounds.append((1.0, n/2.0))      # sigma
    for _ in range(K):
        bounds.append((-5.0, 5.0))       # alpha

    # set DE options
    max_iter = 1000  # will be limited by time
    popsize = 15

    # callback to stop if time
    def de_callback(xk, convergence):
        if time.monotonic() > deadline:
            return True

    result = scipy.optimize.differential_evolution(
        obj_params,
        bounds,
        seed=seed,
        strategy='best1bin',
        popsize=popsize,
        maxiter=max_iter,
        init='latinhypercube',
        polish=False,
        mutation=(0.5, 1),
        recombination=0.7,
        callback=de_callback,
        disp=False,
        tol=1e-6
    )

    best_params = result.x
    best_a = construct_a(best_params)
    best_score = compute_J(best_a)

    print(f"[{time.monotonic()-start_time:.2f}s] DE best J = {best_score:.6f}")

    # Phase 2: refine via projected Adam with smoothed max (log-sum-exp)
    # Use the optimum a as starting point.
    a = best_a.copy()
    # align sum to 1 (already)
    # Precompute indices for gradient of smoothed max
    # We'll choose lambda = 50 initially, then increase.

    # compute conv and gradient using FFT.
    # We'll implement function to compute smoothed objective and gradient.
    def smooth_objective_and_grad(a, lam):
        # a shape (n,)
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]   # real, length conv_len
        # compute smooth max: logsumexp
        max_val = np.max(conv)
        # For numeric stability:
        exp_term = np.exp(lam * (conv - max_val))
        denom = exp_term.sum()
        weights = exp_term / denom  # weights sum to 1
        # smoothed max value:
        smooth_max = (np.log(denom) / lam) + max_val
        # gradient: grad = 2 * (weights convolve a) (cross-correlation)
        # compute convolution of weights and a (cross-correlation)
        # Equivalent to convolution of weights with a reversed?
        # Let's compute using FFT: cross-correlation = irfft(rfft(weights, n=fft_len) * np.conj(rfft(a, n=fft_len)))? Wait need careful.
        # But easier: compute g_i = Σ_j weights[j] * a[j-i] for each i where indices valid.

        # Let A = rfft(a, n=fft_len)
        # Let W = rfft(weights, n=fft_len)
        # cross-corr = irfft(np.conj(A) * W, n=fft_len) # because correlation = IFFT( conj(A) * W )

        # So g = 2 * cross_corr

        grad = 2.0 * cross_corr
        return smooth_max, grad

    # Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

        # compute gradient of smooth max
        smax, grad = smooth_objective_and_grad(a, lam)

        grad_J = 2.0 * n * grad

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_J
        v = beta2 * v + (1 - beta2) * (grad_J * grad_J)
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)

        # compute learning rate schedule linear dec.
        frac = (time.monotonic() - start_time) / (budget_s - 5.0)
        lr = lr_start * (1 - frac) + lr_end * frac

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            print(f"[{last_improve - start_time:.2f}s] Smooth Adam new best J = {best_score:.6f}")

        a = a_new

        if time.monotonic() - last_improve > 30.0:

            # add random jitter
            a = _random_block_perturb(a, rng)
            m[:] = 0
            v[:] = 0
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:.2f}s] Smooth Adam restart – best J = {best_score:.6f}")

    # After time limit, return best_vec as list
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]

"""
Propose a step‑function that (hopefully) beats the previous J≈1.5132.

Algorithm
---------
1. Choose a problem size *n* (use the length of a previously found
   sequence if available, otherwise a power‑of‑two ≈4000 for fast FFT).

2. Global search in a low‑dimensional “mixture‑of‑Gaussians” family
   using SciPy’s differential‑evolution (DE).  A candidate is
   characterised by K means, K scales and K log‑weights → a dense,
   non‑negative vector that is normalised to sum 1.  DE is purely
   black‑box, tolerates the non‑smooth max‑convolution objective and
   quickly discovers a promising shape.

3. Refine that shape with a projected Adam optimiser that works on a
   *smooth* surrogate of the max‑convolution:
       Sλ(a) = (1/λ)·log∑exp(λ·(a⋆a))
   Gradient of Sλ can be computed exactly via FFT
   (cross‑correlation of a with the soft‑max weights).  λ is increased
   gradually, turning the surrogate into the true max.

4. Final polish: a few hundred Adam steps on the *exact* max‑convolution
   (sub‑gradient) – exactly the same update as the reference code but
   starting from a far better region.

The whole routine respects the 1000 s budget (5‑second safety margin)
and prints intermediate J‑values.  The returned list is already clipped
to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative components that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine consists of:
      • a global DE search in a 3K‑parameter Gaussian‑mixture family,
      • a smooth‑max Adam refinement,
      • a final exact‑max Adam polish.
    It returns the best coefficient list discovered before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size ---------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n0 = len(height_sequence_1)
        # use a power of two close to the existing size (speed‑up FFT)
        n = 1 << ((max(1024, n0) - 1).bit_length())
    else:
        n = 4096                     # default power of two

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)   # length for zero‑padded FFT

    # ------------------------------------------------------------------
    # 2) Objective helper (exact J = 2·n·max(a⋆a)) -----------------------
    # ------------------------------------------------------------------
    def _compute_J(a: np.ndarray) -> float:
        """Exact objective J for a normalised vector a (sum = 1)."""
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 3) Gaussian‑mixture parametrisation --------------------------------
    # ------------------------------------------------------------------
    K = 6                                  # number of Gaussian components

    def _construct_a(params: np.ndarray) -> np.ndarray:
        """Build a normalised, non‑negative sequence from 3K parameters."""
        mus = params[:K]                   # means  ∈ [0, n-1]
        sigs = params[K:2 * K]             # scales ∈ [1, n/2]
        alphas = params[2 * K:]            # log‑weights (unbounded)
        # log‑softmax for the mixture weights
        max_alpha = np.max(alphas)
        w = np.exp(alphas - max_alpha)
        w /= w.sum()
        i = np.arange(n, dtype=np.float64)
        a = np.zeros(n, dtype=np.float64)
        for k in range(K):
            # Gaussian – safe for very small sigma because we bound it ≥1
            a += w[k] * np.exp(-0.5 * ((i - mus[k]) / sigs[k]) ** 2)
        total = a.sum()
        if total == 0.0:
            return np.full(n, 1.0 / n)
        return a / total                     # project onto simplex automatically

    # ------------------------------------------------------------------
    # 4) Global DE search (black‑box) ------------------------------------
    # ------------------------------------------------------------------
    # Try importing SciPy; if unavailable fall back to a cheap random search.
    try:
        from scipy.optimize import differential_evolution
        DE_AVAILABLE = True
    except Exception:          # pragma: no cover
        DE_AVAILABLE = False

    if DE_AVAILABLE:
        # bounds for the 3K parameters
        bounds = []
        for _ in range(K):
            bounds.append((0.0, float(n - 1)))          # μ
        for _ in range(K):
            bounds.append((1.0, float(n) / 2.0))        # σ
        for _ in range(K):
            bounds.append((-5.0, 5.0))                  # α (log‑weight)

        # objective for DE – directly the exact J
        def _de_obj(params):
            a = _construct_a(params)
            return _compute_J(a)

        # callback that aborts DE when we run out of time
        def _de_callback(xk, convergence):
            if time.monotonic() > deadline:
                return True    # stop DE
            return False

        # a relatively aggressive budget: we let DE consume ≈30 % of the time
        de_start = time.monotonic()
        remaining = deadline - de_start
        # we cap the number of generations so that we stay well within the budget
        max_generations = 200
        popsize = 8

        result = differential_evolution(
            _de_obj,
            bounds,
            seed=seed,
            strategy='best1bin',
            popsize=popsize,
            maxiter=max_generations,
            init='latinhypercube',
            polish=False,
            mutation=(0.5, 1.0),
            recombination=0.7,
            callback=_de_callback,
            disp=False,
            tol=1e-6
        )
        best_params = result.x
        best_a = _construct_a(best_params)
        best_score = _compute_J(best_a)
        print(f"[{time.monotonic() - start_time:6.2f}s] DE best J = {best_score:.6f}")

    else:   # -----------------------------------------------------------------
        # Simple random search (fallback) ------------------------------------
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _compute_J(best_a)
        trials = 0
        while time.monotonic() < deadline:
            # random Gaussian‑mixture parameters
            mus = rng.uniform(0, n - 1, K)
            sigs = rng.uniform(1.0, n / 2.0, K)
            alphas = rng.uniform(-5, 5, K)
            params = np.concatenate([mus, sigs, alphas])
            a = _construct_a(params)
            sc = _compute_J(a)
            trials += 1
            if sc < best_score:
                best_score = sc
                best_a = a
                print(f"[{time.monotonic() - start_time:6.2f}s] rand‑search new best J = {best_score:.6f}")
                # stop early if we already beat the target
                if best_score < 1.5025:
                    break
        print(f"[{time.monotonic() - start_time:6.2f}s] Random search done, trials={trials}")

    # ------------------------------------------------------------------
    # 5) Warm‑start with a refined Adam on a *smooth* max ----------------
    # ------------------------------------------------------------------
    # time left after the global phase
    if time.monotonic() >= deadline:
        best_a = np.clip(best_a, 0.0, 1000.0)
        return [float(x) for x in best_a]

    a = best_a.copy()
    m = np.zeros_like(a)      # Adam first moment
    v = np.zeros_like(a)      # Adam second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it_adam = 0
    last_improve = time.monotonic()
    # learning‑rate schedule (linear decay)
    lr_start, lr_end = 0.25, 0.001
    # start smoothing parameter λ and let it grow slowly
    lam = 8.0

    # pre‑allocated FFT buffers (reuse to avoid repeated allocations)
    def _smooth_max_and_grad(x: np.ndarray, lam_val: float):
        """Return smooth max of conv(x,x) and its exact gradient."""
        # forward FFT of x
        X = np.fft.rfft(x, n=fft_len)
        conv = np.fft.irfft(X * X, n=fft_len)[:conv_len]   # a⋆a
        # log‑sum‑exp smoothing
        max_c = conv.max()
        exp_shift = np.exp(lam_val * (conv - max_c))
        Z = exp_shift.sum()
        weights = exp_shift / Z                              # soft‑max weights
        smooth_max = max_c + np.log(Z) / lam_val

        # gradient of smooth max:
        #   ∂conv_i/∂x = 2 * shift_i(x)   (see analysis)
        #   grad = Σ_i w_i * ∂conv_i/∂x = 2 * (cross‑correlation of x and w)
        W = np.fft.rfft(weights, n=fft_len)
        # cross‑correlation via conj(FFT(x)) * FFT(w)
        cross = np.fft.irfft(np.conj(X) * W, n=fft_len)[:n]
        grad = 2.0 * cross
        return smooth_max, grad

    while time.monotonic() < deadline:
        it_adam += 1
        # increase λ slowly so we move from a very smooth surrogate to the true max
        lam = min(200.0, lam * 1.02)

        smooth_val, grad = _smooth_max_and_grad(a, lam)
        # objective J = 2·n·smooth_max  (the smooth version approximates the true J)
        grad_J = 2.0 * n * grad

        # Adam step with linear learning‑rate decay
        frac = (time.monotonic() - start_time) / (budget_s - 5.0)
        lr = lr_start * (1.0 - frac) + lr_end * frac

        m = beta1 * m + (1.0 - beta1) * grad_J
        v = beta2 * v + (1.0 - beta2) * (grad_J * grad_J)
        m_hat = m / (1.0 - beta1 ** it_adam)
        v_hat = v / (1.0 - beta2 ** it_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        new_score = _compute_J(a_new)

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_new.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Smooth‑Adam new best J = {best_score:.6f}")

        a = a_new

        # occasional random jitter to help escape plateaus
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # restart if we have been stagnant for a while
        if time.monotonic() - last_improve > 30.0:
            a = best_a.copy()
            a = _random_block_perturb(a, rng)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Smooth‑Adam restart – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Final polishing with the exact max‑subgradient (optional) ------
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:          # may have some time left
        a = best_a.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it_adam = 0
        last_improve = time.monotonic()
        # learning‑rate schedule for the exact phase
        lr_start2, lr_end2 = 0.18, 0.001

        while time.monotonic() < deadline:
            it_adam += 1
            conv, idx_max, _ = _conv_and_max(a, fft_len, conv_len)  # will be defined below
            # exact sub‑gradient at the (any) arg‑max position
            g = _subgradient_max(a, idx_max)

            # Adam update on the *true* objective (J = 2*n*max(conv))
            grad_J = 2.0 * n * g

            frac = (time.monotonic() - start_time) / (budget_s - 5.0)
            lr = lr_start2 * (1.0 - frac) + lr_end2 * frac

            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J * grad_J)
            m_hat = m / (1.0 - beta1 ** it_adam)
            v_hat = v / (1.0 - beta2 ** it_adam)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = proj_simplex(a_new)

            new_score = _compute_J(a_new)
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start_time:6.2f}s] Exact‑Adam new best J = {best_score:.6f}")

def _conv_and_max(a: np.ndarray, fft_len: int, conv_len: int):
    """Return full self‑convolution, index of its maximum and the max value."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    idx = int(conv.argmax())
    return conv, idx, float(conv.max())


def _subgradient_max(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(a⋆a) at a given arg‑max position.
    For each i, ∂ (a⋆a)[idx] / ∂ a_i = 2·a_{idx‑i} when the index is valid,
    otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(idx, n - 1)               # inclusive
    i_range = np.arange(i_start, i_end + 1)
    j_range = idx - i_range
    g[i_range] = 2.0 * a[j_range]
    return g
