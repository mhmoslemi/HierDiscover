The algorithm therefore repeatedly

* computes the convolution (FFT, O(n log n)),
* builds the sub‑gradient from the indices where the convolution is maximal,
* takes a projected step on the probability simplex,
* accepts the step only if the true objective improves,
* occasionally injects a small random perturbation to escape plateaux,
* restarts from the current best after a prolonged lack of improvement.

The whole routine runs for the allotted time budget (default 1000 s, 5 s safety
margin) and finally returns the best normalised vector found.  The returned
list is ready for ``evaluate_sequence`` – the scaling invariance of the
objective guarantees that the exact numeric scale does not matter.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Return a*a (full discrete convolution) using FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective J = 2·n·max(conv) (the L¹‑norm of a is assumed to be 1)."""
    return float(2.0 * n * conv.max())


def subgradient_max_convolution(
    a: np.ndarray,
    conv: np.ndarray,
    idx: np.ndarray,
    rel_tol: float = 1e-12,
    max_ks: int = 20,
) -> np.ndarray:
    """
    Sub‑gradient of M = max(conv) w.r.t. a.

    For each index k where conv[k] is (almost) maximal we add the shifted
    vector a_{k‑i} (zero outside the support).  The final sub‑gradient is the
    average over the selected k‑indices.
    """
    n = a.size
    max_val = conv.max()
    eps = max(rel_tol * max_val, 1e-15)
    ks = np.where(conv >= max_val - eps)[0]
    if ks.size > max_ks:
        ks = ks[:max_ks]

    grad = np.zeros_like(a)
    for k in ks:
        j = k - idx                     # j = k‑i
        mask = (j >= 0) & (j < n)
        grad[mask] += a[j[mask]]
    grad /= ks.size
    return grad


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method combines a projected sub‑gradient descent with occasional random
    perturbations and a simple stagnation‑restart scheme.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0. Problem size – reuse the length of the previously discovered sequence
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))          # keep FFT cheap
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 1. Starting vector on the probability simplex
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size)])
        elif v.size > n:
            v = v[:n]
        s = v.sum()
        if s == 0.0:
            cur_vec = np.full(n, 1.0 / n)
        else:
            cur_vec = v / s
    else:
        cur_vec = np.full(n, 1.0 / n)

    cur_vec = proj_simplex(cur_vec)            # guarantee feasibility

    # --------------------------------------------------------------
    # 2. Pre‑compute static data
    # --------------------------------------------------------------
    idx = np.arange(n, dtype=np.int64)        # used in sub‑gradient computation

    # --------------------------------------------------------------
    # 3. Initial objective
    # --------------------------------------------------------------
    conv = compute_convolution(cur_vec, fft_len, conv_len)
    cur_J = compute_J_from_conv(conv, n)
    best_vec = cur_vec.copy()
    best_J = cur_J
    print(f"[0.00s] initial J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start

    iteration = 0
    # linear temperature for occasional Metropolis‑type acceptance
    T_start = cur_J * 0.05
    T_end = 1e-8

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / (deadline - start)

        # temperature (only used for the cheap random perturbations)
        T = T_start * (1.0 - frac) + T_end * frac

        # ----------------------------------------------------------
        # a) deterministic sub‑gradient step
        # ----------------------------------------------------------
        subgrad = subgradient_max_convolution(cur_vec, conv, idx)

        # step size schedule – start fairly aggressive, then decay
        step = 0.5 * (1.0 - frac) + 0.001   # between 0.5 and 0.001
        trial_vec = proj_simplex(cur_vec - step * subgrad)
        trial_conv = compute_convolution(trial_vec, fft_len, conv_len)
        trial_J = compute_J_from_conv(trial_conv, n)

        if trial_J < cur_J:
            # clear improvement
            cur_vec, cur_J, conv = trial_vec, trial_J, trial_conv
            if trial_J < best_J - 1e-12:
                best_J, best_vec = trial_J, trial_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best J = {best_J:.6f}")

        else:
            # ------------------------------------------------------
            # b) occasional stochastic perturbation (Metropolis rule)
            # ------------------------------------------------------
            if rng.random() < 0.03:
                # multiplicative log‑normal noise
                noise = np.exp(rng.normal(0.0, 0.08, size=n))
                pert_vec = proj_simplex(cur_vec * noise)
                pert_conv = compute_convolution(pert_vec, fft_len, conv_len)
                pert_J = compute_J_from_conv(pert_conv, n)

                if pert_J < cur_J or rng.random() < np.exp(-(pert_J - cur_J) / max(T, 1e-12)):
                    cur_vec, cur_J, conv = pert_vec, pert_J, pert_conv
                    if pert_J < best_J - 1e-12:
                        best_J, best_vec = pert_J, pert_vec.copy()
                        last_improve = now
                        print(f"[{elapsed:6.2f}s] new best (perturb) J = {best_J:.6f}")

        # ----------------------------------------------------------
        # c) stagnation handling – restart in the neighbourhood
        # ----------------------------------------------------------
        if now - last_improve > 20.0:
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = proj_simplex(best_vec * jitter)
            conv = compute_convolution(cur_vec, fft_len, conv_len)
            cur_J = compute_J_from_conv(conv, n)
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart after stagnation – best J = {best_J:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best discovered vector (still normalised to sum=1)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec = best_vec / s
    return [float(x) for x in best_vec]
