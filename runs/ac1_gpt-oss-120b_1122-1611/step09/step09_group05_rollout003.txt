"""
Propose a step‑function by smooth‑max gradient descent (Adam).

The true objective is   J(a) = 2·n·max(conv(a,a))   (∑a = 1).
We approximate the max with a log‑sum‑exp:

    J_α(a) = 2·n·(1/α)·log Σ_i exp(α·conv_i)

For a fixed α the function is smooth and its gradient can be written
in closed form.  Using the soft‑max weights  
    w_i = exp(α·conv_i) / Σ_j exp(α·conv_j)

the gradient w.r.t. the probability vector a is  

    ∇_a J_α = 2·Σ_i w_i·∂conv_i/∂a
            = 2·(w ⋆ a)                     (⋆ = linear correlation)

The correlation is obtained efficiently by a single FFT.  We optimise
α from a small value (smooth landscape) up to a large value (tight
approximation to the true max) while applying Adam on the logits
z with a = softmax(z).  Periodic restarts after 30 s without
improvement keep the search from stagnating.

The routine respects the 1000 s budget, returns the best vector found,
and works directly on the true objective (no external surrogate).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Returns the full convolution.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * a.size - 1]


def _correlate(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear correlation (valid part) w ⋆ a  = Σ_k w[i+k]·a[k],
    for i = 0 … len(w)-len(a).  Result length = len(w)-len(a)+1.
    Implemented via FFT: ifft(fft(w)·conj(fft(a))).
    """
    W = np.fft.rfft(w, n=fft_len)
    A = np.fft.rfft(a, n=fft_len)
    corr = np.fft.irfft(W * np.conj(A), n=fft_len)
    # valid part starts at lag 0
    return corr[: a.size]  # length = len(w)-len(a)+1 = len(a)


def _objective_J(a: np.ndarray) -> float:
    """Exact objective J = 2·n·max(conv(a,a)) (∑a = 1)."""
    n = a.size
    conv = np.convolve(a, a)                # small n, direct is fine
    return 2.0 * n * conv.max()


def _smooth_objective_and_grad(
    a: np.ndarray,
    alpha: float,
    fft_len_conv: int,
    fft_len_corr: int,
) -> tuple[float, np.ndarray]:
    """
    Returns (J_α, gradient w.r.t. a) for the smooth approximation.
    """
    # 1) convolution a*a
    conv = _conv_self(a, fft_len_conv)          # length 2n-1
    # 2) smooth max via log‑sum‑exp
    #    J_α = 2·n·(1/α)·log Σ exp(α·conv)
    max_val = np.max(alpha * conv)
    exp_scaled = np.exp(alpha * conv - max_val)
    sum_exp = exp_scaled.sum()
    J_alpha = 2.0 * a.size * (np.log(sum_exp) + max_val) / alpha

    # 3) soft‑max weights over the convolution entries
    w = exp_scaled / sum_exp                     # sum_i w_i = 1

    # 4) gradient ∇_a J_α = 2·(w ⋆ a)
    corr = _correlate(w, a, fft_len_corr)       # length = n
    grad = 2.0 * corr

    return J_alpha, grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------
def propose_candidate(
    seed: int = 12345,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Gradient‑descent (Adam) optimiser on a smoothed max‑convolution.
    Returns the best non‑negative probability vector encountered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(256, min(n, 8192))   # keep FFTs cheap
        except Exception:
            n = 2048
    else:
        n = 2048

    # --------------------------------------------------------------
    # Initialise a (probability simplex) – use previous construction if present
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = a0 / s
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # Occasionally a random Dirichlet start can help escaping poor basins
    if rng.random() < 0.2:
        a0 = rng.dirichlet(np.ones(n) * 0.2)

    # --------------------------------------------------------------
    # Adam state on logits (z) – a = softmax(z)
    # --------------------------------------------------------------
    z = np.log(a0 + 1e-12) + rng.normal(scale=0.02, size=n)

    m = np.zeros(n, dtype=np.float64)   # first moment
    v = np.zeros(n, dtype=np.float64)   # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    # --------------------------------------------------------------
    # Schedules
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    lr_start, lr_end = 0.25, 0.0005            # learning‑rate schedule
    alpha_start, alpha_end = 5.0, 5000.0      # smoothing‑parameter schedule

    # Pre‑computed FFT lengths
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)               # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)       # for w ⋆ a

    # --------------------------------------------------------------
    # Track the best solution
    # --------------------------------------------------------------
    a = _softmax(z)
    best_a = a.copy()
    best_J = _objective_J(best_a)
    last_improve = start_time

    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)

        # Linear schedules
        lr = lr_start * (1 - frac) + lr_end * frac
        alpha = alpha_start * (1 - frac) + alpha_end * frac

        # Softmax probabilities from logits
        a = _softmax(z)

        # Smooth objective & gradient w.r.t. a
        J_alpha, grad_a = _smooth_objective_and_grad(
            a, alpha, fft_len_conv, fft_len_corr
        )

        # Convert to gradient w.r.t. logits (chain rule through softmax)
        dot = np.dot(a, grad_a)
        grad_z = a * (grad_a - dot)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        z -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # Occasionally evaluate the true (non‑smoothed) objective
        if t % 10 == 0:
            J_true = _objective_J(a)
            if J_true < best_J - 1e-12:
                best_J = J_true
                best_a = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] New best J = {best_J:.6f}")

        # Stagnation handling – restart from the current best with a jitter
        if now - last_improve > 30.0:
            z = np.log(best_a + 1e-12) + rng.normal(scale=0.05, size=n)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Restart after stagnation")

    # --------------------------------------------------------------
    # Return the best vector (already a probability vector)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)   # respect evaluate_sequence limits
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total
    return [float(v) for v in best_a]
