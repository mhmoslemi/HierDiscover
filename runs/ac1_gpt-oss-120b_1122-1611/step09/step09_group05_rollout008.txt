"""
Propose a step‑function (list of non‑negative heights) that minimizes
``evaluate_sequence``.  The optimiser works in two phases:

1. **LP‑driven mixing** – starting from a good seed (the previous
   ``height_sequence_1`` if it exists) we repeatedly solve a small linear
   program

       maximise   Σ g_i
       subject to   f * g ≤ M   (pointwise)
                  g_i ≥ 0

   where ``f`` is the current distribution, ``M = max(f*f)`` and ``*`` is
   discrete convolution.  The solution ``g`` points to directions that can
   increase the total mass without worsening the current worst convolution
   value.  By mixing ``f`` with the normalised ``g`` (a line‑search over a
   few mixing coefficients) we obtain a strictly smaller ``max(f*f)``.
   The LP is solved with SciPy’s high‑performance HiGHS solver; only a
   few (≈ 10‑15) LP steps are needed.

2. **Fine‑grained simulated‑annealing** – after the LP phase we run a
   classic simulated‑annealing optimiser (very similar to the reference
   implementation) to polish the distribution.  The SA uses pairwise
   mass transfers and multiplicative log‑normal perturbations, always
   projecting back onto the probability simplex.

Both phases use an FFT‑based convolution (O(n log n)) for speed and a
simple Euclidean projection onto the simplex.  The final sequence is
clipped to the limits imposed by ``evaluate_sequence`` and normalised to
unit sum (the objective is scale‑invariant).

The routine respects the global time budget (default 1000 s) and prints
progress information.  It returns a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, Σw = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, fft_len: int | None = None) -> float:
    """Return max(conv(a, a)) using an FFT (fast for large n)."""
    n = a.size
    conv_len = 2 * n - 1
    if fft_len is None:
        fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _objective(a: np.ndarray, fft_len: int | None = None) -> float:
    """Exact objective J = 2·n·max(conv(a,a)).  a is assumed to sum to 1."""
    n = a.size
    return 2.0 * n * _conv_max(a, fft_len)


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ g) == conv(f, g).
    The i‑th column is f placed at rows i … i+n‑1.
    """
    n = f.size
    conv_len = 2 * n - 1
    A = np.empty((conv_len, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A

# ----------------------------------------------------------------------
