"""
Differential‑evolution optimiser with simplex projection and
final sub‑gradient refinement for the step‑function problem.
The objective J = 2·n·max(conv(a,a)) is minimised under
a ≥ 0, Σ a = 1.  The algorithm keeps a small population,
mutates by “rand/1” DE, projects onto the probability simplex,
and occasionally applies Gaussian smoothing and a
sub‑gradient descent that directly reduces the peak of the
self‑convolution.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    # Euclidean projection onto the probability simplex.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _initial_candidates(rng: np.random.Generator, n: int) -> list[np.ndarray]:
    cand = []
    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))
    # exponential families
    for lam in [0.5, 1.0, 2.0, 5.0]:
        v = np.exp(-lam * np.arange(n, dtype=np.float64))
        v /= v.sum()
        cand.append(v)
    # power‑law families
    for p in [0.5, 1.0, 1.5, 2.0]:
        idx = np.arange(1, n+1, dtype=np.float64)
        v = 1.0 / np.sqrt(idx) if p == 0.5 else 1.0 / (idx**p)
        v /= v.sum()
        cand.append(v)
    # two‑sided exponential (symmetric)

        v = np.exp(-lam * np.arange(n, dtype=np.float64)) + \
            np.exp(-lam * np.arange(n-1, -1, -1, dtype=np.float64))
        v /= v.sum()
        cand.append(v)
    # Dirichlet draws with various concentrations
    for conc in [0.2, 0.5, 1.0, 2.0]:
        cand.append(rng.dirichlet(np.full(n, conc)))

    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())

def _smooth(v: np.ndarray, sigma: float) -> np.ndarray:
    radius = int(3 * sigma)
    if radius < 1:
        return v.copy()
    x = np.arange(-radius, radius + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    sm = np.convolve(v, kernel, mode='same')
    return sm

def _subgradient_refine(a: np.ndarray,
                        n: int,
                        conv_len: int,
                        fft_len: int,
                        steps: int = 200) -> np.ndarray:
    """Perform a simple sub‑gradient descent on max(conv)."""
    cur = a.copy()
    # geometric step schedule
    step_vals = np.geomspace(0.1, 1e-6, num=steps)
    for step in step_vals:
        # compute convolution
        A = np.fft.rfft(cur, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        k = int(np.argmax(conv))
        # build gradient = 2 * cur shifted by k
        grad = np.zeros_like(cur)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idxs = np.arange(lo, hi + 1, dtype=np.int64)
        grad[idxs] = 2.0 * cur[k - idxs]
        # gradient step
        cur = cur - step * grad
        cur = proj_simplex(cur)
    return cur

sub‑gradient refinement.  The objective

    J(a) = 2·n·max(conv(a,a)) / (∑a)²

is scale‑invariant, so we keep the vector on the probability simplex
(∑a = 1) and minimise 2·n·max(conv(a,a)).  The optimiser works in three
phases:

1.  Build a small, diverse population of probability vectors
    (uniform, exponential, power‑law, two‑sided exponentials,
    Dirichlet draws and, if present, the previously found
    ``height_sequence_1``).

2.  Run a classic “rand/1” differential‑evolution loop:
    - mutant = x_r1 + F·(x_r2 – x_r3)
    - project mutant back onto the simplex
    - keep the mutant if it improves the individual’s objective.
    The global best is tracked and occasional Gaussian jitter,
    Gaussian smoothing and a tiny random walk are applied to escape
    plateaus.

3.  After the time budget is almost exhausted a simple
    sub‑gradient descent on the peak of the self‑convolution is
    performed.  For the index k where conv(a,a) is maximal the
    gradient of max(conv) w.r.t. a is 2·a shifted by k; a short

All heavy work (the convolution) is performed with an FFT of length

    if s <= 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s
    return best_vec.tolist()
