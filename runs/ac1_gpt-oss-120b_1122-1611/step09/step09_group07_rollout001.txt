"""
Propose a step‑function (non‑negative vector) that minimizes
``evaluate_sequence``.

Algorithm outline
-----------------
0️⃣ **Problem size** – use the length of the provided ``height_sequence_1``,
   otherwise fall back to a moderate size (≈3000).  All vectors are kept on
   the probability simplex (they sum to 1) because the objective is
   scale‑invariant.

1️⃣ **Initial population** – build a small set of seeds:
   * the supplied sequence (if any),
   * the uniform sequence,
   * ~20 random Dirichlet draws.
   Keep the best seed.

2️⃣ **Projected Adam on a smooth max‑convolution** – a few thousand
   iterations of Adam are performed on the soft‑max approximation of the
   infinity‑norm of the self‑convolution.  This quickly drives the search
   into a good basin.

3️⃣ **Simulated‑Annealing (SA)** – from the Adam result we run a long
   SA loop that repeatedly transfers a random amount of mass from one
   index to another (or swaps two entries).  The Metropolis rule with a
   temperature decreasing from 0.02 to 1e‑5 lets the search escape local
   minima while converging to a refined solution.

4️⃣ **Permutation‑swap local search** – a cheap “swap‑only’’ hill‑climb on
   the remaining time further polishes the best vector.

5️⃣ **Hard‑max gradient refinement** – if a few seconds are left we perform
   a short Adam‑style descent on the exact max (weights are the uniform
   distribution over the active constraints).  This often shaves the last
   fraction of a percent from the objective.

All phases respect the 1000 s budget (5 s safety margin).  The function
returns the best vector found, clipped to the limits enforced by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and re‑project.
    Used as a large‑scale jitter to escape stagnation.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    # re‑project onto the simplex to keep sum = 1
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  The routine follows the multi‑phase scheme
    described in the module docstring.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine problem dimension
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)
            n = base.size
        except Exception:
            base = None
            n = 4096
    else:
        base = None
        n = 4096  # default if nothing is supplied

    # keep the size modest (avoid >5000 for speed)
    n = int(max(512, min(n, 4096)))

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (re‑used many times)
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # length for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # length for a * w_rev

    # --------------------------------------------------------------
    # 3) Helper to compute the true objective J = 2·n·max(a*a)
    #    (the denominator sum(a)^2 cancels out after normalisation)
    # --------------------------------------------------------------
    def compute_J(vec: np.ndarray) -> float:
        fa = np.fft.rfft(vec, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 4) Build an initial pool of candidates and pick the best one
    # --------------------------------------------------------------
    candidates = []

    # uniform seed
    candidates.append(np.full(n, 1.0 / n))

    # supplied seed (if compatible)
    if base is not None and base.size == n:
        s = base.sum()
        if s > 0:
            candidates.append(base / s)
        else:
            candidates.append(np.full(n, 1.0 / n))

    # a handful of random Dirichlet draws
    rand_seeds = 30 if n <= 1500 else 15
    for _ in range(rand_seeds):
        candidates.append(rng.dirichlet(np.ones(n)))

    best_vec = None
    best_J = np.inf
    for vec in candidates:
        cur_J = compute_J(vec)
        if cur_J < best_J:
            best_J = cur_J
            best_vec = vec.copy()
    print(f"[init] n={n}, best J from seeds = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5) Phase 1 – Projected Adam on a smoothed max‑convolution
    # --------------------------------------------------------------
    # Allocate roughly 15 % of the remaining budget to this phase.
    now = time.monotonic()
    remaining = deadline - now
    phase1_time = remaining * 0.15
    adam_deadline = now + phase1_time

    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = now

    while time.monotonic() < adam_deadline:
        it += 1
        tnow = time.monotonic()
        frac = (tnow - now) / phase1_time  # ∈[0,1]

        # learning‑rate & temperature schedules (linear annealing)
        lr = 0.5 * (1.0 - frac) + 0.001 * frac
        alpha = 10.0 * (1.0 - frac) + 5000.0 * frac

        # ----- self‑convolution & soft‑max weights -----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

        cmax = conv.max()
        shifted = conv - cmax
        w = np.exp(alpha * shifted)          # soft‑max numerators
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
        start_idx = w_rev.size - 1            # = 2n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- Adam update + projection onto simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # ----- evaluate -----
        cur_J = compute_J(a)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = tnow
            print(f"[{tnow - start_time:6.2f}s] Adam new best J = {best_J:.6f}")

        # occasional jitter to shake out plateaus
        if rng.random() < 0.005:
            a = _random_block_perturb(a, rng)

        # restart after prolonged stagnation
        if tnow - last_improve > 20.0:
            a = best_vec.copy()
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = tnow
            print(f"[{tnow - start_time:6.2f}s] Adam restart (stagnation)")

    # --------------------------------------------------------------
    # 6) Phase 2 – Simulated‑Annealing (transfer & swap moves)
    # --------------------------------------------------------------
    now = time.monotonic()
    remaining = deadline - now
    # allocate ~70 % of the rest to SA, leave the final chunk for polishing
    sa_time = remaining * 0.7
    sa_deadline = now + sa_time

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_report = now

    while time.monotonic() < sa_deadline:
        tnow = time.monotonic()
        # linear temperature schedule
        frac = (tnow - now) / sa_time
        temp = 0.02 * (1.0 - frac) + 1e-5 * frac

        # ----- propose a move -----
        if rng.random() < 0.5:
            # mass‑transfer move
            i, j = rng.integers(0, n, size=2)
            if i != j and cur_vec[i] > 0.0:
                max_frac = 0.4
                delta = cur_vec[i] * rng.random() * max_frac
                cand = cur_vec.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        else:
            # simple swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            cand = cur_vec.copy()
            cand[i], cand[j] = cand[j], cand[i]

        cand_J = compute_J(cand)

        # ----- Metropolis acceptance -----
        if cand_J < cur_J - 1e-12:
            cur_vec, cur_J = cand, cand_J
            if cand_J < best_J - 1e-12:
                best_J, best_vec = cand_J, cand.copy()
                print(f"[{tnow - start_time:6.2f}s] SA new best J = {best_J:.6f}")
        else:
            if rng.random() < np.exp(-(cand_J - cur_J) / temp):
                cur_vec, cur_J = cand, cand_J

        # occasional re‑projection to cure floating‑point drift
        if int((tnow - now) / 5.0) != int(((tnow - now) - 0.001) / 5.0):
            cur_vec = proj_simplex(cur_vec)

        # very occasional random restart to diversify the search
        if rng.random() < 0.001:
            cur_vec = rng.dirichlet(np.ones(n))
            cur_J = compute_J(cur_vec)

    # --------------------------------------------------------------
    # 7) Phase 3 – Cheap permutation‑swap hill‑climb
    # --------------------------------------------------------------
    now = time.monotonic()
    remaining = deadline - now
    swap_deadline = deadline
    cur_vec = best_vec.copy()
    cur_J = best_J
    attempts = accepts = 0

    while time.monotonic() < swap_deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        cand = cur_vec.copy()
        cand[i], cand[j] = cand[j], cand[i]
        cand_J = compute_J(cand)
        if cand_J < cur_J - 1e-12:
            cur_vec, cur_J = cand, cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best_J, best_vec = cand_J, cand.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] swap new best J = {best_J:.6f}")

    print(f"Swap phase finished – attempts: {attempts}, accepts: {accepts}")

    # --------------------------------------------------------------
    # 8) Phase 4 – Final hard‑max gradient refinement (if time permits)
    # --------------------------------------------------------------
    now = time.monotonic()
    remaining = deadline - now
    if remaining > 5.0:                     # need a few seconds to be worthwhile
        # a very short Adam run on the exact max (weights = uniform over active set)
        a = best_vec.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        max_iters = 2000
        refine_start = time.monotonic()
        for t in range(max_iters):
            if time.monotonic() >= deadline:
                break
            # convolution
            fa = np.fft.rfft(a, n=fft_len_self)
            conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
            cmax = conv.max()
            # active set (within a tiny tolerance)
            tol = 1e-12
            mask = conv >= cmax - tol
            w = mask.astype(float)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
                w_sum = 1.0
            w /= w_sum                     # uniform distribution over active indices

            # gradient of the (non‑smoothed) max‑convolution
            w_rev = w[::-1]
            fa_a = np.fft.rfft(a, n=fft_len_grad)
            fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
            full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
            start_idx = w_rev.size - 1
            grad = 2.0 * full[start_idx:start_idx + n]

            # Adam step with a tiny learning‑rate that decays
            frac = t / max_iters
            lr = 0.02 * (1.0 - frac) + 0.0005 * frac
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (t + 1))
            v_hat = v / (1.0 - beta2 ** (t + 1))
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = proj_simplex(a)

            # occasional bookkeeping
            if (t + 1) % 250 == 0:
                cur_J = compute_J(a)
                if cur_J < best_J - 1e-12:
                    best_J, best_vec = cur_J, a.copy()
                    print(f"[{time.monotonic() - start_time:6.2f}s] final refine new best J = {best_J:.6f}")

        # be safe and project once more
        best_vec = proj_simplex(best_vec)

    # --------------------------------------------------------------
    # 9) Return the best sequence (clip to evaluation limits)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    print(f"Optimization finished – best J = {best_J:.6f}")
    return [float(x) for x in best_vec]
