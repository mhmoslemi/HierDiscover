    final_score = evaluate_sequence(best_seq.tolist())
    total_elapsed = time.monotonic() - start
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_seq]

"""
Propose a low‑value step‑function for the max‑convolution objective.

The algorithm consists of three phases:

1. **Continuation‑gradient descent** – minimise a smoothed version of the
   max‑convolution (softmax with an annealing temperature β) using
   projected gradient descent on the probability simplex.
   Starting with a small β gives a smooth landscape, while slowly raising β
   drives the solution toward the true objective.

2. **Multiple random restarts** – the descent is run from several Dirichlet
   seeds (and from the best known construction if available).  The best
   sequence seen over all restarts is retained.

3. **Stochastic local refinement** – tiny directed mass‑transfer moves
   (source → target) are accepted if they lower the exact score.
   Occasional jitter avoids getting stuck.

All operations respect the non‑negativity and ∑aᵢ=1 constraints,
so the final value of ``evaluate_sequence`` equals 2·n·max(a∗a).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft_general(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution of two real vectors using an FFT of length ``fft_len``.
    Returns a vector of length ``len(x)+len(y)-1``.
    """
    X = np.fft.rfft(x, n=fft_len)
    Y = np.fft.rfft(y, n=fft_len)
    res = np.fft.irfft(X * Y, n=fft_len)
    return res[: x.size + y.size - 1]


def _smooth_gradient(a: np.ndarray,
                     beta: float,
                     fft_len_self: int,
                     fft_len_grad: int):
    """
    Compute the gradient of the smoothed max‑convolution
    (softmax with temperature beta) and return it together with the exact
    self‑convolution and its current maximum.
    """
    n = a.size
    # a * a
    conv_self = _conv_fft_general(a, a, fft_len_self)          # length 2n‑1
    max_val = conv_self.max()

    # softmax weights over all convolution entries
    # shift by max for numerical stability
    exp_vals = np.exp(beta * (conv_self - max_val))
    w = exp_vals / exp_vals.sum()                             # shape 2n‑1
    w_rev = w[::-1]

    # a * w  (length 3n‑2)
    conv2 = _conv_fft_general(a, w, fft_len_grad)
    term2 = conv2[:n]

    # a * w_rev  (length 3n‑2)
    conv1 = _conv_fft_general(a, w_rev, fft_len_grad)
    term1 = conv1[2 * n - 2 - np.arange(n)]

    grad = term1 + term2                    # ∂/∂a of the smooth max
    return grad, conv_self, max_val


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Determine problem size (a power of two for fast FFTs)
    # --------------------------------------------------------------
    base_n = 4096
    if "height_sequence_1" in globals():
        base_n = max(base_n, len(globals()["height_sequence_1"]))
    n = _next_pow2(base_n)                     # e.g. 4096, 8192, ...

    # --------------------------------------------------------------
    # 1) FFT bookkeeping
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n)            # for a*a  (length 2n‑1)
    fft_len_grad = _next_pow2(3 * n)            # for a*w   (length 3n‑2)

    # --------------------------------------------------------------
    # 2) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 3) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best_a = None
    best_score = float("inf")

    def _fast_score(vec):
        """2·n·max(conv) for a vector that already sums to 1."""
        conv = _conv_fft_general(vec, vec, fft_len_self)
        return 2.0 * n * conv.max()

    def _maybe_update(vec, score):
        nonlocal best_a, best_score
        if score < best_score - 1e-14:
            best_score = score
            best_a = vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Initialise from the previous construction if available
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            pad = np.full(n - a0.size, 1e-12)
            a0 = np.concatenate([a0, pad])
        a0 /= a0.sum()
        _maybe_update(a0, _fast_score(a0))
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
        _maybe_update(a0, _fast_score(a0))

    # --------------------------------------------------------------
    # 5) Continuation‑gradient descent with random restarts
    # --------------------------------------------------------------
    max_restarts = 8                     # how many independent runs
    restart_time = 30.0                  # seconds allocated per restart
    beta_start = 5.0
    beta_end = 200.0
    step0 = 0.20                         # base step‑size for projected GD

    for restart_idx in range(max_restarts):
        # honour the global time limit
        if time.monotonic() >= deadline:
            break

        # ----------------------------------------------------------
        # pick a starting point for this restart
        # ----------------------------------------------------------
        if restart_idx == 0 and best_a is not None:
            a = best_a.copy()
        else:
            a = rng.dirichlet(np.ones(n))
        a = np.clip(a, 1e-15, None)
        a /= a.sum()

        # ----------------------------------------------------------
        # run projected GD for the allocated slice of time
        # ----------------------------------------------------------
        it = 0
        # using a fixed number of inner iterations; we break early
        # if the slice runs out of time.
        while time.monotonic() < min(deadline,
                                     start_time + (restart_idx + 1) * restart_time):
            # linearly increase β on a log‑scale (continuation)
            if it < 2000:
                # exponential schedule from beta_start to beta_end
                beta = beta_start * (beta_end / beta_start) ** (it / 1999.0)
            else:
                beta = beta_end

            # gradient of the smoothed max‑convolution
            grad, conv_self, cur_max = _smooth_gradient(a, beta,
                                                        fft_len_self,
                                                        fft_len_grad)
            # gradient of the true objective (2·n·... )
            grad_obj = 2.0 * n * grad

            # diminishing step‑size
            step = step0 / np.sqrt(it + 1.0)

            # projected additive update
            a_candidate = a - step * grad_obj
            a_candidate = _project_onto_simplex(a_candidate)

            # evaluate the exact objective (fast convolution)
            cand_score = _fast_score(a_candidate)

            # keep the best ever seen
            _maybe_update(a_candidate, cand_score)

            # accept non‑worsening moves to keep descent monotone
            if cand_score <= 2.0 * n * cur_max + 1e-12:
                a = a_candidate

            it += 1
            if it % 500 == 0:
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] restart={restart_idx} "
                      f"iter={it} cur_score={2.0 * n * cur_max:.6f}")

            # early stop if we already beat the target
            if best_score <= 1.5025:
                print("[early stop] target reached during continuation")
                break

        if best_score <= 1.5025:
            break

    # --------------------------------------------------------------
    # 6) Stochastic local refinement (remaining time)
    # --------------------------------------------------------------
    if best_a is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
        best_a = a.copy()
    else:
        a = best_a.copy()
    cur_score = _fast_score(a)

    it_local = 0
    while time.monotonic() < deadline:
        # pick source weighted by current mass (more mass ⇒ more likely)
        src = int(rng.choice(np.arange(n), p=a))
        tgt = int(rng.integers(0, n))
        if src == tgt:
            continue

        # move a small random fraction of the source mass
        frac = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))  # 0.1%–5%
        delta = a[src] * frac
        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta
        a_candidate = _project_onto_simplex(a_candidate)

        cand_score = _fast_score(a_candidate)
        if cand_score < best_score - 1e-12:
            _maybe_update(a_candidate, cand_score)

        # keep the move if it does not increase the exact objective
        if cand_score < cur_score:
            a = a_candidate
            cur_score = cand_score

        # occasional jitter to shake the solution
        if it_local % 4000 == 0 and it_local > 0:
            width = int(rng.integers(1, max(2, n // 80)))
            start_idx = int(rng.integers(0, n - width + 1))
            factor = np.exp(rng.normal(0.0, 0.018))
            a_jitter = a.copy()
            a_jitter[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_jitter)

        it_local += 1

        # early exit if target achieved
        if best_score <= 1.5025:
            print("[early stop] target reached in refinement")
            break

    # --------------------------------------------------------------
    # 7) Finalisation – clip, evaluate with the official checker,
    #    and return the best sequence found.
    # --------------------------------------------------------------
    best_seq = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_seq]
