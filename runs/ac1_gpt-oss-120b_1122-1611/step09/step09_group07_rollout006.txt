    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Choose length
    n = 4096
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0.0, None)
        # Use length of base if within [512, 4096]
        n = max(512, min(base.size, 4096))
        base = base / base.sum()
    else:
        base = None

    # Setup FFT lengths
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    def compute_J_and_conv(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        max_val = conv.max()
        return float(2.0 * n * max_val), conv

    # variables for best

"""
Optimiser using hard‑max sub‑gradients on the simplex.
The algorithm alternates fast projected‑Adam sub‑gradient steps
(with the exact max‑convolution instead of a soft‑max)
and a short stochastic swap‑local‑search phase.
Multiple random restarts are performed within the time budget.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out

def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Exact sub‑gradient of max(conv) with respect to a (a lives on the simplex)."""
    n = a.size
    M = conv.max()
    # treat all indices that are numerically equal to the maximum as active
    tol = 1e-12
    active = np.where(conv >= M - tol)[0]
    grad = np.zeros_like(a)
    for k in active:
        # indices j where both a_j and a_{k-j} are defined (0 ≤ j < n, 0 ≤ k‑j < n)
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        idx = np.arange(start, end + 1, dtype=np.int64)
        grad[idx] += 2.0 * a[k - idx]
    return grad

    The routine performs several Adam‑based sub‑gradient passes on the exact
    max‑convolution (no soft‑max smoothing), followed by a cheap permutation

    sequence found.

    # --------------------------------------------------------------

    # --------------------------------------------------------------

            base_seq = np.clip(base_seq, 0.0, None)

    else:
        base_seq = None
        n = 4096

    n = max(512, min(n, 4096))

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # Helper to compute objective J = 2·n·max(conv_self)
    # ------------------------------------------------------------------

            lr_start, lr_end = 0.5, 0.001

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, None)
            a = proj_simplex(a)

            # ----- record best -----
            if J_val < best_J - 1e-12:
                best_J = J_val
                best_vec = a.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] restart {it:4d} new best J = {best_J:.6f}")

            # occasional random block jitter
            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)

            # restart on stagnation
            if now - last_improve > 25.0:
                # make a noisy copy of the current best
                a = best_vec.copy()
                noise = np.exp(rng.normal(0.0, 0.07, size=n))
                a = a * noise
                a = proj_simplex(a)
                last_improve = now
                print(f"[{now - start_time:6.2f}s] stagnation restart – best J = {best_J:.6f}")

            # early exit if we have already reached the target bound
            if best_J <= 1.5025:
                print("Target reached – breaking out of restarts.")
                break

        # if target reached, break outer loop as well
        if best_J <= 1.5025:
            break

    # ------------------------------------------------------------------
    # 3)  Permutation‑swap local search (uses the best vector found)
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n)
        best_J, _ = compute_J_and_conv(best_vec)

    cur_vec = best_vec.copy()
    cur_J, _ = compute_J_and_conv(cur_vec)

    attempts = 0
    accepts = 0

        if i == j:
            continue

        cand[i], cand[j] = cand[j], cand[i]
        cand_J, _ = compute_J_and_conv(cand)

        if cand_J < cur_J - 1e-12:
            cur_vec = cand
            cur_J = cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                print(f"[swap] new best J = {best_J:.6f}")
        else:

                cur_J = cand_J
                accepts += 1

            cur_vec = _random_block_perturb(cur_vec, rng)

    print(f"Swap phase finished – attempts {attempts}, accepts {accepts}, best J {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4)  Return clipped sequence
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
