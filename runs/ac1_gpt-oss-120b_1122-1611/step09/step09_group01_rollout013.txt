def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """Return the true objective J = 2·n·max(a*a) for a normalised vector."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())
```

"""
Propose a non‑negative step function that minimises ``evaluate_sequence``.

Algorithm
---------
1. **Problem size** – Use the length of the previously found sequence
   (``height_sequence_1``) if it exists, otherwise fall back to 4096.
2. **Diverse seeds** – Build a small portfolio of analytic shapes
   (uniform, linear ramp, exponentials, symmetric Beta‑like densities)
   together with a few Dirichlet draws.  All candidates are projected onto
   the probability simplex (sum = 1).
3. **Sub‑gradient mirror descent** – The true objective is
   ``J = 2·n·max(conv(a,a))`` (with ``sum(a)=1``).  The gradient of the
   max‑convolution is obtained by a convolution of ``a`` with the
   reversed indicator of the active anti‑diagonal(s).  A mirror‑descent
   step (multiplicative update + Euclidean projection onto the simplex)
   is applied with a slowly decreasing step‑size.
4. **Local polishing** – A cheap pair‑wise mass‑transfer hill‑climber
   finishes the search.
5. **Time‑budget handling** – The routine respects a 1000 s wall‑clock
   budget (5 s safety margin) and always returns the best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Linear‑algebra utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex (non‑negative, sum = 1).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D vector")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective helpers (exact objective – no softening)
# ----------------------------------------------------------------------


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """
    Exact objective J = 2·n·max(conv(a,a)) for a vector that satisfies
    sum(a)=1.
    """
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def compute_conv(a: np.ndarray,
                 n: int,
                 fft_len_self: int,
                 conv_len: int) -> np.ndarray:
    """Full self‑convolution (length 2·n‑1)."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


def subgradient_max_conv(a: np.ndarray,
                         n: int,
                         fft_len_self: int,
                         conv_len: int,
                         fft_len_grad: int):
    """
    Sub‑gradient of the max‑convolution term.
    Returns (grad, max_val) where
        grad = 2·a * w_rev   (w – indicator of active anti‑diagonal(s))
    """
    # current convolution
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

    max_val = conv.max()
    # indicator (average over all currently active anti‑diagonals)
    eps = 1e-12
    active = conv >= max_val - eps
    cnt = active.sum()
    w = np.zeros(conv_len, dtype=np.float64)
    if cnt == 0:
        w[conv.argmax()] = 1.0
    else:
        w[active] = 1.0 / float(cnt)

    # reverse the weight vector
    w_rev = w[::-1]

    # convolution a * w_rev  (linear, not circular)
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)

    start = conv_len - 1          # index where the linear convolution aligns
    grad = 2.0 * full[start:start + n]
    return grad, max_val


# ----------------------------------------------------------------------
# Candidate generators (analytic families)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_descending(n: int) -> np.ndarray:
    v = np.arange(n, 0, -1, dtype=np.float64)
    return v / v.sum()


def _exponential(n: int, r: float) -> np.ndarray:
    v = r ** np.arange(n, dtype=np.float64)
    v = np.clip(v, 0.0, None)
    return v / v.sum()


def _beta_symmetric(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) density on the discrete grid."""
    eps = 1e-12
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    x = np.clip(x, eps, 1.0 - eps)
    v = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    v = np.clip(v, 0.0, None)
    s = v.sum()
    if s == 0.0:
        v = _uniform(n)
    else:
        v /= s
    return v


def _dirichlet_draws(n: int, rng: np.random.Generator, k: int = 10):
    """Draw k Dirichlet samples with concentration 0.3 (moderately sparse)."""
    draws = []
    for _ in range(k):
        draws.append(rng.dirichlet(np.ones(n) * 0.3))
    return draws

def mirror_descent_optimize(init_a: np.ndarray,
                            n: int,
                            fft_len_self: int,
                            conv_len: int,
                            fft_len_grad: int,
                            deadline: float,
                            start_time: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (multiplicative) optimisation using the exact sub‑gradient
    of the max‑convolution term.  Returns the best vector and its score.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_score = compute_J(a, n, fft_len_self, conv_len)

    # exponential‑decay step‑size schedule
    while time.monotonic() < deadline:
        # fraction of the mirror‑descent phase already elapsed
        frac = (time.monotonic() - start_time) / (deadline - start_time)
        eta = 0.5 * (1.0 - frac) + 0.02 * frac   # 0.5 → 0.02

        grad, _ = subgradient_max_conv(a, n, fft_len_self, conv_len, fft_len_grad)

        # multiplicative update (mirror step w.r.t. Euclidean distance)
        a = a * np.exp(-eta * grad)
        a = proj_simplex(a)

        cur_score = compute_J(a, n, fft_len_self, conv_len)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
    return best_a, best_score


def hill_climb_optimize(init_a: np.ndarray,
                        n: int,
                        fft_len_self: int,
                        conv_len: int,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple pair‑wise mass‑transfer hill‑climber that runs as long as time permits.
    """
    rng = np.random.default_rng()
    a = init_a.copy()
    cur_score = compute_J(a, n, fft_len_self, conv_len)
    best_a = a.copy()
    best_score = cur_score

    while time.monotonic() < deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # transfer a modest amount of probability mass from i to j
        max_transfer = min(a[i], 0.005)   # limit step size to stay locally smooth
        if max_transfer <= 0.0:
            continue
        delta = rng.random() * max_transfer
        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_score = compute_J(cand, n, fft_len_self, conv_len)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = cand.copy()
        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
    return best_a, best_score

    best_vec = None
    best_score = float("inf")
    for v in cand_vectors:
        v = proj_simplex(v.astype(np.float64))
        sc = compute_J(v, n, fft_len_self, conv_len)
        if sc < best_score:
            best_score = sc
            best_vec = v.copy()
    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] Initial best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Mirror‑descent refinement (major part of the budget)
    # --------------------------------------------------------------
    md_deadline = start_time + 0.75 * (deadline - start_time)   # 75 % for MD
    best_vec, best_score = mirror_descent_optimize(
        best_vec, n, fft_len_self, conv_len, fft_len_grad,
        md_deadline, start_time)

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] After mirror‑descent best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Fine‑grained hill‑climbing (remaining budget)
    # --------------------------------------------------------------
    hill_deadline = deadline
    best_vec, best_score = hill_climb_optimize(
        best_vec, n, fft_len_self, conv_len, hill_deadline)

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] Final best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6) Return result (values already lie in [0,1000] after projection)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
