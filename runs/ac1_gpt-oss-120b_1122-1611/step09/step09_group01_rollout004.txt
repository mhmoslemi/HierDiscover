"""
Simple Exponentiated‑Gradient descent on the exact max‑convolution objective.

The algorithm works on the simplex (non‑negative entries that sum to 1):
1. Choose a problem size `n` (up to ~8192, larger than the previous run).
2. Build a diverse set of initial vectors (uniform, exponentials, linear
   ramps, Dirichlet draws, the previous best if available).
3. For each start run an exponentiated‑gradient / sub‑gradient loop:
   * compute the full self‑convolution via FFT,
   * locate the active set of indices attaining the maximum,
   * build the indicator `w` of that set (normalized to 1),
   * compute the sub‑gradient `g = a * w_rev` by another FFT,
   * update `a ← a * exp(-η·g)` and re‑project onto the simplex.
   * learning‑rate η decays linearly from 0.5 to 0.01.
   * occasional random “block‑perturb’’ steps are injected to leave plateaus.
4. Keep the best vector seen over all starts.
5. Finish with a cheap pair‑wise mass‑transfer hill‑climb for a few seconds.
6. Return the best sequence (clipped to the limits of `evaluate_sequence`).

The whole search respects the 1000 s wall‑clock budget (5 s safety margin)
and always returns the best sequence found so far.
"""

import time
import numpy as np
from typing import List, Tuple

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def compute_J(a: np.ndarray, n: int, fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a*a). `a` must already sum to 1."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())

def compute_conv_and_J(a: np.ndarray,
                       n: int,
                       fft_len_self: int,
                       conv_len: int) -> Tuple[np.ndarray, float]:
    """Return the full self‑convolution (first ``conv_len`` entries) and J."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv, float(2.0 * n * conv.max())

def subgradient_gradient(a: np.ndarray,
                         w_rev: np.ndarray,
                         n: int,
                         fft_len_grad: int) -> np.ndarray:
    """
    Sub‑gradient of ``max(a*a)`` when the active set is represented by
    the (normalised) indicator ``w_rev`` (reversed order).
    Returns a vector of length ``n``.
    """
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1            # position of the first valid element
    return full[start:start + n]

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and re‑normalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()

"""
Simple Exponentiated‑Gradient descent on the exact max‑convolution objective.

The algorithm works on the simplex (non‑negative entries that sum to 1):
1. Choose a problem size `n` (up to ~8192, larger than the previous run).
2. Build a diverse set of initial vectors (uniform, exponentials, linear
   ramps, Dirichlet draws, the previous best if available).
3. For each start run an exponentiated‑gradient / sub‑gradient loop:
   * compute the full self‑convolution via FFT,
   * locate the active set of indices attaining the maximum,
   * build the indicator `w` of that set (normalised to 1),
   * compute the sub‑gradient `g = a * w_rev` by another FFT,
   * update `a ← a * exp(-η·g)` and re‑project onto the simplex.
   * learning‑rate η decays linearly from 0.5 to 0.01.
   * occasional random “block‑perturb’’ steps are injected to leave plateaus.
4. Keep the best vector seen over all starts.
5. Finish with a cheap pair‑wise mass‑transfer hill‑climb for a few seconds.
6. Return the best sequence (clipped to the limits of `evaluate_sequence`).

The whole search respects the 1000 s wall‑clock budget (5 s safety margin)
and always returns the best sequence found so far.
"""

import time
from typing import List, Tuple

import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(a: np.ndarray, n: int, fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a*a). ``a`` must already sum to 1."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def compute_conv_and_J(a: np.ndarray,
                       n: int,
                       fft_len_self: int,
                       conv_len: int) -> Tuple[np.ndarray, float]:
    """Return the full self‑convolution (first ``conv_len`` entries) and J."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv, float(2.0 * n * conv.max())


def subgradient_gradient(a: np.ndarray,
                         w_rev: np.ndarray,
                         n: int,
                         fft_len_grad: int) -> np.ndarray:

    Sub‑gradient of ``max(a*a)`` when the active set is encoded by the
    (normalised) indicator ``w_rev`` (reversed order).
    Returns a vector of length ``n``.

    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1
    return full[start:start + n]
