def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    # select n:
    if 'height_sequence_1' in globals():
        raw = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        n = raw.size
        a0 = np.clip(raw, 0, None)
        a0 = a0 / a0.sum() if a0.sum()>0 else np.full(n, 1.0/n)
    else:
        n = 2048
        a0 = rng.dirichlet(np.full(n, 0.5))
    # jitter
    a0 = a0 + rng.normal(0, 1e-5, size=n)
    a0 = np.clip(a0, 0, None)
    a0 = a0 / a0.sum()
    # convert to y
    eps = 1e-12
    y = np.log(a0 + eps)
    # precompute FFT length
    conv_len = 2*n - 1
    fft_len_self = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(3*n - 2)  # length for convolution with w_rev
    start_idx = (fft_len_grad - conv_len)//2
    # Adam parameters
    m = np.zeros_like(y)
    v = np.zeros_like(y)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr_initial = 0.05
    lr = lr_initial
    # alpha schedule
    alpha_start = 5.0
    alpha_end = 5000.0
    # keep best
    best_a = a0.copy()
    best_val = float('inf')
    # compute initial evaluate_seq
    best_val = evaluate_sequence(best_a.tolist())
    print(f"[init] evaluate = {best_val:.6f}")
    # iteration loop
    it = 0
    last_improve_time = time.monotonic()
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - (deadline - budget_s + 5.0)
        # schedule alpha linearly based on elapsed fraction
        frac = (now - (deadline - budget_s + 5.0)) / (budget_s - 5.0)
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        # Softmax to get a
        # Use stable softmax: subtract max of y
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()
        # compute convolution via FFT
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        cmax = conv.max()
        # compute w = softmax(alpha*(conv - cmax))
        shifted = conv - cmax
        # To avoid overflow: maybe scaling before exp
        exp_shift = np.exp(alpha * shifted)
        w = exp_shift / exp_shift.sum()
        # gradient wrt a: grad_a = 2 * (a * w_rev) via convolution
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fw, n=fft_len_grad)
        grad_a = 2.0 * full[start_idx:start_idx+n]  # shape n
        # gradient wrt y
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_y
        v = beta2 * v + (1 - beta2) * (grad_y ** 2)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        # optionally decay lr (e.g., lr *= 0.9999 per iteration)
        lr *= 0.9999
        # Periodically evaluate actual objective (e.g., every 10 iterations or per time)
        if it % 20 == 0:
            # compute evaluate (real)
            eval_val = evaluate_sequence(a.tolist())
            if eval_val < best_val - 1e-12:
                best_val = eval_val
                best_a = a.copy()
                last_improve_time = now
                print(f"[{elapsed:6.2f}s] iter {it:5d} new best eval = {best_val:.6f}")
            # also maybe adjust lr if stuck
        # Check stagnation
        if now - last_improve_time > 30.0:
            # jitter the y vector a bit
            # we add Gaussian noise to a subset of y
            jitter_scale = 0.1
            idxs = rng.choice(n, size=max(1, n // 20), replace=False)
            y[idxs] += rng.normal(0.0, jitter_scale, size=idxs.shape)
            # reset Adam moments partially
            m[idxs] *= 0.5
            v[idxs] *= 0.5
            last_improve_time = now
            print(f"[{elapsed:6.2f}s] jitter due to stagnation")
        # optional early termination if eval close to target
        if best_val < 1.5025:
            # we have achieved target
            break
    # final best sequence
    best_a = np.clip(best_a, 0.0, 1000.0)
    return best_a.tolist()

"""
Adam‑softmax optimiser for the step‑function bound.

The algorithm works on the probability simplex by representing the
coefficients a_i = softmax(y_i).  A smooth surrogate of the
max‑convolution (soft‑max with a temperature α) is minimised with
Adam.  The temperature α is increased linearly, turning the surrogate
into the true max.  Whenever the objective (2·n·max(a*a)) improves,
the best sequence is stored.  If no improvement is seen for a while a
random block jitter is applied to escape plateaus.  The routine stops
as soon as the time budget is exhausted or the target bound is reached.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  Returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time budget
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # ------------------------------------------------------------------
    # 1) problem size and initial point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # enforce a reasonable size (do not exceed 5000)
        max_len = 5000
        if raw.size > max_len:
            raw = raw[:max_len]
        elif raw.size < 2:
            raw = rng.dirichlet(np.full(max_len, 0.5))
        a0 = np.clip(raw, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0 = np.full_like(a0, 1.0 / a0.size)
        else:
            a0 = a0 / s
    else:
        # fallback length
        n = 4096
        a0 = rng.dirichlet(np.full(n, 0.5))

    n = a0.size

    # small random jitter to avoid exact stagnation on a previously‑found point
    a0 = a0 + rng.normal(0.0, 1e-5, size=n)
    a0 = np.clip(a0, 0.0, None)
    a0 = a0 / a0.sum()

    # ------------------------------------------------------------------
    # 2) work in the log‑softmax (unconstrained) space
    # ------------------------------------------------------------------
    eps = 1e-12
    y = np.log(a0 + eps)                     # shape (n,)

    # ------------------------------------------------------------------
    # 3) pre‑compute FFT dimensions
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)                 # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)                 # for a * w_rev
    # index where the length‑n slice of the full convolution starts
    start_idx = (fft_len_grad - conv_len) // 2

    # ------------------------------------------------------------------
    # 4) optimiser hyper‑parameters
    # ------------------------------------------------------------------
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(y)
    v = np.zeros_like(y)

    # learning‑rate schedule (initial value – will be decayed gently)
    lr = 0.05

    # soft‑max temperature schedule (α turns the surrogate into the true max)
    alpha_start = 2.0
    alpha_end = 5000.0

    # optimisation bookkeeping
    best_y = y.copy()
    best_a = a0.copy()
    # compute initial true objective (2·n·max(a*a))
    fa0 = np.fft.rfft(a0, n=fft_len_self)
    conv0 = np.fft.irfft(fa0 * fa0, n=fft_len_self)[:conv_len]
    best_J = float(2.0 * n * conv0.max())
    print(f"[init] n={n}  J = {best_J:.6f}")

    # stagnation handling
    last_improve = start_time
    stagnation_limit = 30.0  # seconds

    target_J = 1.5025

    it = 0
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time

        # ------------------------------------------------------------------
        # 5) compute a = softmax(y) (ensures non‑negative, sum = 1)
        # ------------------------------------------------------------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        a = exp_y / exp_y.sum()

        # ------------------------------------------------------------------
        # 6) self‑convolution a*a  (used both for objective and for w)
        # ------------------------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

        # true objective (no scaling needed because sum(a)=1)
        J = float(2.0 * n * conv.max())

        # ------------------------------------------------------------------
        # 7) keep the best solution seen so far
        # ------------------------------------------------------------------
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            best_y = y.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] iter {it:5d} new best J = {best_J:.6f}")

            if best_J <= target_J:
                # target reached – early exit
                break

        # ------------------------------------------------------------------
        # 8) soft‑max weights w (smooth approximation of the max)
        # ------------------------------------------------------------------
        cmax = conv.max()
        shifted = conv - cmax                     # ≤ 0
        # exponentials; α may be large but shifted ≤ 0, so no overflow
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / (budget_s - 5.0))
        exp_shift = np.exp(alpha * shifted)
        w_sum = exp_shift.sum()
        if w_sum == 0.0:
            w = np.full_like(exp_shift, 1.0 / exp_shift.size)
        else:
            w = exp_shift / w_sum                  # shape (conv_len,)

        # ------------------------------------------------------------------
        # 9) gradient of the smooth surrogate w.r.t. a
        # ------------------------------------------------------------------
        w_rev = w[::-1]                           # reversed weight vector
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fw, n=fft_len_grad)
        grad_a = 2.0 * full[start_idx:start_idx + n]   # shape (n,)

        # ------------------------------------------------------------------
        # 10) chain rule to obtain gradient w.r.t. y (softmax variables)
        # ------------------------------------------------------------------
        dot = np.dot(grad_a, a)
        grad_y = a * (grad_a - dot)                # dF/dy

        # ------------------------------------------------------------------
        # 11) Adam update on y
        # ------------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_y
        v = beta2 * v + (1.0 - beta2) * (grad_y ** 2)

        # bias‑corrected moments
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # apply update
        y = y - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # gentle decay of the learning rate
        lr *= 0.9999

        # ------------------------------------------------------------------
        # 12) occasional jitter if we have been stuck for a while
        # ------------------------------------------------------------------
        if now - last_improve > stagnation_limit:
            # choose a random contiguous block and perturb its logits
            width = rng.integers(1, max(2, n // 10))
            start = rng.integers(0, n - width + 1)
            perturb = rng.normal(0.0, 0.1, size=width)
            y[start:start + width] += perturb
            # partially reset the Adam moments for the perturbed coordinates
            m[start:start + width] *= 0.5
            v[start:start + width] *= 0.5
            last_improve = now
            print(f"[{elapsed:6.2f}s] jitter after stagnation – J = {J:.6f}")

    # ------------------------------------------------------------------
    # 13) finalise best sequence (clip to evaluation limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
