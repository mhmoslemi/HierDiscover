    """Generate a smooth non‑negative vector via a random Fourier series."""
    x = np.arange(n) / n
    a = np.zeros(n)

    else:
        a /= a.sum()
    return a

assistantfinal**Solution Overview**

The algorithm builds a *smooth* probability distribution (the step‑function heights) and refines it with an *annealed soft‑max* gradient descent on the exact objective  

\[
J(a)=2\,n\;\max_{k}(a\star a)_k\qquad\bigl(\sum a_i = 1,\ a_i\ge 0\bigr)
\]

The main ideas are

1. **Diverse smooth initialisations** – random Fourier series, exponentials,
   triangular shapes and the previously discovered construction are all
   projected onto the simplex.
2. **Annealed soft‑max surrogate** – for a temperature `t` the smooth objective  

   \[
   J_t(a)=\frac{2n}{t}\log\!\Bigl(\sum_{k}\exp\!\bigl(t\,(a\star a)_k\bigr)\Bigr)
   \]

   approximates the true max.  Its gradient is a *weighted* sum of the
   individual convolution gradients and can be computed with a single FFT.
3. **Mirror‑descent on the simplex** – we use the natural KL‑mirror update  

   \[
   a^{\text{new}}_i = a_i\;\exp\!\bigl(-\eta\,\nabla_iJ_t(a)\bigr)
   \]

   followed by a Euclidean projection onto the probability simplex.
4. **Temperature schedule** – start with a very soft surrogate (`t≈0.2`) and
   increase geometrically up to `t≈1500`.  The step‑size `η` decays linearly.
5. **Restarts & jitter** – if no improvement is seen for 20 s we randomise
   the current best vector (small multiplicative noise) to escape local minima.

All heavy work is done with FFTs, the algorithm is fully vectorised,
runs well within the 1000 s budget and returns the best vector found.

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort decreasing
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def self_convolution(a: np.ndarray,
                     fft_len_self: int,
                     conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


def compute_J(a: np.ndarray,
              n: int,
              fft_len_self: int,
              conv_len: int) -> float:
    """
    Exact objective J = 2·n·max(a∗a)  (the vector a is assumed to sum to 1).
    """
    conv = self_convolution(a, fft_len_self, conv_len)
    return float(2.0 * n * conv.max())


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A modest portfolio of diverse starting points (already normalised)."""
    cand: list[np.ndarray] = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # triangular shape (mountain)
    half = n // 2
    v = np.concatenate([np.arange(1, half + 1),
                        np.arange(half, 0, -1)], dtype=np.float64)
    if v.size > n:
        v = v[:n]
    v /= v.sum()
    cand.append(v)

    # two spikes on the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


def random_fourier_candidate(n: int,
                             rng: np.random.Generator,
                             max_freq: int = 15,
                             scale: float = 0.2) -> np.ndarray:
    """
    Generate a smooth non‑negative vector by a random Fourier series.
    The series is forced to be non‑negative and then projected onto the simplex.
    """
    x = np.arange(n) / n
    a = np.zeros(n, dtype=np.float64)

    # small constant offset to avoid exact zeros
    base = rng.uniform(0.0, 0.1)
    a += base

    for k in range(1, max_freq + 1):
        amp = rng.exponential(scale) / k          # decaying amplitudes
        phase = rng.uniform(0.0, 2.0 * np.pi)
        a += amp * np.cos(2.0 * np.pi * k * x + phase)

    a = np.maximum(a, 0.0)
    sm = a.sum()
    if sm == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= sm
    return a


def annealing_descent(initial_a: np.ndarray,
                      n: int,
                      fft_len_self: int,
                      conv_len: int,
                      fft_len_grad: int,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Annealed soft‑max (log‑sum‑exp) gradient descent on the exact objective.
    Uses a KL‑mirror (multiplicative) update and restarts after stagnation.
    Returns the best vector seen and its objective value.
    """
    a = initial_a.copy()
    best_vec = a.copy()
    best_score = compute_J(a, n, fft_len_self, conv_len)

    # schedule parameters
    total_steps = 300
    eta_start, eta_end = 0.8, 0.001          # step‑size (linear decay)
    t_start, t_end = 0.2, 1500.0             # temperature (geometric growth)

    start_time = time.monotonic()
    last_improve = start_time

    for step in range(total_steps):
        now = time.monotonic()
        if now >= deadline:
            break

        frac = step / total_steps
        # geometric temperature schedule
        t = t_start * ((t_end / t_start) ** frac)
        # linear step‑size schedule
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # ----- self‑convolution (needed for the surrogate) -----
        conv = self_convolution(a, fft_len_self, conv_len)

        # ----- soft‑max weights (stable) -----
        max_c = conv.max()
        shifted = conv - max_c
        w = np.exp(t * shifted)          # all entries ≤ 1, no overflow
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        w_rev = w[::-1]                  # reverse for cross‑convolution
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        start_idx = w_rev.size - 1       # = 2·n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- mirror‑descent update (KL‑projection) -----
        a_new = a * np.exp(-eta * grad)  # element‑wise multiplicative step
        a_new = proj_simplex(a_new)

        # occasional random jitter to keep exploration alive
        if rng.random() < 0.01:
            a_new = _random_block_perturb(a_new, rng)

        cur_score = compute_J(a_new, n, fft_len_self, conv_len)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a_new.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] anneal new best J = {best_score:.6f}")

        # restart after 20 s without improvement
        if now - last_improve > 20.0:
            a_new = best_vec * np.exp(rng.normal(0.0, 0.05, size=n))
            a_new = proj_simplex(a_new)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] anneal restart – best J = {best_score:.6f}")

        a = a_new

    return best_vec, best_score


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Main entry point required by the competition framework.
    Returns a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` while respecting the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(1024, min(5000, n0))          # stay in a moderate range
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for a⋆w_rev

    # --------------------------------------------------------------
    # generate a diverse initial pool
    # --------------------------------------------------------------
    candidates: list[np.ndarray] = _initial_candidates(n, rng)

    # add a handful of random Fourier‑series candidates
    for _ in range(30):
        candidates.append(random_fourier_candidate(n, rng, max_freq=15, scale=0.2))

    # inject the previously discovered construction (if any)
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, prev.size)
            x_new = np.linspace(0.0, 1.0, n)
            prev = np.interp(x_new, x_old, prev)
        candidates.append(proj_simplex(prev))

    # --------------------------------------------------------------
    # evaluate the pool and keep the best few
    # --------------------------------------------------------------
    best_vec = None
    best_score = float("inf")
    scored_cands = []

    for v in candidates:
        v = proj_simplex(v)            # guarantee sum=1 and non‑negativity
        sc = compute_J(v, n, fft_len_self, conv_len)
        scored_cands.append((sc, v))
        if sc < best_score:
            best_score = sc
            best_vec = v.copy()

    print(f"[0.00s] initial best J = {best_score:.6f}")

    # keep the top 5 vectors for further refinement
    scored_cands.sort(key=lambda x: x[0])
    to_refine = [vec for _, vec in scored_cands[:5]]

    # --------------------------------------------------------------
    # annealed soft‑max refinement (phase 2)
    # --------------------------------------------------------------
    for idx, vec in enumerate(to_refine):
        now = time.monotonic()
        if now >= deadline:
            break

        # allocate remaining time proportionally to the remaining vectors
        remaining_vectors = len(to_refine) - idx
        sub_deadline = now + (deadline - now) / remaining_vectors * 0.95

        refined_vec, refined_score = annealing_descent(
            vec,
            n,
            fft_len_self,
            conv_len,
            fft_len_grad,
            sub_deadline,
            rng,
        )
        if refined_score < best_score:
            best_score = refined_score
            best_vec = refined_vec.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] refined best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # final sanity checks and return
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() < 0.01:               # protect against an empty vector
        best_vec[:] = 1.0 / n
    best_vec = proj_simplex(best_vec)

    return [float(x) for x in best_vec]
