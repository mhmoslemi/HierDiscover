def adam_optimize(a_init, num_steps, lr_start=0.5, lr_end=0.001, alpha_start=10.0, alpha_end=5000.0):
    a = a_init.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    for it in range(1, num_steps+1):
        frac = it / num_steps
        lr = lr_start * (1-frac) + lr_end * frac
        alpha = alpha_start * (1-frac) + alpha_end * frac

        # compute conv
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        c_max = conv.max()
        # softmax weights
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum ==0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        # gradient
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
        start_idx = w_rev.size - 1
        grad = 2.0 * full[start_idx:start_idx + n]

        # Adam update
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**it)
        v_hat = v / (1 - beta2**it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project simplex
        a = proj_simplex(a)
    return a

"""
Propose a step‑function (non‑negative heights) that minimizes
``evaluate_sequence``.  

The strategy consists of three stages:

1️⃣  **Random‑seeded projected‑Adam** – many short Adam runs (with a
    soft‑max approximation of the max‑convolution) start from random
    Dirichlet vectors.  The best vector among those runs is kept.

2️⃣  **Long refinement** – the current best vector is refined with a
    longer Adam run (exactly the same optimiser that was used in the
    previous baseline).

3️⃣  **Swap‑local‑search** – a cheap combinatorial phase swaps two
    positions of the best vector; a swap is kept if it reduces the
    true objective (computed via a fast FFT convolution).  This step
    often yields a small extra reduction.

The whole routine respects the 1000 s budget (a 5 s safety margin is
kept).  All numbers are projected onto the probability simplex, thus
the evaluation formula simplifies to `2·n·max(conv)`.  The final
sequence is clipped to the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_dirichlet(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a Dirichlet‑(1,…,1) vector (i.e. normalised exponentials)."""
    g = rng.exponential(1.0, size=n)
    if g.sum() == 0:
        return np.full(n, 1.0 / n)
    return g / g.sum()


def _compute_J(
    a: np.ndarray,
    n: int,
    conv_len: int,
    fft_len_self: int,
) -> float:
    """
    Fast objective  J = 2·n·max(a*a)  (the sum of `a` is assumed to be 1).
    """
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    fft_len_self: int,
    fft_len_grad: int,
    steps: int,
    rng: np.random.Generator,
    lr_start: float = 0.5,
    lr_end: float = 0.001,
    alpha_start: float = 10.0,
    alpha_end: float = 5000.0,
    block_perturb_prob: float = 0.02,
) -> np.ndarray:
    """
    Projected Adam optimiser that minimises the smoothed maximum of the
    self‑convolution.  Returns the final vector (still summing to 1).
    """
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    for it in range(1, steps + 1):
        frac = it / steps
        lr = lr_start * (1.0 - frac) + lr_end * frac
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ----- self‑convolution & soft‑max weights -----
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        cmax = conv.max()
        shifted = conv - cmax                # for numerical stability
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smoothed objective -----
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
        start_idx = w_rev.size - 1                # = 2·n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- Adam update + projection onto simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # occasional block perturbation to escape plateaus
        if rng.random() < block_perturb_prob:
            width = rng.integers(1, max(2, a.size // 10))
            start = rng.integers(0, a.size - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a[start:start + width] *= factor
            a = proj_simplex(a)

    return a


def _swap_local_search(
    a: np.ndarray,
    n: int,
    conv_len: int,
    fft_len_self: int,
    time_deadline: float,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Simple swap‑based hill climbing.  Swaps two positions; the swap is
    kept if it reduces the true objective.  A tiny probability of
    uphill moves (temperature 1e‑4) helps crossing flat plateaus.
    """
    best = a.copy()
    best_J = _compute_J(best, n, conv_len, fft_len_self)

    cur = best.copy()
    cur_J = best_J

    temp = 1e-4
    attempts = 0
    accepts = 0

    while time.monotonic() < time_deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        cand = cur.copy()
        cand[i], cand[j] = cand[j], cand[i]

        cand_J = _compute_J(cand, n, conv_len, fft_len_self)

        if cand_J < cur_J - 1e-12:
            cur, cur_J = cand, cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best, best_J = cand.copy(), cand_J
                print(f"[{time.monotonic():6.2f}s] swap new best J = {best_J:.6f}")
        else:
            # tiny uphill acceptance probability
            if rng.random() < np.exp(-(cand_J - cur_J) / temp):
                cur, cur_J = cand, cand_J
                accepts += 1

        # occasional heavy perturbation
        if attempts % 25000 == 0:
            cur = proj_simplex(cur * np.exp(rng.normal(0.0, 0.10, size=n)))

    print(f"Swap phase – attempts: {attempts}, accepts: {accepts}, best J = {best_J:.6f}")
    return best

def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine:
    * draws a large number of random Dirichlet seeds,
    * runs a short projected‑Adam optimisation from each seed,
    * keeps the best intermediate result,
    * refines that result with a longer Adam run,
    * finishes with a cheap swap‑local‑search.
    The function returns the best sequence found before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem dimensions
    # --------------------------------------------------------------
    # If a previous construction exists, reuse its length; otherwise pick a
    # reasonable default that is fast to handle.
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)
            n = base.size
        except Exception:
            base = None
            n = 2048
    else:
        base = None
        n = 2048

    # keep length modest (avoid >5000 for speed)
    n = max(512, min(n, 4096))

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT sizes
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)          # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)          # for a * w_rev

    # --------------------------------------------------------------
    # 3) Helper to evaluate the (scaled) objective quickly
    # --------------------------------------------------------------
    def compute_J(vec: np.ndarray) -> float:
        return _compute_J(vec, n, conv_len, fft_len_self)

    # --------------------------------------------------------------
    # 4) Initial best – either the provided seed or a uniform
    # --------------------------------------------------------------
    if base is not None and base.size == n:
        a0 = base.copy()
        a0 = proj_simplex(a0 / a0.sum())
        print(f"[init] Using provided sequence (len={n})")
    else:
        a0 = np.full(n, 1.0 / n)
        print(f"[init] No good seed – using uniform (len={n})")

    best_vec = a0.copy()
    best_J = compute_J(best_vec)
    print(f"[init] J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5) Phase 1 – many short random‑seeded Adam runs
    # --------------------------------------------------------------
    short_steps = 180               # cheap runs, enough to get a gradient signal
    short_time_budget = (deadline - time.monotonic()) * 0.65   # ~65 % of remaining time
    min_time_per_run = 0.25        # rough lower bound (seconds) for one short run

    while time.monotonic() + min_time_per_run < deadline - (deadline - time.monotonic()) * 0.30:
        # generate a random Dirichlet start (concentration=1 → uniform‑like)
        a_start = _random_dirichlet(rng, n)

        # run a short Adam optimisation
        a_opt = _adam_optimize(
            a_start,
            n,
            conv_len,
            fft_len_self,
            fft_len_grad,
            short_steps,
            rng,
        )

        cur_J = compute_J(a_opt)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a_opt.copy()
            print(f"[{time.monotonic():6.2f}s] New best after short run J = {best_J:.6f}")

        # break if we are about to run out of the allocated short‑run budget
        if time.monotonic() - (deadline - short_time_budget) > short_time_budget:
            break

    # --------------------------------------------------------------
    # 6) Phase 2 – long refinement from the current best
    # --------------------------------------------------------------
    # Reserve ~20 s for the final permutation phase
    long_deadline = deadline - 20.0
    if time.monotonic() < long_deadline:
        long_steps = 2500
        print(f"[refine] Starting long Adam run (steps={long_steps})")
        best_vec = _adam_optimize(
            best_vec,
            n,
            conv_len,
            fft_len_self,
            fft_len_grad,
            long_steps,
            rng,
        )
        best_J = compute_J(best_vec)
        print(f"[refine] After long run J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 7) Phase 3 – permutation (swap) local search
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        best_vec = _swap_local_search(
            best_vec,
            n,
            conv_len,
            fft_len_self,
            deadline,
            rng,
        )
        best_J = compute_J(best_vec)
        print(f"[final] Best J after all phases = {best_J:.6f}")

    # --------------------------------------------------------------
    # 8) Return a list clipped to the limits enforced by evaluate_sequence
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
