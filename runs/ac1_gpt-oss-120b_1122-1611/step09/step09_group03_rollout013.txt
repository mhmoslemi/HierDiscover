"""
Optimization of the step‑function bound.

The algorithm consists of three stages:

1. **Global parametric sweep** – we evaluate a few families of
   analytically‑parameterised sequences (exponential, polynomial,
   cosine‑shaped, logistic).  The sweep is cheap and quickly produces
   candidates better than the uniform construction.

2. **Multi‑restart Adam optimisation** – from each promising candidate
   we run a short projected Adam optimiser on the smooth soft‑max
   surrogate of the max‑convolution.  The optimiser works directly on
   the simplex (the vector is projected after each update).  The
   temperature of the soft‑max is linearly increased so that the
   surrogate approaches the true max.  Multiple random restarts are
   performed until the time budget runs out.

3. **Final polishing** – the best vector found is refined a little
   more with a higher‑resolution schedule and then returned.

The whole routine respects the 1000‑second budget, uses only NumPy
(and the already‑imported ``evaluate_sequence``) and prints progress
information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(a: np.ndarray) -> float:
    """Exact objective 2·n·max(convolution) for a vector that sums to 1."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * np.max(conv))


# ----------------------------------------------------------------------
# Parametric families of non‑negative sequences (all sum to 1)
# ----------------------------------------------------------------------


def shape_exponential(n: int, c: float) -> np.ndarray:
    """
    a_i ∝ r^i  with  r = exp(-c / n) .
    The normalisation factor is chosen analytically.
    """
    gamma = c / n
    r = np.exp(-gamma)
    # avoid loss of precision for very small gamma
    denom = 1.0 - np.exp(-c)          # 1 - r**n
    if denom == 0.0:
        denom = 1e-12
    lam = (1.0 - r) / denom
    i = np.arange(n, dtype=np.float64)
    a = lam * np.exp(-gamma * i)
    # renormalise to guard against floating errors
    a /= a.sum()
    return a


def shape_polynomial(n: int, p: float) -> np.ndarray:
    """a_i ∝ (1 - i/(n‑1))^p , i = 0,…,n‑1."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - i / (n - 1)) ** p
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


def shape_cosine(n: int, p: float) -> np.ndarray:
    """a_i ∝ cos(π·i / (2·(n‑1)))^p , a non‑negative half‑cosine."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / (2.0 * (n - 1))) ** p
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


def shape_logistic(n: int, s: float) -> np.ndarray:
    """a_i ∝ 1/(1+exp((i‑μ)/s)) with μ centred."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp((i - mu) / s))
    if a.sum() == 0.0:
        a[0] = 1.0
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Core optimisation routine (Adam on a soft‑max surrogate)
# ----------------------------------------------------------------------


def _adam_optimize(a0: np.ndarray,
                   max_iter: int,
                   rng: np.random.Generator,
                   lr: float = 0.1,
                   alpha_end: float = 20000.0) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the smooth soft‑max surrogate of the
    max‑convolution.  Returns the best vector encountered and its J‑value.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(3 * n - 2)
    start_idx = (fft_len_grad - conv_len) // 2

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_J = compute_J(a)
    best_a = a.copy()

    for t in range(1, max_iter + 1):
        # ----- convolution of a with itself -----
        fa_self = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa_self * fa_self, n=fft_len_self)[:conv_len]

        # ----- soft‑max weights (surrogate of max) -----
        alpha = 1.0 + (alpha_end - 1.0) * (t - 1) / (max_iter - 1)
        shift = conv - conv.max()
        w = np.exp(alpha * shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the surrogate -----
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = proj_simplex(a + step)
        J_candidate = compute_J(a_candidate)

        # keep the best point seen so far
        if J_candidate < best_J:
            best_J, best_a = J_candidate, a_candidate.copy()

        # always move – Adam is an exploratory optimiser
        a = a_candidate

    return best_a, best_J


# ----------------------------------------------------------------------
# Public entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine performs a quick
    parametric sweep followed by many short Adam restarts with random
    initialisations.  The best vector found before the time budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)

    # safety margin – leave a few seconds for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Global parametric sweep (very cheap)
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096]          # candidate sizes
    best_a = None
    best_J = float('inf')

    print("[0.00s] Starting parametric sweep")
    for n in lengths:
        # Uniform baseline
        a = np.full(n, 1.0 / n)
        J = compute_J(a)
        if J < best_J:
            best_J, best_a = J, a.copy()
            print(f"[---] n={n:4d} uniform      J={J:.6f}")

        # Exponential family
        for c in np.linspace(0.5, 3.0, 30):
            a = shape_exponential(n, c)
            J = compute_J(a)
            if J < best_J:
                best_J, best_a = J, a.copy()
                print(f"[---] n={n:4d} exponential c={c:4.2f} J={J:.6f}")

        # Polynomial family
        for p in np.linspace(0.5, 6.0, 30):
            a = shape_polynomial(n, p)
            J = compute_J(a)
            if J < best_J:
                best_J, best_a = J, a.copy()
                print(f"[---] n={n:4d} polynomial  p={p:4.2f} J={J:.6f}")

        # Cosine family
        for p in np.linspace(0.5, 6.0, 30):
            a = shape_cosine(n, p)
            J = compute_J(a)
            if J < best_J:
                best_J, best_a = J, a.copy()
                print(f"[---] n={n:4d} cosine      p={p:4.2f} J={J:.6f}")

        # Logistic family
        for s in np.linspace(0.1, 3.0, 20):
            a = shape_logistic(n, s)
            J = compute_J(a)
            if J < best_J:
                best_J, best_a = J, a.copy()
                print(f"[---] n={n:4d} logistic    s={s:4.2f} J={J:.6f}")

    # ------------------------------------------------------------------
    # 2) Multi‑restart Adam optimisation
    # ------------------------------------------------------------------
    restart = 0
    while time.monotonic() < deadline:
        restart += 1
        # choose a length (favor larger sizes for finer resolution)
        probs = np.array([1, 2, 4, 4], dtype=float)
        probs = probs / probs.sum()
        n = rng.choice(lengths, p=probs)

        # initialise – either random Dirichlet or the current global best
        if best_a is not None and best_a.shape[0] == n and rng.random() < 0.3:
            a0 = best_a.copy()
        else:
            a0 = rng.dirichlet(np.full(n, 0.5))
            a0 = proj_simplex(a0)

        # a short run of Adam; the number of iterations adapts to the
        # remaining time budget (max 1500, but never exceed the time left)
        max_iter = min(1500, int(1e5 * (deadline - time.monotonic()) / (budget_s - 5)))
        max_iter = max(200, max_iter)   # at least a few hundred steps

        a_opt, J_opt = _adam_optimize(a0, max_iter, rng)

        if J_opt < best_J:
            best_J, best_a = J_opt, a_opt.copy()
            elapsed = time.monotonic() - (deadline - budget_s + 5.0)
            print(f"[{elapsed:6.2f}s] restart {restart:3d} new best n={n:4d} J={J_opt:.6f}")

        # early exit if we already beat the target bound
        if best_J <= 1.5025:
            print("[INFO] target reached – stopping early")
            break

    # ------------------------------------------------------------------
    # 3) Final polishing (a few more Adam steps on the best vector)
    # ------------------------------------------------------------------
    if best_a is not None:
        # a final, higher‑precision Adam run
        a_final, J_final = _adam_optimize(best_a, 2000, rng,
                                          lr=0.05, alpha_end=50000.0)
        if J_final < best_J:
            best_J, best_a = J_final, a_final.copy()
            print(f"[FIN] polishing improved J to {J_final:.6f}")

    # ------------------------------------------------------------------
    # 4) Return result (clip to allowed range – the evaluation function
    #    does the same, but we keep it tidy)
    # ------------------------------------------------------------------
    if best_a is None:
        raise RuntimeError("No candidate was generated – unexpected failure.")
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"[RESULT] final J = {best_J:.6f} (length {best_a.size})")
    return best_a.tolist()
