def _smooth_gradient(a, lam, n, fft_len, conv_len):
    Fa = np.fft.fft(a, n=fft_len)
    conv = np.fft.ifft(Fa * Fa).real[:conv_len]
    max_c = conv.max()
    # Compute weights w
    # stable computation
    scaled = lam * (conv - max_c)  # (conv-max) negative or zero for max
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    # compute F_w:
    Fw = np.fft.fft(w, n=fft_len)
    # gradient of J_s
    g_complex = np.fft.ifft(Fa * np.conj(Fw))
    # gradient of J = 4*n * real part
    g = 4.0 * n * np.real(g_complex[:n])
    # compute exact J (not smoothed)
    J_exact = 2.0 * n * max_c
    return J_exact, g, conv, max_c, w

"""
Proposed algorithm
------------------

A two‑phase optimizer for the step‑function objective
    J(a) = 2·n·max(a∗a)   (a lies on the simplex Σa_i = 1, a_i ≥ 0).

Phase 1 – Smoothed‑max Adam:
    * The non‑smooth max is replaced by a log‑sum‑exp (soft‑max) with a
      temperature λ that grows from 0.5 to ~20.  This yields a differentiable
      surrogate J_λ(a) = 2·n·LSE_λ(a∗a) whose gradient can be written in closed
      form using FFTs:
          ∇J_λ = 4·n·⟂FFT⁻¹(FFT(a)·conj(FFT(w))),
      where w are the soft‑max weights of the convolution.
    * Adam with a decaying learning‑rate (0.05 → 0.001) and simplex projection
      drives the search.  Random Dirichlet mixing and occasional jitter avoid
      stagnation.

Phase 2 – Local discrete refinement:
    * A short simulated‑annealing / mass‑transfer loop runs on the best
      vector found by Adam, using the exact max‑convolution sub‑gradient.
    * Restarts are triggered after 15 s of no improvement.

The routine respects a 1000 s budget (5 s safety margin) and returns the
best coefficient list encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator):
    """Generate a modest pool of diverse seeds, normalized to sum = 1."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        prev /= prev.sum()
        cand.append(prev)

    return cand


def _compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) for a already on the simplex."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _smooth_gradient(a: np.ndarray, lam: float,
                     n: int, fft_len: int, conv_len: int):
    """
    Return (J_exact, grad) where `grad` is the gradient of the
    smoothed objective J_λ = 2·n·LSE_λ(conv) and J_exact is the true
    objective 2·n·max(conv).  All FFT work is done here.
    """
    # FFT of a
    Fa = np.fft.fft(a, n=fft_len)

    # full self‑convolution (real part)
    conv_full = np.fft.ifft(Fa * Fa).real
    conv = conv_full[:conv_len]                     # length 2·n‑1
    max_c = conv.max()

    # soft‑max weights (stable)
    shifted = lam * (conv - max_c)                  # max entry → 0
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()             # length conv_len

    # FFT of weights (zero‑padded to fft_len)
    Fw = np.fft.fft(w, n=fft_len)

    # gradient of the smoothed J (see analysis)
    g_complex = np.fft.ifft(Fa * np.conj(Fw))
    grad = 4.0 * n * np.real(g_complex[:n])

    # exact (non‑smoothed) objective
    J_exact = 2.0 * n * max_c
    return J_exact, grad, conv, max_c, w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimizes
    ``evaluate_sequence``.  The method consists of a smoothed‑max Adam optimisation
    on the simplex followed by a short local simulated‑annealing refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size – aim for a power‑of‑two length (fast FFTs)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n0 = len(height_sequence_1)
    else:
        n0 = 2048
    n0 = max(512, min(n0, 4096))
    n = _next_pow2(n0)               # e.g. 3809 → 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)   # power‑of‑two for FFTs

    # --------------------------------------------------------------
    # 2) Initialise a diverse pool of seeds and keep the best so far
    # --------------------------------------------------------------
    seeds = _initial_candidates(n, rng)
    best_vec = None
    best_score = np.inf

    for idx, cand in enumerate(seeds):
        cand = proj_simplex(cand.astype(np.float64))
        sc = _compute_J(cand, n, fft_len, conv_len)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
        print(f"[{time.monotonic() - start_time:6.2f}s] seed {idx} J={sc:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – Smoothed‑max Adam on the best seed
    # --------------------------------------------------------------
    a = best_vec.copy()
    m = np.zeros_like(a)          # 1st Adam moment
    v = np.zeros_like(a)          # 2nd Adam moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.monotonic()

    # schedules
    lam_start, lam_end = 0.5, 20.0                     # temperature schedule
    lr_start, lr_end = 0.05, 0.001                    # learning‑rate schedule

    while time.monotonic() < deadline:
        it += 1
        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)      # 0 … 1

        # exponential growth of λ, exponential decay of LR
        lam = lam_start * ((lam_end / lam_start) ** frac)
        lr = lr_start * ((lr_end / lr_start) ** frac)

        # gradient of the smoothed objective
        J_exact, g, conv, max_c, w = _smooth_gradient(a, lam, n, fft_len, conv_len)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = proj_simplex(a_new)

        # exact objective after the step
        J_new = _compute_J(a_new, n, fft_len, conv_len)

        if J_new < best_score - 1e-12:
            best_score = J_new
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Adam it {it} new best J={best_score:.6f}")

        a = a_new

        # ------------------------------------------------------------------
        # Stagnation handling
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > 15.0:
            # re‑inject a tiny random perturbation around the current best
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            a = proj_simplex(a)
            m.fill(0.0)
            v.fill(0.0)
            it = 0
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Adam restart (stagnation)")

        # occasional Dirichlet mixing to jump out of deep valleys
        if rng.random() < 0.005:
            mix = 0.2
            dirich = rng.dirichlet(np.ones(n) * 0.3)
            trial = (1.0 - mix) * a + mix * dirich
            trial = proj_simplex(trial)
            J_trial = _compute_J(trial, n, fft_len, conv_len)
            if J_trial < J_new:
                a = trial
                print(f"[{time.monotonic() - start_time:6.2f}s] Dirichlet jump improved J={J_trial:.6f}")

    # --------------------------------------------------------------
    # 4) Phase 2 – Very short local simulated‑annealing refinement
    # --------------------------------------------------------------
    # Use any remaining time for a few mass‑transfer moves.
    remaining = deadline - time.monotonic()
    if remaining > 2.0:
        a = best_vec.copy()
        cur_J = best_score
        last_temp_improve = time.monotonic()
        T_start, T_end = 0.08, 1e-4
        sa_deadline = time.monotonic() + remaining

        while time.monotonic() < sa_deadline:
            # linear temperature cooling (based on SA elapsed fraction)
            frac_sa = (time.monotonic() - (deadline - remaining)) / remaining
            T = T_start * (1.0 - frac_sa) + T_end * frac_sa

            # propose a simple mass‑transfer move
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
            eps = min(a[i], rng.exponential(scale=0.01))
            if eps <= 0.0:
                continue
            a_new = a.copy()
            a_new[i] -= eps
            a_new[j] += eps
            a_new = np.maximum(a_new, 0.0)
            s = a_new.sum()
            if s == 0.0:
                a_new[:] = 1.0 / n
            else:
                a_new /= s

            J_new = _compute_J(a_new, n, fft_len, conv_len)
            delta = J_new - cur_J
            accept = False
            if delta < 0.0:
                accept = True
            elif rng.random() < np.exp(-delta / max(T, 1e-12)):
                accept = True

            if accept:
                a = a_new
                cur_J = J_new
                if J_new < best_score - 1e-12:
                    best_score = J_new
                    best_vec = a.copy()
                    last_temp_improve = time.monotonic()
                    print(f"[{last_temp_improve - start_time:6.2f}s] SA new best J={best_score:.6f}")

            # occasional restart after stagnation
            if time.monotonic() - last_temp_improve > 20.0:
                a = best_vec.copy()
                a = _random_block_perturb(a, rng)
                a = proj_simplex(a)
                cur_J = _compute_J(a, n, fft_len, conv_len)
                last_temp_improve = time.monotonic()
                print(f"[{last_temp_improve - start_time:6.2f}s] SA restart (stagnation)")

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
