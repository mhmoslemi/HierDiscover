import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]                     # sort descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_a(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Initialise the probability vector.
    If a previous construction exists it is used (padded / truncated as
    required); otherwise a uniform vector is returned.
    """
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        if a.size < n:
            a = np.concatenate([a, np.zeros(n - a.size, dtype=np.float64)])
        elif a.size > n:
            a = a[:n]
        a = proj_simplex(a)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Numerically stable soft‑max on the convolution values."""
    max_b = b.max()
    exp_term = np.exp(beta * (b - max_b))
    w = exp_term / exp_term.sum()
    return w


def _compute_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full self‑convolution a⋆a using an FFT of length ``fft_len``."""
    Fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(Fa * Fa, n=fft_len)[:conv_len]
    return conv


def _correlate_w_a(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation of w (length 2n‑1) with a (length n) returning
    the ``valid`` part of length n:
        g_i = Σ_j w_{i+j}·a_j  ,  i = 0…n‑1
    Implemented with NumPy's highly optimised C routine.
    """
    # np.correlate flips the second argument internally, which is exactly
    # the definition we need for the “valid” correlation.
    return np.correlate(w, a, mode='valid')


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method uses a smoothed‑max gradient descent
    with a simplex projection and occasional random jumps.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # keep the size moderate to keep FFT cheap
        n = max(512, min(n, 4096))
    else:
        n = 2048                     # fallback size
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)   # FFT length for a⋆a

    # ------------------------------------------------------------------
    # 2) Initialise a probability vector on the simplex
    # ------------------------------------------------------------------
    a = _initial_a(n, rng)
    best_a = a.copy()
    # compute initial objective
    b = _compute_convolution(a, fft_len_conv, conv_len)
    best_score = float(2.0 * n * b.max())
    print(f"[{time.monotonic() - start_time:6.2f}s] Initial J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Hyper‑parameter schedules
    # ------------------------------------------------------------------
    # soft‑max temperature (β) – start low, increase to sharpen the max.
    beta_start, beta_end = 0.5, 30.0
    # learning‑rate – start relatively large, decay to a fine step.
    lr_start, lr_end = 0.05, 0.001
    # momentum for the gradient (simple exponential moving average)
    momentum = np.zeros_like(a)
    mom_coeff = 0.9

    # ------------------------------------------------------------------
    # 4) Main optimisation loop
    # ------------------------------------------------------------------
    it = 0
    last_improve_time = start_time
    stagnation_counter = 0

    while time.monotonic() < deadline:
        it += 1
        # --------------------------------------------------------------
        # progress‑dependent schedules
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        total_allowed = deadline - start_time
        prog = elapsed / total_allowed

        beta = beta_start + prog * (beta_end - beta_start)
        lr = lr_start + prog * (lr_end - lr_start)

        # --------------------------------------------------------------
        # a) self‑convolution and current score
        # --------------------------------------------------------------
        b = _compute_convolution(a, fft_len_conv, conv_len)
        cur_max = b.max()
        cur_score = float(2.0 * n * cur_max)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve_time = time.monotonic()
            stagnation_counter = 0
            print(f"[{last_improve_time - start_time:6.2f}s] New best J = {best_score:.6f}")

        # --------------------------------------------------------------
        # b) smooth the max with a soft‑max → weights w_k
        # --------------------------------------------------------------
        w = _softmax_weights(b, beta)            # sum(w)=1, length=2n‑1

        # --------------------------------------------------------------
        # c) gradient of the smoothed objective:
        #    g = 4·n·corr(w, a)
        # --------------------------------------------------------------
        corr = _correlate_w_a(w, a)               # length n
        grad = 4.0 * n * corr

        # --------------------------------------------------------------
        # d) momentum + projected gradient step
        # --------------------------------------------------------------
        momentum = mom_coeff * momentum + (1.0 - mom_coeff) * grad
        a_new = a - lr * momentum
        a_new = proj_simplex(a_new)

        # --------------------------------------------------------------
        # e) occasional random Dirichlet jumps (escape from plateaus)
        # --------------------------------------------------------------
        if rng.random() < 0.003:
            p = 0.12
            dirich = rng.dirichlet(np.ones(n) * 0.2)
            a_new = (1.0 - p) * a_new + p * dirich
            a_new = proj_simplex(a_new)

        a = a_new

        # --------------------------------------------------------------
        # f) stagnation handling
        # --------------------------------------------------------------
        if time.monotonic() - last_improve_time > 20.0:
            stagnation_counter += 1
            # after a while, inject a tiny random perturbation around the best
            if stagnation_counter % 3 == 0:
                perturb = np.exp(rng.normal(0.0, 0.07, size=n))
                a = best_a * perturb
                a = proj_simplex(a)
                momentum.fill(0.0)
                last_improve_time = time.monotonic()
                print(f"[{last_improve_time - start_time:6.2f}s] Restart after stagnation")

    # ------------------------------------------------------------------
    # 5) Return the best sequence found (clipped to limits of the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
```

This implementation replaces the previous simulated‑annealing + Adam pipeline with a smooth‑max gradient descent that directly attacks the max‑convolution objective.  
The temperature β gradually sharpens the soft‑max, turning the smooth objective into the true max, while the learning‑rate decays and a simple momentum term accelerate convergence. Random Dirichlet jumps and occasional perturbations prevent the optimizer from getting trapped in local minima. The routine respects the 1000 s budget (with a 5 s safety margin) and always returns the best sequence encountered.
