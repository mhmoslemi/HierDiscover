"""
Propose a step‑function using an alternating linear‑program / mixing scheme.
The algorithm:
1. Start from a simplex‑projected vector (previous best if available).
2. Repeatedly
   * compute M = max(f∗f),
   * solve the LP  max 1ᵀg  s.t. g≥0, f∗g ≤ M,
   * normalise g to sum = 1,
   * replace f ← (1‑t) f + t g and re‑project onto the simplex.
   The mixing factor t is gradually decreased.
3. When the time budget is almost exhausted a short projected Adam
   refinement (same surrogate as the reference code) is performed.
The LP is built with a sparse Toeplitz matrix (HiGHS solver) – each
iteration is cheap enough to allow several hundred updates within the
1000 s budget.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex (sum(v)=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) (the vector is assumed to sum to 1)."""
    conv = self_convolution(a, fft_len_self, conv_len)
    return float(2.0 * n * conv.max())

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The method uses an alternating linear‑program
    / mixing scheme followed by a short Adam refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (0.5 s safety margin)
    # --------------------------------------------------------------
    now = time.monotonic()
    deadline = now + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n_guess = len(globals()["height_sequence_1"])
            n = max(1024, min(5000, n_guess))
        except Exception:
            n = 3000
    else:
        n = 3000

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2) Initial vector (simplex‑projected)
    # --------------------------------------------------------------
    # start from the previously known construction if it exists,
    # otherwise use a uniform distribution.
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, seq.size)
            x_new = np.linspace(0, 1, n)
            seq = np.interp(x_new, x_old, seq)
        f = proj_simplex(seq)
    else:
        f = np.full(n, 1.0 / n, dtype=np.float64)

    best_f = f.copy()
    best_J = compute_J(best_f, n, fft_len_self, conv_len)
    print(f"[{0.0:6.2f}s] start – J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3) Alternating LP / mixing loop
    # --------------------------------------------------------------
    # mixing factor schedule – start aggressive and anneal
    t0, t_final = 0.3, 0.05
    # number of outer LP iterations (guarded by time)
    lp_iter = 0
    while time.monotonic() < deadline:
        lp_iter += 1
        # compute current bound M = max(f∗f)
        conv_f = self_convolution(f, fft_len_self, conv_len)
        M = conv_f.max()

        # build sparse Toeplitz matrix  A_ub  (rows = conv_len, cols = n)
        # A_ub @ g  =  f ∗ g   (linear convolution, zero‑padded)
        A_ub = sp.diags(f, offsets=np.arange(n), shape=(conv_len, n), format='csr')
        b_ub = np.full(conv_len, M, dtype=np.float64)

        # solve LP:   max 1ᵀg   s.t.  A_ub g ≤ M ,  g ≥ 0
        # linprog solves a minimisation; we minimise -1ᵀg.
        res = opt.linprog(c=-np.ones(n),
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=(0, None),
                          method='highs',
                          options={'presolve': True})
        if not res.success:
            # LP failed – perturb the current vector a little and continue
            print(f"[{time.monotonic()-now:6.2f}s] LP {lp_iter} failed, jittering")
            f = _random_block_perturb(f, rng)
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            # nothing useful was found – skip mixing
            continue

        # normalise g to lie on the simplex (same total mass as f)
        g = g / sum_g

        # mixing factor annealed linearly between t0 and t_final
        elapsed = time.monotonic() - now
        frac = elapsed / (deadline - now)
        t = t0 * (1.0 - frac) + t_final * frac

        f_candidate = (1.0 - t) * f + t * g
        f_candidate = proj_simplex(f_candidate)

        J_candidate = compute_J(f_candidate, n, fft_len_self, conv_len)

        if J_candidate < best_J - 1e-12:
            best_J = J_candidate
            best_f = f_candidate.copy()
            f = f_candidate
            print(f"[{time.monotonic()-now:6.2f}s] LP {lp_iter} new best J = {best_J:.6f}")
        else:
            # keep the old f, but occasionally accept a worse move to
            # escape flat regions
            if rng.random() < 0.02:
                f = f_candidate
            # occasional random block jitter to avoid stalling
            if rng.random() < 0.01:
                f = _random_block_perturb(f, rng)

        # stop after a reasonable number of LP steps if we are close to the deadline
        if time.monotonic() > deadline - 5.0:
            break

    # --------------------------------------------------------------
    # 4) Short Adam refinement (soft‑max surrogate) – optional
    # --------------------------------------------------------------
    # If there is still some time left we run a few Adam steps to polish the
    # best vector found so far.
    remaining = deadline - time.monotonic()
    if remaining > 3.0:
        adam_deadline = time.monotonic() + remaining
        a = best_f.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        it = 0
        start_adam = time.monotonic()
        total_adam = adam_deadline - start_adam
        while time.monotonic() < adam_deadline:
            it += 1
            now_i = time.monotonic()
            frac = (now_i - start_adam) / total_adam
            lr = 0.25 * (1.0 - frac) + 0.001 * frac
            alpha = 2.0 * (1.0 - frac) + 3000.0 * frac

            # linear convolution a∗a
            conv = self_convolution(a, fft_len_self, conv_len)

            # soft‑max weights over the convolution entries
            c_max = conv.max()
            shifted = conv - c_max
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed surrogate:
            # grad = 2 * (a ⋆ w_rev)   (circular‑like convolution)
            w_rev = w[::-1]
            fft_len_grad = _next_pow2(3 * n - 2)
            fa = np.fft.rfft(a, n=fft_len_grad)
            fw = np.fft.rfft(w_rev, n=fft_len_grad)
            full = np.fft.irfft(fa * fw, n=fft_len_grad)
            start_idx = w_rev.size - 1
            grad = 2.0 * full[start_idx:start_idx + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = proj_simplex(a)

            cur_J = compute_J(a, n, fft_len_self, conv_len)
            if cur_J < best_J - 1e-12:

                best_f = a.copy()
                print(f"[{now_i - now:6.2f}s] Adam new best J = {best_J:.6f}")

            # occasional jitter to keep exploration alive
            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)

    # --------------------------------------------------------------
    # 5) Return the best sequence
    # --------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)
    return [float(x) for x in best_f]

Algorithm
---------
1. Start from a simplex‑projected vector (the previous best if it exists,
   otherwise uniform).
2. Repeat while time permits
   * Compute M = max(f∗f) (linear self‑convolution).

        maximise   Σ g_j
        subject to  f∗g ≤ M   (linear convolution, element‑wise)
                    g ≥ 0

     LP in a few hundred ms.

   * Keep the best f seen so far.
3. If a small amount of time is left, run a few Adam steps on the
   soft‑max surrogate of the true objective (identical to the reference
   code) to polish the final vector.
4. Return the best vector, clipped to the limits of `evaluate_sequence`.

The LP‑mixing loop aggressively pushes “mass” into the vector while
respecting the current maximal convolution, and the annealed mixing

    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) (vector assumed to sum to 1)."""
    conv = self_convolution(a, fft_len_self, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method consists of an LP‑mixing phase
    followed by a short Adam refinement.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (0.5 s safety margin)
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # 1) Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n_guess = len(globals()["height_sequence_1"])
            n = max(1024, min(5000, n_guess))
        except Exception:
            n = 3000
    else:
        n = 3000

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2) Initial vector (projected onto the simplex)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        seq = np.asarray(height_sequence_1, dtype=np.float64)
        if seq.size != n:
            # linear interpolation to required length
            x_old = np.linspace(0, 1, seq.size)
            x_new = np.linspace(0, 1, n)
            seq = np.interp(x_new, x_old, seq)
        f = proj_simplex(seq)
    else:
        f = np.full(n, 1.0 / n, dtype=np.float64)

    best_f = f.copy()
    best_J = compute_J(best_f, n, fft_len_self, conv_len)
    print(f"[{0.0:6.2f}s] start – J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3) Alternating LP / mixing loop
    # --------------------------------------------------------------
    t0, t_final = 0.3, 0.05   # mixing factor annealing schedule
    lp_iter = 0
    while time.monotonic() < deadline:
        lp_iter += 1

        # current bound M = max(f∗f)
        conv_f = self_convolution(f, fft_len_self, conv_len)
        M = conv_f.max()

        # Build the Toeplitz matrix A_ub where (A_ub @ g) = f∗g
        # A_ub is (conv_len × n) and very sparse.
        A_ub = sp.diags(f, offsets=np.arange(n),
                        shape=(conv_len, n), format='csr')
        b_ub = np.full(conv_len, M, dtype=np.float64)

        # Linear program: maximise Σ g  ⇔  minimise −Σ g
        res = opt.linprog(c=-np.ones(n),
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=(0, None),
                          method='highs',
                          options={'presolve': True})

        if not res.success:
            # LP failed – jitter f a little and continue
            print(f"[{time.monotonic()-start:6.2f}s] LP {lp_iter} failed, jittering")
            f = _random_block_perturb(f, rng)
            continue

        g = res.x
        sum_g = g.sum()
        if sum_g == 0.0:
            # nothing useful discovered – break out of the LP loop
            break

        # Normalise g to the simplex (same total mass as f)
        g = g / sum_g

        # Annealed mixing factor
        elapsed = time.monotonic() - start
        frac = elapsed / (deadline - start)
        t = t0 * (1.0 - frac) + t_final * frac

        f_candidate = (1.0 - t) * f + t * g
        f_candidate = proj_simplex(f_candidate)

        J_candidate = compute_J(f_candidate, n, fft_len_self, conv_len)

        if J_candidate < best_J - 1e-12:
            best_J = J_candidate
            best_f = f_candidate.copy()
            f = f_candidate
            print(f"[{time.monotonic()-start:6.2f}s] LP {lp_iter} new best J = {best_J:.6f}")
        else:
            # occasional acceptance of a worse move to avoid stagnation
            if rng.random() < 0.02:
                f = f_candidate
            # random block jitter with small probability
            if rng.random() < 0.01:
                f = _random_block_perturb(f, rng)

        # leave a tiny margin for the final Adam phase
        if time.monotonic() > deadline - 5.0:
            break

    # --------------------------------------------------------------
    # 4) Short Adam refinement (optional, uses same surrogate as reference)
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 3.0:
        adam_deadline = time.monotonic() + remaining
        a = best_f.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        it = 0
        start_adam = time.monotonic()
        total_adam = adam_deadline - start_adam

        while time.monotonic() < adam_deadline:
            it += 1
            now_i = time.monotonic()
            frac = (now_i - start_adam) / total_adam
            lr = 0.25 * (1.0 - frac) + 0.001 * frac
            alpha = 2.0 * (1.0 - frac) + 3000.0 * frac

            # exact convolution a∗a
            conv = self_convolution(a, fft_len_self, conv_len)

            # soft‑max over the convolution entries
            c_max = conv.max()
            shifted = conv - c_max
            w = np.exp(alpha * shifted)
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # gradient of the smoothed surrogate:
            w_rev = w[::-1]
            fft_len_grad = _next_pow2(3 * n - 2)
            fa = np.fft.rfft(a, n=fft_len_grad)
            fw = np.fft.rfft(w_rev, n=fft_len_grad)
            full = np.fft.irfft(fa * fw, n=fft_len_grad)
            start_idx = w_rev.size - 1
            grad = 2.0 * full[start_idx:start_idx + n]

            # Adam update + projection onto simplex
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = proj_simplex(a)

            cur_J = compute_J(a, n, fft_len_self, conv_len)
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_f = a.copy()
                print(f"[{now_i - start:6.2f}s] Adam new best J = {best_J:.6f}")

            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)

    # --------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)
    return [float(x) for x in best_f]
