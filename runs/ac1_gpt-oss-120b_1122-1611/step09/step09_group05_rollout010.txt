"""
Gradient‑based simulated‑annealing optimiser for the step‑function problem.

The objective      J(a) = 2·n·max(conv(a,a)) / (∑a)²
is scale‑invariant, therefore we normalise the vector to a probability
simplex (∑a = 1).  The non‑smooth “max” is replaced by a smooth
log‑sum‑exp approximation with a temperature α, which yields an explicit
gradient

    ∇J ≈ 2·n·corr(a, w) ,   w_k ∝ exp(α·conv_k)

where “corr” denotes the linear correlation of a with the weight
vector w.  The gradient is computed with a single FFT (O(N log N)).  A
simple Metropolis acceptance rule with a linearly cooling temperature
guides the optimisation; occasional Gaussian jitter restarts the walk
when progress stalls.

The routine starts from a small portfolio of diverse simplex vectors
(uniform, Dirichlet draws, the previously found construction if
available) and keeps the best solution found before the time budget
expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, Σw = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def _conv_and_grad(a: np.ndarray,
                   alpha: float,
                   fft_len: int,
                   conv_len: int) -> tuple[float, np.ndarray]:
    """
    Returns (max_conv, grad) for the smooth approximation with parameter
    ``alpha``.  ``a`` is assumed to sum to 1.
    """
    # FFT of a (zero‑padded to fft_len)
    A = np.fft.rfft(a, n=fft_len)

    # Linear convolution a*a (full length)
    conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]

    # smooth max – log‑sum‑exp trick for stability
    scaled = alpha * conv_full
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    w = exp_scaled / np.sum(exp_scaled)          # soft‑max weights, sum = 1

    # FFT of the weight vector (same padding)
    W = np.fft.rfft(w, n=fft_len)

    # Correlation a ⋆ w  =  irfft(conj(A) * W)
    corr = np.fft.irfft(np.conj(A) * W, n=fft_len)[:a.size]

    # Gradient of 2·n·max(conv) approximated by the smooth max
    grad = 2.0 * a.size * corr   # a.size = n
    max_conv = np.max(conv_full)
    return max_conv, grad


def _objective(a: np.ndarray,
               conv_len: int,
               fft_len: int) -> float:
    """
    Exact objective J = 2·n·max(conv(a,a)), n = len(a).
    """
    A = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * a.size * np.max(conv_full))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------

    Search for a non‑negative step‑function (list of heights) that

    """

    # 3. Gradient‑based simulated annealing
    ######################################################################
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # temperature schedule (linear cooling)
    T_start = best_J * 0.05          # 5 % of current best
    T_end = 1e-8

    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time
    iteration = 0

    # smooth‑max schedule – start gentle, finish sharp
    alpha_start = 5.0
    alpha_end = 80.0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)

        # temperature for Metropolis acceptance
        T = T_start * (1.0 - frac) + T_end * frac

        # smooth‑max "temperature" (large α → closer to true max)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # learning‑rate proportional to temperature (large steps early)
        lr = 0.4 * (T / T_start) + 1e-4   # never zero

        # --------------------------------------------------------------
        # gradient of the smooth max
        # --------------------------------------------------------------
        max_conv, grad = _conv_and_grad(cur_vec, alpha, fft_len, conv_len)

        # propose a gradient step and project back onto the simplex
        proposal = cur_vec - lr * grad
        proposal = proj_simplex(proposal)

        # exact objective for the proposal
        new_J = _objective(proposal, conv_len, fft_len)

        # Metropolis acceptance
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            # probability of uphill move
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = proposal
            cur_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # Stall detection – inject a tiny random jitter around the best so far
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            cur_vec = best_vec * jitter
            cur_vec = proj_simplex(cur_vec)
            cur_J = _objective(cur_vec, conv_len, fft_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        iteration += 1

    # ----------------------------------------------------------------------
    # 4. Final sanitisation and return
    # ----------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s
    return [float(x) for x in best_vec]
