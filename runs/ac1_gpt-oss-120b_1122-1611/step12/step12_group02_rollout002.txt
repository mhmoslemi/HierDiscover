"""
Multi‑restart soft‑max subgradient descent.

We treat the step‑function as a probability vector (∑a_i = 1, a_i ≥ 0).
The objective that evaluate_sequence returns is
    J(a) = 2·n·max_k (a * a)[k] .
Since J is invariant under scaling we may work on the simplex.

The algorithm performs several independent restarts.
For each restart we run a projected sub‑gradient descent where the
gradient is taken w.r.t. a smooth approximation of the max‑convolution:
    f_β(a) = (1/β)·log Σ_k exp(β·(a*a)[k]) .
The gradient of f_β is a weighted sum of the correlations,
weights being the soft‑max of the current convolution.
β (the “inverse temperature”) is increased slowly so that the
approximation focuses on the true maximum.
Updates are additive (a ← a – η·∇) followed by Euclidean projection
onto the probability simplex (guaranteeing non‑negativity and unit sum).

Whenever the improvement stalls for ~30 s we inject a fresh random
Dirichlet vector to regain diversity.  The best vector seen across all
restarts is returned (clipped to the limits required by
evaluate_sequence).  The whole routine respects the 1000 s wall‑clock
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute the gradient of Σ_k weight_k·(a*a)[k] efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = np.full(n, 1.0 / n)
        init_vectors.append(v)

    # c) a few Dirichlet draws with varying concentration (sparseness)
    for alpha in (0.01, 0.03, 0.1, 0.3, 1.0):
        for _ in range(2):
            init_vectors.append(rng.dirichlet(np.ones(n) * alpha))

    # --------------------------------------------------------------
    # 4. Main multi‑restart loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # keep a small safety margin for final clean‑up / printing
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    best_vec = None
    best_score = np.inf
    restart_id = 0

    for init in init_vectors:
        if time.monotonic() >= deadline:
            break
        restart_id += 1
        a = init.copy()
        cur_score = _objective(a)

        # local best for this restart
        local_best_a = a.copy()
        local_best_score = cur_score

        # hyper‑parameters (will be annealed)
        eta = 0.12      # step size for additive update
        beta = 0.4      # inverse temperature for soft‑max
        min_eta = 0.02
        max_beta = 32.0

        last_improve = time.monotonic()
        iter_cnt = 0

        while time.monotonic() < deadline:
            # ------------------------------------------------------
            # 4.1 compute convolution and soft‑max weights
            # ------------------------------------------------------
            conv = _conv_full(a, fft_len, conv_len)
            max_c = conv.max()
            # soft‑max (stable version)
            exp_term = np.exp(beta * (conv - max_c))
            weight = exp_term / exp_term.sum()

            # ------------------------------------------------------
            # 4.2 gradient of the smoothed objective
            # ------------------------------------------------------
            grad = _weighted_gradient(a, weight, fft_len_corr, n)
            # clip gradient to avoid extreme exponentials (optional safety)
            grad = np.clip(grad, -10.0, 10.0)

            # ------------------------------------------------------
            # 4.3 additive update + projection onto simplex
            # ------------------------------------------------------
            a_new = proj_simplex(a - eta * grad)

            new_score = _objective(a_new)

            # ------------------------------------------------------
            # 4.4 accept if better (always keep best seen)
            # ------------------------------------------------------
            if new_score <= cur_score:
                a = a_new
                cur_score = new_score
                if new_score < local_best_score - 1e-12:
                    local_best_score = new_score
                    local_best_a = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] Restart {restart_id} new local best J = {new_score:.6f}")

            # ------------------------------------------------------
            # 4.5 annealing of hyper‑parameters
            # ------------------------------------------------------
            eta = max(eta * 0.9995, min_eta)      # slow decay
            beta = min(beta * 1.005, max_beta)    # gradual tightening

            iter_cnt += 1

            # ------------------------------------------------------
            # 4.6 stagnation handling
            # ------------------------------------------------------
            if time.monotonic() - last_improve > 30.0:
                # replace current vector with a fresh Dirichlet sample
                a = rng.dirichlet(np.ones(n) * 0.1)
                cur_score = _objective(a)
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] Restart {restart_id} jittered (stagnation)")

        # -----------------------------------------------------------
        # 5. Update global best after this restart
        # -----------------------------------------------------------
        if local_best_score < best_score - 1e-12:
            best_score = local_best_score
            best_vec = local_best_a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Global best updated to J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 6. Fallback (should never happen)
    # --------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_vec)

    # --------------------------------------------------------------
    # 7. Final sanitisation – clip, renormalise, convert to list
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s

    # (optional) sanity check using the provided evaluate_sequence
    try:
        val = evaluate_sequence(best_vec.tolist())
        print(f"Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    return [float(v) for v in best_vec]
