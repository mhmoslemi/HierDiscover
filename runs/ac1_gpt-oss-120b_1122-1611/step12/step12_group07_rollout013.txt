def simplex_projection(v):
    """Project v onto the probability simplex."""
    # ensure nonnegative after projection, result sums to 1.
    # Implementation: sort descending, find rho.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv/(np.arange(1, len(v)+1)) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative list) that minimises the
`evaluate_sequence` objective.

The search combines:
* a diverse set of deterministic and random initialisations,
* many cheap evaluations of the objective using FFT,
* an outer restart loop (jittered best or fresh Dirichlet draws),
* an inner exponentiated‑gradient descent that optimises a smooth
  approximation of the max‑convolution (soft‑max with increasing β).
  The update is a mirror‑descent step on the probability simplex
  (multiplicative update + renormalisation).

The method is deliberately different from the previous Adam‑on‑logits
approach and uses a simple exponential update together with aggressive
re‑starts.  It works for any length up to ~8000 (the whole budget is
kept within the 1000 s limit).  The best sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (FFT based convolutions, power‑of‑two helpers)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Result length = conv_len = 2*len(a)-1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Smooth approximation of max: soft‑max weights w_k = exp(β·conv_k) /
    Σ_j exp(β·conv_j).  The weights sum to 1 and give the gradient of the
    smooth max.
    """
    m = conv.max()
    shifted = beta * (conv - m)
    # prevent overflow / underflow
    shifted = np.clip(shifted, -700, 0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    return w


def _grad_smooth_max(a: np.ndarray, w: np.ndarray,
                     grad_fft_len: int) -> np.ndarray:
    """
    Gradient of smooth_max w.r.t. a.
    For each k, weight w_k multiplies the contribution of a shifted
    copy of a.  The computation is performed via FFT for O(n log n).
    """
    rev_a = a[::-1]                                 # reversed a
    W = np.fft.rfft(w, grad_fft_len)
    A_rev = np.fft.rfft(rev_a, grad_fft_len)
    full = np.fft.irfft(W * A_rev, n=grad_fft_len)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]               # slice the relevant part
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (number of steps)
    # --------------------------------------------------------------
    # Use the length of a previously found construction if available,
    # otherwise fall back to a power‑of‑two in the allowed range.
    if "height_sequence_1" in globals():
        try:
            n0 = len(height_sequence_1)
            n = max(256, min(n0, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    grad_fft_len = _next_pow2(3 * n - 2)          # for w * rev(a)

    # --------------------------------------------------------------
    # 2. Helper to compute the true objective quickly
    # --------------------------------------------------------------
    def _objective(a_vec: np.ndarray) -> float:
        conv = _conv_full(a_vec, fft_len, conv_len)
        maxb = conv.max()
        s = a_vec.sum()
        # Protect against pathological zero‑sum vectors (should not happen)
        if s < 1e-12:
            return np.inf
        return float(2.0 * n * maxb / (s * s))

    # --------------------------------------------------------------
    # 3. Build a diverse set of initial probability vectors
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # simple linear ramps
    idx = np.arange(n, dtype=np.float64)
    ramp_up = (idx + 1) / n
    ramp_up /= ramp_up.sum()
    init_candidates.append(ramp_up)
    ramp_down = (n - idx) / n
    ramp_down /= ramp_down.sum()
    init_candidates.append(ramp_down)

    # V‑shaped (large at the edges, small in the centre)
    v = 1.0 / ((idx + 1) * (n - idx))
    v = np.clip(v, 0, None)
    v /= v.sum()
    init_candidates.append(v)

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        g = r ** idx
        g /= g.sum()
        init_candidates.append(g)

    # Gaussian bumps of various widths
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        g = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        g /= g.sum()
        init_candidates.append(g)

    # Dirichlet draws (different concentrations)
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0):
        for _ in range(4):
            init_candidates.append(rng.dirichlet(np.full(n, alpha)))

    # previous construction, if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_candidates.append(v)

    # --------------------------------------------------------------
    # 4. Pick the best initial candidate (hard objective)
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        J = _objective(cand)
        if J < best_J:
            best_J = J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Main optimisation loop – many restarts + exponentiated gradient
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # hyper‑parameters for the inner exponentiated‑gradient phase
    eta0 = 0.8                # base step‑size for the multiplicative update
    beta0 = 2.0               # start smoothing parameter
    beta_max = 200.0
    beta_inc = 1.007          # geometric increase per iteration

    # keep a global best for the whole run
    global_best_a = best_a.copy()
    global_best_J = best_J

    # outer restart loop (creates a fresh start each time)
    run = 0
    while time.monotonic() < deadline:
        run += 1
        # ------------------------------------------------------------------
        # 5.1 Choose a start point: either jittered current best or a fresh draw
        # ------------------------------------------------------------------
        if rng.random() < 0.35 and global_best_a is not None:
            # jitter around the best solution seen so far
            log_vec = np.log(np.maximum(global_best_a, 1e-12))
            log_vec += rng.normal(0.0, 0.12, size=n)   # modest Gaussian jitter
            a = np.exp(log_vec - np.max(log_vec))
            a /= a.sum()
        else:
            # fresh Dirichlet draw with random concentration
            alpha = rng.choice([0.05, 0.1, 0.2, 0.5, 0.8, 1.0])
            a = rng.dirichlet(np.full(n, alpha))

        # ------------------------------------------------------------------
        # 5.2 Inner exponentiated‑gradient optimisation (fixed budget)
        # ------------------------------------------------------------------
        beta = beta0
        # number of inner steps is limited by remaining time
        max_inner = 800
        for it in range(max_inner):
            now = time.monotonic()
            if now >= deadline:
                break

            # convolution and smooth‑max weights
            conv = _conv_full(a, fft_len, conv_len)
            w = _smooth_max_weights(conv, beta)

            # gradient of the smooth‑max (via FFT)
            grad = _grad_smooth_max(a, w, grad_fft_len)

            # multiplicative (exponentiated) mirror‑descent update
            eta = eta0 / np.sqrt(it + 1.0)
            a = a * np.exp(-eta * grad)

            # renormalise onto the simplex
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a /= s

            # slowly increase the smoothing parameter β
            beta = min(beta_max, beta * beta_inc)

            # occasional exact evaluation (every ~30 inner steps)
            if (it & 31) == 0:
                cur_J = _objective(a)
                if cur_J < global_best_J - 1e-12:
                    global_best_J = cur_J
                    global_best_a = a.copy()
                    elapsed = now - start_time
                    print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # end of one restart
        # (no explicit stagnation handling – the outer loop will launch a new seed)

    # --------------------------------------------------------------
    # 6. Return the best sequence (clip, renormalise for safety)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total        # keep sum≈1 (scale‑invariant anyway)

    print(f"Finished. Best J = {global_best_J:.6f}")
    return [float(v) for v in best_vec]
