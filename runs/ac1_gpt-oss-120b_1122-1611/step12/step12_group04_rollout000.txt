"""
step‑function optimiser
-----------------------

The objective to minimise is

    J(a) = 2·n·max_k (a * a)[k]          (a lives on the probability simplex)

where `*` is the full linear convolution.
The function is homogeneous, therefore we can restrict `sum(a)=1`.

The optimiser works in three stages within the 1000 s budget:

1. **Exploratory Adam** – a smooth approximation of the max‑operator via a
   soft‑max with an exponentially increasing temperature β.  The gradient of
   the smoothed objective is obtained by an FFT‑based correlation.  This stage
   quickly discovers a good basin.

2. **Hard‑max sub‑gradient descent** – once the soft‑max is sharp we switch
   to the true sub‑gradient of the max‑entry (computed in O(n) without FFT)
   and continue with an Adam‑style update.  This directly pushes the worst
   convolution peak down.

3. **Local stochastic polishing** – a simulated‑annealing style pairwise mass
   swap that explores the neighbourhood of the best solution found so far.

All updates are projected onto the probability simplex to keep the
normalisation constraint.  The routine respects the 1000 s budget, prints
progress, and finally returns a plain Python list of non‑negative heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all zeros -> uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear auto‑convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Gradient of 2·∑_k w_k·(a*a)[k]  (w is a probability vector over the
    convolution indices).  Computed via FFT as 2·corr(weight, a).
    """
    a_rev = a[::-1]                         # reverse for correlation
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # the correlation result starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)).  a must already sum to 1."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _hard_gradient(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑entry (index k_max) of the convolution.
    For each i, derivative w.r.t a[i] is a[k_max-i] (if in range) plus
    a[i-k_max] (if in range).  The factor 2 accounts for the two symmetric
    contributions of the product a_i·a_j.
    """
    n = a.size
    g = np.zeros_like(a)

    # contribution a_{k_max-i}
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    if i_start <= i_end:
        i_rng = np.arange(i_start, i_end + 1, dtype=np.int64)
        j = k_max - i_rng
        g[i_rng] += a[j]

    # contribution a_{i-k_max}
    i_start2 = max(0, k_max)
    i_end2 = min(n - 1, k_max + n - 1)
    if i_start2 <= i_end2:
        i_rng2 = np.arange(i_start2, i_end2 + 1, dtype=np.int64)
        j2 = i_rng2 - k_max
        g[i_rng2] += a[j2]

    return 2.0 * g


def _random_dirichlet(rng: np.random.Generator, n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration `alpha`."""
    return rng.dirichlet(np.full(n, alpha))

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the allocated
    time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size (length of the vector)
    # ------------------------------------------------------------------
    # If a previous construction exists we try to enlarge it, otherwise we
    # start with a medium‑size power of two.
    if "height_sequence_1" in globals():
        try:
            n0 = len(height_sequence_1)
            n = max(2048, min(8192, n0 * 2))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlation

    # ------------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors (simplex)
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.94, 0.97, 0.985, 0.99, 0.995]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (both directions)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps of various widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 12.0, n / 9.0, n / 6.0, n / 4.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws with many concentrations (including sparse)
    for alpha in [0.02, 0.05, 0.1, 0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(_random_dirichlet(rng, n, alpha))

    # Very sparse random support (pick a random subset and give it equal mass)
    for _ in range(5):
        support = rng.choice(n, size=rng.integers(2, max(3, n // 200)), replace=False)
        v = np.zeros(n, dtype=np.float64)
        v[support] = 1.0
        v /= v.sum()
        init_cands.append(v)

    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        if v.size == n:
            init_cands.append(v)
        else:
            # interpolate or truncate to the current n
            if v.size > n:
                init_cands.append(v[:n] / v[:n].sum())
            else:
                # pad with zeros then re‑normalise
                tmp = np.zeros(n, dtype=np.float64)
                tmp[:v.size] = v
                init_cands.append(tmp / tmp.sum())

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate (fast exact objective)
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = _objective(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Stage‑1: Adam optimisation on a soft‑max smoothed max‑operator
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    # Allocate about 60 % of the budget to this stage
    stage1_deadline = start_time + 0.60 * max(0.0, budget_s)

    # Adam parameters
    lr0 = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Soft‑max schedule
    beta_start = 1.0
    beta_end = 4000.0         # stronger sharpening than the previous run

    # Metropolis temperature (linear decay)
    T_initial = best_J * 0.1
    T_final = 1e-9

    cur_vec = best_vec.copy()
    cur_J = best_J
    best_vec_stage = cur_vec.copy()
    best_J_stage = cur_J

    m = np.zeros_like(cur_vec)
    v_adam = np.zeros_like(cur_vec)

    iteration = 0
    last_improve = start_time

    while time.monotonic() < stage1_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total_stage = stage1_deadline - start_time
        t_frac = (now - start_time) / total_stage

        # ---- temperature schedule ----
        T = (T_initial * (stage1_deadline - now) + T_final * (now - start_time)) / total_stage

        # ---- convolution & soft‑max weights ----
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ---- keep global best ----
        if cur_J < best_J_stage - 1e-12:
            best_J_stage = cur_J
            best_vec_stage = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stage‑1 new best J = {best_J_stage:.6f}")

        # ---- stagnation handling (jitter) ----
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec = best_vec_stage * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = _objective(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stage‑1 jitter restart – best J = {best_J_stage:.6f}")

        # ---- soft‑max weighting ----
        beta = beta_start * (beta_end / beta_start) ** t_frac
        shifted = conv - max_conv               # numerical stability
        w_exp = np.exp(beta * shifted)
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum

        # ---- gradient of the smoothed objective ----
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)

        # ---- Adam update ----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_adam / (1.0 - beta2 ** iteration)
        lr = lr0 * (1.0 - t_frac) + 1e-3
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        proposal = cur_vec + step
        proposal = proj_simplex(proposal)

        prop_J = _objective(proposal, n, fft_len, conv_len)

        # ---- Metropolis acceptance ----
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

            cur_vec = proposal
            cur_J = prop_J

    # ------------------------------------------------------------------

    cur_J = best_J_stage
    best_vec_stage = cur_vec.copy()
    best_J_stage = cur_J

    m = np.zeros_like(cur_vec)
    v_adam = np.zeros_like(cur_vec)
    iteration = 0
    last_improve = time.monotonic()

    while time.monotonic() < stage2_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total_stage = stage2_deadline - start_time
        t_frac = (now - start_time) / total_stage

        # compute convolution and locate the maximum entry
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_idx = int(np.argmax(conv))
        max_conv = conv[max_idx]
        cur_J = float(2.0 * n * max_conv)

        # keep best
        if cur_J < best_J_stage - 1e-12:
            best_J_stage = cur_J
            best_vec_stage = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stage‑2 new best J = {best_J_stage:.6f}")

        # stagnation jitter (if needed)
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_vec_stage * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = _objective(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stage‑2 jitter – best J = {best_J_stage:.6f}")

        # gradient of the true max entry (sub‑gradient)
        grad = _hard_gradient(cur_vec, max_idx)

        # Adam‑style step
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_adam / (1.0 - beta2 ** iteration)
        lr = 0.5 * (1.0 - t_frac) + 1e-4          # slightly smaller learning rate

        prop_J = _objective(proposal, n, fft_len, conv_len)

        # accept if improves, otherwise Metropolis‐style uphill move
        T = max(1e-8, (best_J_stage * 0.01) * (1.0 - t_frac))
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = proposal
            cur_J = prop_J

    # ------------------------------------------------------------------
    # 6. Stage‑3: Local stochastic polishing (pairwise mass swaps)
    # ------------------------------------------------------------------
    stage3_deadline = start_time + max(0.0, float(budget_s) - 2.0)   # leave few seconds for finalisation

    cur_vec = best_vec_stage.copy()
    cur_J = best_J_stage
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    # temperature schedule for the annealer
    T0_polish = cur_J * 0.01
    T_end_polish = 1e-9

    while time.monotonic() < stage3_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = stage3_deadline - start_time
        t_frac = (now - start_time) / total
        T = T0_polish * (1.0 - t_frac) + T_end_polish * t_frac

        # randomly pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # transfer a small random fraction of mass from i to j
        if cur_vec[i] <= 0.0:
            continue

        if eps <= 0.0:

        cand = np.maximum(cand, 0.0)

            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur_vec = cand
                cur_J = cand_J

    # ------------------------------------------------------------------
    # 7. Final normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_vec_stage = np.clip(best_vec_stage, 0.0, 1000.0)
    total = best_vec_stage.sum()
    if total == 0.0:
        best_vec_stage[:] = 1.0 / n
    else:
        best_vec_stage /= total

    # Return as a plain list of floats
    return [float(v) for v in best_vec_stage]
