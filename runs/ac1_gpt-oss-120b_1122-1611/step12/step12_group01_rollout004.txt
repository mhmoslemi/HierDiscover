def grad_approx(a, alpha, n, conv_len, L):
    # a: np.ndarray length n
    # L: FFT length >= conv_len + n -1
    # Compute FFT of a padded to L
    A = np.fft.rfft(a, L)
    # Compute self convolution c via IFFT
    conv_full = np.fft.irfft(A * A, n=L)[:conv_len]
    # Exponential weights
    max_c = conv_full.max()
    w = np.exp(alpha * (conv_full - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum
    # Reverse w
    w_rev = w_norm[::-1]
    B = np.fft.rfft(w_rev, L)
    conv_rev = np.fft.irfft(A * B, n=L)
    start = conv_len - 1
    grad = 2.0 * conv_rev[start:start + n]
    return grad

"""
Propose a step‑function that minimises the evaluation metric.

The algorithm is a homotopy of smooth approximations of the L‑∞ norm.
For a probability vector a (∑a=1, a_i≥0) we want to minimise
      max(conv(a,a))  =  max_{k} (a * a)_k .

We replace the max by a log‑sum‑exp with a temperature α>0:
      f_α(a)= (1/α)·log Σ_k exp(α·c_k) ,   c = conv(a,a).

∇f_α(a) = 2·conv_rev(a, w)  with   w_k = exp(α·(c_k‑max c))/Σ_j exp(α·(c_j‑max c)),
and conv_rev denotes the linear convolution of a with the reversed weight vector w.
The gradient points towards increasing the worst convolution entries,
so we move in the opposite direction using an exponentiated‑gradient (mirror‑descent)
step that automatically respects the simplex constraints.

A schedule of increasing α (from a smooth to a sharp approximation of max)
guides the optimisation.  After the homotopy we finish with a short
multiplicative‑jitter local search.

All operations are performed on the probability simplex; the scaling invariance
of the original objective guarantees that the returned vector is valid for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) assuming ∑a = 1."""
    A = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv_full.max())


def grad_approx(a: np.ndarray, alpha: float,
                n: int, conv_len: int, L: int) -> np.ndarray:
    """
    Gradient of the smooth max approximation with temperature `alpha`.

    Returns 2·conv_rev(a, w), where w is the soft‑max over the self‑convolution.
    """
    # FFT of a (padded to length L)
    A = np.fft.rfft(a, L)

    # self‑convolution c = conv(a,a)
    c = np.fft.irfft(A * A, n=L)[:conv_len]

    # soft‑max weights (numerically stable)
    max_c = c.max()
    w = np.exp(alpha * (c - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w_norm = np.full_like(w, 1.0 / w.size)
    else:
        w_norm = w / w_sum

    # convolution with reversed weights
    w_rev = w_norm[::-1]
    B = np.fft.rfft(w_rev, L)
    conv_rev = np.fft.irfft(A * B, n=L)

    start = conv_len - 1               # index where the full linear convolution begins
    grad = 2.0 * conv_rev[start:start + n]
    return grad


def mirror_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative (exponentiated) mirror‑descent step.
    a is on the simplex, grad ≥ 0.
    """
    factor = np.exp(-eta * grad)
    new = a * factor
    s = new.sum()
    if s == 0.0:
        # fallback to uniform if everything vanished
        return np.full_like(a, 1.0 / a.size)
    return new / s


# ----------------------------------------------------------------------
#  Main search routine
# ----------------------------------------------------------------------

                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low value of
    `evaluate_sequence`.  The method combines a homotopy of smooth
    max‑approximations (log‑sum‑exp) with exponentiated‑gradient descent,
    followed by a short stochastic jitter phase.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a small safety margin for the host environment

    else:
        n = 2048

    conv_len = 2 * n - 1

    L = next_pow2(conv_len + n - 1)       # ≥ 3·n‑2

    # --------------------------------------------------------------

    def objective(v: np.ndarray) -> float:

    # --------------------------------------------------------------

            else:

                eta *= 0.8

                    break   # move on to the next α

    # --------------------------------------------------------------
