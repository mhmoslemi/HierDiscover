"""
Hybrid local search for minimizing
    J(a) = 2·n·max(conv(a,a)),      a_i ≥ 0,  Σ a_i = 1

The algorithm starts from the best known construction (global variable
``height_sequence_1`` when present) and performs a greedy hill‑climbing
procedure that repeatedly moves a small amount of probability mass away
from the indices that contribute most to the current peak of the
self‑convolution.  The update of the convolution is performed with an
O(n) incremental formula (no full FFT per trial), which makes it possible
to evaluate millions of candidates within the 1000 s time budget.

If the search stagnates, the step size is reduced and a random restart
is performed.  After the main phase a short multiplicative‑noise local
refinement (as in the previous DE code) is applied.

The returned list is a probability vector; ``evaluate_sequence`` will
automatically clip and normalise it.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a using a power‑of‑two real FFT."""
    n = a.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def trial_move(a: np.ndarray, conv: np.ndarray, src: int,
               dst: int, delta: float) -> (np.ndarray, float):
    """
    Return (new_conv, new_J) for the candidate obtained by moving ``delta``
    mass from ``src`` to ``dst`` (the move is evaluated on the *current*
    vector a, while conv is the current convolution).
    """
    n = a.size
    conv_new = conv.copy()
    # linear part 2·δ·(shift_dst - shift_src)
    conv_new[dst:dst + n] += 2.0 * delta * a
    conv_new[src:src + n] -= 2.0 * delta * a
    # quadratic part δ²·(e_{2·src}+e_{2·dst}-2·e_{src+dst})
    d2 = delta * delta
    conv_new[2 * src] += d2
    conv_new[2 * dst] += d2
    conv_new[src + dst] -= 2.0 * d2
    max_conv = conv_new.max()
    J = 2.0 * n * max_conv
    return conv_new, J


def apply_move_inplace(a: np.ndarray, conv: np.ndarray,
                       src: int, dst: int, delta: float) -> None:
    """
    Perform the move ``src → dst`` of mass ``delta`` *in place* on ``a``
    and ``conv`` (the same incremental formula as in ``trial_move``).
    """
    n = a.size
    a_old = a.copy()                   # needed for the linear part
    a[src] -= delta
    a[dst] += delta
    conv[dst:dst + n] += 2.0 * delta * a_old
    conv[src:src + n] -= 2.0 * delta * a_old
    d2 = delta * delta
    conv[2 * src] += d2
    conv[2 * dst] += d2
    conv[src + dst] -= 2.0 * d2


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    ``evaluate_sequence``.  The method consists of:

    1. Initialise from the global best ``height_sequence_1`` (or a Dirichlet
       draw if unavailable) and compute its full convolution.
    2. Repeatedly move a small amount of probability mass away from the
       indices that dominate the current convolution peak (greedy hill‑climb
       with an O(n) incremental convolution update).
    3. Reduce the move size on stagnation and occasionally restart from a
       fresh random simplex point.
    4. Finish with a short multiplicative‑noise local search (as in the
       previous DE script) to polish the final vector.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size and initial vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_vec = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = init_vec.size
        a = init_vec.astype(np.float64)
    else:
        n = 2048
        a = rng.dirichlet(np.ones(n))

    # Normalise to the probability simplex (protect against numerical issues)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s

    # --------------------------------------------------------------
    # 2. Initial convolution (full, via FFT – done only once)
    # --------------------------------------------------------------
    conv = conv_full_fft(a)                     # length 2·n‑1
    max_conv = conv.max()
    cur_J = best_J = 2.0 * n * max_conv
    best_vec = a.copy()
    best_conv = conv.copy()

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin
    print(f"[0.00s] Initial J = {best_J:.6f} (n = {n})")

    # --------------------------------------------------------------
    # 3. Parameters for the greedy local search
    # --------------------------------------------------------------
    step_frac = 0.30          # maximal fraction of a source entry that can be moved
    min_step = 0.005
    max_step = 0.5
    stagnation_counter = 0    # attempts without improvement
    stagnation_limit = 4000   # after this many failed attempts we shrink the step
    recompute_every = 12000   # recompute convolution from scratch to avoid drift
    moves_since_recompute = 0

    # --------------------------------------------------------------
    # 4. Main hill‑climbing loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # Identify the index where the convolution attains its maximum
        k_max = int(conv.argmax())

        # Compute contributions a[i]·a[k_max‑i] only where the indices are valid
        i_low = max(0, k_max - (n - 1))
        i_high = min(n - 1, k_max)
        idx = np.arange(i_low, i_high + 1, dtype=np.int64)
        partner = k_max - idx
        contrib = a[idx] * a[partner]

        # Source (the index with the largest single contribution)
        src_local = int(np.argmax(contrib))
        src = int(idx[src_local])
        partner_src = int(partner[src_local])

        # Choose a destination that currently carries little mass
        # (weights proportional to “available capacity” = 1‑a)
        cap = 1.0 - a
        cap[src] = 0.0
        if partner_src != src:
            cap[partner_src] = 0.0
        if cap.sum() > 0:
            dst = int(rng.choice(n, p=cap / cap.sum()))
        else:
            # fallback – uniform random distinct index
            cand = list(set(range(n)) - {src, partner_src})
            dst = rng.choice(cand)

        # Maximum mass we may shift from the sources (must stay non‑negative)
        max_move = a[src]
        if partner_src != src:
            max_move = min(max_move, a[partner_src])

        if max_move <= 0.0:
            # nothing to move – skip this iteration
            continue

        # Random step size (fraction of the allowed move)
        delta = rng.uniform(0.0, step_frac * max_move)
        if delta <= 0.0:
            continue

        # ----------------------------------------------------------
        # Evaluate the trial move
        # ----------------------------------------------------------
        # First move src → dst, then (if distinct) partner_src → dst
        conv_trial, J_trial = trial_move(a, conv, src, dst, delta)
        if partner_src != src:
            conv_trial, J_trial = trial_move(a, conv_trial, partner_src, dst, delta)

        # Accept only if we improve the current objective
        if J_trial + 1e-12 < cur_J:
            # Apply the two moves permanently (in place)
            apply_move_inplace(a, conv, src, dst, delta)
            if partner_src != src:
                apply_move_inplace(a, conv, partner_src, dst, delta)

            cur_J = J_trial
            moves_since_recompute += 1
            stagnation_counter = 0

            # Track the best vector ever seen
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = a.copy()
                best_conv = conv.copy()
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        else:
            stagnation_counter += 1

        # ----------------------------------------------------------
        # Periodic maintenance
        # ----------------------------------------------------------
        # 1) Reduce step size on long stagnation
        if stagnation_counter >= stagnation_limit:
            step_frac = max(step_frac * 0.7, min_step)
            stagnation_counter = 0
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Reducing step_frac → {step_frac:.4f}")

        # 2) Occasionally recompute the full convolution from scratch
        if moves_since_recompute >= recompute_every:
            conv = conv_full_fft(a)
            cur_J = 2.0 * n * conv.max()
            moves_since_recompute = 0
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Full recompute of convolution")

        # 3) Random restart if we have been stuck for a very long time
        if stagnation_counter > 40000:
            # Re‑initialise from a fresh Dirichlet draw
            a = rng.dirichlet(np.ones(n))
            conv = conv_full_fft(a)
            cur_J = 2.0 * n * conv.max()
            step_frac = max_step
            stagnation_counter = 0
            moves_since_recompute = 0
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Random restart performed")

    # --------------------------------------------------------------
    # 5. Final fine‑grained random search (multiplicative noise)
    # --------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start) * 0.05)
    a = best_vec.copy()                     # resume from the best found
    conv = best_conv.copy()

    while time.monotonic() < fine_deadline:
        # log‑normal multiplicative perturbation
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = a * noise
        cand = proj_simplex(cand)           # re‑project onto the simplex
        # recompute convolution (full, cheap for the modest number of trials)
        conv_cand = conv_full_fft(cand)
        J_cand = 2.0 * n * conv_cand.max()
        if J_cand + 1e-12 < best_J:
            best_J = J_cand
            best_vec = cand
            conv = conv_cand
            elapsed = time.monotonic() - start
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best vector (as a plain Python list)
    # --------------------------------------------------------------
    # Clip to the allowed range (the evaluate_sequence routine also does it)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    # Normalise – the evaluator requires a non‑zero sum (it will normalize anyway)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s

    return [float(v) for v in best_vec]
