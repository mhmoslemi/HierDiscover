def greedy_improve(a, cur_J, cur_conv, conv_len, n, rng):

    k_max = np.argmax(cur_conv)
    # compute contributions array
    low = max(0, k_max - (n - 1))
    high = min(n - 1, k_max)
    idx = np.arange(low, high+1, dtype=np.int64)
    j = k_max - idx
    contrib = a[idx] * a[j]
    i_max = idx[np.argmax(contrib)]

    # choose j_target among indices with smallest a values

    smallest_idxs = np.argpartition(a, 10)[:10]

            n = len(height_sequence_1)
            n = max(512, min(n, 8192))

            n = 4096
    else:

    else:

    beta2 = 0.999

"""
Multi‑phase optimisation for a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. Initialise a probability vector (simplex) from the previous best
   sequence if available, otherwise from a uniform distribution.
2. **Phase 1 – Adam optimisation**  
   Optimise a smooth approximation of the max‑convolution using a
   soft‑max weighting (β grows from 1 to 5000).  The gradient of the
   smoothed objective is computed by FFT‑based correlation and the
   update follows Adam with Euclidean projection onto the simplex.
   A temperature‑controlled Metropolis rule and occasional multiplicative
   jitter help to escape plateaux.
3. **Phase 2 – Greedy targeted mass moves**  
   Repeatedly identify the index pair that contributes most to the
   current maximal convolution entry and move a fraction of its mass to
   a location with very little weight.  The move is accepted if it
   lowers the objective; otherwise it may be accepted with a small
   uphill probability.
4. **Phase 3 – Polishing (pairwise swaps)**  
   A cheap simulated‑annealing schedule performs random mass‑swaps
   between two positions, always keeping the vector on the simplex.
5. Finally the vector is clipped to the limits required by
   `evaluate_sequence`, re‑normalised and returned as a plain list.

The routine respects a total time budget (default 1000 s) and returns
the best sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    weight length = 2·n‑1, a length = n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found
    within the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(512, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted corr

    # ------------------------------------------------------------------
    # 2. Initialise best candidate (previous answer if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_vec = np.asarray(height_sequence_1, dtype=np.float64)
        init_vec = np.clip(init_vec, 0.0, None)
        s = init_vec.sum()
        if s == 0.0:
            init_vec = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            init_vec = init_vec / s
    else:
        init_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = init_vec.copy()
    best_J = _objective(best_vec, n, fft_len, conv_len)
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Time bookkeeping
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + float(budget_s)

    # allocate time for the three phases
    phase1_deadline = start_time + 0.55 * budget_s
    phase2_deadline = start_time + 0.90 * budget_s      # 35 % for phase 2
    # remaining time is kept for phase 3 (polishing)

    # ------------------------------------------------------------------
    # 4. Phase 1 – Adam optimisation on a soft‑max smoothed max‑conv
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.8           # base learning rate

    # soft‑max schedule
    beta_start = 1.0
    beta_end = 5000.0   # larger final temperature gives a sharper max

    # Metropolis temperature schedule (linear decay)
    T_initial = cur_J * 0.2
    T_final = 1e-9

    m = np.zeros_like(cur_vec)          # 1st moment
    v_adam = np.zeros_like(cur_vec)     # 2nd moment
    iteration = 0
    last_improve = start_time

    while time.monotonic() < phase1_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # fraction of phase 1 elapsed
        t_frac = (now - start_time) / (phase1_deadline - start_time)

        # --- convolution & max entry ---
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # --- update global best ---
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase 1 new best J = {best_J:.6f}")

        # --- stagnation handling: jitter after 20 s without improvement ---
        if now - last_improve > 20.0:
            # multiplicative jitter preserving the simplex
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = _objective(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase 1 jitter – best J = {best_J:.6f}")

        # --- soft‑max weight vector (smooth approximation of max) ---
        beta = beta_start * (beta_end / beta_start) ** t_frac
        shifted = conv - max_conv          # for numerical stability
        w_exp = np.exp(beta * shifted)
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum

        # --- gradient of the smoothed objective ---
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)

        # --- Adam update ---
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_adam / (1.0 - beta2 ** iteration)
        lr = lr0 * (1.0 - t_frac) + 1e-3
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        proposal = cur_vec + step
        proposal = proj_simplex(proposal)

        # --- evaluate proposal ---
        prop_J = _objective(proposal, n, fft_len, conv_len)

        # --- Metropolis acceptance ---
        if prop_J < cur_J:
            accept = True
        else:
            # linear temperature drop
            T = (T_initial * (phase1_deadline - now) +
                 T_final * (now - start_time)) / (phase1_deadline - start_time)
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = proposal
            cur_J = prop_J

    # ------------------------------------------------------------------
    # 5. Phase 2 – Greedy targeted mass relocation
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    while time.monotonic() < phase2_deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # index where convolution is maximal
        k_max = int(np.argmax(cur_conv))

        # contributions a[i]·a[k_max‑i] for all feasible i
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        idx = np.arange(low, high + 1, dtype=np.int64)
        j = k_max - idx
        contrib = cur_vec[idx] * cur_vec[j]

        # donor – the index with the largest contribution to the peak
        i_max = int(idx[np.argmax(contrib)])

        # recipient – a location with very little weight (among the 10 smallest)
        smallest = np.argpartition(cur_vec, 10)[:10]
        j_target = int(rng.choice(smallest))

        # move a modest fraction of mass from donor to recipient
        delta = float(rng.random() * cur_vec[i_max] * 0.30)   # ≤30 % of donor mass
        if delta <= 1e-14:
            continue

        cand = cur_vec.copy()
        cand[i_max] -= delta
        cand[j_target] += delta
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        if cand_J < cur_J - 1e-12:
            cur_vec = cand
            cur_J = cand_J
            cur_conv = _conv_full(cur_vec, fft_len, conv_len)
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{elapsed:6.2f}s] Phase 2 improvement J = {best_J:.6f}")
        else:
            # occasional uphill step (slowly cooling)
            T_up = max(0.01 * best_J *
                       (1.0 - (now - start_time) / (phase2_deadline - start_time)),
                       1e-12)
            if rng.random() < np.exp(-(cand_J - cur_J) / T_up):
                cur_vec = cand
                cur_J = cand_J
                cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    # ------------------------------------------------------------------
    # 6. Phase 3 – Polishing with random pairwise swaps (simulated annealing)
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    # simple temperature schedule for polishing
    T0_polish = cur_J * 0.02
    T_end_polish = 1e-9

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / (deadline - start_time)
        T = T0_polish * (1.0 - t_frac) + T_end_polish * t_frac

        # pick two distinct indices
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # move a small amount of mass from i to j
        eps = float(rng.random() * cur_vec[i] * 0.5)   # at most half of donor mass
        if eps <= 0.0:
            continue

        cand = cur_vec.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        if cand_J < cur_J:
            cur_vec = cand
            cur_J = cand_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{elapsed:6.2f}s] Polishing improvement J = {best_J:.6f}")
        else:
            if rng.random() < np.exp(-(cand_J - cur_J) / max(T, 1e-12)):
                cur_vec = cand
                cur_J = cand_J

    # ------------------------------------------------------------------
    # 7. Final safety normalisation & conversion to Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"Finished. Best J = {best_J:.6f}")
    return [float(v) for v in best_vec]
