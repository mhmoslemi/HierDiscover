"""
Projected Adam sub‑gradient descent on the max‑auto–convolution.

 * Works directly with the exact max‑convolution (no soft‑max smoothing).
 * Gradient for the max is the average of the true sub‑gradients
   at all active (tight) convolution indices.
 * Uses Adam’s adaptive step size and a simple simulated‑annealing
   acceptance rule to escape shallow plateaus.
 * If no improvement for a while, multiplicative jitter centred on the
   current best solution is applied.
 * The routine runs for the allotted time budget (≈1000 s) and returns the
   best non‑negative step‑function (list of heights) it has seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to ``x``."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | sum(x)=1, x_i ≥ 0}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are equal (or zero)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution of ``x`` and ``y`` using an FFT length ``L``.
    Result length = len(x) + len(y) - 1 (trimmed to that size).
    """
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[: len(x) + len(y) - 1]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Exact objective J = 2·n·max(conv(a,a)) for a vector ``a`` already
    normalised to sum = 1.  Convolution is performed by an FFT for speed.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size.
    # --------------------------------------------------------------
    # Use the length of a previously found construction if available;
    # otherwise fall back to a moderate default.  Upper‑bound it at 8192
    # to keep FFTs cheap.
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(8192, n))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for a*a
    grad_fft_len = _next_pow2(conv_len + n - 1)        # for w * a_rev

    # --------------------------------------------------------------
    # 2. Initialise a point on the simplex.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        # truncate / pad to the chosen size
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            a0 = np.pad(a0, (0, n - a0.size))
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= s
    else:
        # Dirichlet‑like random start; makes the algorithm less likely to stick
        a0 = rng.exponential(scale=1.0, size=n)
        a0 /= a0.sum()

    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)
    cur_J = best_J

    print(f"[0.00s] Initialise n={n}, J={best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Adam hyper‑parameters (working directly on the true max‑gradient).
    # --------------------------------------------------------------
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    lr = 0.15               # base learning rate (tuned for the sub‑gradient)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # Simulated‑annealing temperature schedule (only used for uphill moves)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)    # leave a safety margin
    total_time = deadline - start_time
    T_start = best_J
    T_end = 1e-8

    # Stagnation handling
    last_improve = start_time
    iteration = 1

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # 3.1 Temperature (for occasional uphill moves)
        # ----------------------------------------------------------
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # ----------------------------------------------------------
        # 3.2 Compute the full convolution of a with itself (FFT)
        # ----------------------------------------------------------
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        # ----------------------------------------------------------
        # 3.3 Identify active (tight) constraints
        # ----------------------------------------------------------
        # use a tiny tolerance to catch ties caused by floating‑point noise
        tol = max(1e-14, max_conv * 1e-12)
        active = np.where(conv >= max_conv - tol)[0]

        # ----------------------------------------------------------
        # 3.4 Sub‑gradient of the max‑convolution
        # ----------------------------------------------------------
        # w is a mask that averages the sub‑gradients of all active indices
        w = np.zeros(conv_len, dtype=np.float64)
        w[active] = 1.0 / active.size

        a_rev = a[::-1]                                    # reversed a
        W = np.fft.rfft(w, grad_fft_len)
        A_rev = np.fft.rfft(a_rev, grad_fft_len)
        grad_full = np.fft.irfft(W * A_rev, n=grad_fft_len)

        # slice the part that corresponds to the gradient w.r.t. each a_i
        grad = 2.0 * grad_full[n - 1:n - 1 + n]            # shape (n,)

        # ----------------------------------------------------------
        # 3.5 Adam update (projected)
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_proposed = a - step
        a_proposed = proj_simplex(a_proposed)

        # ----------------------------------------------------------
        # 3.6 Evaluate the proposal (exact J)
        # ----------------------------------------------------------
        A_p = np.fft.rfft(a_proposed, fft_len)
        conv_p = np.fft.irfft(A_p * A_p, n=fft_len)[:conv_len]
        new_J = 2.0 * n * conv_p.max()

        # ----------------------------------------------------------
        # 3.7 Metropolis‑type acceptance (allows occasional uphill moves)
        # ----------------------------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_proposed
            cur_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 3.8 Stagnation detector – jitter after ~15 s without improvement
        # ----------------------------------------------------------
        if now - last_improve > 15.0:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = proj_simplex(best_a * noise)
            # recompute J for the jittered point (cheap)
            A = np.fft.rfft(a, fft_len)
            conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
            cur_J = 2.0 * n * conv.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – current J = {cur_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final sanitisation (clip, re‑normalise) and return
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
