Hybrid probabilistic search for a non‑negative step function that minimises
evaluate_sequence().  The routine works in three stages:

1. **Structured initialisation** – a short grid‑search over several
   parametrised families (beta‑like PDFs, powered sine windows,
   triangular and uniform shapes).  Each candidate is projected onto the
   probability simplex and its objective value J = 2·n·max(conv) is
   recorded.  The best candidate from this cheap exploration seeds the
   subsequent optimisation.

2. **Exponentiated‑gradient (EG) sub‑gradient descent** – the exact
   sub‑gradient of the ‖f * f‖∞ term is obtained via a binary mask of
   the active (maximal) convolution entries and a single FFT‑based
   convolution.  The EG update
        a_i ← a_i·exp(−η·∂J/∂a_i)
   stays on the simplex after renormalisation and converges quickly.
   An annealing schedule (η decreasing with the iteration count) and a
   simple stagnation‑detector trigger short random jitter moves to jump
   out of plateaus.

3. **Optional LP polishing** – if the problem size is modest (n ≤ 3000)
   a tiny linear‑programming refinement is performed once:
   maximise 1ᵀg under the linear constraints (f * g) ≤ max(f * f).
   The LP solution is mixed with the current iterate.

The whole procedure respects the 1000 s budget, prints progress, and
returns the best probability vector found (scaled to sum = 1, clipped to
the allowed range [0, 1000]).  Because `evaluate_sequence` is
scale‑invariant, any normalised vector yields the same score.
"""

import time
import numpy as np
import scipy.sparse
from scipy.optimize import linprog

# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """ Euclidean projection of a vector onto the probability simplex
        {x | sum(x) = 1, x_i ≥ 0}.  O(n log n) implementation. """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear (full) convolution a·a using an FFT of length ``fft_len``."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of ``x`` and ``y`` using an FFT of length ``L**."""
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[: len(x) + len(y) - 1]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective J = 2·n·max(conv(a,a)) for a vector already on the simplex."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgradient_max_conv(a: np.ndarray,
                         n: int,
                         fft_len: int,
                         conv_len: int,
                         grad_fft_len: int):
    """
    Returns (max_conv, subgradient) for the L∞–norm of the auto‑convolution.
    The subgradient is 2·∑_{k∈S} a_{k‑i} where S are the active shifts.
    """
    conv = _conv_full(a, fft_len, conv_len)
    max_c = conv.max()
    # active set tolerance (absolute, but never smaller than 1e‑12)
    tol = max(1e-12, max_c * 1e-12)
    act = (conv >= max_c - tol).astype(np.float64)  # shape (2n‑1,)
    a_rev = a[::-1]
    # gradient via one extra convolution
    grad = 2.0 * conv_fft(act, a_rev, grad_fft_len)[n - 1:n - 1 + n]
    return max_c, grad

def beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete beta‑like PDF on {0,…,n‑1} (unnormalised)."""
    x = np.arange(n, dtype=np.float64) + 0.5
    t = x / (n + 1.0)
    # avoid overflow for extreme exponents by using log‑space
    loga = alpha * np.log(t + 1e-300) + beta * np.log(1.0 - t + 1e-300)
    a = np.exp(loga)
    return a


def sin_power_shape(n: int, p: float) -> np.ndarray:
    """sin(π·(i+0.5)/(n+1)) raised to power p."""
    x = np.arange(n, dtype=np.float64) + 0.5
    a = np.sin(np.pi * x / (n + 1.0)) ** p
    return a

    """Symmetric triangular distribution (mountain) on {0,…,n‑1}."""
    idx = np.arange(n, dtype=np.float64)
    a = np.minimum(idx + 1.0, n - idx)
    return a

                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] LP polishing improvement: J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Exponentiated‑gradient sub‑gradient descent
    # ------------------------------------------------------------------
    # Reset timing for the long descent phase
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 150.0)   # remaining budget
    total_time = deadline - start_time

    # EG hyper‑parameters
    eta0 = 0.9            # initial step size
    decay = 5e-5          # controls how fast η decays
    jitter_sigma = 0.02   # strength of the log‑normal jitter
    last_improve = start_time
    iteration = 1

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # Compute exact sub‑gradient of the max‑convolution term
        max_c, grad = subgradient_max_conv(best_a, n, fft_len,
                                           conv_len, grad_fft_len)

        # Adaptive learning rate (simple 1/(1+decay·t) schedule)
        eta = eta0 / (1.0 + decay * iteration)

        # Exponentiated‑gradient update (move opposite to the sub‑gradient)
        log_factor = -eta * grad
        # avoid overflow/underflow in the exponent
        log_factor = np.clip(log_factor, -50.0, 50.0)
        cand = best_a * np.exp(log_factor)
        s = cand.sum()
        if s == 0.0:
            cand = np.full(n, 1.0 / n)
        else:
            cand = cand / s

        # Evaluate candidate
        Jcand = compute_J(cand, n, fft_len, conv_len)

        # Simple improvement test
        if Jcand < best_J - 1e-12:
            best_J = Jcand
            best_a = cand.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] EG iteration {iteration:<5d}: new best J = {best_J:.6f}")

        # Stagnation handling – occasional jitter
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            cand = proj_simplex(best_a * noise)
            Jcand = compute_J(cand, n, fft_len, conv_len)
            if Jcand < best_J:
                best_J = Jcand
                best_a = cand.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] Jitter escaped plateau: J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 6. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / total

    print(f"Finished: best J = {best_J:.6f} with length {n}")
    return [float(v) for v in best_a]
