Sub‑gradient optimisation for the max‑convolution objective.

We treat the evaluation
    J(a) = 2·n·max_k (a * a)_k      (a lives on the probability simplex)

as a convex piece‑wise linear function: the maximum of a family of linear
functions (one for every shift k).  A simple sub‑gradient is obtained by
picking the shift where the convolution attains its current maximum and
using the corresponding correlation vector as a descent direction.
The algorithm therefore performs:

* a fast full convolution (FFT) to obtain the current max‑shift,
* an O(n) construction of the sub‑gradient for that shift,
* a projected (momentum) step onto the simplex,
* occasional Metropolis jumps and a jitter‑restart when progress stalls,
* a final short simulated‑annealing polishing phase (pairwise mass swaps).

Multiple diverse initial points (uniform, geometric, Dirichlet draws,
previous best if available) are tried and the best one is used as a
starting vector.  All work fits comfortably in the ~1000 s budget and
uses only NumPy.  The returned list is already clipped to the limits of
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

        elapsed = now - start_time
        t_frac = (now - start_time) / total_time
        T = T0_polish * (1.0 - t_frac) + T_end_polish * t_frac

        # pick donor and receiver indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a random fraction of donor mass to receiver
        eps = rng.random() * cur_vec[i] * 0.5
        if eps <= 0.0:
            continue

        cand = cur_vec.copy()
        cand[i] -= eps
        cand[j] += eps

        # keep numerical noise from creating negative entries
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        if cand_J < cur_J:
            cur_vec = cand
            cur_J = cand_J
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                print(f"[{elapsed:6.2f}s] Polishing improvement: J = {best_J:.6f}")
        else:
            # uphill acceptance with annealing temperature
            prob = np.exp(-(cand_J - cur_J) / max(T, eps_adam))
            if rng.random() < prob:
                cur_vec = cand
                cur_J = cand_J

    # ------------------------------------------------------------------
    # 6. Final safety normalisation & conversion to Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
