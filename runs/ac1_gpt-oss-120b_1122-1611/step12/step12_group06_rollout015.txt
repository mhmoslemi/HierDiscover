def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

"""
Optimisation of a non‑negative step function for the evaluate_sequence objective.

The method uses a projected‑gradient descent on the probability simplex.
At every iteration the full convolution a*a is computed (via an FFT) and the
index attaining the maximum is identified.  A sub‑gradient of the exact
maximum (i.e. the hard max) is then back‑propagated through a soft‑max
parameterisation a = softmax(z) and an Adam optimiser updates the logits
*z*.  When progress stalls a stochastic jitter is applied to the logits and
the Adam moments are reset.  Several simple initialisations (uniform, geometric,
linear ramps, Gaussian bumps and a Dirichlet draw) are tried and the best
candidate is retained as the starting point.

The routine respects a 1000 s wall‑time budget, prints progress and finally
returns the best sequence found (normalised to sum = 1, values are clipped to
the allowed range so that `evaluate_sequence` can be called directly).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns the first `conv_len` entries of the linear convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _grad_hard_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Gradient of the hard‑max of the convolution with respect to `a`.
    For the index `max_idx` where the convolution attains its maximum
    the derivative is `2 * a[max_idx - i]` for all i where the shifted
    index lies inside the array, otherwise 0.
    """
    n = a.size
    grad = np.zeros_like(a)
    # valid i satisfy 0 <= max_idx - i < n
    i_start = max(0, max_idx - (n - 1))
    i_end = min(max_idx, n - 1)
    i_range = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = max_idx - i_range
    grad[i_range] = 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds and
    returns the best sequence discovered.

    Optimisation is performed on the logits of a soft‑max parametrisation
    (so the vector always stays on the probability simplex).  At each step
    the exact maximum of the convolution is located and a sub‑gradient of
    this max is back‑propagated; Adam updates the logits.  Stagnation triggers
    a jitter restart.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            # keep n in a reasonable band
            n = max(256, min(n, 8192))
        except Exception:
            n = 3809
    else:
        n = 3809  # the length used in the previous best construction

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Generate a portfolio of initial distributions
    # --------------------------------------------------------------
    init_cands = []

    # Uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Geometric decays
    for r in (0.90, 0.95, 0.975, 0.985, 0.990, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 10.0, n / 8.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws (different concentration levels)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Previous best construction, if any
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the starter with the smallest exact objective J = 2·n·max(conv)
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation on logits (Adam, hard‑max sub‑gradient, jitter)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # a small safety margin so we always return before the external timeout
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    eps = 1e-12
    logits = np.log(np.maximum(best_a, eps))
    m = np.zeros_like(logits)
    v_m = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.7  # a slightly larger initial learning rate

    iteration = 0
    last_improve = start_time

    # Global best (track exact J)
    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # ----- 4.1 Current a from logits (soft‑max) -----
        a = _softmax(logits)

        # ----- 4.2 Compute convolution and exact objective -----
        conv = _conv_fft(a, fft_len, conv_len)
        max_val = conv.max()
        cur_J = float(2.0 * n * max_val)

        # ----- 4.3 Update global best if improved -----
        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----- 4.4 Stagnation detection & jitter restart -----
        if now - last_improve > 15.0:          # restart after ~15 s of no improvement
            jitter_sigma = 0.20
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v_m.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

        # ----- 4.5 Compute hard‑max gradient -----
        max_idx = int(conv.argmax())
        grad_a = _grad_hard_max(a, max_idx)

        # ----- 4.6 Transform gradient to logits space (soft‑max chain rule) -----
        avg = np.dot(a, grad_a)               # scalar <a,∇_a J>
        grad_z = a * (grad_a - avg)           # ∂J/∂z

        # ----- 4.7 Adam update on logits -----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)          # 1/√t decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_m = beta2 * v_m + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_m / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # Clamp logits to avoid numerical overflow
        logits = np.clip(logits, -500.0, 500.0)

    # --------------------------------------------------------------
    # 5. Return the best sequence (clipped, renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    final_J = float(2.0 * n * np.convolve(best_vec, best_vec).max())
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
