"""
Sub‑gradient descent on the ∞‑norm of the aperiodic self‑convolution.
The objective (equal to ``evaluate_sequence`` when the vector is
normalised to sum 1) is

    J(a) = 2·n·max_k (a * a)[k] ,

where ``*`` is the discrete linear convolution.
The algorithm starts from the best known construction (if any) or from a
random Dirichlet draw, then repeatedly

* evaluates the full convolution by a real FFT,
* builds a sub‑gradient of the active (maximal) lag(s),
* takes a decreasing step in the opposite direction,
* projects back onto the probability simplex.

A simple stagnation rule restarts the walk with a fresh random point.
The walk is continued until the 1000 s budget expires and the best
vector seen is returned (scaled to the conventional sum = √(2 n) for
compatibility with the reference code).
"""

import time
import numpy as np
from scipy.optimize import linprog  # only imported for the fallback random walk

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n log n) algorithm of Wang & Carreira‑Perpiñán
    (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Fast full convolution a*a using a real FFT.
    Returns the first 2·n‑1 entries (the linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * a.size - 1]


def compute_J(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Fast evaluation of the objective J for a vector that lies on the
    simplex (sum = 1).  J = 2·n·max(conv(a,a)).
    """
    conv = conv_full_fft(a, fft_len)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


def subgradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Return a sub‑gradient of the function max(conv(a,a)) at the current
    point ``a``.  The active set consists of all lags where the
    convolution attains its maximum (within a tiny tolerance).
    """
    n = a.size
    max_val = conv.max()
    # active lags (a few of them are typical)
    idx = np.where(conv >= max_val - 1e-12)[0]

    grad = np.zeros_like(a)
    for k in idx:
        if k < n:
            # contribution from the left part of the convolution
            grad[: k + 1] += a[k::-1]
        else:
            start = k - n + 1
            grad[start:] += a[: 2 * n - 1 - k][::-1]

        # the diagonal term a_i * a_i appears only once in the sum,
        # its derivative is 2·a_i – we have added a_i once above,
        # so we add the missing a_i when k is even and i = k//2 in range.
        if (k & 1) == 0:          # k even
            i = k // 2
            if i < n:
                grad[i] += a[i]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.
    The method performs a projected sub‑gradient descent on the
    probability simplex, periodically restarting from random points.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size – reuse a previous construction if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = max(512, min(old.size, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    fft_len = _next_pow2(2 * n - 1)          # length for the FFT‑based convolution

    # --------------------------------------------------------------
    # 2. Initialise a pool of starting points (simplex vectors).
    # --------------------------------------------------------------
    starters = []

    # uniform start
    starters.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous best, truncated / padded to the current length
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if v.size > n:
                v = v[:n]
            elif v.size < n:
                v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            v = np.maximum(v, 0.0)
            s = v.sum()
            if s > 0.0:
                v = v / s
            else:
                v = np.full(n, 1.0 / n, dtype=np.float64)
            starters.append(v)
        except Exception:
            pass

    # a few random Dirichlet draws
    for _ in range(5):
        v = rng.dirichlet(np.ones(n))
        starters.append(v)

    # pick the best starter (smallest J)
    best_vec = None
    best_J = float("inf")
    for vec in starters:
        J = compute_J(vec, n, fft_len)
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
    cur = best_vec.copy()
    cur_J = best_J

    print(f"[0.00s] Initial J = {best_J:.6f} (n = {n})")

    # --------------------------------------------------------------
    # 3. Main optimisation loop (projected sub‑gradient descent)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5.0          # tiny safety margin
    last_improve = start_time
    it = 0
    stagnation_time = 30.0                          # seconds without improvement → restart
    step0 = 0.5                                      # initial step‑size multiplier

    while time.monotonic() < deadline:
        it += 1
        # full self‑convolution via FFT
        conv = conv_full_fft(cur, fft_len)

        # sub‑gradient of the active (maximal) lag(s)
        g = subgradient(cur, conv)

        # step size decays like 1/√t
        step = step0 / np.sqrt(it)

        # trial update
        trial = cur - step * g
        trial = np.maximum(trial, 0.0)
        trial = proj_simplex(trial)

        trial_J = compute_J(trial, n, fft_len)

        # accept only if we strictly improve the current point
        if trial_J < cur_J - 1e-12:
            cur, cur_J = trial, trial_J
            # also keep track of the global best
            if trial_J < best_J - 1e-12:
                best_J, best_vec = trial_J, trial.copy()
                now = time.monotonic()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")
                last_improve = now

        # stagnation handling – occasional random restart
        now = time.monotonic()
        if now - last_improve > stagnation_time:
            # fresh Dirichlet start
            cur = rng.dirichlet(np.ones(n))
            cur_J = compute_J(cur, n, fft_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restarted with a random point (J = {cur_J:.6f})")

        # a very light random‑noise refinement once in a while
        if it % 500 == 0:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = proj_simplex(cur * noise)
            cand_J = compute_J(cand, n, fft_len)
            if cand_J < cur_J:
                cur, cur_J = cand, cand_J
                if cand_J < best_J:
                    best_J, best_vec = cand_J, cand.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] Noise‑refine improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Return the best construction (scaled to sum = √(2 n) for
    #    compatibility with the reference formulation).
    # ------------------------------------------------------------------
    scale = np.sqrt(2.0 * n) / best_vec.sum()
    best_scaled = best_vec * scale
    # final safety clamp (evaluate_sequence itself also clamps)
    best_scaled = np.clip(best_scaled, 0.0, 1000.0)

    # sanity check (optional)
    try:
        import __main__
        _ = __main__.evaluate_sequence(best_scaled.tolist())
    except Exception:
        # should never happen – fall back to a uniform vector
        best_scaled = np.full(n, np.sqrt(2.0 * n) / n, dtype=np.float64)

    return [float(v) for v in best_scaled]
