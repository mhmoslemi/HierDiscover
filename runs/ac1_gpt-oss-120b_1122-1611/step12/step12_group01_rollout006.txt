"""
Hybrid simple DE + targeted sub‑gradient descent.

* The objective J(a) = 2·n·max(conv(a,a)) / (sum(a)²) is scale‑invariant,
  therefore all candidates are kept on the probability simplex (sum = 1,
  a_i ≥ 0).
* A small DE/best/1 population quickly explores the search space.
* Afterwards an explicit sub‑gradient of the worst convolution entry is
  computed; a projected line‑search step moves the vector away from that
  “active” lag.
* If the sub‑gradient step stagnates a tiny multiplicative jitter is tried
  to escape flat regions.
* The whole routine respects the provided time budget (default 1000 s)
  and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a 1‑D vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works only on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Compute the maximum entry of the linear self‑convolution of *vec*.
    Uses FFT for O(n log n) speed.
    """
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max()), conv  # also return the full convolution


def compute_obj(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Full objective 2·n·max(conv(vec,vec)) / (sum(vec)²)."""
    m, _ = conv_max(vec, fft_len, conv_len)
    s = vec.sum()
    if s < 0.01:                     # safety guard – matches evaluate_sequence
        return float("inf")
    return 2.0 * n * m / (s * s)


def gradient_of_lag(a: np.ndarray, lag: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry at *lag*.
    For a_i ≥ 0 the derivative of
        conv_lag(a) = Σ_i a_i·a_{lag−i}
    is 2·a_{lag−i} when the index is in range, otherwise 0.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, lag - (n - 1))
    i_end = min(lag, n - 1)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[idx] = 2.0 * a[lag - idx]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step function (list of heights) that minimizes
    ``evaluate_sequence``.  The method mixes a tiny DE population with a
    directed sub‑gradient descent on the worst convolution lag.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (use the previously found construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        # keep the size reasonable – the original run used 3809 elements
        n = max(256, min(int(n), 8192))
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)

    # --------------------------------------------------------------
    # Initial candidate(s)
    # --------------------------------------------------------------
    init_vectors = []

    # 1) uniform
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) Hann window – smooth bell shape
    x = np.arange(n, dtype=np.float64)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * x / (n - 1))
    hann /= hann.sum()
    init_vectors.append(hann)

    # 3) Gaussian bell
    sigma = n / 6.0
    gauss = np.exp(-0.5 * ((x - (n - 1) / 2.0) / sigma) ** 2)
    gauss /= gauss.sum()
    init_vectors.append(gauss)

    # 4) a few Dirichlet draws with different concentrations
    for alpha in (0.05, 0.1, 0.3, 0.6, 1.0):
        for _ in range(2):
            init_vectors.append(rng.dirichlet(np.ones(n) * alpha))

    # 5) the previously best known sequence (if any)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        init_vectors.append(v)

    # --------------------------------------------------------------
    # Evaluate initial vectors and keep the best
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for v in init_vectors:
        J = compute_obj(v, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = v.copy()
    print(f"[{0.0:6.2f}s] Initial best J = {best_J:.8f}")

    # --------------------------------------------------------------
    # Timing control
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a 2 s safety margin
    total_allowed = deadline - start_time

    # --------------------------------------------------------------
    # Very short Differential Evolution phase (exploratory)
    # --------------------------------------------------------------
    NP = 30                               # modest population size
    population = []
    for _ in range(NP):
        # start around the current best with a small perturbation
        noise = np.exp(rng.normal(0.0, 0.05, size=n))
        cand = best_vec * noise
        cand = proj_simplex(cand)
        population.append(cand)

    fitness = np.empty(NP, dtype=np.float64)
    for i, v in enumerate(population):
        fitness[i] = compute_obj(v, n, fft_len, conv_len)

    # DE parameters (best/1 strategy)
    F = 0.78
    CR = 0.92
    de_stop = start_time + 0.15 * total_allowed   # ~15 % of the budget

    while time.monotonic() < de_stop:
        for i in range(NP):
            # three distinct donors, none equal to i
            idxs = list(range(NP))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

            donor = (population[i] +
                     F * (best_vec - population[i]) +
                     F * (population[r1] - population[r2]))
            donor = proj_simplex(donor)

            cross = rng.random(n) < CR
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, donor, population[i])
            trial = proj_simplex(trial)

            trial_J = compute_obj(trial, n, fft_len, conv_len)
            if trial_J < fitness[i]:
                population[i] = trial
                fitness[i] = trial_J
                if trial_J < best_J - 1e-12:
                    best_J = trial_J
                    best_vec = trial.copy()
                    # keep the best vector also inside the population
                    best_idx = np.argmin(fitness)
                    population[best_idx] = best_vec
                    fitness[best_idx] = best_J
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] DE improvement → J = {best_J:.8f}")

    # --------------------------------------------------------------
    # Sub‑gradient descent on the active lag(s)
    # --------------------------------------------------------------
    grad_stop = deadline - 0.5  # reserve half a second for final jitter
    while time.monotonic() < grad_stop:
        # current convolution & worst lag
        cur_max, conv_full = conv_max(best_vec, fft_len, conv_len)
        lag = int(np.argmax(conv_full))

        # compute the (sub‑)gradient of the worst entry
        grad = gradient_of_lag(best_vec, lag)

        # accumulate gradients of all lags that are essentially tied with the maximum
        # (helps when the maximum plateau is flat)
        eps = 1e-12 * cur_max
        tied = np.where(conv_full >= cur_max - eps)[0]
        if tied.size > 1:
            grad = np.zeros_like(best_vec)
            for k in tied:
                grad += gradient_of_lag(best_vec, int(k))

        norm = np.linalg.norm(grad)
        if norm == 0.0:
            break

        direction = -grad / norm

        improved = False
        # simple back‑tracking line search
        for alpha in (0.6, 0.3, 0.15, 0.07, 0.03, 0.015, 0.007):
            cand = proj_simplex(best_vec + alpha * direction)
            cand_J = compute_obj(cand, n, fft_len, conv_len)
            if cand_J < best_J - 1e-12:
                best_vec = cand
                best_J = cand_J
                improved = True
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Grad step (α={alpha:.3f}) → J = {best_J:.8f}")
                break
        if not improved:
            # fallback: tiny multiplicative jitter to shake out of a flat region
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = proj_simplex(best_vec * noise)
            cand_J = compute_obj(cand, n, fft_len, conv_len)
            if cand_J < best_J - 1e-12:
                best_vec = cand
                best_J = cand_J
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Jitter escape → J = {best_J:.8f}")
            else:
                # no progress – stop the gradient phase
                break

    # --------------------------------------------------------------
    # Final fine‑grained jitter (few milliseconds left)
    # --------------------------------------------------------------
    final_deadline = deadline
    while time.monotonic() < final_deadline:
        sigma = 0.01
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_J = compute_obj(cand, n, fft_len, conv_len)
        if cand_J < best_J - 1e-12:
            best_vec = cand
            best_J = cand_J
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Final jitter → J = {best_J:.8f}")

    # --------------------------------------------------------------
    # Normalise & safety clip (matches evaluate_sequence's guards)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
