def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 5.0  # margin

    # Determine initial n based on previous seq length or default
    if 'height_sequence_1' in globals():
        base_seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        n0 = base_seq.size
        # clamp min and max for computational feasibility
        n0 = max(256, min(n0, 5000))
    else:
        n0 = 2048

    # multi-scale size schedule
    sizes = [int(n0/2), n0, min(8192, n0*2)]  # keep within memory
    sizes = [s for s in sizes if s >= 256]

    best_vec = None
    best_J = float('inf')

    for n in sizes:
        # If we have previous best, upsample
        if best_vec is not None:
            # interpolate
            x_old = np.linspace(0, 1, best_vec.shape[0])
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, best_vec)
            a = proj_simplex(a)
        else:
            a = np.full(n, 1.0/n, dtype=np.float64)

        # Precompute conv parameters
        conv_len = 2*n - 1
        fft_len = _next_pow2(conv_len)

        # Initialize population for DE
        popsize = max(30, min(70, n // 200))
        pop = []
        pop_J = []
        for _ in range(popsize):
            v = rng.dirichlet(np.ones(n))
            pop.append(v)
            pop_J.append(eval_J(v, n, fft_len, conv_len))
        # Add uniform and best guess
        pop.append(a.copy())
        pop_J.append(eval_J(a, n, fft_len, conv_len))

        # DE loop limited to part of remaining time
        sub_deadline = min(deadline, time.monotonic() + (deadline - start)*0.5)

        # parameters
        F = 0.5
        CR = 0.9
        generation = 0
        last_improve = time.monotonic()
        while time.monotonic() < sub_deadline:
            for i in range(len(pop)):
                # mutation: DE/rand/1
                idxs = list(range(len(pop)))
                idxs.remove(i)
                a1, a2, a3 = rng.choice(idxs, size=3, replace=False)
                donor = pop[a1] + F * (pop[a2] - pop[a3])
                donor = np.maximum(donor, 0.0)
                donor = proj_simplex(donor)

                # crossover
                j_rand = rng.integers(n)
                mask = (rng.random(n) < CR) | (np.arange(n) == j_rand)
                trial = np.where(mask, donor, pop[i])
                trial = proj_simplex(trial)

                trial_J = eval_J(trial, n, fft_len, conv_len)
                if trial_J <= pop_J[i]:
                    pop[i] = trial
                    pop_J[i] = trial_J
                    if trial_J < best_J - 1e-12:
                        best_J = trial_J
                        best_vec = trial.copy()
                        last_improve = time.monotonic()
                        print(f"[{last_improve - start:6.2f}s] new DE best J={best_J:.6f} (n={n})")
            generation += 1
            # adjust F gradually
            F = 0.5 * (1.0 - generation / (sub_deadline - start + 1.0))
            # restart if stagnation
            if time.monotonic() - last_improve > 20.0:
                # replace half of population with fresh Dirichlet
                half = len(pop)//2
                for k in range(half):
                    idx = rng.integers(len(pop))
                    v = rng.dirichlet(np.ones(n))
                    pop[idx] = v
                    pop_J[idx] = eval_J(v, n, fft_len, conv_len)
                last_improve = time.monotonic()
                print(f"[{last_improve - start:6.2f}s] DE reinit {half} individuals")
        # after DE, perform gradient descent on best candidate for this size
        if best_vec is not None:
            a = best_vec.copy()
        else:
            a = a.copy()
        # gradient descent schedule
        gd_deadline = min(deadline, time.monotonic() + (deadline - start)*0.5)
        iteration = 0
        while time.monotonic() < gd_deadline:
            conv = conv_full_fft(a, fft_len, conv_len)
            max_idx = int(np.argmax(conv))
            max_val = conv[max_idx]
            J = 2.0 * n * max_val
            if J < best_J - 1e-12:
                best_J = J
                best_vec = a.copy()
                print(f"[{time.monotonic() - start:6.2f}s] GD improved J={best_J:.6f} (n={n})")
            # compute gradient for max index
            g = np.zeros_like(a)
            # left part: indices 0..max_idx
            left_len = max_idx + 1
            g[:left_len] += a[max_idx::-1]
            # right part: indices max_idx..n-1
            right_len = n - max_idx
            g[max_idx:] += a[:right_len]

            # step size schedule
            eta = 0.5 / np.sqrt(iteration + 1)
            # multiplicative update
            a = a * np.exp(-eta * g)
            a = proj_simplex(a)

            iteration += 1

        # end of size loop
    # final best vector may be of size last size; optionally trim or pad to original length of height_sequence_1

    # Ensure sum>0

"""
Propose a step‑function (non‑negative heights) that minimises
evaluate_sequence().

The algorithm works in three stages:

1.  **Initial seeding** – a handful of diverse probability vectors
    (uniform, geometric, Dirichlet draws and possibly the previous best
    construction) are generated and the best of them (lowest J) is kept.

2.  **Projected exponential gradient descent** – the objective is
    J = 2·n·max(conv(a,a)).  On each iteration the index i* that attains
    the maximum convolution is located, the sub‑gradient of conv_i* w.r.t.
    a is computed analytically (g_k = a_{i*‑k}+a_{k‑i*}) and a is updated
    by a multiplicative (exponential‑gradient) step followed by a simple
    normalisation onto the probability simplex.  The step‑size decays as
    η_t = 0.5 / sqrt(t).  This quickly drives the maximum of the
    convolution down.

3.  **Periodic random perturbations** – every few hundred iterations a
    small log‑normal noise is applied to the current best vector; if it
    yields an improvement it is accepted.  This helps escape shallow
    local minima.

The whole procedure respects the 1000 s time‑budget, prints progress,
and finally returns the best vector found as a plain Python list of
floats.  All helper functions are top‑level; no closures or I/O are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using an FFT of length `fft_len`.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def eval_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute J = 2·n·max(conv(a,a)).
    The vector `a` is assumed to be non‑negative and sum(a)=1.
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

        cand = proj_simplex(cand)  # ensure a valid probability vector
        J = eval_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f} (n = {n})")

    # ------------------------------------------------------------------
    # 3. Main optimisation loop – projected exponential gradient descent
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a small margin
    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        # 3.1 compute full convolution and the index that attains the max
        conv = conv_full_fft(best_vec, fft_len, conv_len)
        i_max = int(np.argmax(conv))
        max_val = conv[i_max]
        J = 2.0 * n * max_val

        # record improvement
        if J < best_J - 1e-12:
            best_J = J
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # 3.2 compute sub‑gradient of conv_i_max w.r.t. a
        g = np.zeros_like(best_vec)

        # contribution a_{i_max‑k}  (k = 0 … i_max)
        left_len = i_max + 1
        g[:left_len] += best_vec[i_max::-1]

        # contribution a_{k‑i_max}  (k = i_max … n‑1)
        right_len = n - i_max
        g[i_max:] += best_vec[:right_len]

        # 3.3 exponential‑gradient update + simplex normalisation
        eta = 0.5 / np.sqrt(iteration + 1)          # decaying step size
        best_vec = best_vec * np.exp(-eta * g)      # stays non‑negative
        # normalise to the probability simplex (simple division is enough)
        s = best_vec.sum()
        if s == 0.0:
            best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            best_vec = best_vec / s

        # 3.4 occasional random perturbation to jump out of plateaus
        if iteration % 500 == 0 and iteration > 0:
            noise = np.exp(rng.normal(0.0, 0.03, size=n))
            pert = proj_simplex(best_vec * noise)
            pert_J = eval_J(pert, n, fft_len, conv_len)
            if pert_J < best_J - 1e-12:
                best_J = pert_J
                best_vec = pert
                now = time.monotonic()
                print(f"[{now - start_time:6.2f}s] Perturbation improved J = {best_J:.6f}")

        # stagnation handling – if no improvement for 30 s, re‑randomise a bit
        if time.monotonic() - last_improve > 30.0:
            # replace 20 % of the mass with fresh Dirichlet noise
            mask = rng.random(n) < 0.20
            fresh = rng.dirichlet(np.ones(mask.sum())) if mask.sum() > 0 else np.array([])
            if fresh.size:
                best_vec[mask] = fresh
                best_vec = proj_simplex(best_vec)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Stagnation reset applied")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final sanitisation and return
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    # ensure a non‑zero sum (should already be 1)
    if best_vec.sum() == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    return [float(v) for v in best_vec]
