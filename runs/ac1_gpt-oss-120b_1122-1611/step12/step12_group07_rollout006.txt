def proj_simplex(v):
    # ensures sum=1, v_i>=0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def hill_climb(a, n, fft_len, conv_len, time_budget):
    best_a = a.copy()
    best_J = evaluate_J(best_a)
    # dynamic step size
    step = 0.1  # initial perturbation magnitude
    iter = 0
    while time remaining:
        # compute conv and max index
        conv = conv_full(a, fft_len, conv_len)
        k_max = int(np.argmax(conv))
        # find list of i that have partner j = k_max - i within [0,n)
        # choose random i among those
        i_candidates = np.arange(max(0, k_max - (n-1)), min(n-1, k_max) + 1)
        i = rng.choice(i_candidates)
        j = k_max - i
        # Determine indices to move mass from: choose the larger of a[i], a[j]
        # We'll try moving epsilon from that to random other index p
        p = rng.integers(0, n)
        while p == i or p == j:
            p = rng.integers(0, n)
        # epsilon: small proportion of a[big]
        big_idx = i if a[i] > a[j] else j
        eps = step * a[big_idx]  # relative
        if eps > a[big_idx]:
            eps = a[big_idx]  # can't exceed
        if eps == 0:
            continue
        # Create new a
        a_new = a.copy()
        a_new[big_idx] -= eps
        a_new[p] += eps
        # ensure non-negative; but always.
        # Normalize to sum=1 (should be same)
        # Evaluate J
        cur_J = evaluate_J(a_new)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a_new.copy()
            a = a_new  # accept move
            last_improve = time.monotonic()
        else:
            # maybe accept with small probability for simulated annealing
            # set temperature decreasing
            T = max(0.01, (time_remaining / time_budget))
            if rng.random() < np.exp(-(cur_J - best_J) / (T + 1e-9)):
                a = a_new
        # Annealing step: gradually reduce step size
        iter += 1
        if iter % 500 == 0:
            step *= 0.9  # reduce step gradually

"""
Particle‑Swarm optimisation on the simplex (logits + softmax) followed by a
simple sub‑gradient refinement.  The objective is the evaluation function
``evaluate_sequence``:
    J(a) = 2·n·max(conv(a,a)) / (∑a)² .
Because J is scale‑invariant we keep ∑a = 1 (probability simplex) and
optimise only the shape of the sequence.

The algorithm consists of three stages:

1.  **Initial seeding** – a small portfolio of hand‑crafted candidates
    (uniform, V‑shaped, geometric decays, Gaussian bumps, a Dirichlet
    draw, and the previous solution if available).  The best of these
    seeds provides a good starting point.

2.  **Particle Swarm Optimisation (PSO)** – particles are represented by
    unconstrained logits; a soft‑max turns them into a probability vector.
    The swarm searches the simplex using the classic PSO update
    (inertia + cognitive + social terms) while keeping track of personal
    and global best positions.  Velocity is clipped to avoid overflow.
    The PSO runs for as long as the time budget allows (leaving a few
    seconds for the final refinement).

3.  **Sub‑gradient refinement** – from the best PSO solution we perform
    a projected sub‑gradient descent on the true max‑convolution
    (the gradient of the active convolution entry is simply
    ``2·a[ k_max−i ]``).  A simple diminishing step‑size together with a
    projection onto the probability simplex drives the solution to a
    (local) minimum of the exact objective.

All heavy work (FFT‑based convolutions, soft‑max, simplex projection) is
implemented as top‑level helper functions.  The routine respects the
``budget_s`` time limit and prints occasional progress reports.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    ``conv_len`` must equal ``2*len(a)-1``.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Row‑wise soft‑max (logits shape (batch, n))."""
    max_log = np.max(logits, axis=1, keepdims=True)
    exp = np.exp(logits - max_log)
    sum_exp = np.sum(exp, axis=1, keepdims=True)
    return exp / sum_exp


def _j_value(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact evaluation J = 2·n·max(conv(a,a)) (∑a=1)."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Projects a vector onto the probability simplex (sum=1, v_i≥0).
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if np.sum(v) == 1.0 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, n: int, k_max: int, lr: float) -> np.ndarray:
    """
    One projected sub‑gradient step for the exact max‑convolution.
    The sub‑gradient of ``max(conv)`` at the active index ``k_max`` is:
        g_i = 2·a_{k_max‑i}  (zero when the index is out of range).
    """
    grad = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    jdx = k_max - idx
    grad[idx] = 2.0 * a[jdx]
    a_new = a - lr * grad
    return _simplex_projection(a_new)

# Main optimisation routine
# ----------------------------------------------------------------------

        if Jc < best_J:
            best_J = Jc
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Particle Swarm Optimisation on the simplex (logits)
    # --------------------------------------------------------------
    m = 30                               # swarm size
    w_inertia = 0.7
    c1 = 1.5
    c2 = 1.5

    # Initialise logits for the swarm
    logits = np.empty((m, n), dtype=np.float64)

    # First few particles are perturbations of the best hand‑crafted seed
    for i in range(min(5, m)):
        logits[i] = np.log(np.maximum(best_a, eps)) + rng.normal(0.0, 0.3, size=n)

    # The remaining particles are random Dirichlet draws (different alphas)
    for i in range(5, m):
        alpha = rng.choice([0.2, 0.5, 1.0, 2.0])
        p = rng.dirichlet(np.ones(n) * alpha)
        logits[i] = np.log(np.maximum(p, eps))

    # PSO bookkeeping
    pbest_logits = logits.copy()
    pbest_J = np.full(m, np.inf)
    gbest_logits = None
    gbest_J = np.inf
    last_gbest_update = time.monotonic()

    iteration = 0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 8.0)  # reserve ~8 s for refinement

    while time.monotonic() < deadline:
        # -----   evaluate swarm   -----
        a_batch = _softmax(logits)                # shape (m, n)
        J_vals = np.empty(m, dtype=np.float64)

        for i in range(m):
            J_vals[i] = _j_value(a_batch[i], n, fft_len, conv_len)

        # -----   update personal and global bests   -----
        improved = J_vals < pbest_J
        pbest_J[improved] = J_vals[improved]
        pbest_logits[improved] = logits[improved]

        idx_best = int(np.argmin(J_vals))
        if J_vals[idx_best] < gbest_J - 1e-12:
            gbest_J = J_vals[idx_best]
            gbest_logits = logits[idx_best].copy()
            last_gbest_update = time.monotonic()
            elapsed = last_gbest_update - start_time
            print(f"[{elapsed:6.2f}s] New global best J = {gbest_J:.6f}")

        # -----   stochastic velocity update   -----
        r1 = rng.random((m, n))
        r2 = rng.random((m, n))
        vel = (w_inertia * getattr(propose_candidate, "_velocity", np.zeros_like(logits))
               + c1 * r1 * (pbest_logits - logits)
               + c2 * r2 * (gbest_logits - logits))
        # Clip velocities to avoid extreme values
        vel = np.clip(vel, -10.0, 10.0)

        # Store velocity for next iteration (as a function attribute)
        propose_candidate._velocity = vel

        # -----   move particles   -----
        logits = logits + vel
        # keep logits in a numerically safe range
        np.clip(logits, -500.0, 500.0, out=logits)

        # -----   stagnation handling   -----
        now = time.monotonic()
        if now - last_gbest_update > 20.0:
            # re‑randomise half of the swarm (keep the best half untouched)
            half = m // 2
            for i in range(half, m):
                alpha = rng.choice([0.2, 0.5, 1.0, 2.0])
                p = rng.dirichlet(np.ones(n) * alpha)
                logits[i] = np.log(np.maximum(p, eps))
                # reset velocity for those particles
            propose_candidate._velocity = np.zeros_like(logits)
            last_gbest_update = now
            print(f"[{now - start_time:6.2f}s] Restarted half of the swarm due to stagnation")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Sub‑gradient refinement from the best PSO solution
    # --------------------------------------------------------------
    if gbest_logits is None:
        # Fallback – use the best handcrafted seed
        a_refine = best_a.copy()
    else:
        a_refine = _softmax(gbest_logits.reshape(1, -1))[0]

    best_refine_J = _j_value(a_refine, n, fft_len, conv_len)
    last_improve = time.monotonic()
    refine_iter = 0
    lr0 = 0.12                          # initial step size
    while time.monotonic() < start_time + budget_s - 1.0:   # keep 1 s buffer
        # compute full convolution once
        conv = _conv_full(a_refine, fft_len, conv_len)
        k_max = int(np.argmax(conv))

        lr = lr0 / np.sqrt(refine_iter + 1)
        a_new = _subgradient_step(a_refine, n, k_max, lr)

        J_new = _j_value(a_new, n, fft_len, conv_len)
        if J_new < best_refine_J - 1e-12:
            best_refine_J = J_new
            a_refine = a_new
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Sub‑gradient improvement: J = {best_refine_J:.6f}")

        # occasional aggressive jitter if stuck
        if time.monotonic() - last_improve > 12.0:
            jitter = rng.normal(0.0, 0.08, size=n)
            a_refine = _simplex_projection(a_refine + jitter)
            last_improve = time.monotonic()

        refine_iter += 1

    # --------------------------------------------------------------
    # 6. Return the best sequence found (as a plain Python list)
    # --------------------------------------------------------------
    final_vec = np.clip(a_refine, 0.0, 1000.0)
    total = final_vec.sum()
    if total == 0.0:
        final_vec = np.full_like(final_vec, 1.0 / n)
    else:
        final_vec /= total   # ensure proper normalisation (optional)

    print(f"Finished. Best J = {best_refine_J:.6f}")
    return [float(v) for v in final_vec]
