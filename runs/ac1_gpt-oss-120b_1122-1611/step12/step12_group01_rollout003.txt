"""
Hybrid DE + Projected Sub‑gradient refinement + tiny random jitter.

The evaluation function  J(a)=2·n·max(conv(a,a))  is scale‑invariant.
We work on the probability simplex (∑a_i=1, a_i≥0) and minimise
the peak of the self‑convolution.  

Algorithm
---------
1. Choose a problem size  (n).  If a previous construction
   `height_sequence_1` exists we reuse its length, otherwise n≈2048.
2. Initialise a diverse population (uniform, geometric, ramps,
   Gaussians, Dirichlet draws, sparse seeds).  All vectors are
   projected onto the simplex.
3. DE/best/1 optimisation for roughly half the time budget.
4. Projected sub‑gradient descent on the *maximum* convolution
   entry for the remaining time (except a few seconds for a final
   stochastic jitter).  At each iteration we:
   * evaluate the full convolution via FFT,
   * collect the active shifts where the convolution is (almost) maximal,
   * build the sub‑gradient  g_k[i]=2·a_{k‑i}  for each active shift,
   * perform a back‑tracking line‑search and project back onto the simplex.
5. A short multiplicative‑noise jitter (shrinking σ) to polish the result.
6. Return the best vector found (clipped to the allowed range and
   re‑normalised – this is exactly what `evaluate_sequence` expects).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a 1‑D vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_full(vec: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution of *vec* obtained via FFT (real‑valued)."""
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def max_conv(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the linear self‑convolution of *vec*."""
    return float(conv_full(vec, fft_len, conv_len).max())


def compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Full objective 2·n·max(conv(vec,vec))."""
    return 2.0 * n * max_conv(vec, fft_len, conv_len)


def subgradient_max_conv(a: np.ndarray, conv_arr: np.ndarray,
                         tol: float = 1e-12) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)_k .
    For each active shift k we add 2·a_{k‑i} to the i‑th component.
    """
    n = a.size
    max_val = conv_arr.max()
    active = np.where(conv_arr >= max_val - tol)[0]
    grad = np.zeros_like(a)

    for k in active:
        # indices i for which both i and k‑i lie in [0,n‑1]
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1, dtype=np.int64)
        # contribution 2·a_{k‑i}
        grad[idx] += 2.0 * a[k - idx]

    return grad

    Return a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The algorithm mixes a differential‑evolution
    phase with a deterministic projected sub‑gradient descent and a final
    stochastic jitter.
    """
    rng = np.random.default_rng(seed)

    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])

    else:

    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)

    # 2. Objective helpers (closed over n, fft_len, conv_len)
    # ------------------------------------------------------------------
    def objective(v: np.ndarray) -> float:
        return compute_J(v, n, fft_len, conv_len)

    # ------------------------------------------------------------------
    # 3. Initial population (size NP)
    # ------------------------------------------------------------------
    NP = 60                               # larger population for better diversity
    population = []

    # hand‑crafted seeds

    # linear ramps
    for asc in (True, False):
        v = np.arange(1, n + 1, dtype=np.float64) if asc else np.arange(n, 0, -1, dtype=np.float64)
        v /= v.sum()
        population.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        population.append(v)

    # Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.1, 0.2, 0.3, 0.6, 1.0):
        for _ in range(2):
            population.append(rng.dirichlet(np.ones(n) * alpha))

    # sparse random draws (small support)
    for _ in range(8):
        k = rng.integers(3, min(200, n // 4))
        idx = rng.choice(n, size=k, replace=False)
        weights = rng.random(k)
        weights /= weights.sum()
        v = np.zeros(n, dtype=np.float64)
        v[idx] = weights
        population.append(v)

    # include previous construction if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = 1.0 / n
        else:
            v /= v.sum()
        population.append(v)

    # Trim/pad to exactly NP members
    if len(population) > NP:
        population = rng.choice(population, size=NP, replace=False).tolist()
    else:
        while len(population) < NP:
            population.append(rng.dirichlet(np.ones(n)))

    # ensure simplex feasibility
    population = [proj_simplex(v) for v in population]

    # ------------------------------------------------------------------
    # 4. Initial evaluation
    # ------------------------------------------------------------------
    fitness = np.empty(NP, dtype=np.float64)
    for i, v in enumerate(population):
        fitness[i] = objective(v)

    best_idx = int(np.argmin(fitness))
    best_vec = population[best_idx].copy()
    best_J = fitness[best_idx]

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a few seconds safety margin
    total_time = deadline - start_time

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Differential Evolution phase (≈ 45 % of the budget)
    # ------------------------------------------------------------------
    de_deadline = start_time + 0.45 * budget_s
    F = 0.8
    CR = 0.9
    last_improve = start_time
    iteration = 0

    while time.monotonic() < de_deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- stagnation handling -----
        if now - last_improve > 30.0:
            # re‑initialise half the population with fresh Dirichlet draws
            for i in range(NP // 2, NP):
                alpha = 10 ** rng.uniform(-2, 0.5)
                vec = rng.dirichlet(np.ones(n) * alpha)
                population[i] = proj_simplex(vec)
                fitness[i] = objective(population[i])
            print(f"[{elapsed:6.2f}s] Restart – half population re‑initialised")
            last_improve = now

        # ----- DE generation -----
        for i in range(NP):
            # distinct donors
            idxs = list(range(NP))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, size=3, replace=False)

            # DE/best/1 donor vector
            donor = population[i] + F * (best_vec - population[i]) + F * (population[r1] - population[r2])
            donor = proj_simplex(donor)

            # binomial crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, donor, population[i])
            trial = proj_simplex(trial)

            trial_J = objective(trial)

            # selection
            if trial_J < fitness[i]:
                population[i] = trial
                fitness[i] = trial_J
                if trial_J < best_J - 1e-12:
                    best_J = trial_J
                    best_vec = trial.copy()
                    last_improve = time.monotonic()
                    print(f"[{time.monotonic() - start_time:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 6. Projected Sub‑gradient descent (≈ 45 % of the remaining budget)
    # ------------------------------------------------------------------
    sgd_deadline = deadline - 8.0                     # leave a few seconds for jitter
    it = 0
    while time.monotonic() < sgd_deadline:
        # compute full convolution once
        conv_arr = conv_full(best_vec, fft_len, conv_len)
        max_val = conv_arr.max()
        cur_J = 2.0 * n * max_val

        # stop if we already cannot beat the target (tiny improvements left)
        if cur_J >= best_J - 1e-12:
            # compute sub‑gradient
            grad = subgradient_max_conv(best_vec, conv_arr, tol=1e-12)

            # step size schedule (decreasing)
            eta0 = 0.25
            eta = eta0 / np.sqrt(it + 1.0)

            improved = False
            # simple back‑tracking line search
            for factor in (1.0, 0.5, 0.25, 0.125, 0.0625):
                step = eta * factor
                candidate = best_vec - step * grad
                candidate = proj_simplex(candidate)
                cand_max = max_conv(candidate, fft_len, conv_len)
                cand_J = 2.0 * n * cand_max
                if cand_J < best_J - 1e-12:
                    best_J = cand_J
                    best_vec = candidate
                    improved = True
                    print(f"[{time.monotonic() - start_time:6.2f}s] SG‑desc iter {it} → J = {best_J:.6f}")
                    break

            if not improved:
                # If simple back‑track fails, reduce step more aggressively
                # (or just break after many failed attempts)
                if eta < 1e-6:
                    break
        else:
            # Ideally we should never get here; keep the best vector.
            pass

        it += 1

    # ------------------------------------------------------------------
    # 7. Final jitter (tiny multiplicative noise) – polishing
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        sigma = max(0.003, 0.15 * remaining / total_time)   # gradually shrink
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_vec * noise
        cand = proj_simplex(cand)
        cand_J = objective(cand)
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand
            print(f"[{time.monotonic() - start_time:6.2f}s] Jitter improvement → J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 8. Return – safety clipping and normalisation
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
