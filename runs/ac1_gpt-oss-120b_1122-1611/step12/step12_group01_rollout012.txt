#!/usr/bin/env python
# -*- coding: utf-8 -*-

"""
Propose a step‐function sequence that minimizes
    J(a) = 2·n·max(conv(a,a)) / (∑a)²
(the scale of a does not matter, we work on the probability simplex).

Algorithm
---------
1.  Initialise a probability vector `a` (uniform, previous best or a random Dirichlet).
2.  Run a *smooth* projected gradient descent.
    *   The non‑smooth max is replaced by a log‑sum‑exp with a temperature β.
        For β → ∞ the smoothed objective converges to the true max.
    *   The gradient of the smoothed objective can be written as a
        convolution:   g = 2·(w * a_rev)[n‑1 : n‑1+n],
        where w are soft‑max weights over the convolution values.
        Fast FFT based convolution gives an O(n log n) gradient.
    *   An exponentiated‑gradient (mirror) step preserves non‑negativity;
        the vector is renormalised to stay on the simplex.
    *   β is increased slowly so the optimisation gradually focuses on the
        true maximum.
3.  Detect stagnation (no improvement >30 s) and restart from a fresh random
    Dirichlet draw.
4.  After the time budget a short stochastic jitter phase tries tiny
    multiplicative perturbations of the current best solution.
5.  Return the best sequence found (clipped to [0,1000] and normalised).

The whole routine respects the 1000 s budget and uses only NumPy
(and the already‑imported `evaluate_sequence` for scoring).
"""

import time
import math
import numpy as np

# -------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas.
# -------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works only on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(conv(vec,vec)) for a probability vector."""
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return 2.0 * n * float(conv.max())


def grad_from_weights(w: np.ndarray, a: np.ndarray,
                     fft_len: int, n: int) -> np.ndarray:
    """
    Compute the gradient of the smoothed objective.

    grad_i = 2 * Σ_k w_k * a_{k-i}
    which equals 2·(w * a_rev)[n-1 : n-1+n].
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, fft_len)
    AR = np.fft.rfft(a_rev, fft_len)
    conv = np.fft.irfft(W * AR, n=fft_len)[: (2 * n - 1 + n - 1)]
    start = n - 1
    return 2.0 * conv[start:start + n]

# -------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs a smoothed gradient descent
    with occasional restarts and a final jitter phase.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # 1. Problem size
    # ---------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        # keep n in a comfortable range for FFT speed
        n = max(256, min(n, 8000))
    else:
        n = 2048

    conv_len = 2 * n - 1
    # we need enough FFT points for (w * a_rev) where w has length conv_len
    fft_len = next_pow2(conv_len + n - 1)

    # ---------------------------------------------------------
    # 2. Initialise the population / current vector
    # ---------------------------------------------------------
    def normalise(v):
        total = v.sum()
        if total == 0.0:
            return np.full_like(v, 1.0 / n)
        return v / total

    # start from the previously discovered construction if it exists,
    # otherwise from the uniform distribution.
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        a0 = normalise(a0)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)

    print(f"[0.00s] Initial J = {best_J:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)  # 1 s safety margin

    # ---------------------------------------------------------
    # 3. Main optimisation loop (smooth gradient descent + restarts)
    # ---------------------------------------------------------
    beta = 1.0                # temperature for the log‑sum‑exp smoothing
    beta_max = 500.0
    eta0 = 0.5                # base step size for exponentiated update
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        # ---- compute convolution of the current vector ----
        A = np.fft.rfft(a, fft_len)
        conv_a = np.fft.irfft(A * A, n=fft_len)[:conv_len]

        a_new = a * np.exp(-eta * grad)
        a_new = normalise(a_new)

        # ---- evaluate exact objective ----
        J_new = compute_J(a_new, n, fft_len, conv_len)

        # ---- keep the best seen so far ----
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Iter {iteration:5d} → new best J = {best_J:.6f}")

        # ---- accept the new point (always, to keep the search moving) ----
        a = a_new

        # ---- increase smoothing temperature slowly ----
        if beta < beta_max:
            beta = min(beta_max, beta * 1.005)

        # ---- stagnation handling – random restart ----
        if time.monotonic() - last_improve > 30.0:
            # fresh Dirichlet draw with moderate concentration
            alpha = 10 ** rng.uniform(-2, 0.5)
            a = rng.dirichlet(np.ones(n) * alpha)
            a = normalise(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Restart – random Dirichlet (α={alpha:.3g})")
            # reset smoothing to give the new point a chance
            beta = 1.0

        iteration += 1

        # occasional cheap jitter to shake out plateaus (no more than once per second)
        if iteration % 500 == 0:
            sigma = max(0.01, 0.2 * (deadline - time.monotonic()) /
                        (deadline - start_time))
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            a = proj_simplex(a * noise)

    # ---------------------------------------------------------
    # 4. Final stochastic jitter phase (uses the remaining budget)
    # ---------------------------------------------------------
    final_deadline = start_time + float(budget_s) - 0.1
    while time.monotonic() < final_deadline:
        remaining = final_deadline - time.monotonic()
        sigma = max(0.003, 0.1 * remaining / (final_deadline - start_time))
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = proj_simplex(best_a * noise)
        Jcand = compute_J(cand, n, fft_len, conv_len)
        if Jcand < best_J - 1e-12:
            best_J = Jcand
            best_a = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Jitter improvement → J = {best_J:.6f}")

    # ---------------------------------------------------------
    # 5. Return result (clipped to allowed range, normalised)
    # ---------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_a = best_a / total

    return [float(v) for v in best_a]
