"""
Projected exponentiated‑gradient descent on the max‑convolution objective.

* The objective that evaluate_sequence() returns is
      J(a) = 2·n·max_k (a * a)[k]   (the sum of a is kept equal to 1)
  because the function clips and normalises the vector internally.
* We minimise the ℓ∞‑norm of the autocorrelation a*a.
* For a given shift k the gradient of (a*a)[k] w.r.t. a is
      ∇_a (a*a)[k]_i = 2·a_{k‑i}   (if the index is inside the support).
  When k is the current maximal shift we obtain a simple sub‑gradient.
* The algorithm uses
      - exponentiated‑gradient (multiplicative) updates, which keep the
        vector non‑negative and stay on the probability simplex after a
        normalisation step;
      - a simulated‑annealing acceptance rule to allow occasional uphill
        moves;
      - periodic “jitter’’ (random multiplicative noise) and full restarts
        to escape plateaus;
      - a cheap exchange move (transfer a tiny amount of mass between two
        random coordinates) to explore the neighbourhood.
* The search runs for the whole allotted time (≈ 1000 s).  The best
  sequence found is returned as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | Σ x_i = 1, x_i ≥ 0}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal or zero → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear (full) convolution a * a via FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Return the objective value J = 2·n·max(conv(a,a)).
    The vector a is assumed to be normalised (Σ a = 1) – this is the
    quantity evaluated by ``evaluate_sequence`` up to the clipping that
    occurs there (our values stay well within the allowed range).
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _max_convolution_gradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution at the active indices ``max_idx``.
    For each active shift k we add 2·a_{k‑i} to the i‑th component.
    """
    n = a.size
    grad = np.zeros_like(a)
    for k in max_idx:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i = np.arange(start, end + 1, dtype=int)
        j = k - i
        grad[i] += 2.0 * a[j]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep it in a reasonable range
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # for a*a convolution

    # ------------------------------------------------------------------
    # 2. Initial vector (simple random start; occasionally blend with a known
    #    good construction if it exists).
    # ------------------------------------------------------------------
    a = rng.random(n).astype(np.float64)
    a /= a.sum()
    if "height_sequence_1" in globals() and rng.random() < 0.4:
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
        a = a0 / max(a0.sum(), 1e-12)

    # ------------------------------------------------------------------
    # 3. Book‑keeping
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)
    cur_J = best_J

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety margin
    total_time = deadline - start_time

    iteration = 1
    last_improve = start_time

    # temperature schedule for simulated annealing
    T_start = best_J
    T_end = 1e-4

    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # 4.1 Compute convolution and locate the maximal shift(s)
        # --------------------------------------------------------------
        conv = _conv_full(a, fft_len, conv_len)
        max_b = conv.max()
        cur_J = 2.0 * n * max_b

        # active set – allow a tiny tolerance to capture ties
        tol = max_b * 1e-12
        active_idx = np.where(conv >= max_b - tol)[0]

        # --------------------------------------------------------------
        # 4.2 Sub‑gradient for the current max‑convolution
        # --------------------------------------------------------------
        grad = _max_convolution_gradient(a, active_idx)

        # --------------------------------------------------------------
        # 4.3 Exponentiated‑gradient update (mirror descent on the simplex)
        # --------------------------------------------------------------
        eta = 0.5 / np.sqrt(iteration)               # decreasing step size
        a_new = a * np.exp(-eta * grad)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= s

        # --------------------------------------------------------------
        # 4.4 Small random exchange move (helps to explore neighbourhood)
        # --------------------------------------------------------------
        if rng.random() < 0.07:
            i, j = rng.choice(n, size=2, replace=False)
            # move at most 0.8 % of the total mass from i to j
            delta = min(a_new[i], rng.random() * 0.008)
            a_new[i] -= delta
            a_new[j] += delta
            # re‑normalise to keep the sum exactly one
            a_new /= a_new.sum()

        # --------------------------------------------------------------
        # 4.5 Evaluate the candidate
        # --------------------------------------------------------------
        conv_new = _conv_full(a_new, fft_len, conv_len)
        new_J = 2.0 * n * conv_new.max()

        # --------------------------------------------------------------
        # 4.6 Simulated‑annealing acceptance
        # --------------------------------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            # temperature decays exponentially from T_start to T_end
            T = T_start * (T_end / T_start) ** (elapsed / total_time)
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_J = new_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 4.7 Stagnation handling – jitter / full restart
        # --------------------------------------------------------------
        if now - last_improve > 20.0:
            # Multiplicative jitter centred on the best solution found so far
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = proj_simplex(best_a * noise)
            last_improve = now
            # recompute J after the jump
            conv = _conv_full(a, fft_len, conv_len)
            cur_J = 2.0 * n * conv.max()
            print(f"[{elapsed:6.2f}s] Jitter – current J = {cur_J:.6f}")

        if now - last_improve > 120.0:
            # Full random restart (blend a bit of the best solution to keep progress)
            a = rng.random(n)
            a /= a.sum()
            if rng.random() < 0.5:
                a = 0.6 * a + 0.4 * best_a
                a = proj_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Random restart")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
