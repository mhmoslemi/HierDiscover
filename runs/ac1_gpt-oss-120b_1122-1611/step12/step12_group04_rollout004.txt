"""
Hybrid simulated‑annealing + short Adam refinement for the
`evaluate_sequence` bound.

The algorithm works in three stages:

1. **Problem size** – we use a power‑of‑two length (≤ 8192) – either the
   length of a previously found construction (`height_sequence_1`) or
   4096 as a default.

2. **Global stochastic search** – a Metropolis simulated‑annealing loop
   that repeatedly proposes three kinds of moves:
        * pairwise mass transfer,
        * redistribution of the mass on a random subset (Dirichlet draw),
        * a small Gaussian smoothing of the whole vector.
   Moves are accepted according to a temperature schedule that cools
   from 10 % of the current objective to near 0.  When no improvement
   occurs for a while a multiplicative jitter is applied to escape the
   basin.

3. **Local polishing** – a short Adam optimiser on a smooth
   log‑sum‑exp approximation of the max‑operator, starting from the
   best vector found by the SA loop.

All vectors are kept on the probability simplex (non‑negative,
summing to 1) by an Euclidean projection.  Convolutions and gradients
are evaluated with real FFTs (`O(n log n)`).  The routine respects the
1000 s time budget and finally returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Gradient of  Σ_k weight_k * (a*a)[k] w.r.t. a.
    weight length = 2·n‑1, a length = n.
    Returned gradient has length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

    T_end = 1e-12

    last_improve = start_time

    # move‑type probabilities
    move_probs = np.array([0.5, 0.3, 0.2])   # pair, subset, smooth

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - start_time) / total_sa_time   # ∈[0,1]
        T = T0 * (1.0 - frac) + T_end * frac

        # ------- pick a move ------------------------------------------------
        move_type = rng.choice(['pair', 'subset', 'smooth'], p=move_probs)

        cand = cur_vec.copy()

        if move_type == 'pair':
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps = rng.random() * cand[i] * 0.5   # move at most half of donor mass
            if eps <= 0.0:
                continue
            cand[i] -= eps
            cand[j] += eps

        elif move_type == 'subset':
            # choose a random subset (between 2 and 5 % of n)
            max_sz = max(2, int(0.05 * n))
            sz = rng.integers(2, max_sz + 1)
            idx = rng.choice(n, size=sz, replace=False)
            total = cand[idx].sum()
            if total <= 0.0:
                continue
            # new Dirichlet proportions (α=1)
            new_prop = rng.dirichlet(np.ones(sz))
            cand[idx] = total * new_prop

        else:   # 'smooth'
            sigma = rng.uniform(0.5, 4.0)
            radius = int(3 * sigma)
            if radius == 0:
                continue
            x = np.arange(-radius, radius + 1, dtype=np.float64)
            kernel = np.exp(-0.5 * (x / sigma) ** 2)
            kernel /= kernel.sum()
            cand = np.convolve(cand, kernel, mode='same')

        # keep on the simplex
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        # ------- Metropolis acceptance ------------------------------------
        if cand_J < cur_J or rng.random() < np.exp(-(cand_J - cur_J) / max(T, 1e-12)):
            cur_vec = cand
            cur_J = cand_J
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------- stagnation handling (jitter) ----------------------------
        if now - last_improve > 60.0:
            # multiplicative jitter around the best vector
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = _objective(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – cur J = {cur_J:.6f}")

    # -------------------- 5. Local Adam refinement --------------------
    # a short run (≈200 iterations) on a smooth log‑sum‑exp of the convolution
    cur = best_vec.copy()
    cur_J = best_J
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    # schedule for the soft‑max temperature (β grows to make the max sharper)
    beta_start = 5.0
    beta_end = 2000.0

    for it in range(200):
        conv = _conv_full(cur, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # soft‑max weights
        beta = beta_start * (beta_end / beta_start) ** (it / 199.0)
        shifted = conv - max_conv
        w_exp = np.exp(beta * shifted)
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum

        # gradient of the smoothed objective
        grad = _weighted_gradient(cur, w, fft_len_corr, n)

        # Adam update with projection onto simplex
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = -0.1 * m_hat / (np.sqrt(v_hat) + eps_adam)
        cur = cur + step
        cur = proj_simplex(cur)

        # keep the best ever
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur.copy()
            print(f"[refine] improvement – J = {best_J:.6f}")

    # -------------------- 6. final normalisation & return -------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
