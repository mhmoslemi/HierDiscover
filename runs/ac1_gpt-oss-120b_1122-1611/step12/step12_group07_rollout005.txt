"""
Propose a candidate step‑function using a simple LP‑based coordinate descent.

The algorithm works on a probability vector a (sum = 1).  In each inner
iteration we

1. compute the self‑convolution a∗a and its current peak M,
2. solve the linear program
       maximise   1ᵀb
       s.t.       a∗b ≤ M            (element‑wise)
                b ≥ 0
   to obtain a new feasible vector b,
3. mix the current a and the LP solution (a←(1‑t)a + t b) and renormalise.

Because the constraint a∗b ≤ M forces the new vector to stay within the
same peak level it can only move mass to positions where the convolution
is small, thus monotonically decreasing the true objective
J(a)=2·n·max(a∗a)/(∑a)².  The method is cheap for n≈256–512 (the LP
has ≤ 511 constraints and ≈ 512 variables) and we repeat it from many
random starts until the time budget expires.  The best sequence found
during the run (according to the true evaluate_sequence) is returned.

If SciPy is not available the routine falls back to the smooth‑max Adam
optimizer that was used previously.
"""

import time
import numpy as np

# try to import a fast LP solver; fall back if unavailable
try:
    from scipy.optimize import linprog
    _HAVE_LINPROG = True
except Exception:  # pragma: no cover
    _HAVE_LINPROG = False


def _lp_step(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    One LP step: maximise sum(b) subject to a⋆b ≤ M and b≥0.
    Returns the optimal b (unnormalised) or None if the LP failed.
    """
    n = a.shape[0]
    m = 2 * n - 1

    # Build the constraint matrix C where (C b)_k = (a⋆b)_k
    C = np.zeros((m, n), dtype=np.float64)
    for k in range(m):
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1, dtype=np.intp)
        C[k, idx] = a[k - idx]          # a_{k-i}

    # Linear programme: maximise 1ᵀb  ⇔ minimise -1ᵀb
    c = -np.ones(n, dtype=np.float64)          # objective
    b_ub = np.full(m, M, dtype=np.float64)     # right‑hand side

    res = linprog(c, A_ub=C, b_ub=b_ub, bounds=(0, None),
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x

    The routine runs for at most `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)   # 1 s safety margin

    best_seq = None
    best_J = np.inf
    iter_count = 0

    # admissible lengths – small enough for a dense LP, large enough to
    # give a decent search space.
    length_choices = [256, 384, 512]

    # Mixing weight: how strongly we move towards the LP solution.
    mix_factor = 0.3

    while time.time() < deadline:
        # ---------------------------------------------------------
        # 1. pick a length and a random starting distribution
        # ---------------------------------------------------------
        n = int(rng.choice(length_choices))

        # random Dirichlet – the concentration parameter is itself random
        alpha = rng.uniform(0.2, 2.5)
        a = rng.dirichlet(np.ones(n) * alpha)          # sums to 1

        # ---------------------------------------------------------
        # 2. inner optimisation loop (LP + mixing)
        # ---------------------------------------------------------
        max_inner = 30
        for inner in range(max_inner):
            now = time.time()
            if now >= deadline:
                break

            # evaluate (cheap, O(n²) but n is ≤ 512)
            cur_J = evaluate_sequence(a.tolist())
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_seq = a.copy()
                print(f"[{now - start_time:6.2f}s] New best J = {best_J:.6f}"
                      f" (n={n}, inner={inner})")

            # -------------------------------------------------
            # 2a. compute current peak of the self‑convolution
            # -------------------------------------------------
            conv = np.convolve(a, a)
            M = conv.max()

            # -------------------------------------------------
            # 2b. LP step (if we have SciPy)
            # -------------------------------------------------
            if _HAVE_LINPROG and n <= 512:
                b_opt = _lp_step(a, M)
                if b_opt is None:
                    # LP failed – give up this inner loop
                    break

                sum_b = b_opt.sum()
                if sum_b == 0.0:
                    break

                # scale b to keep the total mass unchanged (scale‑invariant)
                b_scaled = b_opt * (a.sum() / sum_b)

                # mixing & renormalisation
                a = (1.0 - mix_factor) * a + mix_factor * b_scaled
                a = np.clip(a, 0.0, None)
                total = a.sum()
                if total > 0.0:
                    a /= total
            else:
                # -------------------------------------------------
                # 2c. fallback: a few Adam‑style smooth‑max steps
                # -------------------------------------------------
                # This branch is only reached when SciPy is unavailable.
                # It re‑uses the gradient code from the previous solution.
                # (kept extremely lightweight to avoid extra imports.)
                from math import log, exp

                # smooth‑max parameters (very small β at the beginning)
                beta = 5.0
                beta_max = 200.0
                beta_inc = 1.01

                # compute smooth‑max weights
                m = conv.max()
                shifted = beta * (conv - m)
                shifted = np.clip(shifted, -700, 0)
                w = np.exp(shifted)
                w /= w.sum()

                # gradient of smooth‑max w.r.t. a (FFT version)
                rev_a = a[::-1]
                fft_len = 1 << ((2 * n - 1 + n - 1).bit_length())
                W = np.fft.rfft(w, fft_len)
                A_rev = np.fft.rfft(rev_a, fft_len)
                full = np.fft.irfft(W * A_rev, n=fft_len)[: n + n - 1]
                grad = 2.0 * full[n - 1: n - 1 + n]

                # soft‑max Jacobian
                avg = np.dot(a, grad)
                grad_z = a * (grad - avg)

                # simple Adam‑like update (no moments for speed)
                lr = 0.5 / np.sqrt(inner + 1)
                a = a - lr * grad_z
                a = np.clip(a, 0.0, None)
                total = a.sum()
                if total > 0.0:
                    a /= total
                beta = min(beta_max, beta * beta_inc)

        iter_count += 1
        # occasional global jitter if we are stuck for a while
        if time.time() - start_time > 0.9 * budget_s and best_seq is not None:
            break

    # -------------------------------------------------------------
    # 3. Return the best sequence found (normalised, clipped)
    # -------------------------------------------------------------
    if best_seq is None:
        # very unlikely – fall back to a uniform vector of modest size
        best_seq = np.full(256, 1.0 / 256, dtype=np.float64)

    best_vec = np.clip(best_seq, 0.0, 1000.0)
    total = best_vec.sum()
    if total > 0.0:
        best_vec /= total           # keep sum≈1 (scale‑invariant)

    print(f"Search finished (iterations={iter_count}). Best J = {best_J:.6f}")
    return [float(v) for v in best_vec.tolist()]
