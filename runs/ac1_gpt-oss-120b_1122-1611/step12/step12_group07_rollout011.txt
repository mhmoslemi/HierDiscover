```
def propose_candidate(seed=0, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    n = 4096  # maybe adapt size to 5000? Use max 8192.
    # choose n depending on time: if environment slow, we can adjust.
    
    # Precompute FFT length for full conv
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    
    def compute_J(a):
        a = np.asarray(a, dtype=np.float64)
        total = a.sum()
        if total <= 0:
            return np.inf, None, None
        # Normalise to sum=1 to keep scale invariance (makes easier compute)
        a_norm = a / total
        # Compute conv via FFT
        A = np.fft.rfft(a_norm, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_conv = conv.max()
        J = 2.0 * n * max_conv / (total ** 2)   # total^2 = 1 after norm -> 2 n max
        return J, conv, max_conv
```

"""
Simple simulated‑annealing optimizer for the step‑function problem.

The algorithm proceeds in two stages:
1) a cheap sweep of several low‑dimensional parametric families
   (uniform, beta‑type, cosine‑type, Gaussian, geometric, random Dirichlet)
   to obtain a good initial distribution;
2) a long simulated‑annealing / hill‑climbing phase that mutates the
   probability vector while keeping the sum fixed.  Convolutions are
   evaluated by FFT (O(n log n)) so that many thousands of iterations
   fit comfortably into the 1000 s budget.

The final vector is returned as a list of floats; it satisfies the
non‑negativity constraints and is normalised to unit sum (scaling does
not affect the objective).
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _prepare_fft_params(n: int):
    """Return the useful lengths for full linear convolution via FFT."""
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    return conv_len, fft_len

def _max_conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Return max_k (a*a)[k] using real FFT (a is assumed to sum to 1)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())

def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective J = 2 * n * max(conv)  (a is assumed to sum to 1)."""
    max_conv = _max_conv_fft(a, fft_len, conv_len)
    return 2.0 * n * max_conv

def _candidate_from_beta(n: int, p: float) -> np.ndarray:
    """Beta‑type symmetric shape: (i+½)^{p‑1} (n‑i‑½)^{p‑1}."""
    idx = np.arange(n, dtype=np.float64)
    a = (idx + 0.5) ** (p - 1) * (n - idx - 0.5) ** (p - 1)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _candidate_from_cosine(n: int, p: float) -> np.ndarray:
    """Cosine‑type shape: max(cos(pi*i/(n‑1)),0)^p."""
    idx = np.arange(n, dtype=np.float64)
    theta = np.pi * idx / (n - 1)
    a = np.clip(np.cos(theta), 0.0, None) ** p
    a /= a.sum()
    return a

def _candidate_from_gauss(n: int, sigma: float) -> np.ndarray:
    centre = (n - 1) / 2.0
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
    a /= a.sum()
    return a

def _candidate_from_geo(n: int, r: float) -> np.ndarray:
    """Geometric decay (or its mirror)."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    a /= a.sum()
    return a

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a low‑value sequence for ``evaluate_sequence``.
    The routine runs for at most ``budget_s`` seconds and returns the
    best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size – pick a reasonable n.
    # --------------------------------------------------------------
    # If a previous construction is available, use its length as a hint,
    # otherwise fall back to a power‑of‑two that gives a good trade‑off.
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            # keep n in a comfortable range
            n = max(1024, min(n, 16384))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len, fft_len = _prepare_fft_params(n)

    # --------------------------------------------------------------
    # 2. Build a diverse set of cheap candidates.
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # beta‑type (symmetric) – sweep exponent
    for p in np.linspace(0.3, 6.0, 17):   # 0.3 … 6.0
        candidates.append(_candidate_from_beta(n, p))

    # cosine‑type – sweep exponent
    for p in np.linspace(0.5, 10.0, 13):
        candidates.append(_candidate_from_cosine(n, p))

    # Gaussian – sweep sigma
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 9.0, n / 6.0, n / 4.0):
        candidates.append(_candidate_from_gauss(n, sigma))

    # geometric decays
    for r in (0.90, 0.94, 0.97, 0.99, 0.995):
        candidates.append(_candidate_from_geo(n, r))
        candidates.append(_candidate_from_geo(n, 1.0 / r)[::-1])  # mirror

    # random Dirichlet draws – different concentration levels
    for alpha in (0.1, 0.3, 0.6, 1.0):
        for _ in range(8):
            vec = rng.dirichlet(np.full(n, alpha))
            candidates.append(vec)

    # previous construction, if any (normalised)
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            prev = np.clip(prev, 0.0, None)
            if prev.size != n:
                if prev.size > n:
                    prev = prev[:n]
                else:
                    prev = np.concatenate([prev,
                                           np.full(n - prev.size, 1.0 / n)])
            total = prev.sum()
            if total == 0.0:
                prev = np.full(n, 1.0 / n)
            else:
                prev /= total
            candidates.append(prev)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Evaluate the cheap candidates, keep the best.
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for a in candidates:
        J = _objective(a, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    start_time = time.monotonic()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing / hill‑climbing refinement.
    # --------------------------------------------------------------
    # Keep a separate copy of the current state
    cur_a = best_a.copy()
    cur_J = best_J

    # Hyper‑parameters for the annealing schedule
    T = 0.02                       # initial temperature
    T_min = 1e-5
    cooling = 0.99996              # multiplicative cooling per iteration
    move_scale = 0.5               # initial fraction of the source mass to move
    move_decay = 0.99994           # decay of move_scale per iteration

    # Stagnation handling
    last_improve = start_time
    restart_interval = 30.0        # seconds without improvement → jitter

    iteration = 0
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # 4.1 Propose a small mass transfer i → j
        # ----------------------------------------------------------
        i = rng.integers(n)
        # ensure we pick a different destination
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if cur_a[i] == 0.0:
            # cannot take mass from an empty cell – skip this loop
            continue

        # choose an amount proportional to the source amplitude
        eps = cur_a[i] * move_scale * rng.random()
        # enforce non‑negativity (eps ≤ cur_a[i] automatically)
        a_new = cur_a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # ----------------------------------------------------------
        # 4.2 Compute the objective for the proposal
        # ----------------------------------------------------------
        J_new = _objective(a_new, n, fft_len, conv_len)

        # ----------------------------------------------------------
        # 4.3 Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if J_new < cur_J:
            accept = True
        else:
            # uphill move – accept with probability exp(-(Δ)/T)
            delta = J_new - cur_J
            if T > 0.0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_a = a_new
            cur_J = J_new
            iteration += 1

            # record a new global best
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = cur_a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 4.4 Update schedules
        # ----------------------------------------------------------
        T = max(T_min, T * cooling)
        move_scale = max(0.001, move_scale * move_decay)

        # ----------------------------------------------------------
        # 4.5 Stagnation detection – apply jitter around the best point
        # ----------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > restart_interval:
            # Create a jittered copy of the current best solution.
            jitter = rng.normal(0.0, 0.08, size=n)
            # Work in log‑space to keep the simplex constraint simple.
            log_a = np.log(np.maximum(best_a, 1e-30))
            log_a += jitter
            a_jit = np.exp(log_a)
            a_jit /= a_jit.sum()
            cur_a = a_jit
            cur_J = _objective(cur_a, n, fft_len, conv_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {best_J:.6f})")

    # --------------------------------------------------------------
    # 5. Return the best vector (clip just in case, normalise again).
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / total

    print(f"Finished. Best J = {best_J:.6f}")
    return [float(v) for v in best_a]

"""
Simple simulated‑annealing optimizer for the step‑function problem.

The algorithm runs in two phases:

1. **Cheap parametric sweep** – a handful of low‑dimensional families
   (uniform, beta‑type, cosine‑type, Gaussian, geometric and random
   Dirichlet draws) are evaluated with an FFT‑based convolution.  The best
   candidate from this sweep is used as the starting point.

2. **Refinement** – a long simulated‑annealing / hill‑climbing phase
   repeatedly moves a small amount of probability mass from one index to
   another while keeping the total sum equal to one.  The objective
   ``J = 2·n·max(a*a)`` (the scaling‑invariant form of
   ``evaluate_sequence``) is computed by a real FFT, so thousands of
   iterations fit comfortably into the 1000 s budget.  Temperature,
   move‑size and a periodic jitter restart are scheduled automatically.
   The best vector found is returned as a list of floats; all entries are
   non‑negative and small enough to avoid clipping inside
   ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _prepare_fft_params(n: int):
    """Return convolution length and a suitable FFT length."""
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    return conv_len, fft_len


def _max_conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Compute max_k (a * a)[k] using a real FFT.
    The vector ``a`` is assumed to sum to one (scaling does not matter).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective J = 2·n·max(conv) for a probability vector a."""
    return 2.0 * n * _max_conv_fft(a, fft_len, conv_len)


# ----------------------------------------------------------------------
# Families of parametrised candidates
# ----------------------------------------------------------------------


def _candidate_from_beta(n: int, p: float) -> np.ndarray:
    """Symmetric beta‑type shape: (i+½)^{p‑1}·(n‑i‑½)^{p‑1}."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 0.5) ** (p - 1) * (n - i - 0.5) ** (p - 1)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_from_cosine(n: int, p: float) -> np.ndarray:
    """Cosine shape: max(cos(π·i/(n‑1)),0)^p."""
    i = np.arange(n, dtype=np.float64)
    theta = np.pi * i / (n - 1)
    a = np.clip(np.cos(theta), 0.0, None) ** p
    a /= a.sum()
    return a


def _candidate_from_gauss(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the interval."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def _candidate_from_geo(n: int, r: float) -> np.ndarray:
    """Geometric decay (or its mirror)."""
    i = np.arange(n, dtype=np.float64)
    a = r ** i
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    evaluation function ``evaluate_sequence``.  The routine runs for at
    most ``budget_s`` seconds and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(1024, min(n, 16384))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len, fft_len = _prepare_fft_params(n)

    # --------------------------------------------------------------
    # 2. Build a diverse set of cheap candidates.
    # --------------------------------------------------------------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # beta‑type sweep
    for p in np.linspace(0.3, 6.0, 17):
        candidates.append(_candidate_from_beta(n, p))

    # cosine‑type sweep
    for p in np.linspace(0.5, 10.0, 13):
        candidates.append(_candidate_from_cosine(n, p))

    # Gaussian sweep
    for sigma in (n / 12.0, n / 9.0, n / 6.0, n / 4.0):
        candidates.append(_candidate_from_gauss(n, sigma))

    # geometric decays (both directions)
    for r in (0.90, 0.94, 0.97, 0.99, 0.995):
        candidates.append(_candidate_from_geo(n, r))
        candidates.append(_candidate_from_geo(n, 1.0 / r)[::-1])

    # random Dirichlet draws (various concentrations)
    for alpha in (0.1, 0.3, 0.6, 1.0):
        for _ in range(8):
            candidates.append(rng.dirichlet(np.full(n, alpha)))

    # previous construction, if present
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(height_sequence_1, dtype=np.float64)
            v = np.clip(v, 0.0, None)
            if v.size != n:
                if v.size > n:
                    v = v[:n]
                else:
                    v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
            total = v.sum()
            if total == 0.0:
                v = np.full(n, 1.0 / n)
            else:
                v /= total
            candidates.append(v)
        except Exception:
            pass

    # --------------------------------------------------------------
    # 3. Evaluate the cheap candidates and keep the best.
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for a in candidates:
        J = _objective(a, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_a = a.copy()
    start_time = time.monotonic()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing / hill‑climbing refinement.
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_J = best_J

    # annealing schedule
    T = 0.02            # initial temperature
    T_min = 1e-5
    cooling = 0.99996   # geometric cooling per iteration
    move_scale = 0.5    # initial fraction of the source mass to move
    move_decay = 0.99994

    last_improve = start_time
    restart_interval = 30.0   # seconds without improvement → jitter
    iteration = 0

    # time budget (leave a small safety margin)
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    while time.monotonic() < deadline:
        # ------------------------------------------------------
        # 4.1 Propose a small transfer i → j
        # ------------------------------------------------------
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        if cur_a[i] == 0.0:
            continue

        eps = cur_a[i] * move_scale * rng.random()
        a_new = cur_a.copy()
        a_new[i] -= eps
        a_new[j] += eps

        # ------------------------------------------------------
        # 4.2 Evaluate the proposal
        # ------------------------------------------------------
        J_new = _objective(a_new, n, fft_len, conv_len)

        # ------------------------------------------------------
        # 4.3 Metropolis acceptance rule
        # ------------------------------------------------------
        accept = False
        if J_new < cur_J:
            accept = True
        else:
            delta = J_new - cur_J
            if T > 0.0 and rng.random() < np.exp(-delta / T):
                accept = True

        if accept:
            cur_a = a_new
            cur_J = J_new
            iteration += 1

            # global improvement?
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = cur_a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------
        # 4.4 Update schedules
        # ------------------------------------------------------
        T = max(T_min, T * cooling)
        move_scale = max(0.001, move_scale * move_decay)

        # ------------------------------------------------------
        # 4.5 Stagnation handling – jitter around the best point
        # ------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > restart_interval:
            # add log‑space Gaussian jitter
            log_a = np.log(np.maximum(best_a, 1e-30))
            log_a += rng.normal(0.0, 0.08, size=n)
            a_jit = np.exp(log_a)
            a_jit /= a_jit.sum()
            cur_a = a_jit
            cur_J = _objective(cur_a, n, fft_len, conv_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {best_J:.6f})")

    # --------------------------------------------------------------
    # 5. Normalise and return the best solution.
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a = best_a / total

    print(f"Finished. Best J = {best_J:.6f}")
    return [float(v) for v in best_a]
