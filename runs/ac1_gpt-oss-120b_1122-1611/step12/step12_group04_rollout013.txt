def weighted_gradient(a, w, fft_len_corr, n):
    a_rev = a[::-1]
    W_hat = np.fft.rfft(w, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    grad = 2.0 * full[n-1:n-1 + n]
    return grad

Define softmax weight:

```
def softmax_weights(conv, beta):
    max_val = conv.max()
    shifted = conv - max_val
    exp_vals = np.exp(beta * shifted)
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        return np.full_like(exp_vals, 1.0 / exp_vals.size)
    return exp_vals / sum_exp

"""
Hybrid optimizer for the self‑convolution max problem.

The objective J = 2·n·max(conv(a,a)) is scale‑invariant, so we constrain
∑ a_i = 1 and a_i ≥ 0.  The search proceeds in four stages:

1.  Diverse initial simplex points are generated and the best is kept.
2.  Adam optimisation on a smooth soft‑max surrogate of the max, with a
    temperature‑controlled Metropolis rule and a schedule that turns the
    soft‑max into a hard max (γ→1).  Stagnation triggers a multiplicative
    jitter.
3.  Simulated‑annealing local search that mixes random pairwise mass swaps
    with targeted moves that directly attack the dominant convolution peak.
4.  A small linear‑programming refinement that maximises the total mass
    under the currently active convolution constraints; after renormalisation
    this typically lowers the peak further.
5.  A final short polishing pass with simple pairwise swaps.

All heavy work (convolution, gradients) uses real FFTs (O(n log n)).
The algorithm adapts to the remaining wall‑clock time (default 1000 s) and
returns the best probability vector found, clipped to the limits of
`evaluate_sequence`.
"""

import time
import numpy as np
from scipy import sparse
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear auto‑convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """2·corr(weight, a) via FFT (valid part of length n)."""
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(conv: np.ndarray, n: int) -> float:
    """Return J = 2·n·max(conv)."""
    return float(2.0 * n * conv.max())


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max on the convolution vector."""
    max_val = conv.max()
    shifted = conv - max_val
    exp_vals = np.exp(beta * shifted)
    s = exp_vals.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_vals / s


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found within
    the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(512, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    # Use a power of two for FFT efficiency
    n_pow2 = _next_pow2(n)
    if n_pow2 != n:
        n = n_pow2

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for correlation

    # ------------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.94, 0.97, 0.985, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 12.0, n / 9.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws – various concentrations
    for alpha in [0.02, 0.05, 0.1, 0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Random exponential draws
    for _ in range(3):
        v = rng.exponential(scale=1.0, size=n)
        v /= v.sum()
        init_cands.append(v)

    # previous construction, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v /= s
        else:
            v = np.full(n, 1.0 / n)
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best initial candidate (fast evaluation)
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_full_fft(cand, fft_len, conv_len)
        J = _objective(conv, n)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    start_time = time.monotonic()
    deadline = start_time + float(budget_s)

    # ------------------------------------------------------------------
    # 4. Phase‑1: Adam optimisation on a soft‑max surrogate
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.25 * budget_s
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_vec_local = cur_vec.copy()
    best_J_local = cur_J

    # Adam hyper‑parameters
    lr0 = 0.8
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-9

    # Soft‑max schedule (β from 1 to 8000)
    beta_start, beta_end = 1.0, 8000.0

    # Metropolis temperature schedule
    T_initial = best_J_local * 0.1
    T_final = 1e-9

    m = np.zeros_like(cur_vec)
    v = np.zeros_like(cur_vec)
    it = 0
    last_improve = start_time

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / (phase1_end - start_time)

        # --- current convolution and objective ---
        conv = _conv_full_fft(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        # --- record best ---
        if cur_J < best_J_local - 1e-12:
            best_J_local = cur_J
            best_vec_local = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Adam new best J = {best_J_local:.6f}")

        # --- stagnation handling ---
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_vec_local * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Adam jitter restart (best J = {best_J_local:.6f})")

        # --- soft‑max weights ---
        beta = beta_start * (beta_end / beta_start) ** t_frac
        w_soft = _softmax_weights(conv, beta)

        # --- hard max weight (active set) ---
        tol = 1e-12
        max_inds = np.where(conv >= max_conv - tol)[0]
        w_hard = np.zeros_like(conv)
        w_hard[max_inds] = 1.0 / len(max_inds)

        # blend soft and hard (γ ramps from 0 to 1)
        gamma = t_frac
        w = (1.0 - gamma) * w_soft + gamma * w_hard

        # --- gradient ---
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)

        # --- Adam update ---
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 * (1.0 - t_frac) + 1e-3
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        proposal = cur_vec + step
        proposal = np.maximum(proposal, 0.0)
        proposal = proj_simplex(proposal)

        # --- evaluate proposal ---
        conv_prop = _conv_full_fft(proposal, fft_len, conv_len)
        J_prop = 2.0 * n * conv_prop.max()

        # --- Metropolis acceptance ---
        T = (T_initial * (phase1_end - now) + T_final * (now - start_time)) / (phase1_end - start_time)
        if J_prop < cur_J:
            accept = True
        else:
            prob = np.exp(-(J_prop - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = proposal
            cur_J = J_prop

    # ------------------------------------------------------------------
    # 5. Phase‑2: Simulated‑annealing local search (random + targeted)
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * budget_s
    cur_vec = best_vec_local.copy()
    conv = _conv_full_fft(cur_vec, fft_len, conv_len)
    cur_J = 2.0 * n * conv.max()
    best_vec = cur_vec.copy()
    best_J = cur_J

    # temperature schedule for annealing
    T0 = cur_J * 0.01
    T_end = 1e-9
    phase2_start = time.monotonic()

    while time.monotonic() < phase2_end:
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - phase2_start) / (phase2_end - phase2_start)
        T = T0 * (1.0 - frac) + T_end * frac

        # with probability 0.2 perform a targeted move aimed at the peak
        if rng.random() < 0.2:
            # index of current peak
            k_max = int(conv.argmax())
            i_min = max(0, k_max - (n - 1))
            i_max = min(n - 1, k_max)
            i_vals = np.arange(i_min, i_max + 1, dtype=int)
            j_vals = k_max - i_vals
            contribs = cur_vec[i_vals] * cur_vec[j_vals]
            idx = int(contribs.argmax())
            i = int(i_vals[idx])
            j = int(j_vals[idx])
            donor = i if cur_vec[i] >= cur_vec[j] else j
            # pick a random receiver that is not the donor
            recv = rng.integers(0, n)
            while recv == donor:
                recv = rng.integers(0, n)
            eps = rng.random() * cur_vec[donor] * 0.5
            if eps <= 0.0:
                continue
            cand = cur_vec.copy()
            cand[donor] -= eps
            cand[recv] += eps
        else:
            # pure random pairwise swap
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            eps = rng.random() * cur_vec[i] * 0.5
            if eps <= 0.0:
                continue
            cand = cur_vec.copy()
            cand[i] -= eps
            cand[j] += eps

        # protect against negativity and normalise back onto the simplex
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        # evaluate
        conv_cand = _conv_full_fft(cand, fft_len, conv_len)
        J_cand = 2.0 * n * conv_cand.max()

        # Metropolis rule
        if J_cand < cur_J:
            accept = True
        else:
            prob = np.exp(-(J_cand - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = cand
            cur_J = J_cand
            conv = conv_cand
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{elapsed:6.2f}s] Annealing improved best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Phase‑3: Active‑set linear‑programming refinement
    # ------------------------------------------------------------------
    phase3_end = start_time + 0.85 * budget_s
    # Use only the most critical convolution constraints (top‑k peaks)
    while time.monotonic() < phase3_end:
        # recompute convolution for the current best vector
        conv = _conv_full_fft(best_vec, fft_len, conv_len)
        M = conv.max()

        # pick a modest number of active constraints (e.g. 150)
        k_cons = min(150, conv_len)
        top_idx = np.argpartition(conv, -k_cons)[-k_cons:]

        # build sparse matrix A_ub of shape (k_cons, n)
        rows = []
        cols = []
        data = []
        for row_id, idx in enumerate(top_idx):
            i_min = max(0, idx - (n - 1))
            i_max = min(n - 1, idx)
            cols_range = np.arange(i_min, i_max + 1, dtype=np.int32)
            rows_range = np.full(cols_range.shape, row_id, dtype=np.int32)
            rows.append(rows_range)
            cols.append(cols_range)
            data.append(best_vec[i_min:i_max + 1])
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        data = np.concatenate(data)

        A_ub = sparse.csr_matrix((data, (rows, cols)), shape=(k_cons, n))
        b_ub = np.full(k_cons, M)

        # objective: maximise sum(b)  → minimise -sum(b)
        c = -np.ones(n)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs')
        if not res.success:
            # if the LP fails we simply abort this phase
            print("LP solver failed – exiting LP refinement.")
            break

        b_opt = res.x
        sum_b = b_opt.sum()
        if sum_b <= 1.0 + 1e-12:
            # nothing to improve
            break

        # renormalise to keep sum = 1
        g = b_opt / sum_b
        conv_g = _conv_full_fft(g, fft_len, conv_len)
        J_g = 2.0 * n * conv_g.max()
        if J_g < best_J - 1e-12:
            best_J = J_g
            best_vec = g.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] LP refined J = {best_J:.6f}")
        else:
            # no improvement → stop the LP loop
            break

    # ------------------------------------------------------------------
    # 7. Phase‑4: Final cheap polishing (pairwise swaps, simulated anneal)
    # ------------------------------------------------------------------
    polish_deadline = deadline
    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full_fft(cur_vec, fft_len, conv_len)

    T0_polish = cur_J * 0.005
    T_end_polish = 1e-9

    while time.monotonic() < polish_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / budget_s
        T = T0_polish * (1.0 - t_frac) + T_end_polish * t_frac

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps = rng.random() * cur_vec[i] * 0.3
        if eps <= 0.0:
            continue
        cand = cur_vec.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_conv = _conv_full_fft(cand, fft_len, conv_len)
        cand_J = 2.0 * n * cand_conv.max()

        if cand_J < cur_J:
            cur_vec = cand
            cur_J = cand_J
            cur_conv = cand_conv
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{elapsed:6.2f}s] Polishing improved J = {best_J:.6f}")
        else:
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur_vec = cand
                cur_J = cand_J
                cur_conv = cand_conv

    # ------------------------------------------------------------------
    # 8. Final normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
