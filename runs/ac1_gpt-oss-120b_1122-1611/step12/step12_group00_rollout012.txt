def propose_candidate(...):

    import numpy as np, time
    rng = np.random.default_rng(seed)

    # problem size
    if 'height_sequence_1' in globals():
        v0 = np.array(height_sequence_1, dtype=np.float64)
        # enforce nonneg, clip
        v0 = np.clip(v0, 0.0, None)
        n_original = v0.size
        n = n_original  # use same length; could adapt
    else:
        n = 4096
        v0 = np.full(n, 1.0/n)  # uniform

    # project to simplex
    v0 = proj_simplex(v0)

    # conv params
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    # initial convolution
    conv = conv_full_fft(v0, fft_len, conv_len)

    # compute J
    max_b = conv.max()
    current_J = 2.0*n*max_b

    # best
    best_a = v0.copy()
    best_J = current_J

    print(f"[0.00s] Start length {n}, init J {current_J:.6f}")

    # Simulated annealing loop
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # leave margin
    # Temperature schedule
    init_temp = best_J * 0.01   # 1% of current value as start
    min_temp = best_J * 1e-6
    # define counters
    it = 0
    last_improve_time = start
    recompute_interval = 5000

    while time.monotonic() < deadline:
        it += 1
        # progress for temp
        now = time.monotonic()
        elapsed = now - start
        progress = elapsed / (deadline - start)
        temperature = init_temp * (1.0 - progress) + min_temp

        # pick i with positive mass
        # optionally choose i based on probability proportional to a_i to bias away from 0
        # Weighted choice using a (non zero)
        # if many zeros, use rng.choice of indices with weight = a + epsilon
        # simple: select i = rng.choice(n, p=v0) may produce zero sometimes due to zero probability.
        # but we need index where a_i >0. Use while loop.
        i_idx = rng.integers(0, n)
        # ensure a_i > epsilon
        attempts = 0
        while v0[i_idx] < 1e-12 and attempts < 10:
            i_idx = rng.integers(0, n)
            attempts += 1
        # if still zero, skip iteration
        if v0[i_idx] < 1e-12:

        # pick j different
        j_idx = rng.integers(0, n-1)
        if j_idx >= i_idx:
            j_idx += 1

        # choose delta up to some fraction
        max_move = min(v0[i_idx], 0.2)   # cap at 0.2
        delta = rng.random() * max_move

        # Build candidate conv
        c_new = conv.copy()

        # slice for i (shift_i) subtract
        c_new[i_idx:i_idx+n] -= 2.0 * delta * v0
        # slice for j (shift_j) add

        center = n - 1
        c_new[center] += 2.0 * delta * delta

            idx_neg = center - d

            if 0 <= idx_neg < conv_len:
                c_new[idx_neg] -= delta * delta

"""
Simulated‑Annealing step‑function optimiser.

The evaluation metric is   J(a) = 2·n·max(conv(a,a)) / (sum(a)²),
which is homogeneous – we can fix   sum(a)=1   and minimise
   J = 2·n·max(conv) .
The algorithm starts from the best known construction (global
variable ``height_sequence_1``) if it exists, otherwise from a uniform
vector.  It performs a temperature‑controlled stochastic walk on the
probability simplex:

* pick a source index i with positive weight,
* pick a distinct target index j,
* move a random amount δ ≤ a[i] from i to j,
* update the linear convolution a∗a analytically (O(n) slice ops
  plus three impulse updates) and evaluate the new maximum,
* accept the move if it improves J, otherwise with Metropolis
  probability exp(-(ΔJ)/T).

Periodically the full convolution is recomputed by FFT to avoid drift,
and a short jitter (multiplicative Gaussian noise) is applied after
30 s of stagnation.  The best vector seen is returned (normalised,
clipped to the limits of ``evaluate_sequence``).  The routine respects a
time budget (default 1000 s) and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]

# Main search routine
# ----------------------------------------------------------------------

    """

        n = seq0.size

    else:

        elapsed = now - start_time
        # linear cooling schedule
        progress = (elapsed) / (deadline - start_time)
        T = max(min_temp, init_temp * (1.0 - progress))

        # ---- pick a move -------------------------------------------------
        # source index with positive weight
        src = rng.integers(0, n)
        # try a few times to find a non‑zero source
        for _ in range(10):
            if a[src] > 1e-12:
                break
            src = rng.integers(0, n)
        if a[src] <= 1e-12:
            continue  # nothing to move

        # distinct target index
        tgt = rng.integers(0, n - 1)
        if tgt >= src:
            tgt += 1

        # random amount – up to 20 % of the source mass (but not more than the source)
        max_move = min(a[src], 0.20)
        delta = rng.random() * max_move
        if delta <= 0.0:
            continue

        # ---- compute candidate convolution --------------------------------
        c_new = conv.copy()
        # slice updates – shift_i and shift_j
        c_new[src:src + n] -= 2.0 * delta * a
        c_new[tgt:tgt + n] += 2.0 * delta * a

        # ΔΔ impulse contributions
        centre = n - 1                     # index of lag 0 (not used directly here)
        # impulses at 2*src, 2*tgt and -(i+j) offset
        c_new[2 * src] += delta * delta
        c_new[2 * tgt] += delta * delta
        ij = src + tgt
        c_new[ij] -= 2.0 * delta * delta

        # ---- evaluate candidate -------------------------------------------
        new_max = c_new.max()
        new_J = 2.0 * n * new_max

        # ---- Metropolis acceptance ----------------------------------------
        accept = False
        if new_J < current_J:
            accept = True
        else:
            diff = new_J - current_J
            if T > 0.0 and rng.random() < np.exp(-diff / T):
                accept = True

        # ---- commit if accepted -------------------------------------------
        if accept:
            a[src] -= delta
            a[tgt] += delta
            conv = c_new
            current_J = new_J

            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

                # early exit if target reached
                if best_J <= 1.5025:
                    print("[INFO] Target reached – terminating early.")
                    break

        # ---- periodic full recompute (numerical drift) --------------------
        if it % recompute_every == 0:
            conv = conv_full_fft(a, fft_len, conv_len)
            max_b = conv.max()
            current_J = 2.0 * n * max_b

        # ---- stagnation handling – jitter ---------------------------------
        if now - last_improve > stagnation_limit:
            # small multiplicative Gaussian perturbation around the current best
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a_tmp = best_a * jitter
            a_tmp = proj_simplex(a_tmp)
            conv_tmp = conv_full_fft(a_tmp, fft_len, conv_len)
            J_tmp = 2.0 * n * conv_tmp.max()
            if J_tmp < best_J:
                best_J = J_tmp
                best_a = a_tmp
                a = a_tmp
                conv = conv_tmp
                current_J = J_tmp
                print(f"[{elapsed:6.2f}s] Jitter improved J = {best_J:.6f}")
            else:
                # restart from the best without jitter
                a = best_a.copy()
                conv = conv_full_fft(a, fft_len, conv_len)
                current_J = 2.0 * n * conv.max()
                print(f"[{elapsed:6.2f}s] Restarted from best after stagnation.")
            last_improve = now

    # ---------------------------------------------------------------
    # 4. Final fine‑grained random jitter (last ~5 s) -----------------
    # ---------------------------------------------------------------
    fine_deadline = time.monotonic() + 4.0
    while time.monotonic() < fine_deadline:
        jitter = np.exp(rng.normal(0.0, 0.01, size=n))
        cand = best_a * jitter
        cand = proj_simplex(cand)
        conv_c = conv_full_fft(cand, fft_len, conv_len)
        cand_J = 2.0 * n * conv_c.max()
        if cand_J < best_J:
            best_J = cand_J
            best_a = cand
            print(f"[Fine] Improved J = {best_J:.6f}")

    # ---------------------------------------------------------------
    # 5. Normalise and clip for safety (matches evaluate_sequence) ----
    # ---------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
