    # Determine length n.
    if 'height_sequence_1' in globals():
        try:
            h = np.asarray(globals()['height_sequence_1'], dtype=np.float64)

            n = 2048
    else:

    # Initialize candidates
    init_cands = []
    # Uniform

    # Triangular decreasing
    tri = np.arange(n, 0, -1, dtype=np.float64)

    # Gaussian

"""
Simulated‑Annealing + Gradient‑Descent search on the probability simplex.

The routine builds a simplex‑constrained probability vector `a` (non‑negative,
sum = 1) of length *n* (taken from a previous solution or set to 2048).  The
objective   J(a) = 2·n·max(conv(a,a))  is evaluated via a real FFT for
speed.  A mixed proposal distribution is used:

* multiplicative log‑normal perturbations (global random search),
* a deterministic gradient step that lowers the current maximal convolution
  entry (∂max/∂a_i = 2·a_{k_max‑i}) with a decreasing step size η.

These moves are accepted with a classic simulated‑annealing rule, the
temperature T decreasing linearly during the allotted budget.  Periodic
stagnation handling restarts the current state with a fresh Dirichlet draw.
After the main loop a short fine‑grained hill‑climbing phase (tiny
log‑normal noise) refines the best solution found.  All vectors are projected
onto the probability simplex after each modification, guaranteeing validity
for `evaluate_sequence`.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Objective J = 2·n·max(conv(a,a)) for a simplex‑constrained vector.
    The vector is first normalized to sum = 1 (the evaluation routine does the
    same internally).  Returns np.inf for degenerate inputs.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = conv_full_fft(a_norm, fft_len, conv_len)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


def gradient_step(a: np.ndarray, peak_idx: int, eta: float) -> np.ndarray:
    """
    One deterministic step that decreases the current maximal convolution
    entry.  The gradient of max(conv) w.r.t. a_i is 2·a_{peak_idx‑i} (0 if out
    of range).  The step moves opposite to the gradient and the result is
    projected back onto the simplex.
    """
    n = a.size
    grad = np.zeros_like(a)
    i_start = max(0, peak_idx - (n - 1))
    i_end = min(n - 1, peak_idx)
    idxs = np.arange(i_start, i_end + 1)
    grad[idxs] = 2.0 * a[peak_idx - idxs]
    new = a - eta * grad
    new = np.maximum(new, 0.0)
    new = proj_simplex(new)
    return new


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The method combines simulated annealing with a
    gradient‑descent style move that directly attacks the peak of the
    self‑convolution.  The best sequence found before the time budget expires
    is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            h_prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = max(256, min(h_prev.size, 5000))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Build a small set of diverse initial candidates.
    # ------------------------------------------------------------------
    init_cands = []

    # Uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Linear decreasing (triangular)
    tri = np.arange(n, 0, -1, dtype=np.float64)
    init_cands.append(tri / tri.sum())

    # Gaussian bell
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = n / 6.0
    gau = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    init_cands.append(gau / gau.sum())

    # Arcsine‑like (Beta(0.5, 0.5) shape)
    i = np.arange(1, n + 1, dtype=np.float64)
    arcs = 1.0 / np.sqrt(i * (n + 1 - i) + 1e-12)
    init_cands.append(arcs / arcs.sum())

    # Dirichlet draws with several concentration parameters
    for alpha in (0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # Previous best if it exists
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if v.size != n:
                if v.size > n:
                    v = v[:n]
                else:
                    v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            init_cands.append(v)
        except Exception:
            pass

    # Evaluate initial candidates and keep the best
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)
        J = compute_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()

    # ------------------------------------------------------------------
    # 3. Initialise the annealing state.
    # ------------------------------------------------------------------
    cur = best_vec.copy()
    cur_J = best_J

    start_time = time.monotonic()
    # keep a small safety margin for the final refinement
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Annealing schedules (linear interpolation)
    T0 = cur_J * 0.3 + 1e-12          # initial temperature
    T_end = cur_J * 1e-4 + 1e-12     # final temperature
    sigma0 = 0.5                     # initial std‑dev for log‑normal noise
    sigma_end = 0.001                # final std‑dev
    eta0 = 0.2                       # initial step size for gradient move
    eta_end = 0.001                  # final step size

    last_improve = start_time
    prob_grad = 0.4                  # fraction of gradient proposals

    print(f"[0.00s] Initial best J = {best_J:.6f} (n = {n})")

    # ------------------------------------------------------------------
    # 4. Main simulated‑annealing loop.
    # ------------------------------------------------------------------
    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # progress ∈ [0,1]
        progress = (now - start_time) / (deadline - start_time)
        T = T0 * (1 - progress) + T_end * progress
        sigma = sigma0 * (1 - progress) + sigma_end * progress
        eta = eta0 * (1 - progress) + eta_end * progress

        # Current convolution and its peak index
        conv_cur = conv_full_fft(cur, fft_len, conv_len)
        peak_idx = int(np.argmax(conv_cur))

        # ------------------------------------------------------------------
        # 4.1 Propose a new candidate (gradient move or random perturbation)
        # ------------------------------------------------------------------
        if rng.random() < prob_grad:
            cand = gradient_step(cur, peak_idx, eta)
        else:
            # multiplicative log‑normal perturbation
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            cand = cur * noise
            cand = proj_simplex(cand)

        cand_J = compute_J(cand, n, fft_len, conv_len)

        # ------------------------------------------------------------------
        # 4.2 Acceptance decision (Metropolis criterion)
        # ------------------------------------------------------------------
        accept = False
        if cand_J <= cur_J:
            accept = True
        else:
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur = cand
            cur_J = cand_J
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.3 Stagnation handling – occasional random restart.
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            cur = proj_simplex(rng.dirichlet(np.ones(n)))
            cur_J = compute_J(cur, n, fft_len, conv_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restarted random candidate J = {cur_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Final fine‑grained hill climbing (tiny log‑normal noise, accept only
    #    improvements).  A short window (≤15 s) is used.
    # ------------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(15.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fine_deadline:
        # Very small perturbation around the best vector
        noise = np.exp(rng.normal(0.0, 0.005, size=n))
        cand = best_vec * noise
        cand = proj_simplex(cand)
        cand_J = compute_J(cand, n, fft_len, conv_len)
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Normalise and return the final sequence.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
