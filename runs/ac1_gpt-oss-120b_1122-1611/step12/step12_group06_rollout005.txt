    cond = u - cssv / ind > 0

"""
Simulated‑annealing + sub‑gradient descent for the step‑function bound.

The algorithm works on the probability simplex (∑a_i = 1, a_i ≥ 0) because
the evaluation function is invariant under a uniform scaling of the vector.
It repeatedly takes a sub‑gradient of the current maximal convolution entry,
does a projected gradient step and accepts the move with a temperature‑controlled
Metropolis rule.  Random jitter and occasional restarts from a diverse pool of
initial vectors help escape local minima.  The best sequence found before the
time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns the first `conv_len` entries of the linear convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgrad_at_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of   max_k (a * a)_k   with respect to a.
    The active index is the (first) argmax of the convolution.
    """
    n = a.shape[0]
    k = int(conv.argmax())               # index of the maximal entry
    i_min = max(0, k - (n - 1))
    i_max = min(n - 1, k)
    i = np.arange(i_min, i_max + 1, dtype=np.int64)
    j = k - i
    grad = np.zeros_like(a)
    # derivative of Σ a_i a_{k-i} w.r.t. a_i is a_{k-i} (twice because of symmetry)
    grad[i] = a[j]
    grad *= 2.0
    return grad


def _compute_J(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Exact evaluation of the objective for a vector that already sums to 1.
    Returns J = 2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _conv_fft(a, fft_len, conv_len)
    return 2.0 * n * float(conv.max())

# Main search routine
# ----------------------------------------------------------------------

    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size -------------------------------------------------
    # --------------------------------------------------------------
    # Prefer the length of an existing construction if it exists,
    # otherwise fall back to a moderate size.
    if "height_sequence_1" in globals():

        n = int(seq0.size)
        if n > 8192:                     # cap to keep FFT cheap

    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # Build a diverse set of starter vectors (all sum to 1) ----------
    # --------------------------------------------------------------
    init_cands = []

    # uniform

    # geometric decays
