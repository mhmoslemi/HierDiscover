def random_shape(n, rng):
    shape_type = rng.integers(0, 4)  # 0 triangular,1 cosine,2 gaussian,3 exponential
    center = (n-1)/2
    if shape_type == 0:
        w_frac = rng.uniform(0.05, 0.5)
        w = int(w_frac * n)
        p = rng.uniform(0.5, 4.0)
        idx = np.arange(n)
        d = np.abs(idx - center)
        a = np.maximum(0.0, 1.0 - (d / w) ** p)
    elif shape_type == 1:
        w_frac = rng.uniform(0.05, 0.5)
        w = int(w_frac * n)
        p = rng.uniform(0.5, 4.0)
        idx = np.arange(n)
        d = np.abs(idx - center)
        mask = d <= w
        a = np.zeros_like(idx, dtype=np.float64)
        a[mask] = np.cos(np.pi * d[mask] / (2 * w)) ** p
    elif shape_type == 2:
        sigma = rng.uniform(0.05, 0.5) * n
        idx = np.arange(n)
        d = idx - center
        a = np.exp(- (d ** 2) / (2 * sigma ** 2))
    elif shape_type == 3:
        tau = rng.uniform(0.05, 0.5) * n
        idx = np.arange(n)
        d = np.abs(idx - center)
        a = np.exp(- d / tau)
    else:
        # fallback uniform
        a = np.full(n, 1.0 / n)
    # normalize
    s = a.sum()
    if s <= 0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / s
    return a
```

"""
propose_candidate
-----------------
Hybrid search for a low‑maximum auto‑correlation step function.

The routine consists of three phases that run within the 1000‑second
budget:

1. **Parametric sampling** – generate many symmetric shapes
   (triangular, cosine, Gaussian, exponential) with random widths,
   exponents and scales.  Each candidate is normalised to sum = 1 and
   its objective J = 2·n·max(conv(a,a)) is evaluated with a single FFT.
   The best shape is kept as a starting point.

2. **Random Dirichlet + jitter** – a fast Monte‑Carlo phase that
   draws Dirichlet‑distributed vectors and performs a cheap jitter
   around the current best solution.  This often discovers vectors
   that are not well described by the simple parametric families.

3. **Projected sub‑gradient descent** – starting from the best vector
   found so far, we minimise the exact non‑smooth objective
   `max(conv(a,a))`.  The active set of indices attaining the current
   maximum is identified and the exact sub‑gradient is computed via a
   single FFT (convolution of the active‑set mask with the reversed
   vector).  An Adam‑style update with a time‑dependent learning rate is
   applied and the iterate is projected onto the probability simplex.
   A Metropolis acceptance rule with a slowly‑cooling temperature and
   occasional random re‑balancing moves helps escape shallow plateaus.

All three phases run until the global time limit is reached; the best
vector encountered is returned as a plain ``list[float]`` ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas – top level only)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | sum(x)=1, x_i ≥ 0 }.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries were non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear (full) convolution a*a with a pre‑allocated FFT length."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of ``x`` and ``y`` using an FFT of length ``L``."""
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[: len(x) + len(y) - 1]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute the objective J = 2·n·max(conv(a,a)) for a vector already
    on the simplex (∑a = 1).  The convolution is performed via FFT for speed.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _active_set_grad(a: np.ndarray, conv: np.ndarray,
                     fft_len_grad: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)).
    ``conv`` is the full convolution a*a.
    Returns a vector of length len(a) (the gradient of max(conv)).
    """
    max_val = conv.max()
    # tolerance: treat values within 1e‑12·max_val as active
    tol = max(1e-12, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]
    w = np.zeros_like(conv)
    if active.size == 0:                     # should never happen
        return np.zeros_like(a)
    w[active] = 1.0 / active.size            # uniform weight on the active set
    # gradient = 2 * (w * a_rev)  (full conv, then slice central part)
    a_rev = a[::-1]
    grad_full = conv_fft(w, a_rev, fft_len_grad)   # length = 3n‑2
    n = a.shape[0]
    grad = 2.0 * grad_full[n - 1:n - 1 + n]        # centre slice
    return grad


def _sample_triangular(n: int, rng: np.random.Generator) -> np.ndarray:
    center = (n - 1) / 2.0
    w_frac = rng.uniform(0.05, 0.5)
    w = max(1, int(w_frac * n))
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    d = np.abs(idx - center)
    a = np.maximum(0.0, 1.0 - (d / w) ** p)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def _sample_cosine(n: int, rng: np.random.Generator) -> np.ndarray:
    center = (n - 1) / 2.0
    w_frac = rng.uniform(0.05, 0.5)
    w = max(1, int(w_frac * n))
    p = rng.uniform(0.5, 4.0)
    idx = np.arange(n, dtype=np.float64)
    d = np.abs(idx - center)
    mask = d <= w
    a = np.zeros_like(idx)
    a[mask] = np.cos(np.pi * d[mask] / (2.0 * w)) ** p
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def _sample_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    center = (n - 1) / 2.0
    sigma = rng.uniform(0.05, 0.5) * n
    idx = np.arange(n, dtype=np.float64)
    d = idx - center
    a = np.exp(-0.5 * (d / sigma) ** 2)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def _sample_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    center = (n - 1) / 2.0
    tau = rng.uniform(0.05, 0.5) * n
    idx = np.arange(n, dtype=np.float64)
    d = np.abs(idx - center)
    a = np.exp(-d / tau)
    a_sum = a.sum()
    return a / a_sum if a_sum > 0 else np.full(n, 1.0 / n)


def _sample_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    # Gamma(1,1) is exponential – yields a Dirichlet with α=1 (uniform)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    s = g.sum()
    return g / s if s > 0 else np.full(n, 1.0 / n)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The function respects the given time budget
    and returns the best sequence found (as a plain list of floats).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine problem size ------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # Clamp to a reasonable power‑of‑two for FFT efficiency
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # for a*a convolution
    fft_len_grad = _next_pow2(conv_len + n - 1)  # for gradient convolution

    # ------------------------------------------------------------------
    # 2. Initialise with a known good vector if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64).copy()
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
        a0_sum = a0.sum()
        if a0_sum > 0:
            a0 /= a0_sum
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = a0.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)
    cur_a = best_a.copy()
    cur_J = best_J

    print(f"[0.00s] Initial J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Time budgeting
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 0.1   # tiny safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 4. Phase 1 – parametric random sampling
    # ------------------------------------------------------------------
    phase1_end = start_time + 0.30 * total_time   # 30 % of the budget
    sampler_fns = [_sample_triangular, _sample_cosine,
                   _sample_gaussian, _sample_exponential]
    while time.monotonic() < phase1_end:
        # Choose a sampler uniformly at random
        fn = rng.choice(sampler_fns)
        a_cand = fn(n, rng)
        Jcand = compute_J(a_cand, n, fft_len, conv_len)
        if Jcand < best_J:
            best_J = Jcand
            best_a = a_cand.copy()
            cur_a = a_cand.copy()
            cur_J = Jcand
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Phase 1 new best J = {best_J:.6f}")

        # Occasionally also test a pure Dirichlet draw (quick, cheap)
        if rng.random() < 0.02:
            a_cand = _sample_dirichlet(n, rng)
            Jcand = compute_J(a_cand, n, fft_len, conv_len)
            if Jcand < best_J:
                best_J = Jcand
                best_a = a_cand.copy()
                cur_a = a_cand.copy()
                cur_J = Jcand
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Phase 1 (Dirichlet) best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Phase 2 – random jitter around current best
    # ------------------------------------------------------------------
    phase2_end = start_time + 0.55 * total_time   # another 25 %
    while time.monotonic() < phase2_end:
        # generate a jittered copy: multiplicative log‑normal noise
        noise = np.exp(rng.normal(0.0, 0.04, size=n))
        a_cand = proj_simplex(best_a * noise)
        Jcand = compute_J(a_cand, n, fft_len, conv_len)
        if Jcand < best_J:
            best_J = Jcand
            best_a = a_cand.copy()
            cur_a = a_cand.copy()
            cur_J = Jcand
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Phase 2 jitter improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Phase 3 – projected (sub‑)gradient descent with Adam & SA
    # ------------------------------------------------------------------
    phase3_end = deadline
    # Adam moments
    m = np.zeros_like(best_a)
    v = np.zeros_like(best_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-12
    # Learning‑rate schedule (high at start, low at the end)
    lr_start, lr_end = 0.25, 0.005
    # Temperature schedule for occasional uphill moves
    T_start = best_J
    T_end = 1e-8
    iter_cnt = 1
    last_improve = start_time

    while time.monotonic() < phase3_end:
        now = time.monotonic()
        elapsed = now - start_time
        # --------------------------------------------------------------
        # 6.1. Adaptive schedules
        # --------------------------------------------------------------
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        # --------------------------------------------------------------
        # 6.2. Compute convolution and exact sub‑gradient (active set)
        # --------------------------------------------------------------
        conv_cur = _conv_full(cur_a, fft_len, conv_len)
        max_conv = conv_cur.max()
        grad = _active_set_grad(cur_a, conv_cur, fft_len_grad)

        # --------------------------------------------------------------
        # 6.3. Adam update + simplex projection
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iter_cnt)
        v_hat = v / (1.0 - beta2 ** iter_cnt)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_prop = cur_a - step
        a_prop = proj_simplex(a_prop)

        # --------------------------------------------------------------
        # 6.4. Small random re‑balancing move (helps escape flat regions)
        # --------------------------------------------------------------
        if rng.random() < 0.07:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(a_prop[i], rng.random() * 0.006)   # ≤ 0.6 % of total mass
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = proj_simplex(a_prop)

        # --------------------------------------------------------------
        # 6.5. Evaluate proposal
        # --------------------------------------------------------------
        new_J = compute_J(a_prop, n, fft_len, conv_len)

        # --------------------------------------------------------------
        # 6.6. Metropolis acceptance
        # --------------------------------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = a_prop
            cur_J = new_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = cur_a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] Phase 3 new best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 6.7. Stagnation handling – inject a fresh jitter if no improvement
        # --------------------------------------------------------------
        if now - last_improve > 22.0:
            # re‑centre around the incumbent best with modest noise
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_a = proj_simplex(best_a * noise)
            cur_J = compute_J(cur_a, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Stagnation jitter – current J = {cur_J:.6f}")

        iter_cnt += 1

    # ------------------------------------------------------------------
    # 7. Final sanitisation (clip to allowed range, renormalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    print(f"[{time.monotonic() - start_time:6.2f}s] Finished – best J = {best_J:.6f}")
    return [float(v) for v in best_a]
