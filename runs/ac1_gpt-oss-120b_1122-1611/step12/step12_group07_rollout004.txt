"""
Propose a step‑function (non‑negative vector) that minimises
evaluate_sequence().  The method works directly on the probability simplex:
* a is kept on the simplex (∑a = 1, a_i ≥ 0) – the scale is irrelevant for the
  evaluation function.
* The objective we try to reduce is the variance of the self‑convolution
  (which is a smooth proxy for the true max‑norm).  In addition we add the
  exact sub‑gradient of the current maximal entry to drive the peak down.
* A projected Adam optimiser is used together with simple stagnation
  detection and Gaussian jitter to escape local plateaus.
* The length of the sequence is increased (up to 8192) – larger n gives a
  lower theoretical bound 2 n/(2 n−1)≈1, so we can improve over the previous
  best ≈1.51.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a * a (full length) using real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _grad_variance(a: np.ndarray, diff: np.ndarray,
                   grad_fft_len: int) -> np.ndarray:
    """
    Gradient of   L = Σ_k ( (a*a)_k - μ )²   w.r.t. a .
    Here  diff = (a*a) - μ, μ = mean of the convolution.
    The gradient equals 4 * (diff * rev(a))  (full convolution,
    then sliced to obtain the part corresponding to a).
    """
    rev_a = a[::-1]
    W = np.fft.rfft(diff, grad_fft_len)
    A_rev = np.fft.rfft(rev_a, grad_fft_len)
    full = np.fft.irfft(W * A_rev, n=grad_fft_len)
    n = a.shape[0]
    grad = 4.0 * full[n - 1:n - 1 + n]
    return grad


def _subgrad_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of  max(conv)  w.r.t. a.
    For the index k* where conv is maximal,
        (∂/∂a_i) conv_{k*} = 2 * a_{k*‑i}   (if the index is in range).
    """
    k_star = int(np.argmax(conv))
    n = a.shape[0]
    i = np.arange(n)
    j = k_star - i
    mask = (j >= 0) & (j < n)
    g = np.zeros_like(a)
    g[mask] = 2.0 * a[j[mask]]
    return g


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for vectors")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero; return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) with a small
    evaluation score.  The routine runs for at most `budget_s` seconds and
    returns the best sequence seen.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size
    # ------------------------------------------------------------------
    # Use a fairly large size (up to 8192) – larger n gives a lower bound.
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        n = max(256, min(prev.size, 8192))
    else:
        n = 8192
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)           # for a*a
    grad_fft_len = _next_pow2(3 * n - 2)     # for variance‑gradient

    # ------------------------------------------------------------------
    # 2. Build a portfolio of initial probability vectors
    # ------------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # random Dirichlet draws (different concentrations)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            init_candidates.append(rng.dirichlet(np.full(n, alpha)))

    # geometric decays
    for r in (0.97, 0.99, 0.995, 0.999):
        g = r ** np.arange(n, dtype=np.float64)
        g /= g.sum()
        init_candidates.append(g)

    # if a previous construction exists, insert it (cropped / padded)
    if "height_sequence_1" in globals():
        v = np.copy(prev)
        if v.size < n:
            v = np.concatenate([v, np.full(n - v.size, 0.0)])
        elif v.size > n:
            v = v[:n]
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_candidates.append(v)

    # ------------------------------------------------------------------
    # 3. Pick the best starter (using the true hard objective)
    # ------------------------------------------------------------------
    def hard_J(a_vec: np.ndarray) -> float:
        conv = _conv_full(a_vec, fft_len, conv_len)
        max_conv = conv.max()
        s = a_vec.sum()
        return float(2.0 * n * max_conv / (s * s))

    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        cur_J = hard_J(cand)
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Starter best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Optimisation loop (projected Adam on the simplex)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # Adam state (working on the simplex directly)
    m = np.zeros_like(best_a)
    v_adam = np.zeros_like(best_a)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.8               # a bit larger than the previous code

    iteration = 0
    last_improve = start_time
    global_best_a = best_a.copy()
    global_best_J = best_J

    # Parameters for the variance/peak trade‑off
    lambda_var = 0.15       # weight of the variance gradient
    jitter_sigma = 0.02     # magnitude of random restarts

    while time.monotonic() < deadline:
        # ----- 4.1 compute full convolution and the current hard J -----
        conv = _conv_full(best_a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv / (best_a.sum() ** 2))

        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = best_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----- 4.2 check stagnation → jitter restart -----
        if now - last_improve > 20.0:   # shorter stagnation window than before
            # add a small Gaussian perturbation and re‑project onto the simplex
            perturbed = best_a + rng.normal(0.0, jitter_sigma, size=n)
            best_a = _project_simplex(perturbed)
            m.fill(0.0)
            v_adam.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (jitter) – best J = {global_best_J:.6f}")

        # ----- 4.3 gradients -------------------------------------------------
        # Sub‑gradient of the maximal entry
        g_max = _subgrad_max(best_a, conv)

        # Gradient of the variance of the convolution (smooth proxy)
        ave = best_a.sum() ** 2 / conv_len           # the exact mean of conv
        diff = conv - ave
        g_var = _grad_variance(best_a, diff, grad_fft_len)

        # Combined descent direction (we move opposite to the gradient)
        grad = g_max + lambda_var * g_var

        # ----- 4.4 Adam update on the simplex (project after step) -----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)      # classic 1/√t decay

        m = beta1 * m + (1.0 - beta1) * grad
        v_adam = beta2 * v_adam + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_adam / (1.0 - beta2 ** iteration)

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        best_a = best_a - step
        # project back onto the probability simplex
        best_a = _project_simplex(best_a)

    # ------------------------------------------------------------------
    # 5. Return the best sequence (clipped to the limits required by evaluate_sequence)
    # ------------------------------------------------------------------
    final_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = final_vec.sum()
    if total == 0.0:
        final_vec = np.full_like(final_vec, 1.0 / n)
    else:
        final_vec /= total          # keep it a proper probability vector

    print(f"Finished. Best J = {global_best_J:.6f}")
    return [float(v) for v in final_vec]
