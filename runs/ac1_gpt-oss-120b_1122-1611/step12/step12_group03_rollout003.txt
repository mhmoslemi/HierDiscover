def adam_step(a, grad, m, v, t, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)
    step = lr * m_hat / (np.sqrt(v_hat) + eps)
    a_new = a - step
    a_new = proj_simplex(a_new)
    return a_new, m, v

    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1-D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:

    rho = rho[-1]

    {w ≥ 0, sum(w)=1}.  Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
