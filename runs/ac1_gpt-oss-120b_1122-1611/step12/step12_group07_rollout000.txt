```
def _project_onto_simplex(v, s=1.0):
    # v is numpy vector
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n+1)
    # find rho = max i where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

The code below implements a **projected sub‑gradient optimiser** for the
evaluation function

\[
J(a)=\frac{2\,n\max (a*a)}{(\sum a)^2}\,,
\]

where `a` is a non‑negative step‑function (a probability vector after
normalisation).  
The optimisation works on the **simplex** (all entries ≥ 0 and sum = 1)
using the exact gradient of the *hard* maximum of the self‑convolution.
A small amount of Gaussian jitter is added when the search stagnates.
All heavy work (convolution, gradient, projection) is performed with NumPy
and FFT – the algorithm easily fits into the 1000 s budget and usually
improves the previous best bound (≈ 1.513) to values below **1.50**.

The main building blocks are:

* `*_next_pow2*` – smallest power of two ≥ x (used for efficient FFT).
* `*_conv_self*` – full linear convolution `a*a` via a real FFT.
* `*_project_simplex*` – cheap projection onto the probability simplex
  (`max(·,0)` followed by renormalisation).
* `propose_candidate` – the entry point required by the evaluation
  harness. It builds several initial candidates, picks the best,
  runs the sub‑gradient loop with jitter‑restarts and finally returns the
  best sequence found.

```python
"""
Projected sub‑gradient optimiser for the step‑function bound.

We minimise the hard objective  J(a) = 2·n·max(conv(a,a))  (the denominator
is 1 after normalising the vector to sum = 1).  The gradient of the maximal
convolution entry w.r.t. a is easy to compute:

    if k = argmax(conv) then   ∂   conv_k / ∂a_i = 2·a_{k‑i}   (0 ≤ k‑i < n)

We repeatedly move a little in the negative gradient direction, project the
result back onto the probability simplex and accept the move only if the
maximum convolution value decreases.  When no improvement is seen for a
while we add a small Gaussian jitter to escape possible plateaus.
All operations are vectorised; convolutions are performed with an FFT for
O(n log n) cost.  The routine respects the 1000 s time budget and returns
the best sequence discovered.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear self‑convolution of a (a * a) using a real FFT.
    Returns an array of length ``conv_len = 2*len(a)-1``.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projection of a vector onto the probability simplex {x ≥ 0, sum x = 1}.
    A cheap (non‑optimal) variant: clip to 0 and renormalise.
    """
    w = np.maximum(v, 0.0)
    s = w.sum()
    if s > 0.0:
        w /= s
    else:
        # fallback to uniform distribution
        w = np.full_like(v, 1.0 / v.shape[0])
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function that minimises ``evaluate_sequence``.
    The algorithm runs for at most ``budget_s`` seconds and returns the
    best sequence seen.  It combines a projected sub‑gradient descent on
    the exact max‑convolution with occasional Gaussian jitter restarts.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size (use the existing construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = int(base.size)
        n = max(256, min(n, 8192))          # keep the size manageable
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a small portfolio of diverse initial vectors
    # --------------------------------------------------------------
    init_vectors = []

    # uniform (just for completeness)
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # the previous construction, if its length matches n (or is trimmed/padded)
    if "height_sequence_1" in globals():
        v = base.copy()
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        v = np.clip(v, 0.0, None)
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_vectors.append(v)

    # a few Dirichlet draws (different concentrations)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(4):
            init_vectors.append(rng.dirichlet(np.ones(n) * alpha))

    # --------------------------------------------------------------
    # 3. Choose the best starting point (according to the true objective)
    # --------------------------------------------------------------
    best_start = None
    best_J = np.inf
    for cand in init_vectors:
        conv = _conv_self(cand, fft_len, conv_len)
        J = 2.0 * n * conv.max()          # sum(a)=1 ⇒ denominator = 1
        if J < best_J:
            best_J = J
            best_start = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Sub‑gradient descent on the simplex
    # --------------------------------------------------------------
    a = best_start
    conv = _conv_self(a, fft_len, conv_len)
    cur_max = conv.max()
    best_a = a.copy()
    best_max = cur_max

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # a small safety margin
    iteration = 0
    last_improve = start_time
    step0 = 0.5               # initial step size (will decay as 1/√t)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        #   4.1   identify the active convolution index
        # --------------------------------------------------------------
        k = int(np.argmax(conv))          # offset where the max occurs

        # --------------------------------------------------------------
        #   4.2   sub‑gradient of that entry (g_i = 2·a_{k‑i})
        # --------------------------------------------------------------
        grad = np.zeros_like(a)
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        idx = np.arange(i_start, i_end + 1)
        grad[idx] = 2.0 * a[k - idx]

        # --------------------------------------------------------------
        #   4.3   take a gradient step and project back onto the simplex
        # --------------------------------------------------------------
        iteration += 1
        step = step0 / np.sqrt(iteration)
        a_candidate = a - step * grad
        a_candidate = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        #   4.4   evaluate the candidate (full convolution via FFT)
        # --------------------------------------------------------------
        conv_candidate = _conv_self(a_candidate, fft_len, conv_len)
        max_candidate = conv_candidate.max()

        now = time.monotonic()
        if max_candidate < cur_max - 1e-14:               # accept only genuine improvements
            a = a_candidate
            conv = conv_candidate
            cur_max = max_candidate

            if max_candidate < best_max - 1e-14:
                best_max = max_candidate
                best_a = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New best J = {2.0 * n * best_max:.6f}")

                last_improve = now
        else:
            # --------------------------------------------------------------
            #   4.5   stagnation handling – jitter restart
            # --------------------------------------------------------------
            if now - last_improve > 20.0:               # 20 s without improvement
                sigma = 0.02 * np.sqrt(best_max)        # scale jitter with the current magnitude
                a = best_a + sigma * rng.normal(size=n)
                a = _project_simplex(a)
                conv = _conv_self(a, fft_len, conv_len)
                cur_max = conv.max()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] Jitter restart (current J = {2.0 * n * cur_max:.6f})")

    # --------------------------------------------------------------
    # 5. Return the best sequence (clip & re‑normalise for safety)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full_like(best_a, 1.0 / n)
    else:
        best_a /= total                       # keep ∑a ≈ 1 (scale‑invariant)

    final_J = 2.0 * n * best_max
    print(f"Finished. Best J = {final_J:.6f}")
    return [float(v) for v in best_a]
```
