"""
Gradient‑descent on a smooth approximation of the ∞‑norm of the
autocorrelation (a*a).  The algorithm works on the probability simplex
by multiplicative (exponentiated‑gradient) updates, projects back onto
the simplex each step, and gradually sharpens the smoothing parameter
β so that the objective approaches the true max‑convolution.
A simple restart‑and‑jitter scheme is added to escape possible local
minima.  The routine runs for the allotted time (≤ 1000 s) and returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):          # all entries are equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Small utility
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a*a using real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(A * A, n=fft_len)
    return c[:conv_len]


def max_conv_value(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return max(a*a) needed for the exact evaluation."""
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Gradient‑descent optimiser on the probability simplex.
    Returns a list of non‑negative heights (already normalised to sum‑one).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    # Use a length similar to the best known construction, but bound it.
    if "height_sequence_1" in globals():
        n_prev = len(globals()["height_sequence_1"])
        n = max(2048, min(n_prev, 8192))
    else:
        n = 4096                      # comfortable default

    conv_len = 2 * n - 1
    # FFT length needs to accommodate a (len n) and w (len 2n‑1) for the
    # correlation used in the gradient.  3 n – 2 is sufficient; we round up.
    fft_len = _next_pow2(3 * n)

    # --------------------------------------------------------------
    # 2. Initial candidate (simplex‑projected)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.clip(np.asarray(globals()["height_sequence_1"], dtype=np.float64), 0.0, None)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.concatenate([a0, np.zeros(n - a0.size, dtype=np.float64)])
        a0 = proj_simplex(a0)
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a = a0.copy()

    # --------------------------------------------------------------
    # 3. Optimiser parameters
    # --------------------------------------------------------------
    beta = 5.0                     # initial smoothing parameter
    beta_max = 200.0               # final smoothing (close to true max)
    beta_mul = 1.015               # multiplicative increase per iteration

    lr0 = 0.08                     # base learning‑rate for multiplicative update
    lr_min = 0.001                 # lower bound to keep progress
    # we will use a simple 1/√t decay
    iter_cnt = 0

    # bookkeeping
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s margin for clean‑up
    best_a = a.copy()
    best_J = 2.0 * n * max_conv_value(best_a, n, fft_len, conv_len)
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    last_improve = start_time
    restart_interval = 120.0                     # seconds
    next_restart = start_time + restart_interval

    while time.monotonic() < deadline:
        # ----- FFT of current a -------------------------------------------------
        A = np.fft.rfft(a, fft_len)               # shape (fft_len//2+1,)

        # ----- linear convolution a*a (needed for soft‑max) ----------------------
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]

        # ----- smooth max (log‑sum‑exp) -----------------------------------------
        m = conv.max()
        # stable soft‑max
        exp_term = np.exp(beta * (conv - m))
        w = exp_term / exp_term.sum()             # length conv_len, non‑negative, sums to 1

        # ----- prefix sums for the S‑term in the gradient -----------------------
        # S_i = sum_{k=i}^{i+n-1} w_k   (window of length n)
        csum = np.concatenate(([0.0], np.cumsum(w)))
        S = csum[n:] - csum[:n]                  # length n

        # ----- correlation a ⋆ w (gives conv1_i = Σ_j a_j w_{i+j}) -------------
        W = np.fft.rfft(w, fft_len)
        corr = np.fft.irfft(np.conj(A) * W, n=fft_len)   # linear correlation, length fft_len
        conv1 = corr[:n]                               # first n entries are Σ_j a_j w_{i+j}

        # ----- gradient of the smooth max ---------------------------------------
        grad = conv1 + a * S                            # shape (n,)

        # ----- step size (decays like 1/√t) ------------------------------------
        lr = max(lr0 / np.sqrt(iter_cnt + 1), lr_min)

        # ----- multiplicative (exponentiated‑gradient) update -------------------
        a_new = a * np.exp(-lr * grad)

        # ----- back to the simplex ---------------------------------------------
        a_new = proj_simplex(a_new)

        # ----- exact objective for the new point --------------------------------
        J_new = 2.0 * n * max_conv_value(a_new, n, fft_len, conv_len)

        # ----- accept & bookkeeping ------------------------------------------------
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_a = a_new.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        a = a_new
        iter_cnt += 1

        # ----- slowly increase β (sharpen the approximation) --------------------
        if beta < beta_max:
            beta = min(beta_max, beta * beta_mul)

        # ----- periodic restart to escape stagnation ----------------------------
        now = time.monotonic()
        if now > next_restart:
            # re‑initialise from a random Dirichlet draw (slightly perturbed)
            alpha = rng.uniform(0.2, 2.0)
            a = proj_simplex(rng.dirichlet(np.ones(n) * alpha))
            next_restart = now + restart_interval
            print(f"[{now - start_time:6.2f}s] Restarted with random Dirichlet")

    # --------------------------------------------------------------
    # 5. Fine‑grained random jitter around the best solution found
    # --------------------------------------------------------------
    fine_deadline = time.monotonic() + min(4.0, (deadline - start_time) * 0.05)
    while time.monotonic() < fine_deadline:
        # multiplicative log‑normal noise (σ small)
        noise = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = proj_simplex(best_a * noise)
        J_cand = 2.0 * n * max_conv_value(cand, n, fft_len, conv_len)
        if J_cand < best_J:
            best_J = J_cand
            best_a = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Final normalisation and return
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
