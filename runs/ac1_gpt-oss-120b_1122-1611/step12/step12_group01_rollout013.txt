"""
Simulated‑Annealing + Gradient‑Guided local moves.

* Starts from the best known construction if ``height_sequence_1`` is present,
  otherwise falls back to a uniform distribution.
* The objective to minimise is
       J(a) = 2·n·max(conv(a,a))          (scale‑invariant, a on the simplex)
* A simple annealing loop repeatedly proposes two kinds of moves:
    – a directed “gradient” step that moves mass away from the index that
      contributes most to the current maximal convolution,
    – a random pairwise mass‑transfer move.
  Moves are accepted if they improve the objective, otherwise with a
  temperature‑dependent probability.
* The temperature decays linearly with the remaining budget; the maximum
  transferred fraction also shrinks, allowing coarse exploration early and
  fine tuning later.
* A short stochastic jitter is performed at the end to polish the best
  solution found.
* All vectors are kept on the probability simplex by simple renormalisation.
"""

import time
import numpy as np


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            while j_dst == i_src:
                j_dst = rng.integers(n)
            delta = cur_vec[i_src] * rng.uniform(0.0, move_frac)
            if delta > 0.0:
                cand_vec = cur_vec.copy()
                cand_vec[i_src] -= delta
                cand_vec[j_dst] += delta

        # ensure feasibility (non‑negative, sum to 1)
        cand_vec = np.maximum(cand_vec, 0.0)
        s = cand_vec.sum()
        if s == 0.0:
            continue
        cand_vec /= s

        # ------------------------------------------------------------------
        # 4.2 Evaluate the candidate
        # ------------------------------------------------------------------
        cand_conv = full_conv(cand_vec)
        cand_J = obj_from_conv(cand_conv)

        # ------------------------------------------------------------------
        # 4.3 Acceptance test
        # ------------------------------------------------------------------
        accept = False
        if cand_J < cur_J - 1e-12:
            accept = True
        else:
            # uphill move with Metropolis probability
            if T > 0.0:
                prob = np.exp(-(cand_J - cur_J) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur_vec = cand_vec
            cur_J = cand_J
            cur_conv = cand_conv
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand_vec.copy()
                last_improve = now
                print(f"[{now - start_time:6.2f}s] new best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.4 Simple stagnation handling – occasional global jitter
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            sigma = 0.1
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            jitter = cur_vec * noise
            jitter = np.maximum(jitter, 0.0)
            jitter /= jitter.sum()
            j_conv = full_conv(jitter)
            j_J = obj_from_conv(j_conv)
            if j_J < cur_J:
                cur_vec, cur_J, cur_conv = jitter, j_J, j_conv
                print(f"[{now - start_time:6.2f}s] jitter reset J = {cur_J:.6f}")
            last_improve = now

        it += 1

    # ----------------------------------------------------------------------
    # 5. Final fine‑grained jitter (very small multiplicative noise)
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline - 0.2:
        remaining = deadline - time.monotonic()
        sigma = max(0.001, 0.05 * remaining / total_time)
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_vec * noise
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        cand_J = obj_from_conv(full_conv(cand))
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] final jitter improved J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # 6. Return the best sequence (clipped and normalised for safety)
    # ----------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    print(f"finished – best J = {best_J:.6f}, length = {n}")
    return [float(v) for v in best_vec]
