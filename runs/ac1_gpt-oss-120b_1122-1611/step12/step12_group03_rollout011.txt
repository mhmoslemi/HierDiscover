    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            break

        # Adam‑style update (element‑wise step size)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** sub_iter)
        v_hat = v / (1 - beta2 ** sub_iter)
        step = 0.1 / (np.sqrt(v_hat) + eps)
        a = a - step * m_hat
        a = proj_simplex(a)

        # occasional jitter to escape plateaus
        if sub_iter % 200 == 0:
            noise = np.exp(rng.normal(0.0, 0.01, size=n))
            a = proj_simplex(a * noise)

    # --------------------------------------------------------------
    # 3) Final random‑noise sweep (log‑normal perturbations) for the remaining time.
    # --------------------------------------------------------------
    fine_deadline = time.monotonic() + min(30.0, max(0.0, deadline - time.monotonic()))
    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        J_cand, _, _ = evaluate_vec(cand)
        if J_cand < best_J - 1e-12:
            best_J = J_cand
            best_vec = cand
            elapsed = time.monotonic() - start
            print(f"[Fine {elapsed:.2f}s] improved J={best_J:.6f}")

    # --------------------------------------------------------------
    # Normalise final vector (protect against zero‑sum) and return.
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s <= 1e-12:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec /= s

    print(f"[{time.monotonic() - start:.2f}s] finished – best J={best_J:.6f}")
    return best_vec.tolist()
