import time
import numpy as np

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def proj_simplex(v):
    # Euclidean projection onto probability simplex
    # from Condat (2016)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _conv_full(a, fft_len, conv_len):
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

def _weighted_gradient(a, weight, fft_len_corr, n):
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    grad = 2.0 * full[n-1:n-1+n]
    return grad

def _objective_maxconv(a, n, fft_len, conv_len):
    conv = _conv_full(a, fft_len, conv_len)
    return float(conv.max())

    # returns J = 2 * n * max(conv)
    conv = _conv_full(a, fft_len, conv_len)

    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – fall back to the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / (rho + 0.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray,
                       weight: np.ndarray,
                       fft_len_corr: int,
                       n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    weight length = 2·n‑1, a length = n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # the valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective_J(a: np.ndarray,
                 n: int,
                 fft_len: int,
                 conv_len: int) -> float:
    """
    Return the exact objective used by `evaluate_sequence`,
    i.e.  J = 2·n·max(conv(a,a)).
    The vector *a* is assumed to already sum to 1.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Random‑restart projected‑Adam optimisation of a smooth approximation
    of ``max(conv)`` (soft‑max with an annealing temperature) on the
    probability simplex, followed by a simulated‑annealing polishing
    phase.  The routine works for up to ~8000 coefficients and stays
    within the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlation

    start_time = time.monotonic()
    deadline = start_time + float(budget_s)

    # ------------------------------------------------------------------
    # Reserve a chunk of time for the final polishing phase
    # ------------------------------------------------------------------
    POLISH_TIME = 120.0                       # seconds
    search_deadline = deadline - POLISH_TIME
    if search_deadline < start_time + 5.0:     # fall back if the budget is tiny
        search_deadline = deadline - 10.0

    # ------------------------------------------------------------------
    # 2. Build a pool of diverse initial vectors (all are normalised)
    # ------------------------------------------------------------------
    init_pool = []

    # uniform
    init_pool.append(np.full(n, 1.0 / n, dtype=np.float64))

    # linear ramps
    init_pool.append(np.arange(1, n + 1, dtype=np.float64) /
                     np.arange(1, n + 1).sum())
    init_pool.append(np.arange(n, 0, -1, dtype=np.float64) /
                     np.arange(n, 0, -1).sum())

    # geometric decays
    for r in (0.97, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_pool.append(v)

    # sinusoidal bumps (shifted upwards to stay non‑negative)
    x = np.arange(n, dtype=np.float64)
    for period in (n / 3.0, n / 5.0, n / 7.0):
        v = np.sin(2.0 * np.pi * x / period) + 1.0
        v = np.maximum(v, 0.0)
        v /= v.sum()
        init_pool.append(v)

    # Dirichlet draws with a random concentration
    for _ in range(8):
        alpha = 10 ** rng.uniform(-2.0, 1.0)   # between 0.01 and 10
        init_pool.append(rng.dirichlet(np.full(n, alpha)))

    # previous construction, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_pool.append(v)

    # ------------------------------------------------------------------
    # 3. Global bookkeeping
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    def maybe_update(vec: np.ndarray):
        """Update the global best if the candidate improves the objective."""
        nonlocal best_vec, best_J
        J = _objective_J(vec, n, fft_len, conv_len)
        if J < best_J - 1e-12:
            best_J = J
            best_vec = vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")

    # ------------------------------------------------------------
    # Quick evaluation of the static pool – gives us a decent
    # starting point for the first Adam run.
    # ------------------------------------------------------------
    for cand in init_pool:
        maybe_update(cand)

    # ------------------------------------------------------------
    # 4. Random‑restart Adam optimisation (soft‑max smoothing)
    # ------------------------------------------------------------
    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr0 = 0.5                     # base learning rate
    beta_start = 1.0
    beta_end = 3000.0            # final soft‑max temperature

    restart_id = 0
    while time.monotonic() < search_deadline:
        restart_id += 1

        # ---- pick a random seed from the pool and add a tiny jitter ----
        base = init_pool[rng.integers(len(init_pool))]
        jitter = np.exp(rng.normal(0.0, 0.025, size=n))
        cur_vec = proj_simplex(base * jitter)

        # ---- Adam state ----
        m = np.zeros_like(cur_vec)
        v_hat = np.zeros_like(cur_vec)

        # ---- limits for this run (time‑aware) ----
        max_iters = 1500
        it = 0
        last_improve = time.monotonic()

        while it < max_iters and time.monotonic() < search_deadline:
            it += 1
            now = time.monotonic()
            elapsed = now - start_time
            t_frac = (now - start_time) / (search_deadline - start_time)

            # ----- convolution and current objective -----
            conv = _conv_full(cur_vec, fft_len, conv_len)
            max_conv = conv.max()
            cur_J = 2.0 * n * max_conv

            # ----- update global best -----
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] Restart {restart_id} iter {it} "
                      f"new best J = {best_J:.6f}")

            # ----- soft‑max weights (temperature schedule‑controlled) -----
            beta = beta_start * (beta_end / beta_start) ** t_frac
            shifted = conv - max_conv          # stabilise numerical range
            w_exp = np.exp(beta * shifted)
            w_sum = w_exp.sum()
            if w_sum == 0.0:
                weight = np.full_like(w_exp, 1.0 / w_exp.size)
            else:
                weight = w_exp / w_sum

            # ----- gradient of the smoothed max objective -----
            grad = _weighted_gradient(cur_vec, weight, fft_len_corr, n)

            # ----- Adam update (projected back onto simplex) -----
            m = beta1 * m + (1.0 - beta1) * grad
            v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)

            m_hat = m / (1.0 - beta1 ** it)
            v_corr = v_hat / (1.0 - beta2 ** it)

            lr = lr0 * (1.0 - t_frac) + 1e-3
            step = - lr * m_hat / (np.sqrt(v_corr) + eps_adam)

            proposal = cur_vec + step
            cur_vec = proj_simplex(proposal)

            # ----- escape from stagnation (optional jitter) -----
            if now - last_improve > 30.0:
                noise = np.exp(rng.normal(0.0, 0.06, size=n))
                cur_vec = proj_simplex(best_vec * noise)
                last_improve = now
                m.fill(0.0)
                v_hat.fill(0.0)

    # ------------------------------------------------------------------
    # 5. Polishing phase – simulated‑annealing pairwise mass swaps
    # ------------------------------------------------------------------
    polish_deadline = deadline
    cur_vec = best_vec.copy()
    cur_J = _objective_J(cur_vec, n, fft_len, conv_len)

    # temperature schedule for polishing
    T0 = cur_J * 0.02
    T_end = 1e-9

    while time.monotonic() < polish_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - frac) + T_end * frac

        # randomly select two distinct indices
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a small fraction of mass from the richer index to the poorer one
        donor, receiver = (i, j) if cur_vec[i] > cur_vec[j] else (j, i)
        if cur_vec[donor] <= 0.0:
            continue
        eps = rng.random() * cur_vec[donor] * 0.5   # at most half of donor mass
        if eps <= 0.0:
            continue

        cand = cur_vec.copy()
        cand[donor] -= eps
        cand[receiver] += eps
        cand = proj_simplex(cand)

        cand_J = _objective_J(cand, n, fft_len, conv_len)

        if cand_J < cur_J:
            cur_vec = cand
            cur_J = cand_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{elapsed:6.2f}s] Polishing improvement: J = {best_J:.6f}")
        else:
            # Metropolis‑style uphill move
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur_vec = cand
                cur_J = cand_J

    # ------------------------------------------------------------------
    # 6. Final normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total <= 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
