def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n and initial vector
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        n = 4096
        a = np.full(n, 1.0/n)
    # Clip, renorm
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0:
        a = np.full_like(a, 1.0 / a.size)
    else:
        a = a / total

    n = a.size

    # Precompute FFT length
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    def compute_J(vec):
        conv = _conv_full(vec, fft_len, conv_len)
        return 2.0 * n * conv.max()

    best_vec = a.copy()
    best_J = compute_J(a)
    print(f"Initial J = {best_J}")

    # Simulated annealing parameters
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Temperature schedule
    T0 = 1e-2 # maybe high
    T_end = 1e-6
    # Steps schedule based on time fraction
    # We'll compute timeline fraction f = (t - start)/(deadline-start)
    # then T = T0 * (T_end/T0) ** f

    # Step size schedule - maximum delta to move per iteration
    max_delta_initial = 0.05 # up to 5% of mass moved
    # Decrease to 0.001.

    iter_count = 0
    while time.monotonic() < deadline:
        # elapsed fraction
        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)
        # temperature
        T = T0 * (T_end / T0) ** frac
        # step size
        max_delta = max_delta_initial * (1 - frac) + 0.001 * frac

        # random move
        i, j = rng.choice(n, size=2, replace=False)
        # ensure we move from i (donor) with some amount up to min(a[i], max_delta)
        donor = i
        recipient = j
        # choose amount
        delta = min(a[donor], max_delta) * rng.random()
        # propose move
        a_new = a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta

        # compute J
        J_new = compute_J(a_new)
        # accept criteria
        if J_new < best_J:
            best_J = J_new
            best_vec = a_new.copy()
            a = a_new
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
            continue
        # else accept with SA prob
        if T > 0:
            prob = np.exp(-(J_new - best_J) / T)
            if rng.random() < prob:
                a = a_new
        # else reject

        iter_count += 1
        # optionally every 1000 iter do reproject to simplex (should be sum=1)
        if iter_count % 1000 == 0:
            # re-normalize due to rounding
            s = a.sum()
            if s != 0:
                a = a / s

    # finally return best_vec normalized/clipped
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]

"""
Simulated‑Annealing + Gradient‑Guided mass transfer.

* Start from the best known construction (``height_sequence_1``) if it is
  present, otherwise fall back to a uniform distribution.
* The vector lives on the probability simplex (∑a_i = 1, a_i ≥ 0).
* At each iteration we
  – compute the full convolution a*a (FFT based, O(n log n))
  – build a binary mask of the (near‑)maximal convolution entries
  – obtain the sub‑gradient of the max‑convolution via a correlation
    (the same weighted‑gradient routine used by the reference solver)
  – move a small amount of mass from the index with the largest gradient
    (i.e. contributing most to the peak) to the index with the smallest
    gradient.
* Moves are accepted if they lower the objective; otherwise they are
  accepted with a temperature‑dependent probability (simulated annealing)
  to escape plateaus.
* The step size and temperature decay linearly with elapsed time.
* If the search stagnates for >30 s we jitter a fraction of the entries
  to regain diversity.
* After the annealing phase we perform a few aggressive “polish”
  multiplicative updates (the same as the baseline code) to fine‑tune
  the final vector.

The routine respects the 1000 s wall‑clock budget, prints occasional
progress messages and finally returns the best sequence found (as a
Python list of floats) ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] with respect to a.
    ``weight`` is a binary mask (length 2·n‑1) that marks the active
    convolution indices (those attaining the current maximum).
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine combines a gradient‑guided simulated
    annealing phase with a final multiplicative polishing step.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size and initial vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        init = np.clip(init, 0.0, None)
        if init.size == 0:
            n = 4096
            cur = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = init.size
            cur = init.copy()
    else:
        n = 4096
        cur = np.full(n, 1.0 / n, dtype=np.float64)

    # Normalise onto the simplex (∑a_i = 1)
    s = cur.sum()
    if s == 0.0:
        cur[:] = 1.0 / n
    else:
        cur /= s

    # --------------------------------------------------------------
    # 2. FFT bookkeeping (same for the whole run)
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for the weighted gradient

    # --------------------------------------------------------------
    # 3. Helper that evaluates the objective J = 2·n·max(conv)
    # --------------------------------------------------------------
    def _objective(vec: np.ndarray) -> float:
        conv = _conv_full(vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 4. Initial assessment
    # --------------------------------------------------------------
    cur_J = _objective(cur)
    best_vec = cur.copy()
    best_J = cur_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # keep a safety margin
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Simulated annealing + gradient‑guided mass transfer
    # --------------------------------------------------------------
    # Annealing schedule parameters
    T_start = 2e-2          # initial temperature
    T_end = 5e-7            # final temperature
    step_start = 0.06       # initial maximal fraction of donor mass to move
    step_end = 0.001        # final maximal fraction

    last_improve_time = start_time
    stagnation_counter = 0

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)   # 0 → 1

        # Linear temperature and step‑size decay
        T = T_start * (T_end / T_start) ** frac
        max_step_frac = step_start * (1.0 - frac) + step_end * frac

        # ---------------------------------------------------------
        # Compute current convolution, max and the active mask
        # ---------------------------------------------------------
        conv = _conv_full(cur, fft_len, conv_len)
        cur_max = conv.max()
        # mask of indices that are (almost) maximal (tolerance tiny)
        eps_mask = max(cur_max * 1e-12, 1e-15)
        active = (conv >= cur_max - eps_mask).astype(np.float64)

        # ---------------------------------------------------------
        # Gradient of the max‑convolution (sub‑gradient)
        # ---------------------------------------------------------
        grad = _weighted_gradient(cur, active, fft_len_corr, n)

        # Choose donor (largest gradient) and recipient (smallest)
        donor = int(np.argmax(grad))
        recipient = int(np.argmin(grad))
        if donor == recipient:
            # fallback to a random pair if the gradient is flat
            donor, recipient = rng.choice(n, size=2, replace=False)

        # Amount to transfer – a fraction of the donor’s current mass
        delta = max_step_frac * cur[donor]
        # Ensure we do not make the donor negative
        delta = min(delta, cur[donor] * 0.999999)

        # Propose the new vector
        cand = cur.copy()
        cand[donor] -= delta
        cand[recipient] += delta

        # ---------------------------------------------------------
        # Evaluate candidate
        # ---------------------------------------------------------
        cand_J = _objective(cand)

        # Acceptance rule (Metropolis)
        accept = False
        if cand_J < cur_J:
            accept = True
        else:
            # probability = exp(-(ΔJ)/T)
            if T > 0.0:
                prob = np.exp(-(cand_J - cur_J) / T)
                if rng.random() < prob:
                    accept = True

        if accept:
            cur = cand
            cur_J = cand_J

            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                last_improve_time = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---------------------------------------------------------
        # Stagnation handling – occasional jitter
        # ---------------------------------------------------------
        if now - last_improve_time > 30.0:
            # Randomly jitter a small fraction of the components
            n_jitter = max(1, n // 200)          # ~0.5 % of the entries
            idxs = rng.choice(n, size=n_jitter, replace=False)
            # replace those entries by fresh Dirichlet draws
            jitter_vals = rng.dirichlet(np.ones(n_jitter))
            cur[idxs] = jitter_vals
            # re‑project onto the simplex
            cur = proj_simplex(cur)
            cur_J = _objective(cur)
            last_improve_time = now
            stagnation_counter += 1
            print(f"[{elapsed:6.2f}s] Stagnation #{stagnation_counter} – jitter applied")

        iteration += 1

    # ------------------------------------------------------------------
    # 6. Final polishing – a few aggressive multiplicative updates
    # ------------------------------------------------------------------
    # Perform 6 polish steps (similar to the baseline solver but with a
    # slightly larger step multiplier).  This often squeezes a few
    # extra 1e‑5 improvements.
    for polish_iter in range(6):
        conv = _conv_full(best_vec, fft_len, conv_len)
        max_conv = conv.max()
        active = (conv >= max_conv - 1e-12).astype(np.float64)
        grad = _weighted_gradient(best_vec, active, fft_len_corr, n)
        step_mul = 0.85
        factor = np.exp(-step_mul * grad)
        polished = best_vec * factor
        polished = proj_simplex(polished)
        polished_J = _objective(polished)
        if polished_J < best_J - 1e-12:
            best_J = polished_J
            best_vec = polished
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Polished best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 7. Clean‑up: clip and renormalise for the outer evaluator
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # (Optional) sanity check – can be commented out if not desired.
    # try:
    #     val = evaluate_sequence(best_vec.tolist())
    #     print(f"Final evaluate_sequence = {val:.6f}")
    # except Exception:
    #     pass

    return [float(v) for v in best_vec]
