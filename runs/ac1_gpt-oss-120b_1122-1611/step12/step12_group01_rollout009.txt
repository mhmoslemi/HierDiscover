"""
Propose a step‑function using a projected multiplicative‑gradient descent.
The objective  J(a)=2·n·max(conv(a,a))  is scale‑invariant, so the search works on
the probability simplex (∑a_i=1, a_i≥0).  At each iteration we

1. evaluate the full self‑convolution via FFT,
2. form a soft‑max weighting of the convolution entries (β controls the
   smoothness – we anneal β from 0.5 up to ≈200),
3. compute the gradient of the weighted sum w.r.t. a efficiently as a
   convolution of the reversed a with the weight vector,
4. update a multiplicatively (a←a·exp(−η·grad)) and renormalise.
The step size η and the jitter amplitude are annealed with the elapsed time.
Whenever a new best J is seen it is reported.  The algorithm restarts
periodically (≈2 min) from the current best or from a few hand‑crafted seeds
(including the previously found sequence if available) to avoid local traps.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Helper functions – all top‑level, no closures or lambdas
# -------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def resize_vector(v: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate a vector to a new length and re‑normalise."""
    old_n = v.shape[0]
    if old_n == new_n:
        return v.copy()
    # interpolate on [0,1]
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    v_new = np.interp(x_new, x_old, v)
    v_new = np.maximum(v_new, 0.0)
    s = v_new.sum()
    if s == 0.0:
        v_new[:] = 1.0 / new_n
    else:
        v_new /= s
    return v_new


def self_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear self‑convolution of a using FFT."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def gradient_of_weighted_conv(a: np.ndarray,
                              w: np.ndarray,
                              grad_fft_len: int) -> np.ndarray:
    """
    Compute ∇_a ( Σ_k w_k·conv_k(a,a) ) .
    The formula gives 2·(rev(a) * w) restricted to n entries.
    """
    rev_a = a[::-1]
    rev_fft = np.fft.rfft(rev_a, n=grad_fft_len)
    w_fft = np.fft.rfft(w, n=grad_fft_len)
    conv_rev = np.fft.irfft(rev_fft * w_fft, n=grad_fft_len)
    start = a.shape[0] - 1                     # index of the first relevant element
    grad = 2.0 * conv_rev[start:start + a.shape[0]]
    return grad

# Main search routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and uses a projected multiplicative‑gradient descent with simulated‑annealing
    of the smoothing parameter β and of the step size η.
    """
    rng = np.random.default_rng(seed)

    # ---------------------------------------------------------
    # 1. Problem size
    # ---------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
    else:
        n = 4096                     # a comfortable power of two
    n = max(256, min(n, 8192))       # respect the allowed limits

    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)                # for self‑convolution
    grad_fft_len = next_pow2(3 * n - 2)           # for gradient convolution

    # ---------------------------------------------------------
    # 2. Prepare a few diverse seed vectors
    # ---------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (more mass at the beginning)
    for r in (0.99, 0.995, 0.997):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        seeds.append(v)

    # Gaussian bumps centred in the middle
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 10.0, n / 8.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        seeds.append(v)

    # Dirichlet draws with a range of concentration parameters
    for alpha in (0.05, 0.2, 0.6, 1.0):
        seeds.append(rng.dirichlet(np.full(n, alpha)))

    # include the previously discovered construction if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s == 0.0:
            prev[:] = 1.0 / prev.size
        else:
            prev /= s
        if prev.shape[0] != n:
            prev = resize_vector(prev, n)
        seeds.append(prev)

    # ---------------------------------------------------------
    # 3. Search control parameters
    # ---------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # 0.5 s safety margin
    total_budget = deadline - start_time

    # annealing schedules (linear in elapsed fraction)
    eta_start, eta_end = 0.20, 0.001          # step size
    beta_start, beta_max = 0.5, 200.0        # smoothing of the max
    sigma_start, sigma_end = 0.02, 0.0005   # jitter amplitude

    restart_interval = 120.0                 # seconds per independent run
    best_overall_vec = None
    best_overall_J = np.inf

    # ---------------------------------------------------------
    # 4. Main loop – repeated independent runs
    # ---------------------------------------------------------
    while time.monotonic() < deadline:
        run_start = time.monotonic()
        remaining = deadline - run_start
        run_len = min(restart_interval, remaining)
        run_end = run_start + run_len

        # ---- initialise a for this run ---------------------------------
        if best_overall_vec is not None and rng.random() < 0.6:
            a = best_overall_vec.copy()
            # slight random perturbation to escape a possible local basin
            a *= np.exp(rng.normal(0.0, 0.02, size=n))
        else:
            a = rng.choice(seeds)          # pick one of the prepared seeds

        a = np.maximum(a, 0.0)
        s = a.sum()
        a /= s if s != 0.0 else np.full_like(a, 1.0 / n)

        # ---- iterative optimisation ------------------------------------
        while time.monotonic() < run_end:
            # ----- schedule parameters based on total elapsed time -----
            elapsed = time.monotonic() - start_time
            frac = elapsed / total_budget               # in [0,1]
            eta = eta_start * (1.0 - frac) + eta_end * frac
            beta = beta_start * ((beta_max / beta_start) ** frac)
            sigma = sigma_start * (1.0 - frac) + sigma_end * frac

            # ----- self‑convolution and objective --------------------
            conv = self_convolution(a, fft_len, conv_len)
            max_b = conv.max()
            J = 2.0 * n * max_b

            # ----- keep the best seen so far -------------------------
            if J < best_overall_J - 1e-12:
                best_overall_J = J
                best_overall_vec = a.copy()
                print(f"[{elapsed:6.2f}s] New overall best J = {best_overall_J:.6f}")

            # ----- soft‑max weights for a smooth approximation of max --
            # stability trick: subtract the (current) maximum before exponentiating
            exp_vals = np.exp(beta * (conv - max_b))
            w = exp_vals / exp_vals.sum()

            # ----- gradient of the weighted sum ----------------------
            grad = gradient_of_weighted_conv(a, w, grad_fft_len)

            # ----- multiplicative update (projected onto simplex) ---
            # Clip the gradient magnitude to avoid overflow in exp()
            gmax = np.max(np.abs(grad))
            if gmax > 1e3:
                grad = grad * (1e3 / gmax)

            a = a * np.exp(-eta * grad)

            # ----- tiny stochastic jitter (annealed) -----------------
            if sigma > 0.0:
                jitter = np.exp(rng.normal(0.0, sigma, size=n))
                a *= jitter

            # ----- projection onto the probability simplex ------------
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a[:] = 1.0 / n
                s = 1.0
            a /= s

        # end of this run – loop back for another restart (if time permits)

    # ---------------------------------------------------------
    # 5. Return the best vector discovered
    # ---------------------------------------------------------
    if best_overall_vec is None:
        best_overall_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_overall_vec = np.clip(best_overall_vec, 0.0, 1000.0)
    total = best_overall_vec.sum()
    if total == 0.0:
        best_overall_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_overall_vec = best_overall_vec / total

    return [float(v) for v in best_overall_vec]
