"""
propose_candidate
-----------------

Search routine for the step‑function minimisation problem.

The algorithm proceeds in three phases:

1. *Broad shape sweep* – deterministic families (uniform, ramps,
   power‑laws, Gaussians, cosine bumps, double‑Gaussian mixtures) and a few
   random Dirichlet draws are evaluated.  The best vector from this cheap
   sweep is kept as a warm‑start.

2. *Random Dirichlet refinement* – additional Dirichlet samples are drawn
   while a small portion of the time budget remains; any improvement updates
   the current best.

3. *Exact‑gradient Adam* – the objective
   J(a)=2·n·max(conv(a,a)) is scale‑invariant, so we keep a on the simplex via
   a=softmax(z).  The sub‑gradient of the exact max (instead of a soft‑max
   approximation) is computed analytically:
        ∂J/∂a = 4·n·corr(a,w) ,
   where w is the uniform weight over all convolution indices attaining the
   current maximum and ``corr`` denotes the (non‑reversed) correlation.
   The gradient is transferred to the log‑it space, and Adam updates are
   applied.  Stagnation triggers a jitter restart to escape local minima.

All heavy operations (convolutions, correlations) are performed with
real‑FFT for O(n log n) cost.  The routine respects the 1000 s time budget,
prints progress, and finally returns the best sequence (already normalised)
as a plain ``list[float]`` ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _grad_exact_max(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of J = 2·n·max(conv(a,a)) w.r.t. a.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Current probability vector (sum = 1, non‑negative).
    w : np.ndarray, shape (2·n‑1,)
        Weight vector that is uniform over all indices attaining the current
        maximum of the convolution (zero elsewhere).

    Returns
    -------
    grad : np.ndarray, shape (n,)
        Gradient of J w.r.t. a.
    """
    n = a.size
    # Full linear correlation length = n + len(w) - 1 = 3·n‑2
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    A = np.fft.rfft(a, fft_len)
    W = np.fft.rfft(w, fft_len)
    # correlation a ⋆ w (non‑reversed)
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len)[:corr_len]

    start = 2 * n - 2                     # = len(w)‑1
    grad_slice = corr_full[start:start + n]
    # factor 4·n comes from J = 2·n·max and ∂conv/∂a = 2·shift
    return 4.0 * n * grad_slice


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length `fft_len`."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

        iteration += 1
        lr = lr0 / np.sqrt(iteration)                 # simple 1/√t decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_m = beta2 * v_m + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_m / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # keep logits in a numerically safe range
        logits = np.clip(logits, -500.0, 500.0)

    # --------------------------------------------------------------
    # 7. Return the best sequence (already normalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    final_J = float(2.0 * n * np.convolve(best_vec, best_vec).max())
    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:.2f}s] Finished – final best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
