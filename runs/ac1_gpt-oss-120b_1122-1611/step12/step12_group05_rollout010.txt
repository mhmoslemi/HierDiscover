def run_optimize_length(n, budget_s, rng, init_seq=None):
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    L_grad = _next_pow2(conv_len + n - 1)
    # Initialize a
    if init_seq is not None:
        # Upsample init_seq to length n
        a0 = upsample_sequence(init_seq, n)
    else:
        a0 = np.full(n, 1.0/n)
    # Ensure sum to 1
    a0 = np.clip(a0, 0.0, None)
    s = a0.sum()
    if s == 0:
        a0[:] = 1.0/n
    else:
        a0 /= s

    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(a, n, fft_len, conv_len)
    cur_J = best_J
    print(f"Start n={n} init J={best_J:.6f}")

    # Adam params
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smoothing schedule
    alpha_start = 30.0
    alpha_end = 500.0

    # Simulated annealing temperature parameters
    T_start = best_J
    T_end = 1e-8

    start = time.monotonic()
    deadline_local = start + budget_s
    last_improve = start
    iteration = 1

    while time.monotonic() < deadline_local:
        now = time.monotonic()
        elapsed = now - start
        total_time = budget_s

        # update temps
        T = T_start * (T_end / T_start) ** (elapsed / total_time)
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total_time) ** 1.5  # accelerate

        # jitter if needed
        if now - last_improve > 30.0:
            noise = rng.lognormal(mean=0.0, sigma=0.08, size=n)
            a = proj_simplex(best_a * noise)
            cur_J = compute_J(a, n, fft_len, conv_len)
            last_improve = now
            print(f"[{now-start:.2f}s] Jitter: J={cur_J:.6f}")

        # compute conv and gradient
        conv_a = _conv_full(a, fft_len, conv_len)
        max_conv = conv_a.max()
        shift = conv_a - max_conv
        w_raw = np.exp(alpha * shift)
        w_sum = w_raw.sum()
        if w_sum == 0.0:
            # fallback active set: pick indices near max
            tol = max(1e-12, max_conv * 1e-12)
            active = np.where(conv_a >= max_conv - tol)[0]
            grad_smooth = np.zeros_like(a)
            for p in active:
                start_i = max(0, p - (n - 1))
                end_i = min(p, n - 1)
                # idx = start_i...end_i inclusive
                idx = np.arange(start_i, end_i + 1)
                grad_smooth[idx] += 2.0 * a[p - idx]
        else:
            w = w_raw / w_sum
            a_rev = a[::-1]
            conv2 = conv_fft(w, a_rev, L_grad)
            grad_smooth = 2.0 * conv2[n-1:n-1+n]

        grad_J = 2.0 * n * grad_smooth

        # Adam updates
        m = beta1 * m + (1 - beta1) * grad_J
        v = beta2 * v + (1 - beta2) * (grad_J ** 2)

        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_proposed = a - step
        a_proposed = proj_simplex(a_proposed)

        # occasional large move (swap)
        if rng.random() < 0.1:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(a_proposed[i], rng.random() * 0.01)
            a_proposed[i] -= delta
            a_proposed[j] += delta
            a_proposed = proj_simplex(a_proposed)

        # evaluate new
        new_J = compute_J(a_proposed, n, fft_len, conv_len)

        # accept via Metropolis
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            if rng.random() < np.exp(-(new_J - cur_J) / max(T, 1e-12)):
                accept = True

        if accept:
            a = a_proposed
            cur_J = new_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = a.copy()
                last_improve = now
                print(f"[{now-start:.2f}s] New best J={best_J:.6f}")

        iteration += 1

    return best_a, best_J

"""
Hybrid projected Adam + simulated annealing with multi‑scale restarts.

* Works with any initial global sequence ``height_sequence_1`` (if present);
  otherwise starts from a uniform distribution.
* Tries three problem sizes (original length, doubled, 8192) – longer
  sequences give a tighter theoretical lower bound (≈1).
* For each size:
  – a short random‑Dirichlet warm‑start search picks a promising start;
  – a smooth‑max Adam optimiser (soft‑max of the convolution) runs;
    the smoothing parameter α grows from 30 to 500, the learning rate
    decays quadratically and a simple SA acceptance rule allows occasional
    uphill steps.
  – after 30 s without improvement a log‑normal jitter is applied to the
    current best point to escape plateaus.
* The best vector over all sizes is projected onto the probability simplex,
  clipped to the allowed range and returned.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | sum(x)=1, x_i ≥ 0}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal or zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear (full) convolution of ``a`` with itself using an FFT of length
    ``fft_len``.  Returns the first ``conv_len`` entries (the linear part).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """
    Full linear convolution of ``x`` and ``y`` using an FFT of length ``L``.
    The result is truncated to ``len(x)+len(y)-1`` entries.
    """
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[: len(x) + len(y) - 1]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective  J = 2·n·max(conv(a,a))  for a already on the simplex."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def upsample_sequence(seq: np.ndarray, target_n: int) -> np.ndarray:
    """
    Linear interpolation of a 1‑D sequence to a new length ``target_n``.
    The resulting vector is non‑negative and will later be normalised.
    """
    src_n = seq.shape[0]
    if src_n == target_n:
        return seq.copy()
    # Parameterise both source and target on [0,1)
    x_src = np.linspace(0.0, 1.0, src_n, endpoint=False)
    x_tgt = np.linspace(0.0, 1.0, target_n, endpoint=False)
    return np.interp(x_tgt, x_src, seq)


def _random_simplex(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point uniformly from the probability simplex."""
    # exponential sampling followed by normalisation
    x = rng.exponential(scale=1.0, size=n)
    s = x.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n)
    return x / s


# ----------------------------------------------------------------------
# Core optimiser for a single problem size
# ----------------------------------------------------------------------


def _optimise_length(
    n: int,
    budget_s: float,
    rng: np.random.Generator,
    init_seq: np.ndarray | None,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` for at most ``budget_s`` seconds.
    Returns the best sequence (on the simplex) together with its J‑value.
    """
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # for a*a
    L_grad = _next_pow2(conv_len + n - 1)        # for smooth‑max gradient

    # ----- initialisation -------------------------------------------------
    if init_seq is not None:
        a = upsample_sequence(init_seq, n)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
    else:
        a /= s

    # perform a quick random‑restart warm‑start (few Dirichlet samples)
    best_a = a.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)
    for _ in range(30):
        cand = _random_simplex(rng, n)
        Jc = compute_J(cand, n, fft_len, conv_len)
        if Jc < best_J:
            best_J, best_a = Jc, cand.copy()

    a = best_a.copy()
    cur_J = best_J

    # ----- optimisation hyper‑parameters ---------------------------------
    start_time = time.monotonic()
    deadline = start_time + budget_s

    # Adam
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # smooth‑max schedule
    alpha_start, alpha_end = 30.0, 500.0

    # simulated‑annealing temperature schedule
    T_start = best_J
    T_end = 1e-8

    last_improve = start_time
    iteration = 1

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total = budget_s

        # ---- temperature / smoothing ------------------------------------
        T = T_start * (T_end / T_start) ** (elapsed / total)
        # accelerate α growth (quadratic in time)
        alpha = alpha_start + (alpha_end - alpha_start) * (elapsed / total) ** 2

        # ---- jitter if stuck --------------------------------------------
        if now - last_improve > 30.0:
            # log‑normal jitter centred on the current best point
            noise = rng.lognormal(mean=0.0, sigma=0.08, size=n)
            a = proj_simplex(best_a * noise)
            cur_J = compute_J(a, n, fft_len, conv_len)
            last_improve = now
            print(f"[{now-start_time:6.2f}s] Jitter → J = {cur_J:.6f}")

        # ---- convolution and smooth‑max gradient -------------------------
        conv_a = _conv_full(a, fft_len, conv_len)
        max_conv = conv_a.max()
        shift = conv_a - max_conv               # ≤ 0
        w_raw = np.exp(alpha * shift)
        w_sum = w_raw.sum()

        if w_sum == 0.0:
            # fall back to sub‑gradient on active set (very rare)
            tol = max(1e-12, max_conv * 1e-12)
            active = np.where(conv_a >= max_conv - tol)[0]
            grad_smooth = np.zeros_like(a)
            for p in active:
                lo = max(0, p - (n - 1))
                hi = min(p, n - 1)
                idx = np.arange(lo, hi + 1)
                grad_smooth[idx] += 2.0 * a[p - idx]
        else:
            w = w_raw / w_sum
            a_rev = a[::-1]
            conv2 = conv_fft(w, a_rev, L_grad)
            grad_smooth = 2.0 * conv2[n - 1:n - 1 + n]

        grad_J = 2.0 * n * grad_smooth

        # ---- Adam update & projection ------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_J
        v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # decaying step size (quadratic decay)
        lr = lr0 * (1.0 - elapsed / total) ** 2
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_proposed = a - step
        a_proposed = proj_simplex(a_proposed)

        # ---- occasional large re‑balancing move -------------------------
        if rng.random() < 0.09:                 # ~9 % chance
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(a_proposed[i], rng.random() * 0.01)  # ≤ 1 % of mass
            a_proposed[i] -= delta
            a_proposed[j] += delta
            a_proposed = proj_simplex(a_proposed)

        # ---- evaluate ----------------------------------------------------
        new_J = compute_J(a_proposed, n, fft_len, conv_len)

        # ---- Metropolis acceptance ---------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_proposed
            cur_J = new_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = a.copy()
                last_improve = now
                print(f"[{now-start_time:6.2f}s] new best J = {best_J:.6f}")

        iteration += 1

    return best_a, best_J

def propose_candidate(
    seed: int = 4242,
    budget_s: float = 1000.0,
    **unused_kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem sizes.
    # ------------------------------------------------------------------
    known = globals().get("height_sequence_1")
    if isinstance(known, (list, tuple, np.ndarray)) and len(known) > 0:
        base_n = int(len(known))
        base_seq = np.asarray(known, dtype=np.float64)
    else:
        base_n = 2048
        base_seq = None

    # choose a few increasing lengths – never exceed 8192 (to keep FFT cheap)
    lengths = [base_n]
    while lengths[-1] < 8192:
        nxt = min(8192, lengths[-1] * 2)
        if nxt != lengths[-1]:
            lengths.append(nxt)
        else:
            break

    # ------------------------------------------------------------------
    # Global best across all lengths.
    # ------------------------------------------------------------------
    best_global_seq: np.ndarray | None = None
    best_global_J = np.inf
    start = time.monotonic()
    # allocate time proportionally (simple equal split with a small reserve)
    remaining_lengths = len(lengths)
    for idx, n in enumerate(lengths):
        now = time.monotonic()
        time_left = (start + budget_s) - now - 5.0  # safety margin
        if time_left < 5.0:
            break
        # split the remaining time equally among the unfinished sizes
        allocated = time_left / (remaining_lengths - idx)

        print(f"[{now-start:6.2f}s] Optimising length {n} (≈{allocated:.1f}s)")

        seq, J = _optimise_length(
            n=n,
            budget_s=allocated,
            rng=rng,
            init_seq=best_global_seq if best_global_seq is not None else base_seq,
        )
        print(f"[{time.monotonic()-start:6.2f}s] Finished n={n} → J={J:.6f}")

        if J < best_global_J:
            best_global_J, best_global_seq = J, seq.copy()
            print(f"[{time.monotonic()-start:6.2f}s] *** New overall best J={best_global_J:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation: clip, renormalise and convert to plain list.
    # ------------------------------------------------------------------
    if best_global_seq is None:
        # fallback – uniform vector of length 2048
        best_global_seq = np.full(2048, 1.0 / 2048, dtype=np.float64)

    best_global_seq = np.clip(best_global_seq, 0.0, 1000.0)
    total = best_global_seq.sum()
    if total == 0.0:
        best_global_seq[:] = 1.0 / best_global_seq.size
    else:
        best_global_seq /= total

    return [float(v) for v in best_global_seq]
