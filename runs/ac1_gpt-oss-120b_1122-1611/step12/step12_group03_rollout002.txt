"""
Propose a step‐function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – start from the previously found best
   ``height_sequence_1`` (if present) or from a random Dirichlet draw.
   The vector is projected onto the probability simplex (sum = 1) and
   symmetrised (a_i = a_{n-1-i}) because the convolution objective is
   invariant under reversal.

2. **Iterative local improvement** – repeat while time permits:
      * Compute the full convolution by FFT.
      * Locate the index `k_max` where the convolution attains its
        maximum.
      * The value at `k_max` is the sum of products a_i·a_{k_max‑i}.
        Choose a source index `i` proportionally to the contribution
        a_i·a_{k_max‑i} and a random destination `j`.
      * Transfer a small amount of mass from `i` to `j`,
        re‑symmetrise and re‑project onto the simplex.
      * Accept the move if the objective (2·n·max(conv)) improves;
        otherwise keep the current vector (very occasional uphill
        acceptance for diversification).

   *If the search stagnates for >30 s, a fresh random simplex point is
   inserted.*

3. **Fine‑grained random refinement** – after the main loop a short
   phase of multiplicative log‑normal perturbations (around the best
   solution) is performed.

4. **Return** – the best vector found (clipped to the allowed range
   [0, 1000]).  Because the objective is scale‑invariant, the vector is
   kept on the simplex; ``evaluate_sequence`` will compute the correct
   score.

The routine respects the 1000 s budget, prints progress and finally
returns a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def compute_objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute 2·n·max(conv(a, a)) for a vector `a` whose entries sum to 1.
    This equals the value returned by ``evaluate_sequence`` because the
    scaling is irrelevant (the sum is 1 ⇒ denominator = 1).
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry v_i = v_{n-1-i}."""
    return (v + v[::-1]) * 0.5


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The method combines a simplex
    projection, a deterministic pair‑wise mass‑transfer hill‑climber and
    a short stochastic refinement phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine problem size and initialise a candidate.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            start_vec = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = start_vec.size
        except Exception:
            n = 2048
            start_vec = None
    else:
        n = 2048
        start_vec = None

    # Clip to the allowed numeric range before any processing.
    if start_vec is None:
        a = rng.dirichlet(np.ones(n))
    else:
        a = np.clip(start_vec, 0.0, 1000.0)
        a = proj_simplex(a)

    # Force symmetry and re‑project (symmetry often reduces the max‑conv).
    a = symmetrize(a)
    a = proj_simplex(a)

    # Pre‑compute FFT parameters.
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Track the best solution.
    # ------------------------------------------------------------------
    best_vec = a.copy()
    best_J = compute_objective(best_vec, n, fft_len, conv_len)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5 s margin
    last_improve = start_time

    print(f"[0.00s] Initial J = {best_J:.6f} (n = {n})")

    # ------------------------------------------------------------------
    # 3. Main hill‑climbing loop.
    # ------------------------------------------------------------------
    iteration = 0
    step_factor = 0.05          # proportion of mass moved each try

    while time.monotonic() < deadline:
        # Full convolution – needed for the current max location.
        conv = conv_full_fft(a, fft_len, conv_len)
        max_val = conv.max()
        k_max = int(conv.argmax())

        # Determine the admissible index range for i where i + j = k_max.
        i_min = max(0, k_max - (n - 1))
        i_max = min(n - 1, k_max)
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        j_range = k_max - i_range

        # Contributions a_i * a_j to the maximal entry.
        contrib = a[i_range] * a[j_range]
        total_contrib = contrib.sum()

        if total_contrib > 0:
            # Pick a source index proportional to its contribution.
            probs = contrib / total_contrib
            src = rng.choice(i_range, p=probs)
        else:
            # Fallback – pick a random source.
            src = rng.integers(n)

        # Destination index – a random (different) coordinate.
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # Amount of mass to move; never exceed the source's mass.
        delta = step_factor * a[src]
        if delta <= 0.0:
            delta = step_factor * 1e-6
        delta = min(delta, a[src])

        cand = a.copy()
        cand[src] -= delta
        cand[dst] += delta

        # Symmetrise and enforce the simplex constraints.
        cand = symmetrize(cand)
        cand = proj_simplex(cand)

        cand_J = compute_objective(cand, n, fft_len, conv_len)

        # Accept only improving moves (occasionally accept a worsening move).
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand.copy()
            a = cand                     # continue from the new point
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Improved J = {best_J:.6f}")
            # slowly reduce step size after a success (more fine‑grained moves)
            step_factor = max(step_factor * 0.97, 0.001)
        else:
            # Very occasional uphill step to avoid getting stuck.
            if rng.random() < 0.001:
                a = cand

        iteration += 1

        # Periodic global random perturbation (log‑normal noise) – helps escape flat regions.
        if iteration % 300 == 0:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            pert = proj_simplex(a * noise)
            pert = symmetrize(pert)
            pert = proj_simplex(pert)
            pert_J = compute_objective(pert, n, fft_len, conv_len)
            if pert_J < best_J:
                best_J = pert_J
                best_vec = pert.copy()
                a = pert
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Log‑normal perturb improved J = {best_J:.6f}")

        # Stagnation handling – re‑initialise if no improvement for 30 s.
        if time.monotonic() - last_improve > 30.0:
            a = proj_simplex(rng.dirichlet(np.ones(n)))
            a = symmetrize(a)
            a = proj_simplex(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Stagnation restart (re‑initialised).")

    # ------------------------------------------------------------------
    # 4. Final short stochastic refinement around the best solution.
    # ------------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.1)

    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.01, size=n))
        cand = proj_simplex(best_vec * noise)
        cand = symmetrize(cand)
        cand = proj_simplex(cand)
        cand_J = compute_objective(cand, n, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the final sequence (clipped to the allowed range).
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(v) for v in best_vec]
