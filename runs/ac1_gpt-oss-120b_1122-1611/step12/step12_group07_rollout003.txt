def propose_candidate(seed=12345, budget_s=1000.0):
    import time, numpy as np
    rng = np.random.default_rng(seed)

    # Choose n
    # Try larger than previous: set baseline 8192; ensure not too large maybe 16384.
    # Compute based on available memory: 2* n = conv length. FFT length double.

    # If height_sequence_1 exists:
    #   base_n = len(height_sequence_1)
    # else base_n = 4096
    # We'll set n = max(base_n, 8192) but also bounded to 16384.

    # We can define n = max(2048, min(16384, int(base_n * 2))) maybe.

    # Actually simpler: set n = 8192.

    n = 8192

    # Maybe if existing height_sequence_1 longer than 8192, adjust.

    # if height_sequence_1 present: we can interpolate to n.

    # Build initial a. Start with uniform:
    a = np.full(n, 1.0/n, dtype=np.float64)

    # Add small random Dirichlet mixture: a = rng.dirichlet(np.full(n, 0.5))

    # Or combine: a = (uniform + Dirichlet) / 2.

    # Compute conv length and FFT lengths.

    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())  # next power of 2

    grad_fft_len = 1 << ((3 * n - 2 - 1).bit_length())  # for gradient

    # functions for conv, smooth weights, gradient.

    # Initialize logits = log(a) for softmax param.

    logits = np.log(np.maximum(a, 1e-16))

    m1 = np.zeros_like(logits)
    v1 = np.zeros_like(logits)

    # hyperparameters
    beta = 2.0
    beta_max = 2000.0
    beta_inc = 1.0015

    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 1.0

    iter = 0
    best_J = np.inf
    best_a = None

    start = time.monotonic()
    deadline = start + budget_s - 0.5

    last_improve_time = start

    # Main loop
    while time.monotonic() < deadline:
        # compute softmax probabilities:
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()

        # compute conv (FFT)
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]

        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv # sum a = 1

        # update best
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve_time = time.monotonic()
            elapsed = last_improve_time - start
            print(f"[{elapsed:.2f}s] New best J = {best_J:.6f}")

        # stagnation detection
        now = time.monotonic()
        if now - last_improve_time > 30.0:
            # jitter
            jitter_sigma = 0.2
            logits = np.log(np.maximum(best_a, 1e-16))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m1.fill(0.0)
            v1.fill(0.0)
            last_improve_time = now
            print(f"[{now-start:.2f}s] Restart jitter applied. Best J = {best_J:.6f}")

        # compute smooth max weights
        # use current beta
        m = conv.max()
        shifted = beta * (conv - m)
        # Clip for stability
        shifted = np.clip(shifted, -700, 0)
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()

        # compute gradient w.r.t a via FFT:
        rev_a = a[::-1]
        W = np.fft.rfft(w, grad_fft_len)
        A_rev = np.fft.rfft(rev_a, grad_fft_len)
        full = np.fft.irfft(W * A_rev, n=grad_fft_len)
        grad_a = 2.0 * full[n-1:n-1+n]

        # Gradient w.r.t logits
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)

        # Adam step
        iter += 1
        lr = lr0 / np.sqrt(iter)

        m1 = beta1 * m1 + (1 - beta1) * grad_z
        v1 = beta2 * v1 + (1 - beta2) * (grad_z**2)

        m1_hat = m1 / (1 - beta1**iter)
        v1_hat = v1 / (1 - beta2**iter)

        logits -= lr * m1_hat / (np.sqrt(v1_hat) + adam_eps)

        # Clip logits to something
        logits = np.clip(logits, -500.0, 500.0)

        # increase beta
        beta = min(beta_max, beta * beta_inc)

    # final best sequence:
    final_seq = best_a
    # Ensure non-negative and clip cap.
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    total = final_seq.sum()
    if total == 0:
        final_seq = np.full_like(final_seq, 1.0/len(final_seq))
    else:
        final_seq /= total  # optional normalization

    # return as list
    print(f"Finished. Best J = {best_J:.6f}")
    return [float(v) for v in final_seq]

import time
import numpy as np

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises the
    evaluation function `evaluate_sequence`.

    The optimiser works on the probability simplex by parametrising the
    sequence with a soft‑max (logits).  A smooth approximation of the
    maximum of the self‑convolution (log‑sum‑exp) is used; its gradient is
    obtained efficiently via FFTs.  Adam updates are applied to the logits,
    the temperature (β) is increased slowly, and a Gaussian jitter restart
    is triggered after 30 s of stagnation.  The routine runs for at most
    `budget_s` seconds and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a length larger than the previous construction.
    # ------------------------------------------------------------------
    # If a previous construction exists, make the new length at least twice
    # (but not larger than 8192 for speed).  Otherwise start from 4096.
    if "height_sequence_1" in globals():
        try:
            old_len = len(height_sequence_1)
        except Exception:
            old_len = 4096
    else:
        old_len = 4096

    # Target length: a power of two for fast FFTs
    target_len = max(2048, old_len * 2)
    target_len = 1 << ((target_len - 1).bit_length())      # next power of two
    n = min(8192, target_len)                             # cap at 8192

    # ------------------------------------------------------------------
    # 2. Initialise a probability vector.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        # interpolate the previous best to the new size
        old = np.asarray(height_sequence_1, dtype=np.float64)
        old = np.clip(old, 0.0, None)
        s = old.sum()
        if s == 0.0:
            old = np.full_like(old, 1.0 / old.size)
        else:
            old = old / s
        old_idx = np.linspace(0, old.size - 1, old.size)
        new_idx = np.linspace(0, old.size - 1, n)
        a0 = np.interp(new_idx, old_idx, old)
        a0 = np.clip(a0, 1e-12, None)
        a0 /= a0.sum()
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3. Set up optimisation on the simplex (logits + Adam).
    # ------------------------------------------------------------------
    logits = np.log(np.maximum(a0, 1e-16))
    m1 = np.zeros_like(logits)          # first moment
    v1 = np.zeros_like(logits)          # second moment

    # FFT lengths
    conv_len = 2 * n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())          # >= conv_len
    grad_fft_len = 1 << ((3 * n - 2 - 1).bit_length())    # >= 3n‑2

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.7

    # Smooth‑max temperature schedule
    beta = 2.0
    beta_max = 2000.0
    beta_inc = 1.0012

    iteration = 0
    best_J = np.inf
    best_a = None

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin
    last_improve = start_time

    # ------------------------------------------------------------------
    # 4. Main optimisation loop.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ---- 4.1 probabilities from logits (soft‑max) ----
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)          # shift for stability
        a = exp_logits / exp_logits.sum()                # lies on the simplex

        # ---- 4.2 current objective (exact, because sum(a)=1) ----
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv                        # because sum(a)=1

        # ---- 4.3 keep the best sequence ----
        now = time.monotonic()
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{now-start_time:6.2f}s] New best J = {best_J:.6f}")

        # ---- 4.4 stagnation detection – jitter restart ----
        if now - last_improve > 30.0:
            jitter_sigma = 0.20
            logits = np.log(np.maximum(best_a, 1e-16))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m1.fill(0.0)
            v1.fill(0.0)
            last_improve = now
            print(f"[{now-start_time:6.2f}s] Restart – jitter applied (best J = {best_J:.6f})")

        # ---- 4.5 smooth‑max weights (log‑sum‑exp) ----
        shifted = beta * (conv - max_conv)
        shifted = np.clip(shifted, -700, 0)               # avoid overflow
        exp_shifted = np.exp(shifted)
        w = exp_shifted / exp_shifted.sum()               # sums to 1

        # ---- 4.6 gradient w.r.t. a (via FFT) ----
        rev_a = a[::-1]
        W = np.fft.rfft(w, grad_fft_len)
        A_rev = np.fft.rfft(rev_a, grad_fft_len)
        full = np.fft.irfft(W * A_rev, n=grad_fft_len)
        grad_a = 2.0 * full[n - 1:n - 1 + n]

        # ---- 4.7 chain rule to logits (soft‑max Jacobian) ----
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)

        # ---- 4.8 Adam step on logits ----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)

        m1 = beta1 * m1 + (1.0 - beta1) * grad_z

        m1_hat = m1 / (1.0 - beta1 ** iteration)

        logits -= lr * m1_hat / (np.sqrt(v1_hat) + adam_eps)

        # ---- 4.9 increase the smoothing parameter ----
        beta = min(beta_max, beta * beta_inc)

    # ------------------------------------------------------------------
    # 5. Return the best sequence (clipped, renormalised).
    # ------------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_a, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq = np.full_like(best_seq, 1.0 / n)
    else:
        best_seq = best_seq / total                # normalise (optional)

    print(f"Finished. Best J = {best_J:.6f}")
    return [float(v) for v in best_seq]
