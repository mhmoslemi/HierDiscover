"""
Sub‑gradient descent with Adam on the true max‑convolution.
---------------------------------------------------------

* Works directly on the exact max of the auto‑convolution (no soft‑max
  approximation).  For the current max index `t*` the sub‑gradient of
  `max(conv(a,a))` is simply `g[i] = 2·a[t*‑i]` for those indices that
  contribute to that entry.
* Adam‐style moments (β₁=0.9, β₂=0.999) are used to stabilise the noisy
  updates.
* The iterate is always projected onto the probability simplex
  (∑aᵢ = 1, aᵢ ≥ 0) with the Euclidean projection `proj_simplex`.
* A simple simulated‑annealing acceptance rule permits occasional uphill
  moves; the temperature decays from the current best value to 1e‑8.
* If no improvement is seen for a while, a multiplicative jitter centred
  on the best solution is applied to escape plateaus.
* The routine respects the 1000 s budget, prints occasional progress
  messages and finally returns the best vector found (clipped to the
  allowed range and re‑normalised).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | sum(x)=1, x_i ≥ 0 }.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length ``fft_len``."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Objective value J = 2·n·max(conv(a,a)) for a vector already on the
    probability simplex (∑a = 1). Convolution performed via FFT for speed.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgrad_max_conv(a: np.ndarray, t_star: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at index `t_star`.
    For each i with 0 ≤ t_star‑i < n we have g[i] = 2·a[t_star‑i];
    otherwise g[i] = 0.
    """
    n = a.size
    g = np.zeros_like(a)
    # indices that actually contribute
    i_low = max(0, t_star - (n - 1))
    i_high = min(n - 1, t_star)
    idx = np.arange(i_low, i_high + 1, dtype=int)
    g[idx] = 2.0 * a[t_star - idx]
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The routine respects the 1000 s
    time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size – reuse a known good length if possible.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(glendas := globals()["height_sequence_1"]))
            # Keep it reasonable for FFT speed (max ~8192)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # for a*a convolution

    # --------------------------------------------------------------
    # 2. Initialise the search vector on the simplex.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
        s = a0.sum()
        if s == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= s
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # add a tiny random perturbation to break symmetry
    a0 = a0 * np.exp(rng.normal(0.0, 0.02, size=n))
    a0 = proj_simplex(a0)

    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)
    cur_J = best_J

    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Optimisation hyper‑parameters
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # small safety margin
    total_time = deadline - start_time

    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.12                     # initial learning rate (pre‑scale)
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature (simulated annealing) schedule
    T_start = best_J
    T_end = 1e-8

    # step‑size schedule for sub‑gradient
    eta_start = 1.0
    eta_end = 0.001

    last_improve = start_time
    iteration = 1

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # 3.1 Schedule updates
        # ----------------------------------------------------------
        # temperature decays geometrically
        T = T_start * (T_end / T_start) ** (elapsed / total_time)
        # learning‑rate for Adam stays constant (lr0 is applied later)
        # sub‑gradient scaling factor decays geometrically
        eta = eta_start * (eta_end / eta_start) ** (elapsed / total_time)

        # ----------------------------------------------------------
        # 3.2 Stagnation handling – jitter after 20 s without improvement
        # ----------------------------------------------------------
        if now - last_improve > 20.0:
            # multiplicative jitter centred on the best solution
            sigma = 0.08
            jitter = np.exp(rng.normal(0.0, sigma, size=n))
            a = proj_simplex(best_a * jitter)
            cur_J = compute_J(a, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – current J = {cur_J:.6f}")

        # ----------------------------------------------------------
        # 3.3 Compute convolution and obtain the true max index
        # ----------------------------------------------------------
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        t_star = int(conv.argmax())

        # ----------------------------------------------------------
        # 3.4 Sub‑gradient of the true max
        # ----------------------------------------------------------
        g = subgrad_max_conv(a, t_star)

        # ----------------------------------------------------------
        # 3.5 Adam update (on the sub‑gradient)
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        # incorporate the geometric sub‑gradient scaling
        a_proposed = a - eta * step
        a_proposed = proj_simplex(a_proposed)

        # ----------------------------------------------------------
        # 3.6 Small exchange move (occasionally)
        # ----------------------------------------------------------
        if rng.random() < 0.07:
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(a_proposed[i], rng.random() * 0.004)   # ≤ 0.4 % mass
            a_proposed[i] -= delta
            a_proposed[j] += delta
            a_proposed = proj_simplex(a_proposed)

        # ----------------------------------------------------------
        # 3.7 Evaluate the proposal (exact J)
        # ----------------------------------------------------------
        new_J = compute_J(a_proposed, n, fft_len, conv_len)

        # ----------------------------------------------------------
        # 3.8 Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_proposed
            cur_J = new_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    print(f"[{time.monotonic() - start_time:6.2f}s] Finished – final J = {best_J:.6f}")
    return [float(v) for v in best_a]
