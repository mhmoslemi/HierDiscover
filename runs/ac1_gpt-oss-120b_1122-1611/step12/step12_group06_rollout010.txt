def propose_candidate(seed=..., budget_s=1000.0):
    rng = np.random.default_rng(seed)
    # determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 2048
    # use sum=1
    if 'height_sequence_1' in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0, None)
        if a.size != n: # adjust length
            # similar code as before but we set n accordingly
    else:
        a = np.full(n, 1.0/n)
    # ensure sum=1
    a = a / a.sum()
    # compute best_J = evaluate
    conv = np.convolve(a, a)
    best_J = float(2.0 * n * conv.max() / (a.sum() ** 2))
    best_a = a.copy()
    last_improve_time = time.monotonic()
    start_time = last_improve_time
    deadline = start_time + budget_s - 5
    # main loop
    while time.monotonic() < deadline:
        # compute conv and max
        conv = np.convolve(a, a)
        max_idx = np.argmax(conv)
        max_val = conv[max_idx]
        # compute contributions for indices
        contrib = np.zeros_like(a)
        # compute j = max_idx - i
        i_vals = np.arange(n)
        j_vals = max_idx - i_vals
        mask = (j_vals >= 0) & (j_vals < n)
        # contributions only where mask yes
        contrib[mask] = a[i_vals[mask]] * a[j_vals[mask]]
        # find i_max with largest contrib
        if mask.sum() == 0:
            # fallback
            i_max = rng.integers(n)
            j_max = rng.integers(n)
        else:
            i_max = np.argmax(contrib)
            j_max = max_idx - i_max
        # generate random target index k not equal i_max or j_max
        # choose random k from [0,n)
        k = rng.integers(n)
        while k == i_max or k == j_max:
            k = rng.integers(n)
        # determine delta move size
        # max possible delta = min(a[i_max], a[j_max])
        delta_max = min(a[i_max], a[j_max])
        if delta_max <= 0:
            # can't move, random jitter
            # small random perturbation
            noise = rng.normal(scale=0.01, size=n)
            a = np.maximum(0, a + noise)
            a = a / a.sum()
            continue
        # choose delta = factor * delta_max, factor decays over time
        elapsed = time.monotonic() - start_time
        progress = elapsed / (budget_s - 5)
        # factor start 0.5, then decays to 0.01
        factor = 0.5 * (1 - progress) + 0.01 * progress
        delta = factor * delta_max
        # apply move: subtract delta from i_max and j_max, add 2*delta to k
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[j_max] -= delta
        a_new[k] += 2 * delta
        # ensure nonnegativity (due to rounding)
        if a_new[i_max] < 0 or a_new[j_max] < 0:
            # adjust delta
            delta = min(a[i_max], a[j_max]) * 0.5
            a_new = a.copy()
            a_new[i_max] -= delta
            a_new[j_max] -= delta
            a_new[k] += 2 * delta
        # compute new J
        conv_new = np.convolve(a_new, a_new)
        J_new = float(2.0 * n * conv_new.max() / (a_new.sum() ** 2))
        # accept if improvement
        if J_new < best_J:
            best_J = J_new
            best_a = a_new.copy()
            last_improve_time = time.monotonic()
            a = a_new
            print(f"[{elapsed:.2f}s] Improved J = {best_J:.6f}")
        else:
            # else maybe accept with prob according to temperature
            T = 0.01 * (1 - progress) + 1e-4 * progress
            if rng.random() < np.exp(-(J_new - best_J) / T):

            # apply jitter: small random noise

            a = np.maximum(0, a + noise)
            a = a / a.sum()
            last_improve_time = time.monotonic()
    # final result
    best_vec = np.clip(best_a, 0.0, 1000.0)
    # ensure sum>0.01

    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")
    return [float(v) for v in best_vec]

import numpy as np
import time

def _compute_J(a: np.ndarray) -> float:
    """Return evaluation metric J = 2 n max(conv(a,a)) / (sum(a)^2)."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Heuristic local‑search / simulated‑annealing optimiser.
    Starts from the previously discovered best construction (if any),
    repeatedly moves a small amount of mass away from the indices that
    contribute most to the current peak of the self‑convolution and
    accepts any improvement (or a probabilistic worsening move).  Random
    Dirichlet restarts and occasional jitter are injected to escape
    local minima.  The routine respects the time budget and returns the
    best sequence found.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 1. Problem size and initial sequence
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        # use the length of the previously reported construction
        init_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = max(256, init_seq.size)
    else:
        n = 2048
        init_seq = None

    # ensure a 1‑D array of length n
    if init_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.clip(init_seq, 0.0, None)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.concatenate([a, np.full(n - a.size, 0.0)], dtype=np.float64)
        # normalise to unit L1‑norm (scale does not matter for J)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a.sum()

    # -----------------------------------------------------------------
    # 2. Global bookkeeping
    # -----------------------------------------------------------------
    best_a = a.copy()
    best_J = _compute_J(best_a)
    cur_a = a.copy()
    cur_J = best_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # schedule parameters
    stagnation_limit = 30.0      # seconds without improvement -> jitter
    random_sample_interval = 15.0
    next_random_sample = start_time + random_sample_interval

    # -----------------------------------------------------------------
    # 3. Main optimisation loop
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- 3.1   Focused move that attacks the current peak ----------
        conv = np.convolve(cur_a, cur_a)
        k_max = int(conv.argmax())

        # contributions of each index to the peak at k_max
        i = np.arange(n, dtype=int)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur_a)
        contrib[mask] = cur_a[i[mask]] * cur_a[j[mask]]

        if mask.any():
            i_max = int(np.argmax(contrib))
            j_max = int(k_max - i_max)
        else:
            # degenerate case – pick two random positions
            i_max = rng.integers(n)
            j_max = rng.integers(n)

        # pick a third index to receive the moved mass
        k_target = rng.integers(n)
        while k_target == i_max or k_target == j_max:
            k_target = rng.integers(n)

        # amount we are allowed to move (cannot make entries negative)
        delta_allowed = min(cur_a[i_max], cur_a[j_max])
        if delta_allowed > 0:
            # factor slowly decays from 0.5 to 0.01 as we approach the deadline
            progress = elapsed / (budget_s - 5.0)
            factor = 0.5 * (1.0 - progress) + 0.01 * progress
            delta = factor * delta_allowed

            cand[i_max] -= delta

            # numerical safety – re‑normalise to unit L1‑norm
            total = cand.sum()
            if total > 0.0:
                cand /= total

            cand_J = _compute_J(cand)

            if cand_J < cur_J:
                cur_a, cur_J = cand, cand_J
                if cand_J < best_J:
                    best_a, best_J = cand.copy(), cand_J

    final_J = _compute_J(best_vec)
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

Heuristic local‑search / simulated‑annealing optimiser for the step‑function
attack.  The algorithm starts from the previously discovered best construction
(if available) and repeatedly moves a small amount of mass away from the
indices that contribute most to the current peak of the self‑convolution.
Improvements are accepted greedily; worsening moves are accepted with a
temperature that decays over time.  Random Dirichlet restarts, occasional
jitter and very rare permutation tweaks are injected to escape local minima.
The routine respects a 1000 s budget and returns the best sequence found.

import time
import numpy as np


def _compute_J(a: np.ndarray) -> float:
    """
    Compute the evaluation metric J = 2·n·max(conv(a,a)) / (sum(a)**2).
    This is exactly what `evaluate_sequence` returns (up to clipping that never
    happens for the values used here).
    """

    conv = np.convolve(a, a)
    max_conv = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))

    Search for a non‑negative step function (list of heights) that minimises

    peak‑reduction move with stochastic restarts and a tiny simulated‑annealing
    component.  It returns the best sequence discovered before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 1. Problem size and initial sequence
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        init_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = max(256, init_seq.size)
    else:
        init_seq = None
        n = 2048

    if init_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.clip(init_seq, 0.0, None)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.concatenate([a, np.full(n - a.size, 0.0)], dtype=np.float64)
        if a.sum() == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a / a.sum()          # unit L1‑norm (scale does not affect J)

    # -----------------------------------------------------------------
    # 2. Global bookkeeping
    # -----------------------------------------------------------------
    best_a = a.copy()
    best_J = _compute_J(best_a)
    cur_a = a.copy()
    cur_J = best_J
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    last_improve = start_time

    # Parameters for the stochastic parts
    stagnation_limit = 30.0               # seconds without improvement → jitter
    random_sample_interval = 15.0
    next_random_sample = start_time + random_sample_interval

    # -----------------------------------------------------------------
    # 3. Main optimisation loop
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # -------------------------------------------------------------
        # 3.1  Targeted move that attacks the current convolution peak
        # -------------------------------------------------------------
        conv = np.convolve(cur_a, cur_a)
        k_max = int(conv.argmax())

        # contributions of each index to the peak at k_max
        i = np.arange(n, dtype=int)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(cur_a)
        contrib[mask] = cur_a[i[mask]] * cur_a[j[mask]]

        if mask.any():
            i_max = int(np.argmax(contrib))
            j_max = int(k_max - i_max)
        else:
            # degenerate – pick two random positions
            i_max = rng.integers(n)
            j_max = rng.integers(n)

        # pick a third index that will receive the moved mass
        k_target = rng.integers(n)
        while k_target == i_max or k_target == j_max:
            k_target = rng.integers(n)

        # maximal transferable amount (cannot go negative)
        delta_allowed = min(cur_a[i_max], cur_a[j_max])
        if delta_allowed > 0:
            # step size shrinks from 0.5 → 0.01 during the run
            progress = elapsed / (budget_s - 5.0)
            factor = 0.5 * (1.0 - progress) + 0.01 * progress
            delta = factor * delta_allowed

            cand = cur_a.copy()
            cand[i_max] -= delta
            cand[j_max] -= delta
            cand[k_target] += 2.0 * delta

            # re‑normalise to unit L1 (tiny numerical drift may appear)
            total = cand.sum()
            if total > 0.0:
                cand /= total

            cand_J = _compute_J(cand)

            if cand_J < cur_J:
                cur_a, cur_J = cand, cand_J
                if cand_J < best_J:
                    best_a, best_J = cand.copy(), cand_J
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
            else:
                # simulated‑annealing acceptance
                T = 0.02 * (1.0 - progress) + 1e-4 * progress
                if rng.random() < np.exp(-(cand_J - cur_J) / (T + 1e-12)):
                    cur_a, cur_J = cand, cand_J

        # -------------------------------------------------------------
        # 3.2  Periodic random Dirichlet sampling (exploratory restarts)
        # -------------------------------------------------------------
        if now >= next_random_sample:
            for _ in range(3):
                alpha = rng.choice([0.3, 0.6, 1.0, 2.0])
                cand = rng.dirichlet(np.full(n, alpha))
                cand_J = _compute_J(cand)
                if cand_J < best_J:
                    best_a, best_J = cand.copy(), cand_J
                    cur_a, cur_J = cand.copy(), cand_J
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] Random‑sample improvement J = {best_J:.6f}")
            next_random_sample = now + random_sample_interval

        # -------------------------------------------------------------
        # 3.3  Stagnation handling – jitter around the current best
        # -------------------------------------------------------------
        if now - last_improve > stagnation_limit:
            jitter_sigma = 0.1
            cand = best_a + rng.normal(scale=jitter_sigma, size=n)
            cand = np.clip(cand, 0.0, None)
            s = cand.sum()
            if s > 0.0:
                cand /= s
            cand_J = _compute_J(cand)
            cur_a, cur_J = cand, cand_J
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – J = {cand_J:.6f}")

        # -------------------------------------------------------------
        # 3.4  Very occasional permutation tweak (helps if ordering matters)
        # -------------------------------------------------------------
        if rng.random() < 0.0005:   # ~once per few thousand iterations
            perm = rng.permutation(n)
            cand = cur_a[perm]
            cand_J = _compute_J(cand)
            if cand_J < cur_J:
                cur_a, cur_J = cand, cand_J
                if cand_J < best_J:
                    best_a, best_J = cand.copy(), cand_J
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] Permutation improvement J = {best_J:.6f}")

    # -----------------------------------------------------------------
    # 4. Return the best sequence (clipped to the allowed range)
    # -----------------------------------------------------------------
    best_vec = np.clip(best_a, 0.0, 1000.0)
    if best_vec.sum() < 0.01:
        best_vec = np.full_like(best_vec, 1.0 / best_vec.size)

    final_J = _compute_J(best_vec)
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
