def compute_J(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, None)   # ensure nonneg
    s = a.sum()
    if s < 1e-9:
        return np.inf, np.inf
    a = a / s
    n = a.shape[0]
    # convolution using FFT:
    m = len(a)
    conv_len = 2*m - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    max_b = conv.max()
    J = float(2.0 * m * max_b)
    return J, max_b
```

def generate_window(n, kind, param=None):
    if kind == 'uniform':
        return np.full(n, 1/n)
    elif kind == 'triangle':
        # Bartlett
        return np.bartlett(n) / np.bartlett(n).sum()
    elif kind == 'hann':
        w = np.hanning(n)
        return w / w.sum()
    elif kind == 'hamming':
        w = np.hamming(n)
        return w / w.sum()
    elif kind == 'blackman':
        w = np.blackman(n)
        return w / w.sum()
    elif kind == 'kaiser':
        beta = param if param is not None else 5.0
        w = np.kaiser(n, beta)
        return w / w.sum()
    elif kind == 'cosine':
        idx = np.arange(n)
        w = np.sin(np.pi * (idx + 0.5) / n)   # half-sine
        return w / w.sum()

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns an array of length 2*len(a)-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray) -> float:
    """
    Fast evaluation of the objective J = 2·n·max(conv(a,a)) / (sum(a)**2).
    The vector is automatically projected onto the non‑negative simplex
    (negative entries are clipped, then normalised to sum = 1).
    """
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, None)
    s = a.sum()

        return np.inf
    a = a / s
    n = a.shape[0]
    conv = _conv_full_fft(a)
    max_b = conv.max()
    return float(2.0 * n * max_b)


def _make_window(kind: str, n: int, param: float = None) -> np.ndarray:
    """Return a normalised window of length n."""
    if kind == "uniform":
        w = np.full(n, 1.0 / n, dtype=np.float64)

    elif kind == "bartlett":
        w = np.bartlett(n)
        if w.sum() == 0:

    elif kind == "hann":

        w = w / w.sum()

        w = np.hamming(n)
        w = w / w.sum()

    elif kind == "blackman":
        w = np.blackman(n)
        w = w / w.sum()

    elif kind == "kaiser":

        w = np.kaiser(n, beta)
        w = w / w.sum()

        w = ((idx + 1) * (n - idx)) ** p
        w = w / w.sum()

    elif kind == "exp":
        r = param if param is not None else 0.98
        w = r ** np.arange(n, dtype=np.float64)
        w = w / w.sum()

    else:
        raise ValueError(f"unknown window kind {kind}")

    # ensure non‑negative and finite
    w = np.clip(w, 0.0, None)
    w[np.isinf(w) | np.isnan(w)] = 0.0
    total = w.sum()
    if total == 0.0:
        w = np.full(n, 1.0 / n)
    else:
        w = w / total
    return w


def _lp_optimize(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_i   subject to   conv(f,g) ≤ M   and g_i ≥ 0.
    Returns the optimiser (may be all zeros if infeasible).
    """
    if not _SCIPY_AVAILABLE:          # pragma: no cover
        return np.zeros_like(f)

    n = f.shape[0]
    # Build the Toeplitz matrix that maps g → conv(f,g)
    # First column = f padded with zeros at the bottom
    c = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
    # First row = [f[0], 0, 0, …, 0]
    r = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
    A = scipy.linalg.toeplitz(c, r)    # shape (2n‑1, n)

    # Linear programme: maximise Σ g  ↔ minimise -Σ g
    res = scipy.optimize.linprog(
        c=-np.ones(n, dtype=np.float64),          # minimise -sum(g)
        A_ub=A,
        b_ub=np.full(2 * n - 1, M, dtype=np.float64),
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 15.0},
    )
    if not res.success:
        return np.zeros_like(f)
    return res.x


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low evaluate_sequence value.
    Returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # small safety buffer

    # ------------------------------------------------------------------
    # Phase 1 – generate a diverse pool of seeds on a few problem sizes.
    # ------------------------------------------------------------------
    n_candidates = [256, 512, 1024]          # modest sizes for speed
    window_kinds = [
        ("uniform", None),
        ("bartlett", None),
        ("hann", None),
        ("hamming", None),
        ("blackman", None),
        ("kaiser", 4.0),
        ("kaiser", 8.0),
        ("kaiser", 12.0),
        ("sine", None),
        ("power", 0.3),
        ("power", 0.6),
        ("power", 1.0),
        ("exp", 0.95),
        ("exp", 0.98),
    ]

    best_a = None
    best_J = np.inf
    best_n = None

    for n in n_candidates:
        # deterministic windows
        for kind, param in window_kinds:
            a = _make_window(kind, n, param)
            J = compute_J(a)
            if J < best_J:
                best_J, best_a, best_n = J, a.copy(), n
                print(f"[{time.monotonic() - start_time:6.2f}s] "
                      f"Init win {kind} (n={n}) → J={J:.6f}")

        # random Dirichlet draws (moderately concentrated)
        for _ in range(30):
            a = rng.dirichlet(np.ones(n) * 0.6)
            J = compute_J(a)
            if J < best_J:
                best_J, best_a, best_n = J, a.copy(), n
                print(f"[{time.monotonic() - start_time:6.2f}s] "
                      f"Dirichlet draw (n={n}) → J={J:.6f}")

        # include the historic construction if dimensions match
        if "height_sequence_1" in globals():
            old = np.asarray(height_sequence_1, dtype=np.float64)
            if old.size >= n:
                a = old[:n].copy()
            else:
                a = np.concatenate([old, np.full(n - old.size, 0.0)])
            a = np.clip(a, 0.0, None)
            total = a.sum()
            if total == 0.0:
                a = np.full(n, 1.0 / n)
            else:
                a = a / total
            J = compute_J(a)
            if J < best_J:
                best_J, best_a, best_n = J, a.copy(), n
                print(f"[{time.monotonic() - start_time:6.2f}s] "
                      f"Historic seq (trim/pad n={n}) → J={J:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based polishing (only if SciPy is available).
    # ------------------------------------------------------------------
    if _SCIPY_AVAILABLE and best_a is not None:
        f = best_a.copy()
        n = best_n
        # keep a copy of the best seen so far
        global_best_f = f.copy()
        global_best_J = best_J

        while time.monotonic() < deadline:
            # current bound
            M = _conv_full_fft(f).max()
            # solve the LP
            g = _lp_optimize(f, M)
            if g.sum() == 0.0:
                break
            g = g / g.sum()                      # renormalise to the simplex

            # try several mixing coefficients
            improved = False
            for t in (0.5, 0.3, 0.2, 0.1):
                candidate = (1.0 - t) * f + t * g
                Jc = compute_J(candidate)
                if Jc + 1e-12 < global_best_J:
                    global_best_J = Jc
                    global_best_f = candidate.copy()
                    improved = True
                    print(f"[{time.monotonic() - start_time:6.2f}s] "
                          f"LP mix t={t:.2f} → J={Jc:.6f}")

            if not improved:
                # No improvement – stop the LP phase
                break
            f = global_best_f.copy()
        best_a = global_best_f
        best_J = global_best_J

    # ------------------------------------------------------------------
    # Phase 3 – simple stochastic hill‑climbing from the best seed.
    # ------------------------------------------------------------------
    if best_a is not None:
        a = best_a.copy()
        n = a.shape[0]
        while time.monotonic() < deadline:
            # propose a random transfer of mass
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            if a[i] <= 0.0:
                continue
            # amount to move: a random fraction of a[i] (up to 20%)
            frac = rng.uniform(0.0, 0.2)
            delta = a[i] * frac
            a[i] -= delta
            a[j] += delta

            Jnew = compute_J(a)
            if Jnew + 1e-12 < best_J:
                best_J = Jnew
                best_a = a.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] "
                      f"Hill‑climb improvement → J={Jnew:.6f}")
            else:
                # revert move if not improving
                a[i] += delta
                a[j] -= delta

    # ------------------------------------------------------------------
    # Return the best sequence found (clipped to the allowed range).
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – uniform sequence of modest length
        best_a = np.full(256, 1.0 / 256, dtype=np.float64)

    out = [float(min(1000.0, max(0.0, v))) for v in best_a.tolist()]
    print(f"Finished. Best J ≈ {best_J:.6f} (length {len(out)})")
    return out
