"""
Simulated‑annealing optimiser on the probability simplex.

The objective (ignoring the scaling denominator) is
    J(a) = 2·n·max(conv(a,a)),   a_i ≥ 0,   Σ a_i = 1 .
We start from a uniform or the previously found construction,
project onto the simplex after every move and run a Metropolis‑style
annealing loop (single‑mass transfer moves).  Periodic “re‑heat” steps
replace the current state by a fresh Dirichlet draw when stagnation
exceeds 30 s.  In the last seconds a fine‑grained multiplicative
Gaussian jitter is applied around the best solution found.
The routine respects the 1000 s budget, prints progress timestamps and
returns a list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def obj_from_vec(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute the evaluation objective:
        J = 2·n·max(conv(a,a)) / (sum(a)**2)
    The function works on any non‑negative vector a (not necessarily
    normalised) – the division by the squared sum makes the objective
    scale‑invariant.
    """
    s = a.sum()
    # protect against division by zero (evaluate_sequence would reject)
    if s < 1e-12:
        return np.inf
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max() / (s * s))


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Simulated‑annealing search for a non‑negative step‑function (list of heights)
    that minimises `evaluate_sequence`.  Returns the best sequence found within
    the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # 1. Choose problem size
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n_prev = len(globals()["height_sequence_1"])
            n = max(2048, min(n_prev, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # -------------------------------------------------------------
    # 2. Initialise the current best point
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        best_vec = proj_simplex(v)
    else:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_J = obj_from_vec(best_vec, n, fft_len, conv_len)
    start_time = time.monotonic()
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # -------------------------------------------------------------
    # 3. Simulated‑annealing main loop
    # -------------------------------------------------------------
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # reserve a few seconds
    # temperature schedule (exponential cooling)
    T_initial = best_J * 0.5 if best_J > 0 else 0.1
    T_final = best_J * 1e-4 if best_J > 0 else 1e-6
    max_step = 0.05            # maximal mass transferred in one move
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # exponential cooling based on progress fraction
        progress = (elapsed) / (deadline - start_time)
        temp = T_initial * ((T_final / T_initial) ** progress) if T_initial > 0 else 1e-12

        # pick two distinct coordinates
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1

        # amount moved cannot exceed what is present at i
        max_delta = min(best_vec[i], max_step)
        if max_delta <= 0.0:
            continue
        delta = rng.uniform(0.0, max_delta)

        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_J = obj_from_vec(cand, n, fft_len, conv_len)

        # acceptance
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
        else:
            # Metropolis step
            if temp > 0 and rng.random() < np.exp(-(cand_J - best_J) / temp):
                best_vec = cand
                best_J = cand_J

        # ---------------------------------------------------------
        # 4. Stagnation handling – occasional random restart
        # ---------------------------------------------------------
        if now - last_improve > 30.0:
            # generate a fresh Dirichlet candidate
            alpha = rng.uniform(0.1, 2.0)
            fresh = rng.dirichlet(np.ones(n) * alpha)
            fresh = proj_simplex(fresh)
            fresh_J = obj_from_vec(fresh, n, fft_len, conv_len)
            if fresh_J < best_J:
                best_J = fresh_J
                best_vec = fresh
                last_improve = now
                print(f"[{elapsed:6.2f}s] Stagnation reset improved J = {best_J:.6f}")
            else:
                # probabilistic acceptance just to escape flat regions
                if temp > 0 and rng.random() < np.exp(-(fresh_J - best_J) / temp):
                    best_vec = fresh
                    best_J = fresh_J
            # treat this as a “re‑heat” event
            last_improve = now

    # -------------------------------------------------------------
    # 5. Fine‑grained jitter around the best solution
    # -------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fine_deadline:
        # multiplicative Gaussian jitter (small sigma)
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_J = obj_from_vec(cand, n, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # -------------------------------------------------------------
    # 6. Final normalisation & clipping (mirrors `evaluate_sequence`)
    # -------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
