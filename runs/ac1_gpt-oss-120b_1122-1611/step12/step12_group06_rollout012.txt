def proj_simplex(v):
    """Project vector v onto probability simplex (nonneg, sum=1)."""
    # based on algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function sequence that (probabilistically) improves the
upper‑bound 1.513196 → lower values.

Algorithm (≈ 1000 s budget)

1. **Initial portfolio** – generate many diverse candidates:
   uniform, geometric, ramps, Gaussians, power‑law, exponential,
   Dirichlet draws and the previously found best solution (if any).
   Keep the one with smallest exact objective J = 2·n·max(conv).

2. **Linear‑programming refinement** (only for moderate size, n≤2500):
   given f, let M = max(f∗f).  Solve the LP
        max   1ᵀg
        s.t.  f∗g ≤ M ,   g≥0
   (A∗g ≤ M with ``A`` the convolution matrix of f).  The optimal g tends
   to place mass where f is small.  Update f ← (1‑t)·f + t·g (t≈0.5) and
   repeat while improvement continues and time permits.

3. **Annealed‑softmax Adam** – exactly the optimiser from the
   reference code, but with a more aggressive temperature schedule
   (β from 10 to 5000) and a slower learning‑rate decay.
   The vector is kept on the probability simplex via the soft‑max.
   Jitter restarts are triggered after 15 s of stagnation.

4. **Sub‑gradient polishing** – for the last seconds a cheap
   sub‑gradient step on the true max‑convolution is performed:
   locate the peak index k, compute ∂max/∂a = shift(a,k), take a step
   and re‑project onto the simplex.

Throughout the run the best exact J and its sequence are remembered.
The final vector is clipped to the allowed range [0,1000] and re‑normalised
so that ``evaluate_sequence`` can be called directly.

The routine respects the 1000 s external timeout and prints progress.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution via real FFT (full length)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z = x - np.max(x)
    e = np.exp(z)
    return e / e.sum()


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights over the convolution entries.
    w_i = exp(beta*(conv_i - conv_max)) / Σ.
    """
    cmax = conv.max()
    exp_part = np.exp(beta * (conv - cmax))
    total = exp_part.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / total


def _grad_softmax(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ w_i·(a∗a)_i w.r.t. a.
    Using correlation: grad = 2 * corr(a, weights).
    """
    n = a.size
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    A = np.fft.rfft(a, fft_len)
    W = np.fft.rfft(weights, fft_len)
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len)[:corr_len]

    start = 2 * n - 2               # (len(weights)-1)
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum=1)."""
    if np.all(v <= 0):
        # all non‑positive -> uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(v.size) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_conv_matrix(f: np.ndarray) -> "scipy.sparse.csr_matrix":
    """Build the sparse matrix A (shape 2n‑1 × n) such that A·g = f∗g."""
    import scipy.sparse as sp

    n = f.size
    conv_len = 2 * n - 1
    # each diagonal offset d (0‑based) contains n copies of f[d]
    data = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(conv_len, n), format="csr")
    return A


# -------------------------------------------------------------
# Main search routine
# -------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    Returns the best sequence found before the budget expires.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 1. Problem size -------------------------------------------------
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # -----------------------------------------------------------------
    # 2. Generate an initial portfolio of candidates -------------------
    # -----------------------------------------------------------------
    init_cands = []

    # 2.1 Uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2.2 Geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # 2.3 Linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # 2.4 Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # 2.5 Power‑law family (large mass at edges)
    for p in (0.2, 0.5, 0.8, 1.0):
        idx = np.arange(n, dtype=np.float64)
        v = (idx + 1) ** (-p) + (n - idx) ** (-p)
        v /= v.sum()
        init_cands.append(v)

    # 2.6 Exponential decay from centre (edge‑heavy)
    for lam in (0.001, 0.002, 0.005):
        v = np.exp(-lam * np.abs(x - centre))
        v /= v.sum()
        init_cands.append(v)

    # 2.7 Dirichlet draws – dense and sparse regimes
    for alpha in (0.05, 0.1, 0.3, 0.6, 1.0):
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.full(n, alpha)))

    # 2.8 Previous best (if available)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        if v.sum() == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= v.sum()
        init_cands.append(v)

    # -----------------------------------------------------------------
    # 3. Pick the best starter (exact objective J = 2·n·max(conv))
    # -----------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        a = np.clip(cand, 0.0, None)
        tot = a.sum()
        if tot == 0.0:
            continue
        a = a / tot
        conv = _conv_fft(a, fft_len, conv_len)
        J = float(2.0 * n * conv.max())
        if J < best_J:
            best_J = J
            best_a = a.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # -----------------------------------------------------------------
    # 4. Timings
    # -----------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    # -----------------------------------------------------------------
    # 5. Phase 1 – linear‑programming refinement (if feasible)
    # -----------------------------------------------------------------
    # Run only for moderately sized problems (memory considerations).
    LP_TIME_FRACTION = 0.15
    lp_deadline = start_time + LP_TIME_FRACTION * budget_s
    if n <= 2500:
        try:
            from scipy.optimize import linprog
            import scipy.sparse as sp
        except Exception as e:
            print("SciPy not available – skipping LP phase.")
        else:
            t_step = 0.5
            no_improve_lp = 0
            while time.monotonic() < lp_deadline and no_improve_lp < 3:
                # current bound
                conv_f = _conv_fft(best_a, fft_len, conv_len)
                M = conv_f.max()
                # build sparse constraint matrix A (f∗g ≤ M)
                A = _build_conv_matrix(best_a)
                c = -np.ones(n)                     # maximise sum(g) → minimise -sum(g)
                b_ub = np.full(conv_len, M)
                bounds = [(0, None)] * n
                # Solve the LP (high‑s performance)
                res = linprog(
                    c,
                    A_ub=A,
                    b_ub=b_ub,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True},
                )
                if not res.success:
                    print("LP solver failed – exiting LP phase.")
                    break
                g = res.x
                sum_g = g.sum()
                if sum_g <= 0.0:
                    break
                g_norm = g / sum_g
                # convex combination
                a_new = (1.0 - t_step) * best_a + t_step * g_norm
                a_new = np.clip(a_new, 0.0, None)
                a_new /= a_new.sum()
                conv_new = _conv_fft(a_new, fft_len, conv_len)
                J_new = float(2.0 * n * conv_new.max())
                if J_new < best_J - 1e-12:
                    best_J, best_a = J_new, a_new.copy()
                    no_improve_lp = 0
                    print(f"[{time.monotonic() - start_time:6.2f}s] LP step → J = {best_J:.6f}")
                else:
                    no_improve_lp += 1
    else:
        print(f"[{time.monotonic() - start_time:6.2f}s] Skipping LP phase (n={n} too large).")

    # -----------------------------------------------------------------
    # 6. Phase 2 – annealed‑softmax Adam optimisation
    # -----------------------------------------------------------------
    # hyper‑parameters
    eps = 1e-12
    logits = np.log(np.maximum(best_a, eps))
    m = np.zeros_like(logits)
    v_m = np.zeros_like(logits)

    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.5  # initial learning rate (decays as 1/√t)

    beta_min = 10.0
    beta_max = 5000.0

    iteration = 0
    last_improve = start_time
    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # -------------------------------------------------------------
        # a) current a from logits
        # -------------------------------------------------------------
        a = _softmax(logits)

        # b) exact convolution and objective
        conv = _conv_fft(a, fft_len, conv_len)
        cur_M = conv.max()
        cur_J = float(2.0 * n * cur_M)

        # c) record global best
        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # d) stagnation detection => jitter restart
        if now - last_improve > 15.0:
            # add jitter to escape flat region
            jitter_sigma = 0.12
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v_m.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J={global_best_J:.6f})")

        # e) temperature schedule (linear anneal)
        progress = min(1.0, (now - start_time) / (deadline - start_time))
        beta = beta_min + (beta_max - beta_min) * progress

        # f) soft‑max weights over convolution entries
        weights = _softmax_weights(conv, beta)

        # g) smooth gradient wrt a
        grad_a = _grad_softmax(a, weights)

        # h) back‑propagate through the soft‑max (logits space)
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)

        # i) Adam update on logits
        iteration += 1
        lr = lr0 / np.sqrt(iteration)
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_m = beta2 * v_m + (1.0 - beta2) * (grad_z ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_m / (1.0 - beta2 ** iteration)
        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # keep logits in a safe numeric range
        logits = np.clip(logits, -500.0, 500.0)

    # -----------------------------------------------------------------
    # 7. Phase 3 – cheap sub‑gradient polishing (if time left)
    # -----------------------------------------------------------------
    sub_start = time.monotonic()
    while time.monotonic() - sub_start < 30.0 and time.monotonic() < deadline:
        a = global_best_a.copy()
        conv = _conv_fft(a, fft_len, conv_len)
        k_max = int(np.argmax(conv))
        # gradient of the max‑entry
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = a[k_max - idx]
        # simple additive step with diminishing step size
        step = 0.2 / np.sqrt(iteration + 1)
        a_new = a - step * grad
        a_new = _proj_simplex(a_new)
        conv_new = _conv_fft(a_new, fft_len, conv_len)
        J_new = float(2.0 * n * conv_new.max())
        if J_new < global_best_J - 1e-12:
            global_best_J, global_best_a = J_new, a_new.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] Sub‑grad polish → J = {global_best_J:.6f}")
        iteration += 1

    # -----------------------------------------------------------------
    # 8. Return best sequence (clipped & re‑normalised)
    # -----------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    final_J = float(2.0 * n * np.convolve(best_vec, best_vec).max())
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
