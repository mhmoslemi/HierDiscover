"""
Subgradient descent on the infinity norm of the self‑convolution.

The objective for a non‑negative vector a on the probability simplex
(sum(a)=1) is

    J(a) = 2·n·max(conv(a,a))

where conv(a,a) is the full discrete linear convolution.  The
convolution sum is always 1, so minimising J is equivalent to flattening
the convolution as much as possible.

The algorithm proceeds as follows:

1. Initialise a from the previously obtained best sequence
   (`height_sequence_1`) if it exists, otherwise from a sparse Dirichlet
   draw.
2. Repeatedly:
   * compute the full convolution by a real FFT,
   * locate the index k_max where the convolution attains its maximum,
   * compute the sub‑gradient of max(conv) at the current point,
     which is simply 2·a reversed and shifted by k_max,
   * take a projected sub‑gradient step onto the probability simplex,
     with a step size decreasing like η₀/(1+decay·iteration).
   * accept the step only if it does not increase the objective;
     keep track of the best vector seen so far.
   * if no improvement occurs for a while, apply a multiplicative
     log‑normal perturbation around the best vector to escape possible
     plateaus.
3. After the time budget is almost exhausted, perform a short random
   local search (small multiplicative noise) around the best vector.
4. Return the best vector (clipped to the allowed range, renormalised).

The method combines a deterministic sub‑gradient optimisation with
occasion‑al stochastic perturbations and typically pushes the score
below the previous differential‑evolution result within the allotted
time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  The method uses a projected
    sub‑gradient descent on the max‑convolution objective together with
    occasional stochastic perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size and initial point
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = a0.size
            # ensure it lies on the simplex
            a0 = proj_simplex(a0)
        except Exception:
            # fallback
            n = 2048
            a0 = proj_simplex(rng.dirichlet(np.full(n, 0.3)))
    else:
        n = 2048
        a0 = proj_simplex(rng.dirichlet(np.full(n, 0.3)))

    # ------------------------------------------------------------------
    # 2. Pre‑computations for FFT‑based convolution
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # power‑of‑two for speed

    # ------------------------------------------------------------------
    # 3. Initialise the optimisation state
    # ------------------------------------------------------------------
    a = a0.copy()                           # current iterate
    b = conv_full_fft(a, fft_len, conv_len) # current convolution
    max_b = b.max()
    best_a = a.copy()
    best_J = float(2.0 * n * max_b)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # keep a margin
    last_improve = start_time

    print(f"[0.00s] Initial J = {best_J:.6f} (n = {n})")

    # ------------------------------------------------------------------
    # 4. Sub‑gradient descent loop
    # ------------------------------------------------------------------
    it = 0
    eta0 = 0.5               # initial step size
    min_eta = 1e-4
    decay = 1e-3             # controls step‑size decay

    while time.monotonic() < deadline:
        # (a) identify the offending index of the convolution
        k_max = int(b.argmax())          # 0 … 2n‑2

        # (b) compute the sub‑gradient of max(conv) w.r.t. a
        grad = np.zeros_like(a)
        start_j = max(0, k_max - (n - 1))
        end_j = min(k_max, n - 1)
        if start_j <= end_j:
            idx = np.arange(start_j, end_j + 1)
            grad[idx] = 2.0 * a[k_max - idx]

        # (c) step size schedule
        eta = max(min_eta, eta0 / (1.0 + decay * it))

        # (d) projected sub‑gradient step
        a_trial = proj_simplex(a - eta * grad)

        # (e) evaluate trial point
        b_trial = conv_full_fft(a_trial, fft_len, conv_len)
        J_trial = float(2.0 * n * b_trial.max())

        # (f) accept only if the objective does not increase
        if J_trial <= float(2.0 * n * max_b) + 1e-12:
            a = a_trial
            b = b_trial
            max_b = b.max()

        # (g) record the best ever seen
        if J_trial < best_J - 1e-12:
            best_J = J_trial
            best_a = a_trial.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 4.1 Stagnation handling – random perturbation around the best
        # ------------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > 20.0:
            # multiplicative log‑normal noise (≈5 % std)
            noise = np.exp(rng.normal(0.0, 0.05, size=n))
            a = proj_simplex(best_a * noise)
            b = conv_full_fft(a, fft_len, conv_len)
            max_b = b.max()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Perturbation applied (stagnation)")

        it += 1

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random search around the current best
    # ------------------------------------------------------------------
    fine_deadline = time.monotonic() + min(30.0, (deadline - start_time) * 0.05)
    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_a * noise)
        b_cand = conv_full_fft(cand, fft_len, conv_len)
        J_cand = float(2.0 * n * b_cand.max())
        if J_cand < best_J - 1e-12:
            best_J = J_cand
            best_a = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Normalise and return (ensure legal range)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = best_a.sum()
    if s > 0.0:
        best_a /= s
    else:
        best_a[:] = 1.0 / n

    return [float(v) for v in best_a]
