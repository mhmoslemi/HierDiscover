"""
Propose a step‑function (non‑negative list) that minimises the evaluation
function.  The method starts from the best known construction (if any),
performs a short random‑restart phase, then iteratively solves a linear
program

        max  Σ g_j
        s.t. f * g ≤ M   (componentwise)
             g ≥ 0

where M = max(f∗f).  The LP pushes mass onto positions where the current
distribution f is small while keeping the convolution bound.  The solution
g is normalised and blended with f; the process is repeated a few dozen
times.  Finally a smoothed‑max gradient descent with projection onto the
probability simplex fine‑tunes the result.

The algorithm respects the 1000 s time budget, prints progress and returns
the best sequence found as a plain list of floats.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper functions – top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The algorithm:
      1. Initialise a candidate (use the global best if available).
      2. Perform a quick random‑restart phase to obtain a decent start.
      3. Iterate:
           * compute max‑convolution M of the current vector,
           * solve a linear program that maximises Σ g_j under f∗g ≤ M,
           * normalise g, blend it into f and re‑project onto the simplex.
      4. Apply a smoothed‑max gradient descent for fine polishing.
    The best vector found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s)

    # --------------------------------------------------------------
    # 1. Choose a working length n (keep it modest for LP speed)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n0 = old.size
        # cap the dimension to avoid huge LP matrices
        n = max(256, min(n0, 2500))
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Initialise a feasible vector on the simplex (sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        v0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # truncate / pad to length n
        if v0.size > n:
            v0 = v0[:n]
        elif v0.size < n:
            v0 = np.concatenate([v0, np.zeros(n - v0.size, dtype=np.float64)])
        v0 = np.clip(v0, 0.0, 1000.0)
        f = proj_simplex(v0)
    else:
        # uniform Dirichlet start
        f = proj_simplex(rng.dirichlet(np.ones(n)))

    # --------------------------------------------------------------
    # 3. Helper to evaluate the objective J = 2·n·max(conv(f,f))
    # --------------------------------------------------------------
    def compute_J(a: np.ndarray) -> float:
        conv = conv_full_fft(a, fft_len, conv_len)
        M = conv.max()
        return float(2.0 * n * M)

    best_f = f.copy()
    best_J = compute_J(best_f)
    print(f"[0.00s] initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Quick random‑restart phase (few dozen random Dirichlet draws)
    # --------------------------------------------------------------
    for i in range(50):
        if time.monotonic() >= deadline:
            break
        cand = proj_simplex(rng.dirichlet(np.ones(n)))
        Jcand = compute_J(cand)
        if Jcand < best_J:
            best_J = Jcand
            best_f = cand.copy()
            f = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] random‑restart improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Main LP‑based improvement loop
    # --------------------------------------------------------------
    lp_iter = 0
    while time.monotonic() < deadline - 25.0:  # keep ~25 s for final polishing
        lp_iter += 1
        # current convolution and bound
        conv_f = conv_full_fft(f, fft_len, conv_len)
        M = conv_f.max()
        J = float(2.0 * n * M)

        if J < best_J - 1e-12:
            best_J = J
            best_f = f.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] LP iter {lp_iter} new best J = {best_J:.6f}")

        # build the Toeplitz matrix representing the linear map g ↦ f∗g
        # first column: f followed by zeros, length = conv_len
        col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
        # first row: f[0] then zeros, length = n
        row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
        A = toeplitz(col, row)               # shape (conv_len, n)

        # linear program: maximise Σ g_j  ⇔ minimise −Σ g_j
        c_obj = -np.ones(n, dtype=np.float64)
        b_ub = np.full(conv_len, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        # solve – HIGHS is fast and handles dense matrices well
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})

        if not res.success:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] LP failed (iteration {lp_iter}): {res.message}")
            break

        g_raw = res.x
        sum_g = g_raw.sum()
        if sum_g <= 0.0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] LP produced zero mass (iteration {lp_iter})")
            break

        # normalise g to the simplex
        g = g_raw / sum_g

        # blend with current f (step size decreasing with iteration)
        t = 0.2 * (1.0 / np.sqrt(lp_iter))
        f = (1.0 - t) * f + t * g
        f = proj_simplex(f)   # keep on the simplex (fix numeric drift)

    # --------------------------------------------------------------
    # 6. Fine‑grained smoothed‑max gradient descent
    # --------------------------------------------------------------
    fine_deadline = deadline - 2.0   # leave a couple of seconds for output
    it = 0
    while time.monotonic() < fine_deadline:
        it += 1
        conv = conv_full_fft(best_f, fft_len, conv_len)
        M = conv.max()
        J = float(2.0 * n * M)

        # soft‑max weights (smooth approximation of the max)
        alpha = 15.0
        w = np.exp(alpha * (conv - M))
        w_sum = w.sum()
        if w_sum == 0.0:
            break
        w /= w_sum

        # gradient of the smoothed objective
        # ∇ = 2 * (a * reverse(w)) evaluated at indices 2n‑2‑i
        rev_w = w[::-1]
        conv_rev = np.convolve(best_f, rev_w)               # length = 3n‑2
        grads = 2.0 * conv_rev[(2 * n - 2) - np.arange(n)]

        # step size schedule
        lr = 0.03 / np.sqrt(it)

        cand = proj_simplex(best_f - lr * grads)
        cand_J = compute_J(cand)

        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_f = cand
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] gradient step improved J = {best_J:.6f}")

        # occasional random multiplicative perturbation
        if it % 30 == 0:
            noise = np.exp(rng.normal(0.0, 0.015, size=n))
            pert = proj_simplex(best_f * noise)
            pert_J = compute_J(pert)
            if pert_J < best_J:
                best_J = pert_J
                best_f = pert
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] random perturb improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 7. Final normalisation (protect against tiny numeric drift)
    # --------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)
    s = best_f.sum()
    if s > 0.0:
        best_f /= s
    else:
        best_f[:] = 1.0 / n

    # return as a plain Python list (evaluate_sequence will handle scaling)
    return [float(v) for v in best_f]
