"""
Fast gradient‑descent search on the probability simplex.

The algorithm works directly on a probability vector p (∑p_i = 1, p_i ≥ 0) of
length *n*.  For a given p the objective is

    J(p) = 2·n·max_k (p * p)[k]

(the scale of the original `evaluate_sequence` is irrelevant because it is
scale‑invariant).

The method iteratively reduces the peak of the auto‑convolution using an
exponentiated‑gradient update:

    w_k ∝ exp(β·(conv_k – max(conv)))      # a smooth approximation of the active set
    ∇J ≈ 2·corr(w, p)                     # computed by FFT (see _weighted_gradient)
    p ← p·exp(-η·∇J)  →  renormalise onto the simplex

β grows automatically when the convolution becomes flatter (the difference
between max and min shrinks), which turns the smooth weight into an almost
binary mask on the current maximal entries.  η (step‑size) decays slowly.
If no improvement of the best value is seen for a while the current vector is
re‑initialised (jitter) to regain diversity.  All operations are O(n log n)
via real FFTs and the search comfortably fits into the 1000 s budget.

The routine returns the best probability vector found, which is already a
valid input for `evaluate_sequence` (non‑negative, sum > 0 and far from the
clipping limits).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.

    The search uses an exponentiated‑gradient scheme with a soft‑max
    approximation of the max‑convolution, all operations performed with FFTs.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (number of steps)
    # --------------------------------------------------------------
    # Larger n gives a tighter theoretical lower bound (≈1), but also costs
    # more per iteration.  16384 is a good compromise for the 1000 s budget.
    n = int(kwargs.get("n", 16384))
    if n < 1024:
        n = 1024
    # ensure n is not absurdly large (memory & time constraints)
    max_n = int(kwargs.get("max_n", 32768))
    n = min(n, max_n)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlations

    # --------------------------------------------------------------
    # 2. Helper to compute the objective (2·n·max(conv)) for a probability vector
    # --------------------------------------------------------------
    def _objective(p: np.ndarray) -> float:
        conv = _conv_full(p, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 3. Initialise a candidate vector
    # --------------------------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    # Start from the previous best construction if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.shape[0] < n:
            v = np.concatenate([v, np.zeros(n - v.shape[0])])
        elif v.shape[0] > n:
            v = v[:n]
        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = uniform.copy()
        cur = v.copy()
        cur_J = _objective(cur)
        best_vec = cur.copy()
        best_J = cur_J
        print(f"[init] Using previous construction, J = {best_J:.6f}")
    else:
        cur = uniform.copy()
        cur_J = _objective(cur)
        best_vec = cur.copy()
        best_J = cur_J
        print(f"[init] Starting from uniform, J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Parameters for the gradient loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # leave a safety margin
    last_best_time = start_time
    iteration = 0

    # hyper‑parameters (can be overridden via kwargs)
    beta_max = float(kwargs.get("beta_max", 200000.0))     # max value for soft‑max sharpness
    step_mul_initial = float(kwargs.get("step_mul", 0.5))  # initial multiplicative step size
    step_mul_min = float(kwargs.get("step_mul_min", 0.01))
    step_decay = float(kwargs.get("step_decay", 1e-4))     # decay of step_mul per iteration
    grad_clip = float(kwargs.get("grad_clip", 5.0))        # clip gradient entries
    jitter_interval = float(kwargs.get("jitter_interval", 30.0))
    target_J = float(kwargs.get("target_J", 1.5025))       # stop early if reached

    # lower bound (theoretical) for reference, useful for early exit
    lower_bound = 2.0 * n / (2.0 * n - 1)

    # --------------------------------------------------------------
    # 5. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- compute current convolution and objective -----
        conv = _conv_full(cur, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv

        # ----- keep global best -----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur.copy()
            last_best_time = time.monotonic()
            elapsed = last_best_time - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

            if best_J <= target_J:
                print(f"[{elapsed:6.2f}s] Target reached. Stopping early.")
                break

        # ----- construct a smooth approximation of the active set -----
        diff = max_conv - conv.min()
        # If the convolution is already flat, use a uniform weight
        if diff <= 0.0:
            weight = np.full(conv_len, 1.0 / conv_len, dtype=np.float64)
        else:
            # Adaptive β: the smaller the spread, the larger β becomes
            beta = min(beta_max, max(5.0, 5.0 / diff))
            # soft‑max (shifted by max to avoid overflow)
            weight = np.exp(beta * (conv - max_conv))
            weight_sum = weight.sum()
            if weight_sum == 0.0:
                weight = np.full_like(weight, 1.0 / weight.size)
            else:
                weight = weight / weight_sum

        # ----- gradient of the weighted sum of convolution entries -----
        grad = _weighted_gradient(cur, weight, fft_len_corr, n)

        # Clip to keep the exponentials numerically stable
        grad = np.clip(grad, -grad_clip, grad_clip)

        # ----- step‑size schedule (decaying multiplicative update) -----
        step_mul = max(step_mul_min,
                       step_mul_initial / (1.0 + iteration * step_decay))

        # Exponentiated update (guarantees non‑negativity)
        factor = np.exp(-step_mul * grad)
        cur = cur * factor
        s = cur.sum()
        if s == 0.0:
            cur = uniform.copy()
        else:
            cur = cur / s

        # ----- stagnation handling (jitter) -----
        now = time.monotonic()
        if now - last_best_time > jitter_interval:
            # re‑inject diversity with a sparse Dirichlet draw
            cur = rng.dirichlet(np.ones(n) * 0.2)
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()
            last_best_time = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Jittered population due to stagnation")

        iteration += 1

    # ------------------------------------------------------------------
    # 6. Final sanitisation
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # (Optional) sanity‑check with the supplied evaluate_sequence, commented out.
    # try:
    #     import __main__ as _m
    #     val = evaluate_sequence(best_vec.tolist())
    #     print(f"Final evaluate_sequence = {val:.6f}")
    # except Exception:
    #     pass

    return [float(v) for v in best_vec]
