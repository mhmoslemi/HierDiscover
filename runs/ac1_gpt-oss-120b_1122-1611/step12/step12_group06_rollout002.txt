"""
Propose a step‑function sequence that minimizes `evaluate_sequence`.

Algorithm
---------
1. Choose a problem size `n` (default 2048, or the length of the
   previously found `height_sequence_1` when available).
2. Initialise a diverse portfolio of distributions (uniform,
   geometric, Gaussian, Dirichlet, etc.) and keep the one with the
   smallest exact objective `J = 2·n·max(conv(a,a))`.
3. Optimise the distribution with a constrained Adam scheme.
   * The coefficients are kept on the probability simplex via a
     soft‑max parametrisation `a = softmax(z)`.
   * For the first ~70 % of the time a *smooth* objective
     `J_β = Σ w_i·conv_i` is used, where the weights `w_i` are a
     soft‑max of the convolution entries with an *exponentially*
     increasing temperature `β`.
   * For the remaining time a *hard* sub‑gradient of the true maximum
     is taken: the active set consists of all convolution entries
     within a tiny tolerance of the current maximum, and the gradient
     is the correlation of this mask with the reversed coefficient
     vector.
   * After each Adam step the logits are transformed back to the
     simplex (soft‑max) and clipped to the allowed range.
   * If no improvement is seen for 20 s a jitter restart is performed.
4. Return the best sequence discovered within the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns the first `conv_len` entries of the linear convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i = exp(beta*(conv_i - conv_max)) / Σ.
    The subtraction of conv_max guarantees numerical stability.
    """
    c_max = conv.max()
    exp_part = np.exp(beta * (conv - c_max))
    total = exp_part.sum()
    if total == 0.0:
        # fallback – should never happen
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / total


def _grad_softmax(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·conv_i  w.r.t. a, where conv = a*a.
    Using the identity ∂conv_i/∂a = shifted copy of a,
    the gradient is 2 * correlation(a, weights) in the appropriate slice.
    """
    n = a.size
    # Length of full linear correlation = n + len(weights) - 1 = 3n-2
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    A = np.fft.rfft(a, fft_len)
    W = np.fft.rfft(weights, fft_len)
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len)[:corr_len]

    # Gradient slice starts at (2n-2) = len(weights)-1
    start = 2 * n - 2
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _grad_exact(a: np.ndarray, mask: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution objective.
    `mask` is a non‑negative weight vector (usually a normalized
    indicator of the active set). The gradient equals
        2 * correlation(mask, a_rev)
    where a_rev is `a` reversed.
    """
    n = a.shape[0]
    a_rev = a[::-1]
    mask_len = mask.size                     # = 2n-1
    corr_len = mask_len + n - 1               # = 3n-2
    fft_len = _next_pow2(corr_len)

    A_rev = np.fft.rfft(a_rev, fft_len)
    M = np.fft.rfft(mask, fft_len)
    corr_full = np.fft.irfft(M * A_rev, n=fft_len)[:corr_len]

    start = n - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine respects the time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a diverse set of initial candidates
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Previous best, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best starter (by exact objective J = 2·n·max(conv))
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        conv = _conv_fft(cand, fft_len_conv, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop (Adam on logits + annealed soft‑max → hard max)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    eps = 1e-12
    logits = np.log(np.maximum(best_a, eps))

    m = np.zeros_like(logits)
    v_m = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.5

    # Temperature schedule (soft‑max phase)
    beta_min = 5.0
    beta_max = 1e5   # aggressive final temperature
    soft_phase_frac = 0.7   # fraction of total time spent in soft‑max stage

    iteration = 0
    last_improve = start_time

    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # ----- 4.1 Current coefficients on the simplex -----
        a = _softmax(logits)

        # ----- 4.2 Convolution and exact objective -----
        conv = _conv_fft(a, fft_len_conv, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ----- 4.3 Track global best -----
        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----- 4.4 Jitter restart on stagnation -----
        if now - last_improve > 20.0:
            jitter_sigma = 0.2
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v_m.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

        # ----- 4.5 Determine phase and compute gradient -----
        progress = (now - start_time) / (deadline - start_time)
        if progress < soft_phase_frac:
            # ----- soft‑max (smooth) phase -----
            rel = progress / soft_phase_frac
            # exponential temperature growth for faster annealing
            beta = beta_min * (beta_max / beta_min) ** rel
            weights = _softmax_weights(conv, beta)
            grad_a = _grad_softmax(a, weights)
        else:
            # ----- hard‑max (sub‑gradient) phase -----
            # tolerance on the maximum (absolute and relative)
            tol_abs = 1e-12
            tol_rel = 1e-6
            tolerance = max(tol_abs, max_conv * tol_rel)
            mask_bool = conv >= (max_conv - tolerance)
            mask = mask_bool.astype(float)
            if mask.sum() > 0.0:
                mask /= mask.sum()        # normalise to keep gradient scale sensible
            else:
                mask = np.full_like(conv, 1.0 / conv.size)
            grad_a = _grad_exact(a, mask)

        # ----- 4.6 Chain rule: gradient w.r.t logits -----
        avg = np.dot(a, grad_a)               # scalar <a,∇_a J>
        grad_z = a * (grad_a - avg)           # ∂J/∂z

        # ----- 4.7 Adam update on logits -----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)        # simple 1/√t decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_m = beta2 * v_m + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_m / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # Keep logits within a reasonable numeric range
        logits = np.clip(logits, -500.0, 500.0)

    # --------------------------------------------------------------
    # 5. Return the best sequence (clipped, renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    final_J = float(2.0 * n * np.convolve(best_vec, best_vec).max())
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
