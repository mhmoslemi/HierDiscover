    # parameters
    alpha0 = 0.5  # starting step size
    eps = 1e-12
    noise_interval = 50
    noise_sigma = 0.05
    while time.monotonic() < deadline:
        # compute conv and J

        max_val = conv.max()
        # gradient of max

        active = (conv >= max_val - 1e-12).astype(np.float64)
        # optionally normalize weight to avoid scaling
        if active.sum() > 0:
            weight = active / active.sum()
        else:
            weight = active
        grad = _weighted_gradient(a, weight, fft_len, n)  # note: conv length maybe partial? Use weighting length 2*n-1

        # step size schedule
        step = alpha0 / np.sqrt(it + 1)
        a_new = a - step * grad
        # ensure nonnegative and project onto simplex

        s = a_new.sum()

        if J_new < best_J:

            best_a = a_new.copy()
            # also update a for continuing descent if improvement
            a = a_new
        else:

            temp = max(0.001, 0.1 * (deadline - time.monotonic()) / (deadline - start))
            if rng.random() < np.exp(-(J_new - best_J) / temp):

def gradient_descent_until_time(init_vec, n, fft_len, conv_len, fft_len_corr, start_time, deadline, rng):
    a = init_vec.copy()
    best_local = a.copy()
    # compute objective
    conv = _conv_full(a, fft_len, conv_len)
    best_J = 2.0 * n * conv.max()
    best_vec = a.copy()
    iter_count = 0
    # parameters
    step0 = 0.5
    eps = 1e-12
    noise_every = 200
    noise_sigma = 0.02
    while time.monotonic() < deadline:
        # compute conv and gradient
        conv = _conv_full(a, fft_len, conv_len)
        max_val = conv.max()
        # binary active mask
        active = (conv >= max_val - eps).astype(np.float64)
        # normalization to avoid scaling
        active_sum = active.sum()
        if active_sum > 0:
            weight = active / active_sum
        else:
            weight = active
        grad = _weighted_gradient(a, weight, fft_len_corr, n)
        # step schedule
        step = step0 / np.sqrt(iter_count + 1)
        a_new = a - step * grad
        # projection to simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / s
        # compute objective
        conv_new = _conv_full(a_new, fft_len, conv_len)
        J_new = 2.0 * n * conv_new.max()
        # acceptance criteria (greedy)
        if J_new < best_J:
            best_J = J_new
            best_vec = a_new.copy()
        # For overall algorithm we continue with a_new as current irrespective of improvement (to continue gradient descent)
        a = a_new
        # noise injection
        if iter_count % noise_every == 0 and iter_count > 0:
            # apply multiplicative noise
            noise = 1.0 + noise_sigma * rng.standard_normal(n)
            a = a * noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s
        iter_count += 1
        # time check maybe break after some iterations to avoid endless loops
        if time.monotonic() >= deadline:
            break
    return best_vec, best_J

"""
gradient‑softmax optimiser for the step‑function upper‑bound problem.

The algorithm lives on the probability simplex (∑a_i = 1, a_i ≥ 0) and
repeatedly reduces the peak of the auto‑convolution using a smooth
approximation of the max‑operator.  At each iteration we:

1. compute the full convolution a∗a by FFT,
2. build a soft‑max weight w_k ∝ exp(β·(c_k−max c)/max c) that
   concentrates on the current largest values,
3. evaluate the gradient of the weighted sum  Σ_k w_k·c_k
   efficiently via another FFT,
4. take a diminishing step in the opposite direction,
5. project back onto the simplex.

The optimiser is run in short “runs’’ (≈30 s each) with random
re‑initialisation between runs to avoid getting stuck in a local
minimum.  The best vector found over the whole wall‑clock budget
(≤ 1000 s) is returned as a plain Python list.  The implementation uses
only NumPy and the helper functions defined below – no closures or
lambdas – and respects the required function signature.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (zero‑padded to fft_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute the gradient of Σ_k weight_k·(a*a)_k efficiently via FFT.
    weight must have length 2·n‑1.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return the evaluation value 2·n·max(a*a) (sum is assumed = 1)."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine (gradient + soft‑max)
# ----------------------------------------------------------------------


def _gradient_descent_run(init_vec: np.ndarray,
                         n: int,
                         fft_len: int,
                         conv_len: int,
                         fft_len_corr: int,
                         start_time: float,
                         deadline: float,
                         rng: np.random.Generator,
                         beta: float = 12.0,
                         step0: float = 0.5,
                         noise_every: int = 500,
                         noise_sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Run a single gradient‑descent optimisation starting from ``init_vec``.
    The run stops when ``deadline`` is reached.  Returns the best vector
    encountered and its objective value.
    """
    a = init_vec.copy()
    best_vec = a.copy()
    best_J = _objective(a, n, fft_len, conv_len)
    eps = 1e-12
    it = 0

    while time.monotonic() < deadline:
        # ----- convolution and current max -----
        conv = _conv_full(a, fft_len, conv_len)
        max_val = conv.max()
        J = 2.0 * n * max_val
        if J < best_J:
            best_J = J
            best_vec = a.copy()

        # ----- soft‑max weighting of the convolution -----
        # relative difference to the current maximum – helps with scaling
        rel_diff = (conv - max_val) / max_val
        w = np.exp(beta * rel_diff)          # w_k = exp(β·(c_k−max)/max)
        w_sum = w.sum()
        if w_sum > 0.0:
            w /= w_sum
        else:
            # fall back to binary mask if something goes wrong
            w = (conv >= max_val - eps).astype(np.float64)
            ws = w.sum()
            if ws > 0:
                w /= ws

        # ----- gradient of the weighted sum -----
        grad = _weighted_gradient(a, w, fft_len_corr, n)

        # ----- diminishing step size (1/√t) -----
        step = step0 / np.sqrt(it + 1.0)
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new = a_new / s

        a = a_new

        # ----- occasional stochastic jitter to escape plateaux -----
        if it > 0 and it % noise_every == 0:
            noise = 1.0 + noise_sigma * rng.standard_normal(n)
            a = a * noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a = a / s

        # ----- progress output (lightweight) -----
        if it % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] iter {it:6d}  best J={best_J:.6f}")

        it += 1

    return best_vec, best_J


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs a series of gradient‑descent
    runs with soft‑max weighting of the convolution peak, periodically
    restarting from random points.  The best sequence found before the time
    budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size – we use a power‑of‑two length that is still fast.
    # --------------------------------------------------------------
    n = 8192                     # comfortable size for FFT, gives a low theoretical bound

    # --------------------------------------------------------------
    # 2. Initialise a starting point (use previous construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.maximum(base, 0.0)
        if base.size < n:
            base = np.concatenate([base, np.zeros(n - base.size)])
        elif base.size > n:
            base = base[:n]
        s = base.sum()
        if s > 0.0:
            base = base / s
        else:
            base = np.full(n, 1.0 / n)
    else:
        base = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 3. Pre‑compute FFT sizes
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                        # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)           # for weighted correlations

    # --------------------------------------------------------------
    # 4. Global best bookkeeping
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a small safety margin
    best_vec = base.copy()
    best_J = _objective(best_vec, n, fft_len, conv_len)
    print(f"[0.00s] initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Main optimisation loop – multiple short runs with random restarts
    # --------------------------------------------------------------
    run_idx = 0
    while time.monotonic() < deadline:
        # allocate a modest slice for a single run (≈30 s) – ensures we can
        # restart many times if needed.
        now = time.monotonic()
        run_deadline = min(deadline, now + 30.0)

        # Choose start point: the very first run uses the known construction,
        # subsequent runs start from a fresh Dirichlet draw.
        if run_idx == 0:
            init = best_vec.copy()
        else:
            # Dirichlet with concentration <1 tends to produce sparse patterns,
            # which is useful for exploration.
            alpha = np.full(n, 0.5)
            init = rng.dirichlet(alpha)

        vec, J = _gradient_descent_run(init, n, fft_len, conv_len,
                                       fft_len_corr, start_time,
                                       run_deadline, rng)

        # Update the global best if needed.
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new global best J = {best_J:.6f}")

        run_idx += 1

    # --------------------------------------------------------------
    # 6. Final clean‑up: clip, renormalise and convert to plain list
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n)
    else:
        best_vec = best_vec / total

    # Return a Python list as required by the evaluation harness.
    return [float(v) for v in best_vec]
