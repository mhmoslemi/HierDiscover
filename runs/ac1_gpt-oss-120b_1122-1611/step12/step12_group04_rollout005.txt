   evaluator).  The vector is returned as a plain Python list.

All heavy work uses real FFTs (O(n log n)) and runs well within the
1000 s budget on a single core.
"""

import time
import numpy as np
from scipy import signal

# ----------------------------------------------------------------------
# Helper functions (no closures or lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    weight length = 2·n‑1, a length = n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Window factories
# ----------------------------------------------------------------------


def _window_kaiser(n: int, beta: float) -> np.ndarray:
    w = signal.windows.kaiser(n, beta, sym=True)
    return np.abs(w)


def _window_chebwin(n: int, atten: float) -> np.ndarray:
    w = signal.windows.chebwin(n, atten, sym=True)
    return np.abs(w)


def _window_dpss(n: int, nw: float) -> np.ndarray:
    # DPSS returns an array of shape (Kmax, n); we take the first (Kmax=1)
    w = signal.windows.dpss(n, nw, Kmax=1, sym=True)[0]
    return np.abs(w)


def _window_hann(n: int) -> np.ndarray:
    return signal.windows.hann(n, sym=True)


def _window_blackman(n: int) -> np.ndarray:
    return signal.windows.blackman(n, sym=True)


def _window_linear_ramp(n: int, reverse: bool = False) -> np.ndarray:
    if reverse:
        v = np.arange(n, 0, -1, dtype=np.float64)
    else:
        v = np.arange(1, n + 1, dtype=np.float64)
    return v


def _window_triangle(n: int) -> np.ndarray:
    # symmetric triangle with peak in the centre
    half = (n + 1) // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    right = left[:-1][::-1] if n % 2 == 1 else left[::-1]
    return np.concatenate([left, right])


def _window_geometric(n: int, r: float) -> np.ndarray:
    return r ** np.arange(n, dtype=np.float64)


# ----------------------------------------------------------------------
# Main search function
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found within
    the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # -------------------------
    # 1. Problem size
    # -------------------------
    n = 8192                     # maximal power‑of‑two ≤ 8192
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlation

    # -------------------------
    # 2. Build a diverse set of initial candidates
    # -------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # deterministic windows
    for beta in [0, 1, 3, 5, 8, 12, 20, 30, 50, 80, 120]:
        init_cands.append(_window_kaiser(n, beta))

    for atten in [20, 40, 60, 80, 100, 120, 150]:
        init_cands.append(_window_chebwin(n, atten))

    for nw in np.arange(2.0, 9.5, 0.5):
        init_cands.append(_window_dpss(n, nw))

    init_cands.append(_window_hann(n))
    init_cands.append(_window_blackman(n))
    init_cands.append(_window_triangle(n))
    init_cands.append(_window_linear_ramp(n, reverse=False))
    init_cands.append(_window_linear_ramp(n, reverse=True))

    # geometric decays
    for r in [0.995, 0.99, 0.985, 0.98, 0.97, 0.95, 0.90]:
        init_cands.append(_window_geometric(n, r))

    # Dirichlet draws – various concentrations (including very sparse)
    for alpha in [0.005, 0.01, 0.02, 0.05, 0.1, 0.3, 0.6, 1.0]:
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if available
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(height_sequence_1, dtype=np.float64)
            v = np.clip(v, 0.0, None)
            s = v.sum()
            if s == 0.0:
                v[:] = 1.0 / n
            else:
                v /= s
            # if the old vector is shorter, pad with zeros
            if v.size < n:
                pad = np.zeros(n - v.size, dtype=np.float64)
                v = np.concatenate([v, pad])
            elif v.size > n:
                v = v[:n]
            init_cands.append(v)
        except Exception:
            pass

    # -------------------------
    # 3. Evaluate initial candidates (fast FFT objective)
    # -------------------------
    best_vec = None
    best_J = np.inf
    start_time = time.monotonic()
    for idx, cand in enumerate(init_cands):
        # Clip negatives (should not happen) and normalise to a simplex
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            continue
        cand = cand / s
        J = _objective(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
        if idx % 10 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Init candidate {idx}/{len(init_cands)} – J = {J:.6f}")

    print(f"[{time.monotonic() - start_time:6.2f}s] Initial best J = {best_J:.6f}")

    # -------------------------
    # 4. Adam‑style optimisation (focus on the max entry)
    # -------------------------
    # Allocate ~70 % of the total budget for this stage
    total_budget = float(budget_s)
    deadline = start_time + total_budget * 0.70
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_vec = cur_vec.copy()
    best_J = cur_J

    # Adam hyper‑parameters
    lr0 = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Soft‑max temperature schedule
    beta_start = 1.0
    beta_end = 5000.0

    # Metropolis temperature (linear cooling)
    T_initial = best_J * 0.1
    T_final = 1e-9

    m = np.zeros_like(cur_vec)
    v_hat = np.zeros_like(cur_vec)
    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / (deadline - start_time)  # [0,1]

        # ----- compute convolution and objective -----
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ----- record improvement -----
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best (Adam) J = {best_J:.6f}")

        # ----- stagnation handling -----
        if now - last_improve > 30.0:
            # multiplicative log‑normal jitter
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – best J = {best_J:.6f}")

        # ----- soft‑max weight (focus on the current maximum) -----
        beta = beta_start * (beta_end / beta_start) ** t_frac
        shifted = conv - max_conv          # max_conv is the maximum entry
        # stable soft‑max
        w_exp = np.exp(beta * shifted)    # all <= 1, no overflow
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum

        # ----- gradient of the smoothed objective -----
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)

        # ----- Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_corr = v_hat / (1.0 - beta2 ** iteration)
        lr = lr0 * (1.0 - t_frac) + 1e-3   # slowly decay learning rate
        step = - lr * m_hat / (np.sqrt(v_corr) + eps_adam)

        proposal = cur_vec + step
        proposal = proj_simplex(proposal)

        # ----- evaluate proposal -----
        prop_conv = _conv_full(proposal, fft_len, conv_len)
        prop_J = float(2.0 * n * prop_conv.max())

        # ----- Metropolis acceptance -----
        if prop_J < cur_J:
            accept = True
        else:
            T = (T_initial * (deadline - now) + T_final * (now - start_time)) / (deadline - start_time)
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = proposal
            cur_J = prop_J

    # -------------------------
    # 5. Local polishing: pairwise mass swaps with SA schedule
    # -------------------------
    polish_deadline = start_time + total_budget - 5.0   # keep a few seconds for finalisation
    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    # Polishing temperature schedule
    T0_polish = cur_J * 0.01
    T_end_polish = 1e-10

    while time.monotonic() < polish_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / total_budget
        T = T0_polish * (1.0 - t_frac) + T_end_polish * t_frac

        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # move a small amount of mass from i to j
        eps = rng.random() * cur_vec[i] * 0.5
        if eps <= 0.0:
            continue

        cand = cur_vec.copy()
        cand[i] -= eps
        cand[j] += eps

        # quick projection (negative rounding errors)
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        if cand_J < cur_J:
            cur_vec = cand
            cur_J = cand_J
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                print(f"[{elapsed:6.2f}s] Polishing improvement: J = {best_J:.6f}")
        else:
            # uphill acceptance with SA probability
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur_vec = cand
                cur_J = cand_J

    # -------------------------
    # 6. Final safety normalisation & conversion
    # -------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    print(f"[{time.monotonic() - start_time:6.2f}s] Finished – final J = {best_J:.6f}")
    return [float(v) for v in best_vec]
```
