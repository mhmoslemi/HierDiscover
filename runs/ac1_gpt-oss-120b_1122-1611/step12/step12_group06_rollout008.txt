"""
Propose a step‑function sequence with a small upper bound.

The search consists of two stages:

1. **Linear‑programming refinement** –  starting from a diverse set of
   normalised candidates we repeatedly solve a linear programme
   `max 1·g  s.t.  conv(a,g) ≤ M` (where `M = max(conv(a,a))`).  The
   optimiser returns an extreme point `g₀`; after normalising `g₀` to the
   required sum we update the current vector by a small convex combination.
   This procedure monotonically reduces the maximum of the self‑convolution.

2. **Annealed soft‑max Adam optimiser** –  the result of stage 1 is used as
   the initial point for a smooth optimisation of the true objective.
   The non‑smooth `max` is approximated by a temperature‑controlled
   soft‑max; the temperature is linearly increased (annealed) while Adam
   updates the logits of a soft‑max parametrisation of the vector.
   Occasional jitter restarts avoid stagnation.

Both stages respect a total time budget (default 1000 s) and keep a global
best solution found at any time.  The final list of heights is returned.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int = None, conv_len: int = None) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns the first `conv_len` entries of the linear convolution.
    """
    n = a.size
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    if conv_len is None:
        conv_len = 2 * n - 1
    A = np.fft.rfft(a, fft_len)
    conv_full = np.fft.irfft(A * A, n=fft_len)
    return conv_full[:conv_len]


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i = exp(beta*(conv_i - conv_max)) / Σ.
    The subtraction of conv_max guarantees numerical stability.
    """
    c_max = conv.max()
    exp_part = np.exp(beta * (conv - c_max))
    total = exp_part.sum()
    # Guard against degenerate case (should never happen)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / total


def _grad_softmax(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·conv_i with respect to a, where conv = a * a.
    The gradient is 2·correlation(a, weights) restricted to the original indices.
    """
    n = a.size
    # full correlation length = n + len(weights) - 1 = 3n-2
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    A = np.fft.rfft(a, fft_len)
    W = np.fft.rfft(weights, fft_len)
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len)[:corr_len]

    start = 2 * n - 2               # len(weights)-1 = 2n-2
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Construct the (2n‑1) × n matrix that maps a vector g to conv(a,g).
    Row k (0‑based) contains a_{k‑j} at column j whenever the index is valid,
    otherwise 0.
    """
    n = a.size
    conv_len = 2 * n - 1
    col = np.concatenate([a, np.zeros(conv_len - n)])
    row = np.concatenate([a[:1], np.zeros(n - 1)])
    return toeplitz(col, row)


def _initial_candidates(n: int, rng: np.random.Generator, target_sum: float):
    """Generate a small portfolio of diverse normalised start vectors."""
    cand = []
    # Uniform
    cand.append(np.full(n, 1.0 / n))
    # Geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)
    # Linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)
    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        cand.append(v)
    # Dirichlet draws
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(2):
            v = rng.dirichlet(np.ones(n) * alpha)
            cand.append(v)
    # Previous best, if available
    try:
        old = np.asarray(height_sequence_1, dtype=np.float64)
        if old.size != n:
            if old.size > n:
                old = old[:n]
            else:
                old = np.concatenate([old, np.full(n - old.size, 0.0)])
        old = np.clip(old, 0.0, None)
        if old.sum() == 0.0:
            old = np.full(n, 1.0 / n)
        else:
            old = old / old.sum()
        cand.append(old)
    except Exception:
        pass
    # Scale to the required sum
    scaled = [c * target_sum for c in cand]
    return scaled

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function that minimises `evaluate_sequence`.
    The routine runs for at most `budget_s` seconds and returns the best
    sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size ---------------------------------------------------
    # ------------------------------------------------------------------
    # Use a length comparable to the previously best construction.
    try:
        n = len(height_sequence_1)          # type: ignore[name-defined]
        # Clip to a reasonable interval for speed / memory
        n = max(1024, min(n, 4096))
    except Exception:
        n = 2048

    target_sum = math.sqrt(2.0 * n)          # the normalisation used in the benchmark
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Choose an initial point (best among a small portfolio) -------
    # ------------------------------------------------------------------
    init_candidates = _initial_candidates(n, rng, target_sum)

    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        cur_J = evaluate_sequence(cand.tolist())
        if cur_J < best_J:
            best_J = cur_J
            best_a = np.array(cand, copy=True)

    p = best_a / target_sum                     # p ≥ 0, sum(p) = 1

    best_J = evaluate_sequence((best_p * target_sum).tolist())
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # 3. Phase 1 – linear‑programming refinement ------------------------
    # ------------------------------------------------------------------

    t_mix = 0.25

    t_decay = 0.98

    iter_lp = 0

    while time.monotonic() < phase1_deadline:
        iter_lp += 1
        # 3.1 current max autocorrelation for the normalised vector p
        conv_p = _conv_fft(p, fft_len=fft_len, conv_len=conv_len)

        A = _build_conv_matrix(p)                 # shape (2n‑1, n)

        c = -np.ones(n)                           # linprog minimises → -sum(g)

            break

            print(f"Linprog iteration {iter_lp} failed (status {res.status}).")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print(f"LP iteration {iter_lp} returned zero sum – aborting LP phase.")
            break

        # 3.4 Normalise g0 to the probability simplex
        g = g0 / sum_g0

        # 3.5 Try a convex combination p ← (1‑t) p + t g
        improved = False
        t_cur = t_mix
        while t_cur >= min_t_mix:
            p_candidate = (1.0 - t_cur) * p + t_cur * g
            J_candidate = evaluate_sequence((p_candidate * target_sum).tolist())
            if J_candidate < best_J - 1e-12:
                p = p_candidate
                best_p = p_candidate.copy()
                best_J = J_candidate
                last_improve = time.monotonic()
                improved = True
                print(f"[{time.monotonic() - start_time:6.2f}s] LP iter {iter_lp:3d}: "
                      f"new best J = {best_J:.6f} (t={t_cur:.4f})")
                break
            else:
                t_cur *= 0.5

                else:

                last_improve = time.monotonic()

    remaining = deadline - time.monotonic()

        logits = np.log(np.maximum(p, eps))

        v = np.zeros_like(logits)

        lr0 = 0.5

                best_p = probs.copy()

                print(f"[{time.monotonic() - start_time:6.2f}s] Adam iter {iteration:5d}: "

                logits += rng.normal(0.0, jitter_sigma, size=n)
                m.fill(0.0)
                v.fill(0.0)
                last_improve_adam = time.monotonic()
                print(f"[{time.monotonic() - start_time:6.2f}s] Adam jitter restart")

            # ------------------------------------------------------------------
            # 4.1 Annealed soft‑max over convolution entries
            # ------------------------------------------------------------------
            progress = (time.monotonic() - (deadline - remaining)) / remaining
            beta = beta_min + (beta_max - beta_min) * progress

            weights = _softmax_weights(conv_a, beta)

            # ------------------------------------------------------------------
            # 4.2 Gradient of the smooth objective wrt the raw vector a
            # ------------------------------------------------------------------
            grad_a = _grad_softmax(a, weights)

            # ------------------------------------------------------------------

            # Prevent overflow / underflow in the logits
            logits = np.clip(logits, -500.0, 500.0)

    # ------------------------------------------------------------------
    # 5. Return the best sequence (clip, renormalise) --------------------
    # ------------------------------------------------------------------
    best_vec = np.clip(best_p * target_sum, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, target_sum / n)
    else:
        best_vec = best_vec * (target_sum / total)   # renormalise to the exact sum

    final_J = evaluate_sequence(best_vec.tolist())
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

    return [float(v) for v in best_vec.tolist()]
