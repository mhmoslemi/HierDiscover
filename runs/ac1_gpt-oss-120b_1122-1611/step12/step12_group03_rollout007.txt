def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)

    if 'height_sequence_1' in globals():
        base = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        n = base.size
        # maybe limit n to <=4096 to reduce computation
        n = min(max(512, n), 4096)
    else:
        n = 2048

    # Build initial candidates list
    init_vectors = []

    # Uniform
    init_vectors.append(np.full(n, 1.0/n, dtype=np.float64))

    # previous best if provided
    if 'height_sequence_1' in globals():
        v = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if v.size != n:
            if v.size > n: v = v[:n]
            else: v = np.concatenate([v, np.zeros(n - v.size)])
        init_vectors.append(proj_simplex(v))

    # Geometric
    for r in (0.9992, 0.9994, 0.9996):
        x = (1 - r) * (r ** np.arange(n, dtype=np.float64))
        init_vectors.append(proj_simplex(x))

    # Linear decreasing
    x = np.arange(n, 0, -1, dtype=np.float64)
    init_vectors.append(proj_simplex(x))

    # Symmetric geometric (both ends)
    r = 0.9995
    left = (1 - r) * (r ** np.arange(n//2, dtype=np.float64))
    right = left[::-1]
    if n % 2 == 1:
        middle = np.array([0.0])  # maybe zero
        sym = np.concatenate([left, middle, right])
    else:
        sym = np.concatenate([left, right])
    init_vectors.append(proj_simplex(sym))

    # random Dirichlet draws

    best_vec = None

        vvec = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999

"""
Propose a step‑function that minimises `evaluate_sequence`.

Algorithm
---------
1. Choose a problem size `n`.  If a previously found sequence
   `height_sequence_1` is present, its length (clipped to ≤4096) is used.
   Otherwise `n = 2048`.

2. Build a small set of diverse initial vectors on the probability simplex:
   • uniform,
   • the previous best (if any),
   • geometric decays with a few decay factors,
   • a decreasing linear ramp,
   • a symmetric geometric shape,
   • a few Dirichlet draws.

3. For each start run a *projected Adam* optimiser that minimises a
   smooth approximation of the max‑convolution:
   - The full linear convolution `c = a * a` is obtained with a real FFT.
   - A soft‑max weight `w_k = exp(gamma * c_k) / Σ exp(gamma * c_k)`
     (γ grows during optimisation) turns the non‑smooth
     `max` into a differentiable objective.
   - The gradient of the smooth objective is
       ∇J_i = 2 Σ_j a_j w_{i+j},
     which is a linear convolution of `a` (reversed) with `w`.  This
     convolution is also carried out with FFTs.
   - Adam updates are applied and the iterate is projected onto the
     probability simplex (Euclidean projection).

   The loop stops when the global time budget is exhausted or a
   per‑seed iteration limit is reached.  The best vector seen over all
   runs is kept.

4. A short random‑perturbation phase (multiplicative log‑normal noise)
   around the best vector refines the solution.

The returned sequence respects all constraints (non‑negative,
sum‑normalised, values ≤ 1000).  Its quality is measured by the
`evaluate_sequence` function, which is equivalent to
`J = 2·n·max(conv(a,a))` for a vector with sum = 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, L: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT of length `L`.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, L)
    conv = np.fft.irfft(A * A, n=L)
    return conv[:conv_len]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The method is a projected Adam optimiser on a
    smooth‑max approximation of the convolution peak, starting from several
    diverse initial points.  The best sequence found before the time budget
    expires is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n0 = old.size
            # Restrict size to a comfortable range for FFT speed & memory
            n = max(512, min(n0, 4096))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1                # length of a*a (full linear conv)
    L_fft = _next_pow2(3 * n - 2)       # length sufficient for all convolutions

    # ------------------------------------------------------------------
    # 2. Build a set of diverse starting points (all on the simplex)
    # ------------------------------------------------------------------
    init_vectors = []

    # uniform
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # previous best, if available
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        init_vectors.append(proj_simplex(v))

    # geometric decays
    for r in (0.9992, 0.9994, 0.9996):
        g = (1.0 - r) * (r ** np.arange(n, dtype=np.float64))
        init_vectors.append(proj_simplex(g))

    # decreasing linear ramp
    lin = np.arange(n, 0, -1, dtype=np.float64)
    init_vectors.append(proj_simplex(lin))

    # symmetric geometric (mass on both ends)
    r = 0.9995
    left = (1.0 - r) * (r ** np.arange(n // 2, dtype=np.float64))
    right = left[::-1]
    if n % 2 == 1:
        sym = np.concatenate([left, np.array([0.0]), right])
    else:
        sym = np.concatenate([left, right])
    init_vectors.append(proj_simplex(sym))

    # a few random Dirichlet draws
    for _ in range(3):
        init_vectors.append(rng.dirichlet(np.ones(n)))

    # ensure all are on the simplex
    init_vectors = [proj_simplex(v) for v in init_vectors]

    # ------------------------------------------------------------------
    # 3. Global best bookkeeping
    # ------------------------------------------------------------------
    best_vec = None
    best_val = float("inf")
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # leave a margin

    # ------------------------------------------------------------------
    # 4. Optimisation from each start point
    # ------------------------------------------------------------------
    for start_id, a0 in enumerate(init_vectors):
        # stop early if we are out of time
        if time.monotonic() >= deadline:
            break

        a = a0.copy()
        # Adam state
        m = np.zeros_like(a)
        vvec = np.zeros_like(a)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        t_iter = 0
        max_iters_per_start = 5000

        while time.monotonic() < deadline and t_iter < max_iters_per_start:
            # ----- convolution a * a (full linear) -----
            A_fft = np.fft.rfft(a, L_fft)
            conv_full = np.fft.irfft(A_fft * A_fft, n=L_fft)[:conv_len]

            max_conv = conv_full.max()
            # because `a` is already normalised (sum = 1)
            J_cur = 2.0 * n * max_conv  # exact objective for sum=1

            if J_cur < best_val - 1e-12:
                best_val = J_cur
                best_vec = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] New global best J = {best_val:.6f} "
                      f"(seed {start_id}, iter {t_iter})")

            # ----- smooth‑max weights -----
            # gamma grows slowly so the surrogate becomes tighter over time
            gamma = min(500.0, 0.5 + t_iter * 0.02)
            weight_raw = np.exp(gamma * conv_full)
            w_sum = weight_raw.sum()
            if w_sum == 0.0:
                w = np.full_like(weight_raw, 1.0 / conv_len)
            else:
                w = weight_raw / w_sum

            # ----- gradient of the smooth objective -----
            a_rev = a[::-1]
            A_rev_fft = np.fft.rfft(a_rev, L_fft)
            W_fft = np.fft.rfft(w, L_fft)
            conv_rev_full = np.fft.irfft(A_rev_fft * W_fft, n=L_fft)
            # grad_i = 2 * Σ_j a_j w_{i+j}
            grad = 2.0 * conv_rev_full[n - 1 : n - 1 + n]

            # ----- Adam update + simplex projection -----
            m = beta1 * m + (1.0 - beta1) * grad
            vvec = beta2 * vvec + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (t_iter + 1))
            v_hat = vvec / (1.0 - beta2 ** (t_iter + 1))
            lr = 0.05
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            a = proj_simplex(a)   # enforce non‑negativity and Σa = 1
            t_iter += 1

        # end while for a single start point

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random refinement around the best vector
    # ------------------------------------------------------------------
    if best_vec is None:
        # fall back to uniform if everything failed
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * np.max(np.convolve(best_vec, best_vec))

    fine_deadline = time.monotonic() + min(30.0, (deadline - start_time) * 0.05)
    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        max_conv = np.max(np.convolve(cand, cand))
        J_cand = 2.0 * n * max_conv / (cand.sum() ** 2)
        if J_cand < best_val - 1e-12:
            best_val = J_cand
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6. Clamp to allowed numeric range and return
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    if best_vec.sum() == 0.0:
        best_vec[:] = 1.0 / n

    return [float(v) for v in best_vec]
