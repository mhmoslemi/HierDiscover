    # objective function
    def compute_J(vec):

    else:
        best_vec = None
        best_J = np.inf

    # Timekeeping
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)  # leave margin

    stagnation_time = 30.0
    last_improve = start

    while time.monotonic() < deadline:
        # generate a starting point
        if best_vec is not None and rng.random() < 0.3:

            noise = rng.normal(scale=0.02, size=n)
            a = a + noise
            a = np.maximum(a, 0.0)
            a = proj_simplex(a)
        else:
            alpha = rng.choice([0.05, 0.1, 0.2, 0.5, 1.0])
            a = rng.dirichlet(np.ones(n) * alpha)

        # gradient descent loop
        eta = base_eta
        beta = base_beta
        momentum = np.zeros_like(a)
        mu = 0.9
        current_J = compute_J(a)
        for step in range(max_steps_per_run):
            if time.monotonic() >= deadline:
                break
            # compute convolution
            conv = _conv_full(a, fft_len, conv_len)
            M = conv.max()
            J = 2.0 * n * M
            # update global best
            if J < best_J - 1e-12:
                best_J = J
                best_vec = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start
                print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")

            # compute softmax weights
            w_raw = np.exp(beta * (conv - M))
            # optional normalization to avoid extremely small sums when beta big
            # but we keep as is
            grad = _weighted_gradient(a, w_raw, fft_len_corr, n)

            # momentum update and step
            momentum = mu * momentum - eta * grad
            a_new = a + momentum
            a_new = np.maximum(a_new, 0.0)
            a_new = proj_simplex(a_new)

            # evaluate new point
            conv_new = _conv_full(a_new, fft_len, conv_len)
            J_new = 2.0 * n * conv_new.max()

            # accept if improves
            if J_new < J:
                a = a_new
                # gradually increase step size and beta
                eta = min(eta * 1.05, 1.0)
                beta = min(beta * 1.02, 200.0)

                a = best_vec.copy()
                noise = rng.normal(scale=0.05, size=n)
                a = a + noise
                a = np.maximum(a, 0.0)
                a = proj_simplex(a)
                # reset momentum
                momentum = np.zeros_like(a)
                last_improve = time.monotonic()
                elapsed = last_improve - start
                print(f"[{elapsed:6.2f}s] Stagnation – global jitter")
    # after loop finalize
    if best_vec is None:
        # fallback uniform
        best_vec = np.full(n, 1.0 / n)
        best_J = compute_J(best_vec)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # optional sanity check
    # try evaluate_sequence(best_vec.tolist()) (allowed)
    # but not needed

"""
Fast‐gradient / parametric search for a step function with low
maximal self‑convolution.  The objective is scale‑invariant, so we keep
the sequence on the probability simplex (∑a_i = 1).  The search combines
three ideas:

1.  A small analytical family a_i ∝ (i+1)^(-p) (including the known
    p = 0.5 “square‑root” shape) and a_i ∝ sqrt(i+1)-sqrt(i).  Random
    values of the exponent give cheap candidates.

2.  A simple projected gradient descent on a smooth surrogate of
    max(conv) (log‑sum‑exp) with momentum.  The gradient is obtained by
    correlating the current vector with a weight vector that concentrates
    on the largest convolution entries.

3.  Occasional random “jitter” restarts to keep diversity.

All heavy work uses real FFTs, which are O(n log n) and fast enough to
run thousands of trials in the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute the gradient of  Σ_k weight_k·(a*a)_k  w.r.t. a.
    The result is 2·corr(weight, a) (real‑valued).
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Objective helpers (scale‑invariant, sum = 1)
# ----------------------------------------------------------------------


def compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return the value 2·n·max(a*a) for a vector that already sums to 1."""
    conv = _conv_full(vec, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def log_sum_exp_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights that approximate the sub‑gradient of max(conv).
    The weights are non‑negative and sum to 1.
    """
    M = conv.max()
    w_raw = np.exp(beta * (conv - M))
    s = w_raw.sum()
    if s == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return w_raw / s


# ----------------------------------------------------------------------
# Parametric families of candidate sequences
# ----------------------------------------------------------------------


def shape_power(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)  (p ≥ 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.power(idx, p)
    return proj_simplex(a)


def shape_sqrt_diff(n: int) -> np.ndarray:
    """
    a_i ∝ sqrt(i+1) - sqrt(i)  (the discrete derivative of sqrt).
    This automatically satisfies ∑ a_i = sqrt(n), so we normalise.
    """
    idx = np.arange(n, dtype=np.float64)
    a = np.sqrt(idx + 1.0) - np.sqrt(idx)
    return proj_simplex(a)


def shape_mixed(n: int, p: float, mix: float) -> np.ndarray:
    """
    Convex combination of the power‑law shape and the uniform vector:
    mix ∈ [0,1] gives the weight of the power‑law component.
    """
    power = shape_power(n, p)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    a = mix * power + (1.0 - mix) * uniform
    return proj_simplex(a)

# Main optimisation routine
# ----------------------------------------------------------------------

            jitter = rng.dirichlet(np.ones(n) * 0.05)
            cand = jitter
            # reset momentum / step‑size for the next round
            momentum = np.zeros_like(cand)
            eta = 0.15
            beta = 5.0
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Stagnation – jittered restart")

    # ----------------------------------------------------------------------
    # final clean‑up: clip, renormalise and convert to python list
    # ----------------------------------------------------------------------
    if best_vec is None:
        # fallback – uniform distribution (never happens in practice)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # return a plain Python list as required by the outer harness
    return [float(v) for v in best_vec]
