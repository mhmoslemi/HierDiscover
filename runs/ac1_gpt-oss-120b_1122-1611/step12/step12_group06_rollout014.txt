"""
Optimiser for the non‑negative step‑function problem.

The routine builds a probability vector on the simplex (by means of a
softmax over logits) and minimises the objective

    J(a) = 2·n·max( a * a )          ( * = linear convolution )

using a hybrid of:
* an annealed soft‑max (smooth max) loss,
* a hard‑max sub‑gradient when the convolution peak is flat,
* Adam updates on the logits,
* periodic stochastic restarts if no improvement is seen.

A simple local stochastic refinement is performed at the end.
The whole search respects the external 1000 s budget and returns the best
sequence discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute smooth soft‑max weights over the entries of `conv`.
    The subtraction of conv.max() guarantees numerical stability.
    """
    c_max = conv.max()
    # exponent ≤ 0, no overflow
    exp_part = np.exp(beta * (conv - c_max))
    total = exp_part.sum()
    # safeguard against pathological case
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / total


def _grad_with_weights(a: np.ndarray, weights: np.ndarray,
                      fft_len_corr: int, corr_len: int) -> np.ndarray:
    """
    Gradient of  Σ_i w_i·(a*a)_i  w.r.t. a.
    Uses FFT‑based correlation.  The factor 2 accounts for the two
    symmetric contributions in the convolution derivative.
    """
    n = a.size
    # FFT of a
    A = np.fft.rfft(a, fft_len_corr)
    # FFT of the weight vector (same length as the full convolution)
    W = np.fft.rfft(weights, fft_len_corr)
    # correlation = irfft( A * conj(W) )
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len_corr)[:corr_len]

    # The slice that corresponds to the gradient entries:
    # start = len(weights) - 1 = (2n-2)
    start = 2 * n - 2
    grad = 2.0 * corr_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds and
    returns the best sequence discovered.

    The optimisation works on the probability simplex via a soft‑max
    parametrisation and combines a smoothed max loss with a hard‑max
    sub‑gradient.  Adam updates with occasional stochastic restarts are used.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size
    # --------------------------------------------------------------
    # Prefer a power‑of‑two length for fast FFTs, limited to ≤8192.
    if "height_sequence_1" in globals():
        try:
            base_len = len(height_sequence_1)
            n = int(2 ** np.ceil(np.log2(base_len)))
            n = max(1024, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    corr_len = 3 * n - 2
    fft_len = _next_pow2(max(conv_len, corr_len))

    # --------------------------------------------------------------
    # 2. Build a diverse set of initial candidates (all on the simplex)
    # --------------------------------------------------------------
    init_candidates = []

    # Uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Geometric decays (several rates)
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_candidates.append(v)

    # Linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_candidates.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_candidates.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_candidates.append(v)

    # Dirichlet draws with different concentrations
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_candidates.append(rng.dirichlet(np.ones(n) * alpha))

    # If a previous construction exists, use it (cropped / padded)
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        old = np.clip(old, 0.0, None)
        if old.size > n:
            old = old[:n]
        elif old.size < n:
            pad = np.full(n - old.size, 1.0 / n)
            old = np.concatenate([old, pad])
        s = old.sum()
        if s == 0.0:
            old = np.full(n, 1.0 / n)
        else:
            old /= s
        init_candidates.append(old)

    # --------------------------------------------------------------
    # 3. Pick the best starter by the exact objective J = 2·n·max(conv)
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        conv = _conv_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop (Adam on logits + hybrid gradient)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # leave a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    eps = 1e-12
    logits = np.log(np.maximum(best_a, eps))
    m = np.zeros_like(logits)
    v_m = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8
    lr0 = 0.8

    # Temperature schedule for the soft‑max surrogate
    beta_min = 5.0
    beta_max = 5000.0  # fairly high – we anneal to a near‑hard max

    iteration = 0
    last_improve = start_time

    # keep the overall best sequence found
    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # 4.1 Current point on the simplex
        # ----------------------------------------------------------
        a = _softmax(logits)

        # ----------------------------------------------------------
        # 4.2 Exact objective
        # ----------------------------------------------------------
        conv = _conv_fft(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # 4.3 Record improvement
        # ----------------------------------------------------------
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----------------------------------------------------------
        # 4.4 Stagnation handling (restart if no improvement for a while)
        # ----------------------------------------------------------
        if now - last_improve > 15.0:
            # strong stochastic jitter on the logits; reset Adam moments
            jitter_sigma = 0.25
            logits = np.log(np.maximum(global_best_a, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v_m.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

        # ----------------------------------------------------------
        # 4.5 Temperature for the smooth max (linear annealing)
        # ----------------------------------------------------------
        progress = (now - start_time) / (deadline - start_time)
        beta = beta_min + (beta_max - beta_min) * progress

        # ----------------------------------------------------------
        # 4.6 Choose gradient source:
        #     * if the peak is narrow – use soft‑max weights (smooth)
        #     * if the peak is flat (many equal maxima) – use a hard mask
        # ----------------------------------------------------------
        # Identify indices that are (practically) maximal
        tolerance = 1e-12
        max_mask = conv >= max_conv - tolerance
        n_max = max_mask.sum()
        if n_max > 10:
            # flat top – use a hard uniform mask over all maximal positions
            weights = max_mask.astype(np.float64) / float(n_max)
        else:
            # narrow peak – smooth soft‑max
            weights = _softmax_weights(conv, beta)

        # ----------------------------------------------------------
        # 4.7 Gradient w.r.t. a (smooth or hard)
        # ----------------------------------------------------------
        grad_a = _grad_with_weights(a, weights, fft_len, corr_len)

        # ----------------------------------------------------------
        # 4.8 Chain‑rule to logits space (soft‑max Jacobian)
        # ----------------------------------------------------------
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)

        # ----------------------------------------------------------
        # 4.9 Adam update on logits
        # ----------------------------------------------------------
        iteration += 1
        # Simple 1/√t decay for stability
        lr = lr0 / np.sqrt(iteration)

        m = beta1 * m + (1.0 - beta1) * grad_z
        v_m = beta2 * v_m + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_m / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # keep logits in a numeric range that avoids overflow
        logits = np.clip(logits, -500.0, 500.0)

        # occasional cheap progress report
        if iteration % 2000 == 0:
            print(f"[{elapsed:6.2f}s] iter {iteration}, cur J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 5. Final local stochastic refinement (few seconds)
    # --------------------------------------------------------------
    time_left = deadline - time.monotonic()
    if time_left > 2.0:
        # we run a tiny random walk around the best point
        a = global_best_a.copy()
        best_local_J = global_best_J
        max_delta = 0.02  # step size relative to probability mass
        iters = 0
        while time.monotonic() < deadline:
            i, j = rng.choice(n, size=2, replace=False)
            delta = (rng.random() - 0.5) * max_delta
            if a[i] + delta < 0.0 or a[j] - delta < 0.0:
                continue
            a[i] += delta
            a[j] -= delta
            a /= a.sum()
            conv = _conv_fft(a, fft_len, conv_len)
            Jcand = float(2.0 * n * conv.max())
            if Jcand < best_local_J:
                best_local_J = Jcand
                global_best_a = a.copy()
                global_best_J = Jcand
                print(f"[{time.monotonic() - start_time:6.2f}s] Local refinement improved J = {global_best_J:.6f}")
                # reset the walk to the improved point
                a = global_best_a.copy()
            iters += 1

    # --------------------------------------------------------------
    # 6. Return the best sequence (clipped, renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    final_J = float(2.0 * n * np.convolve(best_vec, best_vec).max())
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
```
