"""
Hybrid projected sub‑gradient descent with occasional random jitter.

The objective for a probability vector `a` (sum = 1) is
    J(a) = 2·n·max(a * a)                     (a * a = full linear convolution)
which is invariant under scaling of `a`.  The algorithm
   1. Starts from the best known construction (global variable
      `height_sequence_1`) or from a uniform distribution.
   2. Repeatedly computes the index/indices where the convolution
      attains its maximum and forms the sub‑gradient of the
      max‑function:
          ∇J ≈ 2·shift_i(a)  (sum of reversed prefixes)
      for each active index `i`.
   3. Performs a short back‑tracking line‑search on the projected
      update  a ← Π_simplex(a – η·∇J)  and adapts the step size.
   4. If no improvement happens for a while a small random
      multiplicative perturbation is injected to escape plateaus.
   5. After the time budget is almost exhausted a fine‑grained
      random‑jitter search around the best found vector is executed.
The whole routine respects the 1000 s budget and returns a list of
non‑negative heights ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returns a vector w with w_i >= 0 and sum(w) = 1.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all equal case
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Convolution utilities (FFT based, real input)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def objective_from_vec(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute J = 2·n·max(conv(a,a)), assuming Σ a = 1.
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution functional
# ----------------------------------------------------------------------


def subgrad_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Compute a (sub‑)gradient of the function  max(conv(a,a))
    with respect to `a`.  The gradient is
        g = (2 / |I|) * sum_{i∈I} shift_i(a)
    where `I` are the indices at which the convolution attains its
    maximum, and `shift_i(a)[j] = a[i‑j]` if the index is in range,
    otherwise 0.
    """
    max_val = conv.max()
    # treat values within a tiny tolerance as active
    tol = 1e-12
    active = np.where(conv >= max_val - tol)[0]
    grad = np.zeros_like(a)
    for i in active:
        # reversed prefix a[i::-1] has length i+1
        grad[: i + 1] += a[i::-1]
    grad *= 2.0 / max(1, len(active))
    return grad

# Main search routine
# ----------------------------------------------------------------------
