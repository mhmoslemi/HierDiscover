def propose_candidate(seed=..., budget_s=1000.0, **kwargs):
    import time, numpy as np, scipy.sparse as sp, scipy.optimize as opt

    rng = np.random.default_rng(seed)

    # Determine n:
    if 'height_sequence_1' in globals():
        init_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = min(max(int(len(init_seq)), 256), 2048)   # chosen limit
    else:
        n = 1024
    n = max(256, min(n, 2048))

    # Build pattern matrix
    row_idxs = np.arange(n)[:,None] + np.arange(n)[None,:]  # shape (n,n)
    rows = row_idxs.ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.ones_like(rows, dtype=np.float64)
    pattern = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

    # function conv and J
    def compute_conv(a):
        fft_len = 1 << ((2*n-2).bit_length())
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:2*n-1]
        return conv
    def compute_J(a):
        conv = compute_conv(a)
        max_conv = conv.max()
        s = a.sum()
        if s < 1e-12:
            return np.inf
        return float(2.0 * n * max_conv / (s * s))

    # Build init candidates list....
    # We'll replicate same set as previous code but adapt to n.

    # Choose best initial.
    best_a = None; best_J = np.inf
    for cand in init_candidates:
        Jc = compute_J(cand)
        if Jc < best_J:
            best_J = Jc
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # Global best
    global_best_a = best_a.copy()
    global_best_J = best_J
    # iteration counter
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)  # safe margin

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        a = best_a
        # compute conv and M and J
        conv = compute_conv(a)
        M = conv.max()
        cur_J = float(2.0 * n * M / (a.sum() ** 2))
        # Solve LP
        # Build A_ub = pattern @ sp.diags(a)
        A_ub = pattern @ sp.diags(a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
        # Solve
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs',
                          options={'presolve': True, 'time_limit': max(0.1, (deadline - time.monotonic()))})
        if not res.success:
            # LP failed, break
            break
        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()
        if sum_g <= sum_a + 1e-12:
            # No improvement, break
            break

        # mixing strategy
        t = 0.5
        # compute new vector
        new_a = (1.0 - t) * a + t * g
        new_J = compute_J(new_a)
        if new_J < global_best_J - 1e-12:
            global_best_J = new_J
            global_best_a = new_a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] Improved J = {global_best_J:.6f}")

        # accept new if better than current
        if new_J < cur_J * (1 - 1e-9):
            best_a = new_a
        else:
            # try smaller step
            t = 0.1
            new_a = (1.0 - t) * a + t * g
            new_J = compute_J(new_a)
            if new_J < cur_J:
                best_a = new_a
            else:
                # no progress
                # maybe break
                break

    # final best: global_best_a
    final_seq = np.clip(global_best_a, 0.0, 1000.0)
    total = final_seq.sum()
    if total == 0.0:
        final_seq = np.full_like(final_seq, 1.0 / n)
    else:
        final_seq = final_seq / total  # optional unify sum to 1

    print(f"Finished. Best J = {global_best_J:.6f}")
    return [float(v) for v in final_seq.tolist()]

"""
Linear‑programming “sum‑boost” search.

The procedure starts from several diverse probability vectors and keeps the one
with the lowest true objective J = 2·n·max(conv(a,a))/sum(a)^2.
From a current vector f it computes the peak M = max(conv(f,f)) and solves the
linear program

    maximise   Σ g_i
    subject to  (f * g)_k ≤ M   for all k = 0 … 2n‑2
                g_i ≥ 0

Since (f*g)_k is linear in g this LP is easy for the HiGHS solver.
The LP gives a vector g that can carry more total mass than f without raising
the peak M – mixing f and g and renormalising therefore reduces the ratio
M/sum(f)^2 and improves J.  The process is repeated until the time budget is
exhausted (or no further progress is observed).  A short jitter / reset
mechanism is added to escape possible stalls.

The routine returns the best sequence (list of non‑negative floats) found in
the allotted time.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Full linear convolution of a with itself (no FFT – fast enough for modest n)."""
    return np.convolve(a, a)  # length = 2*len(a)-1


def _objective(a: np.ndarray) -> float:
    """Exact evaluation of the objective J (ignores clipping performed in evaluate_sequence)."""
    n = a.shape[0]
    conv = _conv_full(a)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (s * s)


def _build_ones_pattern(n: int) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n sparse matrix whose non‑zero pattern contains a single
    '1' in each column for every row where that column participates in the
    convolution (i.e. rows i … i+n‑1 for column i).
    This matrix is later multiplied with a diagonal matrix of the current
    probabilities to obtain the actual linear constraints.
    """
    rows = (np.arange(n)[:, None] + np.arange(n)[None, :]).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.ones_like(rows, dtype=np.float64)
    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.

    The algorithm:
      1. Choose a problem size n (≈ 1024–2048) and generate a small portfolio of
         initial probability vectors (uniform, geometric, V‑shaped, Gaussian, etc.).
      2. Pick the best initial vector according to the true objective J.
      3. Repeatedly:
           * compute the current peak M of the self‑convolution,
           * solve the LP that maximises the total mass of a new vector g under
             the constraints (f * g)_k ≤ M,
           * mix f and g (convex combination), renormalise to unit sum,
           * keep the best vector ever seen.
         A jitter is applied if no improvement occurs for 30 s.
      4. Return the best sequence found (values are already non‑negative and
         comfortably below the 1000‑clip limit).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    # Prefer the length of a previously discovered construction, but stay
    # within a comfortable range for the LP solver.
    if "height_sequence_1" in globals():
        try:
            n0 = len(height_sequence_1)
            n = max(256, min(n0, 2048))
        except Exception:
            n = 1024
    else:
        n = 1024

    # ------------------------------------------------------------------
    # 2. Helper objects used throughout the run
    # ------------------------------------------------------------------
    pattern_ones = _build_ones_pattern(n)          # shape (2n‑1) × n

    # ------------------------------------------------------------------
    # 3. Build a diverse set of starting probability vectors
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)

    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n))

    # V‑shaped (large at edges)
    v = 1.0 / ((idx + 1) * (n - idx))
    v = np.clip(v, 0, None)
    v /= v.sum()
    init_candidates.append(v)

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        g = r ** idx
        g /= g.sum()
        init_candidates.append(g)

    # linear ramps
    ramp = (n - idx) / n
    ramp /= ramp.sum()
    init_candidates.append(ramp)
    ramp = (idx + 1) / n
    ramp /= ramp.sum()
    init_candidates.append(ramp)

    # Gaussian bumps
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        g = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        g /= g.sum()
        init_candidates.append(g)

    # Dirichlet draws (moderately and strongly concentrated)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(4):
            init_candidates.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size > n:
                prev = prev[:n]
            else:
                prev = np.concatenate([prev, np.full(n - prev.size, 1.0 / n)])
        total = prev.sum()
        if total == 0.0:
            prev = np.full(n, 1.0 / n)
        else:
            prev = prev / total
        init_candidates.append(prev)

    # ------------------------------------------------------------------
    # 4. Pick the best seed according to the true hard objective
    # ------------------------------------------------------------------
    best_a = None
    best_J = np.inf
    start_time = time.monotonic()
    for cand in init_candidates:
        Jc = _objective(cand)
        if Jc < best_J:
            best_J = Jc
            best_a = cand.copy()
    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Main LP‑based improvement loop
    # ------------------------------------------------------------------
    global_best_a = best_a.copy()
    global_best_J = best_J

    # parameters
    max_iters = 30                     # safeguard – the LP is the heavy part
    stagnation_time = 30.0             # seconds before applying jitter
    last_improve = time.monotonic()
    t_mix = 0.5                        # default mixing weight

    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    for it in range(max_iters):
        now = time.monotonic()
        if now >= deadline:
            break

        a = best_a
        conv = _conv_full(a)
        M = conv.max()
        cur_J = _objective(a)

        # ------------------------------------------------------------------
        # Solve the LP: maximise Σ g_i   subject to   (a * g)_k ≤ M
        # ------------------------------------------------------------------
        # Constraint matrix: pattern_ones @ diag(a)
        A_ub = pattern_ones @ sp.diags(a)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)           # minimise –Σ g_i  ⇔ maximise Σ g_i
        # Linprog via HiGHS (sparse support)
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=(0, None), method='highs',
                          options={'presolve': True})
        if not res.success:
            print(f"[{now - start_time:6.2f}s] LP failed (status {res.status}); terminating.")
            break

        g = res.x
        sum_g = g.sum()
        sum_a = a.sum()

        # If the LP did not increase total mass, we are at a (local) optimum.
        if sum_g <= sum_a + 1e-12:
            print(f"[{now - start_time:6.2f}s] No mass increase (Σg={sum_g:.6f} ≤ Σa={sum_a:.6f}); stopping.")
            break

        # ------------------------------------------------------------------
        # Mix the old vector with the LP solution and renormalise.
        # ------------------------------------------------------------------
        mixed = (1.0 - t_mix) * a + t_mix * g
        mixed /= mixed.sum()           # keep total mass = 1  (J invariant under scaling)

        new_J = _objective(mixed)

        if new_J + 1e-12 < global_best_J:
            global_best_J = new_J
            global_best_a = mixed.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] New best J = {global_best_J:.6f}")

        # Accept the mix if it improves the local objective; otherwise try a smaller step.
        if new_J < cur_J:
            best_a = mixed
        else:
            # try a more conservative blend
            t_small = 0.1
            mixed_small = (1.0 - t_small) * a + t_small * g
            mixed_small /= mixed_small.sum()
            new_J_small = _objective(mixed_small)
            if new_J_small < cur_J:
                best_a = mixed_small
                # update best_J locally (optional)
            else:
                # No improvement – apply jitter after a while
                if now - last_improve > stagnation_time:
                    sigma = 0.15
                    jitter = rng.normal(0.0, sigma, size=n)
                    best_a = np.exp(np.log(np.maximum(best_a, 1e-14)) + jitter)
                    best_a = np.clip(best_a, 0.0, None)
                    best_a /= best_a.sum()
                    last_improve = now
                    print(f"[{now - start_time:6.2f}s] Jitter applied to escape stagnation.")
                else:
                    # Unable to progress; exit early.
                    print(f"[{now - start_time:6.2f}s] Stagnated without improvement; terminating.")
                    break

    # ------------------------------------------------------------------
    # 6. Return the best sequence discovered
    # ------------------------------------------------------------------
    final_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = final_vec.sum()
    if total == 0.0:
        final_vec = np.full_like(final_vec, 1.0 / n)
    else:
        final_vec = final_vec / total           # normalise (optional but keeps numbers modest)

    print(f"Finished. Best J = {global_best_J:.6f}")
    return [float(v) for v in final_vec.tolist()]
