"""
Propose a sequence of non‑negative heights that (hopefully) beats the current
best bound (≈1.513).  The optimisation consists of three stages:

1.  Adam on the logits with a smooth‑max approximation (fast exploration).
2.  Exponentiated‑gradient descent on the *hard* max of the self‑convolution
    (directly minimises the true objective).
3.  A short simulated‑annealing style perturbation phase to escape possible
    plateaus.

All stages share a common FFT‑based convolution implementation which runs
in O(n log n) time.  The routine respects the 1000 s budget and returns the
best sequence seen.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Result has length conv_len = 2*len(a)-1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights for a smooth approximation of max(conv).
    Returns a vector of length len(conv) that sums to 1.
    """
    m = conv.max()
    shifted = beta * (conv - m)
    # Clip to avoid overflow/underflow in the exp.
    shifted = np.clip(shifted, -700, 0)
    exp_shifted = np.exp(shifted)
    return exp_shifted / exp_shifted.sum()


def _grad_smooth_max(a: np.ndarray, w: np.ndarray, grad_fft_len: int) -> np.ndarray:
    """
    Gradient of smooth_max w.r.t. a.
    Implements 2 * (w * rev(a))[i] via FFT.
    """
    rev_a = a[::-1]
    W = np.fft.rfft(w, grad_fft_len)
    A_rev = np.fft.rfft(rev_a, grad_fft_len)
    full = np.fft.irfft(W * A_rev, n=grad_fft_len)
    n = a.shape[0]
    return 2.0 * full[n - 1:n - 1 + n]


# ----------------------------------------------------------------------
# Stage 1 – Adam on logits with a smooth max (exploratory)
# ----------------------------------------------------------------------


def _adam_optimize(init_a: np.ndarray,
                   n: int,
                   conv_len: int,
                   fft_len: int,
                   grad_fft_len: int,
                   time_limit: float,
                   rng: np.random.Generator,
                   global_best_updater) -> np.ndarray:
    """
    Adam on the logits (soft‑max parametrisation) using a smooth‐max
    approximation.  Returns the best vector found in the allotted time.
    """
    # hyper‑parameters
    beta = 2.0
    beta_max = 500.0
    beta_inc = 1.015  # accelerated schedule
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # initialise Adam state on logits
    eps = 1e-14
    logits = np.log(np.maximum(init_a, eps))
    m1 = np.zeros_like(logits)
    v1 = np.zeros_like(logits)

    best_local = init_a.copy()
    best_J_local = 2.0 * n * _conv_full(init_a, fft_len, conv_len).max()

    start = time.monotonic()
    it = 0
    last_improve = start

    while time.monotonic() - start < time_limit:
        # probabilities from logits
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()

        # hard J (for record keeping)
        conv = _conv_full(a, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())

        # global best handling (callback)
        global_best_updater(cur_J, a, time.monotonic())

        # local best handling
        if cur_J < best_J_local - 1e-12:
            best_J_local = cur_J
            best_local = a.copy()
            last_improve = time.monotonic()

        # stagnation → jitter
        now = time.monotonic()
        if now - last_improve > 15.0:
            jitter_sigma = 0.35
            logits = np.log(np.maximum(best_local, eps))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m1.fill(0.0)
            v1.fill(0.0)
            last_improve = now

        # smooth‑max weights
        w = _smooth_max_weights(conv, beta)

        # gradient w.r.t. a
        grad_a = _grad_smooth_max(a, w, grad_fft_len)

        # chain rule to logits (soft‑max Jacobian)
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)

        # Adam update on logits
        it += 1
        lr = lr0 / np.sqrt(it)            # simple 1/√t decay
        m1 = beta1 * m1 + (1.0 - beta1) * grad_z
        v1 = beta2 * v1 + (1.0 - beta2) * (grad_z ** 2)

        m1_hat = m1 / (1.0 - beta1 ** it)
        v1_hat = v1 / (1.0 - beta2 ** it)

        logits -= lr * m1_hat / (np.sqrt(v1_hat) + adam_eps)
        logits = np.clip(logits, -500.0, 500.0)

        # increase smoothing
        beta = min(beta_max, beta * beta_inc)

    return best_local


# ----------------------------------------------------------------------
# Stage 2 – Exponentiated gradient on the *hard* max (direct optimisation)
# ----------------------------------------------------------------------


def _expgrad_optimize(init_a: np.ndarray,
                      n: int,
                      conv_len: int,
                      fft_len: int,
                      grad_fft_len: int,
                      time_limit: float,
                      rng: np.random.Generator,
                      global_best_updater) -> np.ndarray:
    """
    Exponentiated‑gradient descent on the true max(convolution) objective.
    Uses a one‑hot weight vector (hard max) for the gradient.
    """
    a = init_a.copy()
    best_local = a.copy()
    best_J_local = 2.0 * n * _conv_full(a, fft_len, conv_len).max()

    start = time.monotonic()
    it = 0
    last_improve = start
    eta0 = 0.60
    decay = 1e-4
    beta = 2.0          # start a very soft smoothing, quickly switch to hard
    beta_inc = 1.02
    beta_max = 2000.0

    while time.monotonic() - start < time_limit:
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # record globally
        global_best_updater(cur_J, a, time.monotonic())

        # record locally
        if cur_J < best_J_local - 1e-12:
            best_J_local = cur_J
            best_local = a.copy()
            last_improve = time.monotonic()

        # identify the (soft) weight vector for the max
        w = _smooth_max_weights(conv, beta)   # gradually moves to one‑hot

        # gradient w.r.t. a (hard or soft according to beta)
        grad_a = _grad_smooth_max(a, w, grad_fft_len)

        # exponentiated‑gradient (mirror descent on the simplex)
        it += 1
        eta = eta0 / (1.0 + decay * it)
        # protect against overflow/under‑flow in the exponent
        step = -eta * grad_a
        step = np.clip(step, -50.0, 50.0)
        a = a * np.exp(step)

        # renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # increase β so that we approach the true max
        beta = min(beta_max, beta * beta_inc)

        # stagnation → large jitter
        now = time.monotonic()
        if now - last_improve > 20.0:
            jitter_sigma = 0.45
            logits = np.log(np.maximum(a, 1e-14))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            a = np.exp(logits - np.max(logits))
            a = a / a.sum()
            last_improve = now

    return best_local


# ----------------------------------------------------------------------
# Stage 3 – Simple simulated‑annealing on the simplex (optional fine‑tune)
# ----------------------------------------------------------------------


def _anneal_optimize(init_a: np.ndarray,
                     n: int,
                     conv_len: int,
                     fft_len: int,
                     time_limit: float,
                     rng: np.random.Generator,
                     global_best_updater) -> np.ndarray:
    """
    Very light‑weight simulated annealing: repeatedly move a tiny amount
    of mass between two random coordinates and accept if the objective
    improves or with a temperature‑dependent probability.
    """
    a = init_a.copy()
    best_local = a.copy()
    best_J_local = 2.0 * n * _conv_full(a, fft_len, conv_len).max()

    start = time.monotonic()
    temperature = 1.0
    cooling = 0.995
    min_temp = 1e-4

    while time.monotonic() - start < time_limit:
        # propose a tiny transfer
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.uniform(0.0, min(a[i], 0.02))   # move at most 2 % of a[i]
        if delta == 0.0:
            continue
        a_prop = a.copy()
        a_prop[i] -= delta
        a_prop[j] += delta

        # linearly normalise (sum is preserved by construction)
        conv = _conv_full(a_prop, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())

        # accept?
        accept = False
        if cur_J < best_J_local:
            accept = True
        else:
            prob = np.exp(-(cur_J - best_J_local) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_prop
            if cur_J < best_J_local - 1e-12:
                best_J_local = cur_J
                best_local = a.copy()
                global_best_updater(cur_J, a, time.monotonic())

        # cooling schedule
        temperature = max(min_temp, temperature * cooling)

    return best_local


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Find a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size n (respect any previous solution)
    # ------------------------------------------------------------------
    try:
        n0 = len(globals()["height_sequence_1"])
        n = max(256, min(n0, 8192))
    except Exception:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                 # for a*a
    grad_fft_len = _next_pow2(3 * n - 2)           # for w * rev(a)

    # ------------------------------------------------------------------
    # 2. Helper to keep a global best record (thread‑safe)
    # ------------------------------------------------------------------
    global_best_a = None
    global_best_J = np.inf

    def _update_global(best_J, best_a, now):
        nonlocal global_best_a, global_best_J
        if best_J < global_best_J - 1e-12:
            global_best_J = best_J
            global_best_a = best_a.copy()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] GLOBAL NEW BEST J = {global_best_J:.6f}")

    # ------------------------------------------------------------------
    # 3. Build a diverse portfolio of starting vectors
    # ------------------------------------------------------------------
    start_vecs = []

    # uniform
    start_vecs.append(np.full(n, 1.0 / n, dtype=np.float64))

    # V‑shaped (large at edges)
    idx = np.arange(n, dtype=np.float64)
    v = 1.0 / ((idx + 1) * (n - idx))
    v = np.clip(v, 0, None)
    v /= v.sum()
    start_vecs.append(v)

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        g = r ** idx
        g /= g.sum()
        start_vecs.append(g)

    # linear ramps
    ramp = (n - idx) / n
    ramp /= ramp.sum()
    start_vecs.append(ramp)
    ramp = (idx + 1) / n
    ramp /= ramp.sum()
    start_vecs.append(ramp)

    # Gaussian bumps
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        g = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        g /= g.sum()
        start_vecs.append(g)

    # Dirichlet draws (varied concentration)
    for alpha in (0.2, 0.4, 0.7, 1.0):
        for _ in range(3):
            start_vecs.append(rng.dirichlet(np.full(n, alpha)))

    # previous construction, if any
    if "height_sequence_1" in globals():
        v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total > 0.0:
            v /= total
        else:
            v = np.full(n, 1.0 / n)
        start_vecs.append(v)

    # shuffle the portfolio to avoid deterministic ordering
    rng.shuffle(start_vecs)

    # ------------------------------------------------------------------
    # 4. Main optimisation schedule
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # Allocate time roughly: 40 % Adam, 45 % exponentiated‑gradient,
    # 15 % annealing (if any time remains)
    total_budget = deadline - start_time
    adam_budget = total_budget * 0.40
    expgrad_budget = total_budget * 0.45
    anneal_budget = total_budget - adam_budget - expgrad_budget

    # ---- Phase 1 : Adam (soft max) -----------------------------------
    remaining = adam_budget
    used = 0.0
    for vec in start_vecs:
        if time.monotonic() >= deadline:
            break
        # split remaining time evenly among the available start vectors
        per_vec = max(0.5, remaining / max(1, len(start_vecs)))
        best_vec = _adam_optimize(vec, n, conv_len, fft_len,
                                  grad_fft_len, per_vec, rng, _update_global)
        used += per_vec
        remaining = adam_budget - used

    # ---- Phase 2 : Exponentiated‑gradient (hard max) -----------------
    remaining = expgrad_budget
    used = 0.0
    # start from the *current* global best if we have one, otherwise uniform
    init_vecs = [global_best_a if global_best_a is not None else np.full(n, 1.0 / n)]
    # add a few random Dirichlet seeds for diversity
    for _ in range(4):
        init_vecs.append(rng.dirichlet(np.full(n, 0.5)))
    for vec in init_vecs:
        if time.monotonic() >= deadline:
            break
        per_vec = max(0.5, remaining / max(1, len(init_vecs)))
        best_vec = _expgrad_optimize(vec, n, conv_len, fft_len,
                                     grad_fft_len, per_vec, rng, _update_global)
        used += per_vec
        remaining = expgrad_budget - used

    # ---- Phase 3 : Tiny annealing pass (optional) --------------------
    if time.monotonic() < deadline and anneal_budget > 1.0:
        init = global_best_a if global_best_a is not None else np.full(n, 1.0 / n)
        _ = _anneal_optimize(init, n, conv_len, fft_len,
                             anneal_budget, rng, _update_global)

    # ------------------------------------------------------------------
    # 5. Return the best sequence discovered (ensure validity)
    # ------------------------------------------------------------------
    if global_best_a is None:
        # fallback – should never happen but keep it safe
        result = np.full(n, 1.0 / n)
    else:
        result = np.clip(global_best_a, 0.0, 1000.0)

    # normalise to keep the sum well‑behaved (scale is irrelevant)
    s = result.sum()
    if s == 0.0:
        result = np.full_like(result, 1.0 / n)
    else:
        result /= s

    print(f"Finished. Best J = {global_best_J:.6f}")
    return [float(v) for v in result]
