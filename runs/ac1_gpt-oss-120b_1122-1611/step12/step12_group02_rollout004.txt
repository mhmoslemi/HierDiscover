"""
Gradient‑descent on a smoothed max‑convolution objective.

The evaluation function is  J(a)=2·n·max(a*a) / (∑a)².
Because J is homogeneous we can enforce ∑a=1 and minimise 2·n·max(a*a).

We optimise a soft‑max (log‑its) parametrisation a_i = exp(u_i)/∑exp(u) by
minimising a differentiable approximation of the max:
    fα(a)=2·n·∑_k w_k·conv_k ,    w_k = softmax(α·conv_k).

For small α the objective is smooth; α is increased gradually so that the
gradient concentrates on the true maximal convolution entry.
The gradient of ∑ w_k·conv_k w.r.t. a is 2·corr(w,a) and is obtained
efficiently with an FFT.  An Adam optimiser updates the logits u.
If progress stalls for >30 s a random restart is performed.
At the end a simple multiplicative “polish” step (as in the baseline) is
applied and the final vector is projected onto the probability simplex.

The routine respects the 1000 s wall‑clock limit and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute the gradient of   Σ_k weight_k·conv_k   w.r.t. a.
    Returns 2·corr(weight, a) (length n vector).
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # -------------------------
    # 1. Choose problem size n
    # -------------------------
    # Prefer the length of the previously found construction,
    # but allow a larger power‑of‑two for FFT efficiency.
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base_len = base.size
        except Exception:
            base = None
            base_len = 0
    else:
        base = None
        base_len = 0

    # Target length – a power of two not smaller than 2048
    # and not larger than 8192 (still cheap for the O(n log n) work).
    target_len = max(2048, min(8192, base_len if base_len > 0 else 4096))
    n = 1 << ((target_len - 1).bit_length())   # round up to next power of 2

    # -------------------------------------------------
    # 2. Initialise probability vector a (sum = 1)
    # -------------------------------------------------
    if base is not None and base_len > 0:
        # pad / truncate the existing construction to length n
        if base_len < n:
            v = np.concatenate([base, np.zeros(n - base_len)])
        else:
            v = base[:n].copy()
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = np.full(n, 1.0 / n)
    else:
        v = np.full(n, 1.0 / n)

    # Add a tiny random perturbation to avoid exact symmetries
    v = v + 1e-6 * rng.random(n)
    v = v / v.sum()

    # Work in log‑it space: u_i = log(v_i)
    u = np.log(v + 1e-12)

    # -------------------------------------------------
    # 3. Pre‑compute FFT sizes
    # -------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)        # for weighted correlation

    # -------------------------------------------------
    # 4. Adam optimiser state
    # -------------------------------------------------
    m = np.zeros_like(u)
    v_hat = np.zeros_like(u)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    global_lr = 0.05           # base learning rate for Adam

    # -------------------------------------------------
    # 5. Smoothed–max schedule
    # -------------------------------------------------
    alpha = 5.0                # start with a gentle softmax
    alpha_max = 40.0
    alpha_mul = 1.001          # geometric increase per iteration

    # -------------------------------------------------
    # 6. Book‑keeping
    # -------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 3.0)   # a tiny safety margin
    best_a = None
    best_J = float('inf')
    last_improve = start_time
    stagnation_counter = 0
    iteration = 0

    # -------------------------------------------------
    # 7. Main optimisation loop
    # -------------------------------------------------
    while time.monotonic() < deadline:
        # ---- a = softmax(u) -------------------------------------------------
        max_u = np.max(u)
        exp_u = np.exp(u - max_u)    # shift for numerical stability
        a = exp_u / exp_u.sum()

        # ---- convolution a*a ------------------------------------------------
        conv = _conv_full(a, fft_len, conv_len)          # length 2n‑1
        max_conv = conv.max()
        J = float(2.0 * n * max_conv)

        # ---- keep the best ---------------------------------------------------
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---- smoothed max (softmax over conv) --------------------------------
        # weight_k ∝ exp(alpha * conv_k)
        # to keep numbers in a sane range we subtract the global max first
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        exp_scaled = np.exp(scaled - max_scaled)
        weight = exp_scaled / exp_scaled.sum()   # length conv_len, sums to 1

        # ---- gradient of the smoothed objective -------------------------------
        # grad_a = d/da Σ_k weight_k·conv_k = 2·corr(weight, a)
        grad_a = _weighted_gradient(a, weight, fft_len_corr, n)

        # chain rule for logits u (softmax)
        grad_a_total = 2.0 * n * grad_a                     # because J_smooth = 2·n·Σ w·conv
        dot = np.dot(grad_a_total, a)
        grad_u = a * (grad_a_total - dot)                  # ∂J/∂u_i

        # ---- Adam update ----------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_u
        v_hat = beta2 * v_hat + (1.0 - beta2) * (grad_u ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat_corr = v_hat / (1.0 - beta2 ** iteration)
        u = u - global_lr * m_hat / (np.sqrt(v_hat_corr) + eps)

        # ---- increase smoothing parameter ------------------------------------
        if alpha < alpha_max:
            alpha = min(alpha_max, alpha * alpha_mul)

        # ---- stagnation handling ---------------------------------------------
        now = time.monotonic()
        if now - last_improve > 30.0:
            # random restart: sample a fresh Dirichlet point
            rand_a = rng.dirichlet(np.ones(n))
            u = np.log(rand_a + 1e-12)
            last_improve = now
            stagnation_counter += 1
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Stagnation #{stagnation_counter} – random restart")

    # -----------------------------------------------------------------
    # 8. Final polishing (multiplicative‑gradient step on the true max)
    # -----------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    # one polishing round – identical to the “polish” step used before
    conv = _conv_full(best_a, fft_len, conv_len)
    max_conv = conv.max()
    active = (conv >= max_conv - 1e-12).astype(np.float64)   # indicator of binding constraints
    grad = _weighted_gradient(best_a, active, fft_len_corr, n)
    step_mul = 0.6                                          # a moderate step size
    factor = np.exp(-step_mul * grad)
    polished = best_a * factor
    polished = proj_simplex(polished)

    # evaluate polished candidate
    conv_polished = _conv_full(polished, fft_len, conv_len)
    J_polished = float(2.0 * n * conv_polished.max())
    if J_polished < best_J:
        best_J = J_polished
        best_a = polished
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] Polished best J = {best_J:.6f}")

    # -------------------------------------------------
    # 9. Normalise, clip and return
    # -------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    # (optional) sanity check with the supplied evaluate_sequence
    try:
        val = evaluate_sequence(best_a.tolist())
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    return [float(v) for v in best_a]
