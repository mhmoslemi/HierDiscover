def softmax_grad(a, beta, fft_len_full):
    n = a.shape[0]
    # compute self-convolution
    conv_len = 2 * n - 1
    fft_len_conv = _next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len_conv)
    conv = np.fft.irfft(A * A, n=fft_len_conv)[:conv_len]
    # compute softmax probabilities
    # Use stable exponent
    max_c = conv.max()
    exp_beta = np.exp(beta * (conv - max_c))
    p = exp_beta / exp_beta.sum()
    # full conv p * a
    # Need length of p + a -1 = (2n-1) + n -1 = 3n-2
    fft_len_full = _next_pow2(3 * n - 2)  # precomputed maybe
    P = np.fft.rfft(p, fft_len_full)

"""
Gradient‐descent with soft‑max smoothing on the probability simplex.

The objective to minimise (for a vector a with Σa=1) is

    J(a) = 2·n·max_k (a*a)[k]      (a*a = linear convolution)

J is scale‑invariant, so we work on the simplex (projected gradient updates).
A smooth approximation of the max is the log‑sum‑exp (soft‑max):
    Sβ(a) = (2·n/β)·log( Σ_k exp( β·(a*a)[k] ) )
Its gradient can be written in closed form using two FFT‑based convolutions,
so each iteration costs O(n log n).  We start with a small smoothing
parameter β and increase it geometrically, performing many projected‑gradient
steps at each β.  After this “deterministic” phase we finish with a short
simulated‑annealing jitter to escape any remaining local minima.

The routine respects the 1000 s budget, returns a non‑negative simplex vector,
and clips values to the limits imposed by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the O(n) algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def full_conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two 1‑D real arrays using the FFT.
    Returns the full result of length len(x)+len(y)-1.
    """
    L = _next_pow2(len(x) + len(y) - 1)
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[: len(x) + len(y) - 1]


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective J(a) = 2·n·max(conv(a,a)), assuming Σa = 1.
    """
    n = a.shape[0]
    conv = full_conv_fft(a, a)               # length 2·n‑1
    return float(2.0 * n * conv.max())


def softmax_gradient(a: np.ndarray, beta: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        Sβ(a) = (2·n/β)·log Σ exp(β·conv(a,a))
    with respect to a on the simplex.
    Returns ∇Sβ(a).
    """
    n = a.shape[0]

    # 1) self‑convolution a*a
    conv = full_conv_fft(a, a)               # length 2·n‑1

    # 2) soft‑max probabilities (stable formulation)
    max_c = conv.max()
    exp_beta = np.exp(beta * (conv - max_c))
    p = exp_beta / exp_beta.sum()            # shape (2·n‑1,)

    # 3) weighted convolutions  p * a   and   a * p_rev
    conv_pa = full_conv_fft(p, a)            # length 3·n‑2
    p_rev = p[::-1]
    conv_a_prevr = full_conv_fft(a, p_rev)   # length 3·n‑2

    # 4) Extract the two terms that form ∂/∂a Σ p_k·conv_k
    term2 = conv_pa[:n]                                      # p * a  (first n entries)
    term1 = conv_a_prevr[2 * n - 2 - np.arange(n)]          # a * p_rev (reversed slice)

    grad = term1 + term2                                     # ∂ Σ p·conv / ∂a
    # 5) Scale to the gradient of Sβ (the factor 2·n comes from the chain rule)
    grad *= 2.0 * n
    return grad


def gradient_step(a: np.ndarray, beta: float, lr: float) -> np.ndarray:
    """
    One projected‑gradient step on the soft‑max surrogate.
    """
    g = softmax_gradient(a, beta)
    a_new = a - lr * g
    return proj_simplex(a_new)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative heights) that minimises
    `evaluate_sequence`.  The routine combines a soft‑max gradient descent
    with a short simulated‑annealing fine‑tuning phase and returns the best
    sequence found before the allotted time expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size --------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        n = prev.shape[0]
        a = np.clip(prev, 0.0, None)
        a = proj_simplex(a)
    else:
        n = 4096
        a = rng.dirichlet(np.ones(n))

    best_vec = a.copy()
    best_J = compute_objective(best_vec)

    print(f"[0.00s] Initial J = {best_J:.6f} (n={n})")

    start_time = time.monotonic()
    # Reserve a few seconds for final cleanup / fine search
    deadline = start_time + max(0.0, float(budget_s) - 8.0)

    # ------------------------------------------------------------------
    # 2. Phase‑1: soft‑max gradient descent with increasing β ----------
    # ------------------------------------------------------------------
    beta = 0.5                     # start with a very smooth approximation
    max_beta = 64.0
    # Number of inner gradient steps per β – we stop early if time runs out
    steps_per_beta = 800          # generous; actual count limited by time

    while time.monotonic() < deadline and beta <= max_beta:
        # Learning‑rate schedule (decreases with β)
        lr = 0.25 / (np.sqrt(beta) + 0.5)

        improved_this_beta = False
        for _ in range(steps_per_beta):
            if time.monotonic() >= deadline:
                break

            # one gradient step
            a = gradient_step(a, beta, lr)
            cur_J = compute_objective(a)

            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] β={beta:.2f}  new best J = {best_J:.6f}")
                improved_this_beta = True

        # If we made no progress for a whole β‑stage we still increase β
        beta *= 2.0

    # ------------------------------------------------------------------
    # 3. Phase‑2: short simulated‑annealing jitter around the best -----
    # ------------------------------------------------------------------
    # Simple geometric cooling, mixing a tiny amount of fresh Dirichlet noise.
    sa_deadline = time.monotonic() + min(15.0, deadline - time.monotonic())
    cur_vec = best_vec.copy()
    cur_J = best_J

    if time.monotonic() < deadline:
        T_start = 0.015
        T_end = 1e-5
        total_sa_iters = 25000
        for it in range(total_sa_iters):
            if time.monotonic() >= sa_deadline:
                break
            t_frac = it / total_sa_iters
            T = T_start * (1 - t_frac) + T_end * t_frac

            # propose a small convex combination with fresh Dirichlet noise
            eps = rng.uniform(0.0, 0.018)     # mixing weight
            proposal = (1.0 - eps) * cur_vec + eps * rng.dirichlet(np.ones(n))
            proposal = proj_simplex(proposal)

            prop_J = compute_objective(proposal)
            delta = prop_J - cur_J
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_vec = proposal
                cur_J = prop_J
                if prop_J < best_J:
                    best_J = prop_J
                    best_vec = proposal.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] SA   new best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Phase‑3: final fine‑grained multiplicative jitter -------------
    # ------------------------------------------------------------------
    fine_deadline = time.monotonic() + min(6.0, deadline - time.monotonic())
    while time.monotonic() < fine_deadline:
        # multiplicative Gaussian perturbation (≈2 % relative noise)
        noise = np.exp(rng.normal(0.0, 0.018, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_J = compute_objective(cand)
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Final normalisation & safety clipping ------------------------
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total <= 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:6.2f}s] Finished – best J = {best_J:.6f}")

    return [float(v) for v in best_vec]
