    # returns projection onto simplex {x>=0, sum=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1

    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def compute_subgradient(a, conv):
    n = a.shape[0]
    max_val = conv.max()
    # gather indices where conv close to max within small tolerance relative to max (like < 1e-12)
    # Since conv may be exact due to floating errors, but we treat equality.
    K = np.where(conv >= max_val - 1e-12)[0]  # indices list
    g = np.zeros_like(a)
    for k in K:
        # compute gradient for shift k
        if k < n:
            # first term: a reversed prefix
            g[:k+1] += a[:k+1][::-1]
            # second term:
            g[k:] += a[:n-k]
        else:
            start = k - (n - 1)
            g[start:] += a[start:][::-1]
    # average if multiple indices
    if len(K) > 0:
        g /= len(K)
    return g

def propose_candidate(budget_s=1000.0, seed=12345, **kwargs):
    rng = np.random.default_rng(seed)
    # determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        n = max(256, min(n, 8192))
    else:
        n = 4096
    # convolution lengths 
    conv_len = 2*n-1
    fft_len = _next_pow2(conv_len)

    # random param search
    best_a = None
    best_J = np.inf
    start = time.time()
    deadline = start + budget_s - 5.0   # keep margin
    # define number of samples based on time
    num_samples = 0
    while time.time() < deadline and num_samples < 10000:
        # random choice of shape type
        shape_type = rng.choice(['uniform','v','geom','linear','power','exp','beta'])
        if shape_type == 'uniform':
            a = np.full(n, 1.0/n, dtype=np.float64)
        elif shape_type == 'v':
            idx = np.arange(n, dtype=np.float64)
            a = 1.0/((idx+1)*(n-idx))
            a = np.clip(a,0,None)
            a /= a.sum()
        elif shape_type == 'geom':
            r = rng.uniform(0.90,0.9999)
            a = r ** np.arange(n)
            a /= a.sum()
        elif shape_type == 'linear':
            # decreasing linear
            a = (n - np.arange(n)) 
            a = np.maximum(a,0)
            a = a.astype(np.float64)
            a /= a.sum()
        elif shape_type == 'power':
            p = rng.uniform(0.0,4.0)
            q = rng.uniform(0.0,4.0)
            i = np.arange(n, dtype=np.float64) + 1
            a = (i**p) * ((n - i + 1)**q)
            a = np.clip(a,0,None)
            a /= a.sum()
        elif shape_type == 'exp':
            c = rng.uniform(0.0, 0.01)  # small slope?
            a = np.exp(-c * np.arange(n))
            a /= a.sum()
        elif shape_type == 'beta':
            alpha = rng.uniform(0.2,5.0)
            beta = rng.uniform(0.2,5.0)
            i = np.arange(1,n+1)
            a = (i**(alpha-1) * (n-i+1)**(beta-1))
            a = np.clip(a,0,None)
            a /= a.sum()
        else:
            a = rng.dirichlet(np.ones(n))
        # compute J quickly (use own conv)
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        J = float(2.0 * n * max_conv / (a.sum()**2))
        if J < best_J:
            best_J = J
            best_a = a.copy()
            print(f"[{time.time()-start:6.2f}s] Random search new best J={best_J:.6f} using {shape_type}")
        num_samples += 1
    # add previous best as candidate
    if 'height_sequence_1' in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.size != n:
            if v.size > n: v = v[:n]
            else: v = np.concatenate([v, np.full(n - v.size, 1.0/n)])
        v = np.clip(v, 0.0, None)
        total = v.sum()
        if total == 0: v = np.full(n, 1.0/n)
        else: v /= total
        conv = _conv_full(v, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max() / (v.sum()**2))
        if cur_J < best_J:
            best_J = cur_J
            best_a = v.copy()
            print(f"[{time.time()-start:6.2f}s] Using previous best J={best_J:.6f}")

    # local refinement using Adam with smooth max + subgradient
    # run optimizer for remaining time
    remaining = deadline - time.time()
    if remaining > 1.0 and best_a is not None:
        refined_a, refined_J = _refine(best_a, remaining, start)
        if refined_J < best_J:
            best_J = refined_J
            best_a = refined_a

    # final clamp and renorm (though sum=1)
    best_vec = np.clip(best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0/n)
    else:
        best_vec /= total

    print(f"Finished search. Best J = {best_J:.6f}")
    return [float(v) for v in best_vec]

def _refine(a_init, time_budget, start_time):
    n = a_init.shape[0]; conv_len = 2*n-1; fft_len = _next_pow2(conv_len)
    logits = np.log(np.maximum(a_init, 1e-14))
    m1 = np.zeros_like(logits)
    v1 = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr0=0.5
    beta = 2.0
    beta_max=200.0
    beta_inc=1.002
    best_a = a_init.copy()
    best_J = float(2.0 * n * _conv_full(best_a, fft_len, conv_len).max())
    deadline = start_time + time_budget
    last_improve = time.time()
    iteration = 0
    while time.time() < deadline:
        # compute a via softmax
        max_log = np.max(logits)
        exp_log = np.exp(logits - max_log)
        a = exp_log / exp_log.sum()
        # conv
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv / (a.sum()**2))
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = time.time()
            print(f"[{time.time()-start_time:6.2f}s] Refinement new best J={best_J:.6f}")
        # jitter restart if stagnated >20 sec
        if time.time() - last_improve > 20.0:
            jitter_sigma = 0.2
            logits = np.log(np.maximum(best_a, 1e-14))
            logits += np.random.normal(0.0, jitter_sigma, size=n)
            m1.fill(0); v1.fill(0)
            last_improve = time.time()
            print(f"[{time.time()-start_time:6.2f}s] Restart jitter")
        # compute smooth weights
        w = _smooth_max_weights(conv, beta)   # using existing defined earlier
        # gradient w.r.t a via FFT (smooth)
        grad_a = _grad_smooth_max(a, w, _next_pow2(3*n-2))
        # optional: also subgradient
        # we could blend them: grad_a = (1-alpha)*grad_a + alpha*subgrad (alpha small)
        # compute subgradient from max shift
        # compute set of indices for max
        max_val = conv.max()
        K = np.where(conv >= max_val - 1e-12)[0]
        subgrad = np.zeros_like(a)
        for k in K:
            # use direct loop to add contributions:
            i_start = max(0, k-(n-1))
            i_end = min(k, n-1)
            i = np.arange(i_start, i_end+1)
            j = k - i
            np.add.at(subgrad, i, a[j])
            np.add.at(subgrad, j, a[i])
        if len(K) > 0:
            subgrad /= len(K)
        # blend
        grad_a = 0.7*grad_a + 0.3*subgrad
        # chain rule
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)
        iteration += 1
        lr = lr0 / np.sqrt(iteration)
        m1 = beta1*m1 + (1-beta1)*grad_z
        v1 = beta2*v1 + (1-beta2)*(grad_z**2)
        m1_hat = m1 / (1 - beta1**iteration)

"""
Search for a non‑negative step‑function that minimizes
    J(a) = 2·n·max(conv(a,a)) / (∑a)² .

The routine consists of two stages:

1. **Global random search** – a large number of simple parametric families
   (geometric, power, beta‑distribution, linear, exponential, etc.) are
   sampled.  Each candidate is normalised to unit L¹‑norm, its self‑convolution
   is evaluated with an FFT (O(n log n)), and the best sequence seen is stored.
   This stage quickly discovers good‑looking shapes.

2. **Local refinement** – starting from the best global candidate, a mirrored‑
   gradient descent on the probability simplex is performed.  The gradient
   is a blend of a smooth‑max approximation (as in the reference code) and the
   exact sub‑gradient of the hard max.  Adam with an adaptive learning rate
   updates the soft‑max logits, jitter restarts break stagnation, and the best
   vector found by the optimiser is returned.

The whole search respects the 1000 s budget and never returns an empty or
invalid sequence.  All values are clipped to the range required by
`evaluate_sequence`.  The algorithm works for any length (default up to 8192)
and can also reuse the previously discovered sequence `height_sequence_1`.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (FFT‑based convolution, smoothing, etc.)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using real FFT.
    Result length = conv_len = 2*len(a)-1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights for a smooth approximation of max(conv).
    Returns a probability vector that sums to 1.
    """
    m = conv.max()
    shifted = beta * (conv - m)
    # clip to avoid overflow in exp
    shifted = np.clip(shifted, -700, 0)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    return w


def _grad_smooth_max(a: np.ndarray, w: np.ndarray, grad_fft_len: int) -> np.ndarray:
    """
    Gradient of smooth‑max w.r.t. a.
    Uses the identity ∂/∂a_i  smooth_max = 2·(w * rev(a))[i].
    """
    rev_a = a[::-1]
    W = np.fft.rfft(w, grad_fft_len)
    A_rev = np.fft.rfft(rev_a, grad_fft_len)
    full = np.fft.irfft(W * A_rev, n=grad_fft_len)
    n = a.shape[0]
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _subgradient_max(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)) w.r.t. a.
    For every shift k attaining the maximum, we add the contributions
    a_{k-i} ↔ i and a_i ↔ k-i.
    """
    n = a.shape[0]
    max_val = conv.max()
    # tolerances for ties (numerical)
    K = np.where(conv >= max_val - 1e-12)[0]
    g = np.zeros_like(a)
    for k in K:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i = np.arange(i_start, i_end + 1, dtype=int)
        j = k - i
        # contribution a[j] to gradient at i
        np.add.at(g, i, a[j])
        # contribution a[i] to gradient at j
        np.add.at(g, j, a[i])
    if len(K) > 0:
        g /= float(len(K))
    return g


# ----------------------------------------------------------------------
# Local optimiser (Adam on the simplex with a mixed gradient)
# ----------------------------------------------------------------------


def _refine(
    a_start: np.ndarray,
    time_budget: float,
    start_time: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that starts from `a_start` and runs for at most
    `time_budget` seconds.  Returns the best vector seen and its J‑value.
    """
    n = a_start.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    grad_fft_len = _next_pow2(3 * n - 2)

    # initialise logits (soft‑max parametrisation)
    eps = 1e-14
    logits = np.log(np.maximum(a_start, eps))
    m1 = np.zeros_like(logits)
    v1 = np.zeros_like(logits)

    # Adam hyper‑parameters
    beta1, beta2 = 0.9, 0.999
    adam_eps = 1e-8
    lr0 = 0.8    # a slightly larger initial step size

    # smooth‑max temperature schedule
    beta = 2.0
    beta_max = 200.0
    beta_inc = 1.002

    # bookkeeping
    best_a = a_start.copy()
    conv = _conv_full(best_a, fft_len, conv_len)
    best_J = float(2.0 * n * conv.max() / (best_a.sum() ** 2))
    last_improve = time.time()
    iteration = 0

    deadline = start_time + time_budget

    while time.time() < deadline:
        # ----- probabilities from logits -----
        max_log = np.max(logits)
        exp_logits = np.exp(logits - max_log)
        a = exp_logits / exp_logits.sum()

        # ----- hard objective (for tracking) -----
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv / (a.sum() ** 2))

        now = time.time()
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s]  Refine new best J = {best_J:.6f}")

        # ----- stagnation handling -----
        if now - last_improve > 20.0:
            jitter_sigma = 0.25
            logits = np.log(np.maximum(best_a, eps))
            logits += np.random.normal(0.0, jitter_sigma, size=n)
            m1.fill(0.0)
            v1.fill(0.0)
            last_improve = now
            print(f"[{now - start_time:6.2f}s]  Restart – jitter applied")

        # ----- mixed gradient (smooth‑max + exact sub‑gradient) -----
        w = _smooth_max_weights(conv, beta)          # soft‑max over convolution
        grad_smooth = _grad_smooth_max(a, w, grad_fft_len)

        grad_sub = _subgradient_max(a, conv)

        # blend the two (empirically 0.7 / 0.3 works well)
        grad_a = 0.7 * grad_smooth + 0.3 * grad_sub

        # ----- back‑prop through the soft‑max -----
        avg = np.dot(a, grad_a)
        grad_z = a * (grad_a - avg)

        # ----- Adam step on logits -----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)

        m1 = beta1 * m1 + (1.0 - beta1) * grad_z
        v1 = beta2 * v1 + (1.0 - beta2) * (grad_z ** 2)

        m1_hat = m1 / (1.0 - beta1 ** iteration)
        v1_hat = v1 / (1.0 - beta2 ** iteration)

        logits -= lr * m1_hat / (np.sqrt(v1_hat) + adam_eps)
        logits = np.clip(logits, -500.0, 500.0)   # keep numerics sane

        # ----- increase smooth‑max temperature -----
        beta = min(beta_max, beta * beta_inc)

    # final J for the best vector
    conv = _conv_full(best_a, fft_len, conv_len)
    best_J = float(2.0 * n * conv.max() / (best_a.sum() ** 2))
    return best_a, best_J


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(budget_s: float = 1000.0, seed: int = 12345, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (a step function) that minimises the
    evaluation function `evaluate_sequence`.  The routine spends most of the
    allotted time exploring simple parametric families, then performs a local
    Adam‑based refinement.  The best sequence found is returned.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size n
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Global random search over simple families
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    start = time.time()
    deadline = start + max(0.0, float(budget_s) - 10.0)   # reserve ~10 s for refinement
    samples = 0

    while time.time() < deadline and samples < 15000:
        shape = rng.choice(
            ["uniform", "v", "geom", "linear", "power", "exp", "beta", "dirichlet"]
        )
        if shape == "uniform":
            a = np.full(n, 1.0 / n, dtype=np.float64)

        elif shape == "v":
            idx = np.arange(n, dtype=np.float64)
            a = 1.0 / ((idx + 1) * (n - idx))
            a = np.clip(a, 0, None)
            a /= a.sum()

        elif shape == "geom":
            r = rng.uniform(0.90, 0.9985)               # geometric decay factor
            a = r ** np.arange(n, dtype=np.float64)
            a /= a.sum()

        elif shape == "linear":
            a = (n - np.arange(n, dtype=np.float64))
            a = np.maximum(a, 0)
            a /= a.sum()

        elif shape == "power":
            p = rng.uniform(0.0, 4.0)
            q = rng.uniform(0.0, 4.0)
            i = np.arange(1, n + 1, dtype=np.float64)
            a = (i ** p) * ((n - i + 1) ** q)
            a = np.clip(a, 0, None)
            a /= a.sum()

        elif shape == "exp":
            c = rng.uniform(0.0, 0.015)                # exponential decay rate
            a = np.exp(-c * np.arange(n, dtype=np.float64))
            a /= a.sum()

        elif shape == "beta":
            alpha = rng.uniform(0.2, 5.0)
            beta = rng.uniform(0.2, 5.0)
            i = np.arange(1, n + 1, dtype=np.float64)
            a = (i ** (alpha - 1)) * ((n - i + 1) ** (beta - 1))
            a = np.clip(a, 0, None)
            a /= a.sum()

        else:   # dirichlet draw
            alpha = rng.uniform(0.2, 2.0)
            a = rng.dirichlet(np.ones(n) * alpha)

        # fast J evaluation (same formula as evaluate_sequence)
        conv = _conv_full(a, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max() / (a.sum() ** 2))

        if cur_J < best_J:
            best_J = cur_J
            best_a = a.copy()
            elapsed = time.time() - start
            print(f"[{elapsed:6.2f}s] Random search new best J = {best_J:.6f} (shape={shape})")

        samples += 1

    # --------------------------------------------------------------
    # 3. Seed with the previously discovered construction (if any)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        v = np.clip(v, 0.0, None)
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        conv = _conv_full(v, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max() / (v.sum() ** 2))
        if cur_J < best_J:
            best_J = cur_J
            best_a = v.copy()
            print(f"[{time.time() - start:6.2f}s] Seeded with previous best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Local refinement (Adam + mixed gradient)
    # --------------------------------------------------------------
    remaining = max(0.0, deadline - time.time())
    if best_a is not None and remaining > 1.0:
        refined_a, refined_J = _refine(best_a, remaining, start)
        if refined_J < best_J:
            best_J = refined_J
            best_a = refined_a
            print(f"[{time.time() - start:6.2f}s] After refinement best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Final normalisation and output
    # --------------------------------------------------------------
    best_vec = np.clip(best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total                     # keep the sum ≈ 1 (scale‑invariant)

    print(f"Finished. Best J = {best_J:.6f}")
    return [float(v) for v in best_vec]
