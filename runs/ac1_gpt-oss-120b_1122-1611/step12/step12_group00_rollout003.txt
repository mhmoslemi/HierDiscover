def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    '''
    Simulated annealing + gradient refinement search for a step function that
    minimises `evaluate_sequence`. ...
    '''
    import numpy as np
    import time

    rng = np.random.default_rng(seed)

    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        n = max(2048, min(n, 8192))
    else:
        n = 4096

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    # Helper: compute objective J = 2*n*max(conv)
    def obj(vec):
        conv = conv_full_fft(vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # Initial population: store best
    # start with uniform
    best_vec = np.full(n, 1.0/n, dtype=np.float64)
    best_J = obj(best_vec)

    # optionally incorporate previous best if exists
    if 'height_sequence_1' in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.zeros(n - v.size)])
        v = proj_simplex(v)
        Jv = obj(v)
        if Jv < best_J:
            best_J = Jv
            best_vec = v.copy()
            print(f"Initial previous best J = {best_J:.6f}")

    # random sampling phase
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 200.0)  # keep 200s for refinement
    while time.monotonic() < deadline:
        # Dirichlet random draws with random alpha
        alpha = rng.uniform(0.1, 5.0)
        cand = rng.dirichlet(np.full(n, alpha))
        Jc = obj(cand)
        if Jc < best_J:
            best_J = Jc
            best_vec = cand
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Random init improved J = {best_J:.6f}")

    # Simulated annealing local search
    # parameters
    total_time = time.monotonic()
    end_time = total_time + min(200.0, budget_s - (total_time - start)) # allocate 200s for SA
    T0 = best_J * 0.5
    Tmin = 1e-4
    # geometric cooling schedule
    while time.monotonic() < end_time:
        now = time.monotonic()
        elapsed = now - start
        # temperature decreasing linearly
        frac = (now - total_time) / (end_time - total_time)
        T = (1 - frac) * T0 + frac * Tmin

        # propose small move
        i, j = rng.choice(n, size=2, replace=False)
        # delta up to fraction of a_i
        delta = rng.random() * min(best_vec[i], 0.1)  # up to 10% of a_i
        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        # ensure nonnegative, then project onto simplex to fix sum
        cand = proj_simplex(cand)

        Jc = obj(cand)
        dJ = Jc - best_J
        if dJ < 0 or rng.random() < np.exp(-dJ / T):
            best_vec = cand
            best_J = Jc
            print(f"[{elapsed:6.2f}s] SA accept J = {best_J:.6f}")

    # Gradient refinement (subgradient descent)
    # We'll run for remaining time
    remaining = start + budget_s - time.monotonic()
    if remaining > 10.0:
        # step schedule
        step = 0.5
        grad_deadline = time.monotonic() + remaining
        while time.monotonic() < grad_deadline:
            conv = conv_full_fft(best_vec, fft_len, conv_len)
            k_star = int(np.argmax(conv))
            # compute gradient
            grad = np.zeros_like(best_vec)
            if k_star < n:
                grad[:k_star+1] = 2.0 * best_vec[:k_star+1][::-1]

            cand = proj_simplex(cand)
            Jc = obj(cand)
            if Jc < best_J:
                best_J = Jc
                best_vec = cand
                print(f"[Grad] Improved J = {best_J:.6f}")

"""
Propose a step‑function (non‑negative vector) that minimises the
`evaluate_sequence` objective.

The algorithm mixes three complementary stages:

1. **Random Dirichlet sampling** – a quick exploration of the simplex
   using different concentration parameters (30 % of the time budget).

2. **Simulated‑annealing hill‑climbing** – repeatedly transfer a small
   amount of mass between two coordinates.  Moves that improve the
   objective are always accepted; worsening moves are accepted with a
   temperature‑dependent probability (≈50 % of the time budget).

3. **Sub‑gradient refinement** – when the current vector `a` is fixed,
   the maximal entry of the auto‑convolution is at some index `k*`.  The
   sub‑gradient of `max(conv(a,a))` at `a` is simply
   `g_i = 2·a_{k*‑i}` (when the index is valid).  A projected
   sub‑gradient step followed by a simplex projection quickly smooths
   large peaks (≈15 % of the time budget).

A short “fine‑jitter’’ phase (the remaining time) adds multiplicative
Gaussian noise and keeps any improvement.  All vectors are kept on the
probability simplex (∑ a_i = 1) so that the evaluated score equals
`2·n·max(conv(a,a))`, which is exactly the value returned by
`evaluate_sequence` for a sum of 1.

The routine respects the security checks of `evaluate_sequence`:
all entries are non‑negative, bounded by 1000, and returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns exactly `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def objective_from_vec(a: np.ndarray, n: int,
                       fft_len: int, conv_len: int) -> float:
    """
    Compute the value returned by `evaluate_sequence` for a vector with
    total sum = 1, i.e. 2·n·max(conv(a,a)).
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------

                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for the best non‑negative step‑function within the allocated
    time budget.  Returns a list of floats whose sum is 1 (the simplex
    normalisation).  The search proceeds through random sampling,
    simulated‑annealing hill‑climbing, sub‑gradient refinement and a
    final fine‑jitter phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size --------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(1024, min(n, 8192))   # keep it reasonable
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # power‑of‑two for FFT

    # ------------------------------------------------------------------
    # 2. Objective helper -----------------------------------------------
    # ------------------------------------------------------------------
    def obj(vec: np.ndarray) -> float:
        """Fast objective (sum of vector is assumed to be 1)."""
        return objective_from_vec(vec, n, fft_len, conv_len)

    # ------------------------------------------------------------------
    # 3. Initialise best candidate --------------------------------------
    # ------------------------------------------------------------------
    best_vec = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    best_J = obj(best_vec)

    # use the previous construction if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)

            else:
                v = np.concatenate([v, np.zeros(n - v.size)])
        v = proj_simplex(v)
        Jv = obj(v)
        if Jv < best_J:
            best_J = Jv
            best_vec = v.copy()
            print(f"[Init] Previous best J = {best_J:.6f}")

    print(f"[Init] Starting J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Timing set‑up -------------------------------------------------
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s))

    # allocate fractions of the total budget
    phase1_end = start_time + 0.30 * budget_s   # random Dirichlet sampling
    phase2_end = start_time + 0.80 * budget_s   # simulated annealing
    phase3_end = deadline - 0.02 * budget_s     # sub‑gradient (leaving a tiny buffer)

    # ------------------------------------------------------------------
    while time.monotonic() < phase1_end:

        if max_delta <= 0.0:

        cand = best_vec.copy()
        cand[i] -= delta
        cand[j] += delta
        # sum is preserved and non‑negativity holds, projection not needed
        Jc = obj(cand)

    # ------------------------------------------------------------------
    step = 0.5                     # initial step length

            elapsed = time.monotonic() - start_time
