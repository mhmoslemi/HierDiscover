def gradient_maxconv(a, conv):
    # conv is full convolution a*a
    k = np.argmax(conv)
    N = a.shape[0]
    # compute gradient vector g
    g = np.zeros_like(a)
    # g[j] = 2*a[k-j] if 0 <= k-j < N else 0
    # For j where idx = k - j in range.
    # So g[j] = 2*a[idx]
    # Equivalent: reversed a shifted by k
    # We can compute g = 2 * np.take(a, k - np.arange(N), mode='clip')? Not straightforward.
    # Let's compute vector of indices
    idx = k - np.arange(N)
    # valid mask
    mask = (idx >= 0) & (idx < N)
    # assign values
    g[mask] = 2.0 * a[idx[mask]]
    return g, k

"""
Propose a step‑function (non‑negative vector) that minimises
`evaluate_sequence`.  

The algorithm uses a collection of diversified initial candidates
(uniform, geometric, ramp, triangular, beta‑shaped and random
Dirichlet draws).  For the best few candidates a simple projected
sub‑gradient descent on the *max* of the auto‑convolution is performed.
The sub‑gradient of `max(conv(a,a))` at a point is
`g_j = 2·a_{k−j}` where `k` is an index attaining the maximum.
After each step we project onto the probability simplex.
Occasional jitter and a short Metropolis acceptance rule help escape
plateaus.  Finally a cheap pairwise‑mass‑swap annealing polish is run.

The whole routine respects the 1000 s time budget, works with any
vector length (the code prefers a power‑of‑two ≤ 8192 for fast FFT) and
returns a plain Python list of floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – all top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute the evaluation function (without the extra clipping performed
    inside `evaluate_sequence`).  The vector `a` is assumed to be
    non‑negative and to sum to 1 (or close to it); the expression
    2·n·max(conv) therefore equals the true objective.
    """
    conv = _conv_full(a, fft_len, conv_len)
    max_conv = conv.max()
    return float(2.0 * n * max_conv)


def _gradient_maxconv(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) at the maximum index `max_idx`.
    For each component j, ∂max/∂a_j = 2·a_{max_idx-j} if the index is
    inside the array, otherwise 0.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    idx = max_idx - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g[mask] = 2.0 * a[idx[mask]]
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found
    within the allocated time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + float(budget_s)

    # --------------------------------------------------------------
    # 1. Choose problem size (prefer a power of two for FFT speed)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    # force a power of two (fast FFT) while staying inside allowed range
    if (n & (n - 1)) != 0:
        n = 1 << (n - 1).bit_length()
        n = max(256, min(n, 8192))

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Build a diverse pool of initial probability vectors
    # --------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.94, 0.97, 0.985, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # triangular shape (increase then decrease)
    mid = n // 2
    if n % 2 == 0:
        left = np.arange(1, mid + 1, dtype=np.float64)
        right = np.arange(mid, 0, -1, dtype=np.float64)
    else:
        left = np.arange(1, mid + 2, dtype=np.float64)
        right = np.arange(mid + 1, 0, -1, dtype=np.float64)
    tri = np.concatenate([left, right])
    tri /= tri.sum()
    init_cands.append(tri)

    # sinusoidal squared (smooth bump)
    x = np.arange(n, dtype=np.float64)
    sin_sq = np.sin(np.pi * x / (n - 1)) ** 2
    sin_sq /= sin_sq.sum()
    init_cands.append(sin_sq)

    # beta‑shaped PDFs (discrete approximations)
    # a_i ∝ (i+0.5)^{α-1} (n-i-0.5)^{β-1}
    for (alpha, beta) in [(0.5, 0.5), (2.0, 2.0), (0.8, 0.8), (1.5, 1.5)]:
        i = np.arange(n, dtype=np.float64)
        x = (i + 0.5) / n
        v = (x ** (alpha - 1)) * ((1.0 - x) ** (beta - 1))
        v = np.maximum(v, 0.0)
        v_sum = v.sum()
        if v_sum > 0:
            v /= v_sum
            init_cands.append(v)

    # random Dirichlet draws with a variety of concentrations
    for alpha in [0.02, 0.05, 0.1, 0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction (if present)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        total = v.sum()
        if total == 0.0:
            v[:] = 1.0 / n
        else:
            v /= total
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Evaluate all candidates quickly and keep the best few
    # --------------------------------------------------------------
    cand_vals = []
    for cand in init_cands:
        J = _objective(cand, n, fft_len, conv_len)
        cand_vals.append((J, cand.copy()))
    cand_vals.sort(key=lambda x: x[0])
    best_J = cand_vals[0][0]
    best_vec = cand_vals[0][1].copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # Number of top candidates to refine (we keep a small diverse set)
    TOP_K = 5
    top_cands = [vec for _, vec in cand_vals[:TOP_K]]

    # --------------------------------------------------------------
    # 4. Sub‑gradient descent on max‑convolution (multiple restarts)
    # --------------------------------------------------------------
    # Reserve a few seconds for final polishing
    POLISH_TIME = 30.0
    sub_deadline = deadline - POLISH_TIME
    if sub_deadline <= start_time:
        sub_deadline = deadline

    for restart_id, a0 in enumerate(top_cands, 1):
        if time.time() >= sub_deadline:
            break

        a = proj_simplex(a0.copy())        # ensure exact simplex membership
        cur_J = _objective(a, n, fft_len, conv_len)
        cur_best_time = time.time()
        last_improve = cur_best_time

        # Step‑size schedule (linear from step0 → step_min)
        step0 = 0.7
        step_min = 0.001

        while time.time() < sub_deadline:
            # --- compute full auto‑convolution and locate the peak ---
            conv = _conv_full(a, fft_len, conv_len)
            max_idx = int(conv.argmax())
            max_conv = float(conv[max_idx])
            cur_J = float(2.0 * n * max_conv)

            # --- update global best if improved ---
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = a.copy()
                last_improve = time.time()
                print(f"[{time.time() - start_time:6.2f}s] Restart {restart_id}: new global best J = {best_J:.6f}")

            # --- sub‑gradient for the active max index ---
            g = _gradient_maxconv(a, max_idx)

            # --- step‑size (decrease as we approach deadline) ---
            now = time.time()
            t_frac = (now - start_time) / (sub_deadline - start_time)
            step = step0 * (1.0 - t_frac) + step_min * t_frac

            # --- propose a projected step ---
            proposal = proj_simplex(a - step * g)
            prop_J = _objective(proposal, n, fft_len, conv_len)

            # --- Metropolis‑style acceptance ---
            if prop_J < cur_J:
                accept = True
            else:
                # temperature proportional to current J, decays over time
                T = best_J * 0.02 * (1.0 - t_frac) + 1e-8
                prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
                accept = rng.random() < prob

            if accept:
                a = proposal
                cur_J = prop_J

            # --- stagnation detection (jitter) ---
            if now - last_improve > 15.0:
                noise = np.exp(rng.normal(0.0, 0.07, size=n))
                a = proj_simplex(best_vec * noise)
                cur_J = _objective(a, n, fft_len, conv_len)
                last_improve = now
                print(f"[{now - start_time:6.2f}s] Restart {restart_id}: jitter – J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 5. Final polishing – pairwise mass swaps with annealing
    # --------------------------------------------------------------
    cur = best_vec.copy()
    cur_J = best_J
    melt_start = time.time()
    while time.time() < deadline:
        # pick two distinct positions
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # move a modest amount of mass from i to j
        eps = rng.random() * cur[i] * 0.35
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        now = time.time()
        # simple annealing temperature (linear decay)
        T = cur_J * 0.01 * (deadline - now) / (deadline - start_time) + 1e-9

        if cand_J < cur_J or rng.random() < np.exp(-(cand_J - cur_J) / max(T, 1e-12)):
            cur = cand
            cur_J = cand_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur.copy()
                print(f"[{now - start_time:6.2f}s] Polishing improvement: J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Normalise, clip and return as plain Python list
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
