"""
Sub‑gradient descent on the exact max‑convolution objective.

The evaluation function is
    J(a) = 2·n·max_k (a * a)[k]      (∑a_i = 1, a_i ≥ 0)

Because the convolution is piecewise linear, the gradient of the
non‑smooth term max_k (a*a)[k] is given by the sum of the
derivatives of all active (max‑attaining) shifts.  For an active shift
p the derivative w.r.t. a_i is 2·a_{p‑i} (if the index is in range).
Hence the sub‑gradient of J is

    ∇J_i = 2·n· Σ_{p∈Active} 2·a_{p‑i}
          = 4·n· Σ_{p∈Active} a_{p‑i}.

The algorithm repeatedly
  * evaluates the full convolution by FFT,
  * builds the exact sub‑gradient from the active shifts,
  * performs a back‑tracking line search on the direction –∇J,
  * projects the candidate onto the probability simplex,
  * accepts the move if it improves the objective,
  * occasionally injects a small random jitter to escape plateaus.

The routine works for any length n (the length of the global variable
`height_sequence_1` is used if present, otherwise a default of 2048 is
chosen).  The whole search respects the 1000 s budget and returns the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | Σx = 1, x ≥ 0 }.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all entries equal or zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT of length ``fft_len``."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective J = 2·n·max(conv) for a vector already normalised to Σa = 1."""
    return float(2.0 * n * conv.max())


def active_gradient(conv: np.ndarray, a: np.ndarray, n: int) -> np.ndarray:
    """
    Exact sub‑gradient of J = 2·n·max(conv(a,a)).
    ``conv`` must be the full convolution of ``a`` with itself.
    """
    max_val = conv.max()
    # tolerance for the active set (numerical safety)
    tol = max(1e-12, max_val * 1e-12)
    active = np.where(conv >= max_val - tol)[0]

    grad = np.zeros_like(a)
    for p in active:
        # indices i such that both i and p-i lie in [0, n-1]
        start = max(0, p - (n - 1))
        end = min(p, n - 1)
        idx = np.arange(start, end + 1)
        # contribution 2·a_{p-i}
        grad[idx] += 2.0 * a[p - idx]

    # scale by the outer factor 2·n (J = 2·n·max_conv)
    grad *= (2.0 * n)
    return grad


def line_search(a: np.ndarray,
                direction: np.ndarray,
                n: int,
                fft_len: int,
                conv_len: int,
                cur_J: float,
                rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple back‑tracking line search on the direction ``direction``.
    Returns a (possibly unchanged) candidate and its J‐value.
    """
    # trial step sizes (geometric series)
    for t in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.0156):
        a_new = proj_simplex(a + t * direction)
        conv_new = _conv_full(a_new, fft_len, conv_len)
        J_new = compute_J_from_conv(conv_new, n)
        if J_new < cur_J:
            return a_new, J_new
    return a, cur_J


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(budget_s: float = 1000.0,
                     seed: int = 424242,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The routine respects the time
    budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size – reuse a known construction if present.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep the size reasonable for the 1 k‑second budget
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # length for a*a convolution

    # --------------------------------------------------------------
    # 2. Initialise the search vector on the simplex.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
        s = a0.sum()
        if s == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= s
    else:
        # start from a slightly perturbed Dirichlet (α=0.6) to avoid perfect symmetry
        a0 = rng.dirichlet(alpha=np.full(n, 0.6))
    a = a0.copy()

    # evaluate the initial point
    conv = _conv_full(a, fft_len, conv_len)
    cur_J = compute_J_from_conv(conv, n)
    best_a = a.copy()
    best_J = cur_J

    print(f"[0.00s] Initial J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 3. Optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep 5 s safety margin
    last_improve = start_time
    iteration = 0

    # parameters for occasional jitter
    jitter_sigma = 0.08
    jitter_interval = 20.0                     # seconds without improvement

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # 3.1 compute convolution and exact sub‑gradient
        # ----------------------------------------------------------
        conv = _conv_full(a, fft_len, conv_len)
        cur_J = compute_J_from_conv(conv, n)

        # update global best if needed
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # 3.2 Stagnation handling – small random jitter
        # ----------------------------------------------------------
        if now - last_improve > jitter_interval:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a = proj_simplex(best_a * jitter)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter performed – resetting to perturbed best")
            continue

        # ----------------------------------------------------------
        # 3.3 Sub‑gradient and line search
        # ----------------------------------------------------------
        grad = active_gradient(conv, a, n)          # shape (n,)
        direction = -grad                           # descent direction

        a_new, new_J = line_search(a, direction, n,
                                   fft_len, conv_len,
                                   cur_J, rng)

        # ----------------------------------------------------------
        # 3.4 Acceptance
        # ----------------------------------------------------------
        if new_J < cur_J - 1e-12:
            a = a_new
            cur_J = new_J
        else:
            # Occasionally try a random cheap exchange move to inject diversity
            if rng.random() < 0.06:
                i, j = rng.choice(n, size=2, replace=False)
                delta = min(a[i], rng.random() * 0.004)   # move up to 0.4 % of mass
                a[i] -= delta
                a[j] += delta
                a = proj_simplex(a)

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    print(f"Finished after {time.monotonic() - start_time:.2f}s – best J = {best_J:.6f}")
    return [float(v) for v in best_a]
