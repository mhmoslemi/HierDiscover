def _initialize_binary(n, m, rng):
    indices = rng.choice(n, size=m, replace=False)

    c[indices] = 1
    S = set(indices.tolist())

"""
Binary‑support simulated‑annealing search.

We treat the step function as a binary vector `b` of length *n* with exactly
*m* ones and interpret the actual coefficients as `a_i = b_i / m`
(so the sum of the coefficients is 1).  Because the evaluation
function is scale‑invariant, minimizing

    J = 2·n·max(conv(b,b)) / m²

is equivalent to minimizing ``evaluate_sequence``.

The algorithm keeps a histogram of the full convolution (the ordered pair
count for each possible sum).  A swap of a “1’’ at position *i_rem* with a
“0’’ at *i_add* can be applied to the histogram by simple vectorised
updates, allowing thousands of iterations per second.  An exponential
cooling schedule (simulated annealing) accepts occasional uphill moves,
and a stagnation‑restart prevents getting stuck.

Multiple support‑densities (different `m/n` ratios) are tried
sequentially; the best sequence found overall is returned as a list of
floats (each entry is either 0 or `1/m`).
"""

import time
import numpy as np


def _init_binary_vector(n: int, m: int, rng: np.random.Generator):
    """Return a random binary vector of length ``n`` with ``m`` ones,
    together with the set of indices of the ones."""
    inds = rng.choice(n, size=m, replace=False)
    b = np.zeros(n, dtype=np.int8)
    b[inds] = 1
    return b, set(int(i) for i in inds)


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget and
    returns the best sequence it found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size – reuse a known good length if possible.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            # keep a reasonable upper bound to avoid excessive FFT sizes
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 4096                     # default length when no prior solution is known

    # ------------------------------------------------------------------
    # 2. Parameters for the different support densities we will try.
    # ------------------------------------------------------------------
    density_ratios = [0.4, 0.5, 0.6]   # fraction of positions that are non‑zero
    best_overall_J = np.inf
    best_overall_vec = None
    best_overall_m = None

    overall_start = time.monotonic()
    deadline = overall_start + max(0.0, float(budget_s) - 2.0)  # safety margin

    # ------------------------------------------------------------------
    # 3. Main optimisation loop (one pass for each density ratio).
    # ------------------------------------------------------------------
    for ratio in density_ratios:
        if time.monotonic() >= deadline:
            break

        m = max(2, int(ratio * n))          # number of ones
        m = min(m, n)

        # ---- initialise a random binary vector and its convolution histogram ----
        b, S = _init_binary_vector(n, m, rng)            # b is np.int8, S is a set
        counts = np.convolve(b.astype(np.int64), b.astype(np.int64))
        max_cnt = int(counts.max())
        cur_J = 2.0 * n * max_cnt / (m * m)

        best_local_J = cur_J
        best_local_b = b.copy()
        best_local_counts = counts.copy()
        best_local_S = S.copy()

        # ---- annealing schedule (exponential cooling) ----
        start_time = time.monotonic()
        total_time = deadline - start_time
        T_start = cur_J
        T_end = 1e-8

        it = 1
        last_improve = start_time

        while time.monotonic() < deadline:
            now = time.monotonic()
            elapsed = now - start_time

            # temperature for Metropolis acceptance
            T = T_start * (T_end / T_start) ** (elapsed / total_time) if total_time > 0 else T_end

            # --------------------------------------------------------------
            # 3.1 Choose a swap (i_rem, i_add)
            # --------------------------------------------------------------
            # try to remove an element that actually participates in the current
            # maximal sum (helps the search); if none are found we fall back to random.
            k_max = int(counts.argmax())
            removal_candidates = [i for i in S
                                  if 0 <= k_max - i < n and (k_max - i) in S]
            if removal_candidates and rng.random() < 0.7:
                i_rem = int(rng.choice(removal_candidates))
            else:
                i_rem = int(rng.choice(list(S)))

            # choose a zero entry uniformly; try a few times to avoid obvious bad picks
            i_add = None
            for _ in range(5):
                cand = int(rng.integers(0, n))
                if cand not in S:
                    i_add = cand
                    break
            if i_add is None:
                zeros = np.where(b == 0)[0]
                if zeros.size == 0:
                    break
                i_add = int(rng.choice(zeros))

            # --------------------------------------------------------------
            # 3.2 Simulate the swap on a copy of the histogram
            # --------------------------------------------------------------
            new_counts = counts.copy()

            # removal contribution (2·shift_i_rem – self‑pair correction)
            S_arr = np.array(list(S), dtype=np.int64)          # current support
            new_counts[S_arr + i_rem] -= 2
            new_counts[2 * i_rem] += 1

            # support after removal
            mask = S_arr != i_rem
            S_without = S_arr[mask]

            # addition contribution
            new_counts[S_without + i_add] += 2
            new_counts[2 * i_add] += 1

            new_max = int(new_counts.max())
            new_J = 2.0 * n * new_max / (m * m)

            # --------------------------------------------------------------
            # 3.3 Acceptance test
            # --------------------------------------------------------------
            accept = False
            if new_J < cur_J:
                accept = True
            else:
                # Metropolis acceptance
                prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                # commit the swap
                counts = new_counts
                b[i_rem] = 0
                b[i_add] = 1
                S.remove(i_rem)
                S.add(i_add)
                cur_J = new_J

                if cur_J < best_local_J - 1e-12:
                    best_local_J = cur_J
                    best_local_b = b.copy()
                    best_local_counts = counts.copy()
                    best_local_S = S.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] ratio {ratio:.2f} new best J = {best_local_J:.6f}")

            # --------------------------------------------------------------
            # 3.4 Stagnation handling – restart if no improvement for a while
            # --------------------------------------------------------------
            if now - last_improve > 30.0:
                # re‑initialise from a fresh random configuration
                b, S = _init_binary_vector(n, m, rng)
                counts = np.convolve(b.astype(np.int64), b.astype(np.int64))
                cur_J = 2.0 * n * int(counts.max()) / (m * m)
                last_improve = now
                print(f"[{elapsed:6.2f}s] ratio {ratio:.2f} restart (stagnation)")

            it += 1

        # ------------------------------------------------------------------
        # 4. Keep the best solution among all density ratios.
        # ------------------------------------------------------------------
        print(f"Finished ratio {ratio:.2f}: best J = {best_local_J:.6f}")
        if best_local_J < best_overall_J:
            best_overall_J = best_local_J
            best_overall_vec = best_local_b.copy()
            best_overall_m = m

        # stop early if time is almost exhausted
        if time.monotonic() + 5.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # 5. Convert the best binary vector to the required float list.
    # ------------------------------------------------------------------
    if best_overall_vec is None:
        # fallback: uniform vector (should never happen)
        best_overall_vec = np.full(n, 1, dtype=np.int8)
        best_overall_m = n

    # Each non‑zero entry gets weight 1/m so that the sum is exactly 1.
    a = best_overall_vec.astype(np.float64) / float(best_overall_m)

    # Clip to the allowed range and re‑normalise just in case (numerical safety).
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total == 0.0:
        a[:] = 1.0 / n
    else:
        a /= total

    return [float(v) for v in a.tolist()]
