def propose_candidate(seed=4242, budget_s=1000.0, **kwargs):

    # Determine n
    global height_sequence_1
    if 'height_sequence_1' in globals():
        n_init = len(height_sequence_1)
    else:
        n_init = 2048
    n = max(256, min(int(n_init), 8192))

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    # Initialize a
    if 'height_sequence_1' in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n-a0.size))
    else:
        a0 = np.full(n, 1.0/n, dtype=np.float64)
    # Normalize sum to 1
    total = a0.sum()
    if total > 0:
        a0 = a0 / total
    else:
        a0 = np.full(n, 1.0/n)

    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)
    cur_J = best_J
    print(f"[0.00s] Initial J = {best_J:.6f}")

    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)
    total_time = deadline - start

    # Phase lengths (seconds)
    # allocate 30% for random jitter + gradient descent warmup, 60% SA, 10% final refinement
    phase1_end = start + 0.30 * total_time
    phase2_end = start + 0.90 * total_time

    # Parameters for gradient descent warmup
    # We'll use Adam with simple parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.2
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 1

    while True:

        if now >= deadline:
            break

        if now < phase1_end:

            # compute smoothing param alpha: start low (~5) to high (~200) linearly over phase
            progress = (now - start) / (phase1_end - start)
            alpha = 5.0 + 195.0 * progress  # from 5 to 200

            max_conv = conv.max()

            w_raw = np.exp(alpha * shift)
            w_sum = w_raw.sum()

                tol = max(1e-12, max_conv * 1e-12)
                active = np.where(conv >= max_conv - tol)[0]
                grad = np.zeros_like(a)
                for p in active:
                    start_i = max(0, p - (n - 1))
                    end_i = min(p, n - 1)
                    inds = np.arange(start_i, end_i+1)
                    grad[inds] += 2.0 * a[p - inds]
            else:

                a_rev = a[::-1]

                conv2 = conv_fft(w, a_rev, L_grad)
                grad = 2.0 * conv2[n-1:n-1+n]   # length n

            grad_J = 2.0 * n * grad

            # Adam update
            m = beta1 * m + (1 - beta1) * grad_J
            v = beta2 * v + (1 - beta2) * (grad_J ** 2)

            v_hat = v / (1 - beta2 ** iteration)

            a_proposed = proj_simplex(a_proposed)

"""
Hybrid search for a low‑value step‑function.

The objective is the evaluation function

    J(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

where n = len(a) and conv(a,a) is the linear convolution of a with itself.
Because J is invariant under scaling we may enforce sum(a)=1 and minimise
2·n·max(conv).  The problem is highly non‑convex; a pure gradient descent
gets stuck in local minima, while a simple simulated‑annealing (SA) walk can
explore far more.

The algorithm is split into three stages (time‑based):

1. **Warm‑up gradient descent** – a smooth‑max (soft‑max) surrogate of the
   maximum is optimised with Adam.  The smoothing parameter α is increased
   from 5 to 200, giving a reasonable direction towards flatter convolutions.

2. **Exploratory SA** – mass is repeatedly transferred between two random
   positions.  The temperature decays geometrically, the step size shrinks
   from ~20 % of the source mass to ~0.1 % and occasional jitter is applied
   after long stagnation periods.

3. **Fine‑grained refinement** – a final high‑α gradient step (α≈300) with a
   smaller learning rate cleans up residual peaks.

The best vector found is returned as a plain list of floats.  All helpers are
top‑level, no closures or file‑IO are used and the routine respects a 1000 s
budget (default) by checking the monotonic clock.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | sum(x)=1, x_i ≥ 0}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works only on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all zero (or equal)
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using an FFT of length ``fft_len``."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of ``x`` and ``y`` using an FFT of length ``L``."""
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[: len(x) + len(y) - 1]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute the objective J = 2·n·max(conv(a,a)).
    The vector ``a`` is assumed to be normalised (∑a = 1) – this does not
    affect the value returned by ``evaluate_sequence``.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # for a*a
    L_grad = _next_pow2(conv_len + n - 1)        # for gradient convolutions

    # --------------------------------------------------------------
    # 2. Initial vector (simplex‑projected)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.pad(a0, (0, n - a0.size))
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    total = a0.sum()
    if total > 0.0:
        a0 = a0 / total
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(best_a, n, fft_len, conv_len)
    cur_J = best_J
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Timing set‑up
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)   # a couple of seconds safety
    total_time = deadline - start

    # Phase split (seconds)
    phase1_end = start + 0.30 * total_time   # gradient warm‑up
    phase2_end = start + 0.90 * total_time   # simulated annealing
    # The remaining 10 % is final refinement

    # Adam hyper‑parameters (shared across phases)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    iteration = 1
    last_improve = start
    phase2_start = None   # will be set the first time we enter SA

    # --------------------------------------------------------------
    # 4. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        # ------------------------------------------------------------------
        # 4.1 Warm‑up gradient descent (smooth‑max)
        # ------------------------------------------------------------------
        if now < phase1_end:
            # linear increase of the smoothing parameter α
            prog = (now - start) / (phase1_end - start)
            alpha = 5.0 + 195.0 * prog           # α ∈ [5, 200]

            conv = _conv_full(a, fft_len, conv_len)
            max_conv = conv.max()
            shift = conv - max_conv               # ≤ 0
            w_raw = np.exp(alpha * shift)
            w_sum = w_raw.sum()

            if w_sum == 0.0:                     # degenerate (should not happen)
                # use sub‑gradient on the active set
                tol = max(1e-12, max_conv * 1e-12)
                active = np.where(conv >= max_conv - tol)[0]
                grad = np.zeros_like(a)
                for p in active:
                    start_i = max(0, p - (n - 1))
                    end_i = min(p, n - 1)
                    idx = np.arange(start_i, end_i + 1)
                    grad[idx] += 2.0 * a[p - idx]
            else:
                w = w_raw / w_sum
                a_rev = a[::-1]
                conv2 = conv_fft(w, a_rev, L_grad)        # length conv_len + n - 1
                grad = 2.0 * conv2[n - 1:n - 1 + n]        # keep n entries

            grad_J = 2.0 * n * grad                        # ∇J

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_J
            v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)

            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)

            lr = 0.15                                       # a modest step size
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a_proposed = a - step
            a_proposed = proj_simplex(a_proposed)

            new_J = compute_J(a_proposed, n, fft_len, conv_len)

            if new_J < cur_J:
                a = a_proposed
                cur_J = new_J
                if new_J < best_J - 1e-12:
                    best_J = new_J
                    best_a = a.copy()
                    last_improve = now
                    elapsed = now - start
                    print(f"[{elapsed:6.2f}s] GD improvement → J = {best_J:.6f}")

            iteration += 1
            continue

        # ------------------------------------------------------------------
        # 4.2 Simulated annealing exploration
        # ------------------------------------------------------------------
        if now < phase2_end:
            if phase2_start is None:
                phase2_start = now
                T0 = cur_J
                T_end = 1e-8

            # temperature (geometric decay)
            prog = (now - phase2_start) / (phase2_end - phase2_start)
            T = T0 * (T_end / T0) ** prog

            # move‑size schedule (large initially, tiny later)
            move_start, move_end = 0.20, 0.001
            max_move_frac = move_start * (1.0 - prog) + move_end * prog

            # choose a source index weighted by its mass (so we move from larger entries)
            probs = a / a.sum()
            i = rng.choice(n, p=probs)
            # choose a distinct destination uniformly
            j = rng.integers(0, n - 1)
            if j >= i:
                j += 1

            # amount to transfer
            max_delta = max_move_frac * a[i]
            if max_delta <= 0.0:
                continue
            delta = rng.random() * max_delta

            old_ai = a[i]
            old_aj = a[j]
            a[i] = old_ai - delta
            a[j] = old_aj + delta

            new_J = compute_J(a, n, fft_len, conv_len)

            accept = False
            if new_J < cur_J:
                accept = True
            else:
                prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                cur_J = new_J
                if new_J < best_J - 1e-12:
                    best_J = new_J
                    best_a = a.copy()
                    last_improve = now
                    elapsed = now - start
                    print(f"[{elapsed:6.2f}s] SA new best → J = {best_J:.6f}")
            else:
                # revert move
                a[i] = old_ai
                a[j] = old_aj

            # occasional jitter if we have been stagnant for a while
            if now - last_improve > 15.0:
                sigma = 0.05 * (1.0 - prog) + 0.005 * prog
                noise = np.exp(rng.normal(0.0, sigma, size=n))
                a = proj_simplex(best_a * noise)
                cur_J = compute_J(a, n, fft_len, conv_len)
                last_improve = now
                elapsed = now - start
                print(f"[{elapsed:6.2f}s] Jitter – J = {cur_J:.6f}")

            iteration += 1
            continue

        # ------------------------------------------------------------------
        # 4.3 Final high‑α refinement
        # ------------------------------------------------------------------
        # (only reached in the last ~10 % of the budget)
        alpha = 300.0
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()
        shift = conv - max_conv
        w_raw = np.exp(alpha * shift)
        w_sum = w_raw.sum()

        if w_sum == 0.0:
            tol = max(1e-12, max_conv * 1e-12)
            active = np.where(conv >= max_conv - tol)[0]
            grad = np.zeros_like(a)
            for p in active:
                start_i = max(0, p - (n - 1))
                end_i = min(p, n - 1)
                idx = np.arange(start_i, end_i + 1)
                grad[idx] += 2.0 * a[p - idx]
        else:
            w = w_raw / w_sum
            a_rev = a[::-1]
            conv2 = conv_fft(w, a_rev, L_grad)
            grad = 2.0 * conv2[n - 1:n - 1 + n]

        grad_J = 2.0 * n * grad

        # Adam with a smaller learning rate
        m = beta1 * m + (1.0 - beta1) * grad_J
        v = beta2 * v + (1.0 - beta2) * (grad_J ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr_refine = 0.05
        step = lr_refine * m_hat / (np.sqrt(v_hat) + eps)

        a_proposed = a - step
        a_proposed = proj_simplex(a_proposed)

        new_J = compute_J(a_proposed, n, fft_len, conv_len)
        if new_J < cur_J:
            a = a_proposed
            cur_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a.copy()
                elapsed = now - start
                print(f"[{elapsed:6.2f}s] Final refinement → J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 5. Sanitise result
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = best_a.sum()
    if s == 0.0:
        best_a = np.full(n, 1.0 / n)
    else:
        best_a = best_a / s

    print(f"Best J found: {best_J:.6f}")
    return [float(v) for v in best_a]
