"""
LP‑guided alternating optimisation.

We start from a random (Dirichlet) distribution on the probability simplex.
For a given sequence `a` we compute `M = max(conv(a,a))`.  The linear program

    maximise   Σ g_i
    subject to  a * g ≤ M               (pointwise)
                g_i ≥ 0

finds a non‑negative vector `g` with the largest possible ℓ₁‑norm while still
respecting the same convolution bound `M`.  If `λ = Σ g_i > 1` then the
scaled vector `ĝ = g/λ` satisfies `a * ĝ ≤ M/λ < M`.  Mixing the current
sequence with `ĝ` (convex combination) strictly reduces the peak of the
self‑convolution.  Repeating this step yields a monotone descent in the
objective `evaluate_sequence`.

The routine below alternates the LP step with a short stochastic
refinement (multiplicative Gaussian jitter) and returns the best sequence
found within the allotted time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# Simplex projection (Wang & Carreira‑Perpiñán 2013)
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # fallback – all equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Build the convolution matrix A such that (A @ g)[k] = (a * g)[k]
def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Given a (length n) array `a`, return a dense matrix A of shape
    (2n‑1, n) with   A[k, i] = a[k‑i]  if 0 ≤ k‑i < n, else 0.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.empty((rows, n), dtype=np.float64)
    A.fill(0.0)
    for i in range(n):
        A[i:i + n, i] = a
    return A


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    LP‑guided steepest‑descent optimiser for the step‑function problem.
    Returns a probability vector (list of floats) that minimises
    `evaluate_sequence` within the time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + float(budget_s) - 5.0               # keep a margin

    # --------------------------------------------------------------
    # 1. Choose a problem size ------------------------------------------------
    # --------------------------------------------------------------
    # If a previous construction exists, reuse its length (but keep it modest).
    if "height_sequence_1" in globals():
        try:
            n = max(1024, min(len(globals()["height_sequence_1"]), 2048))
        except Exception:
            n = 1536
    else:
        n = 1536                                            # a compromise size

    # --------------------------------------------------------------
    # 2. Initialise a starting point (Dirichlet draw) -------------------------
    # --------------------------------------------------------------
    a = rng.dirichlet(np.ones(n) * 0.5)                     # fairly smooth start
    a = proj_simplex(a)                                     # ensure exact simplex

    # Keep the best ever seen
    best_a = a.copy()
    best_J = evaluate_sequence(best_a.tolist())
    print(f"[0.00s] initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3. Main LP‑guided loop ---------------------------------------------------
    # --------------------------------------------------------------
    iteration = 0
    t_mix = 0.5                                            # mixing weight
    while time.monotonic() < deadline:
        # current convolution peak
        conv = np.convolve(a, a)
        M = conv.max()

        # ---------------------------------------------------------
        # Build linear constraints a * g ≤ M
        # ---------------------------------------------------------
        A = build_conv_matrix(a)                           # (2n‑1) × n
        rows = A.shape[0]
        b = np.full(rows, M, dtype=np.float64)

        # ---------------------------------------------------------
        # Linear programme: maximise Σ g_i  (i.e. minimise –Σ g_i)
        # ---------------------------------------------------------
        c = -np.ones(n, dtype=np.float64)                  # minimise –eᵀg
        bounds = [(0.0, None) for _ in range(n)]

        # Use HiGHS – fast interior‑point/simplex hybrid
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})

        if not res.success:
            # LP failed – fall back to a tiny random perturbation
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = proj_simplex(a * noise)
            cand_J = evaluate_sequence(cand.tolist())
            if cand_J < best_J:
                best_J, best_a = cand_J, cand.copy()
                print(f"[{time.monotonic() - start:.2f}s] jitter improved J = {best_J:.6f}")
            else:
                # nothing useful, abort the LP loop
                break

        g = res.x                                           # optimal g
        lam = g.sum()                                       # ℓ₁‑norm of g

        # ---------------------------------------------------------
        # If we managed to increase the ℓ₁‑norm, we can improve.
        # ---------------------------------------------------------
        improved = False
        if lam > 1.0 + 1e-9:
            g_normalised = g / lam                          # now Σ ĝ = 1
            # --- try the pure replacement first -----------------
            cand1 = g_normalised
            cand1_J = evaluate_sequence(cand1.tolist())
            if cand1_J < best_J:
                best_J, best_a = cand1_J, cand1.copy()
                improved = True
                print(f"[{time.monotonic() - start:.2f}s] LP replace improved J = {best_J:.6f}")

            # --- otherwise try a convex mix with the old vector
            if not improved:
                cand2 = (1.0 - t_mix) * a + t_mix * g_normalised
                cand2 = proj_simplex(cand2)                 # keep on simplex
                cand2_J = evaluate_sequence(cand2.tolist())
                if cand2_J < best_J:
                    best_J, best_a = cand2_J, cand2.copy()
                    improved = True
                    print(f"[{time.monotonic() - start:.2f}s] LP mix improved J = {best_J:.6f}")

        # ---------------------------------------------------------
        # If LP did not give a better point, perform a small random
        # jitter around the current best (helps escape tiny plateaux).
        # ---------------------------------------------------------
        if not improved:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = proj_simplex(best_a * noise)
            cand_J = evaluate_sequence(cand.tolist())
            if cand_J < best_J:
                best_J, best_a = cand_J, cand.copy()
                print(f"[{time.monotonic() - start:.2f}s] jitter improved J = {best_J:.6f}")

        # ---------------------------------------------------------
        # Prepare the next iteration (use the newest best as a)
        # ---------------------------------------------------------
        a = best_a.copy()
        iteration += 1

        # Occasionally shrink the mixing weight to avoid overshooting
        if iteration % 15 == 0:
            t_mix = max(0.1, t_mix * 0.9)

    # --------------------------------------------------------------
    # 4. Final fine‑grained stochastic search (tiny Gaussian jitter) ----------
    # --------------------------------------------------------------
    fine_deadline = time.monotonic() + min(30.0, (deadline - start) * 0.05)
    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.01, size=n))
        cand = proj_simplex(best_a * noise)
        cand_J = evaluate_sequence(cand.tolist())
        if cand_J < best_J:
            best_J, best_a = cand_J, cand.copy()
            print(f"[Fine {time.monotonic() - start:.2f}s] improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Return the final sequence (already on the simplex) -------------------
    # --------------------------------------------------------------
    # Clip to the hard bounds the evaluator imposes – purely defensive.
    best_a = np.clip(best_a, 0.0, 1000.0)
    # Normalise one last time (should already sum to 1)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    print(f"[{time.monotonic() - start:.2f}s] finished – best J = {best_J:.6f}")
    return [float(v) for v in best_a]
