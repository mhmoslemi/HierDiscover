def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the linear self‑convolution of *vec*."""
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Full objective 2·n·max(conv(vec,vec))."""
    return 2.0 * n * conv_max(vec, fft_len, conv_len)

    """
