The routine works in three stages within the allotted time:

1. **Annealed soft‑max Adam** – the same smooth‑max idea as the

   Periodic jitter restarts keep the search from stalling.

   temperature is very high we switch to the true sub‑gradient of the
   max‑convolution (β→∞).  The EG update keeps the iterates on the
   probability simplex without any projection step.

3. **Random‑candidate sweep** – any remaining time is spent sampling
   diverse families (Dirichlet, geometric decays, linear ramps,
   piece‑wise constant) and keeping the best vector seen.

All helper functions are defined at module level; the optimiser respects
the 1000 s budget, prints progress, and finally returns the best
non‑negative sequence (normalised to sum = 1) as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns the first `conv_len` entries of the linear convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute soft‑max weights w_i = exp(beta*(conv_i - conv_max)) / Σ.
    The subtraction of conv_max guarantees numerical stability.
    """
    c_max = conv.max()
    exp_part = np.exp(beta * (conv - c_max))
    total = exp_part.sum()
    if total == 0.0:                     # safety fallback
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / total


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _grad_softmax(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·conv_i  w.r.t. a, where conv = a*a.
    Uses the correlation identity ∂conv_i/∂a = shifted copy of a,
    giving a gradient = 2 * correlation(a, weights) on the appropriate slice.
    """
    n = a.size
    # Full linear correlation length = n + len(weights) - 1 = 3n‑2
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    A = np.fft.rfft(a, fft_len)
    W = np.fft.rfft(weights, fft_len)
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len)[:corr_len]

    # The slice that corresponds to the gradient (see derivation in docs)
    start = 2 * n - 2               # = len(weights)-1
    grad = 2.0 * corr_full[start:start + n]
    return grad

                if change > 0:
                    lengths[idx] += 1
                    diff -= 1
                else:
                    if lengths[idx] > 1:
                        lengths[idx] -= 1
                        diff += 1
            heights = rng.dirichlet(np.ones(K))
            a = np.repeat(heights, lengths)[:n]
        else:
            # A small Gaussian bump centred in the middle
            x = np.arange(n, dtype=float)
            centre = (n - 1) / 2.0
            sigma = rng.uniform(n / 12.0, n / 6.0)
            a = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
            a /= a.sum()
        cur_J = _exact_J(a)
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] Random sweep new best J = {global_best_J:.6f}")

    # --------------------------------------------------------------
    # 8. Return the best sequence (clipped, normalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    final_J = float(2.0 * n * np.convolve(best_vec, best_vec).max())
    print(f"[{budget_s:.0f}s] Finished – final best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
