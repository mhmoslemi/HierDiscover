def solve_lp_greedy(a, M):
    n = len(a)
    capacity = np.full(2*n-1, M, dtype=np.float64)
    g = np.zeros(n, dtype=np.float64)
    # Process indices in order of a ascending (prefer small a values to maximize sum)
    order = np.argsort(a)
    # Precompute mask and a_nonzero = a[mask]
    mask = a > 1e-12
    a_nonzero = a[mask]
    # Use vectorized operations for each i
    for i in order:
        # capacity segment:
        seg = capacity[i:i+n]  # shape (n,)
        # compute ratios where a>0
        ratios = seg[mask] / a_nonzero
        # epsilon = min(ratios)
        eps = ratios.min()
        if eps <= 0:

        g[i] = eps
        # update capacity

        # we can update seg directly: seg[mask] -= eps * a_nonzero
        seg[mask] -= eps * a_nonzero

    return g

def conv_max(v):
    n = v.size
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())
    # compute rfft
    A = np.fft.rfft(v, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len].max()

"""
Propose a candidate sequence that improves the upper bound on
evaluate_sequence.

Method
------
1. Initialise a probability vector a (non‑negative, sum = 1) – either from
   the global `height_sequence_1` (if present) or a fresh Dirichlet draw.
2. Repeatedly perform a *Frank–Wolfe style* refinement:
   * Compute M = max_k (a*a)_k.
   * Solve the linear programme
         maximise   Σ g_i
         subject to a∗g ≤ M   (component‑wise)
                    g_i ≥ 0
     The LP has a very special structure; it can be solved exactly by a
     greedy “water‑filling’’ algorithm that runs in O(n²) without any
     external solver.
   * Rescale g to a probability vector (Σ g = 1) and line‑search the mixing
     weight t∈[0,1] for a_new = (1−t)·a + t·g, keeping the best J.
   * Accept the new a if J improves; otherwise stop the refinements.
3. A short stochastic fine‑tuning phase (multiplicative log‑normal noise)
   finishes the search.
4. Return the best probability vector found (values are already in the
   allowed range, sum > 0.01).  The evaluation function is scale‑invariant,
   therefore we keep the sum equal to 1.

The algorithm runs only with NumPy (no external LP solver) and respects
the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_max(v: np.ndarray) -> float:
    """Full linear convolution of v with itself and return the maximal entry."""
    n = v.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(v, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return float(conv[:conv_len].max())


def greedy_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  maximise Σ g_i   subject to a∗g ≤ M, g≥0.
    Because the constraint matrix is a Toeplitz matrix with non‑negative
    entries, the optimal solution can be obtained by a greedy water‑filling
    algorithm that processes the columns of the matrix in order of increasing
    a_i.
    """
    n = a.shape[0]
    # remaining capacity for each convolution index
    capacity = np.full(2 * n - 1, M, dtype=np.float64)
    g = np.zeros(n, dtype=np.float64)

    # columns with non‑zero a‑coefficients are the only ones that affect
    # the capacities – store a mask for speed.
    mask = a > 1e-12
    a_pos = a[mask]          # positive entries of a (1‑D)
    # process indices of a in increasing order (small coefficients first)
    order = np.argsort(a, kind="mergesort")

    for i in order:
        # segment of capacities that column i touches
        seg = capacity[i : i + n]          # view, shape (n,)
        # ratios = remaining capacity / coefficient for all active rows
        ratios = seg[mask] / a_pos
        eps = ratios.min()
        if eps <= 0.0:
            continue
        g[i] = eps
        # consume capacity
        seg[mask] -= eps * a_pos
        # the view `seg` already writes back into `capacity`
    return g


def objective(v: np.ndarray, n: int) -> float:
    """J(v) = 2·n·max(v∗v) for a probability vector (Σv = 1)."""
    return 2.0 * n * conv_max(v)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimizes `evaluate_sequence`.  The routine uses a deterministic Frank–
    Wolfe style refinement followed by a short stochastic polishing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            # clamp extreme lengths for speed / memory
            n = max(256, min(old.size, 5000))
            a0 = old[:n].copy()
        except Exception:
            n = 2048
            a0 = rng.dirichlet(np.ones(n))
    else:
        n = 2048
        a0 = rng.dirichlet(np.ones(n))

    # ensure we are on the simplex
    a0 = np.maximum(a0, 0.0)
    a0 = proj_simplex(a0)
    a = a0.copy()

    best_J = objective(a, n)
    best_a = a.copy()
    start_time = time.monotonic()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    print(f"[0.00s] Initial J = {best_J:.6f} (n = {n})")

    # --------------------------------------------------------------
    # 2. Frank–Wolfe style refinement (up to ~100 iterations or time‑out)
    # --------------------------------------------------------------
    iter_cnt = 0
    while time.monotonic() < deadline and iter_cnt < 120:
        iter_cnt += 1
        M = conv_max(a)                     # max of a*a
        # Solve the LP (greedy, O(n²))
        g_raw = greedy_max_sum(a, M)
        sum_g = g_raw.sum()
        if sum_g <= 0.0:
            # degenerate – abort the refinement loop
            break
        # normalise to a probability vector
        g = g_raw / sum_g

        # ----------------------------------------------------------
        # 3. Line search for the best mixing weight.
        # ----------------------------------------------------------
        # candidates – finer near zero because we expect only a small move
        t_candidates = np.concatenate(
            (np.linspace(0.0, 0.3, 7)[1:],   # 0.05 … 0.30
             np.linspace(0.4, 1.0, 7))       # 0.40 … 1.00
        )
        best_local_J = best_J
        best_local_a = None
        best_t = 0.0

        for t in t_candidates:
            a_new = (1.0 - t) * a + t * g
            J_new = objective(a_new, n)
            if J_new < best_local_J - 1e-12:
                best_local_J = J_new
                best_local_a = a_new
                best_t = t

        if best_local_a is not None:
            # accept the improvement
            a = best_local_a
            best_J = best_local_J
            best_a = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Iter {iter_cnt:3d} – "
                  f"t={best_t:.3f}  J={best_J:.6f}")
        else:
            # no improvement – stop the deterministic phase
            break

    # --------------------------------------------------------------
    # 4. Final stochastic polishing (log‑normal perturbations)
    # --------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.015, size=n))   # smaller sigma
        cand = proj_simplex(best_a * noise)
        cand_J = objective(cand, n)
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_a = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Normalise to the range required by evaluate_sequence
    #    (the function is scale‑invariant, we simply keep sum≈1)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    # Ensure a tiny mass to avoid the sum‑too‑small guard
    s = best_a.sum()
    if s < 1e-12:
        best_a[:] = 1.0 / n
    else:
        best_a /= s

    return [float(v) for v in best_a]
