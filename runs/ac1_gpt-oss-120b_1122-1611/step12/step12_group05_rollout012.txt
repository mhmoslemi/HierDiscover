import numpy as np
import time

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def proj_simplex(v):
    # Euclidean projection onto simplex
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def conv_fft(x, y, L):
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[:len(x) + len(y) - 1]

def _conv_full(a, fft_len, conv_len):
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

    combines an exponentiated‑gradient descent on a smooth approximation of the
    max‑convolution with occasional random jitter and a simple exchange move,
    and finally returns the best vector found.

            n = max(256, min(n, 8192))

    # ------------------------------------------------------------------

        frac = elapsed / total_time

        # ----- schedule parameters ------------------------------------
        alpha = alpha_start + (alpha_end - alpha_start) * frac
        eta   = eta_start   + (eta_end   - eta_start)   * frac
        jitter_sigma = jitter_sigma_start + (jitter_sigma_end - jitter_sigma_start) * frac

        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()

        w_raw = np.exp(alpha * shift)               # all ≤ 1, max = 1
        w_sum = w_raw.sum()

            tol = max(1e-12, max_conv * 1e-12)
            active = np.where(conv >= max_conv - tol)[0]
            w = np.zeros_like(conv)
            w[active] = 1.0 / len(active)
        else:

        grad_smooth = 2.0 * conv2[n-1:n-1+n]               # shape (n,)

        # ----- exponentiated‑gradient update -------------------------
        # a_i ← a_i * exp(-η * grad_i)
        a = a * np.exp(-eta * (2.0 * n * grad_smooth))

        if rng.random() < exchange_prob:

            delta = min(a[i], rng.random() * 0.008)   # move at most 0.8 % of total probability
            a[i] -= delta
            a[j] += delta

        a = proj_simplex(a)

        # ----- evaluate current J ------------------------------------
        conv = _conv_full(a, fft_len, conv_len)
        cur_J = 2.0 * n * conv.max()

            best_J = cur_J
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

    if total == 0.0:

    else:

"""
Exponentiated‑gradient optimisation on a smooth approximation of
the max‑convolution.  The algorithm works on the probability simplex
(∑a_i=1, a_i≥0) using a soft‑max surrogate for the max term,
updates the weights with a multiplicative (mirror‑descent) step and
periodically applies a small random jitter to escape plateaus.
The whole search runs for ≤ budget_s seconds and returns the best
vector found, clipped to the allowed range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries equal or zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of ``x`` and ``y`` using an FFT of length ``L``."""
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[: len(x) + len(y) - 1]


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT of length ``fft_len``."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

# ----------------------------------------------------------------------

    else:

    # --------------------------------------------------------------
    # 2. Initialise the search vector on the simplex.

    if "height_sequence_1" in globals():
