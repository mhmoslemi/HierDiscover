"""
Propose a step‑function (non‑negative vector on the simplex) that minimises
`evaluate_sequence`.  

The algorithm is a multi‑start greedy “swap‑descent” that works directly on
the original non‑smooth objective  J(a)=2·n·max(conv(a,a)).
For a given vector a we

  * compute the full auto‑convolution b = a*a,
  * identify the active set 𝒜 = {k : b_k = max(b)},
  * form the sub‑gradient g = 2·corr(w,a) where w is the normalised indicator of
    𝒜 (w_k = 1/|𝒜| for k∈𝒜, else 0),

  * move a small amount of probability mass from the index with the largest
    gradient component (most harmful for the objective) to the index with the
    smallest component (most beneficial).  The move is accepted
    deterministically if it improves J, otherwise it may be accepted with a
    Metropolis probability controlled by a simple linear cooling schedule.

Several diverse random Dirichlet seeds are tried; each seed undergoes a short
swap‑descent, and the best trajectory is continued for a longer refinement.
Finally a cheap simulated‑annealing pairwise‑swap polish is applied.

All heavy linear algebra is performed with real FFTs (O(n log n)), so even
n≈4000 fits easily into the 1000 s budget.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = rfft(a, fft_len)
    conv = irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Efficiently compute 2·corr(weight, a) via FFT.
    weight length = 2·n‑1, a length = n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = rfft(weight, fft_len_corr)
    A_rev_hat = rfft(a_rev, fft_len_corr)
    full = irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)) for a probability vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`. Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size ------------------------------------------------
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            # keep the size reasonable for the LP‑free method
            n = max(512, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlation

    # ------------------------------------------------------------------
    # 2. Build a pool of diverse initial probability vectors -------------
    # ------------------------------------------------------------------
    init_cands = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.90, 0.94, 0.97, 0.985, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # linear ramps (ascending / descending)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 12.0, n / 9.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws with a range of concentration parameters
    for alpha in [0.001, 0.005, 0.01, 0.05, 0.1, 0.3, 0.6, 1.0]:
        for _ in range(2):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # use the previous construction if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # ------------------------------------------------------------------
    # 3. Pick several promising seeds (fast evaluation) -----------------
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 5.0   # keep a few seconds for wrap‑up
    total_time = deadline - start_time

    # evaluate all candidates quickly and keep the ten best
    cand_scores = []
    for cand in init_cands:
        J = _objective(cand, n, fft_len, conv_len)
        cand_scores.append((J, cand))
    cand_scores.sort(key=lambda t: t[0])
    top_seeds = [vec.copy() for (_, vec) in cand_scores[:8]]
    best_J = cand_scores[0][0]
    best_vec = cand_scores[0][1].copy()
    print(f"[{0.0:6.2f}s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Swap‑descent on each seed ---------------------------------------
    # ------------------------------------------------------------------
    # Parameters for the descent
    max_iters_per_seed = 2000            # enough for a quick rough improvement
    eps_active = 1e-12                  # tolerance for the active set
    move_factor = 0.5                   # fraction of donor mass moved each step
    T_initial_factor = 0.1              # initial temperature = factor * current best
    T_final = 1e-8

    # helper to compute J and convolution
    def _conv_and_J(vec):
        conv = _conv_full(vec, fft_len, conv_len)
        return conv, float(2.0 * n * conv.max())

    # run a short swap‑descent from each seed
    for seed_idx, cur_vec in enumerate(top_seeds):
        cur_conv, cur_J = _conv_and_J(cur_vec)
        best_local_J = cur_J
        best_local_vec = cur_vec.copy()
        last_improve = time.monotonic()

        for it in range(max_iters_per_seed):
            now = time.monotonic()
            if now >= deadline:
                break

            # ---- active set (indices of the maximal convolution value) ----
            max_conv = cur_conv.max()
            active = np.where(cur_conv >= max_conv - max(eps_active, max_conv * eps_active))[0]
            if active.size == 0:
                active = np.array([np.argmax(cur_conv)])
            w = np.zeros_like(cur_conv)
            w[active] = 1.0 / active.size

            # ---- sub‑gradient of the max‑operator -------------------------
            grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)

            # ---- decide donor and receiver --------------------------------
            i_max = int(np.argmax(grad))   # most harmful component
            i_min = int(np.argmin(grad))   # most beneficial component
            if i_max == i_min or cur_vec[i_max] <= 0.0:
                # cannot move mass, abort this seed early
                break

            # ---- amount of mass to move ----------------------------------
            eps = cur_vec[i_max] * move_factor
            eps = min(eps, cur_vec[i_max])   # safety

            # propose new vector by a simple swap
            cand = cur_vec.copy()
            cand[i_max] -= eps
            cand[i_min] += eps

            # ---- evaluate new candidate -----------------------------------
            cand_conv, cand_J = _conv_and_J(cand)

            # ---- temperature‑controlled acceptance (Metropolis) ----------
            if cand_J < cur_J:
                accept = True
            else:
                # linear temperature schedule
                elapsed = now - start_time
                T = (T_initial_factor * cur_J) * (deadline - now) / total_time + T_final * (now - start_time) / total_time
                prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
                accept = rng.random() < prob

            if accept:
                cur_vec = cand
                cur_conv = cand_conv
                cur_J = cand_J
                if cur_J < best_local_J - 1e-12:
                    best_local_J = cur_J
                    best_local_vec = cur_vec.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] Seed {seed_idx} new local best J = {cur_J:.6f}")

            # ---- stagnation handling (jitter) -----------------------------
            if now - last_improve > 30.0:
                # random multiplicative jitter
                noise = np.exp(rng.normal(0.0, 0.08, size=n))
                cur_vec = best_local_vec * noise
                cur_vec = proj_simplex(cur_vec)
                cur_conv, cur_J = _conv_and_J(cur_vec)
                last_improve = now
                print(f"[{now - start_time:6.2f}s] Seed {seed_idx} jitter – reset from best local J = {best_local_J:.6f}")

        # keep the best overall vector seen so far
        if best_local_J < best_J - 1e-12:
            best_J = best_local_J
            best_vec = best_local_vec.copy()
            print(f"[{now - start_time:6.2f}s] Global best updated to J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Long refinement of the current best (swap‑descent + Adam) -------
    # ------------------------------------------------------------------
    cur_vec = best_vec.copy()
    cur_conv, cur_J = _conv_and_J(cur_vec)
    best_J = cur_J
    best_vec = cur_vec.copy()
    last_improve = time.monotonic()

    # Adam‑style parameters (used only after a while)
    use_adam = False
    lr0_adam = 0.6
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    m = np.zeros_like(cur_vec)
    v = np.zeros_like(cur_vec)

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / total_time           # [0,1]

        # ----------------------------------------------------------------
        # Switch to Adam after a few thousand primitive swaps
        # ----------------------------------------------------------------
        if not use_adam and iteration > 8000:
            use_adam = True
            print(f"[{elapsed:6.2f}s] Switching to Adam‑style optimisation")
            # Warm‑up moments for Adam
            m.fill(0.0)
            v.fill(0.0)

        # ----------------------------------------------------------------
        # Compute active set and sub‑gradient
        # ----------------------------------------------------------------
        max_conv = cur_conv.max()
        active = np.where(cur_conv >= max_conv - max(eps_active, max_conv * eps_active))[0]
        w = np.zeros_like(cur_conv)
        w[active] = 1.0 / active.size
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)

        # ----------------------------------------------------------------
        # Propose a new point
        # ----------------------------------------------------------------
        iteration += 1
        if use_adam:
            # Adam update (projected back onto simplex)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** iteration)
            v_hat = v / (1.0 - beta2 ** iteration)
            lr = lr0_adam * (1.0 - t_frac) + 1e-3
            step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
            proposal = cur_vec + step
        else:
            # primitive swap step (same as in the short runs)
            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))
            eps_move = cur_vec[i_max] * move_factor
            eps_move = min(eps_move, cur_vec[i_max])
            proposal = cur_vec.copy()
            proposal[i_max] -= eps_move
            proposal[i_min] += eps_move

        # projection onto the simplex (always needed)
        proposal = proj_simplex(proposal)
        prop_conv, prop_J = _conv_and_J(proposal)

        # ----------------------------------------------------------------
        # Acceptance test (Metropolis)
        # ----------------------------------------------------------------
        T = (T_initial_factor * best_J) * (deadline - now) / total_time + T_final * (now - start_time) / total_time
        if prop_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = proposal
            cur_conv = prop_conv
            cur_J = prop_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------------
        # Stagnation handling – jitter after 30 s without improvement
        # ----------------------------------------------------------------
        if now - last_improve > 30.0:

            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            cur_conv, cur_J = _conv_and_J(cur_vec)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Global jitter – best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Final cheap simulated‑annealing polish (pairwise mass swaps) ----
    # ------------------------------------------------------------------
    polish_deadline = start_time + float(budget_s) - 0.5

    T0_polish = cur_J * 0.01
    T_end_polish = 1e-9

    while time.monotonic() < polish_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / total_time
        T = T0_polish * (1.0 - t_frac) + T_end_polish * t_frac

    return [float(v) for v in best_vec]
