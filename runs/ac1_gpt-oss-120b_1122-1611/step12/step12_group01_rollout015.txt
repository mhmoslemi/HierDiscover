"""
Propose a candidate step‐function using a focused local refinement.
The algorithm starts from the best known construction (if any) or a uniform
distribution, then repeatedly:
1. computes the self‑convolution,
2. identifies the shift that attains the current maximum,
3. evaluates the sensitivity of the objective to each coefficient
   (the derivative of the maximal convolution entry),
4. moves a small amount of mass from the most sensitive locations to the
   least sensitive ones,
5. accepts the move if it improves the objective (or with a small
   simulated‑annealing probability).

The procedure is cheap (FFT based convolution, O(n log n) per iteration) and
fits comfortably into the 1000 s budget, typically performing many tens of
thousands of improvement steps and often beating the previous bound.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_and_max(vec: np.ndarray, fft_len: int, conv_len: int):
    """Return the full linear self‑convolution and its maximum."""
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv, float(conv.max())


def eval_obj(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective used by evaluate_sequence (fast surrogate)."""
    # the sequence is already non‑negative and ≤1000, so no extra checks needed
    conv, mx = conv_and_max(vec, fft_len, conv_len)
    s = vec.sum()
    if s < 0.01:            # replicate the “inf” guard of evaluate_sequence
        return np.inf
    return 2.0 * n * mx / (s * s)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a step‑function (list of non‑negative heights) that improves on
    the current best bound.  The method follows a fast local search that
    moves probability mass away from the indices most responsible for the
    current maximal autocorrelation.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size (use the length of the existing construction if
    #    available, otherwise fall back to a moderate size)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        n = max(256, min(n, 8192))
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)

    # --------------------------------------------------------------
    # 2. Initial vector – start from the known best or from a uniform
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            # pad / truncate to the selected length
            if v.size < n:
                v = np.pad(v, (0, n - v.size), mode='constant')
            else:
                v = v[:n]
        # project onto simplex (sum = 1)
        v = proj_simplex(v)
    else:
        v = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 3. Initial evaluation
    # --------------------------------------------------------------
    best_vec = v.copy()
    best_J = eval_obj(best_vec, n, fft_len, conv_len)

    start = time.monotonic()
    deadline = start + float(budget_s) - 0.5          # a small safety margin
    total_time = deadline - start

    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main local‑search loop
    # --------------------------------------------------------------
    # Parameters governing the size of moves and the simulated‑annealing schedule
    move_factor = 0.25               # fraction of the donor’s mass that can be moved
    temperature = 0.02
    cooling = 0.9998

    iteration = 0
    last_improve = start

    while time.monotonic() < deadline:
        # recompute convolution and locate the dominant shift
        conv, cur_max = conv_and_max(best_vec, fft_len, conv_len)
        max_idx = int(np.argmax(conv))
        shift = max_idx - (n - 1)                # ‑(n‑1)…+(n‑1)

        # compute the sensitivity (partial derivative of the maximal
        # convolution entry) for every index
        w = np.zeros(n, dtype=np.float64)
        if shift >= 0:
            k = shift
            w[:n - k] += best_vec[k:]            # a[i+k]
            w[k:] += best_vec[:n - k]            # a[i‑k]
        else:
            k = -shift
            w[:n - k] += best_vec[k:]            # a[i‑k] (shift is negative)
            w[k:] += best_vec[:n - k]            # a[i+k]

        # pick a donor (high sensitivity) and a receiver (low sensitivity)
        # use a small random subset to keep the operation cheap
        top_k = max(2, int(0.01 * n))            # 1 % of the entries
        low_k = max(2, int(0.01 * n))

        high_candidates = np.argpartition(-w, top_k)[:top_k]
        low_candidates = np.argpartition(w, low_k)[:low_k]

        donor = int(rng.choice(high_candidates))
        receiver = int(rng.choice(low_candidates))

        # ensure we actually move something
        available = best_vec[donor]
        if available <= 1e-12:
            continue

        # propose a transfer of a modest amount of mass
        delta = min(available,
                    move_factor * available * rng.random())
        cand = best_vec.copy()
        cand[donor] -= delta
        cand[receiver] += delta

        # optional clipping (the values stay far below 1000, but we stay safe)
        np.clip(cand, 0.0, 1000.0, out=cand)

        # evaluate the candidate
        cand_J = eval_obj(cand, n, fft_len, conv_len)

        # acceptance rule: always take improvement, otherwise accept with
        # a probability that slowly cools down (simulated annealing)
        accept = False
        if cand_J < best_J - 1e-12:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(cand_J - best_J) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            best_vec = cand
            best_J = cand_J
            last_improve = time.monotonic()
            if cand_J < best_J - 1e-9:
                # (the condition above already ensures strict improvement)
                pass
            print(f"[{time.monotonic() - start:6.2f}s] J = {best_J:.6f}")

        # cooling
        temperature *= cooling
        iteration += 1

        # occasional random restart if we have stagnated for a while
        if time.monotonic() - last_improve > 30.0:
            # keep the current best but re‑randomise a fraction of the vector
            mask = rng.random(n) < 0.10               # 10 % fresh noise
            fresh = rng.dirichlet(np.ones(n))[mask]
            best_vec[mask] = fresh
            best_vec = proj_simplex(best_vec)
            best_J = eval_obj(best_vec, n, fft_len, conv_len)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start:6.2f}s] Stagnation – partial restart, J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Final normalisation (clip, enforce sum>0 and project onto simplex)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    # Convert to plain python floats for the caller
    return [float(x) for x in best_vec]
