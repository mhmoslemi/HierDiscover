"""Propose a candidate step‐function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Initialise the vector on the probability simplex.
   If a previously found construction ``height_sequence_1`` exists we start
   from it, otherwise we use the uniform distribution.
2. Repeatedly (until the time budget is exhausted) perform a
   *sub‑gradient* step on the objective  
      J(a) = 2·n·max(conv(a,a))   with  Σa_i = 1 , a_i ≥ 0 .
   The sub‑gradient is built from the worst convolution entries:
   * pick the top *p* fraction of convolution values,
   * form a weight vector w (uniform on the selected indices),
   * compute  g = 2·(rev(a) * w)  (linear convolution, via FFT),
   * update a ← proj_simplex(a – η·g) with a small step size η.
   The fraction *p* and the step size are slowly decreased so that the
   optimisation focuses on fewer peaks and makes finer adjustments.
3. If the algorithm stagnates for a while we perform a small random
   perturbation around the current best solution.
4. After the main loop a short multiplicative‑jitter phase is run
   (exponential noise) to try to escape tiny plateaus.
5. The best vector found is finally normalised to sum to one,
   clipped to the allowed range and returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def compute_self_conv(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear self‑convolution of *a* via FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_gradient(a: np.ndarray, w: np.ndarray, rev_fft_len: int, n: int) -> np.ndarray:
    """
    Gradient of the weighted sum   Σ_k w_k·conv_k(a,a)   with respect to a.
    For any weight vector w (sum w_k = 1) the gradient is
        g_i = 2 * Σ_j a_j * w_{i+j}
    which is the (linear) convolution of rev(a) with w, extracted on the
    appropriate slice.
    """
    rev_a = a[::-1]                          # reversed a
    # FFTs of appropriate size
    Ra = np.fft.rfft(rev_a, rev_fft_len)
    Rw = np.fft.rfft(w, rev_fft_len)
    conv_rev = np.fft.irfft(Ra * Rw, n=rev_fft_len)[: (3 * n - 2)]
    start = n - 1
    h = conv_rev[start:start + n]            # h_i = Σ_j a_j w_{i+j}
    g = 2.0 * h
    return g


# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """Return a non‑negative step function that (hopefully) improves the bound.

    The routine uses a sub‑gradient descent on the L∞‑norm of the
    autocorrelation, combined with simple restarts and a final jitter phase.
    It respects the 1000 s time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size & initial vector
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        # use the previously discovered construction as a warm‑start
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            n = 256
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = a.size
            a = a / a.sum()
    else:
        # fallback: uniform vector of a moderate size
        n = 2048
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = proj_simplex(a)                     # guarantee feasibility

    # --------------------------------------------------------------
    # 2. Pre‑computed FFT sizes
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    self_fft_len = next_pow2(conv_len)                     # for a*a
    rev_fft_len = next_pow2(3 * n - 2)                     # for rev(a)*w

    # --------------------------------------------------------------
    # 3. Initial evaluation
    # --------------------------------------------------------------
    conv = compute_self_conv(a, self_fft_len, conv_len)
    best_J = 2.0 * n * conv.max()
    best_a = a.copy()
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a little reserve
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Main optimisation loop (sub‑gradient descent)
    # --------------------------------------------------------------
    # parameters – they shrink slowly as the search proceeds
    step_scale = 0.02                     # base step size (will be shrunk)
    top_fraction = 0.05                   # start focusing on the top 5 % of entries
    min_top_frac = 0.001
    decay = 0.9995                         # multiplicative decay per iteration
    min_step = 1e-4
    last_improve = start_time

    iteration = 0
    while time.monotonic() < deadline:
        # ----- evaluate current vector -----
        conv = compute_self_conv(a, self_fft_len, conv_len)
        cur_J = 2.0 * n * conv.max()
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] New best J = {best_J:.6f}")

        # ----- pick the worst convolution entries (top fraction) -----
        k_top = max(1, int(top_fraction * conv_len))
        if k_top < conv_len:
            # argpartition is O(conv_len); we only need the top k indices
            top_idx = np.argpartition(-conv, k_top - 1)[:k_top]
        else:
            top_idx = np.arange(conv_len)
        w = np.zeros(conv_len, dtype=np.float64)
        w[top_idx] = 1.0 / k_top               # uniform weight on the selected indices

        # ----- compute sub‑gradient -----
        grad = compute_gradient(a, w, rev_fft_len, n)

        # ----- update (additive step followed by projection) -----
        step = step_scale                      # constant step for this iteration
        a = a - step * grad
        a = proj_simplex(a)

        # ----- decay the meta‑parameters -----
        top_fraction = max(min_top_frac, top_fraction * decay)
        step_scale = max(min_step, step_scale * decay)

        # ----- stagnation handling (re‑initialise a little) -----
        now = time.monotonic()
        if now - last_improve > 30.0:
            # random jitter around the best solution found so far
            sigma = 0.08
            jitter = np.exp(rng.normal(0.0, sigma, size=n))
            a = best_a * jitter
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Stagnation restart")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Final stochastic jitter phase (few seconds left)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        # gradually shrink the jitter radius
        sigma = max(0.005, 0.2 * remaining / (budget_s - 2.0))
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_a * noise
        cand = proj_simplex(cand)
        conv_cand = compute_self_conv(cand, self_fft_len, conv_len)
        cand_J = 2.0 * n * conv_cand.max()
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_a = cand
            print(f"[{time.monotonic() - start_time:6.2f}s] Jitter improvement → J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Normalise / clip for safety and return
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_a = best_a / total

    return [float(v) for v in best_a]
