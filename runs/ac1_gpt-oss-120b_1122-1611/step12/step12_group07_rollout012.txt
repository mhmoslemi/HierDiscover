"""
Propose a step‐function (non‑negative vector) that minimises
J(a)=2·n·max(conv(a,a))  (the same quantity returned by evaluate_sequence
when the vector is normalised to sum=1).

Algorithm
---------
1. Choose a problem size n (use the length of the previously found
   `height_sequence_1` if it exists, otherwise 4096).
2. Build a diverse set of initial probability vectors (uniform,
   ramps, geometric, Gaussian, Dirichlet draws, sparse‑mask patterns …)
   and keep the one with the smallest J.
3. Optimise the vector on the probability simplex by **projected sub‑gradient**
   descent:
   * evaluate the full linear convolution via an FFT,
   * locate the shift(s) where the convolution attains its maximum,
   * compute the exact sub‑gradient of `max(conv)` (for a single active
     shift the gradient entry i is a[k‑i] if the index is valid, otherwise 0),
   * take a gradient step and project the result back onto the simplex
     (Duchi–Shalev‑Shwartz projection).
4. To avoid stagnation a small Gaussian jitter is added to the current
   vector if no improvement has been seen for a while; the step size
   decays with the iteration count.
5. When the time budget (≈1000 s) is exhausted the best vector seen so
   far is returned (clipped to the limits required by `evaluate_sequence`).

The method is deliberately lightweight (FFT based convolution + O(n) work
per iteration) so that many thousands of iterations can be performed
within the allowed time, giving a good chance of improving the previous
record (J≈1.513).
"""

import time
import numpy as np
from scipy import sparse  # only for the possible sparse‐mask initialisation

# ----------------------------------------------------------------------
# FFT based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT (length >= 2·len(a)-1)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Duchi et al., 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a real vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return a uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / float(rho)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective (fast version, assumes the vector already sums to 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return 2·n·max(conv(a,a)) .  a is assumed non‑negative and Σa = 1."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (for a single active shift)
# ----------------------------------------------------------------------
def _max_conv_subgrad(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv) at shift `k`.
    For shift k the gradient entry i is a[k‑i] (or 0 if out of range).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    # vectorised assignment
    idx_i = np.arange(i_start, i_end + 1, dtype=np.int64)
    idx_j = k - idx_i
    grad[idx_i] = a[idx_j]
    return 2.0 * grad   # factor 2 comes from derivative of a*a


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # ≥ conv_len, used for every FFT

    # --------------------------------------------------------------
    # 2. Build a large portfolio of diverse starting vectors
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    idx = np.arange(n, dtype=np.float64)

    # linear ramps
    init_candidates.append((n - idx) / n)                # decreasing
    init_candidates.append((idx + 1) / n)                # increasing

    # geometric decays (both directions)
    for r in (0.90, 0.93, 0.95, 0.97, 0.985, 0.99, 0.995, 0.998):
        g = r ** idx
        init_candidates.append(g / g.sum())
        init_candidates.append(g[::-1] / g.sum())

    # power‑law shapes
    for p in (-2.0, -1.5, -1.0, -0.5, 0.5, 1.0, 2.0):
        g = (idx + 1) ** p
        g = np.clip(g, 0.0, None)
        init_candidates.append(g / g.sum())

    # Gaussian bumps of different widths
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        g = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        init_candidates.append(g / g.sum())

    # Sparse random masks (few spikes)
    for _ in range(6):
        mask = np.zeros(n, dtype=np.float64)
        k = rng.integers(5, 30)                 # number of spikes
        positions = rng.choice(n, size=k, replace=False)
        mask[positions] = rng.random(k) + 0.5   # positive values
        init_candidates.append(mask / mask.sum())

    # Dirichlet draws (various concentration)
    for alpha in (0.05, 0.1, 0.2, 0.3, 0.6, 1.0, 2.0):
        for _ in range(8):
            init_candidates.append(rng.dirichlet(np.full(n, alpha)))

    # If a previous construction exists, add a (slightly perturbed) copy
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total > 0:
            v = v / total
        else:
            v = np.full(n, 1.0 / n)
        init_candidates.append(v)

    # --------------------------------------------------------------
    # 3. Pick the best initial candidate (exact J)
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        # guarantee non‑negativity and sum‑to‑one
        cand = np.maximum(cand, 0.0)
        if cand.sum() == 0:
            cand = np.full_like(cand, 1.0 / n)
        else:
            cand = cand / cand.sum()
        cur_J = _objective(cand, n, fft_len, conv_len)
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Projected sub‑gradient descent on the simplex
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # spare few seconds for finalisation
    iteration = 0
    last_improve = start_time

    # step‑size parameters (tuned empirically)
    lr0 = 0.6                # initial learning rate
    min_lr = 1e-5

    # parameters for jitter / restart
    jitter_interval = 30.0   # seconds without improvement → jitter
    jitter_sigma = 0.12

    # keep the globally best vector seen
    global_best_a = best_a.copy()
    global_best_J = best_J

    while time.monotonic() < deadline:
        # ---- 4.1 evaluate current vector (hard J, for logging)
        cur_J = _objective(best_a, n, fft_len, conv_len)

        now = time.monotonic()
        elapsed = now - start_time
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = best_a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ---- 4.2 stagnation handling
        if now - last_improve > jitter_interval:
            # inject Gaussian noise and re‑project onto the simplex
            noisy = best_a + rng.normal(0.0, jitter_sigma, size=n)
            best_a = _project_simplex(np.maximum(noisy, 0.0))
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart (best J = {global_best_J:.6f})")

        # ---- 4.3 compute convolution & locate the active shift(s)
        conv = _conv_full(best_a, fft_len, conv_len)
        max_val = conv.max()
        # pick one of the maximising shifts uniformly at random (helps escape plateaus)
        max_shifts = np.flatnonzero(conv == max_val)
        k_max = rng.choice(max_shifts)

        # ---- 4.4 sub‑gradient for the chosen shift
        grad = _max_conv_subgrad(best_a, k_max)   # shape (n,)

        # ---- 4.5 gradient step on the *probability vector* (projected)
        iteration += 1
        lr = max(min_lr, lr0 / np.sqrt(iteration))
        proposed = best_a - lr * grad
        # projection back onto the simplex keeps feasibility
        best_a = _project_simplex(proposed)

    # --------------------------------------------------------------
    # 5. Fine‑grained local search (SA‑style random mass swaps)
    # --------------------------------------------------------------
    # Continue for a few seconds after the main loop (if time permits)
    fine_deadline = deadline + 5.0    # at most a few extra seconds
    temperature = 0.02               # initial temperature for SA
    temp_decay = 0.9995
    step_scale = 0.05                # max fraction of mass moved in a single swap

    while time.monotonic() < fine_deadline:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # amount to move from i to j (cannot exceed a_i)
        max_move = min(step_scale * global_best_a[i], global_best_a[i])
        if max_move <= 0:
            continue
        delta = rng.random() * max_move
        cand = global_best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        cand_J = _objective(cand, n, fft_len, conv_len)
        if cand_J < global_best_J:
            global_best_J = cand_J
            global_best_a = cand
            print(f"[{time.monotonic() - start_time:6.2f}s] SA improvement -> J={global_best_J:.6f}")
        else:
            # accept with Metropolis probability
            prob = np.exp(-(cand_J - global_best_J) / max(temperature, 1e-12))
            if rng.random() < prob:
                global_best_a = cand

        temperature = max(1e-6, temperature * temp_decay)

    # --------------------------------------------------------------
    # 6. Return the best vector (clipped to the limits required by
    #    evaluate_sequence).  Normalise to sum=1 – the scaling is
    #    irrelevant for the evaluation metric.
    # --------------------------------------------------------------
    result = np.clip(global_best_a, 0.0, 1000.0)
    total = result.sum()
    if total == 0.0:
        result = np.full_like(result, 1.0 / n)
    else:
        result = result / total

    print(f"Finished. Best J = {global_best_J:.6f}")
    return [float(v) for v in result]
