"""
Gradient‑based refinement on the simplex.

* Starts from a good heuristic (arcsine density, uniform, or the previously
  discovered construction).
* Uses the exact sub‑gradient of  max(conv(a,a))  (the active convolution
  indices) and a simple back‑tracking line search.
* All updates are projected onto the probability simplex, keeping the
  scaling‑invariance of the objective.
* Stagnation triggers a small random perturbation to escape flat regions.
* A final short multiplicative‑jitter phase exploits any remaining budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex Σv_i = 1, v_i ≥ 0."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works only on 1‑D arrays")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two greater‑or‑equal to *x*."""
    return 1 << ((x - 1).bit_length())


def conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (full length)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray) -> float:
    """Exact evaluation of the objective (scaling‑invariant)."""
    n = a.shape[0]
    conv = conv_self(a)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def arcsine_distribution(n: int) -> np.ndarray:
    """Discretised arcsine density on {0,…,n‑1} (mid‑point rule) and normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = 1.0 / np.sqrt(x * (1.0 - x))
    a /= a.sum()
    return a

        A_fft = np.fft.rfft(a, fft_len_grad)
        P_fft = np.fft.rfft(p, fft_len_grad)
        conv_pa_full = np.fft.irfft(P_fft * A_fft, n=fft_len_grad)
        grad = 2.0 * conv_pa_full[n - 1:n - 1 + n]

        # ----- back‑tracking line search -----
        eta = 0.5                # initial step size
        improved = False
        while eta > 1e-7:
            cand = proj_simplex(a - eta * grad)
            cand_J = compute_J(cand)
            if cand_J < best_J - 1e-12:
                best_vec = cand
                best_J = cand_J
                last_improve_time = time.monotonic()
                improved = True
                print(f"[{time.monotonic() - start_time:6.2f}s] iter {iteration:4d} → J = {best_J:.6f}")
                break
            eta *= 0.5

        # ----- stagnation handling -----
        if not improved:
            # if we have been idle for >30 s, inject a tiny random perturbation
            if now - last_improve_time > 30.0:
                sigma = 0.02
                noise = np.exp(rng.normal(0.0, sigma, size=n))
                pert = proj_simplex(best_vec * noise)
                pert_J = compute_J(pert)
                if pert_J < best_J - 1e-12:
                    best_vec = pert
                    best_J = pert_J
                    last_improve_time = time.monotonic()
                    print(f"[{time.monotonic() - start_time:6.2f}s] jitter rescue → J = {best_J:.6f}")
                else:
                    # give up on this run and re‑initialise from a fresh Dirichlet draw
                    alpha = 10 ** rng.uniform(-2, 0.5)
                    best_vec = proj_simplex(rng.dirichlet(np.ones(n) * alpha))
                    best_J = compute_J(best_vec)
                    last_improve_time = time.monotonic()
                    print(f"[{time.monotonic() - start_time:6.2f}s] restart after stagnation → J = {best_J:.6f}")
        iteration += 1
        # safety break if we are hardly improving any more
        if iteration > 10000:
            # enough iterations – continue to jitter phase
            break

    # --------------------------------------------------------------
    # 5. Final multiplicative jitter (uses the remaining budget)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        # gradually shrink the jitter radius
        sigma = max(0.004, 0.2 * remaining / total_budget)
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_J = compute_J(cand)
        if cand_J < best_J - 1e-12:
            best_vec = cand
            best_J = cand_J
            print(f"[{time.monotonic() - start_time:6.2f}s] jitter improvement → J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Prepare final output (clip to the limits enforced by evaluate_sequence)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    # Return as plain Python list of floats
    return [float(v) for v in best_vec]
