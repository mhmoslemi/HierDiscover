```python
def _project_onto_simplex(v):
    # ensure nonnegative sum to 1
    # from Duchi et al.
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function sequence that (hopefully) beats the previous bound
≈1.5132.  The optimizer uses a *hard‑max* sub‑gradient descent on the
objective J = 2·n·max(a⋆a) (with ∑a = 1, a_i ≥ 0).  At every iteration we

1. compute the full self‑convolution by FFT,
2. locate the (few) indices where the convolution attains its maximum,
3. build the sub‑gradient of the maximum (a sum of shifted copies of a),
4. take a step opposite to the gradient,
5. project back onto the probability simplex (guaranteeing non‑negativity
   and unit sum).

A simple adaptive step‑size (geometric decay) and a jitter‑restart
prevent stagnation.  The routine respects a 1000 s time budget and returns
the best sequence discovered.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a real FFT (full length 2·n‑1)."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x ∈ ℝⁿ | x ≥ 0, ∑ x = 1}
    Implementation of Duchi et al. (ICML 2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1   # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w
