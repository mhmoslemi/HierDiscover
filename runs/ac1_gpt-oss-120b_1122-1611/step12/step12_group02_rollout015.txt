"""Propose a sequence via iterative linear programming improvement.

We start from the known best sequence (or a uniform one) normalized to
sum = 1.  At each iteration we solve the linear program

    maximise   1ᵀ g
    subject to  f * g ≤ M   (full convolution)
                g ≥ 0

where M = max(f * f).  The LP yields a vector g with a larger ℓ₁‑norm
while respecting the current convolution bound.  After scaling g back
to ℓ₁‑norm 1 we take a small convex combination with the current f,
project onto the probability simplex and keep the best vector found.
The process is repeated until the time budget expires.  A final
multiplicative‑gradient polish (the same as in the baseline) is applied
to the best vector before returning it.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _build_constraint_matrix(f: np.ndarray, M: float) -> sp.csr_matrix:
    """
    Build the (2·n‑1) × n matrix A such that  (f*g)[k] ≤ M  ⇔  A·g ≤ 1 .
    The matrix is Toeplitz with constant diagonals equal to f_i/M.
    """
    n = f.size
    # each diagonal d (0 ≤ d < n) contains the constant f[d]/M
    diagonals = [np.full(n, f[d] / M, dtype=np.float64) for d in range(n)]
    offsets = np.arange(n, dtype=np.int64)   # 0,1,…,n‑1
    A = sp.diags(diagonals, offsets, shape=(2 * n - 1, n), format="csr")
    return A

"""Propose a sequence via iterative linear‑programming refinement.

We start from the previously discovered sequence (or from a uniform one)
and normalise it to ℓ₁‑norm = 1.  At each iteration we solve the linear
program

    maximise   1ᵀ g
    subject to  f * g ≤ M               (full convolution)
                g ≥ 0

where M is the current maximum of f * f.  The optimum g has a larger
ℓ₁‑norm while respecting the same convolution bound.  After scaling g
back to unit ℓ₁‑norm we take a modest convex combination with the
current f, project onto the probability simplex and keep the best
vector found.  A few multiplicative‑gradient polishing steps (the same
as in the baseline) are applied at the end.  The routine respects the
1000 s wall‑clock budget and finally returns the best sequence.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
