"""
Iterative population‑mixing + Adam optimisation for minimising
`evaluate_sequence`.  The optimiser works on the probability simplex
(sum = 1, a_i≥0).  It builds a diverse initial pool (uniform,
geometric, Gaussian, Dirichlet draws, and the previous best if
available), exploits cheap random convex combinations of the best
candidates, then refines the current champion with a soft‑max‑smoothed
Adam step (β grows from 1 to 5000).  A short simulated‑annealing
pair‑swap polish finishes the search.  All heavy work uses real FFTs,
so n up to 8192 comfortably fits the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear auto‑convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(
    a: np.ndarray, weight: np.ndarray, fft_len_corr: int, n: int
) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·conv_k  w.r.t. a.
    weight length = 2·n‑1, a length = n.
    Returns a vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # the valid part starts at index n‑1
    grad = 2.0 * full[n - 1 : n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found
    within the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(height_sequence_1)
            n = max(512, min(_next_pow2(n0), 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted corr

    # --------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors
    # --------------------------------------------------------------
    init_cands: list[np.ndarray] = []

    # uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (both forward and reversed)
    for r in [0.97, 0.98, 0.99, 0.995, 0.998]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)
        init_cands.append(v[::-1])

    # linear ramps
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    init_cands.append(v[::-1])

    # Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in [n / 12.0, n / 9.0, n / 6.0]:
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws with a variety of concentrations; also sorted variants
    alphas = [0.001, 0.005, 0.01, 0.05, 0.1, 0.3, 0.6, 1.0]
    for alpha in alphas:
        for _ in range(15):
            v = rng.dirichlet(np.full(n, alpha))
            if rng.random() < 0.5:
                v = np.sort(v)[::-1]          # decreasing
            else:
                v = np.sort(v)                # increasing
            init_cands.append(v)

    # previous construction, if available
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # --------------------------------------------------------------
    # 3. Keep the best K initial candidates (fast evaluation)
    # --------------------------------------------------------------
    K = 20
    top: list[tuple[float, np.ndarray]] = []            # (J, vector)
    best_vec = None
    best_J = np.inf

    for cand in init_cands:
        J = _objective(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
        if len(top) < K:
            top.append((J, cand.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if J < top[-1][0]:
                top[-1] = (J, cand.copy())
                top.sort(key=lambda x: x[0])

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Population mixing & random exploration (≈25 % of the budget)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + float(budget_s)
    mix_deadline = start_time + 0.25 * float(budget_s)

    while time.monotonic() < mix_deadline:
        # pick two distinct elite candidates
        i, j = rng.choice(len(top), size=2, replace=False)
        vi = top[i][1]
        vj = top[j][1]
        w = rng.random()
        cand = w * vi + (1.0 - w) * vj

        # occasional jitter
        if rng.random() < 0.1:
            cand = cand * np.exp(rng.normal(0.0, 0.05, size=n))

        cand = proj_simplex(cand)
        J = _objective(cand, n, fft_len, conv_len)

        if J < best_J:
            best_J = J
            best_vec = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best after mixing J = {best_J:.6f}")

        # maintain the elite list
        if J < top[-1][0]:
            top[-1] = (J, cand.copy())
            top.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 5. Refined Adam optimisation on the current champion
    # --------------------------------------------------------------
    # reserve a few seconds for final polishing
    polish_reserve = 20.0
    opt_deadline = deadline - polish_reserve
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_vec_adam = cur_vec.copy()
    best_J_adam = cur_J

    # Adam hyper‑parameters
    lr0 = 0.8
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Soft‑max temperature schedule
    beta_start = 1.0
    beta_end = 5000.0

    # Metropolis temperature schedule
    T_initial = best_J_adam * 0.1
    T_final = 1e-9

    m = np.zeros_like(cur_vec)
    v = np.zeros_like(cur_vec)
    iteration = 0
    last_improve = time.monotonic()

    while time.monotonic() < opt_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total_opt = opt_deadline - start_time
        t_frac = (now - start_time) / total_opt

        # ----- convolution & current objective -----
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ----- update global best -----
        if cur_J < best_J_adam - 1e-12:
            best_J_adam = cur_J
            best_vec_adam = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Adam improvement J = {best_J_adam:.6f}")

        # ----- escape if stuck -----
        if now - last_improve > 20.0:
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec = best_vec_adam * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = _objective(cur_vec, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Adam jitter restart")

        # ----- soft‑max weights -----
        beta = beta_start * (beta_end / beta_start) ** t_frac
        shifted = conv - max_conv                     # stability shift
        w_exp = np.exp(beta * shifted)
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            w = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            w = w_exp / w_sum

        # ----- gradient of the smoothed max -----
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)

        # ----- Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        lr = lr0 * (1.0 - t_frac) + 1e-3
        step = -lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        proposal = cur_vec + step
        proposal = proj_simplex(proposal)

        prop_J = _objective(proposal, n, fft_len, conv_len)

        # ----- Metropolis acceptance -----
        if prop_J < cur_J:
            accept = True
        else:
            T = (T_initial * (opt_deadline - now) + T_final * (now - start_time)) / total_opt
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = proposal
            cur_J = prop_J

    # keep Adam's champion if it improved over the previous best
    if best_J_adam < best_J:
        best_J = best_J_adam
        best_vec = best_vec_adam.copy()
        print(f"[{time.monotonic() - start_time:6.2f}s] Best after Adam J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Final local polishing (pairwise mass swaps)
    # --------------------------------------------------------------
    polish_deadline = deadline - 2.0
    cur_vec = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    T0_polish = cur_J * 0.01
    T_end_polish = 1e-9

    while time.monotonic() < polish_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / (deadline - start_time)
        T = T0_polish * (1.0 - t_frac) + T_end_polish * t_frac

        # pick two indices and move a small amount of mass
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        eps_move = rng.random() * cur_vec[i] * 0.5
        if eps_move <= 0.0:
            continue

        cand = cur_vec.copy()
        cand[i] -= eps_move
        cand[j] += eps_move
        cand = np.clip(cand, 0.0, None)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        if cand_J < cur_J:
            cur_vec = cand
            cur_J = cand_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_vec = cur_vec.copy()
                print(f"[{elapsed:6.2f}s] Polishing improvement: J = {best_J:.6f}")
        else:
            # uphill acceptance (simulated annealing)
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur_vec = cand
                cur_J = cand_J

    # ------------------------------------------------------------------
    # 7. Normalise and return plain Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(x) for x in best_vec]
