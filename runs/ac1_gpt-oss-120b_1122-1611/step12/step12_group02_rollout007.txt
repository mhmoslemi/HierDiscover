"""
Exponentiated‑gradient descent with soft‑max smoothing.
----------------------------------------------------
* Works directly on the probability simplex (∑a_i = 1, a_i ≥ 0).
* Starts from several heuristic seeds:
  – uniform distribution,
  – “edge‑heavy’’ 1/(i+1)+1/(n‑i) shape,
  – the previously found sequence if available,
  – a few random Dirichlet draws.
* At each iteration the full convolution a∗a is computed with an FFT,
  a smooth approximation of the max‑operator
  (soft‑max with increasing “β’’) yields a weight vector,
  its correlation with a gives a sub‑gradient of the smoothed objective.
* An exponentiated‑gradient (multiplicative) update is performed,
  followed by a Euclidean projection onto the simplex.
* The β‑parameter grows slowly, turning the smooth objective into the
  true max‑objective; the step size η decays slowly.
* Every few thousand iterations a small random jitter (mix with a fresh
  Dirichlet sample) is injected to escape plateaus.
* The routine stops when the provided time budget is exhausted and returns
  the best sequence (lowest value of evaluate_sequence) encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (zero‑padded)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.

    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`. Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a convenient problem size (power of two, not too large)
    # --------------------------------------------------------------
    max_n = 16384          # upper bound to keep FFTs cheap
    min_n = 4096
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(min_n, _next_pow2(n0))
        except Exception:
            n = min_n
    else:
        n = min_n

    n = min(max_n, n)      # respect the upper bound

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)        # for weighted correlation

    # --------------------------------------------------------------
    # 2. Helper to evaluate the objective (2·n·max(conv))
    #    (this is exactly the inner part of evaluate_sequence,
    #     the scaling by the L1‑norm is irrelevant because we keep sum=1)
    # --------------------------------------------------------------
    def _objective(vec: np.ndarray) -> float:
        conv = _conv_full(vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # --------------------------------------------------------------
    # 3. Build a small set of diverse starting points
    # --------------------------------------------------------------
    starts = []

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    starts.append(uniform)

    # edge‑heavy deterministic shape: 1/(i+1) + 1/(n-i)
    idx = np.arange(n, dtype=np.float64)
    edge_heavy = 1.0 / (idx + 1.0) + 1.0 / (n - idx)
    edge_heavy = np.clip(edge_heavy, 0.0, None)
    edge_heavy = edge_heavy / edge_heavy.sum()
    starts.append(edge_heavy)

    # previous best if available
    if "height_sequence_1" in globals():
        old = np.asarray(height_sequence_1, dtype=np.float64)
        old = np.clip(old, 0.0, None)
        if old.size < n:
            old = np.concatenate([old, np.zeros(n - old.size, dtype=np.float64)])
        elif old.size > n:
            old = old[:n]
        if old.sum() > 0.0:
            old = old / old.sum()
        else:
            old = uniform.copy()
        starts.append(old)

    # a few random Dirichlet draws (different concentrations)
    for alpha in (0.02, 0.1, 0.5, 1.0):
        for _ in range(2):
            cand = rng.dirichlet(np.ones(n) * alpha)
            starts.append(cand)

    # -----------------------------------------------------------------
    # 4. Evaluate the initial candidates and pick the best as start
    # -----------------------------------------------------------------
    best_J = float("inf")
    best_vec = None
    for vec in starts:
        J = _objective(vec)
        if J < best_J:
            best_J = J
            best_vec = vec.copy()

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Main optimisation loop (exponentiated gradient + soft‑max)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # safety margin

    # optimisation hyper‑parameters
    eta0 = 0.6               # initial step‑size for exponentiated update
    eta_decay = 1e-5         # gradual decay of η
    beta0 = 0.5              # start of soft‑max temperature
    beta_inc = 1e-4          # linear increase per iteration
    beta_max = 80.0          # final β (makes the weight close to active set)
    grad_clip = 8.0          # clipping of the gradient before exponentiation
    jitter_period = 4000     # how often to inject random perturbations
    jitter_weight = 0.3      # mixing proportion for jitter

    a = best_vec.copy()
    iteration = 0
    last_log = start_time

    while time.monotonic() < deadline:
        # --- 1) compute full convolution a*a (FFT) -----------------
        conv = _conv_full(a, fft_len, conv_len)
        max_conv = conv.max()

        # --- 2) soft‑max weight approximating the max ---------------
        beta = min(beta_max, beta0 + beta_inc * iteration)
        # scale by the current maximum to keep the exponent numerically stable
        scaled = conv / max_conv          # max becomes 1.0
        w_unnorm = np.exp(beta * (scaled - 1.0))   # ≤ 1, =1 at the maximum
        w_sum = w_unnorm.sum()
        if w_sum == 0.0:
            weight = np.full_like(conv, 1.0 / conv_len)
        else:
            weight = w_unnorm / w_sum

        # --- 3) gradient of the smoothed objective -----------------
        grad = _weighted_gradient(a, weight, fft_len_corr, n)

        # --- 4) exponentiated (multiplicative) update -------------
        grad = np.clip(grad, -grad_clip, grad_clip)
        step = eta0 / (1.0 + eta_decay * iteration)
        factor = np.exp(-step * grad)
        a = a * factor

        # --- 5) project back onto the simplex ----------------------
        s = a.sum()
        if s == 0.0:
            a = uniform.copy()
        else:
            a = a / s

        # --- 6) evaluate the true objective (max‑convolution) -------
        J = 2.0 * n * max_conv
        if J < best_J - 1e-12:
            best_J = J
            best_vec = a.copy()
            now = time.monotonic()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --- 7) occasional jitter to regain diversity -------------
        if iteration % jitter_period == 0 and iteration > 0:
            jitter = rng.dirichlet(np.ones(n) * 0.2)
            a = proj_simplex(jitter_weight * jitter + (1.0 - jitter_weight) * a)
            # (re‑evaluate after jitter – cheap, just to keep the loop simple)
        iteration += 1

    # ------------------------------------------------------------------
    # 7. Final clean‑up: clip, renormalise and convert to python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:

    else:
        best_vec /= total

    # optional sanity check (commented out – remove if not desired)

    return [float(v) for v in best_vec]
