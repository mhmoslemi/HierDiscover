def _next_pow2(x):
    return 1 << ((x-1).bit_length())

def proj_simplex(v):
    # Euclidean projection onto simplex
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def conv_full_fft(a, fft_len, conv_len):
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]

def compute_J(a, n, fft_len, conv_len):
    conv = conv_full_fft(a, fft_len, conv_len)
    max_conv = conv.max()
    return float(2.0 * n * max_conv), conv, max_conv

def contributions_at_k(a, k):

"""
Simulated‑annealing / targeted hill‑climbing on the probability simplex.

The objective for a probability vector `a` (sum = 1) is  

    J(a) = 2·n·max(conv(a, a))

where `conv` is the linear convolution.  
The scheme repeatedly moves a tiny amount of mass from an index that
contributes most to the current maximal convolution entry to an index that
contributes little.  Moves are accepted if they improve `J`; otherwise they
are accepted with a temperature‑controlled probability (simulated annealing).
If the search stagnates for a minute a small random jitter around the best
solution is applied.  A short final random‑noise refinement finishes the run.

All helper functions are top‑level (no closures) and the routine respects a
1000 s time budget.  The returned list is a non‑negative probability vector;
`evaluate_sequence` will clip and normalise it automatically.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level only
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of `v` onto the probability simplex."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (only the first `conv_len` entries are returned)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int):
    """Return J(a), the full convolution, and its maximum."""
    conv = conv_full_fft(a, fft_len, conv_len)
    max_conv = conv.max()
    return float(2.0 * n * max_conv), conv, max_conv


def contributions_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given convolution index `k`, return the vector `c` where
    c[i] = a[i]·a[k‑i] (zero if `k‑i` is out of range).  The sum of `c`
    equals conv[k].
    """
    n = a.shape[0]
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    c = np.zeros_like(a)
    i = np.arange(n)[mask]
    j = idx[mask]
    c[i] = a[i] * a[j]
    return c

# ----------------------------------------------------------------------
