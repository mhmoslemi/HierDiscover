    deadline = start_time + budget_s - 5.0

    rng = np.random.default_rng(12345)

    # Create list of initial candidates

    # the same patterns as before.
    ...

        for _ in range(30):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    for cand in init_cands:
        conv = _conv_fft(cand, fft_len, conv_len)
        cur_J = float(2.0 * n * conv.max())
        if cur_J < best_J:
            best_J = cur_J
            best_a = cand.copy()
    print("[...] initial best J = ", best_J)

    # Multi-stage optimization:

    # Stage A: gradient-based (Adam) for a number of iterations

    # Use logits of best_a as initialization.

    # temperature schedule
    beta_min = 0.1
    beta_max = 20000.0

    iteration = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        a = _softmax(logits)
        conv = _conv_fft(a, fft_len, conv_len)
        max_conv = conv.max()

        now = time.monotonic()

            global_best_a = a.copy()

            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

            logits = np.log(np.maximum(global_best_a, 1e-12))

                cand[j] += delta

                cand = np.clip(cand, 0, None)
                cand = cand / cand.sum()
                # compute J
                conv2 = _conv_fft(cand, fft_len, conv_len)
                J2 = float(2.0 * n * conv2.max())
                if J2 < global_best_J - 1e-12:
                    global_best_J = J2
                    global_best_a = cand.copy()
                    logits = np.log(np.maximum(cand, 1e-12))
                    m.fill(0); v.fill(0)
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] Perturb improvement J = {global_best_J:.6f}")

    # Stage B: simple hill climbing / simulated annealing for remaining time
    # Use current global_best_a as start
    a = global_best_a.copy()
    cur_J = global_best_J
    # compute temperature for SA
    # We'll define base_T = initial J - target?? Not needed; maybe start with T = (cur_J - 1) *some factor.
    T0 = (cur_J - 1.0) * 0.1  # less
    # but unify schedule: decay linearly to near zero at deadline.
    # While time left:
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        remaining = deadline - now
        # compute temperature based on progress
        progress = (now - start_time) / (deadline - start_time)
        T = T0 * (1 - progress)  # linear decrease

        # Propose move: pick i leaf, j leaf, delta = random* a_i * scale
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        delta = rng.random() * a[i] * 0.5
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # renorm? sum unchanged.
        # compute new J quickly
        conv_new = _conv_fft(new_a, fft_len, conv_len)
        new_J = float(2.0 * n * conv_new.max())
        if new_J < cur_J or rng.random() < np.exp(-(new_J - cur_J) / max(T, 1e-12)):
            a = new_a
            cur_J = new_J
            if cur_J < global_best_J - 1e-12:
                global_best_J = cur_J
                global_best_a = a.copy()
                print(f"[{elapsed:6.2f}s] SA improvement J = {global_best_J:.6f}")

    # after loops, final best: clip and renormalize (though already sum to 1)
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    print(f"[{budget_s:.0f}s] Final best J = {float(2.0 * n * np.convolve(best_vec, best_vec).max()):.6f}")

    return [float(v) for v in best_vec]

"""
Multi‑phase optimizer for the step‑function problem.

Phase 1 – diverse deterministic and stochastic initialisations (uniform,
geometric, linear ramps, Gaussians, Beta‑shaped U‑curves and many Dirichlet
samples).  Each candidate is evaluated exactly (J = 2·n·max(conv)) and the
best few are kept.

Phase 2 – soft‑max based Adam optimisation on the probability simplex.
The hard max in the objective is replaced by a temperature‑controlled
log‑sum‑exp (soft‑max) with an exponentially growing temperature, which
gives smooth gradients early on and focuses on the true maximiser later.
A simple jitter restart is triggered after a period of stagnation.

Phase 3 – cheap simulated‑annealing hill‑climbing on the simplex to
exploit the remaining time budget.  Small random mass transfers are
accepted either if they improve the exact objective or with a
Metropolis probability that decays to zero.

The routine respects the supplied time budget, prints progress and
returns the best non‑negative sequence found (renormalised so that
`evaluate_sequence` can be called directly).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of `a` with itself using a real FFT.
    Returns the first `conv_len` entries.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Numerically stable soft‑max weights  w_i = exp(beta·(conv_i−conv_max)) / Σ.
    """
    c_max = conv.max()
    exp_part = np.exp(beta * (conv - c_max))
    total = exp_part.sum()
    if total == 0.0:          # should not happen, but guard against NaNs
        return np.full_like(conv, 1.0 / conv.size)
    return exp_part / total


def _grad_softmax(a: np.ndarray, weights: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_i w_i·conv_i w.r.t. a, where conv = a*a.
    Uses FFT‑based correlation (complexity O(n log n)).
    """
    n = a.size
    # Full linear correlation length = n + len(weights) - 1 = 3n-2
    corr_len = 3 * n - 2
    fft_len = _next_pow2(corr_len)

    A = np.fft.rfft(a, fft_len)
    W = np.fft.rfft(weights, fft_len)
    corr_full = np.fft.irfft(A * np.conj(W), n=fft_len)[:corr_len]

    start = 2 * n - 2                      # len(weights) - 1
    grad = 2.0 * corr_full[start:start + n]
    return grad


def _softmax(z: np.ndarray) -> np.ndarray:
    """Standard soft‑max (used to keep the vector on the simplex)."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()

# Main search routine
# ----------------------------------------------------------------------

    `budget_s` seconds and returns the best sequence discovered.
    """

    # --------------------------------------------------------------
    if "height_sequence_1" in globals():

            n = 2048
    else:

            global_best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----- 4.4 stagnation handling -----
        if now - last_improve > 18.0:                     # restart after ~18 s of silence
            jitter_sigma = 0.18
            logits = np.log(np.maximum(global_best_a, 1e-12))
            logits += rng.normal(0.0, jitter_sigma, size=n)
            m.fill(0.0)
            v_m.fill(0.0)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – jitter applied")

        # ----- 4.5 temperature (exponential annealing) -----
        progress = (now - start_time) / (deadline - start_time)
        beta = beta_min * (beta_max / beta_min) ** progress

        # ----- 4.6 soft‑max weights over convolution entries -----
        weights = _softmax_weights(conv, beta)

        # ----- 4.7 smooth gradient w.r.t. a -----
        grad_a = _grad_softmax(a, weights)

        # ----- 4.8 chain rule to logits space -----
        avg = np.dot(a, grad_a)            # <a,∇_a J_s>
        grad_z = a * (grad_a - avg)        # ∂J_s/∂z

        # ----- 4.9 Adam update -----
        iteration += 1
        lr = lr0 / np.sqrt(iteration)      # simple 1/√t decay
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_m = beta2 * v_m + (1.0 - beta2) * (grad_z ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_m / (1.0 - beta2 ** iteration)

        logits -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # keep logits in a safe numeric window
        logits = np.clip(logits, -500.0, 500.0)

        # occasional cheap perturbations (helps escape tiny plateaus)
        if iteration % 800 == 0:
            candidate = a.copy()
            # pick a source with non‑negligible mass
            src = rng.choice(np.where(candidate > 1e-8)[0])
            dst = rng.integers(0, n)
            delta = rng.random() * candidate[src] * 0.4
            candidate[src] -= delta
            candidate[dst] += delta
            candidate = np.clip(candidate, 0.0, None)
            candidate /= candidate.sum()
            conv2 = _conv_fft(candidate, fft_len, conv_len)
            J2 = float(2.0 * n * conv2.max())
            if J2 < global_best_J - 1e-12:
                global_best_J = J2
                global_best_a = candidate.copy()
                logits = np.log(np.maximum(candidate, 1e-12))
                m.fill(0.0)
                v_m.fill(0.0)
                last_improve = now
                print(f"[{elapsed:6.2f}s] Perturb improvement J = {global_best_J:.6f}")

    # --------------------------------------------------------------
    # 5. Final refinement – simulated‑annealing hill climbing
    # --------------------------------------------------------------
    a = global_best_a.copy()
    cur_J = global_best_J
    T0 = max(0.05, (cur_J - 1.0) * 0.2)            # starting temperature
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        progress = (now - start_time) / (deadline - start_time)
        T = T0 * (1.0 - progress)                # linear cooling

        # propose a tiny mass transfer between two random positions
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        j = rng.integers(0, n)
        delta = rng.random() * a[i] * 0.45        # move at most 45 % of src mass
        new_a = a.copy()
        new_a[i] -= delta
        new_a[j] += delta
        # renormalise (sum unchanged but guard against numerical drift)
        new_a = np.clip(new_a, 0.0, None)
        new_a /= new_a.sum()

        conv_new = _conv_fft(new_a, fft_len, conv_len)
        new_J = float(2.0 * n * conv_new.max())

        # Metropolis criterion
        if new_J < cur_J or rng.random() < np.exp(-(new_J - cur_J) / max(T, 1e-12)):
            a = new_a
            cur_J = new_J
            if cur_J < global_best_J - 1e-12:
                global_best_J = cur_J
                global_best_a = a.copy()
                print(f"[{elapsed:6.2f}s] SA improvement J = {global_best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Return the best sequence (clipped, renormalised)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    final_J = float(2.0 * n * np.convolve(best_vec, best_vec).max())
    print(f"[{budget_s:.0f}s] Finished – final best J = {final_J:.6f}")

    return [float(v) for v in best_vec]
