"""
Projected gradient descent (soft‑max smoothing) on the probability simplex.

We minimise  J(a)=2·n·max(conv(a,a))  subject to a_i≥0, Σa_i=1.
A smooth approximation of the max is the log‑sum‑exp with a temperature β.
For each iteration we

  1. compute the full self‑convolution a*a (FFT, O(n log n)),
  2. evaluate the soft‑max weights w∝exp(β·conv),
  3. obtain the gradient ∂L/∂a = 2β·(w_rev * a)  (convolution with the
     reversed weight vector) again via FFT,
  4. take a momentum step a←a−η·grad and re‑project onto the simplex
     (non‑negative, unit‑sum).

β is increased slowly so that the optimisation first sees a smooth
objective and finally concentrates on the true maximum.  If no improvement
is seen for ~30 s we re‑initialise a from a random Dirichlet draw to escape
possible stagnation.  A short fine‑grained random‑jitter phase is performed
after the main loop.

The routine respects the 1000 s budget, prints progress, and returns the
best sequence (as a list of floats) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution of two 1‑D real vectors using a real FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    m = len(x) + len(y) - 1
    L = _next_pow2(m)
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    Z = X * Y
    conv = np.fft.irfft(Z, n=L)
    return conv[:m]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Gradient‑descent optimiser for the step‑function problem.
    Returns the best non‑negative sequence (list of floats) found within
    the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # -------------------------------------------------------------
    # 1. Choose problem size
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            # Keep n reasonably bounded
            n = max(2048, min(n0, 8192))
        except Exception:
            n = 8192
    else:
        n = 8192

    # -------------------------------------------------------------
    # 2. Initial candidate (warm start if previous construction exists)
    # -------------------------------------------------------------
    if "height_sequence_1" in globals():
        v0 = np.asarray(height_sequence_1, dtype=np.float64)
        if v0.size != n:
            if v0.size > n:
                v0 = v0[:n]
            else:
                v0 = np.concatenate([v0, np.zeros(n - v0.size, dtype=np.float64)])
        a = _proj_simplex(np.maximum(v0, 0.0))
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start

    # -------------------------------------------------------------
    # 3. Optimisation hyper‑parameters
    # -------------------------------------------------------------
    best_a = a.copy()
    best_J = float("inf")
    momentum = np.zeros_like(a)

    # learning‑rate schedule
    lr0 = 0.2               # initial step size (will be divided by √t)
    mu = 0.9                # momentum coefficient

    # soft‑max temperature schedule
    beta = 0.5
    beta_target = 15.0
    beta_factor = 1.001     # multiplicative increase per iteration

    start_time = time.monotonic()
    # keep a small safety margin for the final fine‑grained search
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    last_improve = start_time
    it = 0

    # pre‑computed FFT lengths (constant throughout the run)
    conv_self_len = 2 * n - 1
    conv_rev_len = 3 * n - 2
    fft_len_self = _next_pow2(conv_self_len)
    fft_len_rev = _next_pow2(conv_rev_len)

    while time.monotonic() < deadline:
        # ---------- convolution a*a (self‑conv) ----------
        A_self = np.fft.rfft(a, fft_len_self)
        conv_self = np.fft.irfft(A_self * A_self, n=fft_len_self)[:conv_self_len]

        max_conv = conv_self.max()
        J = 2.0 * n * max_conv          # because Σa = 1 after projection

        # ---------- record best ----------
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ---------- soft‑max weights ----------
        # numerical stabilisation: subtract the maximum before exponentiation
        exp_arg = beta * (conv_self - max_conv)
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(conv_self, 1.0 / conv_self_len)
        else:
            w /= w_sum

        # ---------- gradient via convolution with reversed weights ----------
        w_rev = w[::-1]                     # length = conv_self_len
        A_rev = np.fft.rfft(a, fft_len_rev)
        W_rev = np.fft.rfft(w_rev, fft_len_rev)
        conv_rev = np.fft.irfft(A_rev * W_rev, n=fft_len_rev)[:conv_rev_len]

        # indices that pick the needed slice (see derivation in analysis)
        idxs = (2 * n - 2) - np.arange(n)   # descending slice
        grad = 2.0 * beta * conv_rev[idxs]

        # ---------- step with momentum ----------
        lr = lr0 / np.sqrt(it + 1.0)         # diminishing step size
        momentum = mu * momentum - lr * grad
        a = a + momentum

        # ---------- projection onto the probability simplex ----------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # ---------- stagnation handling ----------
        now = time.monotonic()
        if now - last_improve > 30.0:
            # re‑initialise from a fresh Dirichlet draw
            alpha = rng.uniform(0.1, 2.0)
            a = rng.dirichlet(np.ones(n) * alpha)
            a = _proj_simplex(a)
            momentum[:] = 0.0
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Stagnation → random restart")

        # ---------- temperature schedule ----------
        if beta < beta_target:
            beta = min(beta_target, beta * beta_factor)

        it += 1

    # -----------------------------------------------------------------
    # 4. Fine‑grained random jitter around the best found solution
    # -----------------------------------------------------------------
    fine_deadline = time.monotonic() + min(30.0, (deadline - start_time) * 0.05)
    while time.monotonic() < fine_deadline:
        # multiplicative Gaussian jitter (small σ)
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = _proj_simplex(best_a * noise)

        # evaluate candidate J
        A_c = np.fft.rfft(cand, fft_len_self)
        conv_c = np.fft.irfft(A_c * A_c, n=fft_len_self)[:conv_self_len]
        J_c = 2.0 * n * conv_c.max()
        if J_c < best_J:
            best_J = J_c
            best_a = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # -----------------------------------------------------------------
    # 5. Return the best sequence (already respects the limits of
    #    `evaluate_sequence`; we normalise once more for safety)
    # -----------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
