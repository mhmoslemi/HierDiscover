"""
Propose a candidate step‑function sequence.

The algorithm proceeds in three stages:

1. **Exploratory families** – a quick sweep over several parametrised
   families of non‑negative sequences (uniform, geometric, linear ramps,
   Gaussian bumps, edge‑peaked power laws and simple two‑level “step”
   shapes). The best sequence from this sweep is kept as a starter.

2. **Local optimisation** – a lightweight simulated‑annealing / hill‑climbing
   phase that repeatedly moves a small amount of mass from one index to
   another while preserving the total sum.  The acceptance probability follows
   a temperature schedule, allowing occasional uphill moves to escape local
   minima.

3. **Final normalisation** – the best sequence found is clipped to the allowed
   range, re‑normalised (the objective is scale‑invariant) and returned as a
   plain Python list.

All heavy lifting (convolution and objective evaluation) is performed with
real FFTs for speed.  The routine respects the supplied time budget (default
1000 s) and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())


def _conv_max(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """
    Compute the maximum entry of the full linear convolution a*a
    using a real FFT of length `fft_len`.  Returns a Python float.
    """
    # real FFT of a, zero‐padded to fft_len
    A = np.fft.rfft(a, n=fft_len)
    # convolution via element‑wise product in frequency domain
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Exact objective J = 2 * n * max(conv) / (sum(a) ** 2).
    The function is scale‑invariant; the caller must ensure non‑negative a.
    """
    sum_a = a.sum()
    if sum_a < 1e-12:
        return float("inf")
    max_conv = _conv_max(a, fft_len, conv_len)
    return 2.0 * n * max_conv / (sum_a * sum_a)

    # Return a plain Python list as required by the evaluation harness
    return [float(v) for v in best_seq]
