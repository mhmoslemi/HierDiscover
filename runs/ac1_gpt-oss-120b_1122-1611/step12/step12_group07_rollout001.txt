def random_move(a, step_factor):
    n = len(a)
    # choose source i where a[i] > tolerance
    src_idx = rng.choice(np.where(a > 1e-12)[0])
    # dest idx j
    dst_idx = rng.integers(n)
    if src_idx == dst_idx:

    max_transfer = a[src_idx] * step_factor
    if max_transfer <= 0:
        return a.copy()
    delta = rng.random() * max_transfer
    a_new = a.copy()
    a_new[src_idx] -= delta
    a_new[dst_idx] += delta
    # no need to renorm as sum unchanged
    return a_new

def propose_candidate(seed=..., budget_s=1000.0):
    rng = np.random.default_rng(seed)
    Determine n: let n = 4096 or based on height_sequence_1.

    Create initial a candidates: list of arrays.

    Evaluate each by computing J via evaluate_sequence (which returns objective). Choose best a (global_best_a) and global_best_J.

    Initialize current a = global_best_a.copy(), cur_J = global_best_J.

    Prepare conv_len = 2*n-1, fft_len = next_pow2(conv_len).

    Setup parameters:
        step_factor = 0.2 (max fraction moved in random move)
        grad_step = 0.5 (initial η), grad_decay = 0.9999
        top_k = 1 initially; maybe increase after some successes.

        stagnation_time = last_improve = start_time

    Loop while time<deadline:

        # compute conv (FFT)
        conv = conv_full(current a, fft_len, conv_len)
        cur_J = 2.0 * n * conv.max()

        # Compute if we should do gradient step or random move (random)
        if rng.random() < 0.8:  # mostly gradient steps
            # choose top_k indices: get argsort descending, pick first top_k
            top_idxs = np.argpartition(-conv, top_k-1)[:top_k]  # unsorted
            # optionally sort but not necessary.
            g = np.zeros_like(current a)
            # compute gradient as described
            n = current a.shape[0]
            for t in top_idxs:
                low = max(0, t-(n-1))
                high = min(n-1, t)
                i = np.arange(low, high+1)
                g[i] += 2 * current a[t - i]
            # Perform exponentiated gradient update
            eta = grad_step
            # compute exponent safely
            exp_factor = np.exp(-eta * g)
            a_new = current a * exp_factor
            sum_a = a_new.sum()
            if sum_a == 0:
                # fallback to uniform
                a_new = np.full_like(current a, 1.0/n)
                sum_a = 1.0
            a_new /= sum_a

            # compute new conv and J
            conv_new = conv_full(a_new, fft_len, conv_len)
            J_new = 2.0 * n * conv_new.max()

            # accept if improvement or probability
            if J_new < cur_J or rng.random() < np.exp((cur_J - J_new) / (0.01 + cur_J)):
                # accept
                current a = a_new
                cur_J = J_new
                conv = conv_new

        else:
            # random move
            # choose source index with mass > epsilon
            nonzero_idxs = np.where(current a > 1e-12)[0]
            if len(nonzero_idxs) > 1:
                src = rng.choice(nonzero_idxs)
                dst = rng.integers(n)
                while dst == src:
                    dst = rng.integers(n)
                max_transfer = current a[src] * step_factor
                delta = rng.random() * max_transfer
                a_new = current a.copy()
                a_new[src] -= delta
                a_new[dst] += delta
                # compute J for new a
                conv_new = conv_full(a_new, fft_len, conv_len)
                J_new = 2.0 * n * conv_new.max()

                    current a = a_new
                    cur_J = J_new
                    conv = conv_new

        if cur_J < global_best_J - 1e-12:

            global_best_a = current a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time

        # stagnation detection
        now = time.monotonic()
        if now - last_improve > 60.0:

            jitter_sigma = 0.1
            # apply jitter to logits - using log for exponentiated update.

    total = final_seq.sum()
    if total > 0:

    # convert to list of floats
    return [float(v) for v in final_seq]

"""
Local‐search + exponentiated‑gradient optimiser for the max‑autoconvolution
objective.  Works directly on the probability simplex (non‑negative, sum=1).
The algorithm mixes:
  • a fast gradient step that targets the (smoothed) maximum of the
    self‑convolution,
  • occasional random pairwise mass transfers,
  • jitter restarts when progress stalls.
The routine is time‑bounded (default 1000 s) and returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Result length = conv_len = 2*len(a)-1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective  J = 2 * n * max(conv(a,a))  (sum(a) is assumed = 1)."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max()), conv


def _gradient_topk(a: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv) with respect to a, using the set of indices
    `top_idxs` where the convolution is (approximately) maximal.
    For each t in top_idxs the contribution is
        g_i += 2 * a[t-i]   (if 0 ≤ t-i < n)
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    for t in top_idxs:
        low = max(0, t - (n - 1))
        high = min(n - 1, t)
        i = np.arange(low, high + 1, dtype=np.int64)
        g[i] += 2.0 * a[t - i]
    return g


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size -----------------------------------------------------------------
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # for a*a
    # --------------------------------------------------------------
    # 2. Build a diverse set of initial probability vectors -------------------------
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # V‑shaped (large at the edges)
    idx = np.arange(n, dtype=np.float64)
    v = 1.0 / ((idx + 1) * (n - idx))
    v = np.clip(v, 0, None)
    v /= v.sum()
    init_candidates.append(v)

    # geometric decays
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        g = r ** idx
        g /= g.sum()
        init_candidates.append(g)

    # linear ramps
    ramp = (n - idx) / n
    ramp /= ramp.sum()
    init_candidates.append(ramp)
    ramp = (idx + 1) / n
    ramp /= ramp.sum()
    init_candidates.append(ramp)

    # Gaussian bumps (different widths)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        g = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        g /= g.sum()
        init_candidates.append(g)

    # Dirichlet draws (moderately and strongly concentrated)
    for alpha in (0.3, 0.6, 1.0):
        for _ in range(3):
            init_candidates.append(rng.dirichlet(np.ones(n) * alpha))

    # previous construction, if it exists
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.size != n:
            if v.size > n:
                v = v[:n]
            else:
                v = np.concatenate([v, np.full(n - v.size, 1.0 / n)])
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_candidates.append(v)

    # --------------------------------------------------------------
    # 3. Choose the best seed according to the exact objective ----------------------
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        J, _ = _compute_J(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Optimisation loop -----------------------------------------------------------
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # current state
    cur_a = best_a.copy()
    cur_J, cur_conv = _compute_J(cur_a, n, fft_len, conv_len)

    # global best
    global_best_a = cur_a.copy()
    global_best_J = cur_J
    last_improve = start_time

    # hyper‑parameters
    eta0 = 0.5                     # base step for exponentiated gradient
    grad_decay = 0.9999            # slow decay of η
    step_factor = 0.15             # max fraction moved in a random transfer
    jitter_sigma = 0.08            # Gaussian jitter magnitude
    top_k = 1                      # number of peak indices used for the gradient
    iteration = 0
    prob_gradient = 0.80           # probability of performing a gradient step

    while time.monotonic() < deadline:
        iteration += 1
        # --------------------------------------------------
        # 4.1 Decide which type of move to perform
        # --------------------------------------------------
        do_gradient = rng.random() < prob_gradient

        if do_gradient:
            # ----------- Gradient step ---------------------------------
            # Choose the indices of the (few) largest convolution entries
            if top_k == 1:
                # most common case – just the absolute maximum
                top_idxs = np.array([int(np.argmax(cur_conv))], dtype=np.int64)
            else:
                # partially sort to get the top‑k indices (unsorted)
                top_idxs = np.argpartition(-cur_conv, top_k - 1)[:top_k]

            # compute sub‑gradient for those peaks
            g = _gradient_topk(cur_a, top_idxs)

            # exponentiated‑gradient update
            eta = eta0 / np.sqrt(iteration)          # 1/√t decay
            # keep exponent in a safe range
            exp_factor = np.exp(np.clip(-eta * g, -500.0, 500.0))
            a_new = cur_a * exp_factor
            sum_a = a_new.sum()
            if sum_a == 0.0:
                a_new = np.full_like(cur_a, 1.0 / n)
                sum_a = 1.0
            a_new /= sum_a

            # evaluate
            J_new, conv_new = _compute_J(a_new, n, fft_len, conv_len)

            # accept if improvement; otherwise simple Metropolis move
            if J_new < cur_J or rng.random() < np.exp((cur_J - J_new) / (0.01 + cur_J)):
                cur_a = a_new
                cur_J = J_new
                cur_conv = conv_new

        else:
            # ----------- Random mass‑transfer move ----------------------
            # pick a source proportional to its mass (guarantees positivity)
            src = rng.choice(n, p=cur_a)
            dst = rng.integers(n)
            if src == dst:
                continue
            max_transfer = cur_a[src] * step_factor
            if max_transfer <= 0.0:
                continue
            delta = rng.random() * max_transfer
            a_new = cur_a.copy()
            a_new[src] -= delta
            a_new[dst] += delta

            J_new, conv_new = _compute_J(a_new, n, fft_len, conv_len)

            if J_new < cur_J or rng.random() < np.exp((cur_J - J_new) / (0.01 + cur_J)):
                cur_a = a_new
                cur_J = J_new
                cur_conv = conv_new

        # --------------------------------------------------
        # 4.2 Record global best
        # --------------------------------------------------
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = cur_a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New global best J = {global_best_J:.6f}")

        # --------------------------------------------------
        # 4.3 Stagnation detection – jitter restart
        # --------------------------------------------------
        now = time.monotonic()
        if now - last_improve > 45.0:          # 45 s without improvement
            # add a small Gaussian perturbation around the current best
            pert = global_best_a + rng.normal(0.0, jitter_sigma, size=n)
            pert = np.clip(pert, 0.0, None)
            s = pert.sum()
            if s == 0.0:
                pert = np.full_like(pert, 1.0 / n)
                s = 1.0
            cur_a = pert / s
            cur_J, cur_conv = _compute_J(cur_a, n, fft_len, conv_len)
            last_improve = now
            print(f"[{now-start_time:6.2f}s] Jitter restart (best J = {global_best_J:.6f})")

        # --------------------------------------------------
        # 4.4 Optional schedule tweaks
        # --------------------------------------------------
        # slowly increase the number of peaks used for the gradient
        if iteration % 5000 == 0 and top_k < 5:
            top_k += 1

    # --------------------------------------------------------------
    # 5. Return the best sequence (clipped, normalised)
    # --------------------------------------------------------------
    final_vec = np.clip(global_best_a, 0.0, 1000.0)
    total = final_vec.sum()
    if total == 0.0:
        final_vec = np.full_like(final_vec, 1.0 / n)
    else:
        final_vec /= total          # keep sum ≈ 1 (optional)

    print(f"Finished. Best J = {global_best_J:.6f}")
    return [float(v) for v in final_vec]
