"""
Gradient‑descent optimizer for the self‑convolution peak.

We minimise
        J(a) = 2·n·max(conv(a,a))                (a_i ≥ 0, Σ a_i = 1)

The scaling of a cancels, therefore the search is performed on the
probability simplex.  The algorithm is:

1. Initialise a from the best known construction (global variable
   ``height_sequence_1``) or from a random Dirichlet draw.
2. Repeatedly:
   * compute the self‑convolution with a real FFT,
   * build a soft‑max weighting w that concentrates on the largest
     convolution entries (parameter ``alpha`` controls the sharpness),
   * obtain the gradient of the weighted loss via another FFT
     (cross‑correlation of w and a reversed),
   * take a projected gradient step with a decreasing learning rate,
   * occasionally apply a small multiplicative jitter to escape shallow
     local minima,
   * if no improvement is seen for a while, re‑initialise the current
     vector.
3. The best vector seen before the time budget expires is returned.

The implementation uses only NumPy and pure‑Python helpers, runs well
within the 1000 s limit and usually beats the previous DE/LP approach.
"""

from __future__ import annotations
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return the centre of the simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def self_convolution(v: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution of v using a real FFT."""
    V = np.fft.rfft(v, n=fft_len)
    conv = np.fft.irfft(V * V, n=fft_len)[:conv_len]
    return conv


def weighted_gradient(
    v: np.ndarray,
    conv: np.ndarray,
    max_c: float,
    alpha: float,
    fft_len_grad: int,
    conv_len: int,
) -> np.ndarray:
    """
    Compute  ∇  ( Σ_k w_k · conv_k )   with
        w_k ∝ exp(alpha·(conv_k - max_c)).
    The gradient equals 2·(w * rev(v)) evaluated at the indices
    (n‑1)…(2n‑2) of the full convolution.
    """
    # soft‑max weights on the convolution entries
    w = np.exp(alpha * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # cross‑correlation w * rev(v)  →  convolution of w with reversed v
    v_rev = v[::-1]
    W_fft = np.fft.rfft(w, n=fft_len_grad)
    Vrev_fft = np.fft.rfft(v_rev, n=fft_len_grad)
    full = np.fft.irfft(W_fft * Vrev_fft, n=fft_len_grad)

    n = v.size
    # slice that corresponds to indices (n‑1) … (2n‑2)
    grad = 2.0 * full[(n - 1) : (n - 1) + n]
    return grad


def objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Objective  J = 2·n·max(conv)."""
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345, budget_s: float = 1000.0, **kwargs
) -> list[float]:
    """
    Gradient‑descent search for a step function that minimises
    ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Determine problem size (use the known construction if present)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        else:
            base = base / base.sum()
        n = base.size
        # optionally enlarge a little to gain extra freedom (capped at 8000)
        if n < 8000:
            n_target = min(8000, max(4096, n * 2))
            if n_target > n:
                base = np.pad(base, (0, n_target - n), mode="constant")
                n = n_target
                base = base / base.sum()
        init_vec = base.copy()
    else:
        # fall back to a moderate size
        n = 4096
        init_vec = rng.dirichlet(np.ones(n))

    # --------------------------------------------------------------
    # 2. Pre‑compute FFT lengths
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_self = next_pow2(conv_len)                # for self‑conv
    fft_len_grad = next_pow2(conv_len + n - 1)        # for gradient conv

    # --------------------------------------------------------------
    # 3. Define objective helpers
    # --------------------------------------------------------------
    def current_objective(v: np.ndarray) -> tuple[float, np.ndarray, float]:
        """Return (J, conv(v,v), max(conv))."""
        c = self_convolution(v, fft_len_self, conv_len)
        maxc = c.max()
        return objective_from_conv(c, n), c, maxc

    # --------------------------------------------------------------
    # 4. Initialise best candidate
    # --------------------------------------------------------------
    best_vec = proj_simplex(init_vec)
    best_J, best_conv, best_maxc = current_objective(best_vec)
    print(f"[0.00s] Initial J = {best_J:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin
    total_allowed = deadline - start_time

    # --------------------------------------------------------------
    # 5. Main optimisation loop
    # --------------------------------------------------------------
    # hyper‑parameters (will be adapted on‑the‑fly)
    alpha = 5.0                     # initial soft‑max sharpness
    lr = 0.12                       # initial learning‑rate
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ---- adapt schedule -------------------------------------------------
        # sharpen the weighting as we get closer to the deadline
        frac = elapsed / total_allowed
        if frac > 0.75:
            alpha = 35.0
        elif frac > 0.5:
            alpha = 20.0
        elif frac > 0.25:
            alpha = 12.0

        # shrink learning‑rate gradually
        lr = max(0.003, 0.15 * (1.0 - frac))

        # ---- gradient step --------------------------------------------------
        grad = weighted_gradient(best_vec, best_conv, best_maxc,
                                alpha, fft_len_grad, conv_len)
        cand = best_vec - lr * grad
        cand = proj_simplex(cand)

        cand_J, cand_conv, cand_maxc = current_objective(cand)

        if cand_J < best_J - 1e-12:
            best_J, best_conv, best_maxc = cand_J, cand_conv, cand_maxc
            best_vec = cand
            last_improve = time.monotonic()
            print(f"[{elapsed:6.2f}s] Improved J = {best_J:.6f}")

        # ---- occasional stochastic jitter ------------------------------------
        iteration += 1
        if iteration % 80 == 0:
            sigma = max(0.004, 0.2 * (1.0 - frac))
            noise = np.exp(rng.normal(0.0, sigma, size=n))
            jitter = proj_simplex(best_vec * noise)
            jitter_J, _, _ = current_objective(jitter)
            if jitter_J < best_J:
                best_J, best_vec = jitter_J, jitter
                print(f"[{elapsed:6.2f}s] Jitter improvement → J = {best_J:.6f}")

        # ---- stagnation handling -------------------------------------------
        if now - last_improve > 30.0:
            # re‑initialise half the mass with a fresh Dirichlet draw
            k = rng.integers(500, n // 2)          # random support size
            idx = rng.choice(n, size=k, replace=False)
            fresh = np.zeros(n, dtype=np.float64)
            fresh[idx] = rng.random(k)
            fresh /= fresh.sum()
            best_vec = proj_simplex(0.6 * best_vec + 0.4 * fresh)
            best_J, best_conv, best_maxc = current_objective(best_vec)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart – refreshed part of the vector")

    # --------------------------------------------------------------
    # 6. Final safety clipping and normalisation
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
