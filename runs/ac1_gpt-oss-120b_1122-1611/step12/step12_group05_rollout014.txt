def build_lp_constraints(f):
    n = len(f)
    conv_len = 2*n -1
    # A_ub shape (conv_len, n+1)
    A_ub = np.zeros((conv_len, n+1), dtype=np.float64)
    # Fill for each i
    for i in range(n):
        start = i
        end = i + n
        A_ub[start:end, i] = f
    # coefficient for M
    A_ub[:, n] = -1.0
    # b_ub = zeros
    b_ub = np.zeros(conv_len, dtype=np.float64)
    # Equality sum(g)=1
    A_eq = np.zeros((1, n+1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    # bounds: g_i >=0; M>=0
    bounds = [(0, None)]*n + [(0, None)]
    c = np.zeros(n+1)
    c[-1] = 1.0 # minimize M
    return c, A_ub, b_ub, A_eq, b_eq, bounds

import time

import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << ((x - 1).bit_length())

def _full_convolution(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a via FFT.

    Arguments:
        a: 1‑D array of length n.
        fft_len: length of the FFT (power of two ≥ conv_len).
        conv_len: length of the linear convolution, 2·n‑1.

    Returns:
        np.ndarray of shape (conv_len,) containing a*a.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv

def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Heuristic search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The method mixes an exponentiated‑gradient

        max_conv = conv.max()

        if cur_J < best_J - 1e-12:

        print(f"Final evaluate_sequence value = {val:.6f}")
    except Exception:
        # In case the evaluator is not available in this context
        pass

        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a * a
    L_grad = _next_pow2(conv_len + n - 1)         # for w * a_rev

    # ------------------------------------------------------------------
    # 2. Initialise a on the simplex (∑a = 1, a_i ≥ 0).
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(height_sequence_1, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        if a.size != n:
            if a.size > n:
                a = a[:n]
            else:
                a = np.pad(a, (0, n - a.size))
        tot = a.sum()
        if tot == 0.0:
            a = np.full(n, 1.0 / n)
        else:
            a = a / tot
    else:
        # random start to avoid the uniform fixed‑point
        a = rng.random(n)
        a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, fft_len, conv_len)
    best_J = float(2.0 * n * conv.max())
    print(f"[0.00s] Initial J = {best_J:.6f}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 3. Phase‑1 : exponentiated‑gradient descent on a smooth‑max surrogate.
    # ------------------------------------------------------------------
    alpha = 1.0          # smoothing for the soft‑max
    eta = 0.5            # EG learning rate
    iteration = 0
    phase1_deadline = start_time + 0.60 * total_time   # 60 % of budget

    while time.monotonic() < phase1_deadline:
        # --- current loss -------------------------------------------------
        conv = _full_convolution(a, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] New best J (grad) = {best_J:.6f}")

        # --- soft‑max weights ---------------------------------------------
        shift = conv - max_conv                 # ≤ 0
        w_raw = np.exp(alpha * shift)
        w_sum = w_raw.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w_raw, 1.0 / conv_len)
        else:
            w = w_raw / w_sum

        # --- gradient of the smooth‑max (w.r.t. a) ------------------------
        a_rev = a[::-1]
        G = np.fft.rfft(w, L_grad)
        R = np.fft.rfft(a_rev, L_grad)
        grad_full = np.fft.irfft(G * R, n=L_grad)
        grad = 2.0 * grad_full[n - 1:n - 1 + n]   # aligned slice

        # bound gradient to keep exp stable
        grad = np.tanh(grad)                     # now in (‑1, 1)

        # --- exponentiated‑gradient update (mirror descent) ------------
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # --- schedule adaptation -----------------------------------------
        iteration += 1
        if iteration % 1000 == 0:
            alpha = min(alpha * 1.2, 200.0)      # smoother soft‑max
            eta = max(eta * 0.9, 0.01)           # slower steps later

    # ------------------------------------------------------------------
    # 4. Phase‑2 : simulated‑annealing polish around the best point.
    # ------------------------------------------------------------------
    a = best_a.copy()
    conv = _full_convolution(a, fft_len, conv_len)
    max_conv = conv.max()
    cur_J = 2.0 * n * max_conv
    best_J2 = cur_J
    best_a2 = a.copy()

    # exponential temperature schedule
    T_start = max_conv * 0.5 + 1e-12
    T_end = 1e-9

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # progress of the SA part (the last 40 % of the budget)
        frac = (elapsed - 0.60 * total_time) / (0.40 * total_time)
        frac = max(0.0, min(1.0, frac))
        T = T_start * (T_end / T_start) ** frac

        # propose moving a tiny amount of mass from i to j
        i, j = rng.choice(n, size=2, replace=False)
        if a[i] <= 0.0:
            continue
        max_move = min(0.05 * a[i], a[i])       # at most 5 % of source bin
        delta = rng.random() * max_move

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        # evaluate candidate
        conv_cand = _full_convolution(a_candidate, fft_len, conv_len)
        max_conv_cand = conv_cand.max()
        d = max_conv_cand - max_conv

        accept = False
        if d < 0.0:
            accept = True
        else:
            prob = math.exp(-d / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_candidate
            conv = conv_cand
            max_conv = max_conv_cand
            cur_J = 2.0 * n * max_conv
            if cur_J < best_J2 - 1e-12:
                best_J2 = cur_J
                best_a2 = a.copy()
                print(f"[{now - start_time:6.2f}s] New best J (SA) = {best_J2:.6f}")

    # ------------------------------------------------------------------
    # 5. Choose the overall best, sanitise and return.
    # ------------------------------------------------------------------
    final_a = best_a2 if best_J2 < best_J else best_a
    final_a = np.clip(final_a, 0.0, 1000.0)
    tot = final_a.sum()
    if tot == 0.0:
        final_a[:] = 1.0 / n
    else:
        final_a /= tot

    # Optional final report using the supplied evaluator
    try:
        val = evaluate_sequence(final_a.tolist())
        print(f"Final evaluate_sequence value = {val:.6f}")
    except Exception:
        pass

    return [float(v) for v in final_a]
