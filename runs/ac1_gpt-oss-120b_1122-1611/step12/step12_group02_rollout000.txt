"""
Propose a non‑negative step‑function (probability simplex) that minimizes
the evaluation function defined in ``evaluate_sequence``.

The algorithm combines several ideas:

* **Diverse initialisation** – uniform, the previous best (``height_sequence_1``),
  several Dirichlet draws, log‑normal draws and a few deterministic shapes
  (sin, raised‑cosine, triangular, Gaussian).
* **Adaptive Differential Evolution (jDE)** – a small population (≈40) is
  evolved with per‑individual mutation factor ``F`` and crossover
  probability ``CR`` that adapt during the run.
* **Periodic projected gradient polish** – after every few generations the
  current global best is refined by a simple sub‑gradient step on the
  ``max(conv)`` term (the worst autocorrelation lag).  The step is performed
  multiplicatively and the iterate is projected back onto the simplex.
* **Stagnation handling** – if no improvement is seen for a while a fraction
  of the population is re‑initialised (jitter) to restore diversity.
* **Time management** – the routine respects the 1000 s wall‑clock budget
  (a small safety margin is kept).

All helper functions are defined at top‑level (no closures) and only NumPy
and the standard library are used.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT (length = conv_len)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute the gradient 2·corr(weight, a) efficiently via FFT.
    ``weight`` has length (2·n‑1), ``a`` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # the valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(vec: np.ndarray, n: int, fft_len: int) -> float:
    """Return the objective 2·n·max(conv) for a probability vector."""
    conv = _conv_full(vec, fft_len, 2 * n - 1)
    return float(2.0 * n * conv.max())


def _gradient_polish(cur_vec: np.ndarray, n: int, fft_len: int,
                    fft_len_corr: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform a few multiplicative gradient descent steps on ``max(conv)``.
    Stops when time runs out or no improvement is obtained.
    Returns the improved vector and its objective value.
    """
    cur = cur_vec.copy()
    cur_J = _objective(cur, n, fft_len)

    # Parameters for the polish phase
    step_candidates = (0.6, 0.4, 0.25, 0.15, 0.08, 0.04)

    while time.monotonic() < deadline:
        conv = _conv_full(cur, fft_len, 2 * n - 1)
        max_val = conv.max()
        # binary mask of active (max‑tight) lags – a minimal subgradient
        active = (conv >= max_val - 1e-12).astype(np.float64)
        grad = _weighted_gradient(cur, active, fft_len_corr, n)

        improved = False
        for step_mul in step_candidates:
            factor = np.exp(-step_mul * grad)          # multiplicative move
            cand = cur * factor
            cand = proj_simplex(cand)                  # keep on simplex
            cand_J = _objective(cand, n, fft_len)
            if cand_J < cur_J - 1e-12:
                cur = cand
                cur_J = cand_J
                improved = True
                break
        if not improved:
            break
    return cur, cur_J

# Main optimisation routine
# ----------------------------------------------------------------------
