def propose_candidate(seed=..., budget_s=1000.0):
    rng = numpy.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 0.1

    # Determine n have chosen.
    if 'height_sequence_1' in globals():
        base_seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        n = max(1024, min(base_seq.size, 8000))
    else:
        n = 4000  # default

    conv_len = 2 * n - 1
    corr_len = 3 * n - 2
    fft_len = _next_pow2(max(conv_len, corr_len))

    # Helper to compute J and conv and Af
    def compute_metrics(a):
        Af = np.fft.rfft(a, fft_len)
        full_conv = np.fft.irfft(Af * Af, n=fft_len)[:conv_len]
        max_conv = full_conv.max()
        J = float(2.0 * n * max_conv)
        return J, full_conv, Af

    # Initialize best
    best_J = np.inf
    best_a = None

    # initialize candidate list
    init_candidates = []
    init_candidates.append(np.full(n, 1.0 / n))
    if 'height_sequence_1' in globals():
        v = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        # truncate or pad to n
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size)])
        v = np.maximum(v, 0.0)
        sumv = v.sum()
        if sumv == 0:
            v[:] = 1.0 / n
        else:
            v = v / sumv
        init_candidates.append(v)
    # random Dirichlet with various alphas
    for alpha in (0.2, 0.5, 0.8, 1.0):
        for _ in range(3):
            init_candidates.append(rng.dirichlet(np.full(n, alpha)))
    # filter duplicates maybe

    for cand in init_candidates:
        Jc, convc, Afc = compute_metrics(cand)
        if Jc < best_J:
            best_J, best_a, best_conv, best_Af = Jc, cand.copy(), convc, Afc
    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:.2f}s] Initial best J = {best_J:.6f}")

    # Main search loop (random restarts)
    restart = 0
    # we control number of restarts heuristically based on time budget.
    while time.monotonic() < deadline:
        restart += 1
        # start with either perturbation of best or fresh Dirichlet
        if best_a is not None and rng.random() < 0.6:
            # perturb best
            noise = np.exp(rng.normal(0.0, 0.03, size=n))

        step = 0.5
        max_iters_local = 5000  # but break on time
        it = 0
        while time.monotonic() < deadline and it < max_iters_local:
            it += 1

            max_conv = conv_cur.max()
            # using p as softmax scaling factor (higher => sharper)
            # Use stable exponent
            exp_vals = np.exp(p * (conv_cur - max_conv))

            start_idx = 2 * n - 2
            g = corr_full[start_idx:start_idx + n]

            g = 2.0 * g

            # Use normalized step to keep update size moderate
            norm_g = np.linalg.norm(g)
            if norm_g == 0:

                step *= 0.7

            a, J_cur, conv_cur, Af_cur = a_new, J_new, conv_new, Af_new

            if J_cur < best_J - 1e-12:

                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] New best J = {best_J:.6f} (restart {restart}, iter {it})")

            p = min(p * 1.02, 80.0)  # increase temperature gradually
            step = step * 0.999

            # early break if improvement plateau
            if it % 200 == 0 and time.monotonic() - deadline > 0:

    # Final local random refinements around best_a
    fine_start = time.monotonic()

    if s == 0:
        best_a[:] = 1.0 / n
    else:

    # Return list
    return [float(v) for v in best_a]

"""
Gradient descent with softmax smoothing to flatten the convolution of a
probability vector (step‑function heights).  The algorithm works on a
probability simplex of length n (chosen from an existing best or a
default size).  It repeatedly:
  * computes the full auto‑convolution via FFT,
  * builds a smooth approximation of the “max” using a soft‑max with
    temperature parameter p,
  * obtains the gradient of this smooth objective by a cross‑correlation
    (again via FFT),
  * takes a projected gradient step onto the simplex,
  * slowly raises p (sharpening the soft‑max) and decays the step size.
Several random restarts and a final local random perturbation are used
within the 1000‑second budget.  The best sequence found is returned as a
plain Python list of floats.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Uses projected gradient descent on a smooth
    approximation of the infinity‑norm of the self‑convolution, together
    with random restarts and a final fine‑grained perturbation.
    The best sequence found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # leave a small margin for finalisation
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------------
    # Choose problem size
    # ----------------------------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n0 = prev.size
        # Clamp to a comfortable size (larger size gives a lower theoretical bound)
        n = max(1024, min(n0, 8000))
    else:
        n = 4000  # default size

    conv_len = 2 * n - 1
    corr_len = 3 * n - 2
    fft_len = _next_pow2(max(conv_len, corr_len))

    # ----------------------------------------------------------------------
    # Helper to compute objective and reusable intermediate values
    # ----------------------------------------------------------------------
    def compute_metrics(a: np.ndarray):
        """
        Returns (J, full_convolution, rfft_of_a).
        a is assumed to lie on the simplex (sum=1, a>=0).
        """
        Af = np.fft.rfft(a, fft_len)
        full_conv = np.fft.irfft(Af * Af, n=fft_len)[:conv_len]
        max_conv = full_conv.max()
        J = float(2.0 * n * max_conv)
        return J, full_conv, Af

    # ----------------------------------------------------------------------
    # Initialise a set of candidate vectors (uniform, previous best, random)
    # ----------------------------------------------------------------------
    best_J = np.inf
    best_a = None
    best_conv = None
    best_Af = None

    init_candidates = []

    # Uniform vector
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Previous best, truncated / padded
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if v.size != n:
                if v.size > n:
                    v = v[:n]
                else:
                    v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            v = np.maximum(v, 0.0)
            s = v.sum()
            if s == 0.0:
                v[:] = 1.0 / n
            else:
                v = v / s
            init_candidates.append(v)
        except Exception:
            pass

    # Several Dirichlet draws with different concentrations
    for alpha in (0.2, 0.5, 0.8, 1.0):
        for _ in range(3):
            v = rng.dirichlet(np.full(n, alpha))
            init_candidates.append(v)

    # Geometric ramps (decaying and increasing)
    geom = 0.99 ** np.arange(n, dtype=np.float64)
    init_candidates.append(geom / geom.sum())
    rev_geom = (0.99 ** np.arange(n-1, -1, -1, dtype=np.float64))
    init_candidates.append(rev_geom / rev_geom.sum())

    # Evaluate initial candidates
    for cand in init_candidates:
        Jc, convc, Afc = compute_metrics(cand)
        if Jc < best_J:
            best_J, best_a, best_conv, best_Af = Jc, cand.copy(), convc, Afc

    elapsed = time.monotonic() - start_time
    print(f"[{elapsed:.2f}s] Initial best J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # Main optimisation loop – random restarts with smoothed gradient descent
    # ----------------------------------------------------------------------
    restart = 0
    while time.monotonic() < deadline:
        restart += 1

        # Starting point: either a perturbation of the global best or a fresh draw
        if best_a is not None and rng.random() < 0.6:
            noise = np.exp(rng.normal(0.0, 0.025, size=n))
            a = proj_simplex(best_a * noise)
        else:
            a = rng.dirichlet(np.full(n, 0.5))

        J_cur, conv_cur, Af_cur = compute_metrics(a)

        # Local refinement parameters
        p = 2.0               # soft‑max temperature (low → smooth)
        step = 0.8            # base step size for the projected gradient
        max_local_iters = 2500

        it = 0
        while time.monotonic() < deadline and it < max_local_iters:
            it += 1

            # --- soft‑max weights that approximate the max of the convolution ---
            maxc = conv_cur.max()
            # stability: subtract the max before exponentiation
            exp_vals = np.exp(p * (conv_cur - maxc))
            w = exp_vals / exp_vals.sum()

            # --- gradient of the smoothed objective (cross‑correlation) ---
            Wf = np.fft.rfft(w, fft_len)
            corr_full = np.fft.irfft(np.conj(Af_cur) * Wf, n=fft_len)[:corr_len]
            start_idx = 2 * n - 2               # index that corresponds to lag 0
            g = corr_full[start_idx:start_idx + n]
            g = 2.0 * g                         # each product appears twice except on the diagonal

            # Normalise the step size so that updates have comparable magnitude
            norm_g = np.linalg.norm(g)
            if norm_g == 0.0:
                break
            alpha = step / (norm_g + 1e-12)

            # Simple back‑tracking line search (max 5 attempts)
            improved = False
            alpha_tmp = alpha
            for _ in range(5):
                a_new = proj_simplex(a - alpha_tmp * g)
                J_new, conv_new, Af_new = compute_metrics(a_new)
                if J_new <= J_cur + 1e-12:
                    improved = True
                    break
                alpha_tmp *= 0.5

            if not improved:
                # If no improvement, shrink the step and continue
                step *= 0.7
                continue

            # Accept the new point
            a, J_cur, conv_cur, Af_cur = a_new, J_new, conv_new, Af_new

            # Record a global improvement
            if J_cur < best_J - 1e-12:
                best_J, best_a = J_cur, a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] New best J = {best_J:.6f} (restart {restart}, it {it})")

            # Schedule updates – raise the temperature and decay the step size
            p = min(p * 1.025, 80.0)   # make the soft‑max sharper over time
            step *= 0.998

        # Occasionally re‑initialise if we have spent a long time without improvement
        if time.monotonic() - start_time > (restart * 30):
            # inject a few fresh individuals
            pass  # (no explicit population here – the loop simply continues)

    # ----------------------------------------------------------------------
    # Final fine‑grained random perturbation around the best solution
    # ----------------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.012, size=n))
        cand = proj_simplex(best_a * noise)
        J_cand, _, _ = compute_metrics(cand)
        if J_cand < best_J - 1e-12:
            best_J, best_a = J_cand, cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:.2f}s] Improved J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # Normalise for the final output (protect against zero sum, clip range)
    # ----------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]

                a_new = proj_simplex(a - alpha_tmp * g)
                J_new, conv_new, Af_new = compute_metrics(a_new, n, fft_len, conv_len)
                if J_new <= J_cur + 1e-12:
                    improved = True
                    break
                alpha_tmp *= 0.5

            if not improved:
                step *= 0.7            # shrink step if we cannot improve
                continue

            # accept the new point
            a, J_cur, conv_cur, Af_cur = a_new, J_new, conv_new, Af_new

            # record a global improvement
            if J_cur < best_J - 1e-12:
                best_J, best_a = J_cur, a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:.2f}s] New best J = {best_J:.6f} (restart {restart}, it {it})")

            # schedule updates
            p = min(p * 1.025, 80.0)   # make the soft‑max sharper over time
            step *= 0.998

    # ------------------------------------------------------------------
    # Final fine‑grained random perturbation around the best solution
    # ------------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)

    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.012, size=n))
        cand = proj_simplex(best_a * noise)
        J_cand, _, _ = compute_metrics(cand, n, fft_len, conv_len)
        if J_cand < best_J - 1e-12:
            best_J, best_a = J_cand, cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # Normalise for the final output (protect against zero sum, clip range)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    tot = best_a.sum()
    if tot == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= tot

    return [float(v) for v in best_a]
