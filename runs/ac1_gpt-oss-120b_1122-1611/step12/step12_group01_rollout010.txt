"""
Hybrid random-phase + projected subgradient search.

The algorithm searches for a non‑negative step function that minimises

    J(a) = 2·n·max(conv(a, a)) / (sum a)^2

where `conv` is the linear convolution.  The search consists of three
independent ideas that run in parallel until the time budget expires:

1. **Random‑phase Fourier candidates** – a constant‑magnitude spectrum with
   random phases is transformed back to the time domain, clipped to
   non‑negative values and projected onto the probability simplex.  This
   produces diverse vectors whose autocorrelation is close to flat.

2. **Projected sub‑gradient descent** – for the current worst convolution
   shift(s) the exact sub‑gradient of the objective is known analytically.
   The algorithm walks opposite to this sub‑gradient with a diminishing step
   size and projects back onto the simplex.  The method quickly reduces the
   maximal convolution entry.

3. **Multiplicative jitter** – a final short stochastic perturbation phase
   (geometric noise) that refines the best vector found.

All vectors are kept non‑negative, normalised to unit L1 norm
(the scaling does not affect the objective).  The routine respects the
1000 s budget and returns the best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_full(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Full linear convolution a*a using FFT (complex‑valued)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv


def conv_max(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the self‑convolution of *a*."""
    full = conv_full(a, fft_len)[:conv_len]
    return float(full.max())


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective 2·n·max(conv(a,a)) / (sum a)^2."""
    max_b = conv_max(a, fft_len, conv_len)
    sum_a = a.sum()
    return float(2.0 * n * max_b / (sum_a * sum_a))


def gradient_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of conv[k] w.r.t. a.
    For a shift index k (0 ≤ k ≤ 2·n‑2) the derivative is
        g_j = 2·a[k‑j]   if 0 ≤ k‑j < n
        g_j = 0          otherwise
    """
    n = a.size
    g = np.zeros_like(a)
    if k < n:
        # valid indices j = 0 .. k
        g[: k + 1] = 2.0 * a[k :: -1]  # a[k], a[k‑1], … , a[0]
    else:
        start = k - n + 1                # first j with a[k‑j] defined
        length = n - start               # number of valid entries
        # a[:length] correspond to a[k‑j] for j = start … n‑1 (reversed)
        g[start:] = 2.0 * a[:length][::-1]
    return g


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def subgradient_opt(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    conv_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on J = max(conv(a,a)).
    Stops when the *deadline* is reached.
    Returns the best vector seen during the run and its J‑value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(a, n, fft_len, conv_len)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        # full convolution (needed only to locate active shifts)
        full = conv_full(a, fft_len)[:conv_len]
        M = full.max()
        # active indices (within a tiny tolerance)
        active = np.where(full >= M - 1e-12)[0]

        # aggregate gradient of the active shifts
        grad = np.zeros_like(a)
        for k in active:
            grad += gradient_for_shift(a, k)
        grad /= max(1, len(active))

        # diminishing step size
        step = 0.5 / np.sqrt(it)

        # gradient descent step and projection onto the simplex
        a = a - step * grad
        a = proj_simplex(np.maximum(a, 0.0))

        # evaluate
        cur_J = compute_J(a, n, fft_len, conv_len)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_a = a.copy()
    return best_a, best_J


def random_phase_candidate(
    n: int,
    fft_len: int,
    rng: np.random.Generator,
) -> np.ndarray:
    """
    Generate a vector whose DFT has constant magnitude and random phases.
    The resulting time‑domain signal is projected onto the simplex.
    """
    # rfft size
    nr = fft_len // 2 + 1
    phases = rng.uniform(0.0, 2.0 * np.pi, size=nr)
    A = np.exp(1j * phases)

    # ensure the DC component is real (its phase does not matter after projection)
    A[0] = np.abs(A[0])

    # inverse transform, keep only first n entries
    a = np.fft.irfft(A, n=fft_len)[:n]
    a = proj_simplex(np.maximum(a, 0.0))
    return a

# Main entry point
# ----------------------------------------------------------------------

    ``evaluate_sequence``.  The routine employs random Fourier candidates,
    a projected sub‑gradient optimiser and a final jitter phase.  It respects
    the given time budget (default 1000 s) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
        n = max(1024, min(n, 8192))
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)

    # ------------------------------------------------------------------
    # initial best vector (use previous construction if available)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        v0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        v0 = np.clip(v0, 0.0, None)
        if v0.sum() == 0.0:
            v0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            v0 = proj_simplex(v0)
    else:
        v0 = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = v0.copy()
    best_J = compute_J(best_vec, n, fft_len, conv_len)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep 5 s for final jitter

    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # main search loop (random‑phase + sub‑gradient)
    # ------------------------------------------------------------------
    use_subgrad = True  # toggle strategy based on remaining time
    while time.monotonic() < deadline:
        now = time.monotonic()
        remaining = deadline - now

        # decide which primitive to use (probabilistic)
        if use_subgrad and remaining > 2.0 and rng.random() < 0.7:
            # ----- sub‑gradient descent -----
            # pick a starting point: sometimes the current best, otherwise fresh random
            if rng.random() < 0.2:
                a_start = best_vec.copy()
            else:
                alpha = rng.uniform(0.05, 0.6)      # Dirichlet concentration
                a_start = rng.dirichlet(np.ones(n) * alpha)

            # allocate a modest slice of the remaining budget for this run
            local_deadline = now + min(remaining * 0.4, 8.0)  # at most 8 s per run
            a_opt, J_opt = subgradient_opt(a_start, n, fft_len, conv_len, local_deadline)

            if J_opt < best_J - 1e-12:
                best_J = J_opt
                best_vec = a_opt.copy()
                print(f"[{now - start_time:6.2f}s] New best after sub‑grad J = {best_J:.6f}")

        else:
            # ----- random‑phase Fourier candidate -----
            a_rand = random_phase_candidate(n, fft_len, rng)
            J_rand = compute_J(a_rand, n, fft_len, conv_len)
            if J_rand < best_J - 1e-12:
                best_J = J_rand
                best_vec = a_rand.copy()
                print(f"[{now - start_time:6.2f}s] New best random‑phase J = {best_J:.6f}")

        # switch strategy occasionally to keep diversity
        use_subgrad = not use_subgrad

    # ------------------------------------------------------------------
    # final stochastic jitter (tiny multiplicative noise)
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        sigma = max(0.005, 0.2 * remaining / total_time)
        noise = np.exp(rng.normal(0.0, sigma, size=n))
        cand = best_vec * noise
        cand = proj_simplex(cand)
        cand_J = compute_J(cand, n, fft_len, conv_len)
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand
            print(f"[{time.monotonic() - start_time:6.2f}s] Jitter improvement J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # final normalisation & safety clipping (mirrors evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
