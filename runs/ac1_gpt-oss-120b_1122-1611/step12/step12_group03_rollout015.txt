def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)

    # Determine n (length)
    if 'height_sequence_1' in globals():
        old_seq = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        n0 = old_seq.size
        # choose n around also maybe increase by factor if memory allows
        n = max(1024, min(8192, n0))
    else:
        n = 3000

    # Ensure n odd maybe to get central shift exactly n-1
    if n % 2 == 0:
        n += 1

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # evaluation function
    def objective(vec):
        a = np.asarray(vec, dtype=np.float64)
        # maybe clip to max 1000 to match evaluate_sequence if needed.
        a = np.clip(a, 0.0, 1000.0)
        sum_a = a.sum()
        if sum_a < 1e-12:
            return np.inf
        # compute convolution via FFT
        A = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        max_conv = conv.max()
        return float(2.0 * n * max_conv / (sum_a * sum_a))

    # baseline uniform
    uniform_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_vec = uniform_seq
    best_J = objective(uniform_seq)
    print(f"[init] Uniform J = {best_J}")

    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 30.0)   # leave margin for final refine

    # Parameter grid search
    # Exponential r values
    r_vals = np.concatenate([
        np.linspace(0.5, 0.9, 9),
        np.linspace(0.9, 0.999, 11)[1:]
    ])
    # Power law alphas
    alphas = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0, 2.5, 3.0]
    # Linear plateau fractions
    fractions = [0.4, 0.5, 0.6, 0.7, 0.8, 0.9, 1.0]

    # Evaluate each shape
    # Exponential
    for r in r_vals:
        if time.monotonic() >= deadline:
            break
        # compute raw values
        i = np.arange(n, dtype=np.float64)
        raw = r ** i
        raw_sum = raw.sum()
        if raw_sum == 0:
            continue
        seq = raw / raw_sum
        J = objective(seq)
        if J < best_J:
            best_J = J
            best_vec = seq
            print(f"[exp r={r:.4f}] new best J={best_J:.6f}")

    # Power law
    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        i = np.arange(1, n+1, dtype=np.float64)  # start from 1 to avoid division zero
        raw = 1.0 / (i ** alpha)
        seq = raw / raw.sum()
        J = objective(seq)
        if J < best_J:
            best_J = J
            best_vec = seq
            print(f"[pow alpha={alpha}] new best J={best_J:.6f}")

    # Linear decreasing with plateau
    for frac in fractions:
        if time.monotonic() >= deadline:
            break
        m = int(np.ceil(frac * n))
        # decreasing linearly from value a at i=0 to 0 at i=m-1
        # choose a such that sum of sequence =1 after normalization; we can just compute raw = np.arange(m, 0, -1)
        raw = np.concatenate([np.arange(m, 0, -1, dtype=np.float64), np.zeros(n - m, dtype=np.float64)])
        seq = raw / raw.sum()
        J = objective(seq)
        if J < best_J:
            best_J = J
            best_vec = seq
            print(f"[lin frac={frac}] new best J={best_J:.6f}")

    # Mixture of exponential and uniform: generate weighted combinations
    # Eg., mixing parameter w from 0..1: seq = w * exp + (1-w) * uniform
    # We'll try a few w values.
    w_vals = [0.2, 0.4, 0.6, 0.8]
    # use best exponent r among previously examined? Let's pick r=0.99 as representative.
    base_r = 0.99
    base_exp_raw = base_r ** np.arange(n, dtype=np.float64)
    base_exp = base_exp_raw / base_exp_raw.sum()
    for w in w_vals:
        if time.monotonic() >= deadline:
            break
        seq = w * base_exp + (1 - w) * uniform_seq
        seq = seq / seq.sum()   # re-normalize (sum=1)
        J = objective(seq)
        if J < best_J:
            best_J = J
            best_vec = seq
            print(f"[mix w={w}] new best J={best_J:.6f}")

    # Random Dirichlet draws near best
    # We'll randomize around best by adding lognormal noise per element.
    while time.monotonic() < deadline:
        # small perturbations multiplicative
        noise = np.exp(rng.normal(0.0, 0.03, size=n))
        cand = best_vec * noise
        cand = cand / cand.sum()
        J = objective(cand)
        if J < best_J:
            best_J = J
            best_vec = cand
            print(f"[rand ref] improved J={best_J:.6f}")

    # Final fine tuning (short run)
    fine_end = start_time + budget_s
    while time.monotonic() < fine_end:
        # directed subgradient step
        # compute conv via FFT
        A = np.fft.rfft(best_vec, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        k_max = int(np.argmax(conv))
        # compute subgradient: g_i = 2* best_vec[k_max - i] (if valid)
        g = np.zeros_like(best_vec)
        # shift
        idx_low = max(0, k_max - (n - 1))
        idx_high = min(n - 1, k_max)
        # For i from idx_low..idx_high:
        # p_i contributes to conv[k_max] with p_{k_max - i}
        # So g_i = 2 * p_{k_max - i}
        # but careful when k_max - i outside range; we limited.
        i_range = np.arange(idx_low, idx_high+1)
        j_range = k_max - i_range
        g[i_range] = 2 * best_vec[j_range]
        # Step size: maybe adaptive 0.1 / iteration count
        # We'll choose step as small factor
        step = 0.05 * (1.0 / (1 + (time.monotonic() - start_time)))  # decreasing
        cand = best_vec - step * g
        # Ensure non-negative:
        cand = np.maximum(cand, 0.0)
        # Renormalize:
        if cand.sum() == 0:
            cand = uniform_seq
        else:
            cand = cand / cand.sum()
        J = objective(cand)
        if J < best_J:
            best_J = J
            best_vec = cand
            print(f"[subgrad] improved J={best_J:.6f}")

    # Return as list
    # Scale to have moderate values, e.g., multiply by 100 (optional) but scaling doesn't affect objective.
    # To avoid extremely small numbers that might be treated as 0 after clipping? but fine.
    final_seq = best_vec * 1000.0   # scale to max allowed maybe but still <=1000 due to clipping to 1000 in evaluate.
    final_seq = np.clip(final_seq, 0.0, 1000.0)
    # Ensure sum not too low
    if final_seq.sum() < 1e-3:
        final_seq = np.full(n, 1.0 / n) * 1000.0

    return [float(v) for v in final_seq]

"""
Propose a candidate probability vector that minimises
    J(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The search consists of three stages:

1.  Simple parametric families (exponential, power‑law) together with
    uniform mixtures are explored on a grid.  All vectors are normalised
    to sum = 1 so the objective is scale‑invariant.

2.  A smooth‑max (log‑sum‑exp) surrogate is minimised by projected
    gradient descent.  The surrogate:
        F_β(a) = (1/β)·log Σ_k exp(β·conv_k(a))
    approaches the true max as β grows.  The gradient can be written as
    a correlation and is obtained with FFTs, giving O(n log n) per
    iteration while staying on the probability simplex.

3.  A short log‑normal random refinement around the best point.

The routine respects the 1000 s time budget, prints progress and finally
returns the best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a * a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def correlate_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear correlation (w ⋆ a)[ℓ] = Σ_j w_j a_{j+ℓ}.
    Returned length is len(w)+len(a)-1 (circular correlation on zero‑padded
    arrays, which equals the linear correlation for the required slice).
    """
    W = np.fft.rfft(w, fft_len)
    A = np.fft.rfft(a, fft_len)
    corr = np.fft.irfft(np.conj(W) * A, n=fft_len)
    return corr


def objective_vector(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective J(a) for a probability vector a."""
    A = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(A * A, n=conv_fft_len)[: 2 * n - 1]
    max_conv = conv.max()
    sum_a = a.sum()
    if sum_a == 0.0:
        return np.inf
    return float(2.0 * n * max_conv / (sum_a * sum_a))


def normalize_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renormalise to sum = 1."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s

            print(f"[gd {elapsed:6.2f}s] improved J = {best_J:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        shifted = conv - max_conv               # improves numeric stability
        weights = np.exp(beta * shifted)
        weights /= weights.sum()

        # gradient = 2·(weights ⋆ current)
        corr_full = correlate_fft(weights, current, corr_fft_len)[:corr_len]
        grad = 2.0 * corr_full[:n]              # first n lags = Σ_j a_j w_{i+j}

        # gradient descent step + projection onto simplex
        new_vec = current - step * grad
        new_vec = normalize_simplex(new_vec)

        current = new_vec
        it += 1

    # ------------------------------------------------------------------
    # 5. Final fine‑grained random refinement
    # ------------------------------------------------------------------
    fine_deadline = time.monotonic() + min(5.0, (deadline - start_time) * 0.05)
    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = normalize_simplex(best_vec * noise)
        Jc = objective_vector(cand, n, conv_fft_len)
        if Jc < best_J:
            best_J, best_vec = Jc, cand
            print(f"[fine] improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Return the best sequence (non‑negative, sum ≈ 1)
    # ------------------------------------------------------------------
    # No scaling is required – the objective is scale‑invariant.
    return [float(v) for v in best_vec]
