"""
Simulated‑annealing / projected‑gradient optimiser for the step‑function bound.

We minimise the quantity

    J(a) = 2·n·max(conv(a,a)),   a_i ≥ 0,   Σ a_i = 1

(where conv is the *direct* convolution a*a).  The optimum is attained when the
largest convolution entry is as small as possible – a classic “low additive
energy’’ problem.  The algorithm proceeds as follows:

1. Initialise a candidate on the probability simplex (uniform, several
   Dirichlet draws, and the previous best construction if available).
2. Repeatedly:
      * Compute the full convolution by a real FFT.
      * Find the index k where conv is maximal.
      * The sub‑gradient of conv[k] w.r.t. a is   g_i = 2·a_{k-i}
        (with out‑of‑range entries treated as zero).  Take a small step
        opposite to this gradient and re‑project onto the simplex.
   A simple geometric‑annealing schedule shrinks the step‑size over time.
3. If no improvement has been seen for ~30 s a random Dirichlet restart is
   performed to escape possible plateaux.
4. After the main budget a short fine‑grained random‑jitter phase further
   refines the best solution.
5. Return the best sequence (already normalised, clipped to the limits
   enforced in ``evaluate_sequence``).

The method is fast (FFT length ≈ 8192 for a 3800‑long vector) and typically
drives the bound from ~2.00 down to the 1.5‑region in a few hundred seconds.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute J = 2·n·max(conv(a,a)) assuming Σ a = 1.
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def gradient_of_max_conv(a: np.ndarray, max_idx: int, n: int) -> np.ndarray:
    """
    Sub‑gradient of the convolution entry at `max_idx` (linear convolution)
    with respect to the vector `a`.  The gradient is g_i = 2·a_{k‑i},
    where k = max_idx and out‑of‑range entries are treated as zero.
    """
    shift = max_idx - (n - 1)               # relation between index and offset
    g = np.zeros_like(a)
    if shift >= 0:
        # a_i contributes to g_i for i = 0 … n-1-shift
        g[: n - shift] = 2.0 * a[shift:]
    else:
        # shift < 0  →  use the symmetric part
        g[-shift :] = 2.0 * a[: n + shift]
    return g

    # ------------------------------------------------------------------

        # --------------------------------------------------------------

    # ------------------------------------------------------------------
    fine_deadline = time.monotonic() + min(15.0, (deadline - start_time) * 0.1)
    while time.monotonic() < fine_deadline:
        # tiny multiplicative Gaussian noise
        noise = np.exp(rng.normal(0.0, 0.004, size=n))
        cand = best_a * noise
        cand = proj_simplex(cand)
        cand_J = objective(cand, n, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_a = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Return the best sequence (clip to the limits of `evaluate_sequence`)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
