def gradient_max_convolution(a):
    n = a.size
    # compute self-convolution via fft for getting max index
    conv_len = 2*n - 1
    fft_len = next_pow2(conv_len)
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    k = int(conv.argmax())
    # gradient vector
    grad = np.zeros_like(a)
    idx = k - np.arange(n)
    mask = (idx >=0) & (idx < n)
    grad[mask] = 2 * a[idx[mask]]
    return grad

"""
Propose a new step‑function by:
 * starting from a diverse pool of hand‑crafted and random seeds,
 * performing a few hundred exponentiated‑gradient (mirror‑descent) steps
   that directly minimise the peak of the self‑convolution,
 * then refining the current best with stochastic multiplicative jitter and
   a short local gradient polish.
All vectors are kept on the probability simplex (non‑negative, sum = 1).
The algorithm is deliberately different from the previous DE + LP scheme
and works for any length (the default is the length of the previously
found best sequence, or 4096 if none is available).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas as required)
# ----------------------------------------------------------------------


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_max(vec: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the linear self‑convolution of *vec* using FFT."""
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def compute_J(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective used in the evaluation routine: 2·n·max(conv(vec,vec))."""
    return 2.0 * n * conv_max(vec, fft_len, conv_len)


def gradient_max_convolution(vec: np.ndarray) -> np.ndarray:
    """
    Gradient of the *max* entry of the self‑convolution.
    For the index k* where the convolution attains its maximum,
    d conv[k*] / d vec[j] = 2·vec[k*‑j] (zero if the index is out of bounds).
    """
    n = vec.size
    conv_len = 2 * n - 1
    fft_len = next_pow2(conv_len)

    # linear convolution via zero‑padded FFT
    A = np.fft.rfft(vec, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]

    k_star = int(conv.argmax())
    idx = k_star - np.arange(n)
    mask = (idx >= 0) & (idx < n)

    grad = np.zeros_like(vec)
    grad[mask] = 2.0 * vec[idx[mask]]
    return grad


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w
