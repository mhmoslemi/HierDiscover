"""
Propose a step‑function that (hopefully) beats the previous 1.513195 bound.

Algorithm (≈ 1000 s budget)
--------------------------------
* Multiple independent optimisation runs are performed.  Each run
  starts from a random Dirichlet draw (different concentration) or from
  the current globally‑best vector with a jitter.
* Inside a run we run a projected Adam optimiser on a *soft‑max*
  approximation of the max‑operator.
  – β (soft‑max temperature) is increased from 1 to **5000**
    (larger than in the reference code) to focus the gradient on the
    active convolution peaks.
  – The learning‑rate decays slowly (0.6 → 1e‑3) and Adam’s moments are
    reset after every jitter.
  – A Metropolis acceptance rule with a linearly cooling temperature
    allows occasional uphill jumps.
* When a run shows no improvement for 20 s we apply a multiplicative
  jitter to escape the basin.
* After completing all runs we polish the best vector with a cheap
  simulated‑annealing “pairwise‑mass‑swap’’ local search.
* The routine works for any length ≤ 8192 (the code automatically picks
  a power‑of‑two length if no previous solution exists) and respects
  the `evaluate_sequence` constraints (non‑negative, sum normalised to 1).

The implementation is fully vectorised (FFT‑based convolutions) and
runs comfortably within the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero -> uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear auto‑convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    weight length = 2·n‑1, a length = n.
    Returns a gradient vector of length n.
    """
    # reverse a for correlation
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return the evaluation metric J = 2·n·max(conv(a,a))."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

def _run_optimisation(initial: np.ndarray,
                      n: int,
                      total_seconds: float,
                      rng: np.random.Generator,
                      run_id: int = 0) -> tuple[np.ndarray, float]:
    """
    Perform a single Adam‑style optimisation run starting from `initial`.
    Returns the best vector found in this run and its objective value.
    """
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted corr

    # Adam hyper‑parameters
    lr_start = 0.6
    lr_end = 1e-3
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # Soft‑max temperature schedule
    beta_start = 1.0
    beta_end = 5000.0                     # more aggressive than before

    # Metropolis temperature schedule (linear decay)
    # start temperature is a modest fraction of the initial objective
    cur_vec = proj_simplex(initial.copy())
    cur_J = _objective(cur_vec, n, fft_len, conv_len)
    T_start = cur_J * 0.15   # 15 % of current J
    T_end = 1e-9

    best_vec = cur_vec.copy()
    best_J = cur_J

    # Adam state
    m = np.zeros_like(cur_vec)
    v = np.zeros_like(cur_vec)

    start_time = time.monotonic()
    deadline = start_time + total_seconds
    last_improve = start_time
    iteration = 0

    # Stagnation handling
    STAGNATE_SECONDS = 20.0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        t_frac = (now - start_time) / total_seconds  # 0 … 1

        # ---------- evaluate current solution ----------
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # ---------- keep global best ----------
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = cur_vec.copy()
            last_improve = now
            print(f"[run{run_id}] [{elapsed:6.2f}s] new best J = {best_J:.6f}")

        # ---------- stagnation → jitter ----------
        if now - last_improve > STAGNATE_SECONDS:
            # multiplicative log‑normal jitter
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            cur_vec = best_vec * noise
            cur_vec = proj_simplex(cur_vec)
            # reset Adam moments to avoid stale velocity
            m.fill(0.0)
            v.fill(0.0)
            last_improve = now
            print(f"[run{run_id}] [{elapsed:6.2f}s] jitter restart")

        # ---------- soft‑max weights ----------
        beta = beta_start * (beta_end / beta_start) ** (t_frac ** 1.2)
        shifted = conv - max_conv                     # numerical stability
        w_exp = np.exp(beta * shifted)
        w_sum = w_exp.sum()
        if w_sum == 0.0:
            weight = np.full_like(w_exp, 1.0 / w_exp.size)
        else:
            weight = w_exp / w_sum

        # ---------- gradient ----------
        grad = _weighted_gradient(cur_vec, weight, fft_len_corr, n)

        # ---------- Adam update ----------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        lr = lr_start * (1.0 - t_frac) + lr_end
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        proposal = cur_vec + step
        proposal = proj_simplex(proposal)

        # ---------- evaluate proposal ----------
        prop_J = _objective(proposal, n, fft_len, conv_len)

        # ---------- Metropolis acceptance ----------
        if prop_J < cur_J:
            accept = True
        else:
            T = T_start * (deadline - now) + T_end * (now - start_time)
            T /= (deadline - start_time)   # linear interpolation
            prob = np.exp(-(prop_J - cur_J) / max(T, 1e-12))
            accept = rng.random() < prob

        if accept:
            cur_vec = proposal
            cur_J = prop_J

    return best_vec, best_J


def _polish(vec: np.ndarray,
            n: int,
            budget_seconds: float,
            rng: np.random.Generator,
            base_J: float) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing polishing: random pairwise mass swaps.
    """
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    cur_vec = vec.copy()
    cur_J = base_J
    best_vec = cur_vec.copy()
    best_J = cur_J

    start = time.monotonic()
    deadline = start + budget_seconds

    # temperature schedule (log‑linear)
    T0 = cur_J * 0.01
    T_end = 1e-9

    while time.monotonic() < deadline:
        now = time.monotonic()
        t_frac = (now - start) / budget_seconds
        T = T0 * (1.0 - t_frac) + T_end * t_frac

        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue

        # move a random fraction (up to half) of mass from i to j
        eps = rng.random() * cur_vec[i] * 0.5
        if eps <= 0.0:
            continue

        cand = cur_vec.copy()
        cand[i] -= eps
        cand[j] += eps
        # enforce non‑negativity and simplex
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        if cand_J < cur_J:
            cur_vec, cur_J = cand, cand_J
            if cand_J < best_J - 1e-12:
                best_J, best_vec = cand_J, cand.copy()
                elapsed = now - start
                print(f"[polish] [{elapsed:6.2f}s] improvement J = {best_J:.6f}")
        else:
            # uphill move with Metropolis probability
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur_vec, cur_J = cand, cand_J

    return best_vec, best_J

            init = rng.dirichlet(np.full(n, alpha))

        # ----- perform the Adam optimisation -----
        vec, J = _run_optimisation(init, n, run_budget, rng, run_id=run_counter)

        # ----- keep global best -----
        if J < best_global_J - 1e-12:
            best_global_J = J
            best_global_vec = vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[global] [{elapsed:6.2f}s] new global best J = {best_global_J:.6f}")

        run_counter += 1

    # ------------------------------------------------------------------
    # 3. Final polishing (pairwise swaps + simulated annealing)
    # ------------------------------------------------------------------
    if best_global_vec is None:
        # fallback – uniform distribution
        best_global_vec = np.full(n, 1.0 / n, dtype=np.float64)
        best_global_J = _objective(best_global_vec, n, _next_pow2(2 * n - 1), 2 * n - 1)

    polish_time = max(0.0, deadline - time.monotonic())
    if polish_time > 2.0:
        best_global_vec, best_global_J = _polish(best_global_vec, n,
                                                polish_time, rng, best_global_J)

    # ------------------------------------------------------------------
    # 4. Final safety normalisation & conversion to plain list
    # ------------------------------------------------------------------
    best_global_vec = np.clip(best_global_vec, 0.0, 1000.0)
    total = best_global_vec.sum()
    if total == 0.0:
        best_global_vec[:] = 1.0 / n
    else:
        best_global_vec /= total

    final_elapsed = time.monotonic() - start_time
    print(f"[done] total time {final_elapsed:.2f}s – final J = {best_global_J:.6f}")

    return [float(v) for v in best_global_vec]
