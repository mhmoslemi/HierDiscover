def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs):
    """Search for a non‑negative step function that minimises `evaluate_sequence`.

    The algorithm combines several strategies:
    • a diversified set of random initial distributions;
    • simulated‑annealing on the probability simplex (mass‑transfer moves);
    • a final exponentiated‑gradient refinement that directly attacks the
      maximum entry of the auto‑convolution.
    The best exact objective `J = 2·n·max(conv)` encountered during the run is
    returned (after normalising the vector to unit sum)."""
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # Safety margin
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # Determine length n
    # ----------------------------------------------------------------------
    max_n = 8192
    min_n = 256
    if 'height_sequence_1' in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        base_len = base_seq.size
    else:
        base_seq = None
        base_len = 0

    # Use at least 4096 entries if the known construction is shorter,
    # otherwise keep the existing length (capped by max_n).
    if base_len >= min_n:
        n = max(base_len, 4096)
        n = min(n, max_n)
    else:
        n = max(min_n, 4096)
        n = min(n, max_n)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ----------------------------------------------------------------------
    # Build a pool of candidate initial vectors (all sum to 1)
    # ----------------------------------------------------------------------
    init_cands = []

    # Uniform
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Exponential decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Linear ramps
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bump
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    # Dirichlet draws
    for alpha in (0.2, 0.5, 1.0, 3.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    # Random binary‑spike patterns (K equally weighted spikes)
    for K in (int(np.sqrt(n)), int(np.sqrt(n) * 2), int(np.sqrt(n) * 4)):
        for _ in range(2):
            idx = rng.choice(n, size=K, replace=False)
            v = np.zeros(n, dtype=np.float64)
            v[idx] = 1.0 / K
            init_cands.append(v)

    # Include the previous best if it exists
    if base_seq is not None:
        v = np.copy(base_seq)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
        v = np.clip(v, 0.0, None)
        total = v.sum()
        if total == 0.0:
            v = np.full(n, 1.0 / n)
        else:
            v /= total
        init_cands.append(v)

    # ----------------------------------------------------------------------
    # Pick the best starter using the exact objective J
    # ----------------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_cands:
        J, _ = _conv_and_J(cand, fft_len, conv_len, n)
        if J < best_J:
            best_J = J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ----------------------------------------------------------------------
    # Simulated‑annealing phase
    # ----------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_J, _ = _conv_and_J(cur_a, fft_len, conv_len, n)

    global_best_a = cur_a.copy()
    global_best_J = cur_J

    # Temperature schedule parameters
    T_start = 0.05      # relatively high to accept occasional uphill moves
    T_end = 1e-6        # near‑deterministic at the end
    # Step‑size schedule for mass transfer
    scale_start = 0.5
    scale_end = 0.02

    it = 0
    last_improve = start
    # SA runs until 70 % of the time budget is consumed
    sa_deadline = start + 0.70 * (deadline - start)

    while time.monotonic() < sa_deadline:
        now = time.monotonic()
        progress = (now - start) / (sa_deadline - start)
        # linear annealing of temperature and move scale
        T = T_start + (T_end - T_start) * progress
        scale = scale_start + (scale_end - scale_start) * progress

        # Choose a source index proportionally to its current mass – this
        # prefers moving probability out of the biggest entries.
        src = rng.choice(n, p=cur_a)
        dst = rng.integers(n)
        while dst == src:
            dst = rng.integers(n)

        # Transfer a random fraction of the source mass
        delta = scale * rng.random() * cur_a[src]
        if delta <= 0.0:
            continue
        cand = cur_a.copy()
        cand[src] -= delta
        cand[dst] += delta

        cand_J, _ = _conv_and_J(cand, fft_len, conv_len, n)
        dJ = cand_J - cur_J

        # Metropolis acceptance
        if dJ < 0.0 or rng.random() < np.exp(-dJ / max(T, 1e-12)):
            cur_a = cand
            cur_J = cand_J

        # Record global best
        if cur_J < global_best_J - 1e-12:
            global_best_J = cur_J
            global_best_a = cur_a.copy()
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # inactivity.
        if now - last_improve > 30.0:

            cand = rng.dirichlet(np.ones(n) * 0.5)
            cand_J, _ = _conv_and_J(cand, fft_len, conv_len, n)
            cur_a = cand
            cur_J = cand_J
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] Restart – new random start (J = {cur_J:.6f})")

        it += 1
        if it % 5000 == 0:
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] SA iteration {it}, best J = {global_best_J:.6f}")

    # ----------------------------------------------------------------------
    # Exponentiated‑gradient refinement targeting the maximal convolution entry
    # ----------------------------------------------------------------------
    refine_deadline = deadline
    a_ref = global_best_a.copy()
    best_ref_J = global_best_J
    iter_ref = 0

    while time.monotonic() < refine_deadline:
        # Compute current convolution and locate the maximal index
        cur_J, conv = _conv_and_J(a_ref, fft_len, conv_len, n)
        if cur_J < best_ref_J - 1e-12:
            best_ref_J = cur_J
            global_best_a = a_ref.copy()
            global_best_J = cur_J
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] Refinement new best J = {global_best_J:.6f}")

        # Gradient of the active convolution entry (sub‑gradient of max)
        k = int(np.argmax(conv))

        # Build gradient vector: g_i = 2·a_{k-i} for admissible indices
        grad = np.zeros_like(a_ref)
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        # Vectorised fill
        i_range = np.arange(i_min, i_max + 1, dtype=np.int64)
        j_range = k - i_range
        grad[i_range] = 2.0 * a_ref[j_range]

        # Exponentiated‑gradient step with a diminishing learning rate
        step = 0.4 / np.sqrt(iter_ref + 1)
        a_new = a_ref * np.exp(-step * grad)
        # Renormalise to stay on the probability simplex
        total = a_new.sum()
        if total == 0.0:
            a_new = np.full_like(a_new, 1.0 / n)
        else:
            a_new /= total

        # Accept the step only if it does not worsen the exact objective
        new_J, _ = _conv_and_J(a_new, fft_len, conv_len, n)
        if new_J <= cur_J:
            a_ref = a_new
        # Otherwise keep the old vector – the diminishing step size will
        # eventually make a tentative move harmless.

        iter_ref += 1
        # Safety break if we have exhausted the remaining budget
        if time.monotonic() >= refine_deadline:
            break

    # ----------------------------------------------------------------------
    # Return the best vector (clipped and renormalised)
    # ----------------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:

"""
search for a step‑function minimizing `evaluate_sequence`.

The algorithm proceeds in three stages:

1. **Initialisation** – a diverse portfolio of distributions (uniform,
   geometric, ramps, Gaussians, Dirichlet draws and a few sparse “spike”
   patterns) is generated on the probability simplex.  The exact objective

        J(a) = 2·n·max(a * a)

   (where `*` denotes linear convolution) is evaluated and the best starter
   is kept.

2. **Simulated‑annealing** – the current vector lives on the simplex;
   a move transfers a random fraction of probability mass from a source
   entry (chosen proportionally to its current mass) to a destination
   entry.  Moves are accepted with the Metropolis rule using a linearly
   annealed temperature.  If no improvement occurs for 30 s the state is
   re‑initialised from a fresh Dirichlet draw.  The best exact objective
   found during this phase is stored.

3. **Exponentiated‑gradient refinement** – after the annealing phase the
   algorithm switches to a deterministic refinement that directly targets
   the maximal convolution entry.  The sub‑gradient of the active entry is
   `g_i = 2·a_{k−i}` (with `k = argmax conv`).  An exponentiated‑gradient
   step `a ← a·exp(−η·g)` is taken with a diminishing step size and the
   vector is re‑normalised to the simplex; the move is kept only if the
   exact objective does not increase.

All helper functions are defined at module level (no closures).  The
search respects the supplied time budget (default 1000 s) and prints a
few progress messages.  The returned list is a non‑negative sequence
(normalised to sum‑one, clipping to the range accepted by
`evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns the first `conv_len` entries of the linear convolution.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _conv_and_J(a: np.ndarray, fft_len: int, conv_len: int, n: int):
    """
    Compute the convolution a*a and the exact objective
        J(a) = 2·n·max(conv).
    Returns (J, conv).
    """
    conv = _conv_fft(a, fft_len, conv_len)
    return float(2.0 * n * conv.max()), conv

# Main search routine
# ----------------------------------------------------------------------

    Search for a non‑negative step function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for at most `budget_s` seconds and
    returns the best sequence discovered (normalised to unit sum).
    """
    rng = np.random.default_rng(seed)

    # 1. Choose a problem size (length of the sequence)
    # ------------------------------------------------------------------
    max_n = 8192
    min_n = 256

    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        base_len = base_seq.size
    else:
        base_seq = None
        base_len = 0

    if base_len >= min_n:
        # Keep the known construction size but allow a modest up‑scaling
        n = max(base_len, 4096)
        n = min(n, max_n)
    else:
        n = max(min_n, 4096)
        n = min(n, max_n)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Build a diverse pool of initial candidates (all sum to 1)
    # ------------------------------------------------------------------
    init_cands = []

    # Uniform distribution
    init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Geometric decays
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        init_cands.append(v)

    # Linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    init_cands.append(v)

    # Gaussian bumps
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        init_cands.append(v)

    for alpha in (0.2, 0.5, 1.0, 3.0):
        for _ in range(3):
            init_cands.append(rng.dirichlet(np.ones(n) * alpha))

    for K in (int(np.sqrt(n)), int(np.sqrt(n) * 2), int(np.sqrt(n) * 4)):
        for _ in range(2):

            v = np.zeros(n, dtype=np.float64)

        T = T_start + (T_end - T_start) * progress

        src = rng.choice(n, p=cur_a)                     # source index

        dJ = cand_J - cur_J

        # Metropolis acceptance

            cur_J = cand_J

    refine_deadline = deadline

        grad[i_range] = 2.0 * a_ref[j_range]
