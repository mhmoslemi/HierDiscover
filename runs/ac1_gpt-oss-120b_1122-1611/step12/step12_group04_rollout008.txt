    beta1=0.9; beta2=0.999; eps=1e-8
    lr = 0.5  # initial
    iteration=0
    while iteration < max_iter and time.monotonic() < time_budget:
        iteration +=1
        conv = _conv_full(a, fft_len, conv_len)
        residual = conv - m_target
        # compute gradient (2 * corr(residual, a))
        grad = _weighted_gradient(a, residual, fft_len_corr, n)
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a + step
        a = proj_simplex(a)
    return a

where `conv` is the full linear auto‑convolution.  The search proceeds in
four stages:

1.  Build a diverse collection of initial candidates (uniform, geometric,
    ramps, Gaussians, Dirichlet draws, discrete Beta PDFs and a spectral‑
    factorisation of a flat target autocorrelation).  The best few are
    retained.

2.  L2‑flattening: minimise the squared deviation of the autocorrelation
    from the perfectly flat target (½·(2n‑1)⁻¹) using projected Adam updates.
    This yields a distribution with a much flatter convolution profile.

3.  Max‑optimisation: from the L2‑solution we minimise a smooth approximation
    of the max‑operator via a soft‑max (log‑sum‑exp) with a temperature β
    that is increased from 1 to 8000 while Adam steps continue on the
    projected simplex.

4.  Local polishing: fast simulated‑annealing pair‑wise mass swaps and
    occasional multiplicative jitter escape any remaining basins.

The whole routine respects the 1000 s budget, updates the best objective
found, prints timing information and finally returns a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    weight length = 2·n‑1, a length = n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _geometric(n: int, r: float) -> np.ndarray:
    v = r ** np.arange(n, dtype=np.float64)
    v /= v.sum()
    return v


def _linear_ramp(n: int, upward: bool = True) -> np.ndarray:
    if upward:
        v = np.arange(1, n + 1, dtype=np.float64)
    else:
        v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    return v


def _gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    v /= v.sum()
    return v


def _beta_pdf(n: int, alpha: float, beta: float) -> np.ndarray:
    # Discretise the Beta(α,β) pdf on the mid‑points of n bins.
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Guard against α,β < 1 (divergent at the borders) – clip the pdf.
    v = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    v = np.nan_to_num(v, nan=0.0, posinf=0.0, neginf=0.0)
    v = np.maximum(v, 0.0)
    if v.sum() == 0.0:
        v[:] = 1.0 / n
    else:
        v /= v.sum()
    return v


def _spectral_factor(n: int, fft_len_corr: int) -> np.ndarray:
    """
    Approximate a spectral factor of the flat target autocorrelation.
    Takes the square‑root of the PSD of a constant vector and inverts.
    """
    conv_len = 2 * n - 1
    target = np.full(conv_len, 1.0 / conv_len, dtype=np.float64)
    # power‑spectral density (real FFT) of the target
    psd = np.fft.rfft(target, fft_len_corr)
    # enforce non‑negativity (numerical noise)
    psd = np.maximum(psd, 0.0)
    # take square‑root (spectral factor)
    sf = np.sqrt(psd)
    # inverse transform gives a (generally signed) sequence of length fft_len_corr
    a_full = np.fft.irfft(sf, n=fft_len_corr)
    a = a_full[:n]
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a[:] = 1.0 / n
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimization primitives
# ----------------------------------------------------------------------


def _adam_optimize(start_vec: np.ndarray,
                   grad_func,
                   proj_func,
                   n: int,
                   fft_len: int,
                   conv_len: int,
                   fft_len_corr: int,
                   beta_schedule,
                   lr_schedule,
                   time_deadline: float,
                   start_time: float,
                   name: str = "adam") -> tuple[np.ndarray, float]:
    """
    Generic projected Adam optimiser.
    - grad_func(vec) must return the gradient (same length as vec).
    - beta_schedule(t_frac) gives the soft‑max temperature β.
    - lr_schedule(t_frac) gives a learning‑rate multiplier.
    Returns the best vector seen and its objective.
    """
    rng = np.random.default_rng()
    cur = start_vec.copy()
    best = cur.copy()
    best_J = _objective(cur, n, fft_len, conv_len)
    m = np.zeros_like(cur)
    v = np.zeros_like(cur)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    it = 0
    while time.monotonic() < time_deadline:
        it += 1
        tfrac = (time.monotonic() - start_time) / (time_deadline - start_time)
        # gradient from user supplied function
        grad = grad_func(cur, tfrac)
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr_schedule(tfrac)
        step = - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        proposal = proj_func(cur + step)
        cur_J = _objective(proposal, n, fft_len, conv_len)
        # simple accept‑if‑better rule (no Metropolis here)
        if cur_J < best_J:
            best_J = cur_J
            best = proposal.copy()
        cur = proposal
    # final evaluation of best
    final_J = _objective(best, n, fft_len, conv_len)
    return best, final_J


def _grad_l2(a: np.ndarray, tfrac: float,
             n: int, fft_len: int, conv_len: int, fft_len_corr: int,
             target_val: float) -> np.ndarray:
    """Gradient of ∑(conv - target)² (the L2 flattening objective)."""
    conv = _conv_full(a, fft_len, conv_len)
    residual = conv - target_val
    # _weighted_gradient already includes the factor 2
    return _weighted_gradient(a, residual, fft_len_corr, n)


def _grad_softmax_max(a: np.ndarray, tfrac: float,
                     n: int, fft_len: int, conv_len: int, fft_len_corr: int,
                     beta_start: float, beta_end: float) -> np.ndarray:
    """Gradient of the soft‑max smoothed max‑objective."""
    conv = _conv_full(a, fft_len, conv_len)
    mx = conv.max()
    # temperature schedule (exponential)
    beta = beta_start * (beta_end / beta_start) ** tfrac
    # stable soft‑max
    shifted = conv - mx
    w_exp = np.exp(beta * shifted)
    w = w_exp / w_exp.sum()
    # gradient of weighted sum = 2·corr(w, a)
    return _weighted_gradient(a, w, fft_len_corr, n)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine runs for at most ``budget_s`` seconds and returns the best
    sequence found (as a list of floats).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + float(budget_s)

    # ------------------------------------------------------------------
    # 1. Choose problem size (prefer a power of two, up to 8192)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(512, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 8192                     # give us the strongest theoretical bound

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlations

    # ------------------------------------------------------------------
    # 2. Build a diverse initial pool
    # ------------------------------------------------------------------
    init_vectors = []

    # deterministic families
    init_vectors.append(_uniform(n))
    for r in [0.90, 0.94, 0.97, 0.985, 0.99]:
        init_vectors.append(_geometric(n, r))
    init_vectors.append(_linear_ramp(n, upward=False))
    init_vectors.append(_linear_ramp(n, upward=True))
    for sigma in [n / 12.0, n / 9.0, n / 6.0]:
        init_vectors.append(_gaussian(n, sigma))

    # beta‑pdf shapes
    beta_params = [(0.2, 0.2), (0.5, 0.5), (1.0, 1.0), (2.0, 2.0),
                   (0.5, 2.0), (2.0, 0.5), (5.0, 5.0)]
    for a0, b0 in beta_params:
        init_vectors.append(_beta_pdf(n, a0, b0))

    # Dirichlet draws with several concentrations
    alphas = [0.02, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0]
    for a in alphas:
        for _ in range(3):
            v = rng.dirichlet(np.ones(n) * a)
            init_vectors.append(v.astype(np.float64))

    # Spectral‑factorisation candidate
    init_vectors.append(_spectral_factor(n, fft_len_corr))

    # If a previous construction exists, add it (normalised)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_vectors.append(v)

    # ------------------------------------------------------------------
    # 3. Quick evaluation – keep the top K candidates
    # ------------------------------------------------------------------
    K = 8
    scored = []
    for idx, cand in enumerate(init_vectors):
        J = _objective(cand, n, fft_len, conv_len)
        scored.append((J, cand))
    scored.sort(key=lambda x: x[0])
    best_J = scored[0][0]
    best_vec = scored[0][1].copy()
    print(f"[{0.0:6.2f}s] Initial best J = {best_J:.6f}")

    # Keep the best K for the next stage
    top_candidates = [c for _, c in scored[:K]]

    # ------------------------------------------------------------------
    # 4. Phase‑1: L2‑flattening (minimise ‖conv - uniform‖²)
    # ------------------------------------------------------------------
    target_val = 1.0 / conv_len
    phase1_deadline = start_time + 0.30 * budget_s   # 30 % of total time
    for idx, cand in enumerate(top_candidates):
        if time.monotonic() >= phase1_deadline:
            break
        # Adam optimiser specialised for the L2 objective
        def grad_func(vec, _tfrac, _cand=cand):
            return _grad_l2(vec, 0.0,
                            n, fft_len, conv_len, fft_len_corr, target_val)

        # Simple constant learning‑rate schedule (decaying a bit)
        lr_sched = lambda tf: 0.7 * (1.0 - tf) + 0.02

        vec_opt, J_opt = _adam_optimize(
            start_vec=cand,
            grad_func=grad_func,
            proj_func=proj_simplex,
            n=n,
            fft_len=fft_len,
            conv_len=conv_len,
            fft_len_corr=fft_len_corr,
            beta_schedule=lambda _: 1.0,   # unused for L2
            lr_schedule=lr_sched,
            time_deadline=phase1_deadline,
            start_time=start_time,
            name="L2‑adam"
        )
        if J_opt < best_J:
            best_J, best_vec = J_opt, vec_opt.copy()
            print(f"[{time.monotonic()-start_time:6.2f}s] L2 improve → J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Phase‑2: Soft‑max approximation of the max‑objective
    # ------------------------------------------------------------------
    phase2_deadline = start_time + 0.70 * budget_s   # up to 70 % of total time
    beta_start = 1.0
    beta_end = 8000.0

    def grad_softmax(vec, tfrac):
        return _grad_softmax_max(vec, tfrac,
                                 n, fft_len, conv_len, fft_len_corr,
                                 beta_start, beta_end)

    # Learning‑rate decays linearly from 0.8 to 0.01
    lr_sched2 = lambda tf: 0.8 * (1.0 - tf) + 0.01

    vec_soft, J_soft = _adam_optimize(
        start_vec=best_vec,
        grad_func=grad_softmax,
        proj_func=proj_simplex,
        n=n,
        fft_len=fft_len,
        conv_len=conv_len,
        fft_len_corr=fft_len_corr,
        beta_schedule=lambda tf: beta_start * (beta_end / beta_start) ** tf,
        lr_schedule=lr_sched2,
        time_deadline=phase2_deadline,
        start_time=start_time,
        name="soft‑max‑adam"
    )
    if J_soft < best_J:
        best_J, best_vec = J_soft, vec_soft.copy()
        print(f"[{time.monotonic()-start_time:6.2f}s] Soft‑max improve → J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Phase‑3: Local polishing (pairwise mass swaps with SA)
    # ------------------------------------------------------------------
    polish_deadline = deadline - 2.0                     # keep a couple seconds margin
    cur = best_vec.copy()
    cur_J = best_J
    cur_conv = _conv_full(cur, fft_len, conv_len)
    T0 = cur_J * 0.02
    T_end = 1e-9

    while time.monotonic() < polish_deadline:
        t = (time.monotonic() - start_time) / budget_s
        T = T0 * (1.0 - t) + T_end * t
        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # Transfer a random fraction (up to 40 %) from i to j
        eps = rng.random() * cur[i] * 0.4
        if eps <= 0.0:
            continue
        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)
        cand_J = _objective(cand, n, fft_len, conv_len)
        if cand_J < cur_J:
            cur, cur_J = cand, cand_J
            if cand_J < best_J:
                best_J, best_vec = cand_J, cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polishing improvement → J = {best_J:.6f}")
        else:
            # occasional uphill acceptance
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur, cur_J = cand, cand_J

        # occasional jitter if stuck for >20 s
        if time.monotonic() - start_time - (budget_s - (polish_deadline - time.monotonic())) > 20.0:
            if time.monotonic() > polish_deadline:
                break
            jitter = np.exp(rng.normal(0.0, 0.05, size=n))
            cur = proj_simplex(best_vec * jitter)
            cur_J = _objective(cur, n, fft_len, conv_len)

    # ------------------------------------------------------------------
    # 7. Final normalisation & output
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s
    return [float(v) for v in best_vec]
