Gradient‑descent on a smoothed max‑convolution objective.

We minimise

 * performing a mirror‑descent step (exponential‑weights update) and
   projecting back onto the probability simplex.

The temperature β is increased during the run so the method
starts with a very smooth objective (large basin of attraction)
and gradually focuses on the true max.  Stagnation triggers a brief
random restart to escape possible plateaus.  The best vector found
throughout the budget is returned as a plain Python list.

The implementation is deliberately different from the previous
DE + polish approach and uses only a single candidate with
adaptive smoothing.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper routines – all are top‑level, no closures or lambdas
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_corr(a: np.ndarray,
                   weight: np.ndarray,
                   fft_len_corr: int,
                   n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` must have length (2·n‑1); the result is a gradient of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _softmax(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max over the convolution values.
    Larger β puts more weight on the larger entries (approaches the max).
    """
    scaled = beta * conv
    m = np.max(scaled)
    exps = np.exp(scaled - m)          # stable exponentiation
    return exps / exps.sum()


def _eval_J(conv: np.ndarray, n: int) -> float:
    """Objective J = 2·n·max(conv).  `conv` must be the full self‑convolution."""
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size – use the length of a previously found sequence
    #    if it exists, otherwise fall back to a comfortable default.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        init = np.asarray(height_sequence_1, dtype=np.float64)
        init = np.clip(init, 0.0, None)
        total = init.sum()
        if total > 0.0:
            init = init / total
        else:
            init = None
    else:
        init = None

    if init is None:
        # no prior solution → start from a mildly perturbed uniform vector
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)
        a = a * (1.0 + 0.01 * rng.normal(size=n))
        a = proj_simplex(a)
    else:
        n0 = init.shape[0]
        n = max(256, min(n0, 8192))
        if n0 != n:
            if n0 > n:
                init = init[:n]
            else:
                init = np.concatenate([init, np.zeros(n - n0, dtype=np.float64)])
        a = init.copy()

    # --------------------------------------------------------------
    # 2. Pre‑compute FFT lengths for convolution and weighted correlation.
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)        # for weight⋆a_rev

    # --------------------------------------------------------------
    # 3. Initial evaluation.
    # --------------------------------------------------------------
    conv = _conv_full(a, fft_len, conv_len)
    best_J = _eval_J(conv, n)
    best_vec = a.copy()
    print(f"[0.00s] Initial J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Parameters for the smoothing/gradient descent loop.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    beta = 0.5                     # start very smooth
    beta_factor = 1.25             # geometric increase of β
    max_beta = 50.0                # final (almost exact) max

    lr0 = 0.6                      # base learning‑rate for the exponentiated step
    iteration = 0
    last_improve = start_time
    stagnation_limit = 30.0        # seconds without improvement → random restart

    # --------------------------------------------------------------
    # 5. Main loop – smoothed max‑convolution descent with occasional restarts.
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- current convolution & smooth weights -----
        conv = _conv_full(a, fft_len, conv_len)
        J_current = _eval_J(conv, n)

        weight = _softmax(conv, beta)           # ∈ simplex, emphasises large entries
        grad = _weighted_corr(a, weight, fft_len_corr, n)   # subgradient of the smoothed objective

        # ----- mirror‑descent update (exponential weights) -----
        lr = lr0 / np.sqrt(iteration + 1)       # diminishing step size
        factor = np.exp(-lr * grad)
        a_new = a * factor
        a_new = proj_simplex(a_new)

        # ----- evaluate new point -----
        conv_new = _conv_full(a_new, fft_len, conv_len)
        J_new = _eval_J(conv_new, n)

        # ----- keep the best ever found -----
        if J_new < best_J - 1e-12:
            best_J = J_new
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- accept the move if it does not increase the objective -----
        if J_new <= J_current:
            a = a_new
            # raise temperature a little faster when the objective improves
            if iteration % 20 == 0:
                beta = min(beta * beta_factor, max_beta)
        else:
            # a stays unchanged; still increase temperature slowly
            if iteration % 20 == 0:
                beta = min(beta * 1.10, max_beta)

        iteration += 1

        # ----- stagnation handling -----
        now = time.monotonic()
        if now - last_improve > stagnation_limit:
            # inject diversity – fresh Dirichlet draw mixed with current best
            cand = rng.dirichlet(np.ones(n) * 0.2)
            cand = 0.5 * cand + 0.5 * best_vec
            cand = proj_simplex(cand)
            a = cand
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart due to stagnation (β={beta:.3f})")

    # ------------------------------------------------------------------
    # 6. Final deterministic polishing – active‑set (hard max) gradient step.
    # ------------------------------------------------------------------
    conv = _conv_full(best_vec, fft_len, conv_len)
    max_conv = conv.max()
    active = (conv >= max_conv - 1e-12).astype(np.float64)   # binary mask of binding constraints
    grad_active = _weighted_corr(best_vec, active, fft_len_corr, n)

    # try a handful of step‑sizes and keep the best polishing
    for step in (0.3, 0.6, 0.9, 1.2, 1.5):
        cand = best_vec * np.exp(-step * grad_active)
        cand = proj_simplex(cand)
        conv_cand = _conv_full(cand, fft_len, conv_len)
        J_cand = _eval_J(conv_cand, n)
        if J_cand < best_J:
            best_J = J_cand
            best_vec = cand
            print(f"Polish step {step:.2f} improved J to {best_J:.6f}")

    # ------------------------------------------------------------------
    # 7. Clip, renormalise and return as a plain Python list.
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # Optional sanity‑check (commented out to avoid unnecessary imports)
    # try:
    #     val = evaluate_sequence(best_vec.tolist())
    #     print(f"Final evaluate_sequence = {val:.6f}")
    # except Exception:
    #     pass

    return [float(v) for v in best_vec]
```
