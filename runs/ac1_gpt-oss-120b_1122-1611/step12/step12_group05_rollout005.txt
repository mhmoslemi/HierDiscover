def evaluate_sequence(sequence):
    # ... convert each element to float
    sequence = [float(x) for x in sequence]
    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]
    # Protect against numbers too large (>1000)
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against sum too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))
```

def propose_candidate(seed=1234, budget_s=1000.0, **unused_kwargs):
    rng = np.random.default_rng(seed)
    # Config for n
    if 'height_sequence_1' in globals():
        try:
            n = int(len(height_sequence_1))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    L_grad = _next_pow2(conv_len + n - 1)

    # objective compute
    def compute_J_local(a):
        conv = _conv_full(a, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    best_a = None
    best_J = np.inf
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)  # keep small margin

    # function to run a descent for given initial vector and subbudget
    def run_descent(a_init, sub_budget):
        nonlocal best_a, best_J
        # Use copy
        a = a_init.copy()
        cur_J = compute_J_local(a)
        cur_best_J = cur_J
        cur_best_a = a.copy()
        iter_cnt = 0
        last_improve = time.monotonic()
        # step schedule parameters
        step0 = 0.8  # initial step magnitude
        # We may set beta parameter for decreasing step: step = step0 / sqrt(iter+1)
        # Simulated annealing temperature schedule: start T = cur_J, end = 1e-8
        start_T = cur_J
        end_T = 1e-8
        # use local start time
        run_start = time.monotonic()
        while time.monotonic() < run_start + sub_budget:
            now = time.monotonic()
            elapsed_total = now - start_time
            # compute conv and gradient
            conv = _conv_full(a, fft_len, conv_len)
            max_conv = conv.max()
            # compute J
            J = float(2.0 * n * max_conv)
            if J < best_J:
                best_J = J
                best_a = a.copy()
                print(f"[{elapsed_total:.2f}s] New global best J={best_J:.6f}")

            # active set: indices within relative tolerance
            tol = max(1e-12, max_conv * 1e-12)
            active = np.where(conv >= max_conv - tol)[0]
            # Compute gradient using active set indicator
            if len(active) > 0:
                w = np.zeros(conv_len, dtype=np.float64)
                w[active] = 1.0 / len(active)
                conv2 = conv_fft(w, a[::-1], L_grad)
                grad = 2.0 * conv2[n-1:n-1+n]
            else:
                grad = np.zeros_like(a)

            # step size schedule
            step = step0 / np.sqrt(iter_cnt + 1)
            # Proposed step
            a_prop = proj_simplex(a - step * grad)

            # occasional jitter
            if rng.random() < 0.02:
                noise = np.exp(rng.normal(0.0, 0.07, size=n))
                a_prop = proj_simplex(a_prop * noise)

            # evaluate
            conv_prop = _conv_full(a_prop, fft_len, conv_len)
            J_prop = float(2.0 * n * conv_prop.max())

            # Acceptance schedule
            # Temperature
            total_time = deadline - start_time
            T = start_T * (end_T / start_T) ** ((now - start_time) / total_time)
            accept = False
            if J_prop < J:
                accept = True
            else:
                # Metropolis
                prob = np.exp(-(J_prop - J) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                a = a_prop
                cur_J = J_prop
                if J_prop < cur_best_J:
                    cur_best_J = J_prop
                    cur_best_a = a.copy()
                    last_improve = now

            # stagnation: if no improvement for 15 seconds, jitter
            if now - last_improve > 15.0:
                # reinitialize around best
                jitter = np.exp(rng.normal(0.0, 0.06, size=n))
                a = proj_simplex(best_a * jitter)
                cur_J = compute_J_local(a)
                last_improve = now
                print(f"[{now - start_time:.2f}s] Stagnation jitter J={cur_J:.6f}")

            iter_cnt += 1

        # end of run
        return

    # Determine number of restarts based on time and a baseline subbudget,
    # maybe 4 restarts each 200 s.

    remaining_time = deadline - start_time
    # allocate minimal of 150 seconds per restart
    min_sub = 150.0
    num_restarts = max(1, int(remaining_time // min_sub))
    # Ensure at least 1 and at most maybe 6.
    num_restarts = max(1, min(num_restarts, 6))
    sub_budget = remaining_time / num_restarts

    # For each restart, decide initial vector.
    for r in range(num_restarts):
        now = time.monotonic()
        if now >= deadline:
            break
        # Choose init
        if r == 0 and 'height_sequence_1' in globals():
            a0 = np.asarray(height_sequence_1, dtype=np.float64)
            # adjust size
            if a0.size != n:
                if a0.size > n:
                    a0 = a0[:n]
                else:
                    a0 = np.pad(a0, (0, n - a0.size))
            # clip negative, cap
            a0 = np.clip(a0, 0.0, None)
            total = a0.sum()
            if total == 0.0:
                a0[:] = 1.0 / n
            else:
                a0 /= total
        else:
            # Dirichlet random init, maybe sparse
            alpha = 0.5  # encourage sparsity
            a0 = rng.gamma(alpha, 1.0, size=n)
            if a0.sum() == 0:
                a0[:] = 1.0 / n
            else:
                a0 /= a0.sum()
        # Run descent
        run_descent(a0, sub_budget)

    # Ensure best_a set
    if best_a is None:
        # fallback uniform
        best_a = np.full(n, 1.0 / n)

    # final sanitize
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]

"""
Hybrid multi‑restart sub‑gradient descent for the max‑convolution problem.

* The search works on the probability simplex (∑a_i = 1, a_i ≥ 0);
  scaling does not change ``evaluate_sequence`` because the objective is
  homogeneous of degree 0.
* Several independent runs (restarts) are performed within the time budget.
  Each run starts either from the known good sequence (``height_sequence_1``)
  or from a random Dirichlet draw.
* Inside a run we minimise   J(a)=2·n·max(conv(a,a)) .
  The max‑convolution is smoothed by an active‑set (exact sub‑gradient)
  computation: we identify the indices where the convolution attains its
  maximum (up to a tiny tolerance) and average the gradients of those
  active constraints.
* A simple sub‑gradient descent with a √t decreasing step size is used.
  The update is projected back onto the simplex with the Euclidean
  projection routine ``proj_simplex``.
* A simulated‑annealing acceptance rule (temperature decays from the
  current objective to 1e‑8) allows occasional uphill moves, helping the
  algorithm escape shallow local minima.
* If the method stagnates for >15 s a multiplicative jitter around the
  best global solution is performed to explore a new region.
* The best sequence found over all restarts is returned (clipped to the
  allowed range and re‑normalised).

The routine respects the 1000 s budget – if the budget runs out the best
vector obtained so far is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the simplex {x≥0, sum=1}."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1, dtype=v.dtype)
    cond = u - cssv / ind > 0
    if not np.any(cond):                         # all equal or zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_fft(x: np.ndarray, y: np.ndarray, L: int) -> np.ndarray:
    """Full linear convolution of ``x`` and ``y`` using an FFT of length ``L``."""
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    out = np.fft.irfft(X * Y, n=L)
    return out[: len(x) + len(y) - 1]


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT of fixed length ``fft_len``."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Objective J = 2·n·max(conv(a,a)) for a vector already on the simplex.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose a problem size (length of the vector)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                                 # a convenient power of two

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # for a*a convolution
    L_grad = _next_pow2(conv_len + n - 1)        # for gradient convolution

    # --------------------------------------------------------------
    # 2. Global bookkeeping
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 0.2)  # tiny safety margin

    # --------------------------------------------------------------
    # 3. Sub‑gradient descent on a single run
    # --------------------------------------------------------------
    def run_descent(a_init: np.ndarray, run_budget: float) -> None:
        """Perform a sub‑gradient search starting from ``a_init``."""
        nonlocal best_a, best_J

        a = a_init.copy()
        cur_J = compute_J(a, n, fft_len, conv_len)
        # keep a copy of the best local solution (for stagnation jitter)
        local_best_a = a.copy()
        local_best_J = cur_J

        # step‑size schedule parameters
        step0 = 0.8                               # initial magnitude
        # temperature schedule (simulated annealing)
        T_start = cur_J
        T_end = 1e-8

        iter_cnt = 0
        last_improve = time.monotonic()

        while time.monotonic() < start_time + run_budget:
            now = time.monotonic()
            elapsed_total = now - start_time

            # ------------------------------------------------------
            # 3.1 Current convolution and active set
            # ------------------------------------------------------
            conv = _conv_full(a, fft_len, conv_len)
            max_conv = conv.max()
            J = float(2.0 * n * max_conv)

            # global update
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
                print(f"[{elapsed_total:6.2f}s] Global best J = {best_J:.6f}")

            # local update
            if J < local_best_J - 1e-12:
                local_best_J = J
                local_best_a = a.copy()
                last_improve = now

            # ------------------------------------------------------
            # 3.2 Compute a sub‑gradient of the max‑convolution term
            # ------------------------------------------------------
            # tolerance for “active” constraints – relative to the maximum
            tol = max(1e-12, max_conv * 1e-12)
            active = np.where(conv >= max_conv - tol)[0]

            if active.size == 0:
                # fallback (should never happen)
                grad = np.zeros_like(a)
            else:
                # Indicator vector for the active constraints, equally weighted
                w = np.zeros(conv_len, dtype=np.float64)
                w[active] = 1.0 / active.size

                # convolution of w with the reversed a gives the summed gradients
                conv2 = conv_fft(w, a[::-1], L_grad)      # length 3n‑2
                grad = 2.0 * conv2[n - 1:n - 1 + n]       # shape (n,)

            # ------------------------------------------------------
            # 3.3 Step size and proposal
            # ------------------------------------------------------
            step = step0 / np.sqrt(iter_cnt + 1.0)
            a_prop = proj_simplex(a - step * grad)

            # occasional random jitter (helps escape flat regions)
            if rng.random() < 0.02:
                noise = np.exp(rng.normal(0.0, 0.07, size=n))
                a_prop = proj_simplex(a_prop * noise)

            # ------------------------------------------------------
            # 3.4 Evaluate the proposal
            # ------------------------------------------------------
            conv_prop = _conv_full(a_prop, fft_len, conv_len)
            J_prop = float(2.0 * n * conv_prop.max())

            # ------------------------------------------------------
            # 3.5 Simulated‑annealing acceptance
            # ------------------------------------------------------
            total_time = deadline - start_time
            T = T_start * (T_end / T_start) ** ((now - start_time) / total_time)
            accept = False
            if J_prop < J:
                accept = True
            else:
                prob = np.exp(-(J_prop - J) / max(T, 1e-12))
                if rng.random() < prob:
                    accept = True

            if accept:
                a = a_prop
                cur_J = J_prop

            # ------------------------------------------------------
            # 3.6 Stagnation handling – jitter around the global best
            # ------------------------------------------------------
            if now - last_improve > 15.0:
                # Re‑initialise around the best known vector
                if best_a is not None:
                    jitter = np.exp(rng.normal(0.0, 0.06, size=n))
                    a = proj_simplex(best_a * jitter)
                else:
                    a = proj_simplex(rng.random(n))
                cur_J = compute_J(a, n, fft_len, conv_len)
                last_improve = now
                print(f"[{elapsed_total:6.2f}s] Stagnation jitter – J = {cur_J:.6f}")

            iter_cnt += 1

    # --------------------------------------------------------------
    # 4. Allocate time to a handful of independent restarts
    # --------------------------------------------------------------
    remaining = deadline - time.monotonic()
    # aim for at least 4 restarts, but never use less than ~120 s per run
    min_per_restart = 130.0
    nruns = max(1, int(min(6, remaining // min_per_restart)))
    per_run_budget = remaining / nruns

    for run_idx in range(nruns):
        now = time.monotonic()
        if now >= deadline:
            break

        # ----------------------------------------------------------
        # 4.1 Choose an initial vector for this restart
        # ----------------------------------------------------------
        if run_idx == 0 and "height_sequence_1" in globals():
            a0 = np.asarray(height_sequence_1, dtype=np.float64)
            if a0.size != n:
                if a0.size > n:
                    a0 = a0[:n]
                else:
                    a0 = np.pad(a0, (0, n - a0.size))
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s == 0.0:
                a0[:] = 1.0 / n
            else:
                a0 /= s
        else:
            # Dirichlet‑type random start (alpha < 1 favours sparsity)
            alpha = 0.5
            a0 = rng.gamma(alpha, 1.0, size=n)
            s = a0.sum()
            if s == 0.0:
                a0[:] = 1.0 / n
            else:
                a0 /= s

        # ----------------------------------------------------------
        # 4.2 Run the descent for the allocated budget
        # ----------------------------------------------------------
        run_descent(a0, per_run_budget)

    # --------------------------------------------------------------
    # 5. Fallback (in case nothing was ever accepted)
    # --------------------------------------------------------------
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 6. Final sanitisation (clip to allowed domain and renormalise)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    return [float(v) for v in best_a]
