"""
Projected (smooth‑max) gradient descent for minimizing the
self‑convolution L∞‑norm of a non‑negative step function.

The objective that evaluate_sequence() returns (after cancelling the
scale) is

    J(p) = 2·n·max_k (p * p)_k ,   p_i ≥ 0 ,   Σ p_i = 1 .

We treat p as a probability vector and minimise the smooth approximation
of the maximum (log‑sum‑exp).  In each iteration we

1. compute the full linear convolution p∗p (FFT, O(n log n));
2. form soft‑max weights w_i ≈ 1 if (p∗p)_i is large;
3. obtain the gradient  ∇f(p)=2·(w_rev * p)   (convolution again via FFT);
4. take a step in the negative gradient direction, then project back onto
   the probability simplex (Euclidean projection).

A simple line‑search guarantees monotone improvement.  When progress stalls
for ~30 s we inject fresh random Dirichlet candidates to escape local minima.
Finally a short random‑jitter phase explores the neighbourhood of the best
solution found.

The routine respects the 1000 s budget, works for vectors of a few thousand
entries and has no external I/O.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – all equal
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def conv_cross_fft(w: np.ndarray, p: np.ndarray,
                   fft_len: int, out_len: int) -> np.ndarray:
    """
    Convolution of w (length L) with p (length n) using FFT.
    Returns the first `out_len` entries of the linear convolution.
    """
    W = np.fft.rfft(w, fft_len)
    P = np.fft.rfft(p, fft_len)
    conv = np.fft.irfft(W * P, n=fft_len)
    return conv[:out_len]

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Projected smooth‑max gradient descent for the step‑function problem.
    Returns the best non‑negative sequence found within the allotted time.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n_prev = len(globals()["height_sequence_1"])
            n = max(2048, min(n_prev, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    # --------------------------------------------------------------
    # 2. Initialise the probability vector
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        v0 = np.asarray(height_sequence_1, dtype=np.float64)
        v0 = np.clip(v0, 0.0, None)
        if v0.size != n:
            if v0.size > n:
                v0 = v0[:n]
            else:
                pad = np.zeros(n - v0.size, dtype=np.float64)
                v0 = np.concatenate([v0, pad])
        p = proj_simplex(v0)
    else:
        # uniform start + a tiny random perturbation
        p = np.full(n, 1.0 / n, dtype=np.float64)
        p += 0.001 * rng.normal(size=n)
        p = proj_simplex(p)

    # --------------------------------------------------------------
    # 3. Pre‑compute FFT lengths
    # --------------------------------------------------------------
    conv_len = 2 * n - 1                     # length of p * p
    fft_len_conv = _next_pow2(conv_len)      # for self‑convolution
    grad_len = conv_len + n - 1               # length of w_rev * p
    fft_len_grad = _next_pow2(grad_len)      # for gradient convolution

    # --------------------------------------------------------------

        c = conv_self(vec)
        return float(2.0 * n * c.max())

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    start_time = time.monotonic()

    best_vec = p.copy()

    # algorithmic hyper‑parameters
    alpha = 30.0                 # smooth‑max temperature (will slowly increase)
    tau0 = 1.0                   # base step size (will be scaled by gradient norm)
    decay_alpha = 1.001          # gently increase focus on the true max
    stagnation_time = 30.0       # seconds without improvement before a restart
    last_improve = start_time
    it = 0

        elapsed = now - start_time

        conv = conv_self(p)

        w = exps / exps.sum()                      # weights sum to 1

        # ----- gradient of the smooth‑max objective -----
        w_rev = w[::-1]                            # reverse for cross‑convolution

        # Extract the n components that correspond to ∂/∂p_j

        grad = 2.0 * grad_full[start:start + n]    # shape (n,)

        # ----- line‑search step (projected) -----
        # step size is adapted to the gradient magnitude
        grad_norm = np.linalg.norm(grad) + 1e-12
        step = tau0 / (1.0 + 0.001 * it) / grad_norm
        improved = False
        for _ in range(5):                         # up to 5 back‑track attempts
            cand = p - step * grad
            cand = proj_simplex(cand)
            cand_J = objective(cand)
            if cand_J < best_J - 1e-12:

            jitter = 0.02 * rng.normal(size=n)
            p = proj_simplex(p + jitter)

        # ----- schedule updates -----

        it += 1

        # ----- stagnation handling -----
        if now - last_improve > stagnation_time:
            # replace ~20 % of the mass with a fresh Dirichlet draw
            p = rng.dirichlet(np.ones(n) * 0.7)
            p = proj_simplex(p)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (stagnation)")

    # --------------------------------------------------------------
    # 5. Fine‑grained random search around the best found vector
    # --------------------------------------------------------------
    fine_deadline = time.monotonic() + min(30.0, (deadline - start_time) * 0.05)
    while time.monotonic() < fine_deadline:

            best_vec = cand

    # --------------------------------------------------------------

    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
