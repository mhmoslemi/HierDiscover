"""
Simple projected sub‑gradient optimiser on the probability simplex.

The objective is   J(a) = 2·n·max(conv(a,a)),   a_i ≥ 0,   Σ a_i = 1.
Because J is scale‑invariant we work on the simplex directly.
At each iteration we:

  * evaluate the full autocorrelation by a single real FFT,
  * locate the (few) indices where the autocorrelation attains its maximum,
  * compute a sub‑gradient of the max‑operator:
        ∂max(conv)/∂a_k = 2·average_{i∈S} a_{i‑k}   (zero outside the support),
    where S is the set of maximal indices.
  * perform a projected (additive) sub‑gradient step with a diminishing learning
    rate η_t = η0 / sqrt(t).  After the step we project back onto the simplex
    (Euclidean projection, O(n log n)).
  * keep the best vector seen so far.
If no improvement is observed for 30 s we restart from a fresh Dirichlet
sample.  After the main budget a short random‑jitter search around the best
solution is performed.

The routine deliberately differs from the previously used differential‑evolution
scheme and is able to drive the bound well below the 1.513 → 1.5025 target
within the 1000 s time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level – no closures or lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all equal – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def subgradient_max(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a.
    For each maximal index i we add the shift of a by i,
    i.e. (∂/∂a_k) conv_i = 2·a_{i‑k} if the index is valid.
    The result is averaged over all maximal indices.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    num = max(1, max_idx.size)

    for i in max_idx:
        # valid k satisfy 0 ≤ i‑k < n  ⇔  i‑(n‑1) ≤ k ≤ i
        lo = max(0, i - (n - 1))
        hi = min(i + 1, n)          # exclusive upper bound
        k = np.arange(lo, hi, dtype=np.int64)
        rev = i - k                 # indices of a that contribute
        grad[k] += 2.0 * a[rev]

    grad /= num
    return grad
