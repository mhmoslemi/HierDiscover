"""
Differential Evolution (DE) search for a non‑negative step‑function.

The candidate is kept on the probability simplex (∑a_i = 1, a_i ≥ 0) so
that the evaluation function reduces to

    J = 2·n·max(conv(a,a))

where conv(a,a) is the full linear convolution.  DE performs
mutation “rand/1” and binomial crossover, projecting every offspring onto
the simplex.  The known best construction (global ``height_sequence_1``)
is injected as the first individual; the remaining population is drawn
from a Dirichlet distribution.  The algorithm runs until the time budget
expires, periodically jittering the current best solution to escape
plateaus and finishing with a short hill‑climbing refinement.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | sum(x)=1, x_i ≥ 0 }.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute the objective J = 2·n·max(conv(a,a)) for a vector a that already
    satisfies ∑a = 1.  Convolution is performed via FFT for speed.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def random_simplex(rng: np.random.Generator, n: int) -> np.ndarray:
    """Draw a random point uniformly from the n‑dimensional probability simplex."""
    v = rng.exponential(size=n)
    s = v.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return v / s


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size ---------------------------------------------------
    # ------------------------------------------------------------------
    # Use the length of the known best construction if available,
    # otherwise start with a power‑of‑two size that is fast for FFT.
    if "height_sequence_1" in globals():
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        base_seq = np.clip(base_seq, 0.0, None)
        n0 = base_seq.size
        # Force a power of two (helps FFT) but do not exceed 8192.
        n = min(8192, max(256, _next_pow2(n0)))
        if base_seq.size != n:
            if base_seq.size > n:
                base_seq = base_seq[:n]
            else:
                base_seq = np.pad(base_seq, (0, n - base_seq.size))
        # Normalise to sum = 1
        s = base_seq.sum()
        if s == 0.0:
            base_seq[:] = 1.0 / n
        else:
            base_seq /= s
    else:
        # No prior construction – fall back to a modest power of two.
        n = 4096
        base_seq = None

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Population initialisation --------------------------------------
    # ------------------------------------------------------------------
    popsize = 20                     # moderate size – fast enough for many generations
    pop = np.empty((popsize, n), dtype=np.float64)
    J_pop = np.empty(popsize, dtype=np.float64)

    # First individual: the known best (if any)
    if base_seq is not None:
        pop[0] = base_seq
    else:
        pop[0] = random_simplex(rng, n)
    J_pop[0] = compute_J(pop[0], n, fft_len, conv_len)

    # Remaining individuals: random Dirichlet draws
    for i in range(1, popsize):
        pop[i] = random_simplex(rng, n)
        J_pop[i] = compute_J(pop[i], n, fft_len, conv_len)

    best_idx = int(np.argmin(J_pop))
    best_a = pop[best_idx].copy()
    best_J = float(J_pop[best_idx])

    print(f"[0.00s] Initial best J = {best_J:.6f} (n = {n})")

    # ------------------------------------------------------------------
    # 3. Main DE loop ---------------------------------------------------
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # spare a couple of seconds for finalisation

    generation = 0
    while time.monotonic() < deadline:
        generation += 1
        # shuffle order of target vectors each generation
        order = rng.permutation(popsize)
        for i in order:
            # ensure we have three distinct donors different from i
            donors = list(range(popsize))
            donors.remove(i)
            a_idx, b_idx, c_idx = rng.choice(donors, size=3, replace=False)

            # DE mutation (rand/1)
            F = rng.uniform(0.4, 0.9)       # scaling factor
            mutant = pop[a_idx] + F * (pop[b_idx] - pop[c_idx])
            mutant = np.maximum(mutant, 0.0)   # enforce non‑negativity
            mutant = proj_simplex(mutant)

            # binomial crossover
            CR = rng.uniform(0.2, 0.9)      # crossover probability
            mask = rng.random(n) < CR
            if not np.any(mask):
                mask[rng.integers(n)] = True
            trial = np.where(mask, mutant, pop[i])
            trial = proj_simplex(trial)

            # evaluate trial
            J_trial = compute_J(trial, n, fft_len, conv_len)

            # greedy selection (replace if not worse)
            if J_trial <= J_pop[i]:
                pop[i] = trial
                J_pop[i] = J_trial
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_a = trial.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

            # early exit if time is up
            if time.monotonic() >= deadline:
                break

        # ------------------------------------------------------------------
        # 3.1 Periodic jitter to shake the population (every ~10‑s)
        # ------------------------------------------------------------------
        if generation % 50 == 0:
            # create a multiplicative log‑normal perturbation of the best
            jitter = np.exp(rng.normal(0.0, 0.06, size=n))
            cand = proj_simplex(best_a * jitter)
            J_cand = compute_J(cand, n, fft_len, conv_len)
            worst_idx = int(np.argmax(J_pop))
            if J_cand < J_pop[worst_idx]:
                pop[worst_idx] = cand
                J_pop[worst_idx] = J_cand
                if J_cand < best_J:
                    best_J = J_cand
                    best_a = cand.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] Jitter improvement → J = {best_J:.6f}")

        # ------------------------------------------------------------------
        # 3.2 Optional early termination if no progress for a while
        # ------------------------------------------------------------------
        # (not implemented – the time budget dominates)

    # ------------------------------------------------------------------
    # 4. Final hill‑climbing refinement -----------------------------------
    # ------------------------------------------------------------------
    # Small random redistributions of mass that are accepted only if they improve.
    refinement_iters = 2000
    for _ in range(refinement_iters):
        if time.monotonic() >= deadline:
            break
        i = rng.integers(n)
        if best_a[i] <= 1e-14:
            continue
        j = rng.integers(n)
        if i == j:
            continue
        # move a tiny amount of mass from i to j
        delta = min(best_a[i], rng.random() * 0.001)   # ≤0.1 % of the total mass
        trial = best_a.copy()
        trial[i] -= delta
        trial[j] += delta
        trial = proj_simplex(trial)    # re‑normalise (should stay on the simplex)
        J_trial = compute_J(trial, n, fft_len, conv_len)
        if J_trial < best_J:
            best_J = J_trial
            best_a = trial

    # ------------------------------------------------------------------
    # 5. Sanitise and return --------------------------------------------
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    print(f"Finished – best J = {best_J:.6f} (length {n})")
    return [float(v) for v in best_a]
