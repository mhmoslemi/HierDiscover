"""
Exponentiated‑gradient (Adam) optimisation on the soft‑max
parameterisation of a step function.

Goal: minimise  J(a) = 2·n·max(a*a) / (∑a)² .
Because the objective is invariant to scaling we enforce ∑a = 1.
We write a = softmax(z) so that positivity and the simplex constraint
are satisfied automatically.

For a given a we compute the full linear convolution b = a*a with a real‑FFT.
To approximate the non‑smooth max we use a soft‑max weight
   w_i ∝ exp(α·b_i)      (α is increased during the run).
The gradient of ϕ(a)=∑ w_i·b_i  w.r.t. a is 2·corr(w,a) and
the chain‑rule gives the gradient w.r.t. the unconstrained z.
Adam updates on z provide fast convergence.

The routine runs until the supplied time budget expires (≈1000 s) and
returns the best sequence found.  Progress is printed intermittently.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using a real FFT.
    Returns the (2·n‑1)‑length convolution vector.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute ∇_a ( Σ_k weight_k·(a*a)_k ) = 2·corr(weight, a)
    efficiently via FFT.
    Parameters
    ----------
    a : (n,) array, non‑negative, Σa = 1
    weight : (2·n‑1,) array, typically a soft‑max over the convolution
    fft_len_corr : power‑of‑two length for the correlation FFT
    n : length of a
    Returns
    -------
    grad : (n,) array, the gradient w.r.t. a
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    The method uses an Adam optimiser on the soft‑max parameters of the
    sequence and a smooth approximation of the max‑convolution.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Choose problem size (number of steps)
    # --------------------------------------------------------------
    # If a previous construction exists, reuse its length, otherwise 4096.
    global height_sequence_1   # may be absent
    if 'height_sequence_1' in globals():
        try:
            n = len(height_sequence_1)
            # keep n in a comfortable range
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlations

    # --------------------------------------------------------------
    # 2. Initialise optimisation variables (soft‑max parameters)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            a0 = v / s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = rng.dirichlet(np.ones(n))

    # log‑space parameters (soft‑max)
    eps = 1e-20
    z = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(z)
    v_mom = np.zeros_like(z)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8

    # --------------------------------------------------------------
    # 3. Optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    # keep a small safety margin for the interpreter's shutdown
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    best_J = np.inf
    best_a = None
    best_iter = 0
    last_improve = start_time

    # schedule parameters
    max_alpha = 80.0          # final soft‑max sharpness
    alpha_ramp_iters = 4000   # iterations over which α is increased
    lr0 = 0.25                # initial Adam learning‑rate

    iteration = 0
    while time.monotonic() < deadline:
        # ----- convert z → a (soft‑max) -----
        z_max = np.max(z)
        exp_z = np.exp(z - z_max)
        a = exp_z / exp_z.sum()

        # ----- convolution & cheap objective -----
        conv = _conv_full(a, fft_len, conv_len)
        J = 2.0 * n * np.max(conv)   # because Σa = 1
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            best_iter = iteration
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J ≈ {best_J:.6f}")

        # ----- soft‑max weighting of the convolution -----
        if iteration < alpha_ramp_iters:
            alpha = 1.0 + (max_alpha - 1.0) * (iteration / alpha_ramp_iters)
        else:
            alpha = max_alpha

        # stable soft‑max over α·conv
        w_scaled = conv * alpha
        w_max = np.max(w_scaled)
        w_exp = np.exp(w_scaled - w_max)
        w = w_exp / w_exp.sum()                     # w_i ≥ 0, Σ w = 1

        # ----- gradient w.r.t. a (exact) -----
        grad_a = _weighted_gradient(a, w, fft_len_corr, n)

        # ----- chain‑rule to obtain grad z -----
        dot = np.dot(grad_a, a)                     # Σ_i a_i·grad_a_i
        grad_z = a * (grad_a - dot)                 # ∇_z J

        # ----- Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_z
        v_mom = beta2 * v_mom + (1.0 - beta2) * (grad_z ** 2)

        # bias‑corrected estimates
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v_mom / (1.0 - beta2 ** iteration)

        # learning‑rate decay (very gentle)
        lr = lr0 / (1.0 + iteration * 1e-5)

        z -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # ----- occasional status report -----
        if iteration % 2000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] iter {iteration:6d}, best_J ≈ {best_J:.6f}")

        # ----- simple stagnation handling -----
        if time.monotonic() - last_improve > 45.0:
            # re‑initialise to a fresh Dirichlet draw to regain diversity
            a_restart = rng.dirichlet(np.ones(n) * 0.5)
            z = np.log(a_restart + eps)
            m.fill(0.0)
            v_mom.fill(0.0)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Stagnation – re‑initialised at iter {iteration}")

    # ------------------------------------------------------------------
    # 4. Return the best sequence discovered
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – use the final a
        best_a = a

    # final clipping (evaluate_sequence also clips, but we keep it clean)
    best_a = np.clip(best_a, 0.0, 1000.0)

    # renormalise to avoid any tiny drift
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    # optional sanity check with the provided evaluator
    try:
        val = evaluate_sequence(best_a.tolist())
        elapsed = time.monotonic() - start_time
        print(f"[{elapsed:6.2f}s] Final evaluate_sequence = {val:.6f}")
    except Exception:
        pass

    return [float(v) for v in best_a]
