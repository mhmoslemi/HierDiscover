"""
Gradient‑descent on the variance of the self‑convolution (plus a soft
max penalty) with simplex projection and random restarts.

The algorithm works on a probability vector a (∑a=1, a≥0).  For a fixed
length n we minimise

    L(a) = Σ_k ( (a*a)_k – 1/(2n−1) )²

which drives the full convolution a*a towards a flat profile.
A secondary term based on the actual maximum of the convolution is
added with an increasing weight to directly pull down the peak.
Updates are performed with Adam, after each step a is projected onto the
probability simplex.  When no improvement is seen for a while the
current iterate is jittered and the optimisation restarts; the best
vector seen so far is always kept.

The routine respects the 1000 s budget and finally returns the best
sequence discovered (clipped to the allowed range and renormalised so
that `evaluate_sequence` can be called directly).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself (length 2·len(a)−1) using a real FFT.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w
