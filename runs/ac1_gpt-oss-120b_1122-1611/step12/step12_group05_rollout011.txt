    M = conv.max()
    # Get indices of top K conv values
    if K >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        # get K largest indices
        # argpartition returns unsorted; get actual values
        top_idx = np.argpartition(conv, -K)[-K:]
    grad = np.zeros_like(a)
    for k in top_idx:

        end = min(k, n - 1)
        # slice part
        # reversed slice:
        rev = a[start:end+1][::-1]
        grad[start:end+1] += 2.0 * rev
    return grad

"""
Propose a step‑function (probability vector) that minimizes
    J = 2·n·max(conv(a,a))   (sum(a)=1, a_i≥0)

Algorithm
---------
* Work on the probability simplex (projected Euclidean updates).
* At each iteration compute the full auto‑convolution a*a via a single FFT.
* Approximate the sub‑gradient of the max‑convolution by the sum of the
  gradients of the *K* largest convolution entries (active‑set approach).
* Use Adam‑style adaptive steps and a simulated‑annealing acceptance rule.
* When no improvement is seen for ``jitter_interval`` seconds apply a
  multiplicative Gaussian perturbation centred on the best‑so‑far vector.
* Occasionally perform a tiny mass‑exchange move to explore local directions.
* The routine runs for the allotted time (≈ 1000 s) and returns the best
  vector found, clipped to the allowed range and re‑normalised.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    { x | sum(x)=1, x_i ≥ 0 }.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear (full) convolution a*a using an FFT of length ``fft_len``."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Compute J = 2·n·max(conv(a,a)); the vector a is assumed to sum to 1.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def subgradient_active(a: np.ndarray,
                       conv: np.ndarray,
                       n: int,
                       K: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) using the K largest convolution entries.
    Returns the gradient of the *objective* J = 2·n·max(conv).
    """
    # indices of the K largest entries
    if K >= conv.size:
        top_idx = np.arange(conv.size)
    else:
        top_idx = np.argpartition(conv, -K)[-K:]

    grad = np.zeros_like(a)
    for k in top_idx:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)                     # inclusive
        # reversed slice of a contributing to conv[k]
        rev = a[start:end + 1][::-1]
        grad[start:end + 1] += 2.0 * rev
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 4242,
                     budget_s: float = 1000.0,
                     **unused_kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects the 1000 s time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose a problem size.
    # ------------------------------------------------------------------
    # Prefer the length of a previously found solution if it exists.
    if "height_sequence_1" in globals():
        try:
            prev = globals()["height_sequence_1"]
            n0 = int(len(prev))
            # limit to a comfortable size for FFT‑based work
            n = max(256, min(4096, n0))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # length for a*a convolution

    # ------------------------------------------------------------------
    # 2. Initialise a feasible vector on the simplex.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        src = np.asarray(height_sequence_1, dtype=np.float64)
        src = np.clip(src, 0.0, None)
        if src.size > n:
            a0 = src[:n].copy()
        else:
            a0 = np.pad(src, (0, n - src.size), constant_values=0.0)
        s = a0.sum()
        if s == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= s
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    cur_a = a0.copy()
    cur_J = compute_J(cur_a, n, fft_len, conv_len)

    best_a = cur_a.copy()
    best_J = cur_J
    last_improve = time.monotonic()
    start_time = last_improve

    # ------------------------------------------------------------------
    # 3. Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    total_time = deadline - start_time

    # Adam state
    m = np.zeros_like(cur_a)
    v = np.zeros_like(cur_a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr0 = 0.45                         # relatively aggressive base step

    # Sub‑gradient active‑set size – start a bit larger, shrink later
    K_initial = max(5, int(0.02 * n))
    K_final = max(3, int(0.005 * n))

    iteration = 1
    jitter_interval = 18.0            # seconds without improvement before jitter
    sigma_noise = 0.09                # std‑dev for multiplicative jitter

    print(f"[0.00s] Initial J = {cur_J:.6f}  (n={n})")

    while time.time() < deadline:
        now = time.time()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # 3.1 Adaptive active‑set size (linear schedule)
        # --------------------------------------------------------------
        K = int(K_initial - (K_initial - K_final) * (elapsed / total_time))
        K = max(K_final, K)

        # --------------------------------------------------------------
        # 3.2 Compute convolution and sub‑gradient
        # --------------------------------------------------------------
        conv = _conv_full(cur_a, fft_len, conv_len)
        grad = subgradient_active(cur_a, conv, n, K)

        # --------------------------------------------------------------
        # 3.3 Adam update and projection onto simplex
        # --------------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)
        a_proposed = cur_a - step
        a_proposed = proj_simplex(a_proposed)

        # --------------------------------------------------------------
        # 3.4 Occasional small exchange move (helps escape tiny plateaus)
        # --------------------------------------------------------------
        if rng.random() < 0.04:                     # 4 % chance
            i, j = rng.choice(n, size=2, replace=False)
            delta = min(cur_a[i], rng.random() * 0.009)   # ≤ 0.9 % of mass
            cur_a[i] -= delta
            cur_a[j] += delta
            cur_a = proj_simplex(cur_a)

        # --------------------------------------------------------------
        # 3.5 Evaluate proposal
        # --------------------------------------------------------------
        new_J = compute_J(a_proposed, n, fft_len, conv_len)

        # --------------------------------------------------------------
        # 3.6 Simulated‑annealing acceptance
        # --------------------------------------------------------------
        # temperature drops exponentially from the current best value to a tiny floor
        T_start = best_J
        T_end = 1e-9
        T = T_start * (T_end / T_start) ** (elapsed / total_time)

        accept = False
        if new_J < cur_J:
            accept = True
        else:
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = a_proposed
            cur_J = new_J
            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_a = cur_a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # 3.7 Jitter if stagnation persists
        # --------------------------------------------------------------
        if now - last_improve > jitter_interval:
            # multiplicative log‑normal noise centred on the best vector
            noise = np.exp(rng.normal(0.0, sigma_noise, size=n))
            cur_a = proj_simplex(best_a * noise)
            cur_J = compute_J(cur_a, n, fft_len, conv_len)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter – current J = {cur_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 4. Final sanitisation (clip to allowed range, re‑normalise)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    print(f"Finished. Best J = {best_J:.6f} after {iteration-1} iterations.")
    return [float(v) for v in best_a]
