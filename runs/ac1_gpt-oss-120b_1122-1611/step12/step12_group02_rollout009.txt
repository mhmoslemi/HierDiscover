"""
Improved search for the minimum of
    J(a) = 2·n·max(conv(a,a)),
subject to a_i ≥ 0 and Σ a_i = 1.

Algorithm
---------
1. Initialise a population with the known best construction (if any) and a few
   Dirichlet draws.  Keep only the single best vector – we treat the problem
   as a continuous optimisation on the simplex.

2. Repeatedly run a projected (multiplicative) gradient descent on a *smooth*
   approximation of the max‑convolution.
   * The convolution is obtained with a real FFT (O(n log n)).
   * The smooth max uses a soft‑max weighting
         w_k ∝ exp( τ·(conv_k – max(conv)) ),
     where τ is increased slowly so that the optimisation focuses more and more
     on the true maximum.
   * Gradient of the smooth max is 2·corr(w, a) (computed again via FFT).
   * Update:
         a ← a ∘ exp( –α·grad )          (∘ = element‑wise product)
     then project a back onto the probability simplex.
   * The step size α decays as 1/√iteration; τ grows linearly.

3. Perform many *random restarts*: draw a fresh Dirichlet vector, run a short
   gradient run, and keep any improvement.  This prevents getting stuck in a
   local minimum.

4. Occasionally perturb the current best solution with a tiny Gaussian noise,
   re‑project and keep it if it improves the objective.

5. The whole routine respects the supplied time budget (≈1000 s) and finally
   returns the best sequence found as a plain Python list.

All helper functions (simplex projection, FFT‑based convolution, weighted
gradient) are top‑level to satisfy the execution environment.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Select problem size (number of steps)
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(256, min(n0, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096                     # default comfortable size

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                 # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)    # for weighted correlations

    # ------------------------------------------------------------------
    # 2. Objective helpers
    # ------------------------------------------------------------------
    def _objective_vec(vec: np.ndarray) -> float:
        """return J = 2·n·max(conv) for a given vector (already on simplex)."""
        conv = _conv_full(vec, fft_len, conv_len)
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # 3. Initialise best candidate
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        v0 = np.asarray(height_sequence_1, dtype=np.float64)
        v0 = np.clip(v0, 0.0, None)
        if v0.shape[0] < n:
            v0 = np.concatenate([v0, np.zeros(n - v0.shape[0])])
        elif v0.shape[0] > n:
            v0 = v0[:n]
        v0 = proj_simplex(v0)
    else:
        v0 = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = v0.copy()
    best_J = _objective_vec(best_vec)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # keep a 5 s safety margin
    print(f"[0.00s] initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 4. Gradient‑descent optimiser (single‑vector, many restarts)
    # ------------------------------------------------------------------
    def _gradient_descent(a_start: np.ndarray,
                          max_iters: int,
                          tau_start: float,
                          tau_end: float,
                          step_start: float) -> None:
        """Run a projected multiplicative gradient descent from a_start."""
        nonlocal best_vec, best_J

        a = a_start.copy()
        # linear schedules for smoothing parameter τ and step size α
        for it in range(max_iters):
            if time.monotonic() >= deadline:
                break

            # time‑varying hyper‑parameters
            tau = tau_start + (tau_end - tau_start) * (it / max_iters)
            alpha = step_start / np.sqrt(it + 1.0)

            # current convolution
            conv = _conv_full(a, fft_len, conv_len)
            max_conv = conv.max()

            # smooth weight for the “soft‑max” of convolution entries
            w = np.exp(tau * (conv - max_conv))
            w_sum = w.sum()
            if w_sum == 0.0:               # numerically should not happen
                w = np.where(conv >= max_conv - 1e-12, 1.0, 0.0)
                w_sum = w.sum()
            w /= w_sum

            # gradient of the smooth max
            grad = _weighted_gradient(a, w, fft_len_corr, n)

            # multiplicative (exponentiated) gradient step
            a_new = a * np.exp(-alpha * grad)

            # simplex projection (ensures non‑negativity and Σ=1)
            a_new = proj_simplex(a_new)

            # evaluate new objective
            J_new = _objective_vec(a_new)

            # accept only if we improve
            if J_new < best_J - 1e-12:
                best_J = J_new
                best_vec = a_new.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f} (grad run)")

            # simple acceptance rule for the inner iterate (keep the improvement)
            if J_new < _objective_vec(a):
                a = a_new

    # ------------------------------------------------------------------
    # 5. Main search loop (random restarts + occasional perturbations)
    # ------------------------------------------------------------------
    # parameters for the inner gradient runs – they are sampled randomly
    # each restart to increase exploration.
    while time.monotonic() < deadline:
        # ---- random restart ------------------------------------------------
        alpha_dir = rng.uniform(0.02, 2.0)               # Dirichlet concentration
        a0 = rng.dirichlet(np.ones(n) * alpha_dir)

        # random schedule for this run
        max_iters = rng.integers(300, 1200)            # a few hundred iterations
        tau_start = rng.uniform(0.3, 1.5)
        tau_end = rng.uniform(3.0, 8.0)                # end with a sharper max
        step_start = rng.uniform(0.4, 1.2)

        _gradient_descent(a0, max_iters, tau_start, tau_end, step_start)

        # ---- occasional Gaussian perturbation of the global best -----------
        if rng.random() < 0.15:
            pert = rng.normal(0.0, 0.02, size=n)
            a_pert = best_vec + pert
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = proj_simplex(a_pert)
            J_pert = _objective_vec(a_pert)
            if J_pert < best_J - 1e-12:
                best_J = J_pert
                best_vec = a_pert.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f} (perturb)")

        # ---- occasional “jitter” of the whole population to keep diversity -
        # (here we simply re‑initialise a short list of random candidates)
        if int(time.monotonic() - start_time) % 120 == 0:    # every ~2 min
            for _ in range(3):
                a_jit = rng.dirichlet(np.ones(n) * 0.1)
                J_jit = _objective_vec(a_jit)
                if J_jit < best_J - 1e-12:
                    best_J = J_jit
                    best_vec = a_jit.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f} (jitter)")

    # ------------------------------------------------------------------
    # 6. Final clean‑up: clip, renormalise and convert to Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # Optional sanity check (uncomment if you wish to see the exact value)
    # try:
    #     val = evaluate_sequence(best_vec.tolist())
    #     print(f"Final evaluate_sequence = {val:.6f}")
    # except Exception:
    #     pass

    return [float(v) for v in best_vec]
