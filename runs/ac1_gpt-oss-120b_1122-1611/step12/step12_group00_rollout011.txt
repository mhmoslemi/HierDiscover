def solve_lp_max_sum(f, M):

    conv_len = 2*n-1
    # Setup A_ub shape (conv_len, n)
    # For each k, define row: sum_i f_i * g_{k-i} <= M
    # This corresponds to a Toeplitz matrix where only entries where 0 <= k - i < n contribute.
    # Build rows using loops; building dense matrix may be okay for n <= 2048 (conv_len * n ~ 4 million).
    A = np.zeros((conv_len, n), dtype=np.float64)
    for k in range(conv_len):
        # i ranges max(0, k-(n-1)).. min(n-1, k)
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        # for i in range(i_low, i_high+1):
        #   j = k - i is index of g
        #   A[k, j] = f[i]
        # So we fill A[k, k - i] = f[i]
        # Equivalent to: for i in range(i_low,i_high+1): A[k, k-i] = f[i]
        # Using vectorized assignment:
        j_indices = np.arange(i_low, i_high+1)
        i_indices = j_indices  # i = ?
        # No need to allocate new arrays.
        # Actually j = k - i, so we can compute j.
        # Let's just loop.
        for i in range(i_low, i_high+1):
            j = k - i
            A[k, j] = f[i]
    b = np.full(conv_len, M)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    bounds = [(0.0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    g0 = res.x
    return g0

"""
Search for a non‑negative step function that minimizes the security‑check
objective `evaluate_sequence`.  The algorithm consists of three stages:

1. **Initial pool** – a handful of probability vectors (uniform, Dirichlet
   draws, a Gaussian bump and the previously discovered construction if it
   exists) are evaluated and the best one is kept.

2. **Exponentiated‑gradient descent** – many short, independent runs start
   from a random or jittered copy of the current best vector.  Each run
   performs a projected exponential‑gradient update that reduces the
   L∞‑norm of the self‑convolution.  The sub‑gradient is computed only for
   the active (maximal) convolution entries, which is O(n) per iteration.
   Runs are limited by a small time slice, allowing hundreds of restarts
   within the 1000 s budget.

3. **Fine‑grained jitter** – the neighbourhood of the best vector found so
   far is explored by multiplicative Gaussian noise.  Any improvement is
   kept.

The returned list is already normalised (sum = 1) and clipped to the limits
enforced inside `evaluate_sequence`.  Progress messages are printed so that
the improving objective can be observed during execution.
"""

import time
from typing import List

import numpy as np
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def compute_full_conv(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Linear convolution a*a using a real FFT (returns length `conv_len`)."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Objective J = 2·n·max(conv(a,a)) for a probability vector (sum=1)."""
    conv = compute_full_conv(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def compute_gradient(a: np.ndarray, active: np.ndarray, n: int) -> np.ndarray:
    """Sub‑gradient of the maximal convolution entry (active indices supplied)."""
    grad = np.zeros_like(a)
    for k in active:
        low = max(0, k - (n - 1))
        hi = min(n - 1, k)
        start = k - hi
        end = k - low + 1          # exclusive
        # a[start:end] has length hi‑low+1
        grad[low : hi + 1] += 2.0 * a[start:end][::-1]
    return grad


def exponential_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient step followed by normalisation onto the simplex."""
    factor = np.exp(-eta * grad)
    a_new = a * factor
    total = a_new.sum()
    if total == 0.0:
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / total


def exp_gradient_descent(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    conv_len: int,
    max_iters: int,
    time_limit: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Short exponential‑gradient descent run.
    Returns the best vector seen and its J‑value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_J = float("inf")
    start = time.monotonic()

    for it in range(max_iters):
        if time.monotonic() - start > time_limit:
            break

        conv = compute_full_conv(a, fft_len, conv_len)
        max_val = conv.max()
        J = 2.0 * n * max_val
        if J < best_J - 1e-12:
            best_J = J
            best_a = a.copy()
            print(f"[{time.monotonic() - start:6.2f}s] ExpGD iter {it:,} new best J = {best_J:.6f}")

        # active indices – use a tiny absolute tolerance to avoid numerical noise
        tol = max(1e-15, 1e-12 * max_val)
        active = np.where(conv >= max_val - tol)[0]

        grad = compute_gradient(a, active, n)
        # step‑size decays like 1/√t
        eta = 0.5 / np.sqrt(it + 1)
        a = exponential_update(a, grad, eta)

    return best_a, best_J


def lp_max_sum_under_conv(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Linear programme:
        maximise   sum(g)
        subject to  (f * g)_k ≤ M  for all k
                    g ≥ 0
    Returns the optimiser g or None on failure.
    """
    n = f.size
    conv_len = 2 * n - 1
    # Build A_ub where each row corresponds to one convolution position.
    A = np.zeros((conv_len, n), dtype=np.float64)
    for k in range(conv_len):
        i_low = max(0, k - (n - 1))
        i_high = min(n - 1, k)
        for i in range(i_low, i_high + 1):
            j = k - i
            A[k, j] = f[i]
    b = np.full(conv_len, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # minimise -sum(g)  → maximise sum(g)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def lp_refine(f: np.ndarray, n: int, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Apply one LP‑based refinement step:
    maximise the total mass of a vector g subject to f * g ≤ max(f * f),
    then mix a fraction of g into f.
    """
    conv_f = compute_full_conv(f, fft_len, conv_len)
    M = conv_f.max()
    g0 = lp_max_sum_under_conv(f, M + 1e-12)   # slight relaxation for numeric safety
    if g0 is None:
        return f
    sum_g0 = g0.sum()
    if sum_g0 <= 1.0 + 1e-9:
        return f
    g = g0 / sum_g0                           # normalise to a probability vector
    t = 0.2                                    # mixing weight
    f_new = (1.0 - t) * f + t * g
    f_new /= f_new.sum()
    return f_new


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Main entry point required by the evaluation harness.
    Returns a list of non‑negative heights (a probability vector) that tries
    to minimise `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n_prev = len(globals()["height_sequence_1"])
            n = max(1024, min(n_prev, 4096))
        except Exception:
            n = 2048
    else:
        n = 2048

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    print(f"[Setup] n={n}, conv_len={conv_len}, fft_len={fft_len}")

    # ------------------------------------------------------------------
    # 2. Initial pool of candidates
    # ------------------------------------------------------------------
    init_vectors = []

    # uniform
    init_vectors.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws with a few concentration parameters
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(2):
            init_vectors.append(rng.dirichlet(np.ones(n) * alpha))

    # a symmetric Gaussian bump
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigma = n / 6.0
    g = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
    g = np.clip(g, 0.0, None)
    if g.sum() == 0.0:
        g = np.full_like(g, 1.0 / n)
    else:
        g = g / g.sum()
    init_vectors.append(g)

    # previous best if present
    if "height_sequence_1" in globals():
        v_prev = np.asarray(height_sequence_1, dtype=np.float64)
        v_prev = np.clip(v_prev, 0.0, None)
        if v_prev.size > n:
            v_prev = v_prev[:n]
        elif v_prev.size < n:
            v_prev = np.concatenate([v_prev, np.zeros(n - v_prev.size, dtype=np.float64)])
        if v_prev.sum() == 0.0:
            v_prev = np.full_like(v_prev, 1.0 / n)
        else:
            v_prev = v_prev / v_prev.sum()
        init_vectors.append(v_prev)

    # ------------------------------------------------------------------
    # 3. Evaluate initial pool and keep the best
    # ------------------------------------------------------------------
    best_vec = None
    best_J = float("inf")
    for idx, vec in enumerate(init_vectors):
        J = compute_J(vec, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = vec.copy()
        print(f"[Init {idx}] J = {J:.6f}")

    # ------------------------------------------------------------------
    # 4. Exponential‑gradient descent with many restarts
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # reserve a few seconds for polishing
    max_iters_per_run = 3000
    run_id = 0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 0.2:
            break

        # Choose a starting point: fresh random or a jittered copy of the current best
        if run_id % 4 == 0:
            a0 = rng.dirichlet(np.ones(n) * 0.3)   # a spiky start
        else:
            jitter = np.exp(rng.normal(0.0, 0.02, size=n))
            a0 = best_vec * jitter
            a0 = a0 / a0.sum()

        # Allocate a slice of the remaining budget for this run (max 12 s)
        run_time = min(remaining * 0.3, 12.0)

        local_best, local_J = exp_gradient_descent(
            a0,
            n,
            fft_len,
            conv_len,
            max_iters_per_run,
            run_time,
            rng,
        )
        if local_J < best_J - 1e-12:
            best_J = local_J
            best_vec = local_best.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Global new best J = {best_J:.6f}")

        run_id += 1

    # ------------------------------------------------------------------
    # 5. Optional LP polishing (a few quick attempts)
    # ------------------------------------------------------------------
    lp_deadline = start_time + float(budget_s) - 2.0
    while time.monotonic() < lp_deadline:
        # one refinement step
        cand = lp_refine(best_vec, n, fft_len, conv_len)
        cand_J = compute_J(cand, n, fft_len, conv_len)
        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] LP refinement J = {best_J:.6f}")
        else:
            break

    # ------------------------------------------------------------------
    # 6. Fine‑grained random jitter around the current best
    # ------------------------------------------------------------------
    fine_deadline = start_time + float(budget_s) - 0.1
    while time.monotonic() < fine_deadline:
        jitter = np.exp(rng.normal(0.0, 0.01, size=n))
        cand = best_vec * jitter
        cand = cand / cand.sum()
        cand_J = compute_J(cand, n, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 7. Final normalisation / clipping (matches evaluate_sequence guards)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
