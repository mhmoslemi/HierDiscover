"""
Gradient‑based optimizer on the probability simplex.

The objective evaluated by ``evaluate_sequence`` for a non‑negative vector
``a`` (with sum(a)=1) is

    J(a) = 2·n·max(conv(a,a))

where ``conv`` is the full linear convolution.
The algorithm starts from the best known construction (global variable
``height_sequence_1`` if present) and performs many cheap
gradient‑descent steps that respect the simplex constraint.
The gradient of the objective is a sub‑gradient of the maximal
convolution entry and can be obtained by a second convolution,
so both convolutions are computed with real FFTs (O(n log n)).
A decreasing step size together with a simple projection
(clip‑to‑zero and renormalise) guarantees feasibility.
If progress stalls a small random perturbation is applied.
Finally a short fine‑grained random search around the best vector
attempts a last improvement.

The routine respects the 1000 s time budget and returns a list of
floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _proj_simplex_fast(v: np.ndarray) -> np.ndarray:
    """
    Fast (non‑optimal) projection onto the probability simplex:
    clip negatives to zero and renormalise.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full convolution a * a using a real FFT.
    Returns the first 2·len(a)‑1 entries.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_fft_pair(x: np.ndarray, y: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Full convolution of two *real* vectors using FFT.
    Returns the first len(x)+len(y)-1 entries.
    """
    X = np.fft.rfft(x, fft_len)
    Y = np.fft.rfft(y, fft_len)
    conv = np.fft.irfft(X * Y, n=fft_len)
    return conv[: x.size + y.size - 1]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The method combines projected
    gradient descent on the simplex with occasional stochastic
    perturbations and a final fine‑grained random search.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Initial vector (use the best known construction if available)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        # ensure strict positivity (helps the gradient step)
        a = np.maximum(a, 1e-12)
    else:
        # fall back to a modest length if nothing is provided
        a = np.full(2048, 1.0 / 2048, dtype=np.float64)

    a = _proj_simplex_fast(a)
    n = a.size

    # --------------------------------------------------------------
    # 2. Pre‑compute FFT lengths (constant during the run)
    # --------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_a = _next_pow2(conv_len)            # for a*a
    conv_len_mask = conv_len + n - 1            # (2n‑1)+n‑1 = 3n‑2
    fft_len_mask = _next_pow2(conv_len_mask)    # for mask * a_rev

    # --------------------------------------------------------------
    # 3. Initial evaluation
    # --------------------------------------------------------------
    conv_a = _conv_self_fft(a, fft_len_a)
    best_J = float(2.0 * n * conv_a.max())
    best_vec = a.copy()
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # a small safety margin
    last_improve = start
    it = 0

    print(f"[0.00s] Initial J = {best_J:.6f} (n = {n})")

    # --------------------------------------------------------------
    # 4. Main projected gradient descent loop
    # --------------------------------------------------------------
    base_eta = 0.04                     # initial step magnitude
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution of current iterate
        conv_a = _conv_self_fft(a, fft_len_a)
        max_val = conv_a.max()

        # ----- mask of maximal entries (with a tiny tolerance)
        tol = max(1e-15, max_val * 1e-12)
        mask = (np.abs(conv_a - max_val) <= tol).astype(np.float64)

        # ----- gradient of the max‑convolution (sub‑gradient)
        a_rev = a[::-1]
        conv_mask = _conv_fft_pair(mask, a_rev, fft_len_mask)
        grad = 2.0 * conv_mask[n - 1 : n - 1 + n]       # length‑n vector

        # ----- step size (decreases like 1/√iteration)
        eta = base_eta / np.sqrt(it)
        max_g = grad.max()
        if max_g > 0.0:
            # we scale the step such that the largest absolute change is η
            step = eta / max_g
            a_new = a - step * grad
        else:
            a_new = a.copy()

        # ----- keep feasibility (non‑negative, sum = 1)
        a_new = _proj_simplex_fast(a_new)

        # ----- occasional tiny stochastic jitter (helps escape plateaus)
        if it % 2000 == 0:
            noise = np.exp(rng.normal(0.0, 0.018, size=n))
            a_new = a_new * noise
            a_new = _proj_simplex_fast(a_new)

        # ----- evaluate the candidate
        conv_new = _conv_self_fft(a_new, fft_len_a)
        J_new = float(2.0 * n * conv_new.max())

        if J_new < best_J - 1e-12:
            best_J = J_new
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation handling – re‑inject a small random perturbation
        if time.monotonic() - last_improve > 30.0:
            # start from the current best and shake it a little
            a = best_vec.copy()
            a = a * np.exp(rng.normal(0.0, 0.045, size=n))
            a = _proj_simplex_fast(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] Stagnated – re‑randomised")
        else:
            a = a_new

    # --------------------------------------------------------------
    # 5. Final fine‑grained random search around the best vector
    # --------------------------------------------------------------
    fine_start = time.monotonic()
    fine_deadline = fine_start + min(30.0, (deadline - start) * 0.05)
    while time.monotonic() < fine_deadline:
        # very small multiplicative log‑normal perturbation
        noise = np.exp(rng.normal(0.0, 0.010, size=n))
        cand = best_vec * noise
        cand = _proj_simplex_fast(cand)
        conv_cand = _conv_self_fft(cand, fft_len_a)
        J_cand = float(2.0 * n * conv_cand.max())
        if J_cand < best_J - 1e-12:
            best_J = J_cand
            best_vec = cand
            elapsed = time.monotonic() - start
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 6. Return a legal sequence (values already in [0,1000] and sum=1)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    best_vec = _proj_simplex_fast(best_vec)      # final safety normalisation
    return [float(v) for v in best_vec]
