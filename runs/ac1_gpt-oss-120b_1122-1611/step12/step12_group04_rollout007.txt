"""
Hybrid optimisation for ``evaluate_sequence`` (step‑function design).

The routine works in three phases within the 1000 s budget:

1. **Adam on a soft‑max smoothed max‑convolution** – the classic
   gradient‐based search used before, but with an extended β schedule
   (up to 5 000) and a longer optimisation window.

2. **Multiplicative mirror‑descent** – after the Adam phase we focus
   on the currently worst shift (the index where the convolution attains
   its maximum) and apply a KL‑projection update
   `a_i ← a_i·exp(−s·a_{p−i})`.  The step‑size s decays from 0.3 to 0.01.
   This directly reduces the max‑convolution while staying on the
   probability simplex.

3. **Simulated‑annealing pairwise swaps** – a cheap local search that
   moves a small amount of mass between two positions.  It refines the
   solution found by the previous phases.

All heavy lifting (convolutions and weighted‑gradients) uses real FFTs,
so even a length of 8192 fits comfortably.  The routine respects the
time budget, prints a few progress messages and finally returns a plain
Python list of non‑negative heights that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a 1‑D array onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries were zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a * a using a real FFT.

    Parameters
    ----------
    a : np.ndarray, shape (n,)
        Input vector.
    fft_len : int
        Length of the FFT (a power of two ≥ conv_len).
    conv_len : int
        Length of the full convolution (2·n − 1).

    Returns
    -------
    np.ndarray, shape (conv_len,)
        Linear convolution a * a.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray,
                       weight: np.ndarray,
                       fft_len_corr: int,
                       n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    ``weight`` length = 2·n − 1, ``a`` length = n.

    Returns a gradient vector of length n.
    """
    # reverse a for correlation
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective_vec(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a.
    ``a`` is assumed to be non‑negative and sum(a) = 1.
    """
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())
