"""
Subgradient + Simplex Projection optimizer.

We treat the evaluation function (with sum(a)=1) as

    J(a) = 2 * n * max_k (a * a)[k] .

For a fixed index p where the maximum is attained,
the sub‑gradient of max_k (a*a)[k] w.r.t. a is
    g_j = 2 * a_{p-j}        (if 0 ≤ p‑j < n, else 0)

When several indices tie for the maximum we average their
gradients.  A projected sub‑gradient descent (with a
diminishing step size η_t = η0 / sqrt(t+1)) is run on the
probability simplex.  The method is started from several
hand‑crafted seeds (uniform, geometric, Gaussian, Dirichlet and
the previously discovered best construction).  The best
solution seen during the whole run is returned.

The algorithm is deliberately simple but cheap – each iteration
needs one FFT‑based self‑convolution (O(n log n)) and a linear
gradient update, allowing many thousands of iterations within the
1000 s budget.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ------------------------------------------------------------

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def conv_max_and_argmax(a: np.ndarray, fft_len: int, conv_len: int):
    """Return (max_value, list_of_argmax_indices) of a*a (linear convolution)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    max_val = float(conv.max())
    # use tolerance to capture ties caused by floating‑point noise
    eps = 1e-15
    idx = np.where(conv >= max_val - eps)[0]
    return max_val, idx.tolist()


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Full objective 2·n·max(conv(a,a)) (sum(a) is assumed to be 1)."""
    max_val, _ = conv_max_and_argmax(a, fft_len, conv_len)
    return 2.0 * n * max_val


def gradient_of_max(a: np.ndarray, active_idx, n: int) -> np.ndarray:
    """
    Sub‑gradient of max_k conv(a,a)[k] at the active indices.
    Returns (2/|P|) * Σ_p shift_rev_p(a).
    """
    g = np.zeros_like(a)
    rev_a = a[::-1]                     # pre‑compute reversed vector
    for p in active_idx:
        if p < n:
            # add a[p], a[p-1], … , a[0] to positions 0 … p
            g[:p + 1] += a[p::-1]
        else:
            # p ≥ n: add a[n‑1], a[n‑2], … to positions p‑(n‑1) … n‑1
            start = p - (n - 1)
            length = n - start
            g[start:] += rev_a[:length]
    if len(active_idx) > 0:
        g *= 2.0 / len(active_idx)
    return g

# ------------------------------------------------------------

            g = gradient_of_max(cur, active, n)

            # Diminishing step size
            eta = eta0 / np.sqrt(it + 1.0)

            # Gradient step + projection onto simplex
            nxt = cur - eta * g
            nxt = proj_simplex(nxt)

            nxt_J = compute_J(nxt, n, fft_len, conv_len)

            # Accept only improving moves (simple greedy strategy)
            if nxt_J < cur_J - 1e-14:
                cur, cur_J = nxt, nxt_J
                # Record global improvement
                if cur_J < best_J - 1e-14:
                    best_J = cur_J
                    best_vec = cur.copy()
                    elapsed = time.monotonic() - start_time
                    print(f"[{elapsed:6.2f}s] New global best J = {best_J:.6f}")

            # Occasionally inject a tiny multiplicative jitter to escape plateaus
            if it % 500 == 0 and it > 0:
                sigma = 0.05 * (1.0 - it / max_iters_per_restart)   # shrink over time
                noise = np.exp(rng.normal(0.0, sigma, size=n))
                jittered = proj_simplex(cur * noise)
                jitter_J = compute_J(jittered, n, fft_len, conv_len)
                if jitter_J < cur_J:
                    cur, cur_J = jittered, jitter_J
            it += 1

        restart_counter += 1
        # safety break if we are too close to the deadline
        if time.monotonic() > deadline - 0.5:
            break

    # --------------------------------------------------------
    # Final cleanup (clip, re‑normalise)
    # --------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_vec = best_vec / total

    return [float(v) for v in best_vec]
