"""
Propose a step‑function (probability vector) that minimises
`evaluate_sequence`.  The algorithm consists of three stages:

1. **Coarse parametric sweep** –  several one‑parameter families
   (geometric, power‑law, symmetric Beta, arcsine‑like) are generated,
   normalised to the simplex and evaluated with a fast FFT‑based
   convolution.  The best candidate is kept.

2. **Sub‑gradient descent on the true max‑convolution** –  at each
   iteration the index of the current maximal convolution entry is
   identified; its (exact) sub‑gradient 2·a shifted by that index is
   computed, a step is taken on the simplex (via Euclidean projection)
   and the new objective is accepted if it improves, otherwise with a
   temperature‑controlled Metropolis rule.  This directly targets the
   non‑smooth objective and typically flattens the convolution further.

3. **Local random mass‑swap polishing** –  simulated‑annealing style
   pairwise transfers of a small amount of probability mass are tried
   to escape any remaining shallow basin and to squeeze a final
   improvement.

All heavy work uses real FFTs (O(n log n)).  The routine automatically
chooses a problem size `n` (up to 8192) and respects the 1000 s time
budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – same as the reference implementation
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Return J = 2·n·max(conv(a,a)) for a simplex‑normalised vector a."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found within
    the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Problem size
    # ------------------------------------------------------------------
    # Use the length of a previously found sequence if it exists,
    # otherwise fall back to a power‑of‑two not larger than 8192.
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)           # length for a*a
    # ------------------------------------------------------------------
    # 2. Coarse parametric sweep
    # ------------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    def _try_candidate(v_raw: np.ndarray, descr: str = "") -> None:
        """Normalise, evaluate and keep the best candidate."""
        nonlocal best_vec, best_J
        v = np.maximum(v_raw, 0.0)
        s = v.sum()
        if s == 0.0:
            return
        v /= s
        J = _objective(v, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = v.copy()
            print(f"[init] New best ({descr}) J = {best_J:.6f}")

    # uniform
    _try_candidate(np.ones(n, dtype=float), "uniform")

    # geometric decays
    for r in np.linspace(0.90, 0.999, 12):
        _try_candidate(r ** np.arange(n, dtype=float), f"geom r={r:.3f}")

    # symmetric power‑law: (i+½)*(n‑i‑½) raised to p
    for p in np.linspace(-1.5, 1.5, 31):
        idx = np.arange(n, dtype=float) + 0.5
        w = (idx * (n - idx)) ** p
        _try_candidate(w, f"pow p={p:.2f}")

    # symmetric Beta family (x^(α‑1)*(1‑x)^(α‑1))
    for alpha in np.linspace(0.2, 5.0, 31):
        x = (np.arange(n, dtype=float) + 0.5) / n
        w = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        _try_candidate(w, f"beta α={alpha:.2f}")

    # arcsine‑like (inverse‑sqrt) – two exponents for flexibility
    for p in [-0.6, -0.5, -0.4, -0.3]:
        idx = np.arange(n, dtype=float) + 0.5
        w = (idx ** p) * ((n - idx) ** p)
        _try_candidate(w, f"inv‑sqrt p={p:.2f}")

    # previous construction, if available

            if prev.size != n:
                # crop or pad to the target length
                if prev.size > n:
                    prev = prev[:n]
                else:

            _try_candidate(prev, "prev")
        except Exception:
            pass

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------
    cur = best_vec.copy()
    cur_J = best_J
    start_time = time.monotonic()

    main_deadline = start_time + max(0.0, budget_s - 60.0)

        conv = _conv_full(cur, fft_len, conv_len)
        max_idx = int(conv.argmax())
        max_val = float(conv[max_idx])
        cur_J = 2.0 * n * max_val

        # diminishing step size (1/√t schedule)
        step_sz = eta0 / np.sqrt(iteration)

        # exact sub‑gradient of the max‑convolution entry:
        # grad[i] = 2 * a[max_idx - i]  (if the index is inside the array)
        idx = np.arange(n, dtype=int)
        partner = max_idx - idx
        mask = (partner >= 0) & (partner < n)
        grad = np.zeros_like(cur)
        grad[mask] = 2.0 * cur[partner[mask]]

        # take a (projected) sub‑gradient step
        cand = cur - step_sz * grad
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        # accept if we improve, otherwise Metropolis with a linear‑cooling temperature
        if cand_J < cur_J:
            cur = cand
            cur_J = cand_J

                best_vec = cand.copy()

                print(f"[{elapsed:6.2f}s] Sub‑grad improvement: J = {best_J:.6f}")
        else:

            t_frac = (time.monotonic() - start_time) / (main_deadline - start_time + 1e-12)
            T = (0.01 * cur_J) * (1.0 - t_frac) + 1e-9
            if rng.random() < np.exp(-(cand_J - cur_J) / T):
                cur = cand
                cur_J = cand_J

    # 4. Local random‑swap polishing (simulated annealing)
    # ------------------------------------------------------------------
    polish_deadline = start_time + float(budget_s) - 0.5  # keep a tiny margin
    cur = best_vec.copy()
    cur_J = best_J

            continue

        # move a small random fraction of the donor mass to the receiver
        eps = rng.random() * cur[i] * 0.35  # at most 35 % of the donor
        if eps <= 0.0:
            continue

        cand = cur.copy()
        cand[i] -= eps
        cand[j] += eps

        # projection is cheap and guarantees feasibility (non‑negative, sum=1)
        cand = proj_simplex(cand)

        cand_J = _objective(cand, n, fft_len, conv_len)

        if cand_J < cur_J:
            cur = cand
            cur_J = cand_J
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Polishing improvement: J = {best_J:.6f}")
        else:
            # very gentle uphill moves
            t_frac = (time.monotonic() - start_time) / (budget_s + 1e-12)
            T = (0.001 * best_J) * (1.0 - t_frac) + 1e-12
            if rng.random() < np.exp(-(cand_J - cur_J) / T):
                cur = cand
                cur_J = cand_J

    # ------------------------------------------------------------------
    # 5. Final normalisation & conversion to plain Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
