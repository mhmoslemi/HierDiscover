"""
Simulated‑Annealing + variance‑gradient polishing.

* The search works on the probability simplex (∑aᵢ = 1, aᵢ ≥ 0).
* A warm start is taken from ``height_sequence_1`` when it exists,
  otherwise from the uniform distribution.
* The main loop is a classic simulated‑annealing walk:
    – pick two indices,
    – move a small amount of mass from one to the other,
    – accept the move with Metropolis probability.
* Every 2000 accepted moves a “flatten‑convolution” step is performed:
    – the full convolution a*a is computed,
    – its deviation from the average (the only value a * a can have in the
      optimum) is used as a weight,
    – a multiplicative (exponentiated‑gradient) update is applied with a
      short line‑search over a few step‑sizes,
    – the result is projected back onto the simplex.
* Temperature and the maximal transferred mass decay linearly with the
  elapsed time, giving strong exploration early on and fine exploitation
  later.
* The routine respects a 1000 s wall‑clock budget and finally returns
  the best sequence discovered (as a plain Python list of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _conv_max(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of a*a (computed via FFT)."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size (number of steps)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        # Use the supplied construction length as the problem size.
        n = len(height_sequence_1)
        base = np.asarray(height_sequence_1, dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.shape[0] < n:
            base = np.concatenate([base, np.zeros(n - base.shape[0])])
        elif base.shape[0] > n:
            base = base[:n]
        s = base.sum()
        if s > 0.0:
            base = base / s
        else:
            base = np.full(n, 1.0 / n)
    else:

        n = 4096
        base = np.full(n, 1.0 / n, dtype=np.float64)

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                 # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)    # for weighted correlation

    # --------------------------------------------------------------
    # 2. Objective helpers (the constant factor 2·n is kept for clarity)
    # --------------------------------------------------------------
    def objective_max(a_vec: np.ndarray) -> float:
        """Maximum entry of a*a."""
        return _conv_max(a_vec, fft_len, conv_len)

    def objective_J(a_vec: np.ndarray) -> float:
        """Full evaluation expression (2·n·max)/(sum²).  With sum=1 this is 2·n·max."""
        return 2.0 * n * objective_max(a_vec)

    # --------------------------------------------------------------
    # 3. Initialise the current state
    # --------------------------------------------------------------
    # Slight random jitter on the warm‑start to avoid a perfectly flat start.
    a = base + 1e-4 * rng.normal(size=n)
    a = np.maximum(a, 0.0)
    a = proj_simplex(a)

    cur_J = objective_J(a)
    best_J = cur_J
    best_a = a.copy()

    print(f"[0.00s] Initial J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing parameters
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # Temperature schedule – start relatively hot and cool to almost zero.
    init_temp = cur_J * 0.5          # 50 % of the initial value
    final_temp = 1e-9

    # Mass‑transfer schedule (maximum amount moved in a single SA step).
    init_step = 0.05                 # 5 % of the total mass
    final_step = 1e-5

    # Frequency of the “flatten‑convolution” polishing step.
    polish_every = 2000             # after this many *accepted* moves

    accepted_moves = 0
    last_improve = start_time

    # --------------------------------------------------------------
    # 5. Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----- annealing schedule (linear in wall‑clock time) -----
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total if total > 0 else 1.0

        temperature = init_temp * (1.0 - frac) + final_temp * frac
        max_step = init_step * (1.0 - frac) + final_step * frac

        # ----- propose a tiny mass‑transfer move (i ↔ j) -----
        i, j = rng.integers(0, n, size=2)
        while i == j:
            j = rng.integers(0, n)

        # direction: move from the larger entry to the smaller one
        if a[i] >= a[j]:
            delta = rng.random() * min(max_step, a[i])
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
        else:
            delta = rng.random() * min(max_step, a[j])
            a_new = a.copy()
            a_new[i] += delta
            a_new[j] -= delta

        # ----- evaluate new candidate -----
        new_J = objective_J(a_new)

        # ----- Metropolis acceptance -----
        if (new_J < cur_J) or (rng.random() < np.exp(-(new_J - cur_J) / max(temperature, 1e-12))):
            a = a_new
            cur_J = new_J
            accepted_moves += 1

            # ----- keep the global best -----
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----------------------------------------------------------
        # Polishing step: try to flatten the convolution (variance‑grad)
        # ----------------------------------------------------------
        if accepted_moves and (accepted_moves % polish_every == 0):
            # Full convolution and its average (the target value if perfectly flat)
            conv = _conv_full(a, fft_len, conv_len)
            avg = conv.mean()                     # = 1/(2n‑1) because sum(conv)=1
            diff = conv - avg

            # Weighted gradient (2·corr(diff, a))
            grad = _weighted_gradient(a, diff, fft_len_corr, n)

            # Try a few multiplicative step‑sizes and keep the best improvement.
            improved = False
            for lr in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                # exponentiated‑gradient update
                exponent = -lr * grad
                # protect against overflow
                exponent = np.clip(exponent, -20.0, 20.0)
                factor = np.exp(exponent)
                a_cand = a * factor
                a_cand = proj_simplex(a_cand)

                cand_J = objective_J(a_cand)
                if cand_J < cur_J - 1e-12:
                    a = a_cand
                    cur_J = cand_J
                    improved = True
                    if cand_J < best_J - 1e-12:
                        best_J = cand_J
                        best_a = a.copy()
                        last_improve = time.monotonic()
                        print(f"[{time.monotonic() - start_time:6.2f}s] Polished J = {best_J:.6f}")
                    break  # accept the first improvement

            # If none of the line‑search steps helped, fall back to a tiny random jitter.
            if not improved:
                jitter = 1e-4 * rng.standard_normal(n)
                a = proj_simplex(np.maximum(a + jitter, 0.0))
                cur_J = objective_J(a)

        # ----------------------------------------------------------
        # Stagnation handling – if we have not improved for a while,
        # re‑inject diversity by redrawing a fresh Dirichlet candidate.
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # keep the current best but randomise the current position
            a = rng.dirichlet(np.ones(n) * 0.1)
            cur_J = objective_J(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Stagnation – random restart")

    # ------------------------------------------------------------------
    # 6. Final clean‑up: clip, renormalise and return as plain list
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    total = best_a.sum()
    if total == 0.0:
        best_a[:] = 1.0 / n
    else:
        best_a /= total

    # (Optional sanity check – uncomment if you wish to see the final value.)
    # try:
    #     val = evaluate_sequence(best_a.tolist())
    #     print(f"Final evaluate_sequence = {val:.6f}")
    # except Exception:
    #     pass

    return [float(v) for v in best_a]
