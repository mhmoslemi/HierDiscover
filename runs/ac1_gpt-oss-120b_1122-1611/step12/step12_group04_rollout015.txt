def propose_candidate(seed=12345, budget_s=1000., **kwargs):
    # Setup RNG
    # Determine n

    # Candidate generation:
    best_vec = None; best_J = np.inf

    # uniform
    cand = np.full(n, 1.0/n)
    update_best(cand)
    # scan exponential shapes:
    for eps in eps_vals:
        r = 1.0 - eps
        cand = np.exp(np.log(r) * np.arange(n))
        cand /= cand.sum()
        update_best(cand, cand_J)
    # scan power law shapes:
    for p in p_vals:
        cand = (np.arange(1, n+1) ** (-p))
        cand /= cand.sum()
        update_best(cand)

    for _ in range(20):
        cand = rng.dirichlet(np.ones(n))*1.0  # Actually Dirichlet draws sum to 1 already.
        cand = np.clip(cand, 0.0, None)
        update_best(cand)

    # subgradient descent
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_global = best_vec.copy()
    best_global_J = best_J

    # parameters
    step = 0.5
    decay = 0.9995
    last_improve_time = start_time
    while time.monotonic() < deadline_subgrad:
        # compute convolution
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = 2.0 * n * max_conv
        # compute weight indicator
        # find all indices where conv close to max (within tolerance e.g., 1e-12)
        max_indices = np.where(conv >= max_conv - 1e-12)[0]
        w = np.zeros_like(conv)
        # some may many indices, but set weight 1/len for each
        w[max_indices] = 1.0 / max_indices.size
        # compute gradient
        grad = _weighted_gradient(cur_vec, w, fft_len_corr, n)
        # update step size
        step *= decay
        # propose
        cand = cur_vec - step * grad
        cand = proj_simplex(cand)
        # compute J for cand
        cand_conv = _conv_full(cand, fft_len, conv_len)
        cand_J = 2.0 * n * cand_conv.max()

        if cand_J < cur_J:
            cur_vec = cand

            if cand_J < best_global_J - 1e-12:
                best_global_J = cand_J

"""
Propose a step‑function (probability vector) that minimises
`evaluate_sequence`.  The optimiser works in three stages:

1. **Broad scan** – generate a large set of simple parametric families
   (uniform, geometric decay, power‑law, linear ramps, Dirichlet draws)
   and keep the best by the fast FFT‑based objective
   J = 2·n·max(conv(a,a)) (the denominator sum(a)^2 is 1 after the simplex
   projection).

2. **Projected sub‑gradient descent** – starting from the best candidate,
   repeatedly move opposite to the sub‑gradient of the max‑convolution
   (the gradient is the correlation of the indicator of the maximal
   convolution entries with the current vector).  After each step the
   iterate is projected onto the probability simplex.  A simple
   diminishing step‑size plus occasional stochastic jitter avoids
   stagnation.

3. **Local polishing** – cheap pairwise mass‑swaps with a simulated‑annealing
   schedule to squeeze out the last bits of improvement.

All heavy work (convolution, correlation) uses real FFTs, so even a
length of 4096 fits easily into the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear auto‑convolution a*a using a real FFT.
    Returns a vector of length conv_len = 2·n‑1.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` length = 2·n‑1, `a` length = n.
    Returns a gradient vector of length n.
    """
    # reverse a for correlation
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Fast evaluation of J = 2·n·max(conv(a,a)) for a simplex‑normalised vector."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found within
    the allocated time budget.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # length for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # length for weight * a_rev

    # --------------------------------------------------------------
    # 2. Initial candidate generation (fast scan)
    # --------------------------------------------------------------
    best_vec = None
    best_J = np.inf

    def _check_and_update(v: np.ndarray, label: str = ""):
        nonlocal best_vec, best_J
        J = _objective(v, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_vec = v.copy()
            print(f"[init] New best ({label}) J = {best_J:.6f}")

    # uniform
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _check_and_update(uniform, "uniform")

    # exponential families a_i ∝ r^i   (r close to 1 is promising)
    eps_vals = np.logspace(np.log10(1e-5), np.log10(1e-2), num=200)
    idx = np.arange(n, dtype=np.float64)
    for eps in eps_vals:
        r = 1.0 - eps
        if r <= 0.0:
            continue
        v = np.exp(idx * np.log(r))
        v_sum = v.sum()
        if v_sum == 0.0:
            continue
        v /= v_sum
        _check_and_update(v, f"exp_eps={eps:.1e}")

    # power‑law families a_i ∝ (i+1)^{-p}
    p_vals = np.concatenate([np.linspace(0.01, 1.0, 50),
                             np.linspace(1.0, 5.0, 50)])
    for p in p_vals:
        v = (idx + 1.0) ** (-p)
        v_sum = v.sum()
        if v_sum == 0.0:
            continue
        v /= v_sum
        _check_and_update(v, f"power_p={p:.2f}")

    # linear ramps (both directions)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    _check_and_update(v, "linear_desc")
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    _check_and_update(v, "linear_asc")

    # Gaussian bumps – various width
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 12.0, n / 9.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        _check_and_update(v, f"gauss_sigma={sigma:.1f}")

    # a few Dirichlet draws (different concentrations)
    for alpha in (0.02, 0.05, 0.1, 0.3, 0.6, 1.0):
        for _ in range(2):
            v = rng.dirichlet(np.ones(n) * alpha)
            _check_and_update(v, f"dirichlet_a={alpha}")

    # previous construction, if present
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        _check_and_update(v, "prev")

    # --------------------------------------------------------------
    # 3. Projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 100.0)   # reserve ~100 s for polishing
    cur_vec = best_vec.copy()
    cur_J = best_J
    best_global = cur_vec.copy()
    best_global_J = cur_J
    last_improve = start_time

    step_size = 0.5                 # initial step length
    decay = 0.9995                  # multiplicative decay per iteration
    iter_cnt = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        iter_cnt += 1

        # full convolution and maximal value
        conv = _conv_full(cur_vec, fft_len, conv_len)
        max_conv = conv.max()
        cur_J = float(2.0 * n * max_conv)

        # keep the best ever seen
        if cur_J < best_global_J - 1e-12:
            best_global_J = cur_J
            best_global = cur_vec.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best (subgrad) J = {best_global_J:.6f}")

        # stagnation handling – jitter after 30 s without improvement
        if now - last_improve > 30.0:
            noise = np.exp(rng.normal(0.0, 0.06, size=n))
            cur_vec = best_global * noise
            cur_vec = proj_simplex(cur_vec)
            cur_J = float(2.0 * n * _conv_full(cur_vec, fft_len, conv_len).max())
            last_improve = now
            print(f"[{elapsed:6.2f}s] Jitter restart – cur J = {cur_J:.6f}")

        # weight vector: indicator of the (possibly multiple) maximal entries
        max_mask = conv == max_conv
        num_max = np.count_nonzero(max_mask)
        weight = np.zeros_like(conv, dtype=np.float64)
        weight[max_mask] = 1.0 / num_max

        # sub‑gradient (2·corr(weight, a))
        grad = _weighted_gradient(cur_vec, weight, fft_len_corr, n)

        # simple diminishing step
        step = step_size * (decay ** iter_cnt)

        proposal = cur_vec - step * grad
        proposal = proj_simplex(proposal)

        # evaluate proposal
        prop_J = float(2.0 * n * _conv_full(proposal, fft_len, conv_len).max())
        if prop_J < cur_J:
            cur_vec = proposal
            cur_J = prop_J
        # else keep current vector (no uphill moves here – jitter deals with traps)

    # --------------------------------------------------------------
    # 4. Local polishing – pairwise mass swaps + simulated annealing
    # --------------------------------------------------------------
    polish_deadline = start_time + float(budget_s) - 2.0   # keep a couple seconds for finalisation
    cur_vec = best_global.copy()
    cur_J = best_global_J
    cur_conv = _conv_full(cur_vec, fft_len, conv_len)

    # simple temperature schedule
    T0 = cur_J * 0.01
    T_end = 1e-9

    while time.monotonic() < polish_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = (now - start_time) / (budget_s - 2.0)
        T = T0 * (1.0 - frac) + T_end * frac

        i, j = rng.integers(0, n, size=2, endpoint=False)
        if i == j:
            continue
        # move a small amount of mass from i to j
        eps = rng.random() * cur_vec[i] * 0.5
        if eps <= 0.0:
            continue
        cand = cur_vec.copy()
        cand[i] -= eps
        cand[j] += eps
        cand = np.maximum(cand, 0.0)
        cand = proj_simplex(cand)

        cand_J = float(2.0 * n * _conv_full(cand, fft_len, conv_len).max())
        if cand_J < cur_J:
            cur_vec = cand
            cur_J = cand_J
            if cand_J < best_global_J - 1e-12:
                best_global_J = cand_J
                best_global = cand.copy()
                print(f"[{elapsed:6.2f}s] Polishing improvement: J = {best_global_J:.6f}")
        else:
            # uphill acceptance with annealing
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                cur_vec = cand
                cur_J = cand_J

    # --------------------------------------------------------------
    # 5. Final normalisation & conversion
    # --------------------------------------------------------------
    best_global = np.clip(best_global, 0.0, 1000.0)
    total = best_global.sum()
    if total == 0.0:
        best_global[:] = 1.0 / n
    else:
        best_global /= total

    return [float(v) for v in best_global]
