"""
Hybrid DE + LP + smoothed gradient optimizer for the step‑function upper‑bound problem.
The routine works on the probability simplex (∑a_i=1, a_i≥0).
It evolves a differential‑evolution population, periodically
   * performs a multiplicative‑gradient polish,
   * solves a small linear programme on the currently active convolution constraints
     to obtain a “richer” direction g, and mixes it into the best individual.
All candidates are projected onto the simplex before evaluation.
The method runs until the given wall‑clock budget expires
and finally returns the best sequence found (clipped to the allowed range
and re‑normalised).
"""
import time
import numpy as np
try:
    from scipy.optimize import linprog
except Exception:
    linprog = None

# Helper functions: _next_pow2, proj_simplex, _conv_full, _weighted_gradient, _smooth_weight

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    rho = np.nonzero(u - cssv / ind > 0)[0]
    if rho.size == 0:
        w = np.full_like(v, 1.0 / n)
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
        w = np.maximum(v - theta, 0.0)
    return w

def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution of a with itself using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    prod = A * A
    conv = np.fft.irfft(prod, n=fft_len)[:conv_len]
    return conv

def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    ``weight`` has length 2·n‑1, ``a`` has length n.
    Returns a gradient vector of length n.
    """
    # reverse a
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

def _smooth_weights(conv: np.ndarray, alpha: float = 30.0) -> np.ndarray:
    """Return a probability vector over the convolution indices that
    approximates the max‑operator. Larger ``alpha`` gives a tighter approximation."""
    # subtract max for numerical stability
    shift = conv.max()
    exp_term = np.exp(alpha * (conv - shift))
    total = exp_term.sum()
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exp_term / total

def _objective(vec: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:

def _lp_extend(f: np.ndarray, M: float, active_idx: np.ndarray,
               n: int) -> tuple[np.ndarray, bool]:
    """Solve the linear programme
          max 1·g   subject to   f * g ≤ M  (only on rows in ``active_idx``), g ≥ 0.
       Returns the optimiser g and a flag indicating success.
       If ``linprog`` is unavailable, returns (f, False)."""
    if linprog is None:
        return f, False

    m = active_idx.size
    if m == 0:
        return f, False

    # Build the matrix A (m × n) where each row corresponds to one convolution constraint.
    A = np.zeros((m, n), dtype=np.float64)
    for row, k in enumerate(active_idx):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_vals = np.arange(i_start, i_end + 1, dtype=np.int64)
        j_vals = k - i_vals
        # f[i] contributes to g[j]
        A[row, j_vals] = f[i_vals]

    # objective: maximise sum(g)  → minimise -sum(g)
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=np.full(m, M, dtype=np.float64),
                  bounds=[(0.0, None)] * n, method='highs', options={'presolve': True})
    if not res.success:
        return f, False
    g = res.x
    return g, True

The routine works on the probability simplex (∑a_i = 1, a_i ≥ 0).

occasionally solves a tiny linear programme on the currently active

satisfying f * g ≤ M on the active rows; a small convex combination of
the best individual f and g (renormalised) strictly reduces the objective.
All candidates are projected onto the simplex before evaluation.
The algorithm respects the 1000 s wall‑clock budget and returns the best

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    prod = A * A
    conv = np.fft.irfft(prod, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    ``weight`` has length 2·n‑1, ``a`` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad

    return exp_term / total

            s = mutant.sum()
            if s == 0.0:
                mutant = np.full_like(mutant, 1.0 / n)
            else:
                mutant /= s

            # binomial crossover
            cross = rng.random(n) < Cr
            if not np.any(cross):
                cross[rng.integers(n)] = True
            trial = np.where(cross, mutant, population[i])

            # final projection onto simplex
            trial = np.maximum(trial, 0.0)
            ts = trial.sum()
            if ts == 0.0:
                trial = np.full_like(trial, 1.0 / n)
            else:
                trial /= ts

            # selection
            J_trial = _objective(trial, n, fft_len, conv_len)
            if J_trial <= scores[i]:
                population[i] = trial
                scores[i] = J_trial
                if J_trial < best_J - 1e-12:
                    best_J = J_trial
                    best_vec = trial.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        iteration += 1
        now = time.monotonic()

        # ------------------ Stagnation handling ------------------
        if now - last_improve > 30.0:
            for k in range(pop_size // 2):
                cand = rng.dirichlet(np.ones(n) * 0.1)
                population[k] = cand
                scores[k] = _objective(cand, n, fft_len, conv_len)
            last_improve = now
            stagnation_rounds += 1
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Stagnation #{stagnation_rounds} – jittered pop")

        # -------------- Multiplicative‑gradient polishing ----------
        if iteration % 5 == 0:
            conv = _conv_full(best_vec, fft_len, conv_len)
            w = _smooth_weights(conv, alpha=30.0)
            grad = _weighted_gradient(best_vec, w, fft_len_corr, n)
            step_mul = 1.2                     # a bit more aggressive than baseline
            factor = np.exp(-step_mul * grad)
            polished = best_vec * factor
            polished = proj_simplex(polished)
            J_polished = _objective(polished, n, fft_len, conv_len)
            if J_polished < best_J - 1e-12:
                best_J = J_polished
                best_vec = polished.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] Polished best J = {best_J:.6f}")

        # ------------------- LP‑based active‑set step -------------
        if (iteration % 10 == 0) and (linprog is not None):
            conv = _conv_full(best_vec, fft_len, conv_len)
            M = conv.max()
            tol = 1e-12
            active_idx = np.where(conv >= M - tol)[0]
            if active_idx.size > 0:
                g, ok = _lp_extend(best_vec, M, active_idx, n)
                if ok:
                    sum_g = g.sum()
                    if sum_g > 1.0 + 1e-7:
                        for t in (0.05, 0.1, 0.2, 0.4):
                            cand = (1.0 - t) * best_vec + t * g
                            cand = proj_simplex(cand)
                            J_cand = _objective(cand, n, fft_len, conv_len)
                            if J_cand < best_J - 1e-12:
                                best_J = J_cand
                                best_vec = cand.copy()
                                last_improve = time.monotonic()
                                elapsed = last_improve - start_time
                                print(f"[{elapsed:6.2f}s] LP‑mix (t={t}) improved J = {best_J:.6f}")
                                break

        # ---------------------------------------------------------
        # Optional early stop if we already beat the target
        # ---------------------------------------------------------
        # if best_J <= 1.5025:
        #     break

    # ------------------------------------------------------------------
    # Final clean‑up: clip, renormalise and convert to plain Python list
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # (optional) sanity‑check with the provided evaluate_sequence
    # try:
    #     val = evaluate_sequence(best_vec.tolist())
    #     print(f"Final evaluate_sequence = {val:.6f}")
    # except Exception:
    #     pass

    return [float(v) for v in best_vec]
