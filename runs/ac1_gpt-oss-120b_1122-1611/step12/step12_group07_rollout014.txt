"""
Hybrid simulated‑annealing / greedy descent for the step‑function problem.

The objective J(a) = 2·n·max(conv(a,a)) (the sum of a is kept =1).
We work directly on the probability simplex: moves transfer a small amount of
mass from a high‑contributing index (the one that most contributes to the
current maximal convolution entry) to a random index.  The move is accepted
with a Metropolis probability that cools down from an initial temperature
to almost zero.  When the algorithm stagnates we perform a small Gaussian
jitter (or restart from a random Dirichlet) to escape flat regions.
The best vector seen (according to the true evaluation function) is kept
and returned as a plain Python list.

The routine respects the 1000‑second budget, works for any existing global
`height_sequence_1`, and uses only NumPy and the provided `evaluate_sequence`.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Utility functions (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution of a with itself using a real FFT.
    Returns an array of length `conv_len = 2*len(a)-1`.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    # Numerical noise may produce tiny negative values – clip to 0.
    conv = np.maximum(conv, 0.0)
    return conv


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(conv(a,a)).  a is assumed to sum to 1."""
    conv = _conv_full(a, fft_len, conv_len)
    return float(2.0 * n * conv.max())


def _contrib_vector(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Compute the contribution a_i * a_{k_max-i} for all i.
    Entries where the partner index is out of range are set to 0.
    """
    n = a.shape[0]
    idx = np.arange(n, dtype=np.int64)
    partner = k_max - idx
    mask = (partner >= 0) & (partner < n)
    contrib = np.zeros_like(a)
    contrib[mask] = a[idx[mask]] * a[partner[mask]]
    return contrib


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Returned vector is non‑negative and sums to 1.
    """
    # Implementation of the algorithm from
    #  Wang & Carreira‑Perpinán (2013), “Projection onto the probability simplex”
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Main search routine
# ----------------------------------------------------------------------

    # a few deterministic seeds (uniform, linear ramps, geometric)
    init_candidates.append(np.full(n, 1.0 / n))               # uniform
    ramp = np.arange(n, dtype=np.float64)
    init_candidates.append(ramp / ramp.sum())                # increasing
    init_candidates.append((n - 1 - ramp) / (n - 1 - ramp).sum())  # decreasing
    for r in (0.90, 0.95, 0.97, 0.99, 0.995):
        g = r ** ramp
        g /= g.sum()
        init_candidates.append(g)

    # random Dirichlet draws
    for _ in range(4):
        init_candidates.append(_initial_vector())

    # --------------------------------------------------------------
    # 3. Choose the best start according to the exact objective J
    # --------------------------------------------------------------
    best_a = None
    best_J = np.inf
    for cand in init_candidates:
        J = _objective(cand, n, fft_len, conv_len)
        if J < best_J:
            best_J = J
            best_a = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4. Simulated‑annealing / greedy descent loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    cur_a = best_a.copy()
    cur_J = best_J

    global_best_a = cur_a.copy()
    global_best_J = cur_J

    # annealing schedule
    T0 = 0.5                     # initial temperature
    T_end = 1e-6                 # final temperature
    total_time = deadline - start_time

    # stagnation handling
    last_improve_time = start_time
    stagnation_counter = 0

    iteration = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        # linear progress ratio
        progress = min(1.0, elapsed / total_time)
        # exponential temperature decay
        T = T0 * (T_end / T0) ** progress

        # ----------------------------------------------------------
        # 4.1 Compute current convolution and locate the maximal offset
        # ----------------------------------------------------------
        conv = _conv_full(cur_a, fft_len, conv_len)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # ----------------------------------------------------------
        # 4.2 Identify the index that contributes most to this max
        # ----------------------------------------------------------
        contrib = _contrib_vector(cur_a, k_max)
        i_src = int(np.argmax(contrib))
        # avoid a degenerate move when the source contribution is (almost) zero
        if contrib[i_src] <= 1e-15:
            i_src = rng.integers(n)

        # ----------------------------------------------------------
        # 4.3 Propose a move: transfer a fraction of mass from i_src to i_dst
        # ----------------------------------------------------------
        i_dst = rng.integers(n)
        while i_dst == i_src:
            i_dst = rng.integers(n)

        # step size scales with temperature (large jumps early, small later)
        eps = 0.1 * (T / T0) + 0.001
        delta = eps * cur_a[i_src]                 # amount to move
        delta = min(delta, cur_a[i_src])           # stay non‑negative

        cand_a = cur_a.copy()
        cand_a[i_src] -= delta
        cand_a[i_dst] += delta

        # ----------------------------------------------------------
        # 4.4 Compute objective for the candidate
        # ----------------------------------------------------------
        cand_J = _objective(cand_a, n, fft_len, conv_len)

        # ----------------------------------------------------------
        # 4.5 Metropolis acceptance
        # ----------------------------------------------------------
        accept = False
        if cand_J < cur_J:
            accept = True
        else:
            # uphill move accepted with probability exp(-(ΔJ)/T)
            prob = np.exp(-(cand_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_a = cand_a
            cur_J = cand_J
            # record global best
            if cur_J < global_best_J - 1e-12:
                global_best_J = cur_J
                global_best_a = cur_a.copy()
                last_improve_time = time.monotonic()
                print(f"[{elapsed:6.2f}s] New best J = {global_best_J:.6f}")

        # ----------------------------------------------------------
        # 4.6 Stagnation detection – jitter/restart
        # ----------------------------------------------------------
        if time.monotonic() - last_improve_time > 30.0:
            # Apply a small Gaussian jitter to the best‑so‑far vector
            jitter = rng.normal(0.0, 0.02, size=n)
            jittered = global_best_a + jitter
            jittered = np.maximum(jittered, 0.0)
            jittered = _project_simplex(jittered)
            cur_a = jittered
            cur_J = _objective(cur_a, n, fft_len, conv_len)
            last_improve_time = time.monotonic()
            stagnation_counter = 0
            print(f"[{elapsed:6.2f}s] Restart – jitter applied (best J = {global_best_J:.6f})")

        iteration += 1

    # --------------------------------------------------------------
    # 5. Return the best vector found (clipped, optional renorm)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_a, 0.0, 1000.0)

    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec = best_vec / total    # keep it a probability vector

    print(f"Finished after {iteration:,} iterations. Best J = {global_best_J:.6f}")
    return [float(v) for v in best_vec]
