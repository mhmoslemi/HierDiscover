"""
Propose a high‑quality step‑function by (i) building several diverse
initial candidates, (ii) intensive sub‑gradient descent on the
max‑convolution objective, and (iii) a short smoothed‑gradient polishing
stage followed by a tiny random‑perturbation sweep.

All helper functions are top‑level (no closures).  The algorithm works
directly on the probability simplex (sum = 1); the evaluation function
`evaluate_sequence` is invariant under scaling, so the returned list
can be used as‑is.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. Simple utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear convolution a * a using an FFT of length `fft_len`.
    Returns only the first `conv_len` entries (true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


def compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Objective J = 2·n·max(conv(a,a)) for a probability vector a.
    """
    conv = conv_full_fft(a, fft_len, conv_len)
    max_conv = np.max(conv)
    return float(2.0 * n * max_conv)


# ----------------------------------------------------------------------
# 2. Core optimisation procedures
# ----------------------------------------------------------------------

                        conv_len: int,

                        deadline: float,

    Projected sub‑gradient descent on the max‑convolution objective.
    Returns the best vector seen and its J‑value.

    best_a = a.copy()
    best_J = compute_J(a, n, fft_len, conv_len)

    for t in range(max_iters):
        if time.monotonic() >= deadline:
            break

        conv = conv_full_fft(a, fft_len, conv_len)
        max_idx = int(np.argmax(conv))

        # --- sub‑gradient of the active index -----------------------
        start_idx = max(0, max_idx - (n - 1))
        end_idx = min(max_idx, n - 1)
        segment = a[start_idx:end_idx + 1]
        g = np.zeros_like(a)
        g[start_idx:end_idx + 1] = 2.0 * segment[::-1]

        # --- step (diminishing learning rate) -----------------------
        lr = lr0 / np.sqrt(t + 1.0)
        a = proj_simplex(a - lr * g)

        # --- occasional evaluation ----------------------------------
        if (t + 1) % 200 == 0:
            J = compute_J(a, n, fft_len, conv_len)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Candidate {candidate_id} iter {t+1} "
                      f"new best J = {best_J:.6f}")

    return best_a, best_J


def smooth_gradient_descent(a0: np.ndarray,
                            n: int,
                            fft_len: int,
                            conv_len: int,
                            steps: int,
                            lr0: float,
                            alpha_start: float,
                            alpha_end: float,
                            deadline: float,
                            start_time: float):
    """
    A few iterations of gradient descent on a softmax approximation of
    max(conv) to polish the solution.
    """
    a = a0.copy()
    best_a = a.copy()
    best_J = compute_J(a, n, fft_len, conv_len)

    for t in range(steps):
        if time.monotonic() >= deadline:
            break

        # convolution
        conv = conv_full_fft(a, fft_len, conv_len)

        # soft‑max weights
        alpha = alpha_start + (alpha_end - alpha_start) * (t / steps)
        scaled = alpha * conv
        max_scaled = np.max(scaled)
        w = np.exp(scaled - max_scaled)
        p = w / np.sum(w)                     # length = conv_len

        # gradient = 2 * (a ⋆ p)  (cross‑correlation)
        p_rev = p[::-1]
        L_corr = _next_pow2(len(a) + len(p_rev) - 1)
        A = np.fft.rfft(a, L_corr)
        P = np.fft.rfft(p_rev, L_corr)
        corr = np.fft.irfft(A * P, n=L_corr)  # linear convolution a * p_rev
        idx_start = len(p) - 1                # = 2*n-2
        g = 2.0 * corr[idx_start - np.arange(n)]

        # step
        lr = lr0 / np.sqrt(t + 1.0)
        a = proj_simplex(a - lr * g)

        # occasional check
        if (t + 1) % 200 == 0:
            J = compute_J(a, n, fft_len, conv_len)
            if J < best_J - 1e-12:
                best_J = J
                best_a = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Fine‑grad iter {t+1} new best J = {best_J:.6f}")

    return best_a, best_J

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine builds a handful of diverse initial
    vectors, runs a heavy sub‑gradient descent on each, finishes with a
    short smoothed‑gradient polishing stage and a tiny random perturbation
    sweep, and finally returns the best vector found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Problem size (use the length of the known construction if possible)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            old = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            n = old.size                # keep the original length
        except Exception:
            n = 2048
    else:
        n = 2048

    # Convolution parameters
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)      # for a*a

    # --------------------------------------------------------------
    # Build a modest pool of seed vectors (all on the simplex)
    # --------------------------------------------------------------
    seeds = []

    # uniform
    seeds.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays (forward and backward)
    for r in (0.90, 0.95, 0.975, 0.99, 0.995):
        v = (r ** np.arange(n, dtype=np.float64))
        seeds.append(proj_simplex(v))
        v_rev = (r ** np.arange(n - 1, -1, -1, dtype=np.float64))
        seeds.append(proj_simplex(v_rev))

    # linear ramps
    seeds.append(proj_simplex(np.arange(n, 0, -1, dtype=np.float64)))
    seeds.append(proj_simplex(np.arange(1, n + 1, dtype=np.float64)))

    # Gaussian bells of several widths
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    for sigma in (n / 10.0, n / 8.0, n / 6.0):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        seeds.append(proj_simplex(v))

    # Dirichlet draws with different concentrations
    for alpha in (0.3, 0.6, 1.0, 2.0):
        for _ in range(3):
            v = rng.dirichlet(np.full(n, alpha))
            seeds.append(v)

    # previous best construction (if available)
    if "height_sequence_1" in globals():
        try:
            v = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if v.size != n:
                if v.size > n:
                    v = v[:n]
                else:
                    v = np.concatenate([v, np.zeros(n - v.size, dtype=np.float64)])
            seeds.append(proj_simplex(v))
        except Exception:
            pass

    # --------------------------------------------------------------
    # Global bookkeeping
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # small safety margin
    best_vec = None
    best_J = np.inf

    print(f"[0.00s] Starting search with n = {n}, candidate pool = {len(seeds)}")

    # --------------------------------------------------------------
    # 1) Heavy sub‑gradient descent on each seed
    # --------------------------------------------------------------
    max_iters_per_seed = 8000          # adjustable; cheap because gradient is O(n)
    lr0 = 0.5

    for cid, cand in enumerate(seeds):
        if time.monotonic() >= deadline:
            break

        # evaluate the raw candidate first
        J0 = compute_J(cand, n, fft_len, conv_len)
        if J0 < best_J:
            best_J = J0
            best_vec = cand.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Seed {cid} initial J = {J0:.6f} (new global best)")

        # run sub‑gradient descent
        cand_best, cand_J = subgradient_descent(
            a0=cand,
            n=n,
            fft_len=fft_len,
            conv_len=conv_len,
            max_iters=max_iters_per_seed,
            lr0=lr0,
            deadline=deadline,
            start_time=start_time,
            candidate_id=cid,
        )

        if cand_J < best_J - 1e-12:
            best_J = cand_J
            best_vec = cand_best.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Seed {cid} finished with improved J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 2) Polishing stage – smoothed gradient descent on the best so far
    # --------------------------------------------------------------
    if best_vec is not None and time.monotonic() < deadline:
        fine_steps = 2000
        polished_vec, polished_J = smooth_gradient_descent(
            a0=best_vec,
            n=n,
            fft_len=fft_len,
            conv_len=conv_len,
            steps=fine_steps,
            lr0=0.15,
            alpha_start=10.0,
            alpha_end=2000.0,
            deadline=deadline,
            start_time=start_time,
        )
        if polished_J < best_J - 1e-12:
            best_J = polished_J
            best_vec = polished_vec.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Polishing improved J to {best_J:.6f}")

    # --------------------------------------------------------------
    # 3) Tiny random‑perturbation sweep (final local search)
    # --------------------------------------------------------------
    if best_vec is not None and time.monotonic() < deadline:
        fine_start = time.monotonic()
        fine_deadline = fine_start + min(30.0, (deadline - start_time) * 0.05)
        while time.monotonic() < fine_deadline:
            noise = np.exp(rng.normal(0.0, 0.02, size=n))
            cand = proj_simplex(best_vec * noise)
            Jc = compute_J(cand, n, fft_len, conv_len)
            if Jc < best_J - 1e-12:
                best_J = Jc
                best_vec = cand.copy()
                elapsed = time.monotonic() - start_time
                print(f"[Fine {elapsed:6.2f}s] Random perturbation J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Final sanity – keep the vector on the admissible range
    # --------------------------------------------------------------
    if best_vec is None:                     # should never happen
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s                     # enforce sum = 1 (probability simplex)

    final_J = compute_J(best_vec, n, fft_len, conv_len)
    total_time = time.monotonic() - start_time
    print(f"[{total_time:6.2f}s] Search finished – best J = {final_J:.6f}")

    # Return as plain Python list
    return [float(v) for v in best_vec]
