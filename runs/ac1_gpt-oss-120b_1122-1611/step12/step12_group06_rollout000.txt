"""
Simulated‑annealing step‑function optimiser.

The algorithm works directly on the probability vector `a` (sum = 1, a_i ≥ 0)
and repeatedly moves a small amount of mass from one index to another.
Each proposal is evaluated with the exact objective

    J(a) = 2·n·max(conv(a,a))

where `conv` is the full linear convolution (computed by an FFT).  A
temperature schedule allows occasional uphill moves; when no improvement
has been seen for a while a Gaussian jitter is applied to escape shallow
local minima.  The best vector found before the timeout is returned as a
plain Python list (values are already clipped to the allowed range).

The method is deliberately different from the gradient‑soft‑max/Adam
approach used previously, relying on stochastic search rather than
gradient information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two greater than or equal to ``x``."""
    return 1 << ((x - 1).bit_length())


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Compute max(conv(a, a)) using a real FFT of length ``fft_len``.
    ``a`` is assumed to be a 1‑D array of non‑negative numbers that sums to 1.
    """
    # rfft pads with zeros up to fft_len automatically
    A = np.fft.rfft(a, fft_len)
    # element‑wise product gives the spectrum of the convolution
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Stochastic‑annealing search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns the
    best vector discovered (as a list of Python floats).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 1. Problem size ------------------------------------------------
    # --------------------------------------------------------------
    # Prefer to reuse the length of the best known construction if it exists.
    if "height_sequence_1" in globals():
        try:
            n = int(len(height_sequence_1))
        except Exception:
            n = 3000
    else:
        n = 3000

    # A modest upper bound on n keeps the FFT cheap while still giving a
    # sufficiently rich search space.
    n = max(1024, min(n, 8000))

    # FFT length – power of two for speed.
    fft_len = _next_pow2(2 * n)

    # --------------------------------------------------------------
    # 2. Initial vector -----------------------------------------------
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        # Trim or pad to the chosen length
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            a0 = np.concatenate([a0, np.full(n - a0.size, 0.0)])
        a0 = np.clip(a0, 0.0, None)
        total = a0.sum()
        if total == 0.0:
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a = a0 / total
    else:
        # A Dirichlet draw with a small concentration creates a spiky start.
        a = rng.dirichlet(np.full(n, 0.3))
    # Ensure exact normalisation (helps numerical stability)
    a = a / a.sum()

    # Compute the exact objective for the starter.
    cur_max_conv = _max_convolution(a, fft_len)
    cur_J = 2.0 * n * cur_max_conv
    best_J = cur_J
    best_a = a.copy()

    print(f"[0.00s] Start – n = {n}, J = {cur_J:.6f}")

    # --------------------------------------------------------------
    # 3. Simulated annealing parameters --------------------------------
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # leave a safety margin

    # Temperature schedule (linear from T0 to T_end)
    T0 = 0.20          # initial temperature (allows sizeable uphill steps)
    T_end = 1e-5       # final temperature (behaves like hill‑climbing)

    # Jitter parameters – applied when we have been stuck for a while.
    jitter_interval = 30.0   # seconds without improvement before a jitter
    jitter_sigma = 0.02      # standard deviation of Gaussian noise

    last_improve = start_time
    last_status = start_time

    # --------------------------------------------------------------
    # 4. Main annealing loop ------------------------------------------
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        total_allowed = deadline - start_time
        progress = elapsed / total_allowed

        # Linear cooling
        T = T0 * (1.0 - progress) + T_end * progress

        # ---- 4.1 propose a move -----------------------------------
        # Choose a source index i (must have some mass to give)
        i = rng.integers(n)
        # If the chosen i currently holds no mass, retry a few times.
        tries = 0
        while a[i] < 1e-14 and tries < 5:
            i = rng.integers(n)
            tries += 1
        if a[i] < 1e-14:
            # all entries are essentially zero – should not happen, break.
            continue

        # Choose a destination index j ≠ i uniformly.
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        # Amount of mass to move.  Early on we allow larger steps,
        # later we shrink the step size with the temperature.
        max_move = min(0.5, a[i])                      # never move more than half of a[i]
        step_scale = max(0.01, T / T0)                 # shrink as T drops
        delta = rng.random() * max_move * step_scale

        # ---- 4.2 apply the move in‑place (remember old values) ---
        old_i = a[i]
        old_j = a[j]
        a[i] -= delta
        a[j] += delta

        # ---- 4.3 evaluate the new objective ------------------------
        new_max_conv = _max_convolution(a, fft_len)
        new_J = 2.0 * n * new_max_conv

        # ---- 4.4 acceptance test ------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(-(new_J - cur_J) / (T + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_J = new_J
            cur_max_conv = new_max_conv
            # Record a new global best if applicable
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")
        else:
            # revert the move
            a[i] = old_i
            a[j] = old_j

        # ---- 4.5 periodic status output ---------------------------
        if now - last_status > 30.0:
            print(f"[{elapsed:6.2f}s] Current J = {cur_J:.6f}, Best J = {best_J:.6f}")
            last_status = now

        # ---- 4.6 jitter if stuck ------------------------------------
        if now - last_improve > jitter_interval:
            # Apply a modest Gaussian perturbation and renormalise.
            noise = rng.normal(0.0, jitter_sigma, size=n)
            a = a + noise
            a = np.clip(a, 0.0, None)
            total = a.sum()
            if total == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a = a / total
            # Re‑evaluate after jitter
            cur_max_conv = _max_convolution(a, fft_len)
            cur_J = 2.0 * n * cur_max_conv
            print(f"[{elapsed:6.2f}s] Jitter applied – J = {cur_J:.6f}")
            last_improve = now  # reset stall timer after jitter

    # --------------------------------------------------------------
    # 5. Return the best sequence found --------------------------------
    # --------------------------------------------------------------
    # Clip to the allowed numeric range (the optimiser never exceeds 1000,
    # but we enforce the limits explicitly for safety) and renormalise.
    best_vec = np.clip(best_a, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec /= total

    final_J = 2.0 * n * _max_convolution(best_vec, fft_len)
    print(f"[{budget_s:.0f}s] Finished – best J = {final_J:.6f}")

    # Convert to plain Python floats as required by the external harness.
    return [float(v) for v in best_vec]
