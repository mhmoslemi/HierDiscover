        a = a - step
        # project onto simplex
        a = proj_simplex(a)

        # adjust learning rate

            a = a + sigma * rng.normal(size=n)
            a = proj_simplex(a)
            last_improve = time.monotonic()

    fine_deadline = time.monotonic() + min(30.0, (deadline - start_time) * 0.05)
    while time.monotonic() < fine_deadline:

        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        conv_c = conv_full_fft(cand, fft_len, conv_len)

            best_vec = cand
            print(f"[Fine {time.monotonic() - start_time:6.2f}s] Improved J = {best_J:.6f}")

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total
    return [float(v) for v in best_vec]

"""
Simple smooth‑max gradient descent for the autocorrelation minimisation problem.

The algorithm works directly on the probability simplex:
  * The current vector a (sum = 1, a_i ≥ 0) is transformed to the
    full convolution conv = a * a (via a single FFT).
  * A smooth approximation of the max‑operator is built using a soft‑max
    with a temperature τ that is slowly annealed.
  * The gradient of the (smooth) objective J̃(a)=2·n·∑_k w_k·conv_k
    (where w is the soft‑max weight vector) is
        ∇J̃ = 4·n·(w ⋆ a)   (cross‑correlation of w with a).
    The correlation is obtained efficiently with another FFT.
  * An Adam optimiser is applied to the gradient and the iterate is
    projected back onto the probability simplex after each step.
  * If no improvement is seen for 30 s the iterate is perturbed with a
    small Gaussian noise to escape possible stagnation.
  * After the main time budget a quick random‑jitter fine search around the
    best vector is performed.

The routine respects the 1000 s budget, prints progress updates and finally
returns the best sequence (as a list of floats) found within the time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Compute the full linear convolution a * a using a real FFT.
    Returns the first `conv_len` entries (the true linear convolution).
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)
    return conv[:conv_len]


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Gradient‑descent + annealed soft‑max optimisation of the step‑function
    heights.  Returns the best sequence found within the allocated budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Choose problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n_prev = len(height_sequence_1)
            # keep the size in a reasonable window
            n = max(2048, min(n_prev, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1                     # length of a*a
    # Length that can accommodate both the convolution and the correlation
    fft_len = _next_pow2(conv_len + n - 1)   # ≥ 3·n−2

    # ------------------------------------------------------------------
    # 2. Initialise a feasible point on the simplex
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(height_sequence_1, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            if a0.size > n:
                a0 = a0[:n]
            else:
                a0 = np.concatenate([a0, np.zeros(n - a0.size, dtype=np.float64)])
        a = proj_simplex(a0)
    else:
        # Dirichlet draw with moderate concentration (α=0.5 works well)
        a = rng.dirichlet(np.full(n, 0.5, dtype=np.float64))
        a = proj_simplex(a)

    # ------------------------------------------------------------------
    # 3. Optimisation parameters
    # ------------------------------------------------------------------
    # Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                     # base learning rate
    lr_decay = 0.9999

    # Soft‑max temperature
    tau_min = 1e-9
    tau_decay = 0.9995
    tau = None                    # will be set after first evaluation

    # Book‑keeping
    start_time = time.monotonic()
    # keep a small safety margin for the final jitter phase
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Compute initial objective
    conv_full = _conv_full_fft(a, fft_len, conv_len)
    max_conv = conv_full.max()
    best_J = 2.0 * n * max_conv
    best_vec = a.copy()
    last_improve = start_time
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    iteration = 0

    # ------------------------------------------------------------------
    # 4. Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ----- a) current convolution (full) --------------------------------
        A_fft = np.fft.rfft(a, fft_len)                     # FFT of the current a
        conv_full = np.fft.irfft(A_fft * A_fft, n=fft_len)[:conv_len]
        max_conv = conv_full.max()
        J = 2.0 * n * max_conv

        # ----- record best --------------------------------------------------
        if J < best_J - 1e-12:
            best_J = J
            best_vec = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- b) update the soft‑max temperature ---------------------------
        if tau is None:
            tau = max(max_conv * 0.2, tau_min)
        else:
            tau = max(tau * tau_decay, tau_min)

        # ----- c) weight vector w (soft‑max over the convolution) ----------
        shifted = conv_full - max_conv          # ≤ 0
        # avoid overflow/underflow – use safe exponentiation
        exp_vals = np.exp(shifted / tau)
        w_sum = exp_vals.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            # extreme case – fall back to a one‑hot on the maximum location
            w = np.zeros_like(conv_full)
            w[conv_full.argmax()] = 1.0
        else:
            w = exp_vals / w_sum                # shape (conv_len,)

        # ----- d) gradient via cross‑correlation (w ⋆ a) --------------------
        W_fft = np.fft.rfft(w, fft_len)
        corr_full = np.fft.irfft(W_fft * np.conj(A_fft), n=fft_len)
        # cross‑correlation slice giving a vector of length n
        start_idx = n - 1
        corr = corr_full[start_idx:start_idx + n]   # (w ⋆ a)

        # Gradient of the smoothed objective J̃ = 2·n·∑ w·conv
        grad = 4.0 * n * corr                     # shape (n,)

        # ----- e) Adam update -----------------------------------------------
        t = iteration
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- f) projected gradient step ------------------------------------
        a = a - step
        a = proj_simplex(a)                       # stays on the simplex

        # ----- g) learning‑rate decay ----------------------------------------
        lr *= lr_decay

        # ----- h) stagnation handling ----------------------------------------
        if time.monotonic() - last_improve > 30.0:
            # add a small multiplicative Gaussian perturbation
            sigma = 0.02
            a = proj_simplex(a * np.exp(rng.normal(0.0, sigma, size=n)))
            # reset Adam statistics to avoid stale momentum
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Stagnation reset (noise injection)")

    # ------------------------------------------------------------------
    # 5. Fine‑grained random jitter around the best solution
    # ------------------------------------------------------------------
    fine_deadline = time.monotonic() + min(30.0, (deadline - start_time) * 0.05)
    while time.monotonic() < fine_deadline:
        # multiplicative Gaussian jitter (small sigma)
        noise = np.exp(rng.normal(0.0, 0.015, size=n))
        cand = proj_simplex(best_vec * noise)
        conv_c = _conv_full_fft(cand, fft_len, conv_len)
        Jc = 2.0 * n * conv_c.max()
        if Jc < best_J:
            best_J = Jc
            best_vec = cand
            elapsed = time.monotonic() - start_time
            print(f"[Fine {elapsed:6.2f}s] Improved J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 6. Prepare and return the final sequence
    # ------------------------------------------------------------------
    # Clip to the integer limits enforced inside `evaluate_sequence`
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    # Return as plain Python list of floats
    return [float(v) for v in best_vec]
