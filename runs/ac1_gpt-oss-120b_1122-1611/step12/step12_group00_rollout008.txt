"""Hybrid differential‑evolution + LP refinement for the max‑auto‑convolution
task.

The algorithm works in three stages:

1. **Differential Evolution (DE)** – a small, diverse population of
   probability vectors (∑a_i = 1) is evolved for a fraction of the time
   budget.  The DE uses mutation factor `F`, crossover probability `CR`,
   Euclidean projection onto the simplex and occasional population refresh
   to escape stagnation.

2. **Linear‑Programming (LP) refinement** – given the best DE vector `f`,
   we solve
        maximise   1ᵀg
        subject to  conv(f,g) ≤ M   (component‑wise),
                    g ≥ 0,
   where `M = max(conv(f,f))`.  The solution `g` is scaled back to a
   probability vector and mixed with `f` (`f ← (1‑t)f + t·g`).  Repeating
   this step strictly reduces `M` (and therefore the objective
   `2·n·max(conv)`).  The LP is built with a dense Toeplitz matrix; for
   `n = 1024` this is cheap and solves with SciPy’s HiGHS solver in a
   few tenths of a second.

3. **Fine‑grained jitter** – a short random‑perturbation phase around the
   best vector found so far finalises the search.

The returned list sums to 1 (a probability vector); `evaluate_sequence`
therefore yields the same objective used throughout (`2·n·max(conv)`).
"""
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions (all defined at module level – no closures)
# ----------------------------------------------------------------------


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def conv_max(seq: np.ndarray) -> float:
    """Maximum value of the full linear convolution of `seq` with itself."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


def objective(seq: np.ndarray, n: int) -> float:
    """Objective used in the optimisation: 2·n·max(conv(seq,seq))."""
    return 2.0 * n * conv_max(seq)


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that A @ g == conv(f, g) (full linear
    convolution).  Shape of A is (2·n‑1, n).
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contains f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(f: np.ndarray, t: float, max_iter: int, deadline: float) -> tuple:
    """
    Perform LP refinement on probability vector `f`.

    Parameters
    ----------
    f : np.ndarray
        Current probability vector (sums to 1).
    t : float
        Mixing factor (0 < t ≤ 1).  New vector becomes (1‑t)·f + t·g.
    max_iter : int
        Upper bound on the number of refinement steps.
    deadline : float
        Absolute time (as returned by `time.monotonic()`) after which we
        must stop.

    Returns
    -------
    best_f : np.ndarray
        The best probability vector found during refinement.
    best_obj : float
        Corresponding objective value.
    """
    n = f.size
    best_f = f.copy()
    best_obj = objective(best_f, n)

    while time.monotonic() < deadline and max_iter > 0:
        max_iter -= 1
        M = conv_max(f)                     # current peak of f*f
        A = build_conv_matrix(f)            # (2n‑1) × n matrix
        b_ub = np.full(A.shape[0], M)       # all constraints ≤ M
        c = -np.ones(n)                     # maximise sum(g)  → minimise –sum(g)
        bounds = [(0.0, None) for _ in range(n)]

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if not res.success:
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # Scale g back to a probability vector (sum = 1)
        g_norm = g / sum_g

        # Mix with current f
        f_new = (1.0 - t) * f + t * g_norm
        f_new = proj_simplex(f_new)         # keep on the simplex

        obj_new = objective(f_new, n)

        # Keep the best ever seen
        if obj_new < best_obj - 1e-12:
            best_obj = obj_new
            best_f = f_new.copy()
            print(f"[LP] improvement: objective = {best_obj:.6f}")

        # Accept the new f only if it strictly improves the peak
        if obj_new < best_obj - 1e-12 or obj_new < best_obj * 0.9999:
            f = f_new
        else:
            # No improvement – stop early
            break

    return best_f, best_obj


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Hybrid DE + LP optimiser for `evaluate_sequence`.  Returns a list of
    non‑negative heights (probability vector) that minimises the evaluation
    function within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size ----------------------------------------------------
    # ------------------------------------------------------------------
    n = 1024                 # moderate size – LP is cheap at this scale
    pop_size = 30           # DE population size
    F = 0.6                  # mutation factor
    CR = 0.9                 # crossover probability
    t_lp = 0.2               # mixing factor for LP refinement
    max_lp_iters = 50        # safeguard for LP loop

    # ------------------------------------------------------------------
    # Initialise a diverse population ---------------------------------
    # ------------------------------------------------------------------
    pop = np.empty((pop_size, n), dtype=np.float64)
    pop_obj = np.empty(pop_size, dtype=np.float64)

    # Helper to evaluate a candidate and store it into the population
    def evaluate_and_store(idx: int, vec: np.ndarray):
        pop[idx] = vec
        pop_obj[idx] = objective(vec, n)

    # 1) Uniform
    evaluate_and_store(0, np.full(n, 1.0 / n, dtype=np.float64))

    # 2) Gaussian bells
    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    sigmas = [n / 12.0, n / 9.0, n / 6.0]
    for i, sigma in enumerate(sigmas, start=1):
        v = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        v /= v.sum()
        evaluate_and_store(i, v)

    # 3) Linear ramps (decreasing & increasing)
    evaluate_and_store(4, np.linspace(1.0, 0.0, n, dtype=np.float64))
    evaluate_and_store(5, np.linspace(0.0, 1.0, n, dtype=np.float64))

    # 4) Dirichlet draws with different concentration parameters
    idx = 6
    for alpha in (0.3, 0.6, 1.0, 2.0):
        for _ in range(2):
            v = rng.dirichlet(np.ones(n) * alpha)
            evaluate_and_store(idx, v)
            idx += 1

    # 5) If a previous construction is available, truncate / pad it
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.size > n:
            v = v[:n]
        elif v.size < n:
            pad = np.zeros(n - v.size, dtype=np.float64)
            v = np.concatenate([v, pad])
        v = np.clip(v, 0.0, None)
        v = proj_simplex(v)
        evaluate_and_store(idx, v)
        idx += 1

    # 6) Fill the rest with random Dirichlet draws
    while idx < pop_size:
        alpha = rng.uniform(0.1, 2.0)
        v = rng.dirichlet(np.ones(n) * alpha)
        evaluate_and_store(idx, v)
        idx += 1

    # Current global best
    best_idx = int(np.argmin(pop_obj))
    best_vec = pop[best_idx].copy()
    best_obj = float(pop_obj[best_idx])

    print(f"[0.00s] Initial best objective = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # DE main loop ----------------------------------------------------
    # ------------------------------------------------------------------
    start = time.monotonic()
    # Reserve about 40 % of the budget for LP refinement and final jitter
    de_deadline = start + max(0.0, float(budget_s) * 0.55)
    stagnation_timer = start
    last_improve = start

    iteration = 0
    while time.monotonic() < de_deadline:
        now = time.monotonic()
        elapsed = now - start

        # One full generation
        for i in range(pop_size):
            # Choose three distinct individuals different from i
            idxs = list(range(pop_size))
            idxs.remove(i)
            a, b, c = rng.choice(idxs, size=3, replace=False)

            mutant = pop[a] + F * (pop[b] - pop[c])
            mutant = np.maximum(mutant, 0.0)

            cross_mask = rng.random(n) < CR
            trial = np.where(cross_mask, mutant, pop[i])
            trial = proj_simplex(trial)

            trial_obj = objective(trial, n)

            if trial_obj < pop_obj[i]:
                pop[i] = trial
                pop_obj[i] = trial_obj
                if trial_obj < best_obj - 1e-12:
                    best_obj = trial_obj
                    best_vec = trial.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] DE new best = {best_obj:.6f}")

        iteration += 1

        # ------------------------------------------------------------------
        # Stagnation handling – inject fresh individuals --------------------
        # ------------------------------------------------------------------
        if now - last_improve > 30.0:
            n_replace = max(1, pop_size // 5)
            replace_idxs = rng.choice(pop_size, size=n_replace, replace=False)
            for ridx in replace_idxs:
                alpha = rng.uniform(0.1, 2.0)
                v = rng.dirichlet(np.ones(n) * alpha)
                v = proj_simplex(v)
                evaluate_and_store(ridx, v)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Re‑initialised {n_replace} individuals (stagnation)")

    # ------------------------------------------------------------------
    # LP refinement ----------------------------------------------------
    # ------------------------------------------------------------------
    lp_deadline = start + float(budget_s) - 30.0   # keep ~30 s for final jitter
    best_vec, best_obj = lp_refine(best_vec, t_lp, max_lp_iters, lp_deadline)

    # ------------------------------------------------------------------
    # Fine‑grained random jitter ---------------------------------------
    # ------------------------------------------------------------------
    fine_deadline = start + float(budget_s)
    while time.monotonic() < fine_deadline:
        # Small multiplicative Gaussian perturbation
        noise = np.exp(rng.normal(0.0, 0.02, size=n))
        cand = proj_simplex(best_vec * noise)
        cand_obj = objective(cand, n)
        if cand_obj < best_obj:
            best_obj = cand_obj
            best_vec = cand
            elapsed = time.monotonic() - start
            print(f"[Fine {elapsed:6.2f}s] jitter improvement = {best_obj:.6f}")

    # ------------------------------------------------------------------
    # Return the final probability vector (sum = 1) -------------------
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= total

    return [float(v) for v in best_vec]
