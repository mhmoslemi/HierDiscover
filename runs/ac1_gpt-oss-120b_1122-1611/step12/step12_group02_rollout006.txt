"""
Gradient‑based exponential descent on the simplex.

The routine keeps the vector on the probability simplex (∑ a_i = 1,
a_i ≥ 0).  At each iteration we compute the full convolution
b = a * a, build a soft‑max weight
    w_k ∝ exp(β (b_k – max(b)))                     (β grows with time)
and obtain a dense sub‑gradient of the smoothed max‑operator via
the correlation _weighted_gradient.  An exponentiated‑gradient step
a_i ← a_i·exp(−η·g_i) is taken, followed by a Euclidean projection
onto the simplex (proj_simplex).  The algorithm runs several random
restarts (the first one optionally warm‑started from the previously
found construction) and keeps the best sequence seen before the time
budget expires.

The implementation uses only NumPy and pure‑Python helpers – no
closures, no lambdas – and respects the 1000 s wall‑clock limit.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """Full linear convolution a*a using a real FFT."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _weighted_gradient(a: np.ndarray, weight: np.ndarray,
                       fft_len_corr: int, n: int) -> np.ndarray:
    """
    Compute 2·corr(weight, a) efficiently via FFT.
    `weight` has length (2·n‑1), `a` has length n.
    Returns a gradient vector of length n.
    """
    a_rev = a[::-1]
    W_hat = np.fft.rfft(weight, fft_len_corr)
    A_rev_hat = np.fft.rfft(a_rev, fft_len_corr)
    full = np.fft.irfft(W_hat * A_rev_hat, n=fft_len_corr)
    # The valid part of the correlation starts at index n‑1
    grad = 2.0 * full[n - 1:n - 1 + n]
    return grad


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:

    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The method is a soft‑max‑weighted exponentiated‑gradient
    descent on the probability simplex with several random restarts.
    The best vector found before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size (number of steps).  8192 is a comfortable upper
    # bound that still allows fast FFTs.
    # ------------------------------------------------------------------
    try:
        base_seq = globals()["height_sequence_1"]
        n0 = len(base_seq)
    except Exception:
        n0 = 4096
    n = min(8192, max(2048, n0))          # ensure a reasonable size

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                # for a*a
    fft_len_corr = _next_pow2(conv_len + n - 1)   # for weighted correlations

    # ------------------------------------------------------------------
    # Prepare a list of initial vectors (first one warm‑started from the
    # previous construction if it exists, the rest random Dirichlet draws).
    # ------------------------------------------------------------------
    init_vectors = []

    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.shape[0] > n:
            v = v[:n]
        elif v.shape[0] < n:
            v = np.concatenate([v, np.zeros(n - v.shape[0])])
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v = v / s
        else:
            v = np.full(n, 1.0 / n, dtype=np.float64)
        init_vectors.append(v)

    # a few pure Dirichlet draws (α = 1)
    for _ in range(5):
        init_vectors.append(rng.dirichlet(np.ones(n)))

    # ------------------------------------------------------------------
    # Time management
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + float(budget_s) - 1.0   # tiny safety margin

    best_val = np.inf

    # ------------------------------------------------------------------
    restart_id = 0

        # pick a starting point
        if restart_id < len(init_vectors):

        else:
