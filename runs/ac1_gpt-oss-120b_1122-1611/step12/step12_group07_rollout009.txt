"""
Propose a step‑function (non‑negative sequence) that minimises
    J(a) = 2·n·max(conv(a,a)) / (sum(a)²)

The algorithm has three stages:

1. **Fast parametric sweep** – generate a variety of simple analytic
   shapes (uniform, geometric, V‑shaped, end‑heavy, mixtures) and keep
   the best one.

2. **Sub‑gradient descent on the true max‑convolution** – from the best
   candidate we repeatedly:
      * compute the full self‑convolution by FFT,
      * locate the index k where the convolution attains its maximum,
      * form the exact sub‑gradient of the objective (g_i = a_{k‑i}),
      * take a diminishing‑step sub‑gradient update,
      * project back onto the probability simplex.

   The step size decays as 1/√t, guaranteeing convergence for a
   convex‑like objective while respecting the non‑negativity and sum‑one
   constraints.

3. **Return the best vector** (clipped to the limits imposed by
   `evaluate_sequence`).  The whole routine respects the 1000 s budget
   and prints occasional progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (FFT convolution, smooth‑max, simplex projection)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_full(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Full linear convolution a*a using real FFT.
    Returns length ``conv_len = 2*len(a)-1``.
    """
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return conv


def _smooth_max_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights for the smooth approximation of max(conv).
    Returns a vector w that sums to 1.
    """
    m = conv.max()
    shifted = beta * (conv - m)
    shifted = np.clip(shifted, -700, 0)           # safety
    exp_shifted = np.exp(shifted)
    sumexp = exp_shifted.sum()
    return exp_shifted / sumexp


def _grad_smooth_max(a: np.ndarray, w: np.ndarray, grad_fft_len: int) -> np.ndarray:
    """
    Gradient of smooth‑max(conv(a,a)) w.r.t. a.
    Implements 2 * (w * rev(a)) using FFT.
    """
    rev_a = a[::-1]
    W = np.fft.rfft(w, grad_fft_len)
    A_rev = np.fft.rfft(rev_a, grad_fft_len)
    full = np.fft.irfft(W * A_rev, n=grad_fft_len)
    n = a.shape[0]
    # slice that corresponds to the convolution result of length n
    return 2.0 * full[n - 1:n - 1 + n]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max() - 1          # zero‑based index
    theta = cssv[cond].max() / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """
    Exact objective J for a (sum(a) may differ from 1, we keep it general).
    """
    conv = _conv_full(a, fft_len, conv_len)
    max_conv = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_conv / (s * s))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size n
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(height_sequence_1)
            n = max(256, min(n, 8192))
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)                     # for a*a
    grad_fft_len = _next_pow2(3 * n - 2)               # for smooth‑max gradient

    # --------------------------------------------------------------
    # Helper to update best solution
    # --------------------------------------------------------------
    best_seq = None
    best_J = np.inf

    def _update_best(candidate: np.ndarray, Jval: float):
        nonlocal best_seq, best_J
        if Jval < best_J - 1e-12:
            best_J = Jval
            best_seq = candidate.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 1) Parametric sweep over simple families
    # --------------------------------------------------------------
    start_time = time.monotonic()
    idx = np.arange(n, dtype=float)

    # Uniform
    a = np.full(n, 1.0 / n, dtype=np.float64)
    _update_best(a, _objective(a, n, fft_len, conv_len))

    # Geometric decays
    for r in (0.90, 0.93, 0.95, 0.97, 0.99, 0.995):
        g = r ** idx
        g /= g.sum()
        _update_best(g, _objective(g, n, fft_len, conv_len))

    # V‑shaped product‑power family
    for p in np.linspace(0.1, 2.5, 25):
        v = 1.0 / ((idx + 1) * (n - idx)) ** p
        v = np.clip(v, 0, None)
        v /= v.sum()
        _update_best(v, _objective(v, n, fft_len, conv_len))

    # End‑heavy (sum of two power‑laws)
    for p in np.linspace(0.1, 2.5, 25):
        e = (idx + 1) ** (-p) + (n - idx) ** (-p)
        e = np.clip(e, 0, None)
        e /= e.sum()
        _update_best(e, _objective(e, n, fft_len, conv_len))

    # Mixture of uniform and V‑shape (p=1)
    v_base = 1.0 / ((idx + 1) * (n - idx))
    v_base = np.clip(v_base, 0, None)
    v_base /= v_base.sum()
    for lam in np.linspace(0.0, 1.0, 11):
        mix = lam * np.full(n, 1.0 / n) + (1.0 - lam) * v_base
        mix /= mix.sum()
        _update_best(mix, _objective(mix, n, fft_len, conv_len))

    # Random Dirichlet draws (various concentrations)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(15):
            d = rng.dirichlet(np.full(n, alpha))
            _update_best(d, _objective(d, n, fft_len, conv_len))

    # Include previous global solution if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            if prev.size > n:
                prev = prev[:n]
            else:
                prev = np.concatenate([prev, np.full(n - prev.size, 1.0 / n)])
        prev = np.clip(prev, 0, None)
        s = prev.sum()
        if s == 0:
            prev = np.full(n, 1.0 / n)
        else:
            prev /= s
        _update_best(prev, _objective(prev, n, fft_len, conv_len))

    # --------------------------------------------------------------
    # 2) Sub‑gradient descent on the true max‑convolution
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback to uniform
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_J = _objective(best_seq, n, fft_len, conv_len)

    a = best_seq.copy()
    cur_J = best_J
    global_best_seq = a.copy()
    global_best_J = cur_J
    last_improve = time.monotonic()

    # Sub‑gradient parameters
    step0 = 0.02          # initial step magnitude
    iter_cnt = 0

    deadline = start_time + max(0.0, float(budget_s) - 0.5)   # tiny safety margin

    while time.monotonic() < deadline:
        iter_cnt += 1
        # ----- convolution and argmax -----
        conv = _conv_full(a, fft_len, conv_len)
        k_max = int(np.argmax(conv))

        # ----- sub‑gradient of max‑conv w.r.t. a -----
        grad = np.zeros_like(a)
        lo = max(0, k_max - (n - 1))
        hi = min(n - 1, k_max)
        if lo <= hi:
            i_range = np.arange(lo, hi + 1, dtype=int)
            grad[i_range] = a[k_max - i_range]

        # derivative of J = 2*n*max_conv (sum(a)=1)
        grad *= 2.0 * n

        # ----- diminishing step size -----
        step = step0 / np.sqrt(iter_cnt)

        # ----- sub‑gradient update + simplex projection -----
        v = a - step * grad
        a_new = _proj_simplex(v)

        # ----- evaluate -----
        J_new = _objective(a_new, n, fft_len, conv_len)

        # ----- accept only improvements (greedy) -----
        if J_new < cur_J - 1e-12:
            a = a_new
            cur_J = J_new
            if J_new < global_best_J - 1e-12:
                global_best_J = J_new
                global_best_seq = a_new.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] Improved (sub‑grad) J = {global_best_J:.6f}")

        # occasional aggressive jitter if stagnation > 40 s
        now = time.monotonic()
        if now - last_improve > 40.0:
            jitter = rng.normal(0.0, 0.12, size=n)
            a = _proj_simplex(np.log(np.maximum(global_best_seq, 1e-12)) + jitter)
            cur_J = _objective(a, n, fft_len, conv_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Stagnation – jitter restart (J={cur_J:.6f})")

    # --------------------------------------------------------------
    # 3) Return the best sequence (clip to allowed range)
    # --------------------------------------------------------------
    best_vec = np.clip(global_best_seq, 0.0, 1000.0)
    total = best_vec.sum()
    if total == 0.0:
        best_vec = np.full_like(best_vec, 1.0 / n)
    else:
        best_vec = best_vec / total               # keep it a probability vector

    elapsed = time.monotonic() - start_time
    print(f"Finished after {elapsed:.2f}s. Best J = {global_best_J:.6f}")

    return [float(v) for v in best_vec]
