"""
Local search on convolution peak.

The algorithm works on a reduced problem size (≈1024 entries) and
optimises the shape of a probability vector by repeatedly moving a
fraction of mass away from the indices that contribute most to the
current maximal autocorrelation coefficient.  Acceptance follows a
simple simulated‑annealing schedule and a jitter operator is injected
periodically to escape plateaus.  After the allotted time the
optimised vector is interpolated to the required length (the length
of the previously found construction or a default 4096) and returned
as a list of floats.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    # ensure non‑negative for numerical stability
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _conv_max(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Return the maximal entry of the linear convolution a*a."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())

def _objective(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Objective  J = 2·n·max(conv(a,a))  (sum(a) assumed = 1)."""
    n = a.size
    max_b = _conv_max(a, fft_len, conv_len)
    return 2.0 * n * max_b

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function with a low autocorrelation peak.
    Uses a gradient‑guided local search combined with simulated annealing.
    """
    rng = np.random.default_rng(seed)

    # -----------------------------------------------------------------
    # 1. target length and optimisation length
    # -----------------------------------------------------------------
    if "height_sequence_1" in globals():
        target_n = len(height_sequence_1)
    else:
        target_n = 4096

    # use a modest size for the intensive optimisation phase
    # (small enough for fast FFTs, large enough to capture shape)
    opt_n = min(1024, target_n)   # a power of two is not required
    if opt_n < 256:
        opt_n = target_n

    conv_len = 2 * opt_n - 1
    fft_len = _next_pow2(conv_len)

    # -----------------------------------------------------------------
    # 2. initialise the search vector on the simplex
    # -----------------------------------------------------------------
    # start from a uniform distribution; if a previous construction exists,
    # initialise from an interpolated copy of it.
    a = np.full(opt_n, 1.0 / opt_n, dtype=np.float64)

    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        # interpolate to the optimisation size
        if v.size != opt_n:
            src_x = np.linspace(0.0, 1.0, v.size)
            tgt_x = np.linspace(0.0, 1.0, opt_n)
            v_interp = np.interp(tgt_x, src_x, v, left=0.0, right=0.0)
        else:
            v_interp = v.copy()
        v_interp = np.clip(v_interp, 0.0, None)
        if v_interp.sum() > 0:
            a = proj_simplex(v_interp)

    # keep a copy of the best solution seen so far
    best_a = a.copy()
    best_J = _objective(best_a, fft_len, conv_len)

    current_a = a.copy()
    current_J = best_J

    # -----------------------------------------------------------------
    # 3. main local‑search loop
    # -----------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a few seconds margin

    # schedule for the step size (fraction of donor mass that is moved)
    eps_start = 0.20
    eps_end   = 0.001

    # temperature schedule for occasional uphill moves
    T_start = max(1e-6, (best_J - 1.0) * 0.04)   # proportional to the current gap
    T_end   = 1e-8

    # stagnation handling
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # -------------------------------------------------------------
        # 3.1 current convolution and its maximal position
        # -------------------------------------------------------------
        A = np.fft.rfft(current_a, fft_len)

        k_max = int(np.argmax(conv))                 # index of the peak
        # -------------------------------------------------------------

            current_J = trial_J

            # keep the best ever seen
            # ---------------------------------------------------------
            if trial_J < best_J - 1e-12:
                best_J = trial_J
                best_a = trial.copy()
                last_improve = time.monotonic()
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

                current_J = best_J

"""
Local search for a step‑function with a low autocorrelation peak.

The algorithm works on a reduced problem size (≈1024 entries) and
optimises the shape of a probability vector by repeatedly moving a
fraction of mass away from the indices that contribute most to the
current maximal autocorrelation coefficient.  Acceptance follows a
simple simulated‑annealing schedule and a jitter operator is injected
periodically to escape plateaus.  After the allotted time the
optimised vector is interpolated to the required length (the length
of the previously found construction or a default 4096) and returned
as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]                       # sorted in descending order
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _conv_max(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Maximum entry of the linear convolution a*a."""
    A = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(conv.max())


def _objective(a: np.ndarray, fft_len: int, conv_len: int) -> float:
    """Objective J = 2·n·max(conv(a,a))  (∑a = 1 is assumed)."""
    n = a.size
    return 2.0 * n * _conv_max(a, fft_len, conv_len)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    Uses a gradient‑guided local search combined with simulated annealing.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Determine problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        target_n = len(height_sequence_1)
    else:
        target_n = 4096

    # a moderate size for the intensive iterative phase
    opt_n = min(1024, target_n)
    if opt_n < 256:
        opt_n = target_n                      # fallback if the problem is tiny

    conv_len = 2 * opt_n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 2. Initialise a feasible probability vector
    # ------------------------------------------------------------------
    a = np.full(opt_n, 1.0 / opt_n, dtype=np.float64)

    # use the previous construction as a warm start (interpolated)
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        if v.size != opt_n:
            src_x = np.linspace(0.0, 1.0, v.size)
            tgt_x = np.linspace(0.0, 1.0, opt_n)
            v_interp = np.interp(tgt_x, src_x, v, left=0.0, right=0.0)
        else:
            v_interp = v.copy()
        v_interp = np.clip(v_interp, 0.0, None)
        if v_interp.sum() > 0:
            a = proj_simplex(v_interp)

    best_a = a.copy()
    best_J = _objective(best_a, fft_len, conv_len)

    current_a = a.copy()
    current_J = best_J

    # ------------------------------------------------------------------
    # 3. Main local‑search loop
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)   # keep a short safety margin

    # step‑size schedule (fraction of donor mass that is moved)
    eps_start = 0.20
    eps_end   = 0.001

    # temperature schedule for occasional uphill moves
    T_start = max(1e-6, (best_J - 1.0) * 0.04)
    T_end   = 1e-8

    last_improve = start
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # a) current convolution and its peak position
        # --------------------------------------------------------------
        A = np.fft.rfft(current_a, fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        k_max = int(np.argmax(conv))

        # --------------------------------------------------------------
        # b) donor – index that contributes most to the peak
        # --------------------------------------------------------------
        if k_max < opt_n:
            i_vals = np.arange(k_max + 1)
            j_vals = k_max - i_vals
        else:
            i_vals = np.arange(k_max - opt_n + 1, opt_n)
            j_vals = k_max - i_vals
        contrib = current_a[i_vals] * current_a[j_vals]
        donor = int(i_vals[np.argmax(contrib)])

        # --------------------------------------------------------------
        # c) recipient – a low‑mass index (preferably far from the peak)
        # --------------------------------------------------------------
        low_cnt = max(1, opt_n // 20)                    # ~5 % of the positions
        low_idx = np.argsort(current_a)[:low_cnt]
        if low_idx[0] == donor and low_cnt > 1:
            recipient = int(low_idx[1])
        else:
            recipient = int(low_idx[0])

        # --------------------------------------------------------------
        # d) step size (fraction of donor mass to move)
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start
        t_frac = elapsed / budget_s
        eps = eps_start * (eps_end / eps_start) ** t_frac
        delta = current_a[donor] * eps
        if delta < 1e-15:
            # fall back to a tiny random jitter if the donor is almost empty
            noise = np.exp(rng.normal(0.0, 0.02, size=opt_n))
            cand = proj_simplex(current_a * noise)
            cand_J = _objective(cand, fft_len, conv_len)
            if cand_J < current_J:
                current_a = cand
                current_J = cand_J
            continue

        # --------------------------------------------------------------
        # e) construct trial vector
        # --------------------------------------------------------------
        trial = current_a.copy()
        trial[donor] -= delta
        trial[recipient] += delta
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(opt_n, 1.0 / opt_n)
        else:
            trial /= s

        trial_J = _objective(trial, fft_len, conv_len)

        # --------------------------------------------------------------
        # f) acceptance (simulated annealing)
        # --------------------------------------------------------------
        if trial_J < current_J:
            accept = True
        else:
            T = T_start * (T_end / T_start) ** t_frac
            prob = np.exp(-(trial_J - current_J) / (T + 1e-12))
            accept = rng.random() < prob

        if accept:
            current_a = trial
            current_J = trial_J

            # update global best if we improved
            if trial_J < best_J - 1e-12:
                best_J = trial_J
                best_a = trial.copy()
                last_improve = time.monotonic()
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # --------------------------------------------------------------
        # g) stagnation handling – occasional large jitter
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > 20.0:
            # a relatively strong perturbation around the best vector
            big_noise = np.exp(rng.normal(0.0, 0.07, size=opt_n))
            jitter = proj_simplex(best_a * big_noise)
            jitter_J = _objective(jitter, fft_len, conv_len)
            if jitter_J < best_J:
                best_J = jitter_J
                best_a = jitter.copy()
                current_a = jitter.copy()
                current_J = jitter_J
                print(f"[{time.monotonic() - start:6.2f}s] Jitter improved J = {best_J:.6f}")
            else:
                # fall back to the best known point
                current_a = best_a.copy()
                current_J = best_J
            last_improve = time.monotonic()

    # ------------------------------------------------------------------
    # 4. Fine‑grained random search around the best solution
    # ------------------------------------------------------------------
    fine_deadline = time.monotonic() + min(5.0, (deadline - start) * 0.05)
    while time.monotonic() < fine_deadline:
        noise = np.exp(rng.normal(0.0, 0.01, size=opt_n))
        cand = proj_simplex(best_a * noise)
        cand_J = _objective(cand, fft_len, conv_len)
        if cand_J < best_J:
            best_J = cand_J
            best_a = cand
            print(f"[Fine {time.monotonic() - start:6.2f}s] Fine improvement J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 5. Expand to the required target length (if different)
    # ------------------------------------------------------------------
    if target_n != opt_n:
        src_x = np.linspace(0.0, 1.0, opt_n)
        tgt_x = np.linspace(0.0, 1.0, target_n)
        expanded = np.interp(tgt_x, src_x, best_a, left=0.0, right=0.0)
    else:
        expanded = best_a

    # final safety checks (mirroring `evaluate_sequence` constraints)
    expanded = np.clip(expanded, 0.0, 1000.0)
    total = expanded.sum()
    if total < 0.01:
        expanded = np.full_like(expanded, 1.0 / expanded.size)
    else:
        expanded /= total

    return [float(v) for v in expanded]
