    iteration = 0

        iteration += 1

        score = 2 * n * M
        if score < best_score:
            best = a.copy()
            best_score = score
            print(f"[{iteration}] new best score {best_score:.6f}")

        # use tolerance
        eps = max(1e-12, M * 1e-12)
        active = np.where(b >= M - eps)[0]
        if len(active) <= 20:
            # Build LP
            # build matrices
            A_eq = np.ones((1, n+1))
            A_eq[0, -1] = 0.0 # δ not included
            b_eq = np.array([0.0])

            rows = []
            ub = []
            for k in active:

                i_start = max(0, k - (n-1))
                i_end = min(n-1, k)  # inclusive

            if rows:

                b_ub = np.zeros(len(rows))
                # objective

                if res.success and res.x[-1] > 1e-12:
                    delta = res.x[:n]
                    # try full step
                    cand = a + delta

                                best_score = new_score

        # Random perturbation fallback

        move_type = rng.random()

            best = cand.copy()
            best_score = cand_score
            print(f"[{iteration}] random improvement {best_score:.6f}")

        # optional: adapt random step parameters such as sigma that decays over time
        # not needed
    print(f"[final] best score {best_score:.6f} after {iteration} iterations")
    return best.tolist()

"""
Propose a step‑height sequence with a low evaluation score.

The algorithm starts from the classic “alpha” sequence (a_i = C(2i,i)/4^i),
which already yields a value ≈ π/2.  It then iteratively reduces the worst
convolution entry by solving a small linear program that gives a guaranteed
first‑order decrease of all currently maximal shifts (the active set).  The
LP provides a direction Δa that respects the simplex constraints
(a_i ≥ 0, Σa_i = 1).  A line‑search is performed on this direction; if an
actual improvement is observed the new point is accepted.  When the LP
does not succeed or the step does not improve the score, cheap stochastic
moves (multiplicative Gaussian noise, mass‑transfer and block‑scaling) are
applied to keep the search alive.

The whole routine runs until the time budget (≈ 1000 s) expires and returns
the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """
    First n terms of a_i = C(2i,i) / 4^i.
    Recurrence: a_{i+1} = a_i * (2*i+1) / (2*(i+1)).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.  Standard O(n log n) algorithm.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # find rho
    rho = np.where(u - cssv / ind > 0)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def evaluate(a: np.ndarray) -> float:
    """Convenient wrapper for the provided evaluate_sequence."""
    return float(evaluate_sequence(a.tolist()))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a sequence of non‑negative heights that minimises the
    supplied evaluation function.  The method combines a linear‑program
    based sub‑gradient step with occasional random perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise sequence (alpha‑seed or the global construction if present)
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(
        height_sequence_1, (list, np.ndarray)
    ) and len(height_sequence_1) > 0:
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
        n = a.shape[0]
        # make sure we start from a feasible point
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-12:
            a = generate_alpha_sequence(n)
    else:
        # default size – large enough to be close to the asymptotic bound,
        # small enough to keep each iteration cheap
        n = 3000
        a = generate_alpha_sequence(n)

    # Normalise to an L1‑norm of 1 (the objective is scale invariant)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = generate_alpha_sequence(n)
        a_sum = a.sum()
    a = a / a_sum

    best = a.copy()
    best_score = evaluate(best)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Search parameters
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    # probability with which we try a random perturbation instead of the LP
    prob_random = 0.15
    # fallback Gaussian noise magnitude
    sigma = 0.08

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Current objective and active set
        # --------------------------------------------------------------
        conv = np.convolve(a, a)
        max_conv = float(conv.max())
        cur_score = 2.0 * n * max_conv

        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            print(f"[{iteration:6d}] new best  = {best_score:.6f}")

        # We keep a small active set (up to 8 indices) to keep the LP cheap.
        eps_active = max(1e-12, max_conv * 1e-12)
        active_idxs = np.where(conv >= max_conv - eps_active)[0]
        if active_idxs.size > 8:
            # take the 8 largest entries only
            active_idxs = np.argpartition(-conv, np.arange(8))[:8]

        # flat regions and avoids costly LP solves when the LP gives no

            if move_type < 0.5:

                    cand = a.copy()
                    cand[i] -= delta
                    cand[j] += delta
                else:
                    cand = a.copy()
            else:

                start = int(rng.integers(0, max(1, n - width + 1)))
                stop = min(n, start + width)

                cand[start:stop] *= factor

            cand = np.clip(cand, 0.0, None)

            cand_score = evaluate(cand)
            if cand_score < best_score:
                a = cand
                best = cand.copy()
                best_score = cand_score

        # Linear‑programming based sub‑gradient step.
        # --------------------------------------------------------------
        # Build LP:
        #   vars: Δa_0 … Δa_{n-1},   δ  (δ ≥ 0)
        #   max  δ
        #   s.t. Σ Δa_i = 0
        #        for every active k:
        #            Σ a_{k-i} Δa_i + δ ≤ 0   (first‑order decrease of b_k)
        #        Δa_i ≥ -a_i

        # Equality: sum Δa_i = 0
        A_eq = np.zeros((1, n + 1), dtype=np.float64)
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

        rows = []
        for k in active_idxs:
            row = np.zeros(n + 1, dtype=np.float64)
            i_start = max(0, k - (n - 1))
            i_end = min(n - 1, k)  # inclusive
            idx = np.arange(i_start, i_end + 1, dtype=int)
            # coefficient for Δa_i is a_{k-i}
            row[idx] = a[k - idx]
            # coefficient for δ
            row[-1] = 1.0
            rows.append(row)
        if rows:
            A_ub = np.vstack(rows)
            b_ub = np.zeros(A_ub.shape[0], dtype=np.float64)

            # Objective (minimise –δ)
            c = np.concatenate([np.zeros(n), np.array([-1.0])])

            # Bounds: Δa_i ≥ -a_i, no explicit upper bound; δ ≥ 0
            bounds = [(-a[i], None) for i in range(n)] + [(0.0, None)]

            try:
                res = linprog(
                    c,
                    A_ub=A_ub,
                    b_ub=b_ub,
                    A_eq=A_eq,
                    b_eq=b_eq,
                    bounds=bounds,
                    method="highs",
                    options={"presolve": True},
                )
                if res.success and res.x[-1] > 1e-12:
                    delta = res.x[:n]          # direction Δa
                    # try the full step first
                    cand = a + delta
                    cand = np.clip(cand, 0.0, None)
                    cand = project_onto_simplex(cand)

                    cand_score = evaluate(cand)
                    if cand_score < best_score:
                        a = cand
                        best = cand.copy()
                        best_score = cand_score
                        print(
                            f"[{iteration:6d}] LP full step improvement = {best_score:.6f}"
                        )
                        continue

                    # If the full step was too aggressive, back‑track
                    for factor in (0.5, 0.25, 0.1):
                        cand = a + factor * delta
                        cand = np.clip(cand, 0.0, None)
                        cand = project_onto_simplex(cand)
                        cand_score = evaluate(cand)
                        if cand_score < best_score:
                            a = cand
                            best = cand.copy()
                            best_score = cand_score
                            print(
                                f"[{iteration:6d}] LP scaled ({factor:.2f}) improvement = {best_score:.6f}"
                            )
                            break
                # If LP solves but gives no improvement, fall back to random move
            except Exception as exc:
                # LP failed – ignore and continue with the next iteration
                pass

        # --------------------------------------------------------------
        # Occasionally adapt the stochastic parameters (simulated annealing)
        # --------------------------------------------------------------
        if iteration % 2000 == 0:
            sigma *= 0.95
            prob_random = min(0.4, prob_random * 1.02)

    print(f"[final] iterations={iteration}, best score={best_score:.6f}")
    return best.tolist()
