"""
Simple simulated‑annealing search for a non‑negative step function that
minimises 2·n·max(a∗a)/(∑a)².  The algorithm starts from the classical
α‑sequence (a_i = C(2i,i)/4^i) normalised to L¹‑norm = 1 and then performs
a mixture of random mass‑transfer moves, local block scaling, multiplicative
noise and cheap gradient steps.  An annealing schedule allows occasional
up‑hill moves, helping the search escape the flat plateau where the α‑
sequence lives.  Convolution maxima are obtained with an FFT‑based O(n log n)
routine, which makes it possible to evaluate tens of thousands of candidates
within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """
    First n terms of a_i = C(2i,i)/4^i.
    The recurrence a_{i+1}=a_i·(2*i+1)/(2*(i+1)) is used.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def conv_max_fft(a: np.ndarray) -> float:
    """
    Linear convolution a∗a via zero‑padded FFT and return its maximum.
    """
    n = a.shape[0]
    # size >= 2n‑1, power of two for speed
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return float(conv.max())


def objective(a: np.ndarray) -> float:
    """
    Return the exact value of evaluate_sequence for a (up to the
    clamping performed inside evaluate_sequence, which is harmless here).
    We keep the L¹‑norm of a equal to 1, therefore the objective reduces to
    2·n·max(a∗a).
    """
    n = a.shape[0]
    return 2.0 * n * conv_max_fft(a)


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a sequence of non‑negative heights yielding a low evaluation
    score.  The function respects the 1000 s time budget, returns the best
    sequence found (as a plain Python list) and prints occasional progress
    information.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size
    # ------------------------------------------------------------------
    global height_sequence_1
    if isinstance(height_sequence_1, (list, np.ndarray)) and len(height_sequence_1) > 0:
        n = int(len(height_sequence_1))
    else:
        n = 6000                        # a reasonable default

    # ------------------------------------------------------------------
    # Initialise with the (scaled) α‑sequence and normalise to L¹ = 1
    # ------------------------------------------------------------------
    a0 = generate_alpha_sequence(n)
    cur = a0 / a0.sum()                # L¹‑norm = 1
    best = cur.copy()
    cur_score = best_score = objective(cur)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing schedule
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # reserve a few seconds
    it = 0
    init_temp = 0.20
    final_temp = 1e-5

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Temperature (exponential cooling)
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)
        temp = init_temp * (final_temp / init_temp) ** progress

        # --------------------------------------------------------------
        # Pick a move type
        # --------------------------------------------------------------
        move = rng.random()
        cand = cur.copy()

        if move < 0.30:
            # ---- multiplicative log‑normal noise (global) -------------
            sigma = 0.12 * (1.0 - progress) + 0.02   # slowly shrinking
            cand *= np.exp(sigma * rng.standard_normal(n))

        elif move < 0.55:
            # ---- small mass transfer between two positions -------------
            # pick source proportional to current mass
            src = rng.choice(n, p=cur / cur.sum())
            dst = rng.integers(0, n)
            if src != dst:
                frac = rng.uniform(0.0, 0.5)        # up to 50 % transfer
                delta = cur[src] * frac
                cand[src] -= delta
                cand[dst] += delta

        elif move < 0.75:
            # ---- block scaling (local stretch/shrink) -----------------
            width = int(rng.integers(max(1, n // 250), max(2, n // 25)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.08))
            cand[start : start + width] *= factor

        elif move < 0.90:
            # ---- soft‑gradient step using a soft‑max over all b_k -----
            # compute full convolution once (needed for the gradient)
            nlen = cur.shape[0]
            size_fft = 1 << ((2 * nlen - 1).bit_length())
            fa = np.fft.rfft(cur, size_fft)
            conv_full = np.fft.irfft(fa * fa, size_fft)[: 2 * nlen - 1]

            # soft‑max weights – start gentle, become sharper over time
            beta = 5.0 + 30.0 * progress
            w = np.exp(beta * (conv_full - conv_full.max()))
            w_sum = w.sum()
            if w_sum == 0.0:
                w = np.ones_like(w) / w.size
            else:
                w /= w_sum

            # gradient = 2 * (w * a) convolved with reversed a
            # we compute via FFT again (w length 2n‑1, a_rev length n)
            a_rev = cur[::-1]
            m = 1 << ((len(w) + nlen - 1).bit_length())
            fw = np.fft.rfft(w, m)
            fa_rev = np.fft.rfft(a_rev, m)
            grad_full = np.fft.irfft(fw * fa_rev, m)[: len(w) + nlen - 1]
            grad = 2.0 * grad_full[nlen - 1 : nlen - 1 + nlen]  # slice back to length n
            step = 0.004 * (1.0 - progress)  # shrinking step size
            cand = cur - step * grad

        else:

            # (used rarely – resembles the move in the reference code)
            nlen = cur.shape[0]
            size_fft = 1 << ((2 * nlen - 1).bit_length())
            fa = np.fft.rfft(cur, size_fft)
            conv = np.fft.irfft(fa * fa, size_fft)[: 2 * nlen - 1]
            k_max = int(np.argmax(conv))
            # derivative of b_kmax wrt a_i is 2*a_{k_max-i}
            idx = np.arange(nlen)
            j = k_max - idx
            mask = (j >= 0) & (j < nlen)
            grad = np.zeros_like(cur)
            grad[mask] = 2.0 * cur[j[mask]]
            step = 0.006 * (1.0 - progress)
            cand = cur - step * grad

        # --------------------------------------------------------------
        # Project back onto the feasible set (non‑negative, L¹ = 1)
        # --------------------------------------------------------------
        cand = np.clip(cand, 0.0, None)
        total = cand.sum()
        if total < 1e-12:
            continue
        cand /= total

        # --------------------------------------------------------------
        # Evaluate candidate (fast objective)
        # --------------------------------------------------------------
        cand_score = objective(cand)

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance test
        # --------------------------------------------------------------
        if cand_score < cur_score or rng.random() < np.exp((cur_score - cand_score) / max(temp, 1e-12)):
            cur = cand
            cur_score = cand_score

        # --------------------------------------------------------------
        # Keep the overall best
        # --------------------------------------------------------------
        if cand_score < best_score:
            best = cand
            best_score = cand_score
            print(f"[{it:6d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Periodic “re‑heat” from the current best to keep diversity
        # --------------------------------------------------------------
        if it % 5000 == 0:
            cur = best * (1.0 + 0.25 * rng.standard_normal(n))
            cur = np.clip(cur, 0.0, None)
            cur /= cur.sum()
            cur_score = objective(cur)

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    # Return as a plain Python list – the external harness will call
    # evaluate_sequence on it.
    return best.tolist()
