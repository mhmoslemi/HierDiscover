        frac = elapsed / time_budget

def convolve_self(a):

    np.maximum(conv, 0.0, out=conv)

def maximal_conv_gradient(a, k):
    """Return gradient d/d a of the convolution term at index k."""
    n = a.shape[0]
    grad = np.zeros_like(a)
    # i ranges where both i and k-i are valid indices
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    # vector of indices i
    idx = np.arange(i_start, i_end + 1, dtype=np.int64)
    # corresponding partner indices
    partner = k - idx
    grad[idx] = 2.0 * a[partner]
    return grad

# Helper: project onto the non‑negative simplex (L1 = 1)
# ----------------------------------------------------------------------
def project_simplex(v):
    """Project vector v onto {x ≥ 0, sum x = 1}."""
    # ensure non‑negative first to avoid numerical issues
    v = np.maximum(v, 0.0)
    total = v.sum()
    if total <= 0.0:
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    return v / total

def improve_sequence(initial, time_budget, rng):
    """
    Perform a time‑bounded local optimisation starting from `initial`.
    Returns the best sequence found and its score.
    """
    n = initial.shape[0]
    # start from a copy that is already normalised to L1 = 1
    seq = project_simplex(initial.astype(np.float64, copy=False))
    best_seq = seq.copy()
    best_score = _evaluate(best_seq)

    start = time.monotonic()
    # parameters for learning‑rate schedule
    lr_start = 0.5          # relatively aggressive at the beginning
    lr_end = 0.005          # small steps near the end

    # main loop
    while True:
        now = time.monotonic()
        if now - start >= time_budget:
            break
        # fraction of elapsed time (0 … 1)
        frac = (now - start) / time_budget
        # linearly decay learning rate
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # convolution and maximal index
        conv = convolve_self(seq)
        max_val = conv.max()
        # take all indices whose value is within 1 % of the maximum
        # (this gives a smoother descent when several k are close)
        thresh = max_val * 0.99
        max_indices = np.where(conv >= thresh)[0]

        # accumulate gradient over the chosen indices
        grad = np.zeros_like(seq)
        for k in max_indices:
            grad += maximal_conv_gradient(seq, k)

        # project gradient to keep the L1‑norm unchanged:
        # we need a direction d with sum(d)=0.
        # The orthogonal projection of -grad onto the hyperplane sum=0
        # is: d = -grad + mean(grad) * 1
        mean_grad = grad.mean()
        direction = -grad + mean_grad

        # stochastic component – occasional jitter
        if rng.random() < 0.15:
            jitter = np.exp(rng.normal(0.0, 0.03, size=n))
            direction *= jitter

        # proposed step
        new_seq = seq + lr * direction

        # enforce non‑negativity and re‑normalise to the simplex
        new_seq = project_simplex(new_seq)

        # small global log‑normal scaling – does not change the score
        # but helps to shake the system out of flat regions
        if rng.random() < 0.08:
            global_factor = float(np.exp(rng.normal(0.0, 0.02)))
            new_seq = project_simplex(new_seq * global_factor)

        # occasional block perturbation (exploratory move)
        if rng.random() < 0.05:
            block_width = int(rng.integers(max(1, n // 150), max(2, n // 30)))
            start_idx = int(rng.integers(0, max(1, n - block_width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            new_seq[start_idx:start_idx + block_width] *= factor
            new_seq = project_simplex(new_seq)

        # evaluate
        score = _evaluate(new_seq)

        # keep the best ever seen
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = new_seq.copy()
            print(f"[{now - start:6.2f}s] New best score {best_score:.6f}")

        # accept or reject the move (simulated‑annealing style)
        # we use a simple temperature that decays to zero.
        temperature = max(1e-4, (1.0 - frac) * 0.2)
        if score <= best_score:
            seq = new_seq
        else:
            # uphill move with probability exp(-(Δ)/T)
            if rng.random() < math.exp(-(score - best_score) / temperature):
                seq = new_seq
        # loop continues
    return best_seq, best_score
