"""
A gradient‑based optimiser for the “max‑convolution / (sum)²” objective.

The idea
--------
For a sequence a ≥ 0 the evaluation score is

    score(a) = 2·n·max(a*a) / (∑a)² ,

where “*” denotes the linear (aperiodic) convolution.
The average of the convolution is (∑a)²/(2n‑1), so the ideal case would be
a perfectly flat convolution and a score of 2·n/(2n‑1)≈1.  The algorithm
minimises the L²‑distance of the convolution to this flat target; the
gradient of that loss can be written as a simple convolution, allowing a
fast O(n log n) implementation via FFTs.  After each gradient step the
vector is projected onto the non‑negative orthant and re‑normalised
(the score is scale‑invariant, so fixing ∑a=1 is harmless).

To avoid being trapped in local minima the optimiser occasionally applies
small random jitter and a “block‑perturbation’’ (the same cheap move used
by the reference solution).  The best sequence seen during the run is
returned.

The whole routine respects the 1000 s time budget, prints progress and
needs only NumPy (no external LP/GP solvers).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution using an FFT of length a power of two.
def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the linear convolution x*y (real FFT, O(n log n))."""
    m = x.size + y.size - 1
    size = 1 << ((m - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:m]
    return conv


# ----------------------------------------------------------------------
# Score according to the evaluator (identical formula, no clipping needed here).
def _score(a: np.ndarray) -> float:
    """Compute 2·n·max(a*a)/(∑a)² for a non‑negative vector a."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _conv_fft(a, a)
    return float(2.0 * n * b.max() / (s * s))


# ----------------------------------------------------------------------
# Gradient of the L2‑error between the convolution and the flat target.
def _grad_flatten(a: np.ndarray) -> (np.ndarray, np.ndarray):
    """
    Return (grad, conv) where

        grad = ∂/∂a ∑_k ( (a*a)[k] - target )²

    target = (∑a)²/(2n‑1)  (the constant value of a perfectly flat convolution).
    """
    n = a.size
    s = a.sum()
    target = (s * s) / (2 * n - 1)               # constant that the convolution should have
    conv = _conv_fft(a, a)                       # shape (2n‑1,)
    err = conv - target                          # deviation from the flat target
    # Gradient = 2 * a * err (convolution with the reversed error vector)
    err_rev = err[::-1]
    full = _conv_fft(a, err_rev)                 # length 3n‑2
    idx = (2 * n - 2) - np.arange(n)             # pick the part that matches ∂/∂a_i
    grad = 2.0 * full[idx]
    return grad, conv


# ----------------------------------------------------------------------
# Cheap block perturbation (taken from the reference solution).
def _make_block_perturbation(base: np.ndarray,
                            rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:                      # occasional mirroring
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a low‑evaluation step function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget (seconds).  The function stops a few seconds
        before the limit to avoid forced termination.

    Returns
    -------
    list[float]
        The best height sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size.
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                base_seq = arr.copy()
            else:
                base_seq = None
        except Exception:
            base_seq = None
    else:
        base_seq = None

    if base_seq is None:
        # fallback – a modest‑size decreasing sequence (sqrt decay)
        n = 8192                     # a power of two works nicely with FFTs
        base_seq = 1.0 / np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    else:
        n = base_seq.size

    # ------------------------------------------------------------------
    # Normalise the initial vector (the score is scale‑invariant).
    a = np.clip(base_seq.astype(np.float64), 0.0, None)
    if a.sum() == 0:
        a = rng.random(n) + 1e-12
    a = a / a.sum()

    best = a.copy()
    best_score = _score(best)
    print(f"[init] length={n}, start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters (tuned empirically but safe for the time budget).
    lr = 0.10                     # base step size (applied after normalising gradient)
    jitter_sigma = 0.02          # std. dev. of multiplicative jitter (log‑normal)
    jitter_interval = 50        # every ? iterations
    block_interval = 500        # every ? iterations do a block move
    max_iters = 10_000_000       # hard cap – the loop will stop earlier by the time check

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # keep a safety margin

    while time.monotonic() < deadline and it < max_iters:
        # --- gradient step toward a flatter convolution -----------------
        grad, conv = _grad_flatten(a)
        # Normalise the gradient direction; the update size is controlled by lr.
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:

        # Enforce constraints: non‑negative and re‑normalise to keep sum=1.
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0:

            a = rng.random(n) + 1e-12
            s = a.sum()
        a = a / s

        # Occasionally add a tiny random jitter (helps escaping plateaus).
        if it % jitter_interval == 0:
            jitter = np.exp(rng.normal(0.0, jitter_sigma, size=n))
            a *= jitter
            a = a / a.sum()

        # Occasionally perform a cheap block perturbation (non‑gradient move).
        if it % block_interval == 0 and it > 0:
            a = _make_block_perturbation(a, rng)
            a = a / a.sum()                     # keep the sum normalised

        # --- evaluate current score ------------------------------------
        cur_score = _score(a)
        if cur_score < best_score:
            best = a.copy()
            best_score = cur_score
            print(f"[iter {it:08d}] new best {best_score:.6f}")

        # Very occasional “reset to best” if we appear to drift away.
        if it % 2000 == 0 and cur_score > best_score * 1.02:
            a = best.copy()

        it += 1

    # ------------------------------------------------------------------
    print(f"[done] best_score={best_score:.6f} after {it} iterations")
    return [float(x) for x in best]
