def smooth_step(seq, sigma):
    # Apply Gaussian smoothing by convolution with Gaussian kernel of size approx 2*3*sigma
    from math import ceil, exp
    sigma = max(0.1, sigma)
    w = int(ceil(sigma * 6))
    if w % 2 == 0:
        w += 1
    xs = np.arange(w) - w//2
    kernel = np.exp(-0.5 * (xs / sigma) ** 2)
    kernel /= kernel.sum()
    new = np.convolve(seq, kernel, mode='same')
    # ensure nonnegative
    new = np.clip(new, 0, None)
    # renorm
    s = new.sum()
    new = new / s
    return new

propose_candidate
-----------------
Search for a step‐function (non‑negative sequence) that minimises

    evaluate_sequence(seq)

The algorithm consists of three stages:

1. **Deterministic families** – a dense sweep over several analytic
   families (power law, shifted power law, symmetric power law and a
   “inverse‑sine” family).  This quickly finds a good baseline; the
   families are parameterised by only two numbers (exponent and shift)
   so we can afford a fine grid.

2. **Local refinement** – the current best sequence is repeatedly
   improved by cheap, gradient‑inspired updates, Gaussian smoothing and
   the original block‑perturbation operator.  The gradient step is
   derived from the exact derivative of the objective
   ``F(a)=2·n·max(conv(a,a))/sum(a)^2`` and moves mass away from those
   entries that directly contribute to the maximal convolution entry.
   All updates keep the sequence on the probability simplex (sum=1) so
   the objective is scale‑invariant.

3. **Time‑bounded loop** – the while‑loop runs until the supplied budget
   (≈1000 s) expires, keeping the best sequence ever seen.

The whole routine respects the constraints of ``evaluate_sequence``:
non‑negative entries, each ≤1000, and a total sum ≥ 0.01.  The final
list is returned as plain Python floats.

def _clip_and_normalise(arr: np.ndarray) -> np.ndarray:
    """
    Scale the array so that the maximum does not exceed the hard limit
    of 1000 (required by evaluate_sequence) and normalise the sum to 1.
    """
    if arr.size == 0:
        return arr
    # clip to the allowed range first
    arr = np.clip(arr, 0.0, 1000.0)
    total = arr.sum()
    if total < 1e-12:                     # avoid division by zero
        arr = np.ones_like(arr) / arr.size
        total = 1.0
    arr = arr / total                     # sum == 1  (scale invariant)
    return arr


def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift) ** beta   (beta may be negative)."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return _clip_and_normalise(seq)


def generate_power_symmetric(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i+shift)^beta + (n-1-i+shift)^beta  (mirrored power law)."""
    i = np.arange(n, dtype=np.float64) + shift
    rev = (n - 1 - np.arange(n, dtype=np.float64) + shift)
    seq = i ** beta + rev ** beta
    return _clip_and_normalise(seq)


def generate_product_power(beta: float, shift: float, n: int) -> np.ndarray:
    """
    a_i = ((i+shift)*(n-1-i+shift)) ** beta.
    For beta<0 this diverges at the two ends; the result is
    normalised to sum = 1.
    """
    i = np.arange(n, dtype=np.float64) + shift
    j = (n - 1 - np.arange(n, dtype=np.float64) + shift)
    prod = i * j
    # avoid exactly zero – shift guarantees positivity
    seq = prod ** beta
    return _clip_and_normalise(seq)


def generate_inverse_sine(gamma: float, n: int) -> np.ndarray:
    """
    a_i = sin(π·(i+0.5)/n) ** (-gamma)   (gamma > 0).
    This mimics the arcsine density – large near the borders,
    small in the centre.
    """
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** (-gamma)
    return _clip_and_normalise(seq)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative block tweak, used in the original baseline."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width between 0.5 % and ~8 % of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    w = int(rng.integers(lo, hi + 1))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))        # log‑normal tweak
    cand[st:en] *= factor
    # occasional symmetrisation (helps for symmetric shapes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return _clip_and_normalise(cand)


def gradient_step(seq: np.ndarray, eta: float = 0.10) -> np.ndarray:
    """
    One explicit gradient step for the scale‑invariant objective.
    The gradient of 2·n·max(conv) w.r.t. a_i is 4·n·a_{k*-i},
    where k* is the index of the maximal convolution entry.
    We move a_i opposite to that direction, stay on the simplex,
    and re‑normalise.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k_star = int(np.argmax(conv))

    idx = np.arange(n, dtype=np.int64)
    partner = k_star - idx
    mask = (partner >= 0) & (partner < n)
    # partner values (zero where out of bounds)
    partner_vals = np.zeros_like(seq)
    partner_vals[mask] = seq[partner[mask]]

    new = seq - eta * partner_vals
    new = np.clip(new, 0.0, None)        # non‑negative
    return _clip_and_normalise(new)


def smooth_step(seq: np.ndarray, sigma: float) -> np.ndarray:
    """
    Gaussian smoothing of the sequence (mode='same'), followed by
    re‑normalisation.  ``sigma`` is the standard deviation in index units.
    """
    n = seq.size
    sigma = max(0.1, sigma)
    # kernel size ≈ 6σ (odd)
    w = int(np.ceil(sigma * 6))
    if w % 2 == 0:
        w += 1
    x = np.arange(w) - w // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(seq, kernel, mode='same')
    return _clip_and_normalise(smoothed)
