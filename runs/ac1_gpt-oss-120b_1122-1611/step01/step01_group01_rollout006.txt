"""
Propose a step‑function that minimizes
    score = 2 * n * max(a * a) / (sum(a)**2)

The algorithm is a mirror‑descent (exponentiated‑gradient) optimiser
augmented with a few heuristics that proved useful in practice:

*   The objective is scale‑invariant, therefore we keep the vector
    normalised to unit sum and work directly with the *max* of the
    self‑convolution.
*   At each iteration we compute the self‑convolution via FFT,
    take the *top‑k* largest entries (k≈15) and form a weighted sub‑gradient
    by averaging the partner vectors of those indices.
*   The update is `a_i ← a_i * exp( -η * g_i )` (mirror descent on the
    simplex).  The learning rate η decays linearly during the run.
*   Occasionally a random block‑scaling perturbation is injected to
    escape shallow local minima.
*   Every few hundred iterations a light Gaussian smoothing of the
    distribution is applied – this spreads the mass and helps to
    lower the convolution peak.
*   The best vector seen so far (according to the fast score
    `2 * n * max(conv)`) is kept and finally returned.

The routine respects the 1000‑second budget, checks the remaining time
each loop iteration and stops just before the deadline.
All values are clipped to the bounds required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Return the next power‑of‑two length that can hold a full linear
    convolution of two vectors of length n.
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a * a using a real FFT.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa                     # element‑wise product = convolution in frequency domain
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of (a*a)[k] with respect to a:
        w_i = a_{k-i}  (zero when index out of range).
    """
    n = a.size
    w = np.zeros_like(a)

    if k < n:
        # overlap on the left side
        w[: k + 1] = a[k :: -1]          # a[k], a[k‑1], …, a[0]
    else:
        # overlap on the right side
        start = k - (n - 1)               # first i with a partner
        length = (2 * n - 1) - k          # how many overlapping entries
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative perturbation of a random contiguous block,
    followed by re‑normalisation.
    """
    n = a.size
    # block width between 1 and n//5 (but at least 1)
    width = rng.integers(1, max(2, n // 5))
    start = rng.integers(0, n - width + 1)
    # log‑normal scaling factor (σ≈0.2 gives moderate changes)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _smooth(a: np.ndarray, sigma: float = 1.0) -> np.ndarray:
    """
    Apply a short Gaussian blur (zero‑padding, then renormalise).
    """
    if sigma <= 0:
        return a
    radius = int(np.ceil(3 * sigma))
    xs = np.arange(-radius, radius + 1, dtype=np.float64)
    kernel = np.exp(-0.5 * (xs / sigma) ** 2)
    kernel /= kernel.sum()
    sm = np.convolve(a, kernel, mode="same")
    sm = np.clip(sm, 0.0, None)
    s = sm.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.size)
    return sm / s


def _fast_score(a: np.ndarray, conv: np.ndarray) -> float:
    """
    Fast evaluation of the objective for a unit‑sum vector a.
    """
    n = a.size
    return float(2.0 * n * np.max(conv))

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The routine runs for roughly ``budget_s`` seconds
    (leaving a safety margin) and returns the best vector found.
    """
    rng = np.random.default_rng(seed)
    # leave a 2‑second safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise the candidate distribution
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # fallback: modest‑size uniform vector
        base = np.ones(2048, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, None)
    tot = a.sum()
    if tot <= 0.0:
        a[:] = 1.0 / n
    else:
        a /= tot                                 # unit sum → scale‑invariant

    # pre‑allocate parameters
    fft_len = _fft_len_for_n(n)
    best_a = a.copy()
    conv = _conv_self(best_a, fft_len)
    best_score = _fast_score(best_a, conv)

    # momentum term for the mirrored‑gradient (initialized to zero)
    momentum = np.zeros_like(a)

    start_time = time.monotonic()
    iteration = 0
    smoothing_interval = 250          # apply Gaussian smoothing every N iterations
    block_perturb_prob = 0.012        # probability of a random block perturb

    # ------------------------------------------------------------------
    # optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # 1. self‑convolution and current peak
        conv = _conv_self(a, fft_len)
        max_val = float(conv.max())

        # 2. build a weighted sub‑gradient from the top‑k convolution entries
        top_k = min(20, conv.size)                 # keep the active set modest
        # indices of the largest values (unsorted, O(n) via argpartition)
        top_idx = np.argpartition(-conv, top_k - 1)[:top_k]
        # weights proportional to the convolution values (more emphasis on higher peaks)
        wts = conv[top_idx]
        weight_sum = wts.sum()
        if weight_sum == 0.0:
            # degenerate case – fall back to uniform weighting
            wts = np.ones_like(wts) / top_k
        else:
            wts = wts / weight_sum

        # weighted sum of partner vectors
        grad = np.zeros_like(a)
        for idx, wk in zip(top_idx, wts):
            grad += wk * _partner_vector(a, int(idx))

        # 3. momentum (exponential moving average) – helps to stabilise updates
        beta = 0.9
        momentum = beta * momentum + (1.0 - beta) * grad

        # 4. learning‑rate schedule (linear decay)
        elapsed = time.monotonic() - start_time
        total_time = deadline - start_time
        progress = min(1.0, elapsed / total_time)
        eta0 = 1.2
        eta_min = 0.04
        eta = eta0 * (1.0 - progress) + eta_min * progress

        # 5. mirror‑descent (exponentiated‑gradient) step
        #    use the momentum as the descent direction
        step = -eta * momentum
        # clamp exponent to avoid overflow/underflow
        step = np.clip(step, -50.0, 50.0)
        a = a * np.exp(step)

        # 6. renormalise to the simplex
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # 7. occasional stochastic exploration (random block scaling)
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # 8. occasional smoothing – a gentle Gaussian blur
        if iteration % smoothing_interval == 0 and iteration > 0:
            sigma = 2.0 * (1.0 - progress) + 0.5   # start wide, end narrow
            a = _smooth(a, sigma=sigma)

        # 9. evaluate current score (fast version)
        cur_score = 2.0 * n * max_val

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            elapsed_now = time.monotonic() - start_time
            print(f"[{elapsed_now:6.2f}s] new best score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best vector (clipped to the limits enforced by `evaluate_sequence`)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
