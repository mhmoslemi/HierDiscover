def simplex_project(v):
    """Projects vector v onto the probability simplex (sum = 1, v >=0)."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Sub‑gradient optimisation of the peak of the self‑convolution.

The evaluation function is homogeneous:
    E(a) = 2·n·max(a*a) / (sum(a)²)

Thus we may normalise the sequence to sum = 1 without loss of
generality (the evaluator clips to [0,1000] and forces non‑negative
values, which our construction respects).

The algorithm consists of three phases:

1. **Cheap analytic sweep** – power‑law and shifted‑sqrt families.
   The best result of this stage is taken as the starting point.

2. **Projected sub‑gradient descent** – at each iteration we compute
   the full linear convolution a⊛a, locate the shift *k* that attains
   the current worst (max) value, and take the sub‑gradient
   ∂/∂a_i max = 2·a_{k‑i}.  A simple diminishing step size
   ηₜ = η₀/√t is used, and after every update the vector is projected
   back onto the probability simplex (non‑negative, sum = 1).  The
   iteration also contains occasional symmetrisation and a light
   random “block” perturbation to avoid stagnation.

3. **Return the best sequence found** – the sequence with the lowest
   score seen during the whole run (not necessarily the last iterate).

All heavy work (convolution, gradient) is performed with NumPy’s FFT
routines, so a single iteration is O(n log n) and the whole search
fits easily into the allotted 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray) -> np.ndarray:
    """
    Linear (full) convolution of a real vector with itself using FFT.
    Returns an array of length 2·len(x)‑1.
    """
    n = x.shape[0]
    L = 1 << ((2 * n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=L)
    conv = np.fft.irfft(fx * fx, n=L)
    return conv[: 2 * n - 1]


# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        { w | w_i ≥ 0, sum w_i = 1 }.
    Implementation follows Duchi et al. (2008).
    """
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):               # fallback – should never happen
        return np.maximum(v, 0.0)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _score_vector(a: np.ndarray) -> float:
    """
    Compute the exact evaluation‑function value for a (without the
    security checks of ``evaluate_sequence`` – our a is already safe).
    """
    s = a.sum()
    if s < 0.01:                      # matches the internal safety guard
        return np.inf
    conv = _full_convolution(a)
    max_b = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_b / (s * s))

    # combined families
    for p in (0.45, 0.50, 0.55):
        for sh in (0.0, 0.5, 1.0, 2.0, 5.0):
            cand = 1.0 / np.power(idx + sh, p)
            cand /= cand.sum()
            sc = _score_vector(cand)
            if sc < best_score:
                best, best_score = cand, sc
                print(f"[sweep] p={p:.2f} shift={sh:.1f} → {sc:.6f}")

    # --------------------------------------------------------------
    # 2) PROJECTED SUB‑GRADIENT DESCENT
    a = best.copy()                     # current iterate
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 1.0)  # leave <1 s spare

    it = 0
    eta0 = 0.7                          # initial step‑size (tuned empirically)

    while time.monotonic() < deadline:
        it += 1

        # convolution and the worst (max) shift
        conv = _full_convolution(a)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the active shift: g_i = 2·a_{k_max‑i}
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[j[mask]]

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # gradient step + simplex projection
        cand = a - eta * g
        cand = np.maximum(cand, 0.0)
        cand = _project_onto_simplex(cand)

        # occasional symmetry – helps to keep the convolution flat
        if it % 250 == 0:
            cand = 0.5 * (cand + cand[::-1])
            cand = _project_onto_simplex(cand)

        # occasional random block perturbation for escaping shallow minima
        if it % 600 == 0:
            width = max(1, n // 80)                     # 1–2 % of the length
            start = rng.integers(0, n - width + 1)
            cand[start : start + width] *= np.exp(
                rng.normal(0.0, 0.15, size=width)
            )
            cand = np.clip(cand, 0.0, None)
            cand = _project_onto_simplex(cand)

        # evaluate candidate
        sc = _score_vector(cand)
        if sc < best_score:
            best, best_score = cand.copy(), sc
            print(f"[iter {it:05d}] new best {sc:.6f}")

        # move forward
        a = cand

    # --------------------------------------------------------------
    # 3) Return the best sequence (plain python list)
    print(f"[done] best_score={best_score:.6f} after {it} iterations")
    # Scale back to the original magnitude – any positive scaling leaves the
    # score unchanged, we just keep the normalised version.
    return [float(v) for v in best]
