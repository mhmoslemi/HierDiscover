"""
Propose a height sequence via
* soft‑max parametrisation (ensures non‑negativity & unit ℓ₁‑norm),
* a smooth approximation of the max‑convolution using a log‑sum‑exp,
* Adam optimisation on the logit parameters,
* occasional stochastic perturbations to escape plateaus.

The algorithm works with any starting sequence (e.g. the previously
found `height_sequence_1`).  It repeatedly evaluates the true score
2·n·max(conv)/sum² (which for a unit‑sum sequence is simply 2·n·max(conv))
and keeps the best vector seen before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax returning a probability vector (sum = 1)."""
    m = np.max(logits)
    exp_shifted = np.exp(logits - m)
    return exp_shifted / exp_shifted.sum()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises
    the provided `evaluate_sequence` objective.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget (seconds). The routine stops a few
        seconds before the limit to avoid termination.

    Returns
    -------
    list[float]
        The best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time budget (leave a small safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # initialise from previous best if available, otherwise uniform
    if 'height_sequence_1' in globals():
        start = np.array(height_sequence_1, dtype=np.float64)
        start = np.maximum(start, 0.0)          # safety
        if start.sum() < 1e-12:
            n = start.size
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a0 = start / start.sum()
        n = a0.size
    else:
        n = 1024
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # log‑it representation (softmax → feasible point)
    eps = 1e-12
    logits = np.log(a0 + eps)

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.15                     # base learning rate for log‑its

    # ------------------------------------------------------------------
    # helper for a fast linear convolution via FFT (real inputs)
    def _conv_fft(x: np.ndarray, y: np.ndarray, Nfft: int) -> np.ndarray:
        """Linear convolution of real vectors x and y using FFT of size Nfft."""
        X = np.fft.rfft(x, n=Nfft)
        Y = np.fft.rfft(y, n=Nfft)
        z = np.fft.irfft(X * Y, n=Nfft)
        return z[: len(x) + len(y) - 1]

    # pick an FFT size that comfortably contains a_rev + w (length 3·n‑2)
    Nfft = 1
    while Nfft < 3 * n:
        Nfft <<= 1

    # ------------------------------------------------------------------
    # track the best solution found
    a = _softmax(logits)
    conv_full = np.convolve(a, a)
    best_score = float(2.0 * n * conv_full.max())   # exact score, sum(a)==1
    best_a = a.copy()
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # hyper‑parameters for the smooth‑max approximation
    alpha_start = 5.0
    alpha_end = 200.0
    alpha_ramp_iters = 200_000   # iterations over which alpha grows linearly

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # current feasible vector (soft‑max)
        a = _softmax(logits)

        # --------------------------------------------------------------
        # linear convolution a*a (needed for both loss and gradient)
        Fa = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(Fa * Fa, n=Nfft)[: 2 * n - 1]

        # --------------------------------------------------------------
        # smooth‑max temperature (linear schedule)
        if iteration <= alpha_ramp_iters:
            alpha = alpha_start + (alpha_end - alpha_start) * iteration / alpha_ramp_iters
        else:
            alpha = alpha_end

        # --------------------------------------------------------------
        # compute soft weights that approximate the max entry
        max_conv = conv.max()
        shifted = conv - max_conv          # ≤ 0, avoids overflow
        w = np.exp(alpha * shifted)        # weights, sum to Z after normalisation
        Z = w.sum()
        if Z == 0.0 or np.isnan(Z):
            # fallback – should not happen, but keep algorithm alive
            w = np.exp(alpha * shifted - np.max(alpha * shifted))
            Z = w.sum()

        # --------------------------------------------------------------
        # gradient of the smooth‑max (w.r.t. a)
        a_rev = a[::-1]
        conv_rev = _conv_fft(a_rev, w, Nfft)               # length 3·n‑2
        # cross‑correlation slice that yields Σ_j a_j w_{i+j}
        grad_a = (2.0 / Z) * conv_rev[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # transform gradient to logit space (softmax Jacobian)
        dot = np.dot(a, grad_a)
        grad_logits = a * (grad_a - dot)

        # --------------------------------------------------------------
        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # --------------------------------------------------------------
        # occasional random shake‑up (helps escape plateaus)
        if rng.random() < 0.02:
            logits += rng.normal(0.0, 0.12, size=logits.shape)

        # --------------------------------------------------------------
        # evaluate the true objective every few iterations
        if iteration % 10 == 0:
            # true max of the linear convolution (no smoothing)
            conv_true = np.convolve(a, a)
            cur_score = float(2.0 * n * conv_true.max())
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
                print(f"[improve] iter={iteration}, score={best_score:.6f}")

            # early stop if we are already very close to the target
            if best_score <= 1.5025 * 1.01:
                print("[early stop] near‑target bound reached")
                break

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # Return a plain Python list (the evaluation routine will clip it anyway)
    return [float(v) for v in best_a]
