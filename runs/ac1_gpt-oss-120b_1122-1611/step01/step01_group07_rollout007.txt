import time

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """Search for a step‑function (non‑negative sequence) that minimises
    ``evaluate_sequence``.
    
    The algorithm consists of a quick parametric sweep (exponential,
    power‑exponential, Hann, polynomial, inverse‑power families) to obtain
    a decent starting point, followed by a projected sub‑gradient descent
    on the quantity ``2·n·max(conv)`` (the evaluation metric for a
    normalised sequence with total sum = 1).  At each iteration the
    convolution of the current sequence with itself is computed by FFT,
    the index(s) attaining the maximal autocorrelation are identified,
    a sub‑gradient that reduces those terms is taken, and the resulting
    vector is projected back onto the simplex (non‑negative, sum = 1).
    The loop also injects occasional random jitter and symmetry
    projections to avoid stagnation.  The best sequence seen before the
    time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # keep a few seconds free

    # --------------------------------------------------------------------
    # Problem size – use the provided construction if available
    # --------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 4806
        base_seq = np.ones(n, dtype=np.float64)

    # --------------------------------------------------------------------
    # Utility functions (top‑level, no closures)
    # --------------------------------------------------------------------
    def _normalise(v):
        """Scale a non‑negative vector to have sum = 1."""
        s = v.sum()
        if s == 0.0:
            return np.full_like(v, 1.0 / v.size)
