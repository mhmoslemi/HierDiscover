def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # Initialize best sequence from power law with p=0.5 and n=2000
    n = 2000
    seq = (np.arange(n) + 1) ** -0.5
    seq = seq / seq.sum()
    best_seq = seq.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"Initial max conv seq n={n}, score={best_score:.6f}")

    # Hyperparameters
    step = 0.5 # step size for gradient descent
    step_min = 1e-6
    step_max = 2.0
    tol = 1e-12
    iteration = 0
    improve_counter = 0

    while time.monotonic() < deadline:
        iteration += 1
        # compute gradient descent step
        conv = np.convolve(best_seq, best_seq)
        max_val = conv.max()
        # Determine active set tolerance based on relative differences
        active = np.abs(conv - max_val) <= max_val * 1e-9
        # If active set is empty (shouldn't), just continue
        if not np.any(active):
            active = (conv == max_val)
        w = active.astype(float)
        # Normalize weight to average subgradient
        if w.sum() > 0:
            w = w / w.sum()
        # gradient via convolution with reversed weights
        w_rev = w[::-1]
        grad_rev = np.convolve(best_seq, w_rev)
        # get gradient slice
        idx_start = 2 * n - 2
        grad = 2 * grad_rev[idx_start - np.arange(n)]

        # Candidate step
        cand = best_seq - step * grad  # move opposite gradient to decrease max conv
        # Project onto simplex
        cand = project_to_simplex(cand)
        # Clip values within bounds (0-1000)
        cand = np.clip(cand, 0.0, 1000.0)

        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_score = score
            best_seq = cand
            improve_counter += 1
            # if improvement, maybe increase step slightly
            step = min(step * 1.02, step_max)
            print(f"[iter {iteration}] improvement! score={best_score:.6f}, step={step:.4f}")
        else:
            # reject step and shrink step
            step = max(step * 0.5, step_min)
        # occasional random perturbation to escape plateaus (maybe every 100 iterations)
        if iteration % 200 == 0:
            # random block multiplicative perturbation (similar to original) and reproject
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 20))
            stop = min(n, start + width)
            factor = math.exp(rng.normal(0.0, 0.12))
            pert = best_seq.copy()
            pert[start:stop] *= factor
            pert = project_to_simplex(pert)
            pert = np.clip(pert, 0.0, 1000.0)
            pscore = float(evaluate_sequence(pert.tolist()))
            if pscore < best_score:
                best_score = pscore
                best_seq = pert
                print(f"[perturb] improved via random step: {best_score:.6f}")

    # fallback
    if best_seq is None:
        best_seq = np.ones(256) / 256

    return [float(x) for x in best_seq]

"""
Particle Gradient Descent for a Flat Self‑Convolution

We treat the evaluation function
    score = 2·n·max(conv(a,a)) / (sum(a))²
as a scale‑invariant objective: with sum(a)=1 the score is simply
    2·n·max(conv(a,a)).
The gradient of the (non‑smooth) max is approximated by the sub‑gradient
coming from the active convolution indices.
Starting from a power‑law (the known near optimal shape a_i ∝ (i+1)⁻½)
we perform a projected gradient descent on the simplex
{a ≥ 0, Σa = 1}.  The step is adapted by a simple back‑tracking line
search; occasional random block perturbations and optional symmetrisation
help to escape plateaus.  The routine runs until the allotted time expires
and returns the best sequence found.

The algorithm is deterministic given the seed, uses only NumPy and the
provided ``evaluate_sequence`` function, and respects the 1000 s budget.
"""

import time
import math
import numpy as np


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    """
    n = v.size
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the last index where u_i > cssv_i / i
    rho = np.where(u - cssv / ind > 0)[0]
    if rho.size == 0:          # all entries are non‑positive
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq

    # safety margin for the wrapper that calls us
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ----------------------------------------------------------------------
    # 1) Initialisation – a short grid on the exponent of a power law.
    # ----------------------------------------------------------------------
    n = 2000  # base length (good trade‑off between quality and speed)
    best_seq = None
    best_score = float("inf")

    for p in np.linspace(0.45, 0.62, 20):
        seq = _make_power_sequence(n, p)
        sc = float(evaluate_sequence(seq.tolist()))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
        print(f"[init] p={p:.3f}, score={sc:.6f}")

    # ----------------------------------------------------------------------
    # 2) Projected gradient descent on the simplex.
    # ----------------------------------------------------------------------
    step = 0.5                     # initial step size
    step_min = 1e-6
    step_max = 2.0
    iteration = 0
    improve_counter = 0

    while time.monotonic() < deadline:
        iteration += 1
        a = best_seq
        # ----- compute convolution and active set -----
        conv = np.convolve(a, a)
        max_val = conv.max()
        # indices where the convolution attains its maximum (ties possible)
        active_idx = np.flatnonzero(conv == max_val)
        if active_idx.size == 0:
            # numerical safety – fall back to the argmax
            active_idx = np.array([conv.argmax()], dtype=int)

        # weight vector w that is uniform over the active set
        w = np.zeros_like(conv, dtype=np.float64)
        w[active_idx] = 1.0 / active_idx.size

        # ----- gradient of the (smoothed) max -----
        # grad_i = 2 * Σ_j a_j * w_{i+j}
        w_rev = w[::-1]
        rev = np.convolve(a, w_rev)                     # length = 3n‑2
        idx_start = 2 * n - 2                           # offset for correct slice
        grad = 2.0 * rev[idx_start - np.arange(n)]

        # ----- line‑search on the step size -----
        step_try = step
        improved = False
        for _ in range(6):                              # at most 6 halvings
            cand = a - step_try * grad
            cand = _project_to_simplex(cand)
            cand = np.clip(cand, 0.0, 1000.0)            # respects evaluate_sequence limits
            sc = float(evaluate_sequence(cand.tolist()))
            if sc < best_score - 1e-12:                 # strict improvement
                best_seq = cand
                best_score = sc
                step = min(step_try * 1.2, step_max)    # modestly increase on success
                improved = True
                improve_counter += 1
                print(f"[iter {iteration}] new best = {best_score:.6f} (step={step:.4f})")
                break
            else:
                step_try *= 0.5                         # halve and retry

        if not improved:
            # No improvement – shrink the global step a bit
            step = max(step * 0.5, step_min)

            # Occasionally inject a random block perturbation to escape a flat region
            if iteration % 150 == 0:
                start = rng.integers(0, n)
                width = rng.integers(1, max(2, n // 20))
                stop = min(n, start + width)
                factor = math.exp(rng.normal(0.0, 0.12))
                pert = best_seq.copy()
                pert[start:stop] *= factor
                pert = _project_to_simplex(pert)
                pert = np.clip(pert, 0.0, 1000.0)
                sc_pert = float(evaluate_sequence(pert.tolist()))
                if sc_pert < best_score:
                    best_seq = pert
                    best_score = sc_pert
                    print(f"[perturb] improved to {best_score:.6f}")

            # Small chance to enforce symmetry (the problem is symmetric)
            if rng.random() < 0.08:
                sym = 0.5 * (best_seq + best_seq[::-1])
                sym = _project_to_simplex(sym)
                sc_sym = float(evaluate_sequence(sym.tolist()))
                if sc_sym < best_score:
                    best_seq = sym
                    best_score = sc_sym
                    print(f"[sym] improved to {best_score:.6f}")

        # Periodic status report (lightweight)
        if iteration % 500 == 0:
            print(f"[status] iter={iteration}, best_score={best_score:.6f}, step={step:.4f}")

    # ----------------------------------------------------------------------
    # 3) Fallback – if nothing was found (should never happen).
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)
        best_seq /= best_seq.sum()

    # Return plain Python floats as required by the caller
    return [float(x) for x in best_seq]
