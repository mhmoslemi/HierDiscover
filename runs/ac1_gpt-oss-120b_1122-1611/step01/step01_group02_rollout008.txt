def propose_candidate(seed=123, budget_s=1000, **kwargs):

    n = height_sequence_1 size if present else 4000

    target_sum = np.sqrt(2.0*n)

    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"initial score: {best_score}")

    start = time.monotonic()
    deadline = start + budget_s - 2.0

    # parameters
    temperature = 0.1
    cooling = 0.9995

    while time.monotonic() < deadline:
        seq = best_seq.copy()

        # compute conv via FFT
        conv = convolve_self(seq)
        k = int(np.argmax(conv))

        # compute contributions for positions that generate max
        # contributions for i in range(max(0, k-(n-1)), min(n-1,k)+1)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        # Compute contributions array for i indices
        idxs = np.arange(i_start, i_end + 1)
        contrib = seq[idxs] * seq[k - idxs]

        # Choose source index weighted by contrib
        if contrib.sum() == 0:
            # fall back random indices
            i_src = rng.integers(0, n)
        else:
            probs = contrib / contrib.sum()
            i_src = rng.choice(idxs, p=probs)

        # Choose destination index; ideally one with small contribution to any max
        # We could compute total contribution across all k for each index? Too heavy.
        # Simpler: choose random index preferentially away from i_src
        i_dst = rng.integers(0, n)
        while i_dst == i_src:
            i_dst = rng.integers(0, n)

        # amount to transfer
        max_move = seq[i_src] * 0.5
        if max_move <= 0:
            continue
        delta = max_move * rng.random()   # uniformly 0..max_move

        # apply move
        seq[i_src] -= delta
        seq[i_dst] += delta

        # clip
        np.clip(seq, 0.0, 1000.0, out=seq)

        # normalize sum to target_sum (to avoid drift)
        total = seq.sum()
        if total == 0:
            continue
        seq *= target_sum / total

        try:
            score = evaluate_sequence(seq.tolist())
        except Exception:
            continue

        # accept if better or with prob
        if np.isfinite(score):
            if score < best_score:
                best_seq = seq
                best_score = score
                print(f"[improve] score={best_score:.6f} iter time={time.monotonic()-start:.1f}s")
            else:
                # simulated annealing
                if temperature > 0:
                    prob = np.exp((best_score - score) / temperature)
                    if rng.random() < prob:
                        best_seq = seq
                        best_score = score
        # temperature cooling
        temperature *= cooling

    print(f"final best {best_score}")
    return best_seq.tolist()

"""
Propose a step‑function with a lower evaluation than the Catalan construction.

The algorithm starts from the Catalan (constant‑convolution) sequence,
normalised to the standard L¹‑norm √(2·n).  It then performs a large number
of cheap stochastic moves while keeping the total sum fixed:

* **Block perturbations** – a random contiguous block is multiplied by a
  log‑normal factor (the classic “polish” used in the reference code).

* **Targeted mass transfer** – the index k where the self‑convolution attains
  its maximum is identified.  A source index that contributes heavily to this
  peak (chosen proportionally to its product a_i·a_{k‑i}) donates a fraction
  of its value to a random destination index.  This directly reduces the
  dominant term of the convolution.

Both move types are evaluated with the supplied ``evaluate_sequence``.
Improving candidates are kept, and a simple simulated‑annealing acceptance
criterion lets occasional uphill moves escape local minima.

The inner convolution is performed with an FFT‑based routine
(`convolve_self`) which is O(n log n) and therefore fast enough to allow
thousands of iterations within the 1000 s budget.

The returned list satisfies all security checks of ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog  # only used for the auxiliary “LP‑step” (optional)

# ----------------------------------------------------------------------
# Helper functions – all defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    First «length» terms of a_i = C(2i,i) / 4**i.
    The recurrence a_{i+1} = a_i * (2*i+1) / (2*i+2) is used for stability.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale ``seq`` so that its ℓ₁‑norm equals ``target_sum``.
    """
    cur = np.sum(seq)
    if cur == 0:
        return seq
    factor = target_sum / cur
    return seq * factor


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a cheap O(n) perturbation: choose a random contiguous block and
    multiply it by a log‑normal factor; occasionally mirror the sequence.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def convolve_self(seq: np.ndarray) -> np.ndarray:
    """
    Fast self‑convolution using an FFT of length a power‑of‑two.
    Returns the full (2·len‑1) result.
    """
    n = seq.size
    m = 2 * n - 1
    # next power of two ≥ m
    size = 1 << (m - 1).bit_length()
    fft = np.fft.rfft(seq, size)
    conv = np.fft.irfft(fft * fft, size)[:m]
    # tiny negative round‑off can appear; clip to zero for safety
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Main function required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Build a Catalan baseline and then use a hybrid stochastic / targeted
    optimisation (block perturbations + mass‑transfer moves) under a simulated
    annealing scheme to lower the evaluation value.
    The best sequence found before the time deadline is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine a convenient length.  If a previous construction is present
    # we reuse its length, otherwise we fall back to a modest size that is
    # still large enough to give a good search space.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n < 10:                     # sanity check
                raise ValueError
        except Exception:
            n = 2500
    else:
        n = 2500

    # ------------------------------------------------------------------
    # Baseline Catalan sequence, normalised so that sum = √(2·n).
    # After the normalisation the objective equals the maximal
    # convolution entry, i.e.  eval = max(conv) .
    # ------------------------------------------------------------------
    base_seq = catalan_sequence(n)
    target_sum = np.sqrt(2.0 * n)
    base_seq = normalise_to_target_sum(base_seq, target_sum)

    # initial evaluation
    try:
        best_score = float(evaluate_sequence(base_seq.tolist()))
    except Exception:          # should never happen
        best_score = float('inf')
    best_seq = base_seq.copy()

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing style search.
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # leave a small safety margin
    temperature = 0.6
    cooling = 0.9995
    moves = 0

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # Choose a perturbation type.
        #   0 – cheap block perturbation (the “local polish” from the reference)
        #   1 – targeted mass transfer that reduces the current maximal
        #       convolution entry.
        # ------------------------------------------------------------------
        if rng.random() < 0.5:
            cand = block_perturbation(best_seq, rng)
        else:
            # ----- targeted move -------------------------------------------------
            cand = best_seq.copy()
            conv = convolve_self(cand)
            k_max = int(np.argmax(conv))

            # valid i such that both i and k_max-i lie inside the array
            i_start = max(0, k_max - (n - 1))
            i_end = min(n - 1, k_max)
            idxs = np.arange(i_start, i_end + 1, dtype=int)

            # contribution of each admissible index to the peak value
            contrib = cand[idxs] * cand[k_max - idxs]

            if contrib.sum() > 0:
                probs = contrib / contrib.sum()
                i_src = rng.choice(idxs, p=probs)
            else:
                i_src = rng.integers(0, n)

            # pick a destination that is different from the source
            i_dst = rng.integers(0, n)
            while i_dst == i_src:
                i_dst = rng.integers(0, n)

            # move a random fraction (5‑30 %) of the source value to the dest.
            fraction = rng.uniform(0.05, 0.30)
            delta = cand[i_src] * fraction
            cand[i_src] -= delta
            cand[i_dst] += delta
            np.clip(cand, 0.0, 1000.0, out=cand)

        # --------------------------------------------------------------
        # Renormalise to keep the L¹‑norm exactly √(2·n).  Scaling does not
        # affect the objective because it cancels out, but keeping the norm
        # fixed avoids drift and respects the usual normalisation.
        # --------------------------------------------------------------
        total = cand.sum()
        if total == 0.0:
            continue
        cand *= target_sum / total

        # --------------------------------------------------------------
        # Evaluate the candidate.
        # --------------------------------------------------------------
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        # --------------------------------------------------------------
        # Acceptance rule: always keep a strict improvement; otherwise
        # accept with probability exp((best‑score)/temperature) – the classic
        # simulated‑annealing rule.
        # --------------------------------------------------------------
        accept = False
        if np.isfinite(score):
            if score < best_score:
                accept = True
            else:
                # uphill move
                prob = np.exp((best_score - score) / max(temperature, 1e-12))
                if rng.random() < prob:
                    accept = True

        if accept:
            best_seq = cand
            best_score = score
            print(f"[improve] new best {best_score:.6f} (t={temperature:.4f}) "
                  f"elapsed={time.monotonic() - start:.1f}s")
        # cooling
        temperature *= cooling
        moves += 1

    print(f"[final] returning sequence of length {n} with score {best_score:.6f} "
          f"after {moves} moves")
    # Convert to plain Python list as required by the harness.
    return [float(x) for x in best_seq]
