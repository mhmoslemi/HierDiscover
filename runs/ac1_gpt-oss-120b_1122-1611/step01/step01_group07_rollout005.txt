"""Python implementation of a parametric/annealing search for a low‑score
step‑function sequence.

The algorithm works in three phases:

1.  A cheap random parametric sweep over several analytic families
    (single‑parameter exponentials, power‑exponential, power‑law,
    mixtures of two exponentials and simple piecewise constant shapes).
    The best candidate from this sweep seeds the next phase.

2.  A simulated‑annealing / hill‑climbing phase that repeatedly perturbs
    the current best sequence (block scaling, per‑element log‑normal jitter,
    global scaling and a small smoothing move).  A Metropolis criterion
    allows occasional uphill moves, helping the search escape local minima.

3.  The algorithm runs until the time budget (≈ 1000 s) expires and returns
    the best sequence discovered.  The sequence length is limited to a few
    hundred elements (default 800) which makes each evaluation cheap while
    preserving the scale‑invariant objective.

All helper functions are defined at module level and the public entry point
is ``propose_candidate`` as required by the evaluation harness.
"""

import time
import math
import numpy as np

def _evaluate(seq):
    """Wrapper around the provided evaluate_sequence function."""
    # The harness already imports evaluate_sequence into the global namespace.
    return float(evaluate_sequence(seq.tolist()))

def generate_exponential(n, c, p):
    """a_i = exp(-c * (i/(n-1))**p)  – normalized to max = 1."""
    idx = np.arange(n, dtype=np.float64)
    N = n - 1
    a = np.exp(-c * (idx / N) ** p)
    if a.max() == 0:
        return a
    return a / a.max()

def generate_power_exponential(n, alpha, c, p):
    """a_i = (i+1)**alpha * exp(-c * (i/(n-1))**p)  – normalized."""
    idx = np.arange(n, dtype=np.float64)
    N = n - 1
    a = (idx + 1.0) ** alpha * np.exp(-c * (idx / N) ** p)
    a = np.clip(a, 0.0, None)
    if a.max() == 0:
        return a
    return a / a.max()

def generate_power_law(n, beta):
    """a_i = 1/(i+1)**beta  – normalized."""
    idx = np.arange(n, dtype=np.float64)
    a = 1.0 / ((idx + 1.0) ** beta)
    if a.max() == 0:
        return a
    return a / a.max()

def generate_mixture_exponential(n, terms):
    """Linear combination of exponentials; each term is (weight, c, p)."""
    idx = np.arange(n, dtype=np.float64)
    N = n - 1
    a = np.zeros(n, dtype=np.float64)
    for w, c, p in terms:
        a += w * np.exp(-c * (idx / N) ** p)
    a = np.clip(a, 0.0, None)
    if a.max() == 0:
        return a
    return a / a.max()

def generate_piecewise_constant(n, rng, K=4):
    """K decreasing constant blocks with random heights."""
    # split the index axis into K (almost) equal parts
    cuts = np.sort(rng.integers(1, n, size=K - 1))
    cuts = np.concatenate(([0], cuts, [n]))
    # generate decreasing heights
    heights = np.sort(rng.uniform(0.1, 1.0, size=K))[::-1]
    a = np.empty(n, dtype=np.float64)
    for i in range(K):
        a[cuts[i]:cuts[i + 1]] = heights[i]
    # small jitter inside blocks to avoid perfect flatness
    a += rng.normal(0.0, 0.02, size=n)
    a = np.clip(a, 0.0, None)
    if a.max() == 0:
        return a
    return a / a.max()

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a step‑function sequence with a low evaluation score.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds. The function returns before the budget expires.
    kwargs : dict
        Ignored – kept for compatibility with the harness.

    Returns
    -------
    list[float]
        The best non‑negative sequence found (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    # keep a few seconds free for printing & finalisation
    deadline = start_time + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------------
    # Choose a working dimension that is small enough to allow many evals.
    # If a reference sequence is supplied, keep its length modest.
