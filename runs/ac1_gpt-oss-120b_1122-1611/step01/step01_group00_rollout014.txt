def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, cvxpy as cp
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.5
    # choose n
    n = 1024
    # start with Dirichlet
    a = rng.dirichlet(alpha=np.ones(n)*0.5)  # somewhat sparse
    # ensure sum=1 (dirichlet)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print("[init] score:", best_score)
    # parameters
    iteration = 0
    beta = 0.9
    v = np.zeros_like(a)
    # schedule for gamma: start low, increase
    gamma = 1.0
    gamma_inc = 0.001  # per iteration maybe
    # step size schedule: eta0 / sqrt(iter+1)
    eta0 = 0.5
    # main loop
    while time.monotonic() < deadline:
        iteration += 1
        # compute conv
        conv = np.convolve(a, a)
        # compute p = softmax(gamma*conv)
        # shift for stability
        max_val = np.max(gamma * conv)
        exp_vals = np.exp(gamma * conv - max_val)
        p = exp_vals / np.sum(exp_vals)
        # compute gradient = 2 * correlation of p and a
        # we compute conv_rev = np.convolve(p, a[::-1], mode='full')
        conv_rev = np.convolve(p, a[::-1], mode='full')
        # gradient indices: 2*n-2 - np.arange(n)
        grad = 2.0 * conv_rev[2*n - 2 - np.arange(n)]
        # momentum update
        v = beta * v + (1 - beta) * grad
        # step size
        eta = eta0 / np.sqrt(iteration)
        a_candidate = a - eta * v
        # project onto simplex
        a_candidate = _project_simplex(a_candidate)
        # occasional random jitter (5% chance)
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            a_candidate[idx] *= factor
            a_candidate = _project_simplex(a_candidate)
        # evaluate
        score = float(evaluate_sequence(a_candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_candidate.copy()
            print(f"[improve] iter={iteration} score={best_score:.6f}")
        a = a_candidate
        # increase gamma slowly to focus on max
        gamma = min(1000.0, gamma + gamma_inc)
        # early termination if score near target
        if best_score <= 1.5025 * 1.01:
            break
    # LP refinement (optional)
    if time.monotonic() < deadline:
        try:
            # solve LP to minimize max conv bound for current best a
            a_param = cp.Parameter(n, value=best)
            b = cp.Variable(n)
            t = cp.Variable()
            conv_expr = cp.conv(a_param, b)  # vector length 2n-1
            constraints = [b >= 0,
                           cp.sum(b) == 1,
                           conv_expr <= t]
            prob = cp.Problem(cp.Minimize(t), constraints)
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False, max_iters=2000)
            if b.value is not None:
                b_opt = np.maximum(b.value, 0)
                lp_score = float(evaluate_sequence(b_opt.tolist()))
                if lp_score < best_score:

                    best = b_opt.copy()
                    print("[lp improve] score", best_score)
        except Exception as e:
            print("[lp] exception", e)
    print("[final] best score", best_score, "iterations", iteration)
    # return list

"""
Propose a step‑function height sequence with a low evaluation score.

The algorithm consists of three phases:
1. **Randomised warm‑start** – draw a Dirichlet vector (sum = 1) that may
   already be sparse.
2. **Annealed soft‑max gradient descent** – minimise a smooth approximation
   of  max_k (a∗a)[k] while staying on the probability simplex.
   The temperature (γ) is slowly increased so the optimisation first
   smoothes the landscape and then focuses on the true max.  Momentum and
   occasional multiplicative perturbations help to escape plateaus.
3. **Linear‑programming refinement** – once the time budget is almost spent,
   solve a small LP that directly minimises the worst convolution entry for
   the last‑found distribution (variables b, bound t):
        minimise   t
        s.t.       b ≥ 0 ,  Σ b = 1 ,
                   (a ⊛ b)[k] ≤ t  ∀k .
   The LP often reduces the objective a little further.

The whole routine respects the 1000 s time limit, prints progress and
returns the best‑found list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence that minimises `evaluate_sequence`.
    Uses annealed soft‑max gradient descent followed by a final LP refinement.
    """
    import numpy as np
    import time

    # the evaluation routine is assumed to be present in the global namespace
    # (it is injected by the competition harness)
    # from __main__ import evaluate_sequence   # noqa: F401

    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # total time budget (a little safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # problem size – 1024 is a good compromise between quality and speed
    n = 1024

    # ------------------------------------------------------------------
    # initialise a random feasible point (Dirichlet, sparsity controlled by α)
    alpha = 0.5                     # smaller → sparser start
    a = rng.dirichlet(np.full(n, alpha)).astype(np.float64)

    # keep the best ever seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, start score={best_score:.6f}")

    # ------------------------------------------------------------------
    # parameters for the annealed soft‑max descent
    iteration = 0
    eta0 = 0.5                     # base step size
    beta = 0.9                     # momentum factor
    v = np.zeros_like(a)           # momentum accumulator
    gamma = 1.0                    # soft‑max temperature (low → smooth)
    gamma_factor = 1.001           # slow geometric increase
    max_gamma = 2000.0

    # main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1

        # ---- convolution of the current vector with itself --------------------
        conv = np.convolve(a, a)          # length = 2*n-1

        # ---- soft‑max weights (smooth approximation of the max) ---------------
        #   w_i ∝ exp(γ * conv_i)
        exp_shift = np.max(gamma * conv)  # for numerical stability
        weights = np.exp(gamma * conv - exp_shift)
        weights /= np.sum(weights)        # now Σ weights = 1

        # ---- gradient of the smoothed objective --------------------------------
        #   ∇_a M_γ = 2 * Σ_i w_i * shift_i(a)
        #   This equals 2 * correlation(weights, a) at negative lags.
        corr = np.correlate(weights, a, mode='full')   # length = 3*n‑2
        grad = 2.0 * corr[(n - 1) - np.arange(n)]

        # ---- momentum update ---------------------------------------------------
        v = beta * v + (1.0 - beta) * grad

        # ---- step size (diminishing) -------------------------------------------
        eta = eta0 / np.sqrt(iteration)

        # ---- projected gradient step --------------------------------------------
        a_candidate = a - eta * v
        a_candidate = _project_simplex(a_candidate)

        # ---- occasional random shake‑up (helps escaping plateaus) ---------------
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.2)))
            a_candidate[idx] *= factor
            a_candidate = _project_simplex(a_candidate)

        # ---- evaluate -----------------------------------------------------------
        score = float(evaluate_sequence(a_candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a_candidate.copy()
            print(f"[improve] iter={iteration:,} score={best_score:.6f}")

        a = a_candidate

        # ---- anneal temperature -------------------------------------------------
        gamma = min(max_gamma, gamma * gamma_factor)

        # ---- early exit if we are already under the target ----------------------
        if best_score <= 1.5025 * 1.01:
            print("[early stop] target vicinity reached")
            break

    # ------------------------------------------------------------------
    # One‑shot linear‑programming refinement (if time permits)
    try:
        if time.monotonic() < deadline:
            import cvxpy as cp

            # Parameter for the current best vector (allows a cheap reload)
            a_param = cp.Parameter(n, value=best, nonneg=True)

            b = cp.Variable(n)
            t = cp.Variable()
            # convolution a_param * b  (length = 2*n-1)
            conv_expr = cp.conv(a_param, b)

            lp = cp.Problem(cp.Minimize(t),
                            [b >= 0,
                             cp.sum(b) == 1,
                             conv_expr <= t])
            # Warm‑start with previous solution if any
            lp.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, verbose=False)

            if b.value is not None:
                b_opt = np.maximum(b.value, 0.0)
                lp_score = float(evaluate_sequence(b_opt.tolist()))
                if lp_score < best_score:
                    best_score = lp_score
                    best = b_opt.copy()
                    print(f"[lp improve] new score={best_score:.6f}")

    except Exception as exc:  # pragma: no cover
        # LP could fail on some installations – ignore and keep the current best
        print("[lp] exception during refinement:", exc)

    # ------------------------------------------------------------------
    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    return [float(x) for x in best]
