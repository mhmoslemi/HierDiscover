def simplex_projection(v):
    # returns w = argmin ||w - v||_2 s.t. w >= 0, sum w = 1
    # Implementation from literature.
    n = len(v)
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]  # last true
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

    max_val = conv.max()
    # find indices where conv is max

    # gradient g = zeros; for each idx, contribution: for each k: g_k += 2 * a[idx - k] if in range
    g = np.zeros_like(a)

"""
Propose a near‑optimal step‐function for the `evaluate_sequence` task.

Algorithm
---------
1. **Fast random search**  
   * Randomly sample a length `n` (1000‑5000), a power‑law exponent `p`,
     an additive offset `o≥0` and a tiny exponential decay `λ`.  
   * Build the sequence  

        a_i ∝ (i + 1 + o)^{-p} · exp(-λ·i)

     and normalise it to sum = 1 (the objective is scale‑invariant).  
   * Compute the objective cheaply (exactly as `evaluate_sequence` does) and
     keep the best candidate.

2. **Local smoothing optimisation**  
   For the current best sequence run a few hundred iterations of a
   smoothed gradient descent:
   * Compute the full self‑convolution `b = a * a`.
   * Form a soft‑max weight vector over the convolution entries  
     `w_i ∝ exp(τ·(b_i – max(b)))`.  The temperature `τ` starts small
     (≈10) and is increased slowly – this turns the non‑smooth `max`
     into a smooth surrogate, allowing the algorithm to move away from
     symmetric plateaus.
   * The gradient of the (smooth) max‑convolution is  

        g = 2·conv_rev(a, w)        (length‑n slice)

     where `conv_rev(a, w)` is the convolution of the reversed `a`
     with `w`.  The factor 2 follows from ∂(a_i a_j)/∂a_k = 2·a_{i‑k}.
   * Take a projected gradient step `a ← a – α·g` and project back onto the
     probability simplex (non‑negative, sum = 1).
   * Every few iterations evaluate the true objective with `evaluate_sequence`
     and keep the best sequence encountered.

3. **Fallback** – if nothing was found, return a uniform sequence.

The whole procedure respects the 1000 s time budget, prints progress
messages and finally returns a list of Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project vector `v` onto the probability simplex:
        {x | x_i ≥ 0, sum x_i = 1}
    Implements the O(n) algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1.0
    # find the rho index: last i where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return the uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(seq: np.ndarray) -> float:
    """
    Compute the objective used in `evaluate_sequence` for a normalised
    (sum ≈ 1) non‑negative sequence.  This avoids the overhead of the
    security checks inside the provided function.
    """
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    b = np.convolve(seq, seq)
    max_b = b.max()
    return float(2.0 * seq.size * max_b / (s * s))


def _make_general_sequence(n: int, p: float, offset: float, lam: float) -> np.ndarray:
    """
    Return a normalised sequence a_i ∝ (i+1+offset)^{-p}·exp(-lam·i).
    All parameters are non‑negative.  The returned array sums to 1.
    """
    idx = np.arange(n, dtype=np.float64)
    base = (idx + 1.0 + offset) ** (-p)
    if lam != 0.0:
        base *= np.exp(-lam * idx)
    total = base.sum()
    if total == 0.0:
        # degenerate case – fall back to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return base / total


def _smooth_gradient_step(a: np.ndarray, tau: float) -> np.ndarray:
    """
    One gradient step for the smoothed max‑convolution objective.
    Returns the gradient (no step size applied).
    """
    n = a.size
    conv_full = np.convolve(a, a)                     # length 2n‑1
    max_val = conv_full.max()
    # soft‑max weights over the convolution entries
    # shift by max_val for numerical stability (weights ≤ 1)
    w = np.exp(tau * (conv_full - max_val))
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.zeros_like(a)
    w /= w_sum
    # gradient = 2 * conv_rev(a, w)  (take the centre slice of length n)
    grad_full = np.convolve(a[::-1], w)               # length 2n‑1 + n‑1
    start = n - 1
    grad = 2.0 * grad_full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The routine spends the allotted time searching
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # give a tiny safety margin for the wrapper that calls us
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Randomised parameter sweep (fast)
    # ------------------------------------------------------------------
    # Reasonable ranges for the parameters
    length_choices = [1200, 1500, 1800, 2000, 2500, 3000, 3500, 4000, 4500, 5000]
    # main loop – continue sampling until time runs out
    while time.monotonic() < deadline:
        # pick a length, favouring the middle of the range
        n = int(rng.choice(length_choices))

        # power‑law exponent: 0.35‑0.80 is the region used previously,
        # but we also explore a little beyond it.
        p = float(rng.uniform(0.35, 0.80))

        # additive offset (shifts the first term away from 1)
        offset = float(rng.uniform(0.0, 20.0))

        # tiny exponential decay – usually zero, but a small value can
        # break exact symmetry.
        lam = float(rng.uniform(0.0, 0.001))

        # build the candidate
        cand = _make_general_sequence(n, p, offset, lam)

        # quick evaluation
        score = _objective(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[rand] new best_score={best_score:.6f}  n={n}  p={p:.4f} "
                  f"off={offset:.2f}  λ={lam:.6f}")

        # ------------------------------------------------------------------
        # 2) Local refinement on promising candidates
        # ------------------------------------------------------------------
        # When a candidate improves the best score we launch a short
        # gradient‑descent refinement.  The refinement is lightweight
        # (few hundred iterations) and uses the smoothed max‑convolution.
        if score <= best_score + 0.01:  # only refine when we are close to best
            local_a = cand.copy()
            # start with a modest temperature and increase it gradually
            tau = 10.0
            for it in range(250):                     # ≤ a few hundred steps
                if time.monotonic() > deadline:
                    break
                grad = _smooth_gradient_step(local_a, tau)
                # simple step‑size schedule (decreases with √iteration)
                alpha = 0.12 / math.sqrt(it + 1.0)
                local_a = local_a - alpha * grad
                # project back onto the simplex to keep feasibility
                local_a = _simplex_projection(local_a)
                # occasionally evaluate the true objective
                if it % 20 == 0:
                    cur_score = _objective(local_a)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = local_a.copy()
                        print(f"[refine] new best_score={best_score:.6f}  n={n}  it={it}")
                # anneal temperature – higher τ gives a sharper max‑approximation
                tau += 0.5

    # ----------------------------------------------------------------------
    # 3) Fallback – uniform sequence if nothing was found (should never happen)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 256
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _objective(best_seq)
        print("[fallback] using uniform sequence")

    # --------------------------------------------------------------
    # Final sanity check with the official evaluator (optional but cheap)
    # --------------------------------------------------------------
    try:
        # The provided `evaluate_sequence` function is assumed to be in the global scope.
        from __main__ import evaluate_sequence  # type: ignore
        final_score = float(evaluate_sequence(best_seq.tolist()))
        if not math.isfinite(final_score):
            # Should not happen – keep the previously computed score
            final_score = best_score
        print(f"[final] best_score after full evaluation = {final_score:.6f}")
    except Exception:
        # If the evaluator is not importable (e.g., in certain test harnesses)
        # we just rely on the fast internal computation.
        final_score = best_score
        print(f"[final] internal best_score = {final_score:.6f}")

    # Convert to plain Python floats for the caller
    return [float(x) for x in best_seq]
