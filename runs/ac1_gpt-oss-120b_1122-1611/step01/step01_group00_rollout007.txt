    # possible start from previous best if global variable height_sequence_1 present.
    try: # accessing global

"""
Propose a low‑score height sequence for the self‑convolution objective.

The algorithm works in three stages:

1. **Initialisation** – a few hand‑crafted patterns (uniform,
   arcsine‑shaped, triangular) and a random Dirichlet vector are created.
   The best of these initial patterns (including the global
   `height_sequence_1` if present) is taken as the starting point.

2. **Smoothed projected Adam optimisation** – at each iteration the
   full self‑convolution `c = a*a` is computed, a smooth approximation of
   its maximum is built by a soft‑max (temperature µ).  The resulting
   weight vector `w` (∂/∂c) yields the gradient
   `∇a = 2·corr(w, a)` (correlation of `w` with `a`).  An Adam step is
   taken and the iterate is projected back onto the probability simplex
   `{a_i ≥ 0, Σ a_i = 1}`.

   * µ is slowly increased so the optimisation first follows a smooth
     landscape and later focuses on the true peak.
   * The learning‑rate decays as 1/√t.
   * Every few iterations a cheap evaluation of the true objective is
     performed; the best sequence found so far is retained.
   * Small multiplicative perturbations are injected with low probability
     to escape plateaus.

3. **Termination** – the loop runs until the allocated wall‑clock budget
   expires (default ≈ 1000 s).  If the target score (≈ 1.5025) is reached
   early, the current best sequence is returned immediately.

The method improves on a plain sub‑gradient descent by using a smooth
max surrogate, adaptive Adam updates and several diverse starts.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function height sequence that minimises
    `evaluate_sequence`.  Returns the best sequence discovered before the
    time budget expires.

    Parameters
    ----------
    seed : int, default 42
        Random seed for reproducibility.
    budget_s : float, default 1100
        Wall‑clock time (seconds) allocated for the search.  A small
        safety margin is kept for the final return.

    Returns
    -------
    list[float]
        The candidate sequence (list of floats, sum ≈ 1, all entries ≥ 0).
    """
    rng = np.random.default_rng(seed)

    # Reserve a couple of seconds for clean‑up/printing.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # Problem size – 1024 provides a good trade‑off between flexibility
    # and per‑iteration cost.
    n = 1024

    # ------------------------------------------------------------------
    # Build a collection of diverse starting points.
    patterns = []

    # Uniform distribution (guarantees feasibility)
    patterns.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Arcsine‑shaped: heavy at the ends, light in the middle
    i = np.arange(n, dtype=np.float64)
    arcsine = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    arcsine /= arcsine.sum()
    patterns.append(arcsine)

    # Simple triangular (tent) shape – rises then falls linearly
    half = n // 2
    tent = np.concatenate([np.linspace(1, 2, half, endpoint=False),
                           np.linspace(2, 1, n - half)])
    tent /= tent.sum()
    patterns.append(tent)

    # Random Dirichlet vector – provides stochastic diversity
    patterns.append(rng.dirichlet(np.ones(n)).astype(np.float64))

    # If the previous best solution is available, try it as a start.
    if 'height_sequence_1' in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            if prev.size != n:
                # Resize by simple interpolation / truncation.
                if prev.size < n:
                    # pad at the end with a tiny uniform tail
                    pad = np.full(n, 1.0 / n, dtype=np.float64)
                    pad[:prev.size] = prev
                    prev = pad
                else:
                    prev = prev[:n]
            # Normalise to ℓ₁ = 1
            prev = _project_simplex(prev)
            patterns.append(prev)
        except Exception:
            pass  # ignore malformed global variable

    # ------------------------------------------------------------------
    # Optimisation state that will survive across restarts.
    best_seq = None
    best_score = float('inf')
    total_iters = 0

    # Adam hyper‑parameters (the same for every restart).
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # ------------------------------------------------------------------
    # Main optimisation loop – we repeatedly pick a start pattern,
    # re‑initialise Adam and run a bounded number of inner iterations.
    # As soon as the time budget expires we fall back to the best solution.
    while time.monotonic() < deadline:
        # Randomly pick a starting pattern (gives implicit restarts).
        a = rng.choice(patterns).copy()
        a = _project_simplex(a)               # ensure feasibility
        m = np.zeros_like(a)                  # Adam first moment
        v = np.zeros_like(a)                  # Adam second moment

        # Roughly allocate 2000 inner steps per restart; the loop will
        # break early if the global time budget runs out.
        inner_limit = 3000
        for it in range(inner_limit):
            if time.monotonic() >= deadline:
                break

            total_iters += 1

            # ----------------------------------------------------------
            # 1. Full self‑convolution (linear time O(n²) via NumPy C code)
            conv = np.convolve(a, a)                     # length 2n‑1

            # 2. Smooth max – soft‑max with a temperature that grows slowly
            #    (start modest to get a useful gradient, then focus).
            mu = 5.0 * (1.0 + total_iters / 80000.0)     # slowly rising µ
            cmax = conv.max()
            # stabilise the exponential
            exp_args = mu * (conv - cmax)
            w = np.exp(exp_args)
            w /= w.sum()                                 # ∂f/∂conv

            # ----------------------------------------------------------
            # 3. Gradient of the smooth max (via correlation)
            #    grad = 2 * corr(w, a) for lags 0 … n‑1
            corr_full = np.correlate(w, a, mode='full')
            grad = 2.0 * corr_full[n - 1: n - 1 + n]      # length n

            # ----------------------------------------------------------
            # 4. Adam update on the simplex (project after the step)
            lr = 0.2 / np.sqrt(total_iters)               # decaying step‑size
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----------------------------------------------------------
            # 5. Projection back onto the probability simplex
            a = _project_simplex(a)

            # ----------------------------------------------------------
            # 6. Occasional random shake‑up (helps escape shallow plateaux)
            if rng.random() < 0.008:
                idx = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.18))
                a[idx] *= factor
                a = _project_simplex(a)

            # ----------------------------------------------------------
            # 7. Periodic evaluation of the true (non‑smooth) objective.
            #    Doing this every ≈20 iterations keeps overhead low.
            if total_iters % 20 == 0:
                score = float(evaluate_sequence(a.tolist()))
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_seq = a.copy()
                    print(f"[improve] iter={total_iters}, score={best_score:.6f}")

                # Early exit if we have already beaten the target.
                if best_score <= 1.5025:
                    print("[early exit] target reached")
                    return [float(x) for x in best_seq]

    # ------------------------------------------------------------------
    # If no candidate was ever accepted (unlikely), fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    print(f"[final] total_iters={total_iters}, best_score={best_score:.6f}")
    return [float(x) for x in best_seq]
