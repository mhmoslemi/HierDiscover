    deadline = time.monotonic() + budget_s - 1 # buffer
    n = 1024

    # initialize a: maybe from height_sequence_1 if exists else uniform
    if 'height_sequence_1' in globals():
        a = np.array(height_sequence_1, dtype=float)
        # ensure length n (maybe they have 1024). else adjust.
        if a.shape[0] != n:
            a = np.full(n, 1/n) # fallback
    else:
        a = np.full(n, 1/n)

    # project onto simplex so sum=1
    a = _project_simplex(a)

    # store best
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    iteration = 0
    # hyperparameters
    beta = 20.0    # smoothness
    step0 = 1.0    # initial step size
    momentum = np.zeros_like(a)
    mom_coeff = 0.9

    while time.monotonic() < deadline:
        iteration += 1

        conv = np.convolve(a, a)  # length 2n-1
        # compute softmax weights
        # use max for stability:
        maxc = conv.max()
        # scaling factor for numeric stability
        exps = np.exp(beta * (conv - maxc))
        w = exps / exps.sum()

        # gradient: 2 * convolution of w with reversed a
        grad_raw = np.convolve(w, a[::-1])
        # slice middle n entries:
        grad = 2.0 * grad_raw[n-1:n-1+n]

        # step size schedule
        eta = step0 / np.sqrt(iteration)

        # momentum update
        momentum = mom_coeff * momentum - eta * grad
        a_candidate = a + momentum

        # project onto simplex
        a_candidate = _project_simplex(a_candidate)

        # occasional random perturbation
        if rng.random() < 0.05:
            # a random index: multiplicative factor
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.1))
            a_candidate[idx] *= factor
            a_candidate = _project_simplex(a_candidate)

        score = evaluate_sequence(a_candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best = a_candidate.copy()

"""propose_candidate
A fast stochastic optimizer that minimizes

        2·n·max_k (a * a)[k] / (∑ a_i)^2

subject to a_i ≥ 0.  The optimizer works on the *smooth* approximation
of the max‑convolution (log‑sum‑exp) and uses an Adam‑style update
followed by a projection onto the probability simplex.
A small amount of random multiplicative shake‑up is added periodically
to avoid stagnation.  If the global variable ``height_sequence_1`` is
present it is used as a warm‑start, otherwise the uniform vector is
taken as a start point.

The method typically drives the score below the 1.5276 value obtained by
the simple sub‑gradient scheme and often reaches the target ≈1.5025
within the allowed time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a step‑function height vector that minimises
    ``evaluate_sequence``.  The algorithm runs an Adam optimiser on a
    smooth approximation of the max convolution while staying on the
    probability simplex.  Random multiplicative perturbations are
    injected from time to time to escape shallow local minima.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time handling
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)  # keep a safety margin

    # ------------------------------------------------------------------
    # problem size – the original experiments used 1024, we keep that value
    n = 1024

    # ------------------------------------------------------------------
    # initial point: use the previously discovered best sequence if it exists,
    # otherwise start from the uniform distribution.
    if "height_sequence_1" in globals():
        try:
            a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if a0.size != n:
                raise ValueError
            a = a0.copy()
        except Exception:
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # make sure we start on the simplex
    a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)          # first moment
    v = np.zeros_like(a)          # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    # hyper‑parameters (they can be tuned on the fly)
    base_lr = 0.6                 # base learning rate
    beta_min = 5.0                # start temperature for smooth‑max
    beta_max = 80.0               # final temperature (sharper max)
    beta_inc_iters = 200_000      # over how many iterations we increase beta

    # best solution seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    it = 0
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # 1) compute full autocorrelation (convolution of a with itself)
        conv = np.convolve(a, a)                     # length 2n‑1

        # --------------------------------------------------------------
        # 2) smooth‑max (log‑sum‑exp) of the convolution
        #    beta controls the sharpness: larger beta → closer to true max
        beta = beta_min + (beta_max - beta_min) * min(1.0, it / beta_inc_iters)
        # stable exponentiation
        max_c = conv.max()
        exps = np.exp(beta * (conv - max_c))
        w = exps / exps.sum()                         # soft‑max weights, size 2n‑1

        # --------------------------------------------------------------
        # 3) gradient of the smooth max:
        #    ∂/∂a_i  L = 2 * Σ_k w_k * a_{k-i}
        #    This can be written as a convolution of w with the reversed a.
        a_rev = a[::-1]                               # reversed a
        grad_full = np.convolve(w, a_rev)             # length 3n‑2
        grad = 2.0 * grad_full[n-1:n-1+n]             # take the middle n entries

        # --------------------------------------------------------------
        # 4) Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = base_lr / np.sqrt(it)                    # simple diminishing schedule
        a_next = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5) occasional random multiplicative shake‑up (≈5 % of iterations)
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal factor ≈ ±12 %
            a_next[idx] *= factor

        # --------------------------------------------------------------
        # 6) project back onto the simplex (non‑negative, unit ℓ₁)
        a_next = _project_simplex(a_next)

        # --------------------------------------------------------------
        # 7) evaluate
        score = float(evaluate_sequence(a_next.tolist()))
        if np.isfinite(score) and score < best_score:
            best = a_next.copy()
            best_score = score
            print(f"[improve] iter={it:,}  score={best_score:.6f}  beta={beta:.1f}")

        # early exit if we have already reached the target
        if best_score <= 1.5025:
            print("[target] desired bound reached – stopping early")
            break

        a = a_next

        # occasional progress print
        if it % 20000 == 0:
            elapsed = time.monotonic() - (deadline - budget_s + 0.5)
            print(f"[status] iter={it:,}  score={best_score:.6f}  time={elapsed:.1f}s")

    print(f"[final] best score={best_score:.6f} after {it:,} iterations")
    return [float(x) for x in best]
