def propose_candidate(seed=123, budget_s=1100):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, budget_s - 2.0)

    n = 2000 (or 2048)  # choose 2048
    # init with sqrt series
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1,n):
        a[i] = a[i-1] * (2*i - 1) / (2*i)  # using recurrence (2*i-1)/(2*i)
    a = a / np.sum(a)  # normalize sum to 1

    # optionally small random perturb to break symmetry
    a *= np.exp(rng.normal(0, 0.02, size=n))
    a = _project_simplex(a)  # keep sum=1

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"[init] n={n}, score={best_score}")

    # parameters
    eta0 = 0.2
    alpha = 5.0
    momentum = 0.9
    v = np.zeros_like(a)
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        conv = np.convolve(a, a)
        max_conv = np.max(conv)

        # Compute softmax weights
        shifted = conv - max_conv
        w = np.exp(alpha * shifted)  # w_i between 0 and 1
        w_sum = np.sum(w)
        if w_sum == 0:
            w = np.ones_like(conv) / conv.size
        else:
            w /= w_sum

        # Gradient of weighted sum: grad = 2 * reversed_convolution(w, a)
        w_rev = w[::-1]  # length 2n-1
        conv2 = np.convolve(w_rev, a)  # length 3n-2
        grad = 2.0 * np.flip(conv2[n-1:2*n-1])

        # Diminishing step size
        eta = eta0 / np.sqrt(iteration)

        # momentum update
        v = momentum * v - eta * grad
        a_candidate = a + v
        a_candidate = _project_simplex(a_candidate)

        # occasional larger perturbation
        if rng.random() < 0.05:
            pos = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            a_candidate[pos] *= factor
            a_candidate = _project_simplex(a_candidate)

        score = evaluate_sequence(a_candidate.tolist())

        if np.isfinite(score) and score < best_score:
            best = a_candidate.copy()
            best_score = score
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # Update a to candidate for next iteration
        a = a_candidate

        # optional schedule for alpha: increase gradually to sharpen focus
        if iteration % 1000 == 0:
            alpha = min(alpha * 1.02, 50.0)

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]

"""
Propose a high‑quality step‑function by minimizing the worst convolution
entry with a soft‑max weighted gradient descent.  The algorithm
* starts from a deterministic “square‑root” sequence a_i ∝
  (2i choose i)/4^i (the coefficients of (1‑x)^{-1/2}), which already yields
  a fairly flat autocorrelation (score ≈ π/2);
* then repeatedly reduces a smooth approximation of the maximum
  convolution entry using a projected gradient step with momentum;
* the soft‑max temperature (α) is increased during the run so that the
  optimizer first flattens many peaks and finally concentrates on the true
  maximum;
* occasional random multiplicative shocks and a restart‑on‑stagnation
  mechanism help to escape local minima;
* the vector is always projected onto the probability simplex
  (non‑negative, ℓ₁‑norm = 1), which guarantees validity for
  `evaluate_sequence`.

The search runs until the supplied time budget expires (default ~1000 s)
or until the theoretical lower bound 2 n/(2 n‑1) is reached within 1 %.
The best sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# --------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Generate a step‑function height sequence that (approximately) minimizes
    `evaluate_sequence`.  The routine follows the soft‑max weighted gradient
    descent described in the module docstring.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # time budget
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # ------------------------------------------------------------------
    # problem size ------------------------------------------------------
    # If a previously discovered sequence exists we reuse its length,
    # otherwise we pick a generous size that still fits comfortably inside
    # the time budget.
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        try:
            a = np.asarray(height_sequence_1, dtype=np.float64).copy()
            # make sure it is feasible (non‑negative, ℓ₁‑norm = 1)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.full(n, 1.0 / n)
            else:
                a = a / s
        except Exception:
            # fallback to deterministic start
            height_sequence_1 = None

    if "height_sequence_1" not in globals() or height_sequence_1 is None:
        # ----- a deterministic “square‑root” start (truncated (1‑x)^‑½) -----
        n = 2048                     # a reasonable compromise between quality and speed
        a = np.empty(n, dtype=np.float64)
        a[0] = 1.0
        # recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1))
        for i in range(1, n):
            a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
        a = a / a.sum()               # normalise to ℓ₁‑norm = 1
        # a tiny random shake‑up to break perfect symmetry
        a *= np.exp(rng.normal(0.0, 0.02, size=n))
        a = _project_simplex(a)

    # ------------------------------------------------------------------
    # bookkeeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # optimizer hyper‑parameters
    eta0 = 0.25                     # initial step size
    momentum = 0.85                 # classical momentum factor
    velocity = np.zeros_like(a)    # momentum buffer
    alpha = 2.0                     # soft‑max temperature (low → smooth)
    iteration = 0
    last_improve_iter = 0

    # main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1️⃣  current convolution
        conv = np.convolve(a, a)                # length = 2·n‑1
        max_conv = conv.max()

        # --------------------------------------------------------------
        # 2️⃣  soft‑max weights over the convolution entries
        #    w_i ∝ exp(α·(conv_i – max_conv)), thus max entry gets weight 1
        shifted = conv - max_conv
        # to keep numbers in a safe range we subtract the maximum before exp
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:                       # should never happen, but be safe
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # 3️⃣  gradient of the weighted sum  g(a) = Σ w_k·(a∗a)_k
        #    ∇g = 2·reverse(w) ∗ a  (restricted to the original n entries)
        w_rev = w[::-1]                         # length = 2·n‑1
        tmp = np.convolve(w_rev, a)             # length = 3·n‑2
        grad = 2.0 * np.flip(tmp[n - 1 : 2 * n - 1])

        # --------------------------------------------------------------
        # 4️⃣  step‑size schedule (diminishing) and momentum update
        eta = eta0 / np.sqrt(iteration)
        velocity = momentum * velocity - eta * grad
        a_candidate = a + velocity

        # --------------------------------------------------------------
        # 5️⃣  keep the iterate feasible
        a_candidate = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        # 6️⃣  occasional random “shake‑up” to improve exploration
        if rng.random() < 0.04:

            factor = np.exp(rng.normal(0.0, 0.25))
            a_candidate[i] *= factor
            a_candidate = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        # 7️⃣  evaluate
        score = float(evaluate_sequence(a_candidate.tolist()))

        # --------------------------------------------------------------
        # 8️⃣  bookkeeping
        if np.isfinite(score) and score < best_score:
            best = a_candidate.copy()
            best_score = score
            last_improve_iter = iteration
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 9️⃣  prepare for next iteration
        a = a_candidate

        # --------------------------------------------------------------
        # 10️⃣  dynamically increase the soft‑max temperature – sharper focus
        if iteration % 2000 == 0:
            alpha = min(alpha * 1.15, 60.0)

        # --------------------------------------------------------------
        # 11️⃣  restart on stagnation (no improvement for a long stretch)
        if iteration - last_improve_iter > 50000:
            # re‑inject a small random perturbation around the best found so far
            a = best.copy()
            # add a tiny Gaussian noise and re‑project
            a += 0.005 * rng.normal(size=n)
            a = _project_simplex(a)
            last_improve_iter = iteration
            print(f"[restart] iter={iteration}, re‑initialising from best")

        # --------------------------------------------------------------
        # 12️⃣  early‑stop if we reach the (obviously unattainable) theoretical bound
        theoretical_lb = 2.0 * n / (2.0 * n - 1.0)
        if best_score <= theoretical_lb * 1.01:
            print("[early stop] reached near‑optimal theoretical bound")
            break

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # return a plain Python list (the framework expects a list of floats)
    return [float(x) for x in best]
