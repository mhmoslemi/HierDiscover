def refine_seq(seq, max_iter, rng, deadline):
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    for t in range(max_iter):
        if time.monotonic() >= deadline: break
        # subgradient update
        n = best_seq.shape[0]
        b = np.convolve(best_seq, best_seq)
        max_idx = int(np.argmax(b))
        # gradient g = 2 * reversed shift of best_seq
        g = np.zeros_like(best_seq)
        # compute contributions
        lo = max(0, max_idx - (n-1))
        hi = min(n-1, max_idx)
        # For i such that k-i in range => i in [max_idx-(n-1), max_idx]
        # So for i in [lo,hi]:
        # g[i] = 2 * best_seq[max_idx - i]
        i_range = np.arange(lo, hi+1)
        g[i_range] = 2 * best_seq[max_idx - i_range]
        # step size schedule
        eta = 0.01 / (1.0 + t*0.001)   # small decreasing
        # exponentiated gradient update
        cand = best_seq * np.exp(-eta * g)
        # renorm
        cand_sum = cand.sum()
        if cand_sum <= 0:
            # reinitialize random
            cand = _make_power_sequence(n, rng.uniform(0.35,0.80))
        else:
            cand /= cand_sum
        # evaluate
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            # maybe reset step size?
    return best_seq, best_score

"""
Hybrid search for a low‑value step‑function.

The algorithm works in three stages:

1. **Coarse grid** – generate a few families of analytically
   motivated sequences (power‑law, symmetric beta, symmetric exponential)
   for several lengths, keep the best candidate.

2. **Mirror‑descent refinement** – treat the objective
   `max(conv(a,a))` (the sum is scale‑invariant) and perform a
   sub‑gradient descent on the simplex.
   The sub‑gradient of the maximum convolution entry is
   `g_i = 2·a_{k*‑i}` where `k*` is the index of the current peak.
   An exponentiated‑gradient (mirror descent) step
   `a_i ← a_i·exp(‑η·g_i)` followed by renormalisation keeps `a≥0`
   and `∑a=1`.  A softened soft‑max weight gives a smoother
   descent when the peak is broad.

3. **Stochastic perturbations** – occasional random multiplicative
   block tweaks (the “block‑perturbation” used in the reference
   code) help escape shallow local minima.

All work stops before the allotted time budget (≈1000 s) and the
best sequence found is returned as a plain Python list.
"""

def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum‑1."""
    idx = np.arange(n) + 1.0
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _make_beta_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Symmetric Beta(α,β) density discretised on the interval (0,1).
    Uses mid‑point rule to avoid exact 0/1.
    """
    # points in (0,1) – avoid the boundaries where the pdf may blow up
    x = (np.arange(n) + 0.5) / float(n)
    # Beta pdf (unnormalised, we will normalise anyway)
    seq = (x ** (alpha - 1.0)) * ((1.0 - x) ** (beta - 1.0))
    # Guard against NaNs when α or β < 1 (division by zero at the exact ends)
    seq = np.where(np.isnan(seq), 0.0, seq)
    seq /= seq.sum()
    return seq


def _make_exponential_sequence(n: int, lam: float) -> np.ndarray:
    """
    Symmetric two‑sided exponential shape:
        a_i ∝ λ^i + λ^{n‑1‑i}
    """
    i = np.arange(n)
    seq = lam ** i + lam ** (n - 1 - i)
    seq /= seq.sum()
    return seq

# Core optimisation utilities
# ----------------------------------------------------------------------
def _softmax_weights(b: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weighting of the convolution vector b."""
    m = b.max()
    # stability: subtract the max before exponentiation
    w = np.exp(beta * (b - m))
    s = w.sum()
    return w / s if s > 0 else w


def _gradient_via_softmax(a: np.ndarray, b: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute a smooth sub‑gradient of max(conv(a,a)) using a soft‑max
    over the convolution entries.
    """
    n = a.shape[0]
    # soft‑max weights on the full convolution (length 2n‑1)
    w = _softmax_weights(b, beta)

    # Gradient: 2 * (w * a) convolved with a reversed.
    # Using the identity g_i = 2 * Σ_k w_k a_{k‑i}.
    # This is the cross‑correlation of w with a.
    conv_rev = np.convolve(w, a[::-1])
    g = 2.0 * conv_rev[n - 1:n - 1 + n]
    return g


def _mirror_descent_step(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient (mirror‑descent) update on the simplex.
    Returns a new normalised vector.
    """
    # clip the exponent to avoid under‑/overflow
    factor = np.exp(np.clip(-eta * g, -50.0, 0.0))
    a_new = a * factor
    s = a_new.sum()
    if s > 0.0:
        a_new /= s
    else:
        # fallback: uniform distribution
        a_new = np.ones_like(a) / a.shape[0]
    return a_new

    the security checks and list conversion.

    n = seq.shape[0]
    b = np.convolve(seq, seq)
    max_b = b.max()
    s = seq.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)

                best_seq, best_score = seq.copy(), sc
                print(f"[grid] exp    n={n} λ={lam:.4f} -> {sc:.6f}")

        # ---- Random Dirichlet draws (few) ------------------------------------
        for _ in range(5):
            if not time_ok():
                break
            seq = rng.dirichlet(np.ones(n))
            sc = _evaluate_score(seq)
            if sc < best_score:
                best_seq, best_score = seq.copy(), sc
                print(f"[grid] rand   n={n} -> {sc:.6f}")

        print(f"[grid] completed length={n}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2 – iterative refinement of the current best using mirror descent.
    # ------------------------------------------------------------------
    iteration = 0
    # parameters for the descent
    eta0 = 0.06          # initial step size
    beta0 = 5.0          # soft‑max temperature
    while time.monotonic() < deadline:
        if best_seq is None:
            # Should not happen, but guard against it
            n = rng.choice(lengths)
            best_seq = _make_power_sequence(n, 0.6)
            best_score = _evaluate_score(best_seq)

        n = best_seq.shape[0]

        # Compute convolution and current score (reuse for gradient)
        b = np.convolve(best_seq, best_seq)          # length 2n‑1
        cur_score = 2.0 * n * b.max() / (best_seq.sum() ** 2)

        # ----------------------------------------------------------------
        # Gradient step (soft‑max) + mirror‑descent update
        # ----------------------------------------------------------------
        beta = beta0                         # keep temperature fixed (could anneal)
        g = _gradient_via_softmax(best_seq, b, beta)

        # simple diminishing step size
        eta = eta0 / (1.0 + 0.001 * iteration)

        cand = _mirror_descent_step(best_seq, g, eta)

        # ----------------------------------------------------------------
        # Occasionally apply a random block multiplicative perturbation
        # (helps to escape shallow basins)
        # ----------------------------------------------------------------
        if rng.random() < 0.15:
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 30))
            stop = min(n, start + width)
            factor = math.exp(rng.normal(0.0, 0.07))
            cand2 = cand.copy()
            cand2[start:stop] *= factor
            cand2 = np.clip(cand2, 0.0, 1000.0)
            s = cand2.sum()
            if s > 0.0:
                cand2 /= s
            cand = cand2

        # ----------------------------------------------------------------
        # Evaluate candidate (fast internal evaluation)
        # ----------------------------------------------------------------
        cand_score = _evaluate_score(cand)

        # Accept if we improved (greedy acceptance)
        if cand_score < best_score:
            best_seq, best_score = cand.copy(), cand_score
            print(f"[refine] iter={iteration:05d} new best={best_score:.6f} (n={n})")
        else:
            # Occasionally replace the current state with the best so far
            # to avoid drifting too far.
            if rng.random() < 0.02:
                cand = best_seq.copy()

        iteration += 1

        # Occasionally re‑seed from a fresh random start if no progress for a while
        if iteration % 2000 == 0:
            # try a new random Dirichlet of the same length
            fresh = rng.dirichlet(np.ones(n))
            fresh_score = _evaluate_score(fresh)
            if fresh_score < best_score:
                best_seq, best_score = fresh.copy(), fresh_score
                print(f"[restart] new random start at iter={iteration}, score={fresh_score:.6f}")

    # ------------------------------------------------------------------
    # 3 – safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _make_power_sequence(256, 0.6)

    # Convert to plain Python floats for the caller
    return [float(x) for x in best_seq]
