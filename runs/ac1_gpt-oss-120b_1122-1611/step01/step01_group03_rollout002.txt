def project_onto_simplex(v):
    """Project vector v onto simplex {x>=0, sum x=1}."""
    # standard algorithm
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑descent search on the probability simplex.
The objective 2·n·max(conv(a,a)) (sum(a)=1) is minimised by
projected Adam updates where the gradient of a smooth
approximation of the max (soft‑max with a temperature that
grows during the run) is computed via FFT‑based convolutions.
Occasional random perturbations and a restart rule avoid
getting stuck.  The routine starts from the known “alpha”
sequence (or from height_sequence_1 if it exists) and runs
until the time budget expires, returning the best sequence
found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i) / 4**i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum x = 1}."""
    # implementation of the algorithm by Wang & Carreira‑Perpinán (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def self_convolution_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using a power‑of‑two FFT (real)."""
    n = a.shape[0]
    N = 1 << ((2 * n - 1).bit_length())  # smallest power of two ≥ 2n‑1
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)
    return c[: 2 * n - 1]


def gradient_of_smooth_max(a: np.ndarray, conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Compute ∇_a  softmax_β(conv)  where conv = a * a.
    The gradient is 2 * (w * a_rev)  with w_k ∝ exp(β·conv_k).
    """
    n = a.shape[0]
    # soft‑max weights (numerically stable)
    max_c = conv.max()
    w_exp = np.exp(beta * (conv - max_c))
    w = w_exp / w_exp.sum()                     # shape (2n‑1,)

    # convolution w * a_rev  (a_rev = a reversed)
    a_rev = a[::-1]
    M = 1 << ((len(w) + n - 1).bit_length())
    W = np.fft.rfft(w, M)
    A_rev = np.fft.rfft(a_rev, M)
    full = np.fft.irfft(W * A_rev, M)

    start = n - 1
    grad = 2.0 * full[start : start + n]       # shape (n,)
    return grad


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the provided evaluation function.  The search runs
    until the allotted CPU time expires and returns the best
    sequence observed.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the problem size and initialise a starting point
    # ------------------------------------------------------------------
    global height_sequence_1
    if isinstance(height_sequence_1, (list, np.ndarray)) and len(height_sequence_1) > 0:
        n = int(len(height_sequence_1))
        a = np.asarray(height_sequence_1, dtype=np.float64).copy()
    else:
        n = 6000                     # a safe default that works well in practice
        a = generate_alpha_sequence(n)

    # Normalise to L1 = 1 (the objective is scale‑invariant)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:                     # fall back to uniform if everything vanished
        a = np.full(n, 1.0 / n)
    else:
        a = a / s

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Optimisation hyper‑parameters
    # ------------------------------------------------------------------
    # Adam
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                     # base learning rate – will be annealed

    # Soft‑max temperature (β) – starts low, grows slowly
    beta_temp = 5.0
    beta_max = 200.0

    # Book‑keeping
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    it = 0
    last_improve = 0

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ---------- evaluate current candidate ----------
        conv = self_convolution_fft(a)          # length 2n‑1
        max_conv = conv.max()
        score = 2.0 * n * max_conv               # sum(a)=1 ⇒ exact objective
        if score < best_score - 1e-12:
            best = a.copy()
            best_score = score
            last_improve = it
            print(f"[{it:6d}] new best = {best_score:.6f}")

        # ---------- compute a gradient step ----------
        # gradually increase the temperature to approach the true max
        beta_temp = min(beta_max, beta_temp * 1.001)

        grad = gradient_of_smooth_max(a, conv, beta_temp)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = a - step
        a_new = simplex_projection(a_new)

        # Occasionally inject a small random jitter to avoid plateaus
        if rng.random() < 0.02:
            jitter = 0.05 * rng.standard_normal(n)
            a_jit = a_new * (1.0 + jitter)
            a_new = simplex_projection(a_jit)

        a = a_new

        # ---------- simple restart if we are stuck ----------
        if it - last_improve > 50000:
            # re‑initialise around the current best with a tiny perturbation
            a = best + 0.01 * rng.standard_normal(n)
            a = np.clip(a, 0.0, None)
            a = simplex_projection(a)
            print(f"[{it:6d}] restart after stagnation")
            last_improve = it

        # anneal learning‑rate a little from time to time
        if it % 20000 == 0:
            lr *= 0.7

    # ------------------------------------------------------------------
    # Final evaluation (using the provided wrapper for safety)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best.tolist())
    print(f"[final] iterations={it}, best score={final_score:.6f}")
    return best.tolist()

"""
Gradient‑descent search on the probability simplex.

The evaluation function (with a normalised L1‑norm) reduces to

    score = 2·n·max(conv(a,a))

where conv(a,a) is the linear convolution of a with itself.
The algorithm therefore minimises  max(conv) while keeping a on the
simplex {a≥0, Σa=1}.  A smooth approximation of the max (soft‑max with a
temperature that grows during the run) provides a differentiable
objective; its gradient can be written in closed form as a second
convolution.  Both convolutions are evaluated with FFTs, so each
iteration costs O(n log n).

Optimization is performed with projected Adam updates, occasional
random jitter to escape plateaus and a simple restart rule when no
