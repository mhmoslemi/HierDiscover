def propose_candidate(seed=0, budget_s=1000, length=256, ...):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 0.5

    # initialize a: uniform
    a = np.full(length, 1.0/length)

    # Possibly use previous best if provided globally.
    try:
        prev = np.array(height_sequence_1, dtype=float)
        # if its length different, we adjust or use as starting point
        if len(prev) <= length:
            a[:len(prev)] = prev
            a = a / a.sum()
    except NameError:
        pass

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    print(f"[init] length={length}, score={best_score}")

    iteration = 0
    # main loop
    while time.monotonic() < deadline:
        iteration += 1

        # Choose technique: gradient descent 80% of time, random perturb 15%, LP 5%
        r = rng.random()
        if r < 0.78:
            # gradient descent step (smoothmax)
            conv = np.convolve(a, a)
            max_conv = conv.max()
            # smoothing parameter adapt: increase alpha as progress
            alpha = max(5.0, min(50.0, iteration / 1000.0 * 20.0))
            # compute softmax probabilities over conv
            shifted = conv - max_conv   # values ≤0
            exp_term = np.exp(alpha * shifted)
            p = exp_term / exp_term.sum()
            # gradient
            rev_p = p[::-1]
            conv_rev = np.convolve(a, rev_p)
            grad = 2.0 * conv_rev[(2*length-2) - np.arange(length)]
            # step size schedule
            eta = 0.5 / np.sqrt(iteration)
            a_candidate = a - eta * grad
            a_candidate = _project_simplex(a_candidate)
        elif r < 0.93:
            # random mass transfer (swap)
            a_candidate = a.copy()
            i, j = rng.choice(length, size=2, replace=False)
            # transfer epsilon proportion from i to j
            epsilon = rng.random() * min(a_candidate[i], 0.1)  # cap
            a_candidate[i] -= epsilon
            a_candidate[j] += epsilon
            # ensure positivity: project? just ensure nonnegative and renorm
            a_candidate = _project_simplex(a_candidate)
        else:
            # LP improvement
            # compute current max bound
            conv = np.convolve(a, a)
            M = conv.max()
            # build A_ub
            n = length
            idx = np.arange(n)
            A_ub = np.zeros((2*n-1, n), dtype=np.float64)
            for i, fi in enumerate(a):
                rows = i + idx
                A_ub[rows, idx] += fi
            # bounds for b: >=0
            bounds = [(0, None)] * n
            # objective - maximize sum(b), so minimize -sum(b)
            c = -np.ones(n)
            res = linprog(c, A_ub=A_ub, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
            if not res.success:
                # fallback to previous a
                a_candidate = a
            else:
                b = res.x
                sum_b = b.sum()
                if sum_b <= 0:
                    a_candidate = a
                else:
                    b_norm = b / sum_b
                    # convex combination step
                    t = 0.3
                    a_candidate = (1-t)*a + t*b_norm
                    a_candidate = _project_simplex(a_candidate)
        # evaluate
        score = evaluate_sequence(a_candidate.tolist())
        if np.isfinite(score) and score < best_score:
            best = a_candidate.copy()
            best_score = score
            print(f"[improve] iter={iteration}, score={best_score:.6f}")
            # maybe reset step size etc.
        # optionally accept candidate if not worse but near.
        # Update a for next iteration depending on improvement or not
        a = a_candidate
        # early stop if close to target

            print("[early stop reached target]")
            break
    print(f"[final] best score {best_score} after {iteration} iters")
    return [float(x) for x in best]

"""
A fast projected‑Adam optimiser that minimises the worst‑case self‑convolution
of a non‑negative step function.

The evaluation metric (with ℓ₁‑norm fixed to 1) is
      score = 2·n·max_k (a * a)[k] .
We therefore minimise the maximum entry of the convolution.

The algorithm:
1. Initialise a (uniform or the optional global ``height_sequence_1``) and
   normalise it to unit ℓ₁‑norm.
2. Use a smooth‑max (log‑sum‑exp) surrogate of the max‑convolution.
   Its gradient can be obtained by a single FFT‑based convolution,
   giving an **O(n log n)** step.
3. Apply Adam’s adaptive moments, then project back onto the probability
   simplex (non‑negative, unit sum).
4. Occasionally (10 % of the steps) apply a random multiplicative shake‑up
   to escape plateaus.
5. Keep the best vector seen so far and stop early if the score drops
   close to the target (≈ 1.5).

All operations respect the security checks performed inside
``evaluate_sequence``; the final returned list is ready for the scorer.
"""

import time
import math
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _full_convolution(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Return the full (linear) convolution of two real vectors using FFT.
    Length = len(x) + len(y) - 1.
    """
    n = len(x) + len(y) - 1
    # next power of two for speed
    N = 1 << (n - 1).bit_length()
    X = rfft(x, N)
    Y = rfft(y, N)
    c = irfft(X * Y, N)[:n]
    # Numerical noise can produce tiny negatives – clip.
    return np.maximum(c, 0.0)

    surrogate of the worst convolution entry, with occasional random
    perturbations.  The whole routine respects the 1000‑second time
    budget.
