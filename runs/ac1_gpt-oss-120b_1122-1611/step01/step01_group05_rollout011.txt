import time, numpy as np

def fast_max_convolution(a):
    n = a.size
    N = 2 * n - 1
    size = 1 << (N - 1).bit_length()
    A = np.fft.rfft(a, size)
    C = np.fft.irfft(A * A, size)
    conv = C[:N]
    return np.max(conv)

def normalize(a):
    s = a.sum()
    if s == 0:
        return a
    return a / s

def propose_candidate(seed=12345, budget_s=990):
    """Simulated‑annealing / gradient descent search for a step function that
    minimises ``evaluate_sequence`` (i.e. flattens its self‑convolution).
    The algorithm combines:
        * a parameter sweep over simple families (power‑law, exponential)
        * a pairwise balancing move that reduces the current maximal
          convolution entry,
        * random block‑scale / swap / jitter moves,
        * a simple Metropolis acceptance rule to escape local minima.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    # ------------------------------------------------------------------
    # Problem size and initial sequence
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                base_seq = arr.copy()
            else:
                base_seq = None
        except Exception:
            base_seq = None
    else:
        base_seq = None

    if base_seq is None:
        # default length about 6000 to give enough degrees of freedom
        n = 6400
        base_seq = np.ones(n, dtype=np.float64) / n   # uniform, already normalised sum=1
    else:
        n = base_seq.size
        base_seq = np.clip(base_seq.astype(np.float64), 0.0, 1000.0)

    # normalise to unit sum – the objective is scale invariant
    a = normalize(base_seq)
    best = a.copy()
    best_score = 2.0 * n * fast_max_convolution(best)   # internal score matching evaluate_sequence
    print(f"[init] n={n}, start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Parameter sweep – try a few analytic families
    # power‑law
    for p in np.linspace(0.15, 0.90, 16):
        cand = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        cand = normalize(cand)
        score = 2.0 * n * fast_max_convolution(cand)
        if score < best_score:
            best = cand
            best_score = score
            print(f"[sweep] power p={p:.3f} → {score:.6f}")
    # exponential decay
    for lam in np.linspace(0.0005, 0.02, 15):
        cand = np.exp(-lam * np.arange(n, dtype=np.float64))
        cand = normalize(cand)
        score = 2.0 * n * fast_max_convolution(cand)
        if score < best_score:
            best = cand
            best_score = score
            print(f"[sweep] exp λ={lam:.4f} → {score:.6f}")
    # combined power‑law + exp
    for p in (0.35, 0.45, 0.55):
        for lam in (0.001, 0.005, 0.01):
            idx = np.arange(1, n + 1, dtype=np.float64)
            cand = np.exp(-lam * (idx - 1)) / np.power(idx, p)
            cand = normalize(cand)
            score = 2.0 * n * fast_max_convolution(cand)
            if score < best_score:
                best = cand
                best_score = score
                print(f"[sweep] p={p:.2f} λ={lam:.4f} → {score:.6f}")

    # start from the best found so far
    a = best.copy()
    # ------------------------------------------------------------------
    # 2) Simulated‑annealing / gradient descent loop
    deadline = start + max(0.0, float(budget_s) - 2.0)   # keep a couple seconds spare
    iteration = 0
    # temperature schedule: start with 0.02, linearly decay to 0
    T0 = 0.02

        a_new = a.copy()

            conv = np.convolve(a, a)

            # the partner indices also get updated in the same vectorised operation,

        elif r < 0.60:

        else:

            if T > 0 and rng.random() < np.exp((best_score - cur_score) / T):
                a = a_new.copy()   # move to a worse state to escape
        iteration += 1

    # ------------------------------------------------------------------
    print(f"[done] best_score={best_score:.6f} after {iteration} iterations")
    # Return plain Python list – evaluate_sequence will clip again if needed
    return [float(x) for x in best]

"""
Propose a step‑function (list of non‑negative heights) that minimises the
evaluation function.  The method combines a cheap analytic sweep with a
simulated‑annealing / gradient‑descent loop that repeatedly flattens the
largest entry of the self‑convolution.  A “pairwise balancing’’ move
replaces each element a[i] by a convex combination of itself and its
partner at the shift where the convolution is maximal; this directly
reduces the offending product a[i]·a[j].  Random block‑scale, swap and
jitter moves keep the search diverse.  The internal objective
(2 · n · max(conv)) is computed by an FFT‑based convolution for speed;
the final list is returned in plain Python format for the supplied
evaluate_sequence() call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _max_convolution_and_index(arr: np.ndarray):
    """Return (max_value, index_of_max) of the linear convolution arr*arr."""
    n = arr.size
    # length of linear convolution = 2*n-1; use next power of two for FFT
    size = 1 << ((2 * n - 2).bit_length())
    # real FFT of the zero‑padded vector
    A = np.fft.rfft(arr, size)
    # element‑wise square (auto‑correlation in frequency domain)
    C = np.fft.irfft(A * A, size)
    conv = C[: 2 * n - 1]                     # keep only the linear part
    idx = int(np.argmax(conv))
    return float(conv[idx]), idx


def _score(arr: np.ndarray) -> float:
    """Internal fast version of evaluate_sequence for a unit‑sum vector."""
    n = arr.size
    maxb, _ = _max_convolution_and_index(arr)
    return 2.0 * n * maxb                     # identical to evaluate_sequence when sum(arr)=1


def _normalize(arr: np.ndarray) -> np.ndarray:
    """Scale to unit L1 norm (sum = 1)."""
    s = arr.sum()
    if s == 0:
        return arr
    return arr / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 990.0, **_) -> list[float]:
    """Search for a low‑evaluation step function.

    Args:
        seed: random seed.
        budget_s: time budget in seconds (a couple of seconds are kept
                  as a safety margin before the harness timeout).

    Returns:
        The best sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # Determine problem size and an initial vector.
    base_seq = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                base_seq = np.clip(arr, 0.0, 1000.0)
        except Exception:
            base_seq = None

    if base_seq is None:
        # fallback to a moderate‑size uniform vector
        n = 6400
        base_seq = np.ones(n, dtype=np.float64) / n
    else:
        n = base_seq.size

    a = _normalize(base_seq.astype(np.float64))
    best = a.copy()
    best_score = _score(best)
    print(f"[init] n={n}, start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) PARAMETER SWEEP – cheap analytic families
    # power‑law a_i = (i+1)^(-p)
    for p in np.linspace(0.15, 0.90, 16):
        cand = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        cand = _normalize(cand)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] power p={p:.3f} → {sc:.6f}")

    # exponential decay a_i = exp(-λ·i)
    for lam in np.linspace(0.0005, 0.0200, 15):
        cand = np.exp(-lam * np.arange(n, dtype=np.float64))
        cand = _normalize(cand)
        sc = _score(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] exp λ={lam:.4f} → {sc:.6f}")

    # combined power‑law + exponential
    for p in (0.35, 0.45, 0.55):
        for lam in (0.001, 0.005, 0.010):
            idx = np.arange(1, n + 1, dtype=np.float64)
            cand = np.exp(-lam * (idx - 1)) / np.power(idx, p)
            cand = _normalize(cand)
            sc = _score(cand)
            if sc < best_score:
                best, best_score = cand, sc
                print(f"[sweep] p={p:.2f} λ={lam:.3f} → {sc:.6f}")

    # start the local search from the best candidate discovered above
    a = best.copy()

    # ------------------------------------------------------------------
    # 2) MAIN SEARCH – simulated annealing + pairwise balancing
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # leave a couple of seconds spare
    iteration = 0
    T0 = 0.02                     # initial temperature for Metropolis acceptance

    while time.monotonic() < deadline:
        # ----- choose a move type -------------------------------------------------
        r = rng.random()
        a_new = a.copy()

        if r < 0.35:

            maxb, s = _max_convolution_and_index(a)          # also yields the troublesome shift s
            # blending factor – larger in early iterations, decays with sqrt(iter)
            eta = 0.30 / np.sqrt(iteration + 1.0)
            idx = np.arange(n)
            partner = s - idx
            mask = (partner >= 0) & (partner < n)
            # symmetric update using the *old* a values

            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))

            i, j = rng.integers(0, n, size=2)
            if i != j:

                a_new[i] -= delta
                a_new[j] += delta

        else:
            # -------- jitter + optional mirror blend --------------------------------
            if rng.random() < 0.5:
                a_new = 0.5 * (a + a[::-1])
            jitter = np.exp(rng.normal(0.0, 0.05, size=n))
            a_new *= jitter

        # ----- keep feasibility ----------------------------------------------------
        a_new = np.clip(a_new, 0.0, None)           # non‑negativity
        total = a_new.sum()
        if total > 1e-12:
            a_new = a_new / total                    # renormalise to unit sum
        else:
            a_new = a.copy()                         # fallback – should never happen

        # ----- evaluate ------------------------------------------------------------
        cur_score = _score(a_new)

        # ----- acceptance ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best, best_score = a_new.copy(), cur_score
            a = a_new.copy()
            print(f"[improve] iter {iteration:6d} → {best_score:.6f}")
        else:
            # Metropolis rule – allows occasional uphill moves
            remaining = max(0.0, deadline - time.monotonic())
            T = T0 * (remaining / (deadline - start_time))   # linear cooling
            if T > 0.0 and rng.random() < np.exp((best_score - cur_score) / T):
                a = a_new.copy()

        iteration += 1

    # ------------------------------------------------------------------
    print(f"[done] best_score={best_score:.6f} after {iteration} iterations")
    # Convert to pure Python list for the evaluator
    return [float(x) for x in best]
