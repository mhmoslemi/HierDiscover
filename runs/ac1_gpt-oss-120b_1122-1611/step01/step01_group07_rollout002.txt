def gradient_step(a, eps, lr):
    """Perform one mirror‑descent step on a using soft‑max of the convolution.

    Parameters
    ----------
    a : ndarray
        Current probability vector (non‑negative, sum > 0).
    eps : float
        Soft‑max temperature (larger → tighter approximation of max).
    lr : float
        Learning rate.

    Returns
    -------
    a_new : ndarray
        Updated vector (renormalised to sum 1).
    """
    # compute convolution
    b = np.convolve(a, a)                     # length 2n‑1
    # soft‑max weights (stable)
    maxb = b.max()
    w = np.exp(eps * (b - maxb))              # w_k ≤ 1, sum w ≤ len(b)
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # gradient = 2 * (w * a_rev)  (anti‑diagonal convolution)
    grad_full = np.convolve(w, a[::-1])       # length 3n‑2
    n = a.size
    grad = 2.0 * grad_full[n-1:n-1+n]         # slice length n
    # multiplicative mirror update, keep positivity
    a_new = a * np.exp(-lr * grad)
    # renormalise to keep Σa = 1 (the objective is homogeneous)
    s = a_new.sum()

        # fallback to uniform if everything vanished

    else:
        a_new = a_new / s
    return a_new
