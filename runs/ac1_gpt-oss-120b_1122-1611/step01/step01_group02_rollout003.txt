def convolve_fft(x, y):
    n = len(x) + len(y) - 1
    N = int(2**np.ceil(np.log2(n)))
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    Z = X * Y
    conv = np.fft.irfft(Z, N)[:n]
    return conv

"""
propose_candidate
-----------------

A convex‐optimization based search for a step function with a smaller
self‑convolution peak than the classic Catalan construction.

The problem (after scaling invariance) is::

    minimise   max_k (w * w)[k]          (w ≥ 0,  sum w = 1)

where * denotes the discrete (non‑circular) convolution.
The objective is convex because each term (w*w)[k] = wᵀA_k w is a convex
quadratic form and the pointwise maximum of convex functions is convex.

We therefore use a projected sub‑gradient method.
At every iteration we:
    1. compute the convolution b = w * w,
    2. locate an index i with maximal b[i],
    3. take a sub‑gradient ∂b[i]/∂w = 2·w shifted by i,
    4. move w opposite to the sub‑gradient,
    5. project back onto the probability simplex (w ≥ 0, ∑w = 1).

The step size η decays as η₀/√t.  To avoid stagnation we occasionally
apply a cheap random block‑perturbation (the same routine used in the
reference solution) and keep any improvement.

All operations are O(n log n) thanks to FFT‑based convolution, allowing
thousands of iterations within the 1000 s budget.  The final sequence
is scaled to the conventional √(2 n) L¹‑norm (the scaling does not
change the score) and returned as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` terms of a_i = C(2i,i)/4**i.
    Recurrence a_{i+1}=a_i*(2*i+1)/(2*i+2) is numerically stable."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def convolve_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution using real FFT (fast for n≈10⁴)."""
    n = a.size + b.size - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    Fa = np.fft.rfft(a, N)
    Fb = np.fft.rfft(b, N)
    Fc = Fa * Fb
    c = np.fft.irfft(Fc, N)[:n]
    # Small negative round‑off errors are harmless; clip to 0.
    np.maximum(c, 0.0, out=c)
    return c


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0,∑x=1}."""
    # Algorithm of Duchi et al. (2008)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are (almost) equal; fall back to uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) perturbation used as a random jump."""
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # optional symmetric averaging
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Main function required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Returns a non‑negative step‑function (list of floats) whose
    `evaluate_sequence` score is smaller than the Catalan baseline
    (~1.571) and ideally below the target 1.5025.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a length.  If a previous construction exists we reuse its
    # length (it is usually a few thousand) – the search works for any
    # length ≥ 2.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 1:
                raise ValueError
        except Exception:
            n = 5000
    else:
        n = 5000

    # --------------------------------------------------------------
    # Initialise with the Catalan distribution (sum‑normalised to 1).
    # This gives a good starting point (score ≈ π/2 ≈ 1.571).
    # --------------------------------------------------------------
    a = catalan_sequence(n)
    w = a / np.sum(a)                     # probability vector (∑w = 1)

    # current best
    b = convolve_fft(w, w)
    best_score = 2.0 * n * np.max(b)      # same as evaluate_sequence after scaling
    best_w = w.copy()

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Projected sub‑gradient descent.
    # --------------------------------------------------------------
    start_time = time.monotonic()
    eta0 = 0.8                            # initial step size
    iteration = 0
    no_improve = 0

    while time.monotonic() - start_time < max(0.0, float(budget_s) - 5.0):
        # (1) compute convolution and its maximum
        b = convolve_fft(w, w)
        max_val = np.max(b)
        # pick a random location among all maxima (helps escape ties)
        max_idxs = np.flatnonzero(b >= max_val - 1e-15)
        i_max = int(rng.choice(max_idxs))

        # (2) sub‑gradient of the active quadratic form
        #    g_j = 2 * w[i_max - j] for all feasible j
        grad = np.zeros_like(w)
        j_start = max(0, i_max - (n - 1))
        j_end = min(n - 1, i_max)
        j = np.arange(j_start, j_end + 1, dtype=int)
        idx = i_max - j
        grad[j] = 2.0 * w[idx]

        # (3) step size (standard diminishing schedule)
        eta = eta0 / np.sqrt(iteration + 1)

        # (4) gradient step + projection onto the simplex
        w_new = w - eta * grad
        w = simplex_projection(w_new)

        # (5) evaluate new candidate
        b = convolve_fft(w, w)
        score = 2.0 * n * np.max(b)

        if score < best_score - 1e-9:          # strict improvement
            best_score = score
            best_w = w.copy()
            no_improve = 0
            print(f"[improve] iter={iteration:5d}, score={best_score:.6f}")
        else:
            no_improve += 1

        # occasional random jump to avoid being stuck
        if iteration % 1200 == 0 and iteration > 0:
            cand = block_perturbation(best_w, rng)
            cand = cand / np.sum(cand)        # re‑normalise to a probability vector
            b_cand = convolve_fft(cand, cand)
            cand_score = 2.0 * n * np.max(b_cand)
            if cand_score < best_score:
                best_w = cand.copy()
                best_score = cand_score
                w = cand.copy()
                print(f"[block] iter={iteration:5d}, new best score={best_score:.6f}")

        # stop early if we are not improving for a long stretch
        if no_improve > 20000:
            # reset with a small random perturbation of the current best
            w = block_perturbation(best_w, rng)
            w = w / np.sum(w)
            no_improve = 0
            print(f"[reset] iter={iteration:5d}, reset to perturbed best")

        iteration += 1

    # --------------------------------------------------------------
    # Final scaling – the evaluation is scale‑invariant, but we keep the
    # traditional √(2 n) L¹‑norm so the numbers are of a comfortable size.
    # --------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    final_seq = best_w * target_sum / np.sum(best_w)

    final_score = float(evaluate_sequence(final_seq.tolist()))
    print(f"[final] length={n}, best score={best_score:.6f}, "
          f"after scaling evaluate={final_score:.6f}")

    # Convert to plain Python list as required by the harness.
    return [float(x) for x in final_seq]
