def compute_gradient(a):
    n = a.size
    conv = np.convolve(a, a)
    max_val = conv.max()
    max_idxs = np.where(conv == max_val)[0]  # exact comparisons may be ok due to floats.
    # compute dM for each i: sum over each max index
    dM = np.zeros_like(a)
    for k in max_idxs:
        rev_idx = k - np.arange(n)
        mask = (rev_idx >= 0) & (rev_idx < n)
        dM[mask] += 2 * a[rev_idx[mask]]
    # sum dM across all max indices, then average
    if len(max_idxs) > 0:
        dM /= len(max_idxs)
    L = a.sum()
    c1 = 2.0 * n / (L * L)
    c2 = 4.0 * n * max_val / (L * L * L) # 4*n*M / L^3
    grad = c1 * dM - c2
    return grad
```

"""
Propose a candidate sequence for the evaluate_sequence() benchmark.

Algorithm Overview
------------------
1. **Initialisation** – start from the best known construction if it is
   available (`height_sequence_1`).  Otherwise fall back to a power‑law
   shape (a_i ≈ (i+shift)^β) that is known to be reasonably good.
   The sequence is normalised so that `sum(a) = sqrt(2·n)`, which makes
   the evaluation function scale‑invariant.

2. **Gradient descent on the max‑convolution objective** –  
   The score `E = 2·n·max(a★a) / (sum(a)^2)` is differentiable except at
   points where several convolution entries tie for the maximum.
   With a fixed `sum(a)`, the gradient of `E` w.r.t. each `a_i` is

        ∂E/∂a_i = (2·n / sum(a)^2)·∂M/∂a_i –
                  (4·n·M / sum(a)^3),

   where `M = max_k (a★a)_k`.  For a chosen maximal index `k*` we have
   `∂M/∂a_i = 2·a_{k*-i}` if the index is in range, otherwise 0.
   When several indices share the maximum we average their contributions.

   An Adam optimiser (first‑order adaptive moments) is used to keep
   steps stable.  After each update the vector is projected onto the
   feasible set (non‑negative, ≤1000) and re‑normalised to keep the sum
   constant.

3. **Escaping plateaus** – if no improvement is seen for a configurable
   number of iterations, a cheap block‑wise multiplicative perturbation
   (the same technique that gave the previous best score) is applied to
   the current best solution, and optimisation continues from there.

4. **Time‑budget handling** – the routine runs until the allotted
   `budget_s` seconds (minus a small safety margin) are exhausted,
   constantly tracking the best sequence seen.

The method stays entirely within the allowed Python‑only environment,
uses only `numpy`, `math`, and `time`, and prints progress information.

def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift) ** beta, clipped to the allowed range."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape – rises then falls linearly."""
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        seq = np.concatenate([half, half[::-1]])
    else:
        seq = np.concatenate([half, half[-2::-1]])
    return np.clip(seq, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of a contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width ≈ 0.5–5 % of n
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal tweak
    cand[st:en] *= factor
    # occasional symmetrisation (helps for symmetric shapes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def compute_gradient(a: np.ndarray) -> np.ndarray:
    """
    Compute the gradient of the evaluation function with respect to a.
    The gradient uses the exact maximal convolution entry (or averages
    over all maxima if there is a tie).
    """
    n = a.size
    # full convolution (length 2*n-1)
    conv = np.convolve(a, a)
    max_val = np.max(conv)
    # treat all entries that are numerically equal to the max as active
    # (using a tiny relative tolerance)
    tol = max_val * 1e-12
    max_idxs = np.where(conv >= max_val - tol)[0]

    # dM_i = (1/|max_idxs|) * sum_{k in max_idxs} 2 * a_{k-i} (if in range)
    dM = np.zeros_like(a)
    if max_idxs.size > 0:
        for k in max_idxs:
            rev = k - np.arange(n)
            mask = (rev >= 0) & (rev < n)
            dM[mask] += 2.0 * a[rev[mask]]
        dM /= float(max_idxs.size)

    # Constants from the analytic gradient
    sum_a = a.sum()
    if sum_a == 0.0:  # should never happen, but guard against division by zero
        return np.zeros_like(a)

    c1 = 2.0 * n / (sum_a * sum_a)
    c2 = 4.0 * n * max_val / (sum_a * sum_a * sum_a)

    grad = c1 * dM - c2
    return grad


def normalise_sum(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale a so that its sum equals target_sum (preserving non‑negativity)."""
    cur = a.sum()
    if cur == 0.0:
        # fallback to a tiny uniform vector (should never be needed)
        a = np.full_like(a, target_sum / a.size)
    else:
        a = a * (target_sum / cur)
    return a

    # ------------------------------------------------------------------
    m = np.zeros_like(a)               # first moment
    v = np.zeros_like(a)               # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                          # base learning rate (tuned empirically)
