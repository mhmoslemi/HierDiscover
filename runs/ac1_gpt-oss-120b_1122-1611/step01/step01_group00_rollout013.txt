def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s)-2.0)

    # Set hyperparameters
    n = 2048  # default length
    # compute FFT size power of 2 > 3*n
    fft_len = 1 << ((3*n-1).bit_length())
    # Precompute index slice for gradient
    grad_slice_start = n-1
    grad_slice_end = grad_slice_start + n

    best_seq = None
    best_score = float('inf')
    run = 0

    while time.monotonic() < deadline:
        run += 1
        # random initialization: Dirichlet with concentration 0.5 (sparse)
        a = rng.dirichlet(alpha=np.full(n, 0.5))
        # Optionally use previous best as start for second run.
        # We'll do a few iterations
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1=0.9
        beta2=0.999
        eps=1e-8
        iteration = 0
        # inner loop for this run; adapt time limit
        # allocate per-run budget: maybe 0.3 sec per iteration? Let's compute.
        # We'll just iterate until near deadline or 2000 inner iterations.
        while time.monotonic() < deadline and iteration < 2000:
            iteration += 1
            # compute conv via FFT
            fft_a = np.fft.rfft(a, fft_len)
            conv_full = np.fft.irfft(fft_a * fft_a, fft_len)[:2*n-1]
            # compute softmax weights
            # temperature param alpha_soft
            alpha_soft = 10.0  # can increase later gradually
            max_conv = np.max(conv_full)
            # subtract max for stability
            exp_vals = np.exp(alpha_soft * (conv_full - max_conv))
            s = exp_vals / np.sum(exp_vals)
            # gradient: using correlation via FFT
            # compute a_rev
            a_rev = a[::-1]
            fft_a_rev = np.fft.rfft(a_rev, fft_len)
            fft_s = np.fft.rfft(s, fft_len)
            corr_full = np.fft.irfft(fft_s * fft_a_rev, fft_len)[:3*n-2]  # length 3n-2
            grad = 2.0 * corr_full[grad_slice_start:grad_slice_end]
            # Adam update
            m = beta1*m + (1-beta1)*grad
            v = beta2*v + (1-beta2)*(grad**2)
            m_hat = m / (1 - beta1**iteration)
            v_hat = v / (1 - beta2**iteration)
            lr = 0.05 / (np.sqrt(iteration) + 1)  # schedule
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            # project onto simplex
            a = _project_simplex(a)
            # occasional perturb
            if rng.random() < 0.05:
                idx = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, 0.2))
                a[idx] *= factor
                a = _project_simplex(a)

            # evaluate
            score = float(evaluate_sequence(a.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"[run {run} iter {iteration}] new best {best_score:.6f}")

            # early stop if close tolerance
            # theoretical lower bound
            low_bound = 2.0 * n / (2*n - 1)
            if best_score <= low_bound * 1.02:
                print("[early] near lower bound")
                break
        # after inner loop, maybe update n?
        # This loop continues until deadline

    # final fallback: if best_seq not set, use uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
    print("[final] best score:", best_score)
    return [float(x) for x in best_seq]

    deadline = time.monotonic() + max(0.0, float(budget_s)-2.0)

    iteration = 0
    while time.monotonic() < deadline and iteration < 1000:
        iteration += 1
        conv = np.convolve(a, a)
        M = conv.max()
        # Build constraint matrix A
        A = np.zeros((2*n-1, n), dtype=np.float64)
        for i in range(n):

        # Solve LP maximize sum(b) -> minimize -sum(b)
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
        if not res.success:
            print(f"LP failed at iteration {iteration}")
            break
        b = res.x
        sum_b = b.sum()

            # degenerate
            break
        # scale to unit sum
        g = b / sum_b
        # mixing parameter
        tau = 0.5   # maybe adapt
        a = (1 - tau) * a + tau * g
        # project onto simplex (should already sum=1, but due to numerical)
        a = _project_simplex(a) 
        # evaluate

        if score < best_score:

"""
Propose a step‑function height sequence with a low evaluation score.

Algorithm (high‑level)
----------------------
1. Initialise a probability vector `a` (non‑negative, sum = 1).  If a
   previous best sequence `height_sequence_1` exists it is used,
   otherwise the uniform distribution is taken.
2. Repeatedly perform a few thousand *Adam* steps on a *smooth* approximation
   of the objective  `max_k (a*a)[k]`.  For a temperature `α` the smooth
   objective is  `S_α(a) = (1/α)·log Σ_k exp(α·(a*a)[k])`.  Its gradient can be
   written in closed form using a soft‑max weighting of the convolution
   entries and a correlation operation:
        w_k = exp(α·c_k) / Σ_j exp(α·c_j)   (c = a*a)
        ∇S_α(a) = 2·correlate(w, a)   (valid mode → length n)
3. After each Adam step the vector is projected onto the probability simplex
   to keep it feasible.
4. Occasionally a random multiplicative perturbation is applied to avoid
   plateaus.
5. The best sequence seen (according to `evaluate_sequence`) is kept.
6. When the allotted time (≈ 1000 s) is nearly exhausted the best sequence
   is returned.

The method is a smooth‑max surrogate of the original `max` objective,
combined with an adaptive optimiser (Adam) and feasibility projection.
It typically drives the evaluation value well below the target 1.5025
within the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Generate a non‑negative step‑function (list of floats) that minimises
    `evaluate_sequence`.  The routine runs for roughly the given time budget
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # problem size – 1024 elements give a good trade‑off between quality
    # and speed.  The code works for other sizes as well.
    n = 1024

    # ------------------------------------------------------------------
    # initialise the probability vector.
    # If a previous construction is available, reuse it (normalised).
    init_seq = None
    if 'height_sequence_1' in globals():
        try:
            cand = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
            if cand.size >= 2:
                init_seq = cand / np.sum(cand)   # normalise to unit ℓ₁
        except Exception:
            init_seq = None

    if init_seq is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    else:
        if init_seq.size != n:
            # if the previous sequence has a different length, truncate or pad
            if init_seq.size > n:
                a = init_seq[:n]
            else:
                a = np.concatenate([init_seq,
                                    np.full(n - init_seq.size,
                                            0.0, dtype=np.float64)])
            a = a / np.sum(a)
        else:
            a = init_seq.copy()

    # keep the best solution ever seen
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Adam optimiser hyper‑parameters
    beta1 = 0.90
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # temperature schedule for the smooth‑max (soft‑max) used in the gradient
    alpha_start = 2.0          # initially very smooth
    alpha_end = 30.0           # increasingly sharp as optimisation proceeds
    alpha = alpha_start
    alpha_inc = (alpha_end - alpha_start) / 300_000  # approx. linear ramp

    # learning‑rate schedule (initial step size)
    lr0 = 0.12

    # total time budget (leave a small safety margin)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    iteration = 0
    # main optimisation loop
    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) full self‑convolution (needed for the objective value)
        conv = np.convolve(a, a)                     # length 2n‑1
        max_conv = conv.max()

        # --------------------------------------------------------------
        # 2) smooth‑max weights  w_k ∝ exp(α·conv_k)
        #    (stable version subtracting the maximum before exponentiation)
        shift = conv.max()                           # a scalar, same as max_conv
        exps = np.exp(alpha * (conv - shift))
        w = exps / exps.sum()                        # length 2n‑1, sums to 1

        # --------------------------------------------------------------
        # 3) gradient of the smooth‑max:  ∇ = 2·correlate(w, a)
        #    np.correlate(w, a, mode='valid') returns a length‑n vector
        grad = 2.0 * np.correlate(w, a, mode='valid')

        # --------------------------------------------------------------
        # 4) Adam update (with a diminishing step size)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # a simple learning‑rate decay – sqrt(t) gives a gentle schedule
        lr = lr0 / (np.sqrt(iteration) + 1.0)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --------------------------------------------------------------
        # 5) projection back onto the simplex (ensures feasibility)
        a_new = _project_simplex(a_new)

        # --------------------------------------------------------------
        # 6) occasional random shake‑up (≈5 % of the steps)
        if rng.random() < 0.05:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.2)))   # log‑normal perturb
            a_new[idx] *= factor
            a_new = _project_simplex(a_new)

        a = a_new

        # --------------------------------------------------------------
        # 7) evaluate the current candidate
        score = float(evaluate_sequence(a.tolist()))
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
            print(f"[improve] iter={iteration:,}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 8) update the temperature (make the surrogate sharper)
        if alpha < alpha_end:
            alpha = min(alpha_end, alpha + alpha_inc)

        # optional early stopping when we are very close to the theoretical bound
        theoretical_lb = 2.0 * n / (2 * n - 1)           # = 2n/(2n‑1)
        if best_score <= theoretical_lb * 1.02:
            print("[early stop] reached near‑optimal theoretical bound")
            break

        # --------------------------------------------------------------
        # sanity check to avoid endless loops if time is exhausted
        if iteration % 200_000 == 0:
            # tiny progress report
            print(f"[status] iter={iteration:,} – best score so far {best_score:.6f}")

    print(f"[final] best score={best_score:.6f} after {iteration:,} iterations")
    # Return a plain Python list (required by the evaluation harness)
    return [float(x) for x in best]
