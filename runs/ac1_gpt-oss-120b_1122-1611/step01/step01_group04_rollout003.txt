def subgradient_descent(a, deadline):
    n = len(a)
    # renorm sum to 1
    a = a / np.sum(a)
    best = a.copy()
    best_score = compute_score(a)
    iter = 0
    while time.monotonic() < deadline:
        b = np.convolve(a, a)
        M = b.max()
        max_idx = np.where(b == M)[0]
        S = np.sum(a)
        # g:
        g = np.zeros_like(a)
        for k in max_idx:
            low = max(0, k - (n-1))
            high = min(n-1, k)
            idx = np.arange(low, high+1)
            g[idx] += 2.0 * a[k - idx]
        factor1 = 2.0 * n / (S * S)
        factor2 = 4.0 * n * M / (S * S * S)
        grad = factor1 * g - factor2
        # step size schedule
        lr = 0.5 / np.sqrt(iter+1)
        # backtracking
        improved = False
        for _ in range(10):
            a_new = a - lr * grad
            # clip and renorm
            a_new = np.clip(a_new, 0.0, None)
            if a_new.sum() == 0:
                break
            a_new = a_new / np.sum(a_new) * S  # keep same sum as before
            score = compute_score(a_new)
            if score < best_score - 1e-12:
                a = a_new
                best = a_new.copy()
                best_score = score
                improved = True
                print(f"SD iter {iter} lr {lr:.5f} score {best_score:.6f}")
                break
            else:
                lr *= 0.5
        if not improved:
            # occasional random jitter to escape flat region
            # jitter amplitude shrinking with iteration
            jitter = (np.random.default_rng().normal(scale=0.01/np.sqrt(iter+1), size=n))
            a = np.clip(a + jitter, 0.0, None)
            if a.sum() > 0:
                a = a / np.sum(a) * S
        iter += 1
    return best, best_score
```

#!/usr/bin/env python
"""
Propose a coefficient sequence that minimizes

    2·n·max(conv(a,a)) / (sum(a)**2)

The algorithm consists of three stages:

1. **Analytic families** – a quick sweep over several parametric families
   (power law, sine‑hump, power‑law with sinusoidal modulation) to obtain a
   decent starting point.
2. **Stochastic block perturbations** – cheap multiplicative tweaks of random
   blocks (as in the previous baseline) to escape very bad local minima.
3. **Sub‑gradient descent** – a deterministic descent on the exact sub‑gradient
   of the objective (the objective is scale‑invariant, so we keep the L1‑norm
   fixed by renormalising after each step).  A back‑tracking line‑search and
   occasional symmetrisation keep the iterates feasible and drive the maximum
   convolution value down further.

All helper functions are defined at top‑level, no closures or lambda
expressions are used.  The search respects the 1000 s wall‑time budget;
the best sequence found is returned as plain Python floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility functions ----------------------------------------------------
# ----------------------------------------------------------------------
def compute_score(arr: np.ndarray) -> float:
    """
    Direct implementation of the objective used in evaluate_sequence.
    The array is assumed to contain non‑negative numbers already clipped to
    [0, 1000].  Returns np.inf for a degenerate sum.
    """
    if arr.size == 0:
        return float("inf")
    s = arr.sum()
    if s < 0.01:
        return float("inf")
    conv = np.convolve(arr, arr)
    max_conv = conv.max()
    return float(2.0 * arr.size * max_conv / (s * s))


def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift) ** beta, clipped to the allowed range."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_sine(gamma: float, n: int) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n) ** gamma, clipped."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    return np.clip(seq, 0.0, 1000.0)


def generate_modulated_power(beta: float, shift: float, amp: float,
                             n: int) -> np.ndarray:
    """
    Power‑law shape multiplied by (1 + amp·sin(...)).
    When amp=0 this reduces to the pure power law.
    """
    base = generate_power(beta, shift, n)
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    mod = 1.0 + amp * np.sin(angles)
    seq = base * mod
    return np.clip(seq, 0.0, 1000.0)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of a random block."""
    cand = base.copy()
    n = cand.size
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[st:en] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Sub‑gradient descent -------------------------------------------------
# ----------------------------------------------------------------------
def subgradient_descent(start: np.ndarray,
                        deadline: float) -> tuple[np.ndarray, float]:
    """
    Perform sub‑gradient descent on the objective.
    The L1‑norm of the iterate is kept constant (renormalisation after each
    step).  Returns the best vector seen and its score.
    """
    a = start.copy()
    n = a.size
    # Normalise to unit L1 – scale does not affect the objective.
    a = a / a.sum()
    best = a.copy()
    best_score = compute_score(best)

    iteration = 0
    rng = np.random.default_rng(0xC0FFEE)   # deterministic jitter source

    while time.monotonic() < deadline:
        # convolution & max location(s)
        conv = np.convolve(a, a)
        M = conv.max()
        max_idx = np.where(conv == M)[0]

        S = a.sum()
        # ----- sub‑gradient of the objective -----
        g = np.zeros_like(a)
        # g_i = sum_{k in max_idx} 2 * a_{k-i}  (valid indices only)
        for k in max_idx:
            low = max(0, k - (n - 1))
            high = min(n - 1, k)
            idx = np.arange(low, high + 1, dtype=np.int64)
            g[idx] += 2.0 * a[k - idx]

        factor1 = 2.0 * n / (S * S)
        factor2 = 4.0 * n * M / (S * S * S)
        grad = factor1 * g - factor2    # sub‑gradient of the scalar objective

        # ----- back‑tracking line search -----
        # step size ~ 0.5 / sqrt(iteration) with geometric reduction on failure
        lr = 0.5 / math.sqrt(iteration + 1)
        improved = False
        for _ in range(12):
            cand = a - lr * grad
            cand = np.clip(cand, 0.0, None)
            if cand.sum() == 0.0:          # numerical disaster – give up this step
                break
            cand = cand / cand.sum() * S   # re‑normalise to keep L1 constant

            score = compute_score(cand)
            if score < best_score - 1e-12:   # strict improvement
                a = cand
                best = cand.copy()
                best_score = score
                improved = True
                print(f"SD #{iteration:06d} lr={lr:.5g} → score {best_score:.6f}")
                break
            lr *= 0.5                      # shrink step size

        # ----- occasional random jitter (to escape plateaus) -----
        if not improved:
            jitter = rng.normal(scale=0.015 / math.sqrt(iteration + 1), size=n)
            a = np.clip(a + jitter, 0.0, None)
            if a.sum() == 0.0:
                a = best.copy()
            else:
                a = a / a.sum() * S

        # ----- occasional symmetrisation (helps for symmetric optima) -----
        if rng.random() < 0.12:
            a = 0.5 * (a + a[::-1])
            a = a / a.sum() * S

        iteration += 1

    return best, best_score


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises the
    given evaluation function.  The routine combines a coarse sweep over a
    few analytic families, cheap block‑perturbations and a final
    sub‑gradient descent.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # --------------------------------------------------------------
    # Determine problem size – reuse previous construction if available.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        base_seq = np.array(height_sequence_1, dtype=np.float64)
        base_seq = np.clip(base_seq, 0.0, 1000.0)
    else:
        n = 6000                     # a safe default
        base_seq = np.ones(n, dtype=np.float64)

    # --------------------------------------------------------------
    # Initialise with a trivial feasible point.
    # --------------------------------------------------------------
    best = base_seq.copy()
    best_score = compute_score(best)
    print(f"Initial score (length {n}) = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Power‑law family a_i = (i+shift)^beta
    # --------------------------------------------------------------
    betas = np.arange(-0.9, -0.2 + 1e-12, 0.05)
    shifts = np.linspace(0.1, 8.0, 16)
    for beta in betas:
        for shift in shifts:
            cand = generate_power(beta, shift, n)
            score = compute_score(cand)
            if score < best_score:
                best, best_score = cand, score
                print(f"Power β={beta:.3f}, shift={shift:.3f} → {score:.6f}")

    # --------------------------------------------------------------
    # 2) Sine‑hump family a_i = sin(...)^γ
    # --------------------------------------------------------------
    gammas = [0.5, 1.0, 2.0, 3.0]
    for gamma in gammas:
        cand = generate_sine(gamma, n)
        score = compute_score(cand)
        if score < best_score:
            best, best_score = cand, score
            print(f"Sine γ={gamma:.2f} → {score:.6f}")

    # --------------------------------------------------------------
    # 3) Power‑law with sinusoidal modulation (few amplitudes)
    # --------------------------------------------------------------
    amps = [0.0, 0.2, 0.4, 0.6]
    for amp in amps:
        cand = generate_modulated_power(beta=-0.5, shift=1.5, amp=amp, n=n)
        score = compute_score(cand)
        if score < best_score:
            best, best_score = cand, score
            print(f"Modulated power amp={amp:.2f} → {score:.6f}")

    # --------------------------------------------------------------
    # 4) Fast cheap block‑wise perturbations (as a quick local search)
    # --------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline and it < 20000:
        cand = block_perturbation(best, rng)
        score = compute_score(cand)
        if score < best_score:
            best, best_score = cand, score
            print(f"Block #{it:06d} improved → {score:.6f}")
        it += 1

    # --------------------------------------------------------------
    # 5) Sub‑gradient descent from the current best.
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        print("Starting sub‑gradient descent...")
        best, best_score = subgradient_descent(best, deadline)
        print(f"After sub‑gradient descent → {best_score:.6f}")

    print(f"Final best score = {best_score:.6f}")
    # Convert to plain Python floats for the caller.
    return [float(x) for x in best]
