"""
Propose a candidate step‑function sequence.

Algorithm
---------
1. Initialise a moderate‑size (≈2000) non‑negative sequence.
   If a previous best `height_sequence_1` exists it is down‑sampled,
   otherwise a simple exponential decay is used.  The sequence is
   normalised to have unit ℓ₁‑norm (scaling does not affect the
   evaluation metric).

2. Iterative LP‑based refinement (inspired by the “alternating LP”
   method from the problem statement):
   * Let `M = max(conv(f, f))`.
   * Build the full convolution matrix `A` where `A @ g = conv(f, g)`.
   * Solve the linear program
         maximise   sum(g)
         subject to A @ g ≤ M,  g ≥ 0 .
     The constraints are linear in `g`; the LP therefore finds a
     vector `g` with the largest possible ℓ₁‑norm while keeping every
     convolution entry below the current bound `M`.
   * Rescale `g` so that its ℓ₁‑norm equals that of `f` (the bound
     remains valid because the scaling factor ≤ 1).
   * Form a convex combination `f' = (1‑γ)·f + γ·g` (γ≈0.5).  The
     combination preserves the ℓ₁‑norm and provably does not increase
     the maximum convolution value.
   * Keep the best sequence seen so far (according to `evaluate_sequence`).

3. Interleave occasional stochastic block‑perturbations (multiplicative
   log‑normal noise on a random contiguous segment) to escape possible
   plateaus – this is the same perturbation operator used in the
   previous state‑of‑the‑art code.

4. Stop when the time budget (≈ 1000 s) is exhausted or when the score
   drops below the target 1.5025.

The method is deterministic for a given seed apart from the random
perturbations, uses only NumPy, SciPy and the provided `evaluate_sequence`,
and works comfortably within the allowed time‑ and memory‑limits.
"""

import time
import numpy as np
import math

# SciPy is optional; if unavailable we fall back to the previous random
# search (the import will raise ImportError in that case).
try:
    import scipy.optimize
except Exception:  # pragma: no cover
    scipy = None

def _evaluate(seq):
    """Thin wrapper around the externally provided `evaluate_sequence`."""
    # The evaluator expects a Python list.
    return float(evaluate_sequence(seq.tolist()))

def make_block_perturbation(base, rng):
    """
    Return a non‑negative perturbation of ``base``.
    Multiplicative log‑normal factor on a random contiguous block,
    optional mild symmetry.
    """
    cand = np.asarray(base, dtype=np.float64).copy()
    n = cand.size
    # block width between 0.5 % and 8 % of the length
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:           # occasional symmetry
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def _build_convolution_matrix(f):
    """
    Return the dense matrix A (shape (2n-1, n)) such that
    A @ g == np.convolve(f, g, mode='full').
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contains f shifted by j rows
    for j in range(n):
        A[j:j + n, j] = f
    return A

def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a step‑function sequence with a low evaluation score.
    Returns a list of floats (the best sequence found before the deadline).
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)    # keep a margin

    # --------------------------------------------------------------------
    # 1) Determine problem size and initialise a starting sequence.
    # --------------------------------------------------------------------
    # Target length – keep it modest to make the LP tractable.
    TARGET_N = 2000

    try:
        # If a prior best sequence exists, down‑sample it.
        raw = np.asarray(height_sequence_1, dtype=np.float64)
        n_orig = raw.size
        if n_orig > TARGET_N:
            # Linear interpolation to the desired size
            x_old = np.arange(n_orig, dtype=np.float64)
            x_new = np.linspace(0, n_orig - 1, TARGET_N, dtype=np.float64)
            raw = np.interp(x_new, x_old, raw)
        elif n_orig < TARGET_N:
            # Pad with a tiny constant to reach TARGET_N
            pad = np.full(TARGET_N - n_orig, 1e-12, dtype=np.float64)
            raw = np.concatenate([raw, pad])
        f = raw[:TARGET_N]
    except Exception:
        # No historic sequence – start from a simple exponential decay.
        idx = np.arange(TARGET_N, dtype=np.float64)
        decay = math.exp(-2.0 / (TARGET_N - 1))
        f = decay ** idx

    # Ensure strict positivity (helps the LP constraints) and normalise.
    f = np.maximum(f, 1e-12)
    f = f / f.sum()          # ℓ₁‑norm = 1, scaling does not affect the score.
    f = np.clip(f, 0.0, 1000.0)

    best_seq = f.copy()
    best_score = _evaluate(best_seq)
    print(f"Initial score {best_score:.6f}")

    # --------------------------------------------------------------------
    # Helper to (optionally) evaluate a candidate and keep the best one.
    # --------------------------------------------------------------------
    def maybe_update(candidate):
        nonlocal best_seq, best_score
        sc = _evaluate(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"New best score {best_score:.6f}")

    # --------------------------------------------------------------------
    # 2) Main optimisation loop.
    # --------------------------------------------------------------------
    gamma = 0.5                     # mixing weight between f and LP‑solution
    iter_count = 0
    n = f.size

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # (a) Compute the current bound M = max(conv(f, f))
        # --------------------------------------------------------------
        conv_ff = np.convolve(f, f)               # full convolution
        M = conv_ff.max()

        # --------------------------------------------------------------
        # (b) Build the convolution matrix A (dense, still modest size).
        # --------------------------------------------------------------
        A = _build_convolution_matrix(f)

        # --------------------------------------------------------------
        # (c) Solve the LP: maximise sum(g)  s.t. A @ g ≤ M,  g ≥ 0.
        # --------------------------------------------------------------
        if scipy is not None:
            c_obj = -np.ones(n, dtype=np.float64)      # minimise –sum(g)
            bounds = (0.0, 1000.0)                     # same limits as evaluate_sequence
            # The RHS vector is simply M repeated for each row.
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            # Use HiGHS (the default) – it copes well with dense matrices.
            res = scipy.optimize.linprog(
                c=c_obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=bounds,
                method='highs',
                options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())}
            )
        else:
            res = None

        # --------------------------------------------------------------
        # (d) If the LP succeeded, construct the new candidate.
        # --------------------------------------------------------------
        if res is not None and res.success:
            g = res.x
            sum_f = f.sum()
            sum_g = g.sum()
            # Rescale to preserve the ℓ₁‑norm (sum_f == sum_g after scaling).
            if sum_g > 0:
                g = g * (sum_f / sum_g)
            else:
                # Fallback – keep the current f if LP gave the zero vector.
                g = f.copy()

            # Convex combination (parameter gamma) to guarantee a descent.
            candidate = (1.0 - gamma) * f + gamma * g
            candidate = np.clip(candidate, 0.0, 1000.0)

            # Normalise again (purely for numerical stability; scale is irrelevant).
            if candidate.sum() > 0:
                candidate = candidate / candidate.sum()
            else:
                candidate = np.full_like(candidate, 1e-12)

            maybe_update(candidate)
            f = candidate                     # use it as the base for the next iteration
        else:
            # LP failed (unlikely) – fall back to a random block perturbation.
            candidate = make_block_perturbation(f, rng)
            maybe_update(candidate)
            f = candidate

        # --------------------------------------------------------------
        # (e) Occasionally inject a stochastic block perturbation to
        #     avoid being trapped in a flat region.
        # --------------------------------------------------------------
        if rng.random() < 0.07:
            pert = make_block_perturbation(f, rng)
            maybe_update(pert)
            f = pert

        # --------------------------------------------------------------
        # (f) Dynamically adapt the mixing weight.
        # --------------------------------------------------------------
        # If we have not improved for a while, shrink gamma.
        if iter_count % 30 == 0 and iter_count > 0:
            gamma = max(gamma * 0.9, 0.05)

        # Early exit if the target has been reached.
        if best_score <= 1.5025:
            print(f"Target reached after {iter_count} iterations.")
            break

        iter_count += 1

        # --------------------------------------------------------------
        # (g) Safety check for the remaining time.
        # --------------------------------------------------------------
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------------
    # 3) Final fine‑tuning with a short random search (optional).
    # --------------------------------------------------------------------
    parent = best_seq.copy()
    while time.monotonic() < deadline:
        cand = make_block_perturbation(parent, rng)
        maybe_update(cand)
        parent = cand  # continue from the latest improvement

    # --------------------------------------------------------------------
    # Return the best sequence found (clipped to the allowed range).
    # --------------------------------------------------------------------
    final = np.clip(best_seq, 0.0, 1000.0)
    print(f"Search completed – best score {best_score:.6f}")
    return final.tolist()
