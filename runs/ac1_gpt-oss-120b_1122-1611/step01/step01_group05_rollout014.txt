"""
Propose a step‑function (list of non‑negative heights) that minimises the
security‑evaluation function.

Algorithm
---------
1. **Initialisation** – start from the best known construction (the global
   variable ``height_sequence_1`` if it exists) or from a simple power‑law
   sequence.
2. **Broad parameter sweep** – scan a fine grid of analytic families
   (power‑law, shifted power‑law, power‑law multiplied by a tiny exponential).
   The best sequence from this cheap sweep is kept as the current state.
3. **Simulated annealing / stochastic hill‑climbing** – repeatedly apply a
   random move (block scaling, large mass swap, smoothing, mirroring or a
   combination) to the current sequence.  The new state is accepted if it
   improves the score, otherwise it may be accepted with a temperature‑dependent
   probability (standard Metropolis rule).  The temperature is decreased
   linearly from a modest start value to near zero during the allotted time.
4. **Best‑ever tracking** – whenever a candidate beats the best known score,
   it is stored and printed.
5. **Return** – the best sequence found before the time budget expires.

The search works directly on the full coefficient vector (no hidden closures)
and makes heavy use of NumPy for speed.  All operations respect the
non‑negativity and clipping constraints enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions -------------------------------------------------------
def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence."""
    # evaluate_sequence expects a plain list of floats
    return float(evaluate_sequence(seq.tolist()))


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random contiguous block by a log‑normal factor."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 8 % of the length
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # multiplicative perturbation
    cand[start:start + width] *= factor
    # occasional mirroring – helps to reduce the convolution peak
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def move_large_swap(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a possibly large amount of mass from one index to another."""
    cand = base.copy()
    i, j = rng.integers(0, cand.size, size=2)
    if i == j:
        return cand
    # transfer up to the whole mass at i
    delta = rng.uniform(0.0, cand[i])
    cand[i] -= delta
    cand[j] += delta
    return np.clip(cand, 0.0, 1000.0)


def move_scale_block(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random block by a factor drawn from a log‑normal distribution."""
    cand = base.copy()
    n = cand.size
    min_w = max(1, n // 100)          # a bit larger blocks than the simple perturbation
    max_w = max(2, n // 8)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.25)))   # larger variance than block perturbation
    cand[start:start + width] *= factor
    return np.clip(cand, 0.0, 1000.0)


def move_smooth(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """A gentle averaging with neighbours plus a tiny random jitter."""
    cand = base.copy()
    # three‑point moving average
    cand = 0.5 * cand + 0.25 * np.concatenate(([0.0], cand[:-1])) + 0.25 * np.concatenate((cand[1:], [0.0]))
    jitter = np.exp(rng.normal(0.0, 0.03, size=cand.shape))
    cand *= jitter
    return np.clip(cand, 0.0, 1000.0)


def move_mirror_average(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Blend the sequence with its reversal – a strong symmetry move."""
    cand = 0.5 * (base + base[::-1])
    # small random jitter to keep the move stochastic
    jitter = np.exp(rng.normal(0.0, 0.02, size=base.shape))
    cand *= jitter
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
def generate_power_law(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """a_i = 1 / (i+1+shift)^p  (i starts at 0)."""
    idx = np.arange(1, n + 1, dtype=np.float64) + shift
    return np.clip(1.0 / np.power(idx, p), 0.0, 1000.0)


def generate_exp_power_law(n: int, p: float, shift: float, beta: float) -> np.ndarray:
    """a_i = exp(-beta*i) / (i+1+shift)^p."""
    i = np.arange(n, dtype=np.float64)
    base = np.exp(-beta * i) / np.power(i + 1.0 + shift, p)
    return np.clip(base, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a low‑evaluation step function within the allotted time.

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Approximate wall‑clock time budget (seconds).  The algorithm stops
        a couple of seconds early to avoid forced termination.
    **kwargs : dict
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    list[float]
        The best height sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the base length.  Use the length of the provided sequence
    # if it exists, otherwise fall back to a moderate size.
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                n = int(arr.size)
                start_seq = np.clip(arr, 0.0, 1000.0)
            else:
                raise ValueError
        except Exception:
            n = 2000
            start_seq = np.ones(n, dtype=np.float64)
    else:
        n = 2000
        start_seq = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initial best candidate – the supplied one (if any) or a uniform vector.
    best = start_seq.copy()
    best_score = evaluate(best)
    print(f"[init] length={n}, start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) PARAMETER SWEEP -------------------------------------------------
    # The sweep explores analytic families that are known to be close to optimal.
    # It is cheap (few hundred evaluations) and often yields a good starting point.
    p_grid = np.linspace(0.40, 0.60, 21)          # fine grid around 0.5
    shift_grid = np.linspace(0.0, 12.0, 13)      # a modest range of shifts
    beta_grid = [0.0, 1e-4, 5e-4, 1e-3]          # tiny exponential factors

    for p in p_grid:
        for shift in shift_grid:
            cand = generate_power_law(n, p, shift)
            score = evaluate(cand)
            if np.isfinite(score) and score < best_score:
                best, best_score = cand, score
                print(f"[sweep] power p={p:.3f} shift={shift:.2f} → {score:.6f}")

            for beta in beta_grid:
                if beta == 0.0:
                    continue
                cand2 = generate_exp_power_law(n, p, shift, beta)
                score2 = evaluate(cand2)
                if np.isfinite(score2) and score2 < best_score:
                    best, best_score = cand2, score2
                    print(f"[sweep] exp‑power p={p:.3f} shift={shift:.2f} β={beta:.1e} → {score2:.6f}")

    # ------------------------------------------------------------------
    # 2) STOCHASTIC SEARCH (SIMULATED ANNEALING) -------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    cur = best.copy()
    cur_score = best_score

    # simple linear cooling schedule
    start_temp = 0.25
    end_temp = 1e-5

    moves = [
        make_block_perturbation,
        move_large_swap,
        move_scale_block,
        move_smooth,
        move_mirror_average,
    ]

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        # temperature decreasing linearly with time
        t_frac = (iteration * 1.0) / (deadline - (time.monotonic() - (deadline - time.monotonic())) + 1e-12)
        temp = start_temp * (1.0 - t_frac) + end_temp * t_frac

        move_fn = rng.choice(moves)
        cand = move_fn(cur, rng)
        cand_score = evaluate(cand)

        # Metropolis acceptance rule
        accept = False
        if np.isfinite(cand_score) and cand_score < cur_score:
            accept = True
        elif np.isfinite(cand_score):
            prob = np.exp(-(cand_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur, cur_score = cand, cand_score
            # keep the global best as well
            if cur_score < best_score:
                best, best_score = cur.copy(), cur_score
                print(f"[search] new best {best_score:.6f} via {move_fn.__name__}")

    # ------------------------------------------------------------------
    print(f"[done] best_score={best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]
