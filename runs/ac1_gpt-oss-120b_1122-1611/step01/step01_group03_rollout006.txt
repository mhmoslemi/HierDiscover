def propose_candidate(seed=42, budget_s=1100, **kwargs):
    import time, numpy as np

    rng = np.random.default_rng(seed)

    # Determine base sequence
    if isinstance(height_sequence_1, (list, np.ndarray)) and len(height_sequence_1) > 0:
        a_norm = np.array(height_sequence_1, dtype=np.float64)
        n0 = len(a_norm)
    else:
        n0 = 3000
        # generate alpha
        a_norm = generate_alpha_sequence(n0)
        a_norm = a_norm / a_norm.sum()

    # compute raw sequence with max conv = 1
    conv0 = np.convolve(a_norm, a_norm)
    max_conv0 = conv0.max()
    raw = a_norm / np.sqrt(max_conv0)  # scale so max conv = 1

    # define target length N_target = 2 * n0 (or maybe 3*n0 but keep moderate)
    N = int(n0 * 2)
    # allocate array
    a_raw = np.zeros(N, dtype=np.float64)
    a_raw[:n0] = raw

    # Compute initial stats
    sum_raw = a_raw.sum()
    # conv_max should be 1 already
    max_conv = 1.0
    best_raw = a_raw.copy()
    best_sum = sum_raw
    best_score = 2 * N / (best_sum**2)

    print(f"[init] n0={n0}, N={N}, initial_score={best_score:.6f}")

    # Helper function to compute max conv quickly using fft
    def max_convolution(x):
        n = len(x)
        m = 1 << ((2*n-1).bit_length())
        fft_x = np.fft.rfft(x, n=m)
        conv = np.fft.irfft(fft_x * fft_x, n=m)
        # only first 2*n-1 values relevant
        return conv[:2*n-1].max()

    # parameters
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    step_factor = 0.5   # relative to average value
    tail_start = n0
    # we may also allow occasional injection into existing region to balance
    while time.monotonic() < deadline:
        iteration += 1

        # pick index in tail
        p = rng.integers(tail_start, N)

        # compute delta as step_factor * sum_raw / N (mean)
        avg = sum_raw / N
        # protect against avg=0
        delta = step_factor * max(avg, 1e-12)

        a_raw[p] += delta
        sum_raw += delta

        # compute max conv (using fft)
        max_conv = max_convolution(a_raw)

        if max_conv > 1.0:
            scale = np.sqrt(max_conv)
            a_raw /= scale
            sum_raw = a_raw.sum()
            max_conv = 1.0

        # Score
        score = 2 * N / (sum_raw**2)

        if score < best_score:
            best_score = score
            best_raw = a_raw.copy()
            best_sum = sum_raw
            print(f"[{iteration:6d}] new best score={best_score:.6f}, sum={best_sum:.3f}")

        # Anneal step factor
        if iteration % 2000 == 0:
            step_factor *= 0.95
            # occasionally move step_factor up slightly if stagnating?
            if step_factor < 1e-3:
                step_factor = 1e-3

    # Build final sequence normalized to sum=1
    final_seq = (best_raw / best_raw.sum()).tolist()

    # final evaluation (optional)
    final_eval = evaluate(final_seq)
    print(f"[final] iterations={iteration}, best_score (recomputed)={final_eval:.6f}")

    return final_seq

Greedy “mass‐fill” optimization.

The known α–sequence gives a self–convolution whose first half is exactly 1,
which translates after normalisation to a score of π/2≈1.57.  The objective
is  2·n·max(a*a)/(∑a)² and is invariant under scaling, so the problem is
equivalent to: maximise the L₁–norm ∑a subject to max(a*a) ≤ 1.
The α–sequence satisfies max(a*a)=1 but its L₁–norm is only
(2/√π)√n, far from the trivial bound √(2n) (which would give a score of 1).

Starting from the α–sequence we keep the sequence “raw’’ (not normalised)
so that its maximal convolution value is exactly 1.
We then repeatedly add a tiny amount of mass at a position far in the tail,
where the current convolution value is small.  Adding mass at index `p`
increases the convolution value at `p` by roughly `2·a₀·Δ` (with a₀≈1);
as long as the value stays ≤1 we do not need to rescale.
If an addition pushes the global maximum above 1 we immediately renormalise
the whole vector by `1/√max` – this keeps the constraint active while
preserving most of the newly added mass.  The sum of the raw coefficients
grows and the score 2·n/(∑a)² consequently drops.

All operations are performed with an FFT‑based convolution (O(n log n))
so that thousands of iterations fit comfortably into the 1000 s limit.
The best vector found is finally normalised to L₁=1 and returned as the
candidate sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """First n terms of a_i = C(2 i,i) / 4^i ( α‑sequence )."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def max_convolution_fft(x: np.ndarray) -> float:
    """
    Return max value of the linear convolution x*x using an FFT.
    The length of the FFT is the next power of two ≥ 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)
    # Only the first 2·n‑1 entries belong to the linear convolution.
    return float(conv[: 2 * n - 1].max())


def raw_from_normalised(norm: np.ndarray) -> np.ndarray:
    """
    Convert a L₁‑normalised non‑negative sequence `norm` into a
    “raw’’ sequence whose self‑convolution maximal entry equals exactly 1.
    """
    # max of the convolution of the normalised sequence
    max_conv = max_convolution_fft(norm)
    if max_conv <= 0:
        # fallback – should never happen for a valid input
        scale = 1.0
    else:
        scale = 1.0 / np.sqrt(max_conv)
    return norm * scale

most of the newly added mass.  The sum of the raw coefficients therefore
grows, and the score 2·n/(∑a)² drops.

All operations use an FFT‑based convolution (O(n log n)) so that many
thousands of iterations fit comfortably into the 1000 s time limit.
The best raw vector found is finally normalised to L₁=1 and returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (no closures, top‑level)
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """First n terms of a_i = C(2i,i) / 4**i (the α‑sequence)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def max_convolution_fft(x: np.ndarray) -> float:
    """
    Return the maximum value of the linear convolution x*x using an FFT.
    The FFT length is the next power of two ≥ 2·len(x)‑1.
    """
    n = x.shape[0]
    m = 1 << ((2 * n - 1).bit_length())
    fx = np.fft.rfft(x, n=m)
    conv = np.fft.irfft(fx * fx, n=m)
    # Only the first 2·n‑1 entries belong to the linear convolution.
    return float(conv[: 2 * n - 1].max())


def raw_from_normalised(norm: np.ndarray) -> np.ndarray:
    """
    Convert a L₁‑normalised non‑negative sequence `norm` into a
    “raw’’ sequence whose maximal self‑convolution entry is exactly 1.
    """
    max_conv = max_convolution_fft(norm)
    scale = 1.0 / np.sqrt(max_conv) if max_conv > 0 else 1.0
    return norm * scale


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Greedy tail‑mass filling algorithm.
    Returns a list of non‑negative heights with a score lower than the
    α‑sequence (target ≈ 1.50).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1. Base sequence (α‑sequence or the provided global variable)
    # ------------------------------------------------------------------
    if isinstance(height_sequence_1, (list, np.ndarray)) and len(height_sequence_1) > 0:
        a_norm = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base_n = 3000
        a_norm = generate_alpha_sequence(base_n)
        a_norm = a_norm / a_norm.sum()

    base_n = a_norm.shape[0]

    # ------------------------------------------------------------------
    # 2. Build the “raw’’ representation: max(a*a) = 1
    # ------------------------------------------------------------------
    a_raw = raw_from_normalised(a_norm)
    a_raw[0] = 1.0                     # enforce the exact value (numerical noise)

    # ------------------------------------------------------------------
    # 3. Allocate extra slots for future mass insertion (double length)
    # ------------------------------------------------------------------
    N = base_n * 2                     # final length of the candidate
    raw = np.zeros(N, dtype=np.float64)
    raw[:base_n] = a_raw
    a_raw = raw

    n = N
    sum_raw = a_raw.sum()
    best_raw = a_raw.copy()
    best_sum = sum_raw
    best_score = 2.0 * n / (best_sum ** 2)

    print(f"[init] base_n={base_n}, N={N}, init_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Main greedy loop – insert mass in the far tail.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    it = 0
    step_factor = 0.6                     # fraction of the available slack
    tail_start = base_n

    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Current convolution (linear) – needed to know how much slack each
        # tail index has.
        # --------------------------------------------------------------
        conv = np.convolve(a_raw, a_raw)[: 2 * n - 1]

        # --------------------------------------------------------------
        # Choose a tail position with the largest slack (1‑conv).
        # --------------------------------------------------------------
        slack_slice = 1.0 - conv[tail_start:]          # only tail indices
        if np.all(slack_slice <= 0):
            break                                      # already saturated

        # pick a few of the best candidates and select one at random
        top_idxs = np.argpartition(-slack_slice, 5)[:5]
        p = tail_start + rng.choice(top_idxs)

        # --------------------------------------------------------------
        # Compute a safe increment Δ at position p.
        # Adding Δ raises conv[p] by ≈ 2·a₀·Δ (a₀≈1).
        # --------------------------------------------------------------
        cur = conv[p]
        a0 = a_raw[0] if a_raw[0] > 0 else 1.0
        delta_max = max(0.0, (1.0 - cur) / (2.0 * a0))
        if delta_max <= 0:
            continue

        delta = step_factor * delta_max * rng.random()
        a_raw[p] += delta
        sum_raw += delta

        # --------------------------------------------------------------
        # Enforce the global max‑convolution bound.
        # --------------------------------------------------------------
        max_conv = max_convolution_fft(a_raw)
        if max_conv > 1.0:
            scale = np.sqrt(max_conv)
            a_raw /= scale
            sum_raw = a_raw.sum()

        # --------------------------------------------------------------
        # Record the best vector seen so far (largest L₁ ⇒ smallest score).
        # --------------------------------------------------------------
        score = 2.0 * n / (sum_raw ** 2)
        if score < best_score:
            best_score = score
            best_raw = a_raw.copy()
            best_sum = sum_raw
            print(f"[{it:6d}] new best score={best_score:.6f}, sum={best_sum:.3f}")

        # --------------------------------------------------------------
        # Simple annealing on the step size.
        # --------------------------------------------------------------
        if it % 2000 == 0:
            step_factor = max(step_factor * 0.95, 0.05)

    # ------------------------------------------------------------------
    # 5. Normalise the best raw vector to L₁ = 1 and return it.
    # ------------------------------------------------------------------
    final_seq = (best_raw / best_raw.sum()).tolist()

    # One final evaluation for reporting (optional, may be omitted).
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"[final] iterations={it}, best_score={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluation failed:", exc)

    return final_seq
