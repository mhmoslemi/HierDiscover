def convolve_fft(x, y):
    size = 1 << ((len(x) + len(y) - 1).bit_length())
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:len(x)+len(y)-1]
    return conv

"""
Propose a step‑function height sequence using a smooth‑max Adam optimiser.

The algorithm works on a fixed length `n` (default 2048).  It keeps the
vector `a` on the probability simplex (non‑negative, unit ℓ₁‑norm) by
projecting after each update.  At every iteration we

1. compute the full self‑convolution `c = a * a` (FFT based),
2. form a smooth approximation of the maximum using a soft‑max with a
   temperature `β` that is slowly increased,
3. obtain the gradient of this smooth objective:
      ∇_a 𝔼 ≈ 2·(p ⋆ a)      where p are the soft‑max weights,
   i.e. a cross‑correlation that is also evaluated with FFT,
4. update `a` with the Adam optimiser,
5. project back onto the simplex.

Every few hundred steps we evaluate the true score with the provided
`evaluate_sequence`.  The best sequence seen is kept and finally
returned (as a plain Python list).  If a previous construction
`height_sequence_1` exists it is used to initialise the optimisation.

The method is deliberately different from the earlier projected
sub‑gradient descent: it uses a smooth surrogate of the max,
adaptive moment estimation and a temperature schedule, which together
help to escape plateaus and achieve lower worst‑case convolution peaks.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive, return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
def _convolve_fft(x: np.ndarray, y: np.ndarray, size: int = None) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT (real‑valued)."""
    n = x.shape[0] + y.shape[0] - 1
    if size is None:
        size = 1 << (n - 1).bit_length()   # next power of two
    fx = np.fft.rfft(x, size)
    fy = np.fft.rfft(y, size)
    conv = np.fft.irfft(fx * fy, size)[:n]
    return conv

# ----------------------------------------------------------------------
def _softmax(x: np.ndarray, beta: float) -> np.ndarray:
    """Stable soft‑max with temperature β."""
    if beta == 0.0:
        # uniform weights
        w = np.full_like(x, 1.0 / x.size)
        return w
    z = beta * x
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)
    sum_exp = np.sum(exp_z)
    return exp_z / sum_exp

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     length: int = 2048,
                     **kwargs) -> list[float]:
    """
    Search for a low‑score height sequence.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate wall‑clock budget in seconds (the function will stop
        a couple of seconds before this to avoid a hard kill).
    length : int
        Length `n` of the sequence to optimise (default 2048).

    Returns
    -------
    list[float]
        The best sequence discovered (non‑negative, ℓ₁‑norm = 1).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time budget
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    n = int(length)

    # ------------------------------------------------------------------
    # Initialise from a previous construction if available; otherwise use a
    # random Dirichlet start (concentration 0.6 gives a mildly sparse vector).
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        prev = np.maximum(prev, 0.0)
        if prev.size > 0:
            if prev.size == n:
                a0 = prev
            elif prev.size < n:
                a0 = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                a0 = prev[:n]
            s = a0.sum()
            if s > 0:
                a = a0 / s
            else:
                a = np.full(n, 1.0 / n)
        else:
            a = np.full(n, 1.0 / n)
    else:
        # random Dirichlet (alpha=0.6) to start from a “sparse” point.
        a = rng.gamma(0.6, 1.0, size=n)
        a = _project_simplex(a)

    # ------------------------------------------------------------------
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.08   # initial learning rate

    # Soft‑max temperature schedule
    beta_temp = 0.5        # start low (smooth)
    beta_temp_max = 5.0    # final temperature – sharper approximation
    beta_factor = 1.001    # multiplicative increase per iteration

    # Book‑keeping
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    iteration = 0
    last_eval_iter = 0
    eval_every = 200      # evaluate true objective every this many iters

    # Pre‑compute FFT size (same for all convolutions)
    conv_len = 2 * n - 1
    fft_size = 1 << (conv_len - 1).bit_length()  # power of two >= conv_len

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1. Full autocorrelation via FFT
        conv = _convolve_fft(a, a, size=fft_size)          # length 2n‑1

        # --------------------------------------------------------------
        # 2. Soft‑max weights for a smooth surrogate of max(conv)
        p = _softmax(conv, beta_temp)

        # --------------------------------------------------------------
        # 3. Gradient of the smooth surrogate.
        #    grad = 2 * (p ⋆ a)   where ⋆ denotes cross‑correlation.
        a_rev = a[::-1]
        corr_full = _convolve_fft(p, a_rev, size=fft_size)  # length 3n‑2
        grad = 2.0 * corr_full[n - 1:n - 1 + n]              # slice to length n

        # --------------------------------------------------------------
        # 4. Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)
        step = lr0 * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        a_candidate = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        # 5. Occasional stochastic perturbations to escape shallow minima
        if rng.random() < 0.015:
            # multiplicative log‑normal jitter on a random subset (≈5% of entries)
            idxs = rng.choice(n, size=max(1, n // 20), replace=False)
            factors = np.exp(rng.normal(0.0, 0.12, size=idxs.shape))
            a_candidate[idxs] *= factors
            a_candidate = _project_simplex(a_candidate)

        # --------------------------------------------------------------
        # 6. Periodic evaluation with the official objective
        if iteration - last_eval_iter >= eval_every:
            last_eval_iter = iteration
            score = float(evaluate_sequence(a_candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best = a_candidate.copy()
                best_score = score
                print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 7. Prepare for next iteration
        a = a_candidate

        # --------------------------------------------------------------
        # 8. Anneal the soft‑max temperature
        beta_temp = min(beta_temp_max, beta_temp * beta_factor)

        # --------------------------------------------------------------
        # 9. Early exit if we are already very close to the theoretical lower bound
        theoretical_lb = 2.0 * n / (2 * n - 1)   # = 1 + 1/(2n-1)
        if best_score <= theoretical_lb * 1.02:
            print("[early stop] near‑theoretical bound reached")
            break

    # Final evaluation (in case the last iteration was not evaluated)
    final_score = float(evaluate_sequence(best.tolist()))
    if final_score < best_score:
        best_score = final_score
        best = best.copy()
    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]
