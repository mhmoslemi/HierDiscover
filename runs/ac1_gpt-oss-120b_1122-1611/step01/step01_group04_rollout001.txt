"""
Propose a coefficient sequence that aims to beat the previous ~1.574 bound.

Algorithm
---------

1. **Pick a working length** – use the length of the provided
   ``height_sequence_1`` if it exists, otherwise fall back to a modest
   size (≈2000).  This keeps the linear‑program (LP) tractable.

2. **Generate a portfolio of analytic shapes**:
   * power‑law families  (a_i = (i+shift)^β)
   * sine‑hump families  (a_i = sin(π·(i+0.5)/n)^γ)
   * a singular “arcsine” shape   a_i = 1/√((i+½)(n‑½‑i))
   * a parametrised variant   a_i = 1/((i+½)(n‑½‑i))^α
   All shapes are normalised to unit L¹ (the actual scale does not
   affect the evaluation).

3. **Pick the best shape** among the analytic candidates using the
   supplied ``evaluate_sequence`` function.

4. **Linear‑program refinement** (optional – used when `cvxpy` is
   available).  For the current shape `f` we compute its maximal
   convolution value `M = max(f*f)`.  We then solve

        maximise   Σ b_i
        subject to  (f * b)[k] ≤ M   for every shift k
                    b_i ≥ 0

   This is a pure linear program because `f` is fixed.  The solution
   `b` has a larger L¹ norm while respecting the same convolution bound.
   Mixing the two vectors, `f ← (f + b)/2`, strictly reduces the
   maximal convolution (or leaves it unchanged).  The step is repeated
   a few times while time permits.

5. **Local stochastic refinement** – cheap multiplicative block
   perturbations (as in the previous baseline) together with occasional
   random mass transfers between two positions.  Each perturbation that
   improves the score is kept.

6. **Return the best sequence found** as plain Python floats.  All
   intermediate prints are for progress monitoring only.

The whole routine stops well before the 1000 s budget, leaving ample
time for the LP and the stochastic phase.
"""

import time
import math
import numpy as np
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """Return a_i = (i + shift)^beta (beta may be negative)."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    seq = np.clip(seq, 0.0, 1000.0)
    seq /= np.sum(seq)          # normalise L¹ to 1 (scale invariant)
    return seq


def generate_sine(gamma: float, n: int) -> np.ndarray:
    """Return a_i = sin(π·(i+0.5)/n)^gamma (symmetric hump)."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    seq = np.clip(seq, 0.0, 1000.0)
    seq /= np.sum(seq)
    return seq


def generate_arcsine(alpha: float, n: int) -> np.ndarray:
    """Arcsine‑type shape: a_i = 1/((i+½)(n‑½‑i))^α."""
    i = np.arange(n, dtype=np.float64)
    # distance to nearest boundary, shifted by 0.5
    dist = (i + 0.5) * (n - 0.5 - i)
    # avoid division by zero – cap very small distances
    eps = 1e-12
    seq = 1.0 / np.maximum(dist, eps) ** alpha
    seq = np.clip(seq, 0.0, 1000.0)
    seq /= np.sum(seq)
    return seq


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width 0.5 % – 5 % of n
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # log‑normal tweak
    cand[st:en] *= factor
    # occasional symmetrisation helps for symmetric shapes
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def lp_improve(f_vec: np.ndarray, M: float, max_iters: int = 5):
    """
    Linear‑program refinement described in the algorithm description.
    Returns a new vector (average of f and the LP solution) or None if
    the LP could not be solved.
    """
    try:
        import cvxpy as cp
    except Exception:                # cvxpy not available – skip LP
        return None

    n = f_vec.size
    # Build sparse Toeplitz matrix A where (A @ b)[k] = (f * b)[k]
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.repeat(f_vec, n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b = cp.Variable(n, nonneg=True)
    constraints = [A @ b <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    # Solve with a fast first‑order solver; warm_start speeds up repeats
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception as e:
        print("LP solve error:", e)
        return None

    if b.value is None:
        return None

    b_opt = np.maximum(0.0, np.array(b.value).ravel())
    # Mix old and new vectors – the average never exceeds the previous bound
    new_vec = 0.5 * (f_vec + b_opt)
    # Re‑normalise to unit L¹ (purely for numerical stability)
    new_vec /= np.sum(new_vec) + 1e-12
    return new_vec

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------

    print(f"Working length = {n}")

    # ------------------------------------------------------------------

    candidates = []

    gammas = [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
    for gamma in gammas:

            best_score = score

    print(f"final best score = {best_score:.6f}")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
