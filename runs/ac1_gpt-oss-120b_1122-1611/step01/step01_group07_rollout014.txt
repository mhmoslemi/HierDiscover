def proj_simplex(v):
    # returns projection of v onto simplex {x>=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def tukey_window(n, a):
    if a <= 0:
        return np.ones(n)
    if a >= 1:
        # using Hann window
        return 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    w = np.empty(n)
    k = np.arange(n)
    period = n-1
    # first tapered region
    t = (2 * k) / (a * period)
    w[k <= a*period/2] = 0.5 * (1 + np.cos(np.pi * (t[k <= a*period/2] - 1)))
    # middle region flat
    w[(k > a*period/2) & (k < (1 - a/2) * period)] = 1.0
    # second taper
    t2 = (2 * k / period) - (2 / a) + 1
    w[k >= (1 - a/2) * period] = 0.5 * (1 + np.cos(np.pi * t2[k >= (1 - a/2) * period]))
    return w

        else:
            w[i] = 1.0
    return w

"""
Propose a low‑score step‑function sequence.

The algorithm consists of three phases:

1. **Fast parametric sweep** – many families of analytic windows
   (exponential, power‑exponential, sinusoidal, Beta, Gaussian,
   Tukey) are sampled.  The score is computed with a fast FFT‑based
   convolution; the best window becomes the seed for the next stage.

2. **Sub‑gradient refinement** – the current distribution is kept on the
   probability simplex (∑a_i = 1, a_i ≥ 0).  At each iteration the
   convolution shift with maximal value is identified; a sub‑gradient
   2·a_{k‑i} points away from the offending shift.  A diminishing step
   size together with projection onto the simplex reduces the
   maximal autocorrelation.

3. **Stochastic block perturbations** – a few hundred random
   multiplicative perturbations (log‑normal factors on random intervals)
   are applied to escape possible local minima.

All candidates are evaluated with the provided ``evaluate_sequence``.
The routine respects the 1000 s time budget and prints progress.
"""

import numpy as np
import time
import math


def _evaluate(seq):
    """Thin wrapper around the externally provided evaluate_sequence."""
    # The external function expects a plain Python list.
    return float(evaluate_sequence(seq.tolist()))


def fast_score(arr, n, fft_len):
    """
    Compute the evaluation metric using FFT.
    This is much faster than the original evaluate_sequence and
    returns exactly the same value (up to floating‑point noise).
    """
    a = np.clip(arr, 0.0, 1000.0)
    sum_a = np.sum(a)
    if sum_a < 0.01:
        return np.inf
    # linear convolution via FFT (zero‑padded to fft_len)
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
    max_b = np.max(conv)
    return float(2 * n * max_b / (sum_a * sum_a))


def simplex_projection(v, s=1.0):
    """
    Project a vector v onto the simplex {x >= 0, sum x = s}.
    Implements the algorithm from
    Duchi et al., “Efficient Projections onto the L1‑Ball”.
    """
    if s < 0:
        raise ValueError("s must be non‑negative")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1  # zero‑based
        theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def tukey_window(n, a):
    """Return a Tukey window of length n with taper parameter a (0<a<=1)."""
    if a <= 0.0:
        return np.ones(n, dtype=np.float64)
    if a >= 1.0:
        # a==1 gives a Hann window
        return 0.5 * (1 - np.cos(2.0 * np.pi * np.arange(n) / (n - 1)))
    w = np.empty(n, dtype=np.float64)
    N = n - 1
    half = a * N / 2.0
    for i in range(n):
        if i < half:                                   # left taper
            w[i] = 0.5 * (1 + np.cos(np.pi * (2.0 * i / (a * N) - 1.0)))
        elif i > N - half:                             # right taper
            w[i] = 0.5 * (1 + np.cos(np.pi * (2.0 * i / (a * N) - 2.0 / a + 1.0)))
        else:                                          # flat centre
            w[i] = 1.0
    return w


def make_block_perturbation(base, rng):
    """Multiplicative log‑normal perturbation of a contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width roughly between n/200 and n/12
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional mirror symmetry (helps stability)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed=42, budget_s=1000, **_):
    """
    Search for a coefficient sequence that minimises ``evaluate_sequence``.
    Returns the best sequence found before the time limit expires.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # a few seconds margin

    # --------------------------------------------------------------------
    # Determine problem size
    # --------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 4806
        base_seq = np.ones(n, dtype=np.float64)

    # FFT length (next power of two >= 2n‑1)
    fft_len = 1 << ((2 * n - 1).bit_length())

    # --------------------------------------------------------------------
    # Keep the best candidate seen so far
    # --------------------------------------------------------------------
    best_arr = np.clip(base_seq, 0.0, 1000.0)
    best_score = fast_score(best_arr, n, fft_len)

    # --------------------------------------------------------------------
    # 1) Fast parametric exploration
    # --------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    X = idx / (n - 1)                     # normalised coordinate in [0,1]
    N = n - 1

    # --- Exponential ----------------------------------------------------
    for r in np.linspace(0.10, 0.99, 25):
        seq = r ** idx
        sc = fast_score(seq, n, fft_len)
        if sc < best_score:
            best_score, best_arr = sc, seq.copy()
            print(f"New best (exp r={r:.3f}) {best_score:.6f}")

    # --- Power‑exponential ------------------------------------------------
    for alpha in np.linspace(-0.5, 2.0, 12):
        for r in np.linspace(0.30, 0.95, 15):
            seq = ((idx + 1.0) ** alpha) * (r ** idx)
            seq = np.clip(seq, 0.0, None)
            if np.max(seq) == 0:
                continue
            sc = fast_score(seq, n, fft_len)
            if sc < best_score:
                best_score, best_arr = sc, seq.copy()
                print(f"New best (pow‑exp a={alpha:.2f} r={r:.3f}) {best_score:.6f}")

    # --- Sinusoidal power family -----------------------------------------
    for p in np.linspace(0.5, 3.0, 22):
        seq = np.sin(np.pi * X) ** p
        seq = np.clip(seq, 0.0, None)
        if np.max(seq) == 0:
            continue
        sc = fast_score(seq, n, fft_len)
        if sc < best_score:
            best_score, best_arr = sc, seq.copy()
            print(f"New best (sin^p p={p:.2f}) {best_score:.6f}")

    # --- Beta‑type windows (x^α (1‑x)^β) ---------------------------------
    alphas = np.linspace(0.2, 5.0, 15)
    betas = np.linspace(0.2, 5.0, 15)
    for a_ in alphas:
        for b_ in betas:
            seq = (X ** a_) * ((1.0 - X) ** b_)
            # avoid the trivial zero vector
            if np.max(seq) == 0:
                continue
            sc = fast_score(seq, n, fft_len)
            if sc < best_score:
                best_score, best_arr = sc, seq.copy()
                print(f"New best (beta a={a_:.2f} b={b_:.2f}) {best_score:.6f}")

    # --- Gaussian windows -------------------------------------------------
    mu = 0.5
    for sigma in np.linspace(0.05, 0.45, 20):
        seq = np.exp(-0.5 * ((X - mu) / sigma) ** 2)
        sc = fast_score(seq, n, fft_len)
        if sc < best_score:
            best_score, best_arr = sc, seq.copy()
            print(f"New best (gauss σ={sigma:.3f}) {best_score:.6f}")

    # --- Tukey windows ----------------------------------------------------
    for a in np.linspace(0.1, 1.0, 15):
        seq = tukey_window(n, a)
        sc = fast_score(seq, n, fft_len)
        if sc < best_score:
            best_score, best_arr = sc, seq.copy()
            print(f"New best (tukey a={a:.2f}) {best_score:.6f}")

    print(f"Parametric sweep completed – best score {best_score:.6f}")

    # --------------------------------------------------------------------
    # 2) Sub‑gradient descent on the simplex
    # --------------------------------------------------------------------
    # Normalise to a probability vector (sum = 1).  Scaling does not affect
    # the objective, but the simplex constraint simplifies the update.
    p = np.clip(best_arr, 0.0, None)
    total = np.sum(p)
    if total > 0:
        p = p / total
    else:
        p = np.full_like(p, 1.0 / n)

    it = 0
    while time.monotonic() < deadline:
        cur_score = fast_score(p, n, fft_len)
        if cur_score < best_score:
            best_score, best_arr = cur_score, p.copy()
            print(f"Iter {it:05d} – new best {best_score:.6f}")

        # convolution and the index attaining the maximum
        A = np.fft.rfft(p, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[: 2 * n - 1]
        k_max = int(np.argmax(conv))

        # sub‑gradient of the max‑convolution term
        grad = np.zeros_like(p)
        # grad_i = 2 * p_{k_max-i} whenever the index is inside the support
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                grad[i] = 2.0 * p[j]

        # diminishing step size
        eta = 0.2 / np.sqrt(it + 1.0)
        p_new = p - eta * grad
        # project back onto the simplex (keeps non‑negativity and sum=1)
        p = simplex_projection(p_new, s=1.0)

        it += 1

    # --------------------------------------------------------------------
    # 3) Stochastic block perturbations (optional fine‑tuning)
    # --------------------------------------------------------------------
    parent = best_arr.copy()
    while time.monotonic() < deadline:
        cand = make_block_perturbation(parent, rng)
        sc = fast_score(cand, n, fft_len)
        if sc < best_score:
            best_score, best_arr = sc, cand.copy()
            print(f"Block perturb – new best {best_score:.6f}")
        # momentum towards best
        parent = 0.85 * parent + 0.15 * best_arr

    # --------------------------------------------------------------------
    # Return the best sequence (clip to allowed range for safety)
    # --------------------------------------------------------------------
    final_seq = np.clip(best_arr, 0.0, 1000.0)
    # verify with the official scorer (tiny overhead, done once)
    official_score = _evaluate(final_seq)
    print(f"Search completed – final official score {official_score:.6f}")
    return final_seq.tolist()
