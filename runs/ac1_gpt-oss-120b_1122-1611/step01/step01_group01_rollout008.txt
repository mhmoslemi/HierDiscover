"""
Propose a step‑function with a low self‑convolution peak.

The algorithm combines three ideas:

1. **Soft‑max weighted exponentiated‑gradient** – at each iteration we
   compute the self‑convolution `c = a*a`.  Instead of using only the
   single active peak, we form a soft weight
   `p_k ∝ exp(γ·(c_k – max(c)))` (γ grows from 0.1 to ≈1.5).  The
   weighted partner vector  
   `w_i = Σ_k p_k·a_{k‑i}` (zero outside the support) is the gradient
   of the weighted sum of the convolution entries.  An exponentiated‑
   gradient step `a_i ← a_i·exp(‑η·w_i)` (η decays from ≈0.9 to 0.02)
   flattens the whole convolution profile, not just the current maximum.

2. **Occasional block perturbations** – with a small probability we
   multiply a random contiguous block by a log‑normal factor and
   renormalise.  This injects diversity and helps escape plateaus.

3. **Annealing‑style local moves** – in the final part of the budget we
   perform cheap two‑index mass‑transfer proposals (move a random
   amount from one index to another) and accept them if they improve the
   score (or with a tiny Metropolis probability).  This refines the
   solution found by the gradient phase.

All operations are O(n log n) using FFTs; the routine works for the
provided `height_sequence_1` (≈ 3800 entries) and stays well within the
1000 s time limit.  The best vector encountered is returned as a plain
list of Python floats, already clipped to the bounds enforced in
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utility functions
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a non‑negative real vector `a` using the FFT.
    Returns a vector of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _weighted_partner(a: np.ndarray, p: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Weighted partner vector w_i = Σ_k p_k·a_{k‑i} (zero when out of range).

    Parameters
    ----------
    a : (n,) ndarray
        Current step‑function (non‑negative, sum=1).
    p : (2·n‑1,) ndarray
        Non‑negative weights (e.g. soft‑max of the convolution).
    fft_len : int
        Length of the FFT (power of two ≥ 3·n‑2).

    Returns
    -------
    w : (n,) ndarray
        Weighted partner vector.
    """
    # convolution of reversed a with p
    a_rev = a[::-1]
    fa = np.fft.rfft(a_rev, n=fft_len)
    fp = np.fft.rfft(p, n=fft_len)
    conv = np.fft.irfft(fa * fp, n=fft_len)  # length = n + (2n‑1) – 1 = 3n‑2
    start = a.size - 1  # element that aligns with original index 0
    return conv[start : start + a.size]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block of `a` by a log‑normal factor,
    then renormalise to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 5))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.25))  # moderate log‑normal change
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise the
    score computed by `evaluate_sequence`.  The routine runs for roughly
    ``budget_s`` seconds (leaving a small safety margin) and uses a
    combination of soft‑max exponentiated‑gradient descent, occasional
    block perturbations and a short annealing phase.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.2)

    # --------------------------------------------------------------
    # initialise the vector a (unit sum, non‑negative)
    # --------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # fallback – a moderate‑size uniform vector
        base = np.ones(256, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, None)
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total                      # unit sum → scale‑invariant

    # pre‑compute FFT lengths
    fft_self = _next_pow2(2 * n - 1)
    fft_cross = _next_pow2(3 * n - 2)

    # keep track of the best vector seen so far
    conv = _conv_self(a, fft_self)
    best_score = 2.0 * n * np.max(conv)
    best = a.copy()
    start_time = time.monotonic()

    # hyper‑parameter schedules
    eta_start, eta_end = 0.9, 0.02          # step‑size decay
    gamma_start, gamma_end = 0.10, 1.5     # soft‑max sharpening
    perturb_prob = 0.02                    # chance of a block perturb per iteration
    anneal_start_frac = 0.85               # when to start annealing moves
    anneal_prob = 0.03                     # chance per iteration to try a two‑index transfer
    temp_start, temp_end = 1e-3, 1e-5      # temperature for Metropolis acceptance

    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)  # 0 … 1

        # ----------------------------------------------------------
        # 1. compute self‑convolution and current score
        # ----------------------------------------------------------
        conv = _conv_self(a, fft_self)
        max_val = float(conv.max())
        cur_score = 2.0 * n * max_val

        # record improvement
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # 2. weighted exponentiated‑gradient step
        # ----------------------------------------------------------
        eta = eta_start * (1.0 - frac) + eta_end * frac
        gamma = gamma_start * (1.0 - frac) + gamma_end * frac

        # soft‑max over convolution values (max exponent = 0)
        # using double‑precision exponentials is safe because the

        exp_weights = np.exp(gamma * (conv - max_val))
        p = exp_weights / exp_weights.sum()

        w = _weighted_partner(a, p, fft_cross)

        # limit the magnitude of the update to keep numerics stable
        w = np.clip(w, 0.0, 20.0)

        # exponentiated‑gradient multiplicative update
        log_factor = -eta * w
        log_factor = np.clip(log_factor, -20.0, 20.0)   # avoid under/overflow
        a = a * np.exp(log_factor)

        # renormalise to unit sum
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ----------------------------------------------------------
        # 3. occasional stochastic block perturbation
        # ----------------------------------------------------------
        if rng.random() < perturb_prob:
            a = _random_block_perturb(a, rng)

        # ----------------------------------------------------------
        # 4. annealing‑style local tiny moves (only in the last phase)
        # ----------------------------------------------------------
        if frac > anneal_start_frac and rng.random() < anneal_prob:
            # propose moving a random amount from i → j
            i, j = rng.choice(n, size=2, replace=False)
            if a[i] > 0.0:
                delta = a[i] * rng.uniform(0.0, 0.5) * rng.random()
                a2 = a.copy()
                a2[i] -= delta
                a2[j] += delta
                # re‑normalise (the move preserves sum, but rounding may drift)
                a2_sum = a2.sum()
                if a2_sum != 0.0:
                    a2 /= a2_sum
                # evaluate
                conv2 = _conv_self(a2, fft_self)
                score2 = 2.0 * n * np.max(conv2)
                # simple Metropolis acceptance
                if score2 < cur_score:
                    a = a2
                else:
                    # temperature decays linearly with frac
                    temp = temp_start * (1.0 - frac) + temp_end * frac
                    if rng.random() < np.exp(-(score2 - cur_score) / max(temp, 1e-12)):
                        a = a2

        # loop continues until time runs out

    # ------------------------------------------------------------------
    # Return the best vector found (clipped to the limits imposed by
    # `evaluate_sequence` for safety).
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
