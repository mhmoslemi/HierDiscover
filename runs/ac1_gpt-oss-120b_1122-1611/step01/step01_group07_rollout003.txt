def project_simplex(v):
    """Project onto simplex sum(v)=1, v>=0."""
    # see Duchi et al. (2008)
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def step_exponential(a, eta):
    b = np.convolve(a, a)
    max_idx = np.argmax(b)
    # compute gradient contributions
    grad = np.zeros_like(a)
    n = len(a)
    # determine bounds for i
    i_start = max(0, max_idx - (n-1))
    i_end = min(n-1, max_idx)
    # for i in that range, j = max_idx - i
    # gradient = 2 * a[j]
    grad[i_start:i_end+1] = 2 * a[max_idx - np.arange(i_start, i_end+1)]
    # compute multiplicative factor: factor_i = exp(-eta * grad_i)
    factor = np.exp(-eta * grad)
    a_new = a * factor
    # renormalize to sum=1
    sum_a = a_new.sum()
    if sum_a == 0:
        # fallback to uniform
        a_new = np.ones_like(a) / len(a)
        sum_a = 1.0
    a_new = a_new / sum_a
    return a_new

"""
Propose a step‑function sequence with a low evaluation score.

Algorithm
---------

1️⃣ **Initialisation** – Determine the problem size (the length of the
   global ``height_sequence_1`` if it exists, otherwise 4806).  Start from a
   uniform distribution (sum = 1) and keep track of the best sequence seen.

2️⃣ **Broad parametric sweep** – Generate many analytic families (Gaussian,
   super‑Gaussian, Hann / Hamming / Blackman windows, DPSS windows when
   SciPy is available).  Each candidate is normalised to have sum = 1 and
   evaluated; the best candidate becomes the starting point for the next
   stage.

3️⃣ **Guided stochastic refinement** – Repeatedly improve the current
   candidate until the time budget expires.

   * **Gradient‑step** (≈ 60 % of iterations) – Compute the convolution,
     locate the index where the maximum occurs, form the sub‑gradient of
     that maximum (∂bₖ/∂aᵢ = 2·aₖ₋ᵢ for the relevant indices) and apply a
     multiplicative update `a ← a·exp(−η·grad)`.  After the update the
     vector is projected back onto the probability simplex (renormalised
     to sum = 1).

   * **Block perturbation** (≈ 40 % of iterations) – Randomly pick a
     contiguous block, multiply it by a log‑normal factor and renormalise.
     Occasionally enforce mild symmetry.

   * **Momentum & occasional large restart** – The current search point
     is nudged towards the global best (`parent ← 0.9·parent + 0.1·best`);
     rare large‑scale perturbations help escape local minima.

4️⃣ **Return** – The best normalised sequence found before the deadline
   (clipped to the allowed range) is returned as a plain Python list.
"""

import time
import math
import numpy as np

# Helper functions (top‑level – no closures)

def compute_score(seq: np.ndarray, n: int) -> float:
    """
    Compute the objective value 2·n·max(conv(a,a)) / (sum(a)²).

    The sequence is assumed to be non‑negative.
    """
    if seq.size == 0:
        return np.inf
    total = seq.sum()
    if total <= 0.0:
        return np.inf
    # convolution (linear, not circular)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))


def evaluate_and_update(candidate: np.ndarray,
                        best_seq: np.ndarray,
                        best_score: float,
                        n: int) -> tuple[np.ndarray, float]:
    """
    Normalise ``candidate`` to sum = 1, evaluate it and, if it improves,
    replace the current best sequence.  Returns the (possibly) updated
    best sequence and its score.
    """
    total = candidate.sum()
    if total <= 0.0:
        return best_seq, best_score
    cand_norm = candidate / total
    score = compute_score(cand_norm, n)
    if np.isfinite(score) and score < best_score:
        print(f"New best score {score:.6f}")
        return cand_norm, score
    return best_seq, best_score


def make_block_perturbation(base: np.ndarray,
                            rng: np.random.Generator) -> np.ndarray:
    """
    Return a copy of ``base`` where a random contiguous block has been
    multiplied by a log‑normal factor.  The result is renormalised to
    sum = 1.
    """
    cand = base.copy()
    n = cand.size
    # block width roughly between 0.5 % and 8 % of the length
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    # keep values in the allowed range before renormalisation
    np.clip(cand, 0.0, 1000.0, out=cand)
    total = cand.sum()
    if total <= 0.0:
        cand = np.ones_like(cand) / cand.size
    else:
        cand = cand / total
    return cand


def gradient_step(parent: np.ndarray, n: int, eta: float) -> np.ndarray:
    """
    Perform a multiplicative gradient descent step aimed at reducing the
    maximum entry of the self‑convolution.

    ``eta`` is a positive step size (the algorithm reduces it over time).
    The returned array is normalised to sum = 1.
    """
    conv = np.convolve(parent, parent)
    k_max = int(conv.argmax())               # index of the current maximum
    # build the sub‑gradient of b_k_max w.r.t. the entries of ``parent``
    grad = np.zeros_like(parent)
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    if i_end >= i_start:
        i_range = np.arange(i_start, i_end + 1)
        j = k_max - i_range
        grad[i_range] = 2.0 * parent[j]      # derivative of a_i a_{k-i}
    # multiplicative update (mirror‑descent on the simplex)
    factor = np.exp(-eta * grad)
    cand = parent * factor
    total = cand.sum()
    if total <= 0.0:
        cand = np.ones_like(parent) / n
    else:
        cand = cand / total
    return cand

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence (list of floats)
    that minimises the provided ``evaluate_sequence`` metric.
    The routine runs for at most ``budget_s`` seconds and returns the
    best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a few seconds as a safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine problem size (length of the step function)
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))                # type: ignore[name‑defined]
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)  # type: ignore[name‑defined]
    except Exception:
        n = 4806
        base_seq = np.ones(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initialise best sequence (uniform) and best score
    # ------------------------------------------------------------------
    best_seq = np.ones(n, dtype=np.float64) / n
    best_score = compute_score(best_seq, n)
    print(f"Initial uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep
    # ------------------------------------------------------------------
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0

    # Gaussian windows
    for sigma_frac in np.linspace(0.05, 0.5, 20):
        sigma = sigma_frac * n
        gauss = np.exp(-0.5 * ((idx - centre) / sigma) ** 2)
        best_seq, best_score = evaluate_and_update(gauss, best_seq, best_score, n)
        for eps in (0.0, 0.01, 0.05):
            cand = gauss + eps
            best_seq, best_score = evaluate_and_update(cand, best_seq, best_score, n)

    # Super‑Gaussian (higher order exponent)
    for p in (2, 3, 4, 6, 8):
        for sigma_frac in np.linspace(0.05, 0.5, 10):
            sigma = sigma_frac * n
            seq = np.exp(-((np.abs(idx - centre) / sigma) ** p))
            best_seq, best_score = evaluate_and_update(seq, best_seq, best_score, n)

    # Classical window families
    hann = 0.5 * (1 - np.cos(2.0 * np.pi * idx / (n - 1)))
    best_seq, best_score = evaluate_and_update(hann, best_seq, best_score, n)

    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * idx / (n - 1))
    best_seq, best_score = evaluate_and_update(hamming, best_seq, best_score, n)

    blackman = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * idx / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * idx / (n - 1))
    )
    best_seq, best_score = evaluate_and_update(blackman, best_seq, best_score, n)

    # DPSS (Slepian) windows – only if SciPy is available
    try:
        from scipy.signal import windows  # type: ignore[import]

        for NW in (2, 3, 4, 5, 6, 8, 10):
            dpss_seq = windows.dpss(n, NW, Kmax=1, sym=False).flatten()
            best_seq, best_score = evaluate_and_update(dpss_seq, best_seq, best_score, n)
    except Exception:
        # SciPy not installed – silently skip DPSS
        pass

    # ------------------------------------------------------------------
    # 2) Guided stochastic refinement
    # ------------------------------------------------------------------
    parent = best_seq.copy()
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Decide which elementary move to apply
        if rng.random() < 0.6:
            # ----- multiplicative gradient step -----
            # Step size decays linearly with the remaining budget
            elapsed = time.monotonic() - start_time
            frac = max(0.0, (deadline - time.monotonic()) / (deadline - start_time))
            eta = 0.012 * frac          # initial η ≈ 0.012, shrinks to ~0
            cand = gradient_step(parent, n, eta)
        else:
            # ----- random block perturbation -----
            cand = make_block_perturbation(parent, rng)

        # Evaluate and possibly replace the global best
        best_seq, best_score = evaluate_and_update(cand, best_seq, best_score, n)

        # Momentum towards the best known point
        parent = 0.9 * parent + 0.1 * best_seq

        # Rare large‑scale restart to diversify the search
        if rng.random() < 0.01:
            parent = best_seq.copy()
            # Inflate a randomly chosen block dramatically
            width = int(rng.integers(max(1, n // 50), max(2, n // 5)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            stop = min(n, start + width)
            parent[start:stop] *= float(np.exp(rng.normal(0.0, 0.5)))
            np.clip(parent, 0.0, 1000.0, out=parent)
            parent = parent / parent.sum()

        # Optional periodic status report
        if it % 5000 == 0:
            now = time.monotonic()
            print(f"Iter {it:6d} | time {now - start_time:6.1f}s | best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence discovered (clipped to the allowed range)
    # ------------------------------------------------------------------
    print(f"Search completed – best score {best_score:.6f}")
    final = np.clip(best_seq, 0.0, 1000.0)
    return final.tolist()
