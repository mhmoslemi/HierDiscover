def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, scipy.optimize as opt
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    n = 1500
    i = np.arange(n, dtype=np.float64) + 0.5
    a = 1.0 / np.sqrt(i * (n - i))
    a = a / a.sum()

    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, init score={best_score:.6f}")

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05  # initial learning rate
    alpha = 1.0  # smoothing temperature
    iteration = 0
    t_pow1 = 1.0  # beta1^t
    t_pow2 = 1.0
    while time.monotonic() < deadline:
        iteration += 1

        conv = np.convolve(a, a)  # length 2n-1
        max_conv = conv.max()
        # compute smooth weights
        # shift for numerical stability
        shift = conv.max()
        w_unnorm = np.exp(alpha * (conv - shift))
        sum_w = w_unnorm.sum()
        w = w_unnorm / sum_w

        # gradient = 2 * correlation of a with w
        grad = 2.0 * np.correlate(a, w, mode='valid')

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        t_pow1 *= beta1
        t_pow2 *= beta2
        m_hat = m / (1 - t_pow1)
        v_hat = v / (1 - t_pow2)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step
        # projection onto simplex
        a_candidate = _project_simplex(a_candidate)

        # occasional randomized perturbation
        if rng.random() < 0.01:
            idx = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))
            a_candidate[idx] *= factor
            a_candidate = _project_simplex(a_candidate)

        # Evaluate the candidate
        score = float(evaluate_sequence(a_candidate.tolist()))

        if np.isfinite(score) and score < best_score:
            best = a_candidate.copy()
            best_score = score
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        a = a_candidate

        # schedule adjustments
        if iteration % 5000 == 0:
            # increase alpha gradually to sharpen max
            alpha = min(alpha * 1.1, 100.0)
            # decay learning rate
            lr *= 0.9

        # early stop if below target
        if best_score <= 1.5025:
            print("[target reached]")
            break

    print(f"[final] best score={best_score:.6f} after {iteration} iters")
    return [float(x) for x in best]

  – occasional random “shake‑ups” to break symmetry
  – an auxiliary linear‑programming boost that enlarges the ℓ₁‑norm
    while keeping the current peak convolution unchanged.
The routine works on sequences normalised to ℓ₁‑norm 1; the returned

trade‑off between quality and speed).  The algorithm stops early

from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

            A = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')
            # solve   maximise 1ᵀb   s.t. A·b ≤ M,   b≥0
            res = linprog(c=-np.ones(n), A_ub=A, b_ub=np.full(2 * n - 1, M),
                          bounds=(0, None), method='highs')
            if res.success:
                b_opt = res.x
                # mix a little towards the LP solution (convex combination)
                t_mix = 0.25
                a = (1.0 - t_mix) * a + t_mix * (b_opt / b_opt.sum())
                a = _project_simplex(a)

        # --------------------------------------------------------------
        # early stop if we have already hit the target
        if best_score <= 1.5025:
            print("[target reached]")
            break

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # Return as plain Python list – evaluate_sequence will accept it.
    return [float(x) for x in best]
