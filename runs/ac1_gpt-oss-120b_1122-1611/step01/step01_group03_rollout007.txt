"""
Propose a candidate step‑function sequence.

The algorithm starts from the classic “alpha” sequence (C(2i,i)/4^i) which,
after L1‑normalisation, yields the well‑known bound π/2 ≈ 1.5708.  From this
point we iteratively move a small amount of mass from the coefficient that
contributes most to the current maximal convolution entry to a coefficient
that does **not** contribute to that entry.  The move is performed analytically
by updating the convolution vector with O(n) arithmetic (no FFT is needed
after the initial convolution).  A simple simulated‑annealing acceptance rule
allows occasional uphill moves; the temperature and the step size decay
gradually.  The best sequence seen during the run is returned.

The method is completely self‑contained, uses only NumPy and works under the
1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top level – no closures)
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """
    First n terms of a_i = C(2i,i) / 4^i.
    The recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1)) is used.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return a


def initial_convolution(a: np.ndarray) -> np.ndarray:
    """Exact linear convolution a * a (real numbers)."""
    return np.convolve(a, a)


def score_from_convolution(conv: np.ndarray, n: int) -> float:
    """Evaluation score = 2 * n * max(conv) (same as evaluate_sequence)."""
    return float(2.0 * n * np.max(conv))


def update_convolution(conv: np.ndarray, a: np.ndarray,
                       i_src: int, i_dst: int, delta: float) -> np.ndarray:
    """
    Return the new convolution after moving `delta` mass from index `i_src`
    to index `i_dst`.  The update is performed analytically in O(n)
    without recomputing a full convolution.

    Parameters
    ----------
    conv : np.ndarray
        Current convolution (length 2*n-1).
    a : np.ndarray
        Current coefficient vector (length n) **before** the move.
    i_src, i_dst : int
        Source and destination indices (0 ≤ i_src,i_dst < n, i_src≠i_dst).
    delta : float
        Amount of mass transferred (0 < delta ≤ a[i_src]).

    Returns
    -------
    np.ndarray
        Updated convolution of length 2*n-1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    # make a copy that we will modify
    new_conv = conv.copy()

    # contribution of the source term (subtraction)
    # new_conv[k] -= 2*delta*a[k-i_src] for all valid k
    new_conv[i_src:i_src + n] -= 2.0 * delta * a

    # contribution of the destination term (addition)
    new_conv[i_dst:i_dst + n] += 2.0 * delta * a

    # quadratic term (delta^2) handling
    d2 = delta * delta
    new_conv[2 * i_src] += d2
    new_conv[2 * i_dst] += d2
    cross = i_src + i_dst
    if cross < conv_len:
        new_conv[cross] -= 2.0 * d2

    return new_conv


def pick_destination(k_max: int, n: int, rng: np.random.Generator,
                     exclude: int) -> int:
    """
    Choose a destination index where the current maximal convolution entry
    receives **no** contribution (i.e. a_{k_max - j} = 0).  If such indices
    do not exist, pick a random index different from `exclude`.
    """
    # indices where k_max - j < 0  -->  j > k_max
    # or where k_max - j >= n   -->  j < k_max - (n-1)
    candidates = []
    if k_max + 1 < n:
        candidates.append(np.arange(k_max + 1, n, dtype=int))
    low = k_max - (n - 1)
    if low > 0:
        candidates.append(np.arange(0, low, dtype=int))

    if candidates:
        cand = np.concatenate(candidates)
        # ensure we do not pick the source index
        if exclude in cand:
            cand = cand[cand != exclude]
        if cand.size == 0:
            # fall back to any other index
            idx = rng.integers(0, n)
            while idx == exclude:
                idx = rng.integers(0, n)
            return int(idx)
        return int(rng.choice(cand))
    else:
        # No zero‑weight positions – pick any other index uniformly
        idx = rng.integers(0, n)
        while idx == exclude:
            idx = rng.integers(0, n)
        return int(idx)


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a coefficient sequence with a low evaluation score.
    The search runs until the allotted time expires and returns the best
    sequence found.  It starts from the analytic “alpha” sequence (or from
    the globally provided `height_sequence_1` if present) and performs a
    greedy/annealed mass‑transfer optimisation that updates the convolution
    analytically.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Initialise the sequence (use the global if available)
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        hs = globals()['height_sequence_1']
        if isinstance(hs, (list, np.ndarray)) and len(hs) > 0:
            n = int(len(hs))
            a = np.asarray(hs, dtype=np.float64)
        else:
            n = 4000
            a = generate_alpha_sequence(n)
    else:
        n = 4000
        a = generate_alpha_sequence(n)

    # Normalise to L1 = 1 (objective is scale invariant)
    a = np.maximum(0.0, a)
    total = a.sum()
    if total < 1e-12:
        # safety fallback
        a = generate_alpha_sequence(n)
        a = a / a.sum()
    else:
        a = a / total

    # --------------------------------------------------------------
    # Pre‑compute the initial convolution and score
    # --------------------------------------------------------------
    conv = initial_convolution(a)          # length 2*n-1
    cur_score = score_from_convolution(conv, n)
    best = a.copy()
    best_score = cur_score
    print(f"[init] n={n}, initial score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Search parameters
    # --------------------------------------------------------------
    start_time = time.time()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # keep a safety margin
    it = 0
    temperature = 0.006          # initial SA temperature
    temp_decay = 0.99995
    move_factor_start = 0.20     # initial fraction of source mass to move
    move_factor_min = 1e-5
    move_decay = 0.99994

    # Main optimisation loop
    while time.time() < deadline:
        it += 1

        # ----------------------------------------------------------
        # Identify the current maximal convolution entry
        # ----------------------------------------------------------
        k_max = int(np.argmax(conv))

        # Index that contributes the most to this entry (largest a_{k_max-i})
        i_src = min(k_max, n - 1)

        # Choose a destination that does not affect the current max
        i_dst = pick_destination(k_max, n, rng, i_src)

        # Dynamically decreasing step size
        move_factor = max(move_factor_min,
                          move_factor_start * (move_decay ** it))
        delta = move_factor * a[i_src]

        if delta <= 0.0:
            continue  # nothing to move

        # ----------------------------------------------------------
        # Compute the updated convolution analytically
        # ----------------------------------------------------------
        new_conv = update_convolution(conv, a, i_src, i_dst, delta)
        new_score = score_from_convolution(new_conv, n)

        # ----------------------------------------------------------
        # Simulated‑annealing acceptance test
        # ----------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(new_score - cur_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            # Apply the move to the coefficient vector
            a[i_src] -= delta
            a[i_dst] += delta
            # Numerical safety – clip tiny negatives that may arise
            a = np.clip(a, 0.0, None)

            # Update the stored convolution and score
            conv = new_conv
            cur_score = new_score

            # Track the overall best sequence
            if new_score < best_score:
                best_score = new_score
                best = a.copy()
                print(f"[{it:6d}] new best = {best_score:.6f}")

        # ----------------------------------------------------------
        # Temperature and move‑size schedule
        # ----------------------------------------------------------
        temperature *= temp_decay
        move_factor_start *= move_decay

        # ----------------------------------------------------------
        # Occasionally inject a small random perturbation to escape plateaus
        # ----------------------------------------------------------
        if it % 7000 == 0:
            sigma = 0.08 * (0.5 ** (it // 7000))  # gradually shrink
            pert = a * (1.0 + sigma * rng.standard_normal(n))
            pert = np.clip(pert, 0.0, None)
            pert_sum = pert.sum()
            if pert_sum > 1e-12:
                pert = pert / pert_sum
                # recompute convolution from scratch (costly but rare)
                conv = initial_convolution(pert)
                cur_score = score_from_convolution(conv, n)
                a = pert
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()
                    print(f"[{it:6d}] after perturbation best = {best_score:.6f}")

    # --------------------------------------------------------------
    # Finished – report the best found sequence
    # --------------------------------------------------------------
    print(f"[final] iterations={it}, best score={best_score:.6f}")
    return best.tolist()
