def project_simplex(v):
    # v is array, returns projection onto {x >=0, sum=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv/(np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def project_simplex(v):
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted) - 1
    ind = np.arange(len(v)) + 1
    cond = v_sorted - cssv / ind > 0
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

        grad = 2.0 * np.convolve(w, a[::-1])[n-1 : n-1 + n]
        # additive gradient descent step
        a_candidate = a - step * grad
        # enforce non‑negativity
        a_candidate = np.clip(a_candidate, 0.0, None)
        # renormalise to sum 1
        total = a_candidate.sum()
        if total > 0:
            a_candidate = a_candidate / total
        else:
            # degenerate, fallback to a
            a_candidate = a.copy()
        # occasional random mass transfer
        if rng.random() < prob_perturb:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = a_candidate[i] * rng.uniform(0.0, 0.5)
                a_candidate[i] -= delta
                a_candidate[j] += delta
                a_candidate = a_candidate / a_candidate.sum()
        # evaluate candidate score
        conv_cand = np.convolve(a_candidate, a_candidate)
        cand_score = 2.0 * n * conv_cand.max()
        # update best
        if cand_score < best_score:
            best = a_candidate.copy()
            best_score = cand_score
            a = a_candidate.copy()
            cur_score = cand_score
            print(f"[{it:6d}] new best {best_score:.6f}")
        else:
            # if candidate better than current a, move there, else keep a
            if cand_score < cur_score:
                a = a_candidate.copy()
                cur_score = cand_score
        # anneal parameters
        step *= step_decay
        if step < 1e-6:

        if it % 2000 == 0:

        if best_score < 1.5025:
            print(f"[early stop] reached target {best_score:.6f} at iteration {it}")
            break
    print(f"[final] iterations={it}, best score={best_score:.6f}")

"""
Gradient‑descent search for a low‑value step‑function.

The objective returned by `evaluate_sequence` is
    2 * n * max(conv(a, a))    (the L1‑norm is fixed to 1)
for a non‑negative sequence `a` of length `n`.  The algorithm
starts from the classic “α‑sequence’’ (a_i = C(2i,i)/4^i) which
gives the known bound π/2 ≈ 1.5708, then performs a smooth‑max
approximation of the `max` and a mirror‑descent (additive) update
projected onto the probability simplex.  The smoothness parameter
β is slowly increased so that the early iterations move the whole
distribution while later iterations focus on the current maximal
convolution entries.  Random mass‑transfer perturbations are added
occasionally to escape flat plateaus.  The best sequence seen before
the time budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """
    First n terms of a_i = C(2i,i) / 4^i.
    The recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1)) is used.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex:
        { x | x_i >= 0, sum_i x_i = 1 }.
    Implements the algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a sequence with a low value of `evaluate_sequence`.
    Returns the best sequence discovered before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Initialise the starting sequence.
    # Prefer the globally supplied construction (height_sequence_1);
    # otherwise fall back to the α‑sequence of moderate size.
    # --------------------------------------------------------------
    init_seq = None
    if "height_sequence_1" in globals():
        hs = globals()["height_sequence_1"]
        if isinstance(hs, (list, np.ndarray)) and len(hs) > 0:
            init_seq = np.asarray(hs, dtype=np.float64).copy()
            init_seq = np.maximum(init_seq, 0.0)
            total = init_seq.sum()
            if total > 1e-12:
                init_seq = init_seq / total
            else:
                init_seq = None

    if init_seq is None:
        # reasonable default size – enough to see the asymptotic behaviour
        n = 5000
        init_seq = generate_alpha_sequence(n)
        init_seq = init_seq / init_seq.sum()
    else:
        n = len(init_seq)

    a = init_seq.copy()                       # current iterate (always sums to 1)
    cur_score = 2.0 * n * np.max(np.convolve(a, a))   # current objective value
    best = a.copy()
    best_score = cur_score

    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for a fast, monotone descent)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    it = 0
    step = 0.5               # additive step size for gradient descent
    step_decay = 0.9995      # slow cooling of step
    beta = 1.0               # smooth‑max temperature
    beta_inc = 1.02
    beta_max = 200.0
    perturb_prob = 0.02      # probability of a random mass‑transfer

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # ----- convolution and smooth‑max weights -----
        conv = np.convolve(a, a)
        max_conv = conv.max()

        # smooth approximation of the max (log‑sum‑exp trick)
        exp_arg = beta * (conv - max_conv)          # values ≤ 0, stable
        w = np.exp(exp_arg)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----- gradient of the smooth‑max -----
        # grad_i = 2 * sum_j w_{i+j} * a_j
        grad = 2.0 * np.convolve(w, a[::-1])[n - 1 : n - 1 + n]

        # ----- additive mirror‑descent step (projected onto simplex) -----
        cand = a - step * grad
        cand = np.clip(cand, 0.0, None)
        total = cand.sum()
        if total > 0.0:
            cand = cand / total
        else:
            cand = a.copy()          # fallback in the unlikely event of zero mass

        # ----- occasional random perturbation (mass transfer) -----
        if rng.random() < perturb_prob:
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = cand[i] * rng.uniform(0.0, 0.5)
                cand[i] -= delta
                cand[j] += delta
                cand = cand / cand.sum()

        # ----- evaluate candidate (cheap direct evaluation) -----
        cand_score = 2.0 * n * np.max(np.convolve(cand, cand))

        # keep the best ever seen
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            print(f"[{it:6d}] new best score = {best_score:.6f}")

        # decide whether to move the current iterate
        if cand_score < cur_score:
            a = cand
            cur_score = cand_score
        # otherwise keep the old `a` (no uphill moves)

        # ----- schedule updates -----
        step = max(step * step_decay, 1e-6)   # never let step vanish completely
        beta = min(beta * beta_inc, beta_max)

        # periodical re‑centre on the best point to avoid drift
        if it % 2000 == 0:
            a = best.copy()
            cur_score = best_score

        # early exit if the (ad‑hoc) target is reached
        if best_score < 1.5025:
            print(f"[early stop] target reached: {best_score:.6f} at iter {it}")
            break

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    return best.tolist()
