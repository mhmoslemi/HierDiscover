"""
Projected gradient descent on the *soft* maximum of the autocorrelation.
The algorithm works on a probability simplex (∑a = 1, a_i ≥ 0) and
optimises the quantity

    score = 2·n·max_k (a*a)[k]

which is exactly the value returned by `evaluate_sequence` for a normalised
sequence (the denominator (∑a)² is 1).  At every iteration we

1. compute the convolution a*a by means of an FFT (O(n log n));
2. take the ``top_k`` largest entries of the convolution and build a
   sub‑gradient that is the sum of the gradients of those entries;
3. perform an Adam‑style update with a diminishing step‑size;
4. project the result back onto the simplex;
5. occasionally apply a multiplicative shake‑up to escape plateaus.

Multiple random restarts (including the previously discovered sequence
`height_sequence_1` when present) are performed within the allotted time.
The best sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries non‑positive – fall back to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _fft_convolve(a: np.ndarray) -> np.ndarray:
    """
    Return the (real) linear convolution a*a using an FFT.
    Length of the result is 2·len(a)‑1.
    """
    n = a.shape[0]
    # next power of two for efficient FFT
    n_fft = 1 << ((2 * n - 1).bit_length())
    A = np.fft.rfft(a, n=n_fft)
    c = np.fft.irfft(A * A, n=n_fft)
    return c[: 2 * n - 1]


# ----------------------------------------------------------------------
def _grad_from_top(a: np.ndarray, conv: np.ndarray, top_idxs: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of the (non‑smooth) max taken over the entries whose
    indices are supplied in ``top_idxs``.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in top_idxs:
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        idx = np.arange(i_min, i_max + 1, dtype=int)
        grad[idx] += 2.0 * a[k - idx]
    # average over the selected indices to keep the step magnitude moderate
    if top_idxs.size > 1:
        grad /= float(top_idxs.size)
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a low‑score step‑function height sequence.

    The function respects the 1000 s wall‑time limit (``budget_s`` is the
    total time budget, a small safety margin is kept).  It returns a list
    of non‑negative floats that should be accepted by ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # time budget – keep a small safety margin
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    # ------------------------------------------------------------------
    # problem dimension – we start with 1024 (the size used in the
    # reference implementation).  When a global sequence is available we
    # reuse its length.
    if "height_sequence_1" in globals():
        try:
            init_seq = list(globals()["height_sequence_1"])
            n = len(init_seq)
        except Exception:
            n = 1024
    else:
        n = 1024

    # pre‑compute FFT length once (constant throughout the run)
    n_fft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # hyper‑parameters (tuned for speed / quality)
    top_k = 5                     # number of peak entries used for the gradient
    initial_eta = 0.6             # base step size for Adam (learning‑rate)
    beta1, beta2 = 0.9, 0.999     # Adam moments
    eps = 1e-8
    shake_prob = 0.07             # probability of a random shake‑up each iteration
    shake_sigma = 0.18            # log‑normal shake magnitude

    best_score = np.inf
    best_seq = None

    # ------------------------------------------------------------------
    # prepare a collection of starting points (uniform + a few Dirichlet draws)
    start_points = []

    # 1) uniform distribution – always feasible
    start_points.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2) the previously discovered sequence, if it exists and matches length n
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.shape[0] == n:
                cand = np.maximum(cand, 0.0)
                cand = cand / cand.sum()
                start_points.append(cand)
        except Exception:
            pass

    # 3) a few random Dirichlet initialisations (concentration = 0.5 ⇒ spiky)
    for _ in range(3):
        a = rng.gamma(shape=0.5, scale=1.0, size=n)
        a = a / a.sum()
        start_points.append(a)

    # ------------------------------------------------------------------
    # main optimisation loop – each start point runs an Adam‑style descent
    for restart_id, a0 in enumerate(start_points):
        if time.monotonic() >= deadline:
            break

        # make a fresh copy so we never modify the stored start point
        a = a0.copy()
        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)

        iteration = 0
        while time.monotonic() < deadline:
            iteration += 1

            # --------------------------------------------------------------
            # 1) convolution via FFT
            A = np.fft.rfft(a, n=n_fft)
            conv = np.fft.irfft(A * A, n=n_fft)[: 2 * n - 1]

            # --------------------------------------------------------------
            # 2) identify the top_k entries (hard max for speed)
            if top_k == 1:
                top_idxs = np.array([int(np.argmax(conv))], dtype=int)
            else:
                # argpartition is O(N), gives unsorted top‑k indices
                part = np.argpartition(-conv, top_k - 1)[:top_k]
                top_idxs = part.astype(int)

            # --------------------------------------------------------------
            # 3) gradient of the selected peaks
            grad = _grad_from_top(a, conv, top_idxs)

            # --------------------------------------------------------------
            # 4) Adam update (step opposite the gradient)
            beta1_t = beta1 ** iteration
            beta2_t = beta2 ** iteration
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad ** 2)
            m_hat = m / (1.0 - beta1_t)
            v_hat = v / (1.0 - beta2_t)

            # diminishing step size – initial_eta is scaled by 1/√iter
            eta = initial_eta / np.sqrt(iteration)

            step = eta * m_hat / (np.sqrt(v_hat) + eps)
            a_candidate = a - step
            a_candidate = _project_simplex(a_candidate)

            # --------------------------------------------------------------
            # 5) occasional random shake‑up to increase exploration
            if rng.random() < shake_prob:
                pos = rng.integers(0, n)
                factor = np.exp(rng.normal(0.0, shake_sigma))
                a_candidate[pos] *= factor
                a_candidate = _project_simplex(a_candidate)

            # --------------------------------------------------------------
            # 6) evaluate
            score = float(evaluate_sequence(a_candidate.tolist()))
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a_candidate.copy()
                print(f"[improve] restart={restart_id} iter={iteration} score={best_score:.6f}")

            # --------------------------------------------------------------
            # 7) prepare for next iteration
            a = a_candidate

            # --------------------------------------------------------------
            # early termination if we are already extremely close to the
            # theoretical lower bound for this length
            theoretical_lb = 2.0 * n / (2 * n - 1)
            if best_score <= theoretical_lb * 1.012:
                # 1.2 % over the bound is excellent – stop early
                break

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should never happen) return uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] returning uniform sequence")

    print(f"[final] best score={best_score:.6f} after search")
    # convert to plain Python list as required by the framework
    return [float(x) for x in best_seq]
