```
def mass_shift(seq, step):
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = np.argmax(conv)
    # compute gradient contributions
    g = seq.copy()
    low = max(0, k - (n-1))
    high = min(n-1, k)
    idx = np.arange(low, high+1)
    g[idx] += seq[k - idx]
    i_max = idx[np.argmax(g[idx])] # Actually compare across entire g
    i_max = int(np.argmax(g))
    i_min = int(np.argmin(g))
    if seq[i_max] <= 0:
        return seq # can't shift
    delta = step * seq[i_max]
    if delta <= 0:
        return seq
    new_seq = seq.copy()
    new_seq[i_max] -= delta
    new_seq[i_min] += delta
    # ensure nonnegative
    new_seq[i_max] = max(new_seq[i_max], 0.0)
    return new_seq
```

"""
search for a step‑function minimizing the evaluation metric.

Algorithm
---------
1. Initialise from the provided global ``height_sequence_1`` (or a uniform
   vector).  Normalise the candidate to unit L1‑norm – the metric is
   homogeneous, so the scale does not matter.
2. Randomly sample a large family of analytic shapes
   a_i = (i+α+1)^{-p}·exp(-β·i)  (p,α,β are continuous parameters).
   The best sampled shape replaces the current best.
3. Refine the best shape with a cheap *gradient‑guided mass‑shift* move:
   – compute the auto‑convolution and locate the index k where the
     convolution attains its maximum;
   – the contribution of each entry a_j to that peak is g_j = a_j + a_{k‑j}
     (when the partner index is inside the array);
   – move a small fraction of mass from the index with the largest g_j
     to the index with the smallest g_j, preserving the total sum.
   – accept the move only if the metric improves; otherwise shrink the
     step size.
4. Occasionally perform the original “block‑perturbation’’ and a simple
   smoothing step to escape shallow local minima.
5. Continue while the 1000 s time budget permits, then return the best
   sequence found as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _metric(seq: np.ndarray) -> float:
    """
    Fast evaluation of the objective (the same formula used by evaluate_sequence).
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    max_b = float(conv.max())
    sum_a = float(seq.sum())
    if sum_a < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (sum_a * sum_a)

# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) block‑multiplicative perturbation; mirrors the idea used in the
    reference implementation.
    """
    candidate = base.copy()
    n = candidate.size
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    candidate[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        candidate = 0.5 * (candidate + candidate[::-1])
    return np.clip(candidate, 0.0, 1000.0)

# ----------------------------------------------------------------------
def mass_shift_step(seq: np.ndarray, step_frac: float, rng: np.random.Generator) -> np.ndarray:
    """
    One iteration of the gradient‑guided mass‑shift move.

    *seq* is a non‑negative vector with unit L1‑norm.
    *step_frac* is the fraction of the source entry that will be moved.
    The total sum is preserved.
    """
    n = seq.shape[0]
    conv = np.convolve(seq, seq)
    k = int(conv.argmax())                     # index of the current peak
    # contribution vector g_j = a_j + a_{k-j} when the partner exists
    g = seq.copy()
    low = max(0, k - (n - 1))
    high = min(n - 1, k)
    idx = np.arange(low, high + 1, dtype=int)
    if idx.size > 0:
        g[idx] += seq[k - idx]
    # pick source (largest contribution) and target (smallest contribution)
    i_src = int(np.argmax(g))
    i_tgt = int(np.argmin(g))
    if i_src == i_tgt or seq[i_src] <= 0.0:
        # fallback: pick two distinct random indices
        i_src, i_tgt = rng.choice(n, size=2, replace=False)
        if seq[i_src] <= 0.0:
            return seq
    delta = step_frac * seq[i_src]
    if delta <= 0.0:
        return seq
    new_seq = seq.copy()
    new_seq[i_src] -= delta
    new_seq[i_tgt] += delta
    # numerical safety
    new_seq = np.clip(new_seq, 0.0, 1000.0)
    return new_seq

    print(f"[done] final_score={final_score:.6f}")

    if i_src == i_tgt or seq[i_src] <= 0.0:
        i_src, i_tgt = rng.choice(n, size=2, replace=False)
        if seq[i_src] <= 0.0:
            return seq

    delta = step_frac * seq[i_src]
    if delta <= 0.0:
        return seq

    new_seq = seq.copy()
    new_seq[i_src] -= delta
    new_seq[i_tgt] += delta
    # ensure we stay inside the evaluator's limits
    return np.clip(new_seq, 0.0, 1000.0)

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000.0,
                     **kwargs):
    """
    Search for a low‑evaluation step‑function.

    The function is given roughly ``budget_s`` seconds; it stops a couple of
    seconds early to avoid a forced termination.  The returned value is a
    plain ``list[float]`` ready for ``evaluate_sequence``.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 1) initialise from the previously found sequence if it exists
    base_seq = None
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if (arr.ndim == 1 and arr.size > 0 and
                    np.isfinite(arr).all()):
                base_seq = arr.copy()
        except Exception:
            base_seq = None

    if base_seq is None:
        # fallback – a modest‑size uniform vector
        base_seq = np.ones(2048, dtype=np.float64)

    base_seq = np.clip(base_seq, 0.0, 1000.0)
    if base_seq.sum() == 0.0:
        base_seq = np.ones_like(base_seq) / base_seq.size
    else:
        base_seq = base_seq / base_seq.sum()          # unit L1‑norm

    n = base_seq.size
    best_seq = base_seq.copy()
    best_score = _metric(best_seq)
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) random analytic‑shape sweep
    #    a_i = (i+1+α)^{-p} * exp(-β·i)
    #    p∈[0.1,1.3] , α∈[0,30] , β∈[0,0.005]
    num_samples = 1500
    best_params = None
    for _ in range(num_samples):
        p = rng.uniform(0.10, 1.30)
        shift = rng.uniform(0.0, 30.0)
        beta = rng.uniform(0.0, 0.005)

        idx = np.arange(n, dtype=np.float64)
        seq = (idx + 1.0 + shift) ** (-p) * np.exp(-beta * idx)
        total = seq.sum()
        if total == 0.0:
            continue
        seq = seq / total
        score = _metric(seq)
        if score < best_score:
            best_score = score
            best_seq = seq
            best_params = (p, shift, beta)
            print(f"[sweep] new best {score:.6f} (p={p:.4f}, shift={shift:.2f},"
                  f" beta={beta:.6f})")

    # ------------------------------------------------------------------
    # 3) local refinement – gradient‑guided mass shifts + occasional
    #    random moves
    deadline = time.monotonic() + max(0.0, budget_s - 2.0)   # small safety margin
    step_frac = 0.08          # start moving 8 % of the source height
    step_decay = 0.9995       # slowly shrink the step size
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # ---- primary move: mass shift directed at the current convolution peak
        candidate = mass_shift_step(best_seq, step_frac, rng)
        cand_score = _metric(candidate)
        if cand_score < best_score:
            best_seq, best_score = candidate, cand_score
            print(f"[improve] iter {iteration} mass‑shift → {best_score:.6f}")
            step_frac = min(step_frac * 1.03, 0.30)   # gently increase step
            continue

        # ---- occasional block perturbation (adds stochastic exploration)
        if rng.random() < 0.07:
            candidate = make_block_perturbation(best_seq, rng)
            cand_score = _metric(candidate)
            if cand_score < best_score:
                best_seq, best_score = candidate, cand_score
                print(f"[improve] iter {iteration} block → {best_score:.6f}")
                continue

        # ---- occasional smoothing / jitter move
        if rng.random() < 0.04:
            candidate = 0.5 * (best_seq + best_seq[::-1])
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            candidate = np.clip(candidate * jitter, 0.0, 1000.0)
            cand_score = _metric(candidate)
            if cand_score < best_score:
                best_seq, best_score = candidate, cand_score
                print(f"[improve] iter {iteration} smooth → {best_score:.6f}")
                continue

        # decay the step size when nothing improves
        step_frac *= step_decay

    # ------------------------------------------------------------------
    # 4) final normalisation and sanity check
    if best_seq.sum() == 0.0:
        best_seq = np.ones_like(best_seq) / best_seq.size
    else:
        best_seq = best_seq / best_seq.sum()
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # optional verification with the official evaluator
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[done] final_score={final_score:.6f}")

    # convert to plain Python list as required by the harness
    return [float(v) for v in best_seq]
