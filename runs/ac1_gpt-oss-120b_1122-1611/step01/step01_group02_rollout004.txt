"""
Propose a candidate step‑function with a value *better* than π/2.
The algorithm is an implementation of the “LP‑mixing” scheme described in the
statement:

1.  Start from a cheap feasible sequence (uniform heights).
2.  Let M = max_k (f∗f)[k] be the current worst convolution value.
3.  Solve the linear programme

        maximise   Σ g_i
        subject to Σ_i f_i·g_{k‑i} ≤ M   (0 ≤ k < 2n‑1)
                   g_i ≥ 0

   The constraints are linear in the unknown g because f is fixed.
   The LP is solved with SciPy’s HiGHS interior‑point method.

4.  Scale the LP solution to the required L1‑norm
       Σ g_i = √(2 n)   (the “normalisation’’ used by the scoring routine).

5.  Blend the old and the new sequence:

        f ← (1‑t)·f + t·g      (t is a small step size)

   The convex combination guarantees that the new maximum convolution never
   exceeds M, so the score 2 n·max/(Σf)² is non‑increasing.

6.  Repeat steps 2‑5 while time remains, keeping the best sequence seen.
   Occasionally a cheap random block perturbation is applied to help escape
   plateaus.

The whole loop runs well inside the 1000 s budget for a moderate length
(e.g. n≈500–800).  The final sequence returned is a plain list of ``float`` and
conforms to all security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# SciPy is part of the permitted environment.  If it cannot be imported we
# fall back to a very small random search (the code will still be correct,
# just slower).
try:
    from scipy.optimize import linprog
    _HAS_LP = True
except Exception:                     # pragma: no cover
    _HAS_LP = False


def _build_constraint_matrix(f: np.ndarray) -> np.ndarray:
    """
    Return the matrix A (shape (2*n-1, n)) that implements the linear
    constraints  Σ_i f_i·g_{k-i} ≤ M .
    Row k corresponds to the convolution index k.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)

    # for each column i we insert the whole vector f into rows i … i+n-1
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _random_block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) mutation used occasionally to increase diversity.
    A random contiguous block is multiplied by a log‑normal factor.
    """
    cand = seq.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                      # occasional smoothing
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that minimises
    ``evaluate_sequence``.  The implementation follows the LP‑mixing scheme
    described above and falls back to a simple stochastic hill‑climber if SciPy
    is unavailable.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a working length.  If a previous construction is present we try
    # to beat it, otherwise we start with a modest size that keeps the LP
    # tractable.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 0:
                raise ValueError
        except Exception:
            n = 600
    else:
        n = 600                          # a size that is fast for many LP solves

    # ------------------------------------------------------------------
    # Initialise with a uniform sequence (any feasible point will do).
    # The normalisation is Σ a_i = √(2 n) – this is the convention used by the
    # scorer, so after the first LP step the current score equals the max
    # convolution.
    # ------------------------------------------------------------------
    target_sum = np.sqrt(2.0 * n)
    f = np.full(n, target_sum / n, dtype=np.float64)   # uniform start

    # ------------------------------------------------------------------
    # Statistics
    # ------------------------------------------------------------------
    best_seq = f.copy()
    try:
        best_score = float(evaluate_sequence(best_seq.tolist()))
    except Exception:                      # very unlikely
        best_score = float("inf")
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 3.0)
    iteration = 0
    step = 0.25                # convex‑combination weight (can be reduced later)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1) Current max convolution M
        # --------------------------------------------------------------
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # --------------------------------------------------------------
        # 2) Solve the linear programme to obtain a candidate g_0
        # --------------------------------------------------------------
        if _HAS_LP:
            A = _build_constraint_matrix(f)      # shape (2n-1, n)
            b = np.full(2 * n - 1, M, dtype=np.float64)

            # maximise Σ g_i  → minimise -Σ g_i
            c = -np.ones(n, dtype=np.float64)

            # bounds for every variable:  g_i ≥ 0
            bounds = [(0.0, None)] * n

            # HiGHS (default) is very fast for this size
            res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method="highs", options={"presolve": True})
            if not res.success:
                # If the LP failed (numerical pathologies) fall back to a
                # tiny random mutation and continue.
                cand = _random_block_perturb(f, rng)
                g0 = cand
            else:
                g0 = res.x
        else:
            # No LP available – use a random perturbation as a very weak
            # surrogate for the missing linear optimisation.
            g0 = _random_block_perturb(f, rng)

        # --------------------------------------------------------------
        # 3) Normalise g0 so that Σ g = √(2 n)
        # --------------------------------------------------------------
        sum_g0 = np.sum(g0)
        if sum_g0 == 0.0:
            # pathological case – just perturb the current f
            g = _random_block_perturb(f, rng)
        else:
            scale = target_sum / sum_g0
            g = g0 * scale

        # --------------------------------------------------------------
        # 4) Convex blend (mix) the old and new sequence.
        #    The step size is slowly reduced to aid convergence.
        # --------------------------------------------------------------
        f = (1.0 - step) * f + step * g

        # --------------------------------------------------------------
        # 5) Occasionally assess the true score and store the best.
        # --------------------------------------------------------------
        if iteration % 3 == 0:
            try:
                score = float(evaluate_sequence(f.tolist()))
            except Exception:
                score = float("inf")
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_seq = f.copy()
                print(f"[iter {iteration:04d}] improved score = {best_score:.6f}")
                if best_score < 1.5025:
                    # Goal reached – exit early.
                    break

        # --------------------------------------------------------------
        # 6) Small random jitter to avoid being stuck on an LP corner.
        # --------------------------------------------------------------
        if rng.random() < 0.12:
            f = _random_block_perturb(f, rng)

        # --------------------------------------------------------------
        # 7) Reduce the step size a little every few hundred iterations.
        # --------------------------------------------------------------
        if iteration % 200 == 0 and iteration > 0:
            step = max(0.03, step * 0.9)

        iteration += 1

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(f"[final] iterations={iteration}, best score={best_score:.6f}")
    # Convert to plain Python list as required by the harness.
    return [float(x) for x in best_seq]
