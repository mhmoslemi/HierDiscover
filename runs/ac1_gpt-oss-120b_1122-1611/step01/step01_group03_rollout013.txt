def project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def lp_optimize(f):
    n = len(f)
    M = max(convolve(f,f))
    # define variable g (nonneg)
    g = cp.Variable(n)
    constraints = [g >= 0]
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # sum over i: f_i * g_{k-i}
        expr = cp.sum(cp.multiply(f[i_start:i_end+1], g[k-i_end:k-i_start+1]))
        constraints.append(expr <= M)
    # maximize sum(g)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    return g.value

def project_simplex(v):
    # ensure v not all zero
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def gradient_logmax(a, beta):
    c = np.convolve(a,a)
    M = c.max()
    # compute softmax probabilities p
    # Subtract M for numeric stability
    exp_terms = np.exp(beta * (c - M))
    p = exp_terms / exp_terms.sum()
    # gradient = 2 * correlation of a and p
    corr = np.correlate(a, p, mode='full')
    offset = len(p) - 1
    grad = 2 * corr[offset : offset + len(a)]
    return grad, M, p

def solve_lp(f, M):
    import cvxpy as cp
    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]
    for k in range(2*n-1):
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        if i_start <= i_end:
            # slice of f
            f_slice = f[i_start:i_end+1]
            g_slice = g[k-i_end : k-i_start+1]
            constraints.append(cp.sum(cp.multiply(f_slice, g_slice)) <= M)
    # maximize sum(g)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    if g.value is None:
        return None
    return np.maximum(g.value,0)
```

"""
A combined stochastic / smooth‑gradient optimizer.
The algorithm starts from the classical “alpha‑sequence” (a_i = C(2i,i)/4^i)
and then iteratively improves it with a mixture of:

* Gaussian multiplicative noise,
* mass‑transfer moves,
* block‑scaling,
* a smooth‑gradient descent step on the log‑sum‑exp approximation of the
  max‑convolution,
* an occasional linear‑programming boost (if cvxpy is available).

All candidates are projected onto the probability simplex (non‑negative,
unit‑sum) before evaluation.  The best sequence found before the time
budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper routines (top level – no closures)
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def evaluate(a: np.ndarray) -> float:
    """
    Wrapper for the provided evaluate_sequence (expects a Python list).
    The sequence is already non‑negative and its sum is normalised to 1,
    therefore no further clipping is needed.
    """
    return float(evaluate_sequence(a.tolist()))


def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v>=0)."""
    if np.all(v == 0):
        # avoid division by zero – return uniform distribution
        return np.full_like(v, 1.0 / v.size)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety fallback
        theta = cssv[-1] / len(v)
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def gradient_logmax(a: np.ndarray, beta: float = 5.0):
    """
    Smooth‑gradient of the log‑sum‑exp approximation of max(convolution).
    Returns the gradient (same size as a), the current max value, and the
    probability vector used in the soft‑max.
    """
    c = np.convolve(a, a)
    m = c.max()
    # soft‑max over the 2n‑1 convolution entries (controlled by beta)
    exp_terms = np.exp(beta * (c - m))
    prob = exp_terms / exp_terms.sum()
    # gradient = 2 * (a ⋆ prob) where ⋆ is cross‑correlation
    corr = np.correlate(a, prob, mode='full')
    offset = len(prob) - 1
    grad = 2.0 * corr[offset:offset + len(a)]
    return grad, m, prob


# ----------------------------------------------------------------------
# Optional LP boost (requires cvxpy)
# ----------------------------------------------------------------------


def _solve_lp(f: np.ndarray, M: float):
    """Linear program: maximise sum(g) s.t. f * g ≤ M, g ≥ 0."""
    try:
        import cvxpy as cp
    except Exception:          # cvxpy not present – silently skip
        return None

    n = len(f)
    g = cp.Variable(n)
    constraints = [g >= 0]

    # build the convolution constraints f * g ≤ M
    for k in range(2 * n - 1):
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        # slices of f and g that contribute to the k‑th convolution entry
        f_slice = f[i_start:i_end + 1]
        g_slice = g[k - i_end : k - i_start + 1]
        constraints.append(cp.sum(cp.multiply(f_slice, g_slice)) <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # ECOS is a fast interior‑point solver for moderate‑size LPs
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    except Exception:
        return None
    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


# ----------------------------------------------------------------------
# Main entry point required by the test harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a coefficient sequence that minimises
    evaluate_sequence().  The routine runs until the time budget expires
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length and initialise from the global seed if present
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(height_sequence_1, (list, np.ndarray)):
        n = int(len(height_sequence_1))
    else:
        n = 3000                     # a comfortable default size

    # For very large n the search becomes slow; cap it to a reasonable size
    n = min(max(100, n), 3000)

    # ------------------------------------------------------------------
    # Base sequence – the classical alpha‑sequence, normalised to L1=1
    # ------------------------------------------------------------------
    a = generate_alpha_sequence(n)
    a = a / a.sum()

    best = a.copy()
    best_score = evaluate(best)
    current = a.copy()
    current_score = best_score

    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Hyper‑parameters
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    it = 0
    temperature = 0.02               # annealing temperature
    temp_decay = 0.99995
    block_min = max(1, n // 200)
    block_max = max(2, n // 12)

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # --------------------------------------------------------------
        # Propose a new candidate by one of several stochastic moves
        # --------------------------------------------------------------
        move = rng.random()
        if move < 0.45:
            # Gaussian multiplicative noise (global perturbation)
            cand = current * (1.0 + 0.20 * rng.standard_normal(n))
        elif move < 0.75:
            # Mass transfer between two random positions
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                delta = current[i] * rng.uniform(0.0, 0.5)
                cand = current.copy()
                cand[i] -= delta
                cand[j] += delta
            else:
                continue
        elif move < 0.90:
            # Block scaling – multiplicative factor on a contiguous sub‑array
            width = int(rng.integers(block_min, block_max + 1))
            start = int(rng.integers(0, n - width + 1))
            factor = math.exp(rng.normal(0.0, 0.07))
            cand = current.copy()
            cand[start:start + width] *= factor
        else:
            # Smooth‑gradient descent step aiming at the max‑convolution
            grad, _, _ = gradient_logmax(current, beta=8.0)
            step = 0.03 / (np.linalg.norm(grad) + 1e-12)
            cand = current - step * grad

        # --------------------------------------------------------------
        # Projection onto the probability simplex (non‑negative, sum = 1)
        # --------------------------------------------------------------
        cand = project_simplex(cand)

        # --------------------------------------------------------------
        # Evaluate the candidate
        # --------------------------------------------------------------
        cand_score = evaluate(cand)

        # --------------------------------------------------------------
        # Acceptance criterion – Metropolis style with decreasing temperature
        # --------------------------------------------------------------
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            prob = math.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            current = cand
            current_score = cand_score
            if cand_score < best_score:
                best = cand.copy()
                best_score = cand_score
                print(f"[{it:6d}] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Periodic LP boost (if cvxpy is available)
        # --------------------------------------------------------------
        if it % 5000 == 0 and best_score > 1.4:   # extra boost only while we are far
            # Solve the LP that maximises the L1‑norm under the current max bound
            M_current = np.convolve(current, current).max()
            g = _solve_lp(current, M_current)
            if g is not None and np.isfinite(g).all():
                # Normalise the LP solution and mix it with the current point
                g = g / g.sum()
                mix = 0.5
                cand = (1.0 - mix) * current + mix * g
                cand = project_simplex(cand)
                cand_score = evaluate(cand)
                if cand_score < current_score:
                    current = cand
                    current_score = cand_score
                    if cand_score < best_score:
                        best = cand.copy()
                        best_score = cand_score
                        print(f"[{it:6d}] LP‑mix improved best to {best_score:.6f}")

        # --------------------------------------------------------------
        # Temperature schedule and occasional re‑initialisation
        # --------------------------------------------------------------
        temperature *= temp_decay
        if it % 2000 == 0:
            # pull the search back to the best point seen so far
            current = best.copy()
            current_score = best_score

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    # Return a plain Python list as required by the harness
    return best.tolist()
