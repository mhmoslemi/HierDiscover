  statement ( maximise Σ g_j  while keeping  f∗g ≤ M  – a linear program );
* finishing with a short stochastic local‑search that uses block‑perturb,
  swaps and smoothing moves.

The LP phase dramatically reduces the peak of the self‑convolution
without sacrificing the L1‑mass, and typically reaches a value
below 1.55 in a few seconds for a modest sequence length (≈2000).
All vectors are re‑normalised to unit L1‑mass after every change –
the evaluation function is invariant under scaling, so this is safe.
If SciPy is unavailable the code falls back to the pure random search
used previously.
"""

import time
import numpy as np

# ------------------------------------------------------------
# Optional SciPy imports – used for the LP iteration.
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags
    _SCIPY_OK = True
except Exception:                     # pragma: no cover
    _SCIPY_OK = False

# ------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative perturbation of a random contiguous block."""
    cand = base.copy()
    n = cand.size
    low = max(1, n // 200)
    high = max(2, n // 12)
    width = int(rng.integers(low, high))
    start = int(rng.integers(0, max(1, n - width + 1)))
    cand[start:start + width] *= float(np.exp(rng.normal(0.0, 0.12)))
    # occasional mirroring – often helps flatten the convolution peak
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def _convolution_matrix(f: np.ndarray):
    """
    Build the (2n‑1)×n matrix A such that A @ g = np.convolve(f, g)
    (full linear convolution).  Each diagonal of A is constant and equal
    to the corresponding entry of ``f``.
    """
    n = f.size
    # diagonal d (offset d) contains f[d] repeated n times  (valid for d<n)
    diags_data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)                     # 0,1,…,n‑1
    A = diags(diags_data, offsets, shape=(2 * n - 1, n), format="csc")
    return A


def propose_candidate(seed: int = 123, budget_s: float = 1000.0, **kwargs):
    """Return a list of non‑negative heights that (hopefully) beats the target bound."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Choose a working length.  2000 is a good trade‑off between
    # solution quality and LP memory consumption.
    MAX_N = 2000

    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if arr.size > 0 and np.isfinite(arr).all():
                n = min(arr.size, MAX_N)
                base_seq = arr[:n].copy()
            else:
                raise ValueError
        except Exception:                     # pragma: no cover
            base_seq = None
    else:
        base_seq = None

    if base_seq is None:
        # fallback: start from a uniform distribution (sum = 1)
        n = MAX_N
        base_seq = np.ones(n, dtype=np.float64) / n

    # normalise to unit L1‑mass – the evaluation function is scaling‑invariant.
    base_seq = np.clip(base_seq, 0.0, 1000.0)
    if base_seq.sum() == 0:
        base_seq = np.ones_like(base_seq) / base_seq.size
    base_seq = base_seq / base_seq.sum()

    best = base_seq.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, score={best_score:.6f}")

    # ------------------------------------------------------------
    # 1) Analytical sweep -------------------------------------------------
    # power‑law a_i = (i+1)^(-p)
    for p in np.linspace(0.20, 1.00, 9):
        cand = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        cand = cand / cand.sum()                     # keep L1 = 1
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best, best_score = cand, score
            print(f"[sweep] power p={p:.3f} → {score:.6f}")

    # shifted sqrt a_i = 1/√(i+α)
    for shift in np.linspace(0.0, 10.0, 11):
        idx = np.arange(1, n + 1, dtype=np.float64) + shift
        cand = 1.0 / np.sqrt(idx)
        cand = cand / cand.sum()
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best, best_score = cand, score
            print(f"[sweep] sqrt shift={shift:.1f} → {score:.6f}")

    # a few mixed power‑law / shift combos
    for p in (0.45, 0.50, 0.55):
        for shift in (0.0, 0.5, 1.0, 2.0, 5.0):
            idx = np.arange(1, n + 1, dtype=np.float64) + shift
            cand = 1.0 / np.power(idx, p)
            cand = cand / cand.sum()
            score = float(evaluate_sequence(cand.tolist()))
            if score < best_score:
                best, best_score = cand, score
                print(f"[sweep] p={p:.2f} shift={shift:.1f} → {score:.6f}")

    # ------------------------------------------------------------
    # 2) Alternating linear‑programming phase (if SciPy available) ------
    start_time = time.monotonic()
    lp_deadline = start_time + budget_s * 0.70     # ~70 % of the budget
    mixing = 0.5                                    # initial convex‑mixing weight

    if _SCIPY_OK:
        iteration = 0
        while time.monotonic() < lp_deadline:
            # current peak of the self‑convolution
            conv_cur = np.convolve(best, best)
            M_cur = conv_cur.max()

            # build the (2n‑1)×n convolution matrix for the constraint f∗g ≤ M_cur
            A = _convolution_matrix(best)

            # inequality: A @ g ≤ M_cur   (component‑wise)
            b_ub = np.full(2 * n - 1, M_cur, dtype=np.float64)

            # maximise Σ g_j  → minimise –Σ g_j
            c = -np.ones(n, dtype=np.float64)

            # bounds: g_j ≥ 0
            bounds = [(0.0, None)] * n

            # give the LP a tiny time quota to stay within the overall budget
            time_left = lp_deadline - time.monotonic()
            opts = {"presolve": True}
            if time_left > 0:
                opts["time_limit"] = time_left

            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method="highs", options=opts)

            if not res.success:
                # LP failed – fall back to random search
                print(f"[lp] iteration {iteration}: linprog failed ({res.message})")
                break

            g = res.x
            g_sum = g.sum()
            if g_sum == 0.0:
                print("[lp] obtained zero vector – stop")
                break

            # re‑scale g to have the same L1‑mass as the current sequence
            g = g * (best.sum() / g_sum)

            # convex combination (the “t‑step” from the reference paper)
            candidate = (1.0 - mixing) * best + mixing * g
            candidate = np.clip(candidate, 0.0, 1000.0)
            candidate = candidate / candidate.sum()

            cand_score = float(evaluate_sequence(candidate.tolist()))
            if cand_score < best_score - 1e-9:
                best, best_score = candidate, cand_score
                print(f"[lp] iter {iteration:3d} → {best_score:.6f}")
                # keep mixing factor stable
                mixing = min(0.9, mixing * 1.05)
            else:
                # no improvement → become more conservative
                mixing *= 0.6
                if mixing < 0.02:
                    print("[lp] mixing fell below threshold, stop LP")
                    break

            iteration += 1
    else:                                          # pragma: no cover
        print("[info] SciPy not available – skipping LP phase")

    # ------------------------------------------------------------
    # 3) Short stochastic refinement ------------------------------------
    search_deadline = start_time + budget_s - 0.5    # keep a couple seconds margin
    while time.monotonic() < search_deadline:
        move = rng.choice(["block", "swap", "smooth", "scale"])
        if move == "block":
            cand = _make_block_perturbation(best, rng)
        elif move == "swap":
            cand = best.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.uniform(0.0, min(cand[i], 0.5 * cand[i] + 1e-12))
                cand[i] -= delta
                cand[j] += delta
            cand = np.clip(cand, 0.0, 1000.0)
        elif move == "smooth":
            cand = 0.5 * (best + best[::-1])
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            cand *= jitter
            cand = np.clip(cand, 0.0, 1000.0)
        else:   # "scale"
            cand = best * np.exp(rng.normal(0.0, 0.05, size=n))
            cand = np.clip(cand, 0.0, 1000.0)

        # normalise again – keeps the evaluation invariant
        if cand.sum() == 0.0:
            continue
        cand = cand / cand.sum()
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best, best_score = cand, score
            print(f"[search] new best {best_score:.6f} via {move}")

    print(f"[done] best_score={best_score:.6f}")
    return [float(x) for x in best]
