"""
propose_candidate
-----------------

Iteratively improve a Catalan‐type step function by solving a
linear program that “fills” the slack in the self‑convolution
bound.  Starting from the optimal (for the exact infinite problem)
sequence a_i = C(2i,i)/4**i (normalised to Σa = √(2n)),
the LP finds a non‑negative vector g with the same length that
maximises Σg subject to (a * g)[k] ≤ M where M = max(a * a).
After rescaling g to the target L1‑norm we mix it with a:
    a ← (1‑t)·a + t·g .
Repeating this step drives the supremum of the self‑convolution
downwards, which directly lowers the evaluation function

    evaluate_sequence(a) = 2·n·max(a*a) / (Σa)² .

The routine stops when the time budget (≈1000 s) is exhausted
or when no improvement is seen for several iterations, and
returns the best sequence found.  If the optional ``cvxpy`` library
is unavailable the code falls back to a simple stochastic
perturbation search (the method used previously).

The implementation uses only NumPy and (optionally) CVXPY and
conforms to all required constraints of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    First ``length`` terms of a_i = binom(2*i,i) / 4**i.
    The recurrence a_{i+1} = a_i * (2*i+1)/(2*i+2) is used.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale ``seq`` so that Σseq = ``target_sum``.
    """
    cur = np.sum(seq)
    if cur == 0:
        return seq
    factor = target_sum / cur
    return seq * factor


def build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the dense matrix A (size (2n‑1)×n) such that
    A @ g == (f * g) for any vector g of length n.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f[i]
    return A


def linear_program_step(f: np.ndarray, M: float, target_sum: float):
    """
    Solve  max Σg   s.t.   g ≥ 0 , (f * g) ≤ M .
    Returns the scaled vector g (so that Σg = target_sum) or None
    if the solver fails.
    """
    try:
        import cvxpy as cp
    except Exception:          # cvxpy not available – signal caller to skip LP
        return None

    n = f.size
    A = build_conv_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    # solve – a few seconds for n≈500
    prob.solve(solver=cp.ECOS, warm_start=True,
               abstol=1e-8, reltol=1e-8, max_iters=5000)

    if g.value is None:
        return None

    g_val = np.asarray(g.value).ravel()
    sum_g = np.sum(g_val)
    if sum_g <= 0:

    # rescale to the required L1‑norm
    g_scaled = g_val * (target_sum / sum_g)
    return g_scaled

            start_idx = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.12)))
            cand[start_idx:start_idx + width] *= factor
            if rng.random() < 0.35:
                cand = 0.5 * (cand + cand[::-1])
            np.clip(cand, 0.0, 1000.0, out=cand)
        else:
            # mix the new vector into the current candidate
            cand = (1.0 - t_mix) * best_seq + t_mix * g_scaled

        # evaluate the candidate
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if np.isfinite(score) and score < best_score - 1e-7:
            best_seq = cand
            best_score = score
            no_improve = 0
            last_improve = time.monotonic()
            print(f"[improve] iter={iteration} new best score={best_score:.6f}")
        else:
            no_improve += 1

        # occasional status report
        if iteration % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"[status] iter={iteration} best={best_score:.6f} "
                  f"elapsed={elapsed:.1f}s no_improve={no_improve}")

        # stop early if we have stalled for a while
        if no_improve >= 200 and (time.monotonic() - last_improve) > 30.0:
            print("[stop] No improvement for a long time – terminating.")
            break

    print(f"[final] returning sequence length {len(best_seq)} with score {best_score:.6f}")
    # Convert to plain Python list as required by the harness
    return [float(x) for x in best_seq]
