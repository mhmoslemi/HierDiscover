"""
Propose a step‑function (list of non‑negative heights) that minimises the
security‑evaluation function.

Algorithm
---------

1. **Broad parameter sweep** – generate a handful of analytic families
   (power‑law, shifted‑sqrt, exponential, centred Gaussian) for several
   sequence lengths (including the length of the known good sequence).  The
   best‑scoring sequence from this cheap sweep seeds the optimisation.

2. **Simulated‑annealing local search** – starting from the best seed,
   repeatedly apply inexpensive stochastic moves:
      * block‑multiplicative perturbation,
      * random mass‑swap,
      * gentle smoothing,
      * a directed “gradient‑swap’’ that moves mass away from indices that
        contribute most to the current convolution peak.
   Moves that improve the score are always accepted; worse moves are accepted
   with a temperature‑dependent probability that decays exponentially.
   The current best and the overall best sequence are tracked.

3. The routine runs for the allotted wall‑clock budget (≈ 990 s, leaving a
   small safety margin) and finally returns the best sequence found as a
   plain ``list[float]`` – the format expected by ``evaluate_sequence``.
"""

# Helper: evaluate the external objective (already imported in the harness)
def _eval(seq: np.ndarray) -> float:
    """Wrapper around the provided ``evaluate_sequence``."""
    # ``evaluate_sequence`` expects a plain list of floats.
    return float(evaluate_sequence(seq.tolist()))

# Move: multiplicative perturbation of a random contiguous block
def block_perturb(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Scale a random block of ``seq`` by a log‑normal factor."""
    n = seq.size
    # block width between 0.5 % and 20 % of the length
    min_w = max(1, int(n * 0.005))
    max_w = max(2, int(n * 0.20))
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # moderate log‑normal noise
    cand = seq.copy()
    cand[start:start + width] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Move: tiny random mass transfer between two positions
def random_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a random amount of mass from i to j."""
    n = seq.size
    i, j = rng.integers(0, n, size=2)
    if i == j:
        return seq.copy()
    cand = seq.copy()
    # transfer at most 50 % of the source height
    delta = rng.uniform(0.0, 0.5 * cand[i])
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Move: gentle low‑pass smoothing (preserves total mass)
def smooth_sequence(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Replace ``seq`` by a 3‑point moving‑average and add tiny jitter."""
    kernel = np.array([0.25, 0.5, 0.25])
    sm = np.convolve(seq, kernel, mode='same')
    # renormalise to keep the ℓ₁‑norm unchanged
    total = float(seq.sum())
    if sm.sum() > 0:
        sm *= total / sm.sum()
    # small multiplicative jitter (keeps shape but provides stochasticity)
    jitter = np.exp(rng.normal(0.0, 0.02, size=sm.shape))
    cand = sm * jitter
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


# ----------------------------------------------------------------------
# Move: gradient‑guided swap that reduces the current convolution peak
def gradient_swap(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Identify the index ``k`` where the convolution is maximal.
    Pick a source position proportional to its contribution a_i·a_{k‑i},
    move a fraction of its mass to a random target.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    # indices that actually take part in the peak
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=np.int64)
    contrib = seq[idx] * seq[k - idx]

    if contrib.sum() == 0:
        return seq.copy()

    # source biased by contribution size
    probs = contrib / contrib.sum()
    src = int(rng.choice(idx, p=probs))
    # a random destination (may be anywhere)
    dst = int(rng.integers(0, n))
    while dst == src:
        dst = int(rng.integers(0, n))

    # move a modest fraction of the source mass
    delta = rng.uniform(0.0, 0.5 * seq[src])
    cand = seq.copy()
    cand[src] -= delta
    cand[dst] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

# Small library of analytic families (used in the initial sweep)
def make_powerlaw(p: float, n: int) -> np.ndarray:
    """a_i = (i+1)^{-p}."""
    a = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def make_shifted_sqrt(alpha: float, n: int) -> np.ndarray:
    """a_i = 1 / sqrt(i+1+alpha)."""
    i = np.arange(1, n + 1, dtype=np.float64) + alpha
    a = 1.0 / np.sqrt(i)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def make_exponential(tau: float, n: int) -> np.ndarray:
    """a_i = exp(-i/tau)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / tau)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def make_gaussian(sigma: float, n: int) -> np.ndarray:
    """a_i = exp(-0.5 * ((i - µ)/σ)^2) with µ at the centre."""
    i = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    np.clip(a, 0.0, 1000.0, out=a)
    return a

def propose_candidate(seed: int = 42, budget_s: float = 990, **kwargs):

    Search for a low‑evaluation step function within the allocated time budget.

    Args:
        seed:    Random seed for reproducibility.
        budget_s: Approximate time budget in seconds (a few seconds are kept
                  as a safety margin).
        **kwargs: Ignored – present for compatibility with the harness.

    Returns:
        A list of floats – the best height sequence discovered.
    """

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety gap

    # ------------------------------------------------------------------
    # Determine a set of lengths to explore.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    # If the benchmark already supplied a candidate, include its size.
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                lengths.append(int(arr.size))
        except Exception:
            pass

    lengths = sorted(set(lengths))

    # ------------------------------------------------------------------
    # 1) BROAD PARAMETER SWEEP
    best_seq = None
    best_score = float('inf')

    for n in lengths:
        # uniform baseline
        cand = np.ones(n, dtype=np.float64)
        score = _eval(cand)
        if score < best_score:
            best_seq, best_score = cand, score
            print(f"[sweep] uniform n={n} → {score:.6f}")

        # power‑law family
        for p in np.linspace(0.20, 1.20, 11):
            cand = make_powerlaw(p, n)
            score = _eval(cand)
            if score < best_score:
                best_seq, best_score = cand, score
                print(f"[sweep] power p={p:.3f} n={n} → {score:.6f}")

        # shifted √ family
        for alpha in np.linspace(0.0, 10.0, 11):
            cand = make_shifted_sqrt(alpha, n)
            score = _eval(cand)
            if score < best_score:
                best_seq, best_score = cand, score
                print(f"[sweep] sqrt α={alpha:.2f} n={n} → {score:.6f}")

        # exponential decay
        for tau in np.linspace(0.05 * n, 0.30 * n, 7):
            cand = make_exponential(tau, n)
            score = _eval(cand)
            if score < best_score:
                best_seq, best_score = cand, score
                print(f"[sweep] exp τ={tau:.0f} n={n} → {score:.6f}")

        # centred Gaussian
        for sigma in (n / 20.0, n / 10.0, n / 5.0):
            cand = make_gaussian(sigma, n)
            score = _eval(cand)
            if score < best_score:
                best_seq, best_score = cand, score
                print(f"[sweep] gauss σ={sigma:.0f} n={n} → {score:.6f}")

    # ------------------------------------------------------------------
    # Initialise the local optimiser.
    curr_seq = best_seq.copy()
    curr_score = best_score
    global_best_seq = best_seq.copy()
    global_best_score = best_score

    # Simulated‑annealing schedule
    T0 = 0.02                # initial temperature
    Tf = 1e-6                # final temperature
    log_ratio = math.log(Tf / T0) if T0 > 0 else -float('inf')
    max_time = budget_s

    # ------------------------------------------------------------------
    # 2) LOCAL STOCHASTIC SEARCH (Simulated Annealing)
    move_list = ["block", "swap", "smooth", "grad_swap"]
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        elapsed = time.monotonic() - start_time
        t_frac = elapsed / max_time
        T = T0 * math.exp(log_ratio * t_frac)  # exponential cooling

        move = rng.choice(move_list)

        if move == "block":
            cand = block_perturb(curr_seq, rng)
        elif move == "swap":
            cand = random_swap(curr_seq, rng)
        elif move == "smooth":
            cand = smooth_sequence(curr_seq, rng)
        else:  # "grad_swap"
            cand = gradient_swap(curr_seq, rng)

        cand_score = _eval(cand)
        if not np.isfinite(cand_score):
            continue

        delta = cand_score - curr_score
        accept = False
        if delta < 0:
            accept = True
        else:
            # worse move – accept probabilistically
            if T > 0 and rng.random() < math.exp(-delta / T):
                accept = True

        if accept:
            curr_seq = cand
            curr_score = cand_score
            if cand_score < global_best_score:
                global_best_seq = cand.copy()
                global_best_score = cand_score
                print(f"[best] new global best {cand_score:.6f} (move={move}, it={iteration})")

        # occasional “big jitter’’ to escape deep traps
        if rng.random() < 0.001:
            jittered = global_best_seq * np.exp(rng.normal(0.0, 0.03, size=global_best_seq.shape))
            np.clip(jittered, 0.0, 1000.0, out=jittered)
            curr_seq = jittered
            curr_score = _eval(jittered)

    # ------------------------------------------------------------------
    print(f"[done] best score after {iteration} iterations = {global_best_score:.6f}")
    # Return a plain Python list as required by the evaluation harness.
    return [float(v) for v in global_best_seq]
