"""
Propose a step‑function that minimises

    score = 2 * n * max(a * a) / (sum(a) ** 2)

for a non‑negative vector a (the sum is forced to 1, so the score is
2·n·max(conv) where conv = a⋆a).  The algorithm works on the simplex
using a “soft‑max” approximation of the max and Adam updates on the
log‑its of the probabilities (mirror‑descent on the simplex).  The
temperature of the soft‑max is slowly increased so that the optimisation
first smooths the whole convolution landscape and then focuses on the
largest peak.  Occasional random perturbations of the logits are added
to escape shallow local minima.

The routine runs for roughly ``budget_s`` seconds (leaves a 2 s safety
margin) and returns the best vector found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _fft_len_for_len(length: int) -> int:
    """
    Return the smallest power‑of‑two length ≥ ``length``.
    Used for zero‑padded FFTs.
    """
    return 1 << ((length - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution ``a * a`` using an FFT of length ``fft_len``.
    Result length is ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _conv_full(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two arbitrary real vectors ``x`` and ``y``
    using an FFT of suitable length.
    """
    length = x.size + y.size - 1
    fft_len = _fft_len_for_len(length)
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    conv = np.fft.irfft(fx * fy, n=fft_len)
    return conv[:length]


# ----------------------------------------------------------------------
# Core optimiser
# ----------------------------------------------------------------------

    Return a list of non‑negative heights that (approximately) minimise the
    provided ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a small safety margin) and uses an

    # a reasonably large size (≈ 5000) that still fits comfortably in the

    # Uniform distribution corresponds to `z = 0`.
    # ------------------------------------------------------------------
    z = np.zeros(n, dtype=np.float64)          # logits
    a = np.full(n, 1.0 / n, dtype=np.float64)  # current probability vector

    # Adam moments
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.5                                   # base step size for logits

    # Temperature for the soft‑max that approximates the max‑operator.
    # Starts gentle (smooth landscape) and is annealed upwards.
    temp = 0.5
    temp_max = 12.0
    temp_factor = 1.001

    # imposes – this is just a safety net).
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a.tolist()]
