def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, math

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # get n
    try:
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
        n = base_seq.shape[0]
    except Exception:
        n = 4806
        base_seq = np.ones(n, dtype=np.float64)

    target_sum = math.sqrt(2.0 * n)

    # define fast scoring
    def fast_score(a):
        # Assume a is 1D numpy float array
        L = 2 * n - 1
        A = np.fft.fft(a, L)
        conv = np.fft.ifft(A * A).real
        max_b = conv.max()
        sum_a = a.sum()
        return float(2.0 * n * max_b / (sum_a * sum_a))

    # Normalize base seq to target sum and clip
    base_seq = np.clip(base_seq, 0.0, 1000.0)
    if base_seq.sum() == 0:
        base_seq = np.ones(n)
    base_seq = base_seq * (target_sum / base_seq.sum())
    best_seq = base_seq.copy()
    best_score = fast_score(best_seq)
    print(f"Initial best score (fast) {best_score:.6f}")

    # Helper to evaluate with external function for final verification
    def external_score(seq):
        return float(evaluate_sequence(seq.tolist()))

    # Update function
    def maybe_update(candidate):
        nonlocal best_seq, best_score
        sc = fast_score(candidate)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = candidate.copy()
            print(f"New best score {best_score:.6f}")

    # Phase 1: parametric sweep (more thorough)
    idx = np.arange(n, dtype=np.float64)
    N = n - 1

    # Exponential family (with background)
    for c in np.linspace(0.2, 5.0, 60):
        r = math.exp(-c / N)
        seq = r ** idx
        # renorm
        seq = seq * (target_sum / seq.sum())
        maybe_update(seq)
        # add backgrounds
        for eps in (0.0, 0.01, 0.02, 0.05):
            cand = seq + eps
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand * (target_sum / cand.sum())
            maybe_update(cand)

    # Power-exponential combos
    alphas = np.linspace(-1.5, 1.5, 13)
    for alpha in alphas:
        for c in np.linspace(0.5, 4.0, 30):
            r = math.exp(-c / N)
            seq = (idx + 1.0) ** alpha * (r ** idx)
            seq = np.clip(seq, 0.0, None)
            if seq.max() == 0:
                continue
            seq = seq * (target_sum / seq.sum())
            maybe_update(seq)

    # Stretched exponential (Gaussian-like)
    for p in (0.5, 1.0, 2.0, 3.0):
        for c in np.linspace(0.5, 4.0, 30):
            # a_i = exp(-c * (i/N)**p)
            seq = np.exp(-c * (idx / N) ** p)
            seq = np.clip(seq, 0.0, None)
            seq = seq * (target_sum / seq.sum())
            maybe_update(seq)

    # Inverse power law
    for beta in np.linspace(0.3, 2.5, 25):
        seq = 1.0 / ((idx + 1.0) ** beta)
        seq = seq * (target_sum / seq.sum())
        maybe_update(seq)

    # Phase 2: Simulated annealing / gradient descent
    current_seq = best_seq.copy()
    current_score = best_score

    # parameters for annealing
    total_iters = 500000  # may adjust based on time
    iter = 0
    # Base sigmas
    base_block_sigma = 0.15   # lognormal factor sigma
    base_transfer_sigma = 0.02 # fraction of source value
    base_global_sigma = 0.02

    while time.monotonic() < deadline:
        # schedule temperature linearly decreasing
        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)
        T = max(0.0, 1.0 - frac)  # simple linear cooling

        # choose move type probabilistically
        rtype = rng.random()
        cand = current_seq.copy()
        if rtype < 0.5:
            # block scaling move
            # block size with dependence on temperature
            min_width = max(1, n // 500)
            max_width = max(2, n // 10)
            width = int(rng.integers(min_width, max_width))
            start = int(rng.integers(0, n - width + 1))
            stop = start + width
            log_factor = rng.normal(0.0, base_block_sigma * T + 0.001)  # avoid zero sigma later
            factor = float(math.exp(log_factor))
            cand[start:stop] *= factor
        elif rtype < 0.8:
            # transfer move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:

            # ensure nonnegative
            if delta > src:

            cand[i] = max(0.0, src - delta)
            cand[j] = cand[j] + delta
        else:

            log_noise = rng.normal(0.0, base_global_sigma * T, size=n)
            factor_arr = np.exp(log_noise)
            cand *= factor_arr

        # clip and renormalize sum
        np.clip(cand, 0.0, 1000.0, out=cand)
        sum_c = cand.sum()
        if sum_c == 0:

        cand = cand * (target_sum / sum_c)

        cand_score = fast_score(cand)

            current_seq = cand
            current_score = cand_score

        if cand_score < best_score:
            best_seq = cand.copy()
            best_score = cand_score
            print(f"New best score {best_score:.6f} at iteration {iter}")

        iter += 1

    # Final re-evaluation with external function to be safe

    final_score = external_score(final_seq)
    print(f"Search completed – best score {best_score:.6f} (external {final_score:.6f})")
    return final_seq.tolist()

"""
Propose a candidate step function with a low evaluation score.

Algorithm
---------
1. Initialise from the provided `height_sequence_1` (or a uniform fallback) and
   normalise the total sum to sqrt(2 n).  The score does not depend on the
   absolute scale, but fixing the sum simplifies the updates.

2. **Parametric sweep** – evaluate a broad family of analytic shapes:
   exponential, power‑exponential, stretched‑exponential (Gaussian‑like) and
   pure inverse‑power.  For each shape the sequence is normalised and the
   objective 2·n·max(conv)/sum² is computed with a fast FFT‑based convolution.
   The best candidate becomes the current reference.

3. **Simulated annealing** – a temperature‑controlled stochastic search that
   repeatedly perturbs the current sequence:
   * multiplicative log‑normal factor on a random contiguous block,
   * transfer of mass between two random positions,
   * global log‑normal jitter.
   After each move the sequence is clipped to [0, 1000] and re‑normalised.
   The move is accepted if it improves the score or with a Metropolis
   probability that decays with the temperature.  The temperature falls linearly
   to zero over the remaining time budget.

4. **Gradient‑descent fine‑tuning** – after the annealing phase a few
   deterministic steps are taken.  The gradient of the objective with respect
   to the entries (for the currently active maximal convolution index) is
   computed analytically, a small step is applied, the result is clipped,
   re‑normalised and kept only if it improves the score.

5. The best sequence found is returned as a plain Python list.  All helper
   functions are defined at module level to avoid closures.
"""

    # Final verification with the official evaluate_sequence()
    # --------------------------------------------------------------
    final_seq = best_seq.clip(0.0, 1000.0)
    final_score = _external_score(final_seq)
    print(
        f"Search completed – best score (fast) {best_score:.6f}, "
        f"external {final_score:.6f}"
    )
    return final_seq.tolist()
