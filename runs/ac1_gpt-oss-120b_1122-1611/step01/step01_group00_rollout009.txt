"""
Propose a low‑score step‑function by minimizing a smooth approximation of the
maximal convolution entry.  The algorithm runs a projected Adam optimizer on
the loss

    L(a) = (1/α)·log Σ_k exp(α·(a*a)_k)

which is a soft‑max of the auto‑convolution.  Starting from a random Dirichlet
distribution we anneal the temperature α so the loss gradually turns into the
true max.  After each Adam update the iterate is projected onto the probability
simplex {a≥0, Σa=1}.  The best sequence seen so far is kept and returned.
All heavy work (convolutions) is done with FFTs, permitting thousands of
iterations on vectors of length up to a few‑thousand within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x≥0, Σx = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _next_pow2(x: int) -> int:
    """Small utility returning the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()

        # 6) evaluate candidate (fast internal formula)
        Fa_cand = np.fft.rfft(a_candidate, n=L_fft)
        conv_cand = np.fft.irfft(Fa_cand * Fa_cand, n=L_fft)[: 2 * n - 1]
        score = 2.0 * n * np.max(conv_cand)   # sum(a)=1 ⇒ denominator = 1

        # keep the best ever
        if score < best_score:
            best = a_candidate.copy()
            best_score = score
            last_improve = iteration
            print(f"[improve] iter={iteration}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 7) prepare next iterate
        a = a_candidate

        # --------------------------------------------------------------
        # 8) temperature annealing (slow increase towards α_target)
        if alpha < alpha_target:
            alpha *= alpha_mul

        # --------------------------------------------------------------
        # 9) early stop if we are already within 1 % of the theoretical bound
        theoretical_lb = 2.0 * n / (2.0 * n - 1.0)
        if best_score <= theoretical_lb * 1.01:
            print("[early stop] near‑optimal bound reached")
            break

        # --------------------------------------------------------------
        # 10) restart if stagnation persists for a long time
        if iteration - last_improve > RESTART_AFTER:
            # perturb the current best and continue
            a = best.copy()
            # throw a small random reshuffle
            perm = rng.permutation(n)
            a = a[perm]
            a = _project_simplex(a + 1e-4 * rng.random(n))
            # reset Adam moments to avoid carrying stale gradients
            m.fill(0.0)
            v.fill(0.0)
            last_improve = iteration
            print(f"[restart] iter={iteration}, best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # final sanity check with the official evaluator (optional but cheap)
    try:
        # protect against any accidental overflow – clip just in case
        final_seq = [float(x) for x in best]
        # evaluate_sequence is expected to be present in the environment
        # (the scorer will call it anyway).  We call it here only for logging.
        from __main__ import evaluate_sequence  # type: ignore
        final_score = evaluate_sequence(final_seq)
        print(f"[final] best score (official) = {final_score:.6f}")
    except Exception:
        # If evaluate_sequence is not importable, fall back to our internal estimate.
        print(f"[final] best score (internal) = {best_score:.6f}")

    return [float(x) for x in best]
