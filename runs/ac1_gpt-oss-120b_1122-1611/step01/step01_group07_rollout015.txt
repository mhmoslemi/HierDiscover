   “inverse‑square‑root’’ shape   a[i] = 1/√(i+1)   (its maximum is 1).

2. **Global perturbations** – block‑wise multiplicative noise (as in the
   reference code) to explore the neighbourhood of the current best
   sequence.

3. **Permutation moves** – random swaps of two positions.  Swapping does
   not change the sum but reshuffles the way large values overlap with
   each other in the convolution, often lowering the peak.

All moves are accepted if they improve the score; otherwise a simple
simulated‑annealing acceptance rule with a linearly decreasing temperature
is used, allowing occasional uphill steps.  The search runs until the
time budget (≈ 1000 s) is exhausted and returns the best sequence found.

The implementation is completely self‑contained, uses only NumPy and
the provided ``evaluate_sequence`` function, and prints progress
information.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper wrappers
# ----------------------------------------------------------------------
def _eval(seq):
    """Wrapper for the external ``evaluate_sequence`` – works on lists."""
    # ``evaluate_sequence`` already clips negatives and values >1000.
    return float(evaluate_sequence(seq.tolist()))


def _make_convolution(seq):
    """Return the full discrete convolution of ``seq`` with itself."""
    return np.convolve(seq, seq)


def _make_initial_inverse_sqrt(n):
    """
    Deterministic start: a[i] = 1 / sqrt(i+1)  (max = 1).
    The sum is ≈ 2·sqrt(n), which is far larger than the uniform sum n,
    and gives a score close to π/2 ≈ 1.5708.
    """
    idx = np.arange(1, n + 1, dtype=np.float64)
    seq = 1.0 / np.sqrt(idx)
    return seq


def _make_block_perturbation(base, rng):
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    The block size is a small fraction of the whole sequence.
    """
    cand = base.copy()
    n = cand.size
    # block width between n/200 and n/12
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _make_swap(base, rng):
    """
    Randomly swap two positions of ``base``.  The sum stays unchanged.
    This is the key move that reshuffles large‑with‑large overlaps.
    """
    cand = base.copy()
    n = cand.size
    i, j = rng.integers(0, n, size=2)
    # ensure i != j
    while j == i:
        j = rng.integers(0, n)
    cand[i], cand[j] = cand[j], cand[i]
    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed=42, budget_s=1000, **kwargs):
    """
    Search for a sequence with minimal evaluation score.
    Returns a list of floats (the best sequence found before the deadline).

    Parameters
    ----------
    seed : int, optional
        Random seed for reproducibility.
    budget_s : float, optional
        Time budget in seconds (the function must return before this time).

    Returns
    -------
    list[float]
        The best sequence discovered.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – use the provided construction if available.
    # ------------------------------------------------------------------
    try:
        n = int(len(height_sequence_1))
        base_seq = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        n = 4806
        base_seq = _make_initial_inverse_sqrt(n)

    # ------------------------------------------------------------------
    # Initialise the current (parent) sequence.
    # ------------------------------------------------------------------
    parent = np.clip(base_seq, 0.0, 1000.0)
    best_seq = parent.copy()
    best_score = _eval(best_seq)

    # Helper to update the global best
    def maybe_update(candidate, score):
        nonlocal best_seq, best_score
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_seq = candidate.copy()
            print(f"[{time.monotonic() - start_time:6.1f}s] New best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Temperature schedule for occasional uphill moves.
    # Linear cooling from T_start to T_end over the whole budget.
    # ------------------------------------------------------------------
    T_start = 0.1
    T_end = 1e-4

    iteration = 0
    while time.monotonic() < deadline:
        # linear temperature based on elapsed proportion
        elapsed = time.monotonic() - start_time
        prog = elapsed / (deadline - start_time)
        T = T_start * (1 - prog) + T_end * prog

        # Choose a move type
        move_type = rng.random()
        if move_type < 0.55:
            # Block perturbation (exploration)
            cand = _make_block_perturbation(parent, rng)
        elif move_type < 0.85:
            # Swap two positions (re‑ordering)
            cand = _make_swap(parent, rng)
        else:
            # Small global scaling – useful for fine‑tuning
            factor = float(np.exp(rng.normal(0.0, 0.02)))
            cand = np.clip(parent * factor, 0.0, 1000.0)

        # Evaluate candidate
        cand_score = _eval(cand)

        # Metropolis acceptance
        if cand_score < best_score or rng.random() < math.exp(-(cand_score - best_score) / max(T, 1e-12)):
            # Accept as the new parent (even if not a new global best)
            parent = cand
            if cand_score < best_score:
                maybe_update(cand, cand_score)

        iteration += 1
        # occasional momentum pull towards the global best
        if iteration % 30 == 0:
            parent = 0.9 * parent + 0.1 * best_seq
            np.clip(parent, 0.0, 1000.0, out=parent)

        # light progress report each few seconds
        if iteration % 200 == 0:
            print(f"[{elapsed:6.1f}s] iter {iteration:6d}  best {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the allowed range)
    # ------------------------------------------------------------------
    print(f"Search finished – best score {best_score:.6f}")
    final = np.clip(best_seq, 0.0, 1000.0)
    return final.tolist()
