def compute_conv_max_and_idx(a):
    # return max value and index
    # compute via FFT for efficiency
    n = a.shape[0]
    # zero pad length L = 2*n-1
    L = 1 << ((2*n-1)-1).bit_length()
    # compute FFT
    A = np.fft.rfft(a, n=L)
    b = np.fft.irfft(A * A, n=L)  # real convolution length L
    # take first 2*n-1 entries
    b = b[:2*n-1]
    # find max and index
    idx = np.argmax(b)
    max_val = b[idx]
    return max_val, idx

def full_convolution(a):
    n = a.shape[0]

def gradient_step(a, sum_target, lr, idx_max):
    n = len(a)
    g = np.zeros_like(a)
    i_min = max(0, idx_max - (n-1))
    i_max = min(n-1, idx_max)
    i_range = np.arange(i_min, i_max+1)
    j_range = idx_max - i_range
    g[i_range] = a[i_range] + a[j_range]
    # step
    cand = a - lr * g
    # clip
    cand = np.clip(cand, 0.0, 1000.0)
    # renorm
    s = cand.sum()
    if s > 0:
        cand = cand * (sum_target / s)
    else:
        # fallback keep a unchanged
        cand = a.copy()
    return cand
```

"""
Propose a step‑function (list of non‑negative heights) that minimises the
security‑evaluation function.  

Algorithm outline
-----------------
1. **Initialisation** – take the supplied `height_sequence_1` (if present) or a
   modest uniform vector.  Clip to the allowed range and compute its score.

2. **Parameter sweep** – generate a few analytically‑motivated families
   (power‑law, shifted‑sqrt, exponential, hybrid power‑law·exp) and keep the
   best candidate.

3. **Gradient descent phase** – repeatedly reduce the current maximal
   convolution value.  For the lag `k*` where `max(b)= (f*f)[k*]` the gradient
   of `max(b)` w.r.t. `f_i` is `f_i+f_{k*-i}` (zero outside the overlap).  We
   take a few step‑sizes, clip, renormalise the sum to the original total and
   keep any improvement.  This phase consumes ~40 % of the time budget.

4. **Stochastic refinement phase** – a hill‑climber with several cheap moves:
   block multiplicative perturbation, mass‑swap, smoothing, symmetric averaging,
   targeted swap (moves mass from a “dangerous’’ index that contributes heavily
   to the maximal lag to a “safe’’ index), and a small uniform inflation of the
   whole sequence (which can increase the total sum and lower the objective).
   All moves (except inflation) are renormalised to keep the total sum
   constant; inflation is kept permanent when it yields improvement.
   This phase uses the remaining time.

5. Return the best sequence encountered as a plain ``list[float]``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: fast full convolution (self‑convolution) via FFT
def _full_self_convolution(a: np.ndarray) -> np.ndarray:
    """Return np.convolve(a, a) using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    # length needed for full convolution = 2*n-1
    L = 1 << ((2 * n - 2).bit_length())  # next power of two
    A = np.fft.rfft(a, n=L)
    b = np.fft.irfft(A * A, n=L)
    return b[: 2 * n - 1]


# ----------------------------------------------------------------------
def _max_conv_idx(a: np.ndarray):
    """Return (max_value, argmax_index) of the self‑convolution of a."""
    b = _full_self_convolution(a)
    idx = int(b.argmax())
    return float(b[idx]), idx


# ----------------------------------------------------------------------
def _score_from_array(a: np.ndarray) -> float:
    """Score exactly as evaluate_sequence would compute (without safety checks)."""
    n = a.shape[0]
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return float("inf")
    max_b, _ = _max_conv_idx(a)
    return 2.0 * n * max_b / (sum_a * sum_a)


# ----------------------------------------------------------------------
def _make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) block multiplicative perturbation (mirroring optional)."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 8 % of n
    width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
def _gradient_step(a: np.ndarray, sum_target: float,
                   lr: float, lag_idx: int) -> np.ndarray:
    """One gradient‑descent step trying to reduce the maximal convolution entry."""
    n = a.shape[0]
    # build gradient g_i = a_i + a_{lag-i} for overlapping indices
    g = np.zeros_like(a)
    i_min = max(0, lag_idx - (n - 1))
    i_max = min(n - 1, lag_idx)
    if i_min <= i_max:
        i_range = np.arange(i_min, i_max + 1)
        j_range = lag_idx - i_range
        g[i_range] = a[i_range] + a[j_range]
    # take step, clip and renormalise to original sum
    cand = a - lr * g
    cand = np.clip(cand, 0.0, 1000.0)
    s = cand.sum()
    if s > 0:
        cand = cand * (sum_target / s)
    else:
        cand = a.copy()
    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> list[float]:
    """
    Search for a low‑evaluation step function.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Approximate time budget (seconds).  The function stops a few seconds
        before the limit to avoid forced termination.
    **kwargs
        Ignored – kept for compatibility with the evaluation harness.

    Returns
    -------
    List[float]
        The best height sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Retrieve the starting sequence (height_sequence_1 if available)
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                base_seq = np.clip(arr, 0.0, 1000.0).copy()
            else:
                raise ValueError
        except Exception:
            base_seq = None
    else:
        base_seq = None

    if base_seq is None:
        # simple fallback: uniform vector whose sum ≈ √(2n) (the normalisation used in the paper)
        n_fallback = 256
        base_seq = np.full(n_fallback, np.sqrt(2.0 * n_fallback) / n_fallback,
                           dtype=np.float64)

    n = base_seq.shape[0]
    sum_target = float(base_seq.sum())
    if sum_target < 0.01:
        sum_target = float(np.sqrt(2.0 * n))
        base_seq = np.full(n, sum_target / n, dtype=np.float64)

    best = base_seq.copy()
    best_score = _score_from_array(best)
    print(f"[init] n={n}, init_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) PARAMETER SWEEP – a few analytical families
    # ------------------------------------------------------------------
    # power‑law a_i = (i+1)^{-p}
    for p in np.linspace(0.20, 1.10, 10):
        cand = 1.0 / np.power(np.arange(1, n + 1, dtype=np.float64), p)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (sum_target / cand.sum())
        sc = _score_from_array(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] power p={p:.3f} → {sc:.6f}")

    # shifted sqrt a_i = 1/√(i+α)
    for shift in np.linspace(0.0, 10.0, 11):
        idx = np.arange(1, n + 1, dtype=np.float64) + shift
        cand = 1.0 / np.sqrt(idx)
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (sum_target / cand.sum())
        sc = _score_from_array(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] sqrt shift={shift:.1f} → {sc:.6f}")

    # exponential decay a_i = exp(-λ i)
    for lam in np.logspace(-4, -1, 8):
        cand = np.exp(-lam * np.arange(n, dtype=np.float64))
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (sum_target / cand.sum())
        sc = _score_from_array(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] exp λ={lam:.5f} → {sc:.6f}")

    # hybrid a_i = (i+1)^{-p} * exp(-λ i)
    for p in (0.4, 0.6, 0.8):
        for lam in (0.001, 0.003, 0.01):
            idx = np.arange(1, n + 1, dtype=np.float64)
            cand = (1.0 / np.power(idx, p)) * np.exp(-lam * (idx - 1))
            cand = np.clip(cand, 0.0, 1000.0)
            cand = cand * (sum_target / cand.sum())
            sc = _score_from_array(cand)
            if sc < best_score:
                best, best_score = cand, sc
                print(f"[sweep] hybrid p={p:.1f} λ={lam:.4f} → {sc:.6f}")

    # piecewise constant – first m entries constant, rest zero
    for frac in (0.1, 0.25, 0.4, 0.6, 0.8):
        m = max(1, int(frac * n))
        cand = np.zeros(n, dtype=np.float64)
        cand[:m] = 1.0
        cand = np.clip(cand, 0.0, 1000.0)
        cand = cand * (sum_target / cand.sum())
        sc = _score_from_array(cand)
        if sc < best_score:
            best, best_score = cand, sc
            print(f"[sweep] piecewise m={m} → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2) GRADIENT‑DESCENT PHASE (≈40 % of total budget)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    total_deadline = start_time + max(0.0, float(budget_s) - 2.0)  # safety margin
    grad_deadline = start_time + 0.40 * (total_deadline - start_time)

    current = best.copy()
    current_score = best_score
    # learning‑rate schedule – start a bit aggressive, then decay
    lr = 0.08

    while time.monotonic() < grad_deadline:
        max_b, lag_idx = _max_conv_idx(current)

        # try a few scaled step sizes
        best_local = None
        best_local_score = float("inf")
        for factor in (0.5, 1.0, 2.0):
            cand = _gradient_step(current, sum_target, lr * factor, lag_idx)
            sc = _score_from_array(cand)
            if sc < best_local_score:
                best_local_score = sc
                best_local = cand

        # accept if we did not worsen the current state
        if best_local_score <= current_score:
            current = best_local
            current_score = best_local_score
        else:
            # maybe a slightly better global point appeared
            if best_local_score < best_score:
                best = best_local.copy()
                best_score = best_local_score
                current = best_local
                current_score = best_local_score
                print(f"[grad] new global best {best_score:.6f}")

        # update global best if current improved it
        if current_score < best_score:
            best = current.copy()
            best_score = current_score
            print(f"[grad] improved best to {best_score:.6f}")

        # slowly decay learning rate
        lr *= 0.9995
        if lr < 1e-5:
            lr = 1e-5

    # ------------------------------------------------------------------
    # 3) STOCHASTIC REFINEMENT/HILL‑CLIMBING (remaining time)
    # ------------------------------------------------------------------
    current_score = best_score   # sync with best
    while time.monotonic() < total_deadline:
        move = rng.choice(
            ["block", "swap", "smooth", "targeted_swap", "reverse_avg", "inflate"]
        )
        if move == "block":
            cand = _make_block_perturbation(current, rng)

        elif move == "swap":
            cand = current.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.uniform(0.0, min(cand[i], 0.5 * cand[i] + 1e-12))
                cand[i] -= delta
                cand[j] += delta

        elif move == "smooth":
            cand = 0.5 * (current + current[::-1])
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            cand = cand * jitter

        elif move == "reverse_avg":
            cand = 0.5 * (current + current[::-1])

        elif move == "targeted_swap":
            # Identify the lag with maximal convolution
            _, lag_idx = _max_conv_idx(current)
            idx_arr = np.arange(n)
            partner = lag_idx - idx_arr
            valid = (partner >= 0) & (partner < n)
            contrib = np.zeros(n, dtype=np.float64)
            contrib[valid] = current[idx_arr[valid]] * current[partner[valid]]

            # source (dangerous) index – proportional to contribution
            if contrib.sum() > 0:
                src = rng.choice(n, p=contrib / contrib.sum())
            else:
                src = rng.integers(0, n)

            # destination (safe) index – opposite weighting
            safe_weights = 1.0 / (contrib + 1e-12)
            safe_weights = safe_weights / safe_weights.sum()
            dst = rng.choice(n, p=safe_weights)

            cand = current.copy()
            amt = rng.uniform(0.0, min(cand[src], 0.5 * cand[src] + 1e-12))
            cand[src] -= amt
            cand[dst] += amt

        elif move == "inflate":
            # Uniform inflation – may increase total sum and improve the bound
            factor = 1.0 + rng.uniform(0.001, 0.010)
            cand = np.clip(current * factor, 0.0, 1000.0)
            # No renormalisation here – we truly increase the sum
        else:
            cand = current.copy()

        # Clip and (usually) renormalise
        cand = np.clip(cand, 0.0, 1000.0)
        if move != "inflate":
            s = cand.sum()
            if s > 0:
                cand = cand * (sum_target / s)

        sc = _score_from_array(cand)

        # Accept any improvement over the global best
        if sc < best_score:
            best = cand.copy()
            best_score = sc
            current = cand.copy()
            current_score = sc
            # If we inflated, also update the target sum for future moves
            if move == "inflate":
                sum_target = float(cand.sum())
            print(f"[search] new best {best_score:.6f} via {move}")

        # Otherwise accept if it does not worsen the current state (hill‑climb)
        elif sc <= current_score:
            current = cand.copy()
            current_score = sc

    # ------------------------------------------------------------------
    print(f"[done] best_score={best_score:.6f}")
    # Return as plain Python list
    return [float(v) for v in best]
