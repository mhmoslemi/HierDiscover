   candidate of this stage seeds the local optimiser.

4. **Local optimiser** – a stochastic hill‑climber with several move types:
   * *gradient move* – locate the index where the convolution attains its
     maximum, compute each element’s contribution to that peak and transfer a
     small amount of mass from the worst contributor to a low‑mass position.
   * *block perturbation* – multiplicative jitter on a random contiguous block,
     optionally mirrored.

   * *smooth* – short moving‑average smoothing.
   * *jitter* – element‑wise multiplicative noise.

5. **Time budget** – the routine runs for ``budget_s`` seconds (default ≈ 1000 s)

def _fft_score(a: np.ndarray) -> float:
    """Fast replica of ``evaluate_sequence`` using FFT convolution."""
    a = np.clip(a, 0.0, 1000.0)
    total = a.sum()
    if total < 0.01:
        return np.inf
    n = a.shape[0]
    # length for zero‑padded FFT (next power of two for speed)
    L = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b / (total * total))


def _normalise_and_clip(v: np.ndarray, n: int) -> np.ndarray:
    """Clip to [0,1000] and renormalise so that sum(v) == n (average = 1)."""
    v = np.clip(v, 0.0, 1000.0)
    s = v.sum()
    if s <= 0:
        v = np.full_like(v, 1.0)
        s = v.sum()
    return v * (n / s)
