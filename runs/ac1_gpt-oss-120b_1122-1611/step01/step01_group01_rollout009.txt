Projected onto simplex algorithm:

def project_simplex(v):
    # Project onto the probability simplex {x >= 0, sum x = 1}
    # Standard O(n log n) algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _gradient_from_weight(weight: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    rev_a = a[::-1]
    # Convolve weight (len 2n-1) with rev_a (len n) → length 3n-2.
    # Use FFT of length next_pow2( (2n-1)+(n)-1 ) = pow2(3n-2).
    # However we can reuse same fft_len as for self conv? We need length bigger: 3n-2 maybe > 2n-1. So compute conv_len = 1 << (( (2*n-1)+(n)-1 -1).bit_length())
    conv_len = 1 << ((3*n - 2 - 1).bit_length())
    # Compute FFT for weight and rev_a.
    fw = np.fft.rfft(weight, n=conv_len)
    fr = np.fft.rfft(rev_a, n=conv_len)
    fc = fw * fr
    conv = np.fft.irfft(fc, n=conv_len)[: 2*n - 1 + n - 1] # i.e., 3n-2
    # Extract gradient slice.
    start = n - 1
    grad = conv[start : start + n]
    return grad

"""
Propose a step‑function vector that minimises

    score = 2 * n * max(conv(a, a))   (sum(a) is forced to 1)

The algorithm improves on the previous exponentiated‑gradient method by
* using a *soft‑max* (temperature) smoothing of the convolution peak, which
  gives a sub‑gradient that takes many “almost‑active’’ constraints into
  account,
* computing that sub‑gradient efficiently via a single FFT‑based correlation,
* decaying the temperature (α) and the step‑size (η) during the run,
* adding occasional random block perturbations to escape shallow local minima,
* keeping the best vector seen so far.

All operations are O(n log n) and the routine respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_self(n: int) -> int:
    """FFT length required for a linear self‑convolution of length‑n vectors."""
    return _next_pow2(2 * n - 1)


def _fft_len_corr(n: int) -> int:
    """FFT length for the correlation (weight * rev(a)) needed for the gradient."""
    # length(weight) = 2n‑1, length(rev(a)) = n ⇒ linear conv length = 3n‑2
    return _next_pow2(3 * n - 2)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (real input)."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_from_weight(weight: np.ndarray,
                          a: np.ndarray,
                          n: int,
                          fft_len_corr: int) -> np.ndarray:
    """
    Compute the gradient ∇_a  Σ_k w_k (a*a)[k] .
    This equals the cross‑correlation of `weight` with the reversed `a`.
    Returned vector has length n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(weight, n=fft_len_corr)
    fr = np.fft.rfft(rev_a, n=fft_len_corr)
    fc = fw * fr
    full = np.fft.irfft(fc, n=fft_len_corr)
    total_len = weight.size + rev_a.size - 1   # = 3n‑2
    full = full[:total_len]
    start = n - 1
    grad = full[start:start + n]
    return grad


# ----------------------------------------------------------------------
# Random perturbation helper
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

    """

    try:
