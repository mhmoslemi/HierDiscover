"""
Propose a low‑score step‑function by iteratively flattening the self‑convolution.
The method follows the “LP‑update” idea from the reference:

1.  Start from the classic α‑sequence (C(2i,i)/4**i) and scale it so that
    ∑a_i = √(2·n).  With this scaling the evaluation score equals the
    maximal convolution entry M = max_k (a∗a)_k.

2.  At each outer iteration solve a linear program
        maximise   1ᵀg
        subject to   a∗g ≤ M      (element‑wise)
                    g ≥ 0
   The matrix that implements the convolution a∗g is built explicitly as a
   Toeplitz (sparse) matrix.  The LP pushes mass into positions that do not
   increase the peak of the convolution, thereby flattening the autocorrelation.

3.  Rescale the LP solution g to the same ℓ₁‑norm √(2·n) and move a towards g:
        a ← (1‑t)·a + t·g      with a step size t (adaptively reduced when the
        peak does not decrease).

4.  Occasionally perform a cheap random mass‑transfer to escape possible
    plateaus.

The loop runs until the time budget expires (default 1000 s).  The best
sequence found (scaled back to ℓ₁ = 1 for readability) is returned as a
plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """α_i = C(2i,i) / 4**i  (i = 0 .. n‑1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def build_conv_matrix(a: np.ndarray) -> csr_matrix:
    """
    Build the (2n‑1)×n matrix C such that C @ g = a ∗ g (linear convolution).
    C is very sparse: column i contains the value a[i] on rows i … i+n‑1.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    # lil is efficient for incremental column‑wise construction
    C = lil_matrix((rows, n), dtype=np.float64)
    for i, ai in enumerate(a):
        # rows i .. i+n‑1 (clipped to matrix height)
        end = i + n
        if end > rows:
            end = rows
        C[i:end, i] = ai
    return C.tocsr()


def evaluate_sequence_wrapper(seq: np.ndarray) -> float:
    """Thin wrapper around the provided evaluate_sequence (list → float)."""
    # the runner already imported evaluate_sequence
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a coefficient sequence with a low evaluation score.
    The algorithm uses the LP‑flattening scheme described above.
    Returns a list of non‑negative heights (ℓ₁ normalised to 1).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a problem size.
    # Use the length of the previously discovered construction if it exists,
    # otherwise fall back to a moderate size that keeps the LP tractable.
    # --------------------------------------------------------------
    if isinstance(globals().get("height_sequence_1"), (list, np.ndarray)) \
            and len(height_sequence_1) > 0:
        n = min(len(height_sequence_1), 1200)   # cap for speed
    else:
        n = 1200

    # --------------------------------------------------------------
    # Initialise with the α‑sequence and scale to ℓ₁ = sqrt(2·n)
    # (with this scaling the evaluation score equals the maximal
    #  convolution entry, i.e. the quantity we are trying to minimise).
    # --------------------------------------------------------------
    a = generate_alpha_sequence(n)
    target_sum = np.sqrt(2.0 * n)
    a = a / a.sum() * target_sum

    best = a.copy()
    best_score = evaluate_sequence_wrapper(best)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Search parameters
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    t = 0.5           # initial step towards the LP solution
    min_t = 1e-4
    lp_every = 7     # run an LP every few iterations (trades time vs progress)

    while time.monotonic() < deadline:
        iteration += 1

        # ---------- current maximal convolution value ----------
        conv_a = np.convolve(a, a)
        M = conv_a.max()

        # ---------- occasional random mass transfer ----------
        if rng.random() < 0.12:
            i = rng.integers(n)
            j = rng.integers(n)
            if i != j:
                delta = a[i] * rng.uniform(0.0, 0.1)
                a[i] -= delta
                a[j] += delta
                # re‑normalise to keep ℓ₁ = sqrt(2·n)
                a = np.clip(a, 0.0, None)
                a = a / a.sum() * target_sum

        # ---------- LP‑based flattening step ----------
        if iteration % lp_every == 0:
            # build the convolution matrix for the current a
            C = build_conv_matrix(a)

            # maximise 1ᵀg  → minimise -1ᵀg
            c = -np.ones(n, dtype=np.float64)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)

            # solve the LP (sparse formulation, high‑performance HiGHS solver)
            res = linprog(c,
                          A_ub=C,
                          b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method="highs",
                          options={"disp": False})

            if not res.success:
                # fall back to a simple perturbation if LP fails
                print(f"[LP] iteration {iteration} failed: {res.message}")
                continue

            g = res.x
            if g.sum() == 0.0:          # degenerate – skip this round
                continue

            # scale g back to the required ℓ₁‑norm
            g = g * (target_sum / g.sum())

            # try to move a toward g; adapt step size if needed
            step_accepted = False
            cur_t = t
            while cur_t >= min_t:
                a_candidate = (1.0 - cur_t) * a + cur_t * g
                M_candidate = np.convolve(a_candidate, a_candidate).max()
                if M_candidate < M - 1e-12:
                    a = a_candidate
                    step_accepted = True
                    t = min(0.9, cur_t * 1.05)   # slowly increase if we are successful
                    break
                cur_t *= 0.5                     # halve step and retry

            if not step_accepted:
                t = max(min_t, t * 0.7)          # shrink global step if we could not improve

        # ---------- record best candidate ----------
        score = evaluate_sequence_wrapper(a)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{iteration:06d}] new best score={best_score:.6f}")

        # early exit if we already beat the target (optional)
        if best_score < 1.503:   # the required bound 1.5025, give a tiny margin
            break

    # ------------------------------------------------------------------
    # Return the best sequence found, normalised to ℓ₁ = 1 (more natural for
    # the caller).  The evaluation function is scale‑invariant, so this does
    # not affect the score.
    # ------------------------------------------------------------------
    best = best / best.sum()
    print(f"[final] iterations={iteration}, best score={best_score:.6f}")
    return best.tolist()
