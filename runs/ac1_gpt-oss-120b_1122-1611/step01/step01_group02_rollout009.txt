def power_law_sequence(length: int, exponent: float) -> np.ndarray:
    """Return a vector a_i = (i+1)^{-exponent} for i=0..length‑1."""
    idx = np.arange(1, length + 1, dtype=np.float64)
    seq = np.power(idx, -exponent)
    return seq

def compute_score(seq: np.ndarray) -> float:
    """Compute the same objective that evaluate_sequence uses."""
    n = seq.size
    b = np.convolve(seq, seq)
    max_b = np.max(b)
    sum_a = np.sum(seq)
    if sum_a < 0.01:

    if rng.random() < 0.35:

"""

def power_law_sequence(length: int, exponent: float) -> np.ndarray:
