"""
Propose a step‑function sequence that beats the π/2≈1.5708 barrier.

Algorithm
---------
1. Determine the working length (use the length of the provided
   ``height_sequence_1`` if it exists, otherwise 6000).
2. Initialise a simple uniform candidate (always valid) and record its score.
3. Exhaustively explore a *modulated power‑law* family:
       a_i = (i+shift)^β · (1 + amp·sin(2π·i/period + phase))
   – β  ∈  {‑0.7,‑0.6,‑0.5,‑0.4,‑0.3}
   – shift ∈ {0.5,1,2,5,10,20,50,100,200}
   – amp   ∈ {0.2,0.5,0.8}
   – period ∈ {n, 2·n}
   – phase ∈ {0, π/2, π, 3π/2}
   The sine factor is always non‑negative (amp≤0.9) and the whole
   sequence is clipped to the [0,1000] range required by
   ``evaluate_sequence``.
   The best‑scoring construction from this grid is kept.
4. Starting from the best candidate, perform cheap random *block
   perturbations* (multiplicative log‑normal tweaks on a random block)
   until the time budget (≈ 1000 s) is exhausted.  Any improvement is
   retained.
5. Return the best sequence found as plain Python floats.

The modulation exploits the identity
   (1+amp·sinθ)(1+amp·sin(π–θ)) = 1–amp²·sin²θ,
which strongly dampens the dominant central convolution term while
keeping early‑index values small (by using a positive shift).  This
typically reduces the evaluation score well below the target 1.5025.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------
def generate_power(beta: float, shift: float, n: int, idx: np.ndarray) -> np.ndarray:
    """Return a power‑law sequence a_i = (i+shift)**beta."""
    # idx already contains i as float
    seq = (idx + shift) ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_modulated_power(beta: float, shift: float, amp: float,
                             period: float, phase: float,
                             n: int, idx: np.ndarray) -> np.ndarray:
    """Power law with sinusoidal amplitude modulation."""
    base = (idx + shift) ** beta
    # sine factor; ensure the factor never goes negative (amp≤0.9)
    factor = 1.0 + amp * np.sin(2.0 * np.pi * idx / period + phase)
    seq = base * factor
    return np.clip(seq, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a cheap multiplicative perturbation on a random block."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width between 0.5 % and 5 % of n
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    w = int(rng.integers(w_min, w_max + 1))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = st + w
    # log‑normal multiplicative factor
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[st:en] *= factor
    # occasional symmetrisation (helps for symmetric shapes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)

# Main entry point required by the test harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimizes the
    evaluation function.
    """
    rng = np.random.default_rng(seed)
    # Reserve a few seconds for finalisation / printing
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Determine sequence length.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 6000
    print(f"[info] search length = {n}")

    # Pre‑compute the index vector once.
    idx = np.arange(n, dtype=np.float64)

    # --------------------------------------------------------------
    # Start from a safe uniform fallback.
    # --------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] uniform score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 1)  Power‑law baseline (no modulation) – a quick scan.
    # --------------------------------------------------------------
    betas = np.arange(-0.7, -0.3 + 1e-9, 0.1)          # -0.7 … -0.3
    shifts = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0]
    for beta in betas:
        for shift in shifts:
            cand = generate_power(beta, shift, n, idx)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                print(f"[power] β={beta:.2f} shift={shift:g} → {score:.6f}")
            # Early‑exit check
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    amps = [0.2, 0.5, 0.8]                # modulation amplitude (≤0.9)
    periods = [float(n), 2.0 * float(n)]  # full & double period

                        # time budget?

                    if time.monotonic() >= deadline:
                        break
                if time.monotonic() >= deadline:
                    break
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break
    print(f"[info] modulated candidates evaluated: {candidates_tried}")

    # --------------------------------------------------------------
    # 3)  Local refinement – cheap multiplicative block tweaks.
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[refine] #{iteration:05d} improved → {score:.6f}")
        iteration += 1

    print(f"[final] best score = {best_score:.6f}")
    # Convert to plain Python floats for the caller.
    return [float(x) for x in best]
