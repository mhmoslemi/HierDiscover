def targeted_move(seq, rng, frac=0.05):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_idx = int(np.argmax(conv))
    product = np.zeros(n)
    # valid indices i where max_idx-i in [0,n-1]
    lo = max(0, max_idx - (n-1))
    hi = min(n-1, max_idx)
    # Only compute in this range
    i_arr = np.arange(lo, hi+1)
    j_arr = max_idx - i_arr
    product[i_arr] = seq[i_arr] * seq[j_arr]
    # source index with highest product and positive seq
    src_candidates = np.where(seq > 0)[0]
    if src_candidates.size == 0:
        return seq.copy()
    # compute product for those indices (others zero)
    src_idx = src_candidates[np.argmax(product[src_candidates])]
    # dest index with minimal product (maybe includes zeros)
    dest_idx = int(rng.choice(np.where(product == product.min())[0]))
    if src_idx == dest_idx:
        # pick random dest
        dest_idx = int(rng.integers(0, n))
    delta = min(seq[src_idx] * frac, seq[src_idx])
    new_seq = seq.copy()
    new_seq[src_idx] -= delta
    new_seq[dest_idx] += delta
    return np.clip(new_seq, 0.0, 1000.0)

"""
Propose a non‑negative step‑function (list of heights) with a low
evaluation score.

Algorithm
---------

1. **Broad parameter sweep** – generate candidates from a few analytic
   families (power‑law, shifted sqrt, combined) for several lengths.
   The best candidate (lowest value of the objective
   2·n·max(conv)/sum²) is kept.

2. **Targeted max‑convolution reduction** – repeatedly identify the shift
   where the convolution attains its maximum.  The contribution of each
   index to that peak is a_i·a_{k‑i}.  Mass is moved from an index with a
   large contribution to one with a small contribution (a simple greedy
   step).  This directly reduces the peak while preserving the total
   sum.

3. **Diverse local moves** – in addition to the targeted move we apply
   cheap random perturbations (block size change, random swap and a
   local smoothing step).  Any move that improves the score is accepted.

All moves keep the sequence non‑negative and clipped to the range
[0, 1000] required by `evaluate_sequence`.  The search runs until the
time budget expires and returns the best sequence found.

The implementation computes the objective itself (using np.convolve)
instead of calling `evaluate_sequence` on every trial, which is much
faster.  A final call to `evaluate_sequence` sanity‑checks the result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) perturbation: multiply a random contiguous block by a
    log‑normal factor and optionally mirror the whole sequence.
    """
    cand = base.copy()
    n = cand.size
    # width between 0.5 % and 8 % of the length
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= float(np.exp(rng.normal(0.0, 0.12)))
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
def compute_score(seq: np.ndarray) -> float:
    """
    Compute the evaluation function:
        2·n·max(conv(seq,seq)) / (sum(seq)²)
    Returns np.inf if the sum is too small.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * max_b / (sum_a * sum_a)

            # 3) source = index with largest contribution (and positive mass)
            src_mask = (cand > 1e-12) & (contrib > 0.0)
            if src_mask.any():
                src_idx = int(np.argmax(contrib * src_mask))
                # 4) destination = index with smallest contribution
                dst_idx = int(np.argmin(contrib))
                if src_idx != dst_idx:
                    # move a modest fraction of the source mass
                    delta = min(cand[src_idx] * 0.07, cand[src_idx])
                    cand[src_idx] -= delta
                    cand[dst_idx] += delta
        elif move == "block":
            cand = make_block_perturbation(cand, rng)
        elif move == "swap":
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # transfer a random fraction of the smaller of the two buckets
                delta = rng.uniform(0.0, min(cand[i], cand[i] * 0.3 + 1e-12))
                cand[i] -= delta
                cand[j] += delta
        else:  # "smooth"
            # a gentle moving‑average filter of random width
            width = int(rng.integers(3, 9))
            kernel = np.ones(width, dtype=np.float64) / width
            cand = np.convolve(cand, kernel, mode="same")

        cand = np.clip(cand, 0.0, 1000.0)
        score = compute_score(cand)
        if np.isfinite(score) and score < best_score:
            best_seq, best_score = cand, score
            print(f"[search] new best {score:.6f} via {move}")

    # ------------------------------------------------------------------
    # Final sanity‑check with the official evaluator (optional)
    # ------------------------------------------------------------------
    try:
        official = float(evaluate_sequence(best_seq.tolist()))
        if official < best_score:
            best_score = official
            print(f"[final] evaluate_sequence gave an even better score {official:.6f}")
    except Exception:
        pass

    print(f"[done] best_score={best_score:.6f}, length={best_seq.size}")
    return [float(v) for v in best_seq]
