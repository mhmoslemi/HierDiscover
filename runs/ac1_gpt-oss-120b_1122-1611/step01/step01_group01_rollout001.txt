"""
Algorithm: multi‑phase exponentiated‑gradient / mirror‑descent.
The vector a (non‑negative, sum = 1) represents the step‑function heights.
Score (with sum = 1) is 2·n·max(a*a).  We minimise the peak of the
self‑convolution.

Phase 1 (first ~20 % of the time) uses the ordinary sub‑gradient at the
current maximum (partner vector) with an aggressive multiplicative step.
Phase 2 uses a soft‑max weighting of all near‑maximal convolution entries;
the weighted partner is obtained by a second FFT‑based convolution.
An Adam‑style momentum correction is applied to the log‑variables, and
the step‑size η is annealed from 0.8 down to 0.05.  Random block
perturbations are injected occasionally to escape plateaus.
The best vector seen so far is stored and finally returned as a plain
list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _fft_len_for_size(size: int) -> int:
    """Next power‑of‑two ≥ size (for a real FFT)."""
    return 1 << ((size - 1).bit_length())


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Linear convolution of two real vectors using FFT.
    Returns an array of length len(x)+len(y)-1.
    """
    size = x.size + y.size - 1
    L = _fft_len_for_size(size)
    fx = np.fft.rfft(x, n=L)
    fy = np.fft.rfft(y, n=L)
    fz = fx * fy
    z = np.fft.irfft(fz, n=L)[:size]
    return z


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution a*a (linear) using a pre‑computed FFT length."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient helpers
# ----------------------------------------------------------------------


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """Partner vector w_i = a_{k-i} (zero outside the support)."""
    n = a.size
    w = np.zeros_like(a)

    if k < n:
        # overlap 0 … k
        w[: k + 1] = a[k :: -1]
    else:
        start = k - (n - 1)                # first i that has a partner
        length = (2 * n - 1) - k           # number of overlapping entries
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal scaling of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start: start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out
