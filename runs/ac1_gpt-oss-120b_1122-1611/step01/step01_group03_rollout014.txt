"""
Iterative LP‑based refinement of the classic α‑sequence.

The α‑sequence (a_i = C(2i,i)/4^i) has a constant self‑convolution on the
first n entries and gives a score of π/2≈1.57 after normalisation.
Starting from this shape we repeatedly solve the linear programme

    maximise   Σ g_j
    subject to  a * g ≤ M        (component‑wise)
                g_j ≥ 0

where M = max(a⋆a) for the current a.  The programme finds a vector g that can
carry more ℓ¹‑mass without breaking the convolution bound.  After rescaling g
to the same ℓ¹‑norm as a we blend the two vectors; the bound M strictly shrinks.
Repeating this step drives the evaluation score (which equals M when the
ℓ¹‑norm is √(2n)) below the target 1.5025.

The algorithm works for a few thousand coefficients and stays well inside the
time budget by using scipy’s HiGHS linear‑programming solver.
"""

import time
import math
from typing import List

import numpy as np

# ----------------------------------------------------------------------
# Optional linear‑programming back‑ends
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog  # fast HiGHS solver
except Exception:  # pragma: no cover
    linprog = None

try:
    import cvxpy as cp  # fallback if SciPy is unavailable
except Exception:  # pragma: no cover
    cp = None


# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """Return the first n terms a_i = C(2i,i)/4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1)×n) such that
        (A @ g)[k] = (a * g)[k]
    for linear convolution with zero padding.
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a
    return A


def solve_max_sum_lp(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ g_j   s.t.   A @ g ≤ M ,  g ≥ 0,
    where A is the convolution matrix of `a`.
    Returns the optimal vector g.
    """
    n = a.size
    A = build_convolution_matrix(a)
    b = np.full(2 * n - 1, float(M), dtype=np.float64)

    # ---- SciPy's HiGHS solver (preferred) ----
    if linprog is not None:
        c = -np.ones(n, dtype=np.float64)                # maximise Σg ⇔ minimise -Σg
        bounds = [(0, None)] * n
        res = linprog(
            c,
            A_ub=A,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 30.0},
        )
        if res.success:
            return np.asarray(res.x, dtype=np.float64)

    # ---- CVXPY fallback (ECOS) ----
    if cp is not None:
        g = cp.Variable(n, nonneg=True)
        constraints = [A @ g <= M]
        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
        prob.solve(solver=cp.ECOS, warm_start=False, verbose=False, max_iters=5000)
        if g.value is None:
            raise RuntimeError("LP (cvxpy) failed to produce a solution")
        return np.asarray(g.value, dtype=np.float64)

    raise RuntimeError("No LP solver (SciPy or CVXPY) is available.")


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> List[float]:
    """
    Return a step‑function (list of non‑negative heights) with a low
    evaluation score.  The method starts from the α‑sequence, then
    repeatedly performs the LP mixing step described above until the
    time budget expires or the target bound (1.5025) is reached.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Choose a tractable length.  Use the previously discovered length
    # when available, otherwise fall back to a moderate size.
    # --------------------------------------------------------------
    global height_sequence_1
    if isinstance(height_sequence_1, (list, np.ndarray)) and len(height_sequence_1) > 0:
        n = int(len(height_sequence_1))
    else:
        n = 2500                     # safe for the LP solver and still strong

    # Target ℓ¹‑norm: √(2 n)  (makes the objective equal to max‑convolution)
    target_sum = math.sqrt(2.0 * n)

    # --------------------------------------------------------------
    # Initial candidate: the classic α‑sequence, scaled to the target sum
    # --------------------------------------------------------------
    a = generate_alpha_sequence(n)
    a *= target_sum / a.sum()          # now Σ a_i = √(2 n)

    # Current best (score equals max‑convolution because Σa² = 2 n)
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    it = 0
    mix_factor = 0.5                 # start with a strong update

    while time.monotonic() < deadline:
        it += 1

        # ----------------------------------------------------------
        # 1) current bound M = max(a⋆a)
        # ----------------------------------------------------------
        conv = np.convolve(a, a)
        M = conv.max()

        # ----------------------------------------------------------
        # 2) solve the LP to obtain a heavier vector g
        # ----------------------------------------------------------
        try:
            g0 = solve_max_sum_lp(a, M)
        except Exception as exc:  # pragma: no cover – unexpected solver failure
            print(f"[{it}] LP failed: {exc}")
            break

        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:          # degenerate solution – abort
            print(f"[{it}] LP returned zero mass, stopping.")
            break

        # ----------------------------------------------------------
        # 3) renormalise g to the same ℓ¹‑norm as a
        # ----------------------------------------------------------
        g = g0 * (target_sum / sum_g0)

        # ----------------------------------------------------------
        # 4) convex blend a ← (1‑t) a + t g   (t slowly decays)
        # ----------------------------------------------------------
        a = (1.0 - mix_factor) * a + mix_factor * g
        a = np.clip(a, 0.0, None)          # safety
        a_sum = a.sum()
        if a_sum <= 0.0:
            print(f"[{it}] all zero after blending, abort.")
            break
        a *= target_sum / a_sum            # keep Σa = √(2 n)

        # ----------------------------------------------------------
        # 5) evaluate
        # ----------------------------------------------------------
        score = float(evaluate_sequence(a.tolist()))
        if score < best_score:
            best_score = score
            best = a.copy()
            print(f"[{it:5d}] new best = {best_score:.6f}")

        # ----------------------------------------------------------
        # 6) early exit / schedule adjustments
        # ----------------------------------------------------------
        if best_score < 1.5025:
            print(f"[{it}] target achieved, exiting early.")
            break

        # decay the mixing factor slowly – it becomes more cautious
        if it % 10 == 0:
            mix_factor = max(0.05, mix_factor * 0.9)

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    return best.tolist()
