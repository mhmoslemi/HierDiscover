Projection to simplex algorithm:

```
def project_simplex(v):
    n = len(v)
    # ensure non-negative
    v = np.maximum(v, 0)
    if v.sum() == 0:
        return np.full_like(v, 1.0 / n)
    # sorting descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    # find rho: last i where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
    # ensure nonneg
    v = np.asarray(v, dtype=np.float64)
```

The task is to minimise  

\[
F(a)=\frac{2\,n\,\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

where \((a*a)_k=\sum_{i}a_i\,a_{k-i}\) is the linear convolution.
Because the numerator and the denominator are both quadratic in the vector,
the objective is *scale‑invariant*.  
Hence we may fix the \(L_1\)‑norm to one, i.e. \(\sum_i a_i = 1\).  
With this normalisation the objective is simply  

\[
F(a)=2\,n\,\max_k (a*a)_k .
\]

The well‑known “α‑sequence’’ (central binomial coefficients divided by \(4^i\))
yields \(F\approx\pi/2\).  To beat the current best value we use a **projected
gradient descent** on the non‑smooth function \(\max_k (a*a)_k\) together with a
few cheap random perturbations to escape possible plateaus.

*Key ideas*

1. **Projection onto the probability simplex** – after every update we
   re‑normalise the vector to keep it non‑negative and of unit sum.
2. **Sub‑gradient of the max‑convolution** – let  
   \(k^\*=\arg\max_k (a*a)_k\).  
   The derivative of \((a*a)_{k^\*}\) w.r.t. \(a_i\) is  
   \(\displaystyle\frac{\partial}{\partial a_i}(a*a)_{k^\*}=2\,a_{k^\*-i}\)  
   (zero if the index is out of range).  This vector is a sub‑gradient of the
   objective.
3. **Additive gradient step** – a step `a←a−η·grad` followed by the simplex
   projection reduces the maximal convolution value in practice.
4. **Exploratory moves** – occasional Gaussian multiplicative noise,
   mass‑transfer between two positions and block‑scaling keep the search
   stochastic and prevent early stagnation.
5. **Multi‑start & parameter sweep** – besides the provided construction we
   seed the optimiser with a few power‑law and exponential sequences; the
   best among them becomes the starting point.
6. **Time‑bounded loop** – the routine stops automatically when the given
   budget expires (≈ 1000 s).

The algorithm is deterministic apart from the RNG seed and is fully
self‑contained – no file or network access is performed.

```python
"""Projected‑gradient search for a low‑max‑convolution step function.

The routine normalises every candidate to sum 1 (the feasible simplex)
and repeatedly:
  * computes the self‑convolution,
  * takes a sub‑gradient of the maximal entry,
  * performs a scaled gradient descent step with projection,
  * interleaves cheap random perturbations,
  * keeps the best solution seen so far.

The loop stops after the supplied time budget (≈ 1000 s) and returns the
best sequence found as a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, sum x_i = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.shape[0]

    # fast path – already feasible
    if (v >= 0).all() and np.isclose(v.sum(), 1.0):
        return v

    # shift to ensure positivity before sorting
    u = np.sort(v)[::-1]                       # descending order
    cssv = np.cumsum(u) - 1.0                  # cumulative sum minus 1
    ind = np.arange(1, n + 1)                  # 1‑based indices
    # find the last index where u_i > cssv_i / i
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – return uniform vector
        return np.full_like(v, 1.0 / n)

    rho = np.where(cond)[0][-1]                # last true index
    theta = cssv[rho] / (rho + 1.0)             # threshold
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / n
    else:
        w /= w_sum
    return w


def convolution_max(a: np.ndarray) -> float:
    """Return the maximal entry of the linear convolution a*a."""
    b = np.convolve(a, a)
    return float(b.max())


def objective(a: np.ndarray) -> float:
    """Objective 2·n·max(a*a) for a normalised (sum = 1) sequence."""
    n = a.shape[0]
    return 2.0 * n * convolution_max(a)


def gradient_step(a: np.ndarray, eta: float) -> np.ndarray:
    """
    One additive gradient descent step on max(a*a) with step size eta.
    The gradient is the sub‑gradient at the current maximal index.
    The result is projected back onto the simplex.
    """
    n = a.shape[0]
    b = np.convolve(a, a)
    k_max = int(np.argmax(b))

    # build the sub‑gradient
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1, dtype=np.intp)
    grad[idx] = 2.0 * a[k_max - idx]

    cand = a - eta * grad
    cand = np.clip(cand, 0.0, None)
    cand = project_simplex(cand)
    return cand


def random_move(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a cheap stochastic perturbation of a:
      – multiplicative Gaussian noise,
      – a mass‑transfer between two positions,
      – a block scaling.
    The result is projected onto the simplex.
    """
    n = a.shape[0]
    move_type = rng.random()
    if move_type < 0.33:
        # Gaussian multiplicative noise, moderate amplitude
        cand = a * (1.0 + 0.25 * rng.standard_normal(n))
    elif move_type < 0.66:
        # mass transfer (source chosen proportionally to current weight)
        src = rng.choice(n, p=a)
        dst = rng.integers(0, n)
        if src != dst:
            delta = a[src] * rng.uniform(0.0, 0.5)
            cand = a.copy()
            cand[src] -= delta
            cand[dst] += delta
        else:
            cand = a.copy()
    else:
        # block scaling – a smooth local stretch/compression
        width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
        start = int(rng.integers(0, max(1, n - width + 1)))
        stop = min(n, start + width)
        cand = a.copy()
        factor = float(np.exp(rng.normal(0.0, 0.09)))
        cand[start:stop] *= factor

    cand = np.clip(cand, 0.0, None)
    cand = project_simplex(cand)
    return cand


def generate_alpha_sequence(n: int) -> np.ndarray:
    """First n terms of a_i = C(2i,i)/4^i (the classic α‑sequence)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def generate_powerlaw_sequence(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = 1.0 / np.power(i, p)
    a /= a.sum()
    return a


def generate_exponential_sequence(n: int, r: float) -> np.ndarray:
    """a_i ∝ r^{i}, normalised to sum 1."""
    a = np.power(r, np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point – called by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a non‑negative step sequence that minimises the provided
    evaluation function.  The function runs until the time budget expires
    and returns the best sequence found (as a Python list of floats).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the problem size and an initial candidate
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals() and isinstance(
            height_sequence_1, (list, np.ndarray)) and len(height_sequence_1) > 0:
        n = int(len(height_sequence_1))
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        # fallback length – large enough to give a good bound
        n = 6000
        base = generate_alpha_sequence(n)

    # normalise to L1 = 1 (the objective is scale‑invariant)
    base = base / base.sum()

    # --------------------------------------------------------------
    # Build a small portfolio of seed sequences (α, power‑law, exponential)
    # --------------------------------------------------------------
    seeds = [base]

    for p in (0.35, 0.40, 0.45, 0.50, 0.55):
        seeds.append(generate_powerlaw_sequence(n, p))

    for r in (0.999, 0.998, 0.997):
        seeds.append(generate_exponential_sequence(n, r))

    # keep the best among the seeds
    best = None
    best_score = float('inf')
    for s in seeds:
        sc = objective(s)
        if sc < best_score:
            best_score = sc
            best = s.copy()
    cur = best.copy()
    cur_score = best_score

    # --------------------------------------------------------------
    # Search parameters
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    it = 0
    eta = 0.45               # initial gradient step size
    grad_prob = 0.78          # probability to perform a gradient step
    stall_counter = 0        # counts iterations without improvement

    print(f"[init] length={n}, initial best score={best_score:.6f}")

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1

        # pick a move
        if rng.random() < grad_prob:
            # deterministic gradient step
            cand = gradient_step(cur, eta)
        else:
            # cheap stochastic perturbation
            cand = random_move(cur, rng)

        cand_score = objective(cand)

        # accept if we made progress
        if cand_score < cur_score - 1e-12:
            cur = cand
            cur_score = cand_score
            # update global best
            if cand_score < best_score - 1e-12:
                best = cand.copy()
                best_score = cand_score
                print(f"[{it:6d}] new best = {best_score:.6f}")
                stall_counter = 0
        else:
            stall_counter += 1

        # ------------------------------------------------------------------
        # Adaptive schedules
        # ------------------------------------------------------------------
        if it % 1000 == 0:
            # slowly shrink the gradient step
            eta = max(eta * 0.93, 5e-4)
            # increase chance of random moves a bit
            grad_prob = max(grad_prob * 0.98, 0.55)

        # if we have been stuck for a long time, re‑initialise from the best
        if stall_counter > 8000:
            cur = best + 0.02 * rng.standard_normal(n)
            cur = np.clip(cur, 0.0, None)
            cur = project_simplex(cur)
            cur_score = objective(cur)
            stall_counter = 0
            print(f"[{it:6d}] restart after stagnation – current score {cur_score:.6f}")

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    return best.tolist()
```
