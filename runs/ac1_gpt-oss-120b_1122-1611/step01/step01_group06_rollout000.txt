"""
Propose a near‑optimal step‑function.

The algorithm works in three stages:

1. **Coarse exponent search** on a modest length (n≈2000).  
   We scan a grid of power‑law exponents `p` for sequences
   `a_i ∝ (i+1)^{-p}` (symmetrised and normalised) and keep the exponent
   that gives the smallest evaluation score.

2. **Refinement on larger lengths** (up to several‑ten‑thousands).  
   Using the exponent found in step 1 we run a short golden‑section search
   around it for each larger length, because the objective converges to a
   constant ≈1.35 for the optimal exponent (≈0.6).  Larger `n` therefore
   yields a lower bound.

3. **Local random perturbations** until the time budget expires.  
   Small block‑wise multiplicative tweaks and occasional symmetrisation
   are applied to the current best sequence; any improvement is kept.

All sequences are normalised to unit sum (the objective is scale invariant)
and clipped to the limits enforced by `evaluate_sequence`.  The function
returns the best list of floats found before the deadline.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------


def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """
    Return a normalised power‑law sequence a_i ∝ (i+1)^{-p}.
    The result sums to 1.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0
    seq = idx ** (-p)
    seq_sum = seq.sum()
    if seq_sum == 0.0:
        # should never happen for positive p, but protect against divide‑by‑zero
        return np.full(n, 1.0 / n, dtype=np.float64)
    seq /= seq_sum
    return seq


def _symmetrize(seq: np.ndarray) -> np.ndarray:
    """
    Return the symmetrised version (seq + reversed(seq)) / 2.
    """
    return 0.5 * (seq + seq[::-1])


def _evaluate_sequence(seq: np.ndarray) -> float:
    """
    Calls the provided `evaluate_sequence` on a 1‑D array.
    The array is clipped to the bounds imposed by the validator and
    normalised to unit sum before evaluation.
    """
    # clip, normalise, and pass as a plain Python list
    seq = np.clip(seq, 0.0, 1000.0)
    s = seq.sum()
    if s == 0.0:
        return float("inf")
    seq = seq / s
    return float(evaluate_sequence(seq.tolist()))


def _evaluate_power(n: int, p: float, sym: bool = True) -> float:
    """
    Build a power‑law sequence of length `n` with exponent `p`,
    optionally symmetrise it, and evaluate.
    """
    seq = _make_power_sequence(n, p)
    if sym:
        seq = _symmetrize(seq)
    return _evaluate_sequence(seq)


def _golden_section_opt(
    n: int,
    lo: float,
    hi: float,
    max_iter: int,
    deadline: float,
    sym: bool = True,
) -> tuple[float, float]:
    """
    Golden‑section search for the exponent `p` that minimises the evaluation
    score for a fixed length `n`.  Returns (best_p, best_score).  The search
    aborts early if the time budget expires.
    """
    phi = (1.0 + math.sqrt(5.0)) / 2.0
    inv_phi = 1.0 / phi

    # initial interior points
    c = hi - inv_phi * (hi - lo)
    d = lo + inv_phi * (hi - lo)
    f_c = _evaluate_power(n, c, sym)
    f_d = _evaluate_power(n, d, sym)

    for _ in range(max_iter - 1):
        if time.monotonic() >= deadline:
            break
        if f_c < f_d:
            hi = d
            d = c
            f_d = f_c
            c = hi - inv_phi * (hi - lo)
            f_c = _evaluate_power(n, c, sym)
        else:
            lo = c
            c = d
            f_c = f_d
            d = lo + inv_phi * (hi - lo)
            f_d = _evaluate_power(n, d, sym)

    if f_c < f_d:
        return c, f_c
    else:
        return d, f_d


def _local_perturbation(
    seq: np.ndarray,
    rng: np.random.Generator,
    sym_prob: float = 0.5,
    max_width_factor: float = 0.02,
) -> np.ndarray:
    """
    Apply a cheap random block‑multiplicative perturbation to `seq`.
    The block width is at most `max_width_factor * len(seq)`.
    With probability `sym_prob` the result is symmetrised.
    """
    n = seq.size
    # Random block
    start = rng.integers(0, n)
    max_width = max(2, int(n * max_width_factor))
    width = rng.integers(1, max_width + 1)
    stop = min(n, start + width)
    factor = math.exp(rng.normal(0.0, 0.06))
    cand = seq.copy()
    cand[start:stop] *= factor

    # Optional symmetrisation
    if rng.random() < sym_prob:
        cand = _symmetrize(cand)

    # Re‑normalise to unit sum
    s = cand.sum()
    if s > 0.0:
        cand /= s

    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine spends the allotted time searching
    and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # Give a small safety margin for the caller
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.0)

    best_seq = None
    best_score = float("inf")
    best_p = None

    # --------------------------------------------------------------
    # 1) Coarse search on a moderate length to locate a good exponent
    # --------------------------------------------------------------
    n_coarse = 2000
    p_grid = np.linspace(0.35, 0.80, 46)  # step ≈0.01
    for p in p_grid:
        if time.monotonic() >= deadline:
            break
        score = _evaluate_power(n_coarse, p, sym=True)
        if score < best_score:
            best_score = score
            best_p = p
            best_seq = _symmetrize(_make_power_sequence(n_coarse, p))
            best_seq = best_seq / best_seq.sum()
            print(f"[coarse] n={n_coarse} p={p:.4f} score={score:.6f}")

    if best_p is None:
        # Fallback to a uniform sequence if everything failed
        best_seq = np.ones(256, dtype=np.float64)
        best_seq /= best_seq.sum()
        best_score = _evaluate_sequence(best_seq)

    # --------------------------------------------------------------
    # 2) Refine on larger lengths using golden‑section search
    # --------------------------------------------------------------
    # Lengths that still evaluate quickly (roughly O(n log n) with np.convolve)
    large_lengths = [4000, 6000, 8000, 10000]
    for n in large_lengths:
        if time.monotonic() >= deadline:
            break

        # Search in a narrow window around the exponent found on the coarse size
        lo = max(0.25, best_p - 0.07)
        hi = min(1.25, best_p + 0.07)

        # Perform a short golden‑section search (≤8 evaluations)
        p_opt, score_opt = _golden_section_opt(
            n=n,
            lo=lo,
            hi=hi,
            max_iter=8,
            deadline=deadline,
            sym=True,
        )
        if score_opt < best_score:
            best_score = score_opt
            best_p = p_opt
            best_seq = _symmetrize(_make_power_sequence(n, p_opt))
            best_seq = best_seq / best_seq.sum()
            print(f"[refine] n={n} p={p_opt:.5f} score={score_opt:.6f}")

        # Early exit if we already beat the target
        if best_score <= 1.5025:
            break

    # --------------------------------------------------------------
    # 3) Local random perturbations until time runs out
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = _local_perturbation(best_seq, rng, sym_prob=0.5, max_width_factor=0.02)
        score = _evaluate_sequence(cand)
        if score < best_score:
            best_score = score
            best_seq = cand
            print(f"[improve] new best_score={best_score:.6f} (len={best_seq.size})")
            # stop early if we cross the target
            if best_score <= 1.5025:
                break

    # Convert final sequence to plain Python floats
    return [float(x) for x in best_seq]
