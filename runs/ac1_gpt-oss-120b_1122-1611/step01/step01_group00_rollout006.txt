import time
import numpy as np

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x >= 0, sum(x)=1}."""
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):

    Generate a step‑function height sequence by projected gradient descent
    on a smooth (log‑sum‑exp) approximation of the maximal convolution entry.
    The algorithm works in the frequency domain using FFTs for cheap
    O(n log n) convolutions and keeps the iterate on the simplex
    (non‑negative entries with unit ℓ₁ norm).  An annealing schedule
    gradually sharpens the smooth‑max, random multiplicative perturbations
    are added to escape plateaus, and the best sequence encountered is
    returned.
    """
    rng = np.random.default_rng(seed)

    # Determine sequence length.  If a previous construction exists,
    # reuse its length; otherwise fall back to 1024.
    if 'height_sequence_1' in globals():
        init_seq = np.array(height_sequence_1, dtype=np.float64)
        n = init_seq.size
        # Normalise to unit ℓ₁ norm (evaluate_sequence is scale‑invariant)
        if init_seq.sum() > 0:
            a = np.clip(init_seq, 0, None)
            a = a / a.sum()
        else:
            a = np.full(n, 1.0 / n)
    else:
        n = 1024
        a = np.full(n, 1.0 / n)

    best = a.copy()
    # direct evaluation (fast) – the official evaluate_sequence is also called
    # for a reliable score.
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # loop control
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)
    iteration = 0
    no_improve = 0

    # parameters for the schedule
    alpha_start = 5.0
    alpha_end = 150.0
    alpha_ramp_iters = 50000          # number of iterations over which we increase alpha
    eta0 = 0.6                         # initial step‑size multiplier
    target_score = 1.5025

    # pre‑compute FFT size covering the longest convolution we need (w * a_rev)
    sz = 1 << ((3 * n - 2).bit_length())   # power‑of‑two >= 3n‑2

    while time.monotonic() < deadline:
        iteration += 1

        # ----- annealed smooth‑max temperature -----
        ramp = min(1.0, iteration / alpha_ramp_iters)
        alpha = alpha_start + ramp * (alpha_end - alpha_start)

        # ----- forward FFT of current a -----
        A = np.fft.rfft(a, n=sz)

        # ----- linear convolution a * a (length 2n‑1) -----
        conv = np.fft.irfft(A * A, n=sz)[: 2 * n - 1]

        # ----- smooth‑max weights w -----
        max_val = conv.max()
        # shift by the max to avoid overflow
        exp_terms = np.exp(alpha * (conv - max_val))
        w = exp_terms / exp_terms.sum()    # sums to 1

        # ----- gradient of the smooth‑max -----
        # need a_rev = a reversed for cross‑correlation
        a_rev = a[::-1]
        A_rev = np.fft.rfft(a_rev, n=sz)
        W = np.fft.rfft(w, n=sz)
        # convolution of w with a_rev (linear) – length up to 3n‑2
        grad_full = np.fft.irfft(W * A_rev, n=sz)[: 3 * n - 2]
        # extract the n entries that correspond to ∂/∂a_i
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- step size (diminishing) -----
        eta = eta0 / np.sqrt(iteration)

        # ----- gradient step + simplex projection -----
        a_candidate = a - eta * grad
        a_candidate = _project_simplex(a_candidate)

        # ----- occasional random shake‑up (1 % chance) -----
        if rng.random() < 0.01:
            pos = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.25)))
            a_candidate[pos] *= factor
            a_candidate = _project_simplex(a_candidate)

        # ----- evaluate candidate -----
        # Fast evaluation using the convolution we will compute anyway
        A_cand = np.fft.rfft(a_candidate, n=sz)
        conv_cand = np.fft.irfft(A_cand * A_cand, n=sz)[: 2 * n - 1]
        max_b = conv_cand.max()
        sum_a = a_candidate.sum()          # should be 1.0 up to floating‑point noise
        score = 2.0 * n * max_b / (sum_a ** 2)

        # Keep the best sequence seen so far
        if np.isfinite(score) and score < best_score:
            best = a_candidate.copy()
            best_score = score
            no_improve = 0
            print(f"[improve] iter={iteration:07d}, alpha={alpha:.1f}, score={best_score:.6f}")
        else:
            no_improve += 1

        # Early exit when the target has been reached
        if best_score <= target_score:
            print("[target] reached desired bound")
            break

        if no_improve >= 200000:
            a = np.full(n, 1.0 / n)
            no_improve = 0
            print(f"[restart] iter={iteration}, re‑initialising to uniform")
            continue

        # Move to next iterate
        a = a_candidate

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive – fall back to the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w

    evaluation score.  The method is a projected gradient descent on a
    smooth (log‑sum‑exp) approximation of the maximal autocorrelation entry.
    Convolutions are performed via FFT, giving an O(n log n) per‑iteration
    cost.  An annealing schedule gradually sharpens the smooth‑max,
    occasional multiplicative perturbations help escape plateaus, and the
    best sequence encountered is returned.
    """

    # algorithmic hyper‑parameters
    alpha_start = 5.0          # temperature for the smooth‑max (low ⇒ smooth)
    alpha_end = 150.0          # final temperature (sharp)
    ramp_iters = 50000         # iterations over which we increase alpha
