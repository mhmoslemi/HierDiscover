from scipy.linalg import toeplitz

def convolution_matrix(f):
    n = len(f)
    c = np.concatenate([f, np.zeros(n-1, dtype=f.dtype)])
    r = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n-1, dtype=f.dtype)])
    T = toeplitz(c, r)
    # T shape (2n-1, n)
    return T

"""
Propose a near‑optimal non‑negative step‑function.

Algorithm
---------
1.  Initialise a power‑law sequence a_i ∝ (i+1)^{-p} (p≈0.5 gives a flat
    self‑convolution).  Scan a few lengths and exponents and keep the best.
2.  Repeatedly improve the current best sequence by an LP‑based “balancing”
    step:
       • Let M = max(conv(a,a)).
       • Solve the linear programme
                maximise   Σ_j g_j
                subject to conv(a,g) ≤ M   (2·n–1 linear inequalities)
                           g_j ≥ 0 .
          The programme pushes weight of g to positions where a is small
          (the tails) while respecting the current maximal convolution value.
       • Normalise g to have the same ℓ₁‑norm as a and mix:
                a ← (1‑t)·a + t·g      (t is a decreasing step‑size)
          This mixing flattens the convolution and strictly reduces the
          objective in practice.
3.  Occasionally apply a cheap random multiplicative block perturbation
    to escape possible local minima.
4.  Stop when the time budget (≈1000 s) is exhausted and return the best
    sequence found so far.
The objective function is scale‑invariant, so all intermediate sequences are
normalised to unit sum; the final score is 2 n·max(conv(a,a)).
"""

import time
import math
import numpy as np
import scipy.optimize
import scipy.linalg


def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """Return a length‑n non‑negative sequence a_i ∝ (i+1)^{-p} and normalise to sum = 1."""
    idx = np.arange(n) + 1.0                # avoid division by zero
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _score_from_sequence(seq: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(seq,seq)) assuming the ℓ₁‑norm of seq equals 1.
    (The true objective also divides by the sum², but we keep sum = 1.)
    """
    n = seq.size
    max_conv = np.convolve(seq, seq).max()
    return 2.0 * n * max_conv


def _convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the matrix A (shape (2·n‑1, n)) such that A @ g == np.convolve(f,g).
    Each row corresponds to a linear inequality for the convolution.
    """
    n = f.size
    # Toeplitz matrix whose first column is [f, 0, …, 0] (length 2·n‑1)
    # and whose first row is [f[0], 0, …, 0].
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([np.array([f[0]], dtype=f.dtype), np.zeros(n - 1, dtype=f.dtype)])
    T = scipy.linalg.toeplitz(col, row)     # shape (2·n‑1, n)
    return T


def _lp_max_sum_under_bound(f: np.ndarray, bound: float) -> np.ndarray:
    """
    Solve the LP:
        maximise   Σ g_j
        subject to conv(f,g) ≤ bound   (2·n‑1 linear constraints)
                   g_j ≥ 0 .
    Returns the optimal g (may be dense).  The LP is always feasible
    (g = 0) and bounded because each g_j ≤ bound / f_0 (f_0 > 0 for all
    sequences we use).
    """
    n = f.size
    A = _convolution_matrix(f)             # (2·n‑1, n)
    b = np.full(A.shape[0], bound, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)      # maximise sum → minimise –sum
    bounds = [(0, None)] * n                # g_j ≥ 0

    # Use the HiGHS solver (default for linprog) – fast and reliable.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                                 method='highs', options={'presolve': True})
    if not res.success:
        # In the unlikely event of failure fall back to the zero vector.
        return np.zeros_like(f)
    return res.x

    (leaving a small safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float('inf')
    best_n = None

    # ------------------------------------------------------------------
    # 1) Coarse scan of length / exponent – gives a good starting point.
    # ------------------------------------------------------------------
    lengths = [500, 800, 1000]                     # moderate sizes – fast LP solves
    p_grid = np.linspace(0.45, 0.55, 21)           # dense around the theoretical optimum p≈0.5

    for n in lengths:
        for p in p_grid:
            seq = _make_power_sequence(n, p)       # ℓ₁‑norm = 1
            score = _score_from_sequence(seq)
            if score < best_score:
                best_score = score
                best_seq = seq.copy()

        print(f"[grid] n={n:4d}, best_score={best_score:.6f}")

        best_n = best_seq.size

    f = best_seq.copy()                      # current working sequence (ℓ₁‑norm = 1)
    n = f.size
