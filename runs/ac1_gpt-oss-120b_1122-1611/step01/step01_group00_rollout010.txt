"""
Hybrid Adam‑based sub‑gradient descent for the max‑convolution objective.

The algorithm works on a probability simplex (non‑negative entries summing to 1)
so that the denominator of the evaluation function is constant.
At each iteration it

1. computes the self‑convolution,
2. finds the index of the current worst (maximum) entry,
3. builds the exact sub‑gradient of that entry  (∂/∂a_i = 2·a_{k−i}),
4. updates the vector with an Adam optimiser,
5. projects back onto the simplex.

To avoid the uniform distribution being a stationary point, occasional
multiplicative “shakes’’ are applied and, after long periods without
improvement, the algorithm restarts from a fresh Dirichlet sample.
The best sequence seen during the run is returned.
"""

import time
import numpy as np


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a step‑function height sequence that minimises

        2 * n * max_k (a * a)_k / (sum_i a_i)^2

    using an Adam‑optimised sub‑gradient descent with random restarts.
    The returned list is guaranteed to satisfy the security checks of
    `evaluate_sequence`.
    """
    rng = np.random.default_rng(seed)

    # time budget (a small margin is left for the final return)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ------------------------------------------------------------------
    # choose a length that balances expressiveness and speed
    n = 2048

    # ------------------------------------------------------------------
    # initialise from the previous best if it exists, otherwise uniform
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=float)
        a0 = np.zeros(n, dtype=float)
        if prev.size <= n:
            a0[: prev.size] = prev
        else:
            a0[:] = prev[:n]
        a0 = _project_simplex(a0)
    else:
        a0 = np.full(n, 1.0 / n, dtype=float)

    # start the optimisation from this point
    a_curr = a0.copy()

    # Adam state
    m = np.zeros_like(a_curr)
    v = np.zeros_like(a_curr)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    base_lr = 0.23          # base learning rate
    # sub‑gradient is taken on the exact max (no smoothing)
    # ------------------------------------------------------------------
    # bookkeeping
    conv = np.convolve(a_curr, a_curr)
    best_score = 2.0 * n * conv.max()    # sum(a)=1 ⇒ denominator = 1
    best_seq = a_curr.copy()
    total_iter = 0
    last_improve = 0
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        total_iter += 1

        # --------------------------------------------------------------
        # 1) convolution and worst index
        conv = np.convolve(a_curr, a_curr)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # --------------------------------------------------------------
        # 2) sub‑gradient of that entry:
        #    grad_i = 2 * a_{max_idx-i} for all i where the term exists
        grad = np.zeros_like(a_curr)
        i_min = max(0, max_idx - (n - 1))
        i_max = min(n - 1, max_idx)
        i_range = np.arange(i_min, i_max + 1, dtype=int)
        j_range = max_idx - i_range               # complementary index
        grad[i_range] = 2.0 * a_curr[j_range]

        # --------------------------------------------------------------
        # 3) Adam update (bias‑corrected first and second moments)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        # bias‑correction factors
        m_hat = m / (1.0 - beta1 ** total_iter)
        v_hat = v / (1.0 - beta2 ** total_iter)

        # step size shrinks like 1/√t to guarantee convergence
        step = (base_lr / np.sqrt(total_iter)) * (m_hat / (np.sqrt(v_hat) + eps))

        a_next = a_curr - step
        a_next = _project_simplex(a_next)

        # --------------------------------------------------------------
        # 4) occasional random “shake’’ to break symmetry
        if rng.random() < 0.015:
            idx = rng.integers(0, n)
            factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal factor
            a_next[idx] *= factor
            a_next = _project_simplex(a_next)

        # --------------------------------------------------------------
        # 5) local evaluation (no need for the heavy evaluate_sequence)
        conv_next = np.convolve(a_next, a_next)
        cur_score = 2.0 * n * conv_next.max()   # sum(a_next)=1 by construction

        # keep the best ever seen
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a_next.copy()
            last_improve = total_iter
            print(f"[improve] iter={total_iter}, score={best_score:.6f}")

        # --------------------------------------------------------------
        # 6) move forward
        a_curr = a_next

        # --------------------------------------------------------------
        # restart if we haven’t improved for a long while
        if total_iter - last_improve > 250_000:
            # fresh Dirichlet start (α=0.5 gives a mildly sparse start)
            a_curr = rng.dirichlet(np.full(n, 0.5))
            a_curr = _project_simplex(a_curr)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = total_iter
            print("[restart] random Dirichlet start")

        # --------------------------------------------------------------
        # early exit when we are already under the target margin
        if best_score <= 1.5025 * 1.001:
            print("[early exit] target reached")
            break

    print(f"[final] best score={best_score:.6f} after {total_iter} iterations")
    return [float(x) for x in best_seq]
