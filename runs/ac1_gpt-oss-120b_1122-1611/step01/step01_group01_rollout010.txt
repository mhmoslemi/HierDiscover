import time
import numpy as np

def _fft_len_for_n(n):
    size = 2*n - 1
    return 1 << ((size - 1).bit_length())

def _conv_self(a, fft_len):
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:2*a.size-1]
    return conv

def _conv_fft(x, y):
    size = x.size + y.size - 1
    fft_len = 1 << ((size - 1).bit_length())
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)[:size]
    return z

def _random_block_perturb(a, rng, max_width_frac=0.1, sigma=0.15):
    n = a.size
    max_width = max(2, int(n * max_width_frac))
    width = rng.integers(1, max_width)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start+width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """Smooth‑max exponentiated‑gradient optimizer for the step‑function problem.
    
    The algorithm minimises a smoothed version of 
        score = 2 * n * max(a*a) / (sum a)**2
    by repeatedly:
      * computing the self‑convolution,
      * forming a softmax over the convolution entries (parameter β),
      * back‑propagating the softmax as a gradient of the smooth max,
      * applying an exponentiated‑gradient update on the log‑weights,
      * occasionally applying a random block perturbation to escape plateaus.
    The procedure runs for roughly ``budget_s`` seconds and returns the best
    sequence seen.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # ----------------------------------------------------------------------
    # initialise a from the provided global if possible
    # ----------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, None)

    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total  # unit sum

    # pre‑alloc best
    best = a.copy()
    conv_best = _conv_self(best, _fft_len_for_n(n))
    best_score = 2.0 * n * np.max(conv_best)

    # ----------------------------------------------------------------------
    # hyper‑parameters
    # ----------------------------------------------------------------------
    ff_len_self = _fft_len_for_n(n)               # for a*a
    # gradient convolution length (a_rev length n, s length 2n‑1)
    ff_len_grad = 1 << ((3 * n - 2 - 1).bit_length())

    beta_start = 0.5
    beta_end = 30.0
    eta_start = 0.8
    eta_end = 0.07
    perturb_prob = 0.02
    epsilon = 1e-12

    start_time = time.monotonic()

    iteration = 0
    while time.monotonic() < deadline:
        # time‐dependent schedules
        elapsed = time.monotonic() - start_time
        total_time = deadline - start_time
        # linear schedules over the whole run
        beta = beta_start + (beta_end - beta_start) * (elapsed / total_time)
        eta = eta_start + (eta_end - eta_start) * (elapsed / total_time)

        # --------------------------------------------------------------
        # 1. self‑convolution
        # --------------------------------------------------------------
        conv = _conv_self(a, ff_len_self)
        max_b = np.max(conv)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 2. smooth‑max weights (softmax over convolution)
        # --------------------------------------------------------------
        # stabilised softmax
        b_shift = conv - np.max(conv)                # subtract max for numeric stability
        exp_beta = np.exp(beta * b_shift)
        soft = exp_beta / np.sum(exp_beta)

        # --------------------------------------------------------------
        # 3. gradient w.r.t. a  (g_i = sum_j a_j * soft_{i+j})
        # --------------------------------------------------------------
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, soft)          # length 3n‑2
        # slice the part that corresponds to i = 0 … n‑1
        g = grad_full[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # 4. exponentiated‑gradient step on the log‑variables
        # --------------------------------------------------------------
        # multiplicative update a_i ← a_i * exp( - η * g_i )
        factor = np.exp(-eta * g)
        a = a * factor
        # renormalise to keep sum = 1, avoid division by zero
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # --------------------------------------------------------------
        # 5. occasional random block perturbation
        # --------------------------------------------------------------
        if rng.random() < perturb_prob:
            a = _random_block_perturb(a, rng)

        iteration += 1

    # ----------------------------------------------------------------------
    # return the best solution found (as a plain Python list)
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

"""
Smooth‑max + multi‑peak exponentiated‑gradient optimizer.

The objective is
    score = 2 * n * max( a * a ) / (sum a)^2 .
After normalising sum(a)=1 the task reduces to minimising
    M = max_k (a * a)[k] .
The algorithm repeatedly

1. computes the full self‑convolution (FFT, O(n log n));
2. builds a soft‑max over the convolution entries (temperature β);
3. back‑propagates the soft‑max to obtain a smooth gradient
   g_i = Σ_j a_j · soft_{i+j};
4. adds the partner‑vectors of the T largest convolution peaks
   (sub‑gradient of the active constraints);
5. forms a combined gradient, scales it by n (so that the typical
   magnitude is O(1)), and performs an exponentiated‑gradient update
   a_i ← a_i·exp(‑η·gradient_i·n);
6. renormalises a to keep Σa_i = 1;
7. occasionally applies a random block perturbation to escape plateaus;
8. keeps the best sequence seen so far.

All hyper‑parameters (β, η, T, perturb probability) are linearly
scheduled during the allotted time.  The routine stops just before the
budget expires and returns the best sequence found as plain Python
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic convolution utilities (FFT based, linear convolution)
# ----------------------------------------------------------------------
def _fft_len_for_n(n: int) -> int:
    """Next power‑of‑two length ≥ 2·n‑1 (full linear convolution)."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using an FFT."""
    size = x.size + y.size - 1
    fft_len = 1 << ((size - 1).bit_length())
    fx = np.fft.rfft(x, n=fft_len)
    fy = np.fft.rfft(y, n=fft_len)
    fz = fx * fy
    z = np.fft.irfft(fz, n=fft_len)[:size]
    return z


# ----------------------------------------------------------------------
# Partner vector for a single active convolution index
# ----------------------------------------------------------------------
def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of (a*a)[k] w.r.t. ``a``.
    Returns w_i = a_{k-i} (0 where the index is out of range).
    """
    n = a.size
    w = np.zeros_like(a)

    if k < n:
        # indices i = 0 … k
        w[: k + 1] = a[k :: -1]
    else:
        # indices i = k-(n-1) … n-1
        start = k - (n - 1)
        length = (2 * n - 1) - k
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


# ----------------------------------------------------------------------
# Cheap multiplicative block perturbation (helps to escape local minima)
# ----------------------------------------------------------------------
def _random_block_perturb(a: np.ndarray,
                         rng: np.random.Generator,
                         max_width_frac: float = 0.1,
                         sigma: float = 0.15) -> np.ndarray:
    n = a.size
    max_width = max(2, int(n * max_width_frac))
    width = rng.integers(1, max_width)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine (the entry point used by the harness)
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that minimises the supplied
    ``evaluate_sequence`` score.  The optimizer runs for roughly
    ``budget_s`` seconds (leaving a small safety margin) and combines a
    smooth‑max gradient with the partner‑vectors of the T largest
    convolution peaks.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise the candidate (fallback to a modest uniform vector)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(256, dtype=np.float64)

    n = base.size
    a = np.clip(base, 0.0, None)

    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total                               # unit sum → scale‑invariant

    # best solution seen so far
    best = a.copy()
    conv_best = _conv_self(best, _fft_len_for_n(n))
    best_score = 2.0 * n * np.max(conv_best)

    # ------------------------------------------------------------------
    # hyper‑parameters (linearly scheduled)
    # ------------------------------------------------------------------
    ff_len_self = _fft_len_for_n(n)          # length for a*a
    beta_start, beta_end = 0.5, 80.0         # soft‑max temperature schedule
    eta_start, eta_end = 0.8, 0.04           # exponentiated‑gradient step size
    t_max = 5                                # maximum number of active peaks
    perturb_prob = 0.015                     # probability of a random block perturb
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)

    iteration = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        total_time = deadline - start_time

        # ---- linear schedules -------------------------------------------------
        beta = beta_start + (beta_end - beta_start) * (elapsed / total_time)
        eta = eta_start + (eta_end - eta_start) * (elapsed / total_time)

        # ---- 1️⃣  self‑convolution (score & peak indices) --------------------
        conv = _conv_self(a, ff_len_self)
        max_b = np.max(conv)
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ---- 2️⃣  active‑set partner vectors (T largest peaks) --------------
        # linearly increase T from 1 up to ``t_max``
        T = 1 + int((t_max - 1) * (elapsed / total_time))
        T = max(1, min(t_max, T))

        # indices of the T largest convolution entries
        top_idxs = np.argpartition(-conv, T - 1)[:T]    # unsorted but fine

        # sum of the partner vectors for the chosen peaks
        w_total = np.zeros_like(a)
        for k in top_idxs:
            w_total += _partner_vector(a, int(k))

        # ---- 3️⃣  smooth‑max (soft‑max) gradient ------------------------------
        # stable soft‑max
        shift = conv - np.max(conv)
        exp_beta = np.exp(beta * shift)
        soft = exp_beta / np.sum(exp_beta)

        # gradient g_i = Σ_j a_j·soft_{i+j}
        a_rev = a[::-1]
        grad_full = _conv_fft(a_rev, soft)            # length 3n‑2
        g_smooth = grad_full[n - 1 : n - 1 + n]

        # combine the two gradient contributions (equal weight)
        grad = w_total + g_smooth

        # ---- 4️⃣  exponentiated‑gradient update (scaled by n) ----------------
        # scaling by n makes the typical magnitude O(1)
        factor = np.exp(-eta * grad * n)
        a = a * factor

        # renormalise to keep Σa = 1
        s = a.sum()
        if s == 0.0:
            a[:] = 1.0 / n
        else:
            a /= s

        # ---- 5️⃣  occasional stochastic block perturbation -------------------
        if rng.random() < perturb_prob:
            a = _random_block_perturb(a, rng)

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence found (as plain Python floats)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
