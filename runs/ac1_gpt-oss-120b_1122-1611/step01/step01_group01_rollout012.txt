"""
Smoothed multiplicative‑weights optimizer for the max‑self‑convolution problem.

The score to minimise is
    score = 2·n·max(a*a) / (sum(a)²)
with a_i ≥ 0.  The objective is scale‑invariant, so we keep the vector
normalised to unit sum.  For a given a we form the linear self‑convolution
b = a*a (via an FFT).  Instead of moving only against the single maximal
index (as in the reference code) we minimise the smooth approximation
    F_ρ(a) = (1/ρ)·log Σ_k exp(ρ·b_k) .
Its gradient is a weighted cross‑correlation
    ∇F_ρ(a)_i = Σ_k p_k·a_{k-i},
where p is the softmax of the convolution entries.
Both the convolution and the cross‑correlation are evaluated with FFTs,
giving an O(n log n) iteration.  The step is a mirror‑descent /
exponentiated‑gradient update
    a ← a ⊙ exp(−η·∇F_ρ(a)),
followed by renormalisation.  The temperature ρ is increased
linearly during the run (starting with a smooth objective, ending with a
sharp approximation of the max) while the learning rate η decays.
Occasional random block perturbations help to escape shallow local minima.
The best vector seen so far (according to the true `evaluate_sequence`
score) is returned.

The algorithm runs for the allotted budget (default ≈1100 s, leaving a
2 s safety margin) and respects all constraints of `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a (real, non‑negative) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa                     # element‑wise product (self‑convolution)
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _cross_corr(p: np.ndarray, a_rev: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation of p with a_rev.
    Returns the full correlation of length len(p)+len(a_rev)-1.
    """
    fp = np.fft.rfft(p, n=fft_len)
    fa = np.fft.rfft(a_rev, n=fft_len)
    corr = np.fft.irfft(fp * fa, n=fft_len)[: len(p) + len(a_rev) - 1]
    return corr


# ----------------------------------------------------------------------
# Random block perturbation (same as in the reference solution)
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------

    """
    Return a list of non‑negative heights that (approximately) minimise
    `evaluate_sequence`.  The routine runs for roughly ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # initialise the candidate (use the provided global if possible)
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        base = np.ones(512, dtype=np.float64)  # reasonable fallback size

    n = base.size
    a = np.clip(base, 0.0, None)
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= total                     # unit L1 norm (scale‑invariant)

    # pre‑compute FFT lengths (powers of two) for the two convolutions we need
    fft_len_self = _next_pow2(2 * n - 1)       # for a*a
    fft_len_corr = _next_pow2(3 * n - 2)       # for p * a_rev

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    best = a.copy()
    conv_best = _conv_self(best, fft_len_self)
    best_score = 2.0 * n * np.max(conv_best)   # same as evaluate_sequence for unit sum
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)   # safety margin

    # schedule parameters (linear in elapsed fraction)
    rho_start, rho_end = 5.0, 60.0      # temperature for the soft‑max
    eta_start, eta_end = 0.6, 0.03    # multiplicative‑weights step size

    # main optimisation loop ------------------------------------------------
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        fraction = elapsed / (deadline - start_time)   # in [0,1]

        # linearly interpolate schedule
        rho = rho_start + (rho_end - rho_start) * fraction
        eta = eta_start + (eta_end - eta_start) * fraction

        # --------------------------------------------------------------
        # 1) self‑convolution and soft‑max weights
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len_self)               # length 2n‑1
        max_val = np.max(conv)

        # compute soft‑max of convolution entries (stabilised)
        shift = max_val                                 # subtraction for numerical stability
        exp_terms = np.exp(rho * (conv - shift))
        p = exp_terms / exp_terms.sum()                # weight vector, sums to 1

        # --------------------------------------------------------------
        # 2) gradient of the smoothed objective (cross‑correlation)
        # --------------------------------------------------------------
        a_rev = a[::-1]                                 # reversed a
        corr_full = _cross_corr(p, a_rev, fft_len_corr) # length 3n‑2
        # gradient_i = corr_full[n-1 + i]  for i=0..n-1
        grad = corr_full[n - 1 : n - 1 + n]

        # --------------------------------------------------------------
        # 3) multiplicative‑weights (mirror‑descent) update
        # --------------------------------------------------------------
        # protect against overflow/under‑flow
        factor = np.exp(-eta * grad)
        a *= factor

        # --------------------------------------------------------------
        # 4) re‑normalise to unit sum and guard against numerical collapse
        # --------------------------------------------------------------
        s = a.sum()
        if s <= 0.0 or not np.isfinite(s):
            a = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            a /= s

        # --------------------------------------------------------------
        # 5) occasional stochastic exploration
        # --------------------------------------------------------------
        if rng.random() < 0.012:            # ~1 % chance per iteration
            a = _random_block_perturb(a, rng)

        # --------------------------------------------------------------
        # 6) evaluate true score (cheap version, identical for unit‑sum a)
        # --------------------------------------------------------------
        conv_now = _conv_self(a, fft_len_self)
        cur_score = 2.0 * n * np.max(conv_now)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # return the best sequence found (plain Python floats, clipped for safety)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
