"""
Propose a near‑optimal step‑function.

Strategy
--------
1.   Start from any existing good sequence (the global
     ``height_sequence_1`` if present) and remember its score.
2.   Perform a *parameter grid search* over the family  
     a_i ∝ (i + 1 + offset)^{-p} .
     The parameters (p, offset, length) are cheap to tweak and
     the family is known to give very flat self‑convolutions.
3.   Refine the best candidate with two complementary local
     moves:
        * a small multiplicative perturbation of a random block,
        * a small random change of the (p, offset) parameters.
     After each move the sequence is renormalised (the objective
     is scale‑invariant) and evaluated with ``evaluate_sequence``.
4.   Keep the best sequence found before the time budget expires.

The code stays well within the 1000 s limit: the grid consists of a
few hundred evaluations (each a single ``np.convolve`` call) and the
local phase uses cheap stochastic updates.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: generate a normalised power‑law sequence with an optional offset.
# ----------------------------------------------------------------------
def _make_power_offset_sequence(n: int, p: float, offset: float) -> np.ndarray:
    """
    Return a non‑negative sequence a_i ∝ (i + 1 + offset)^{-p}  of length n,
    normalised so that its sum equals 1 (the objective is homogeneous).
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    seq = idx ** (-p)
    total = seq.sum()
    if total == 0.0:
        # fall back to a uniform distribution – will be rejected later
        return np.full(n, 1.0 / n, dtype=np.float64)
    seq /= total
    # ``evaluate_sequence`` later clips to [0,1000]; our values are already safe.
    return seq


# ----------------------------------------------------------------------
# Main entry point required by the harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of heights that (hopefully) minimises ``evaluate_sequence``.
    The routine spends at most *budget_s* seconds searching and returns the
    best sequence discovered.
    """
    rng = np.random.default_rng(seed)

    # give a small safety margin for the outer wrapper
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 0) initialise from a previously found construction, if any.
    # ------------------------------------------------------------------
    best_seq = None
    best_score = float('inf')

    try:
        # ``height_sequence_1`` is injected by the evaluation environment
        init = np.array(height_sequence_1, dtype=np.float64)
        score = float(evaluate_sequence(init.tolist()))
        best_seq = init.copy()
        best_score = score
        print(f"[init] score of provided sequence = {best_score:.6f}")
    except Exception:
        # no useful global start – will be replaced later
        pass

    # ------------------------------------------------------------------
    # 1) Coarse grid search over length, exponent p and offset.
    #    Values are deliberately modest to keep the total runtime tiny.
    # ------------------------------------------------------------------
    # Pick a few sensible lengths (the previous best used 2000)
    lengths = [2000]                     # primary length
    # If we still have time we will also try a longer one later.
    p_grid = np.arange(0.45, 0.71, 0.015)    # 0.45 … 0.70
    offset_grid = np.arange(0, 221, 20)      # 0 … 200

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for p in p_grid:
            for off in offset_grid:
                if time.monotonic() >= deadline:
                    break
                seq = _make_power_offset_sequence(n, float(p), float(off))
                # fast conversion to python list for the evaluator
                score = float(evaluate_sequence(seq.tolist()))
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"[grid] n={n} p={p:.3f} off={off:>3} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Optional longer length if time permits – the asymptotic
    #    constant improves a little for larger n.
    # ------------------------------------------------------------------
    if time.monotonic() < deadline:
        for n in (4000, 6000):
            if time.monotonic() >= deadline:
                break
            for p in p_grid:
                for off in offset_grid:
                    if time.monotonic() >= deadline:
                        break
                    seq = _make_power_offset_sequence(n, float(p), float(off))
                    score = float(evaluate_sequence(seq.tolist()))
                    if score < best_score:
                        best_score = score
                        best_seq = seq.copy()
                        print(f"[grid‑big] n={n} p={p:.3f} off={off:>3} → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fine local optimisation.
    #    Two complementary mutation operators are used:
    #      * block‑wise multiplicative tweak (as in the reference code)
    #      * small random walk in the (p, offset) space.
    # ------------------------------------------------------------------
    # Keep a copy of the parameters that generated the current best
    # sequence – they will be updated when a better candidate appears.
    if best_seq is not None:
        cur_n = best_seq.size
        # crude reverse‑engineer current parameters if possible
        # (use the same formula with a tiny numerical guard)
        cur_p = 0.6
        cur_off = 0.0
    else:
        # fallback – start from a uniform sequence of length 2000
        cur_n = 2000
        cur_p = 0.6
        cur_off = 0.0
        best_seq = np.full(cur_n, 1.0 / cur_n, dtype=np.float64)
        best_score = float(evaluate_sequence(best_seq.tolist()))

    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # a) small block perturbation on the actual array
        # ------------------------------------------------------------------
        cand = best_seq.copy()
        n = cand.size
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 30))
        stop = min(n, start + width)
        factor = math.exp(rng.normal(0.0, 0.07))
        cand[start:stop] *= factor

        # ------------------------------------------------------------------
        # b) occasional global shape mutation (p / offset walk)
        # ------------------------------------------------------------------
        if rng.random() < 0.25:         # 25 % of the iterations

            cur_p = np.clip(cur_p + rng.normal(0.0, 0.008), 0.40, 0.80)
            cur_off = np.clip(cur_off + rng.normal(0.0, 5.0), 0.0, 500.0)
            cand = _make_power_offset_sequence(n, cur_p, cur_off)

        # renormalise to unit sum (scale invariance)
        total = cand.sum()
        if total <= 0.0:

        cand /= total

        # evaluate – the security wrapper will clip values that violate
        # the hard limits; our candidates stay well inside those bounds.
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[improve] new best = {best_score:.6f} (n={n})")

    # ------------------------------------------------------------------
    # 4) final safety net
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64)
        best_seq /= best_seq.sum()

    # Return plain Python floats as required by the harness
    return [float(v) for v in best_seq]
