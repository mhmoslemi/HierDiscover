"""
Optimiser for the self‑convolution bound.

The programme builds the classic Catalan sequence a_i = C(2i,i)/4**i,
which is (asymptotically) optimal for the problem
   minimise  sup (a*a)  with fixed ℓ¹‑norm.
Its evaluation converges to π/2 ≈ 1.5708.
In order to try to push the bound a little further we perform a cheap
gradient‑descent / simulated‑annealing phase that tweaks the coefficients
while keeping the ℓ¹‑norm fixed (the evaluation is invariant under a
global scaling, only the shape matters).  Two complementary perturbation
schemes are used:

* block_perturbation – a random contiguous block is multiplied by a
  log‑normal factor;
* gradient_step       – the sub‑gradient of the current maximum convolution
  entry is moved in the opposite direction.

Both operations are followed by a renormalisation to the target sum
sqrt(2·n) (the same normalisation used for the Catalan construction) and
by clipping to the interval [0,1000] – exactly the preprocessing performed
by ``evaluate_sequence``.  Any improvement is kept, and the loop runs
until the allotted time budget (≈ 995 s) is exhausted.

The result is returned as a plain Python list of floats and is guaranteed
to satisfy all security checks of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be top‑level for the evaluation harness.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    First ``length`` terms of a_i = binom(2*i,i) / 4**i.
    The recurrence a_{i+1} = a_i * (2*i+1)/(2*i+2) is stable
    and produces values with the exact constant self‑convolution
    property a*a == 1 on the interior of the support.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        # a_{i} = a_{i-1} * (2*i-1) / (2*i)
        seq[i] = seq[i - 1] * (2 * i - 1) / (2 * i)
    return seq


def normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale ``seq`` so that its ℓ¹‑norm equals ``target_sum``.
    The scaling factor is positive, therefore the non‑negativity of the
    sequence is preserved.
    """
    current = np.sum(seq)
    if current == 0.0:
        return seq
    factor = target_sum / current
    return seq * factor


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a cheap O(n) perturbation:
    - choose a random contiguous block,
    - multiply the block by a log‑normal factor,
    - occasionally mirror the whole sequence (symmetrisation).
    Afterwards the sequence is clipped to the allowed range.
    """
    cand = base.copy()
    n = cand.size
    # block width between n//200 and n//12 (inclusive)
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    # occasional averaging with the reversed version – keeps the shape smooth
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def gradient_step(seq: np.ndarray, rng: np.random.Generator,
                  step_scale: float = 0.02) -> np.ndarray:
    """
    One projected gradient step on the current maximal convolution entry.
    The sub‑gradient of  max_k (seq*seq)_k  w.r.t. ``seq`` at an index
    where the maximum is attained is  2 * shifted_seq .
    The step moves opposite to that direction, a small step size is used,
    and the result is clipped and renormalised.
    """
    n = seq.size
    # convolution (full length 2n-1)
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # pick a single index attaining the maximum (the first one)
    k = int(np.argmax(conv))

    # build the gradient: g[i] = 2*seq[k-i] if 0<=k-i<n else 0
    grad = np.zeros_like(seq)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    # i runs over indices that actually appear in the sum
    # For those i, the partner index is j = k - i
    # Grad contribution is 2*seq[j]
    idx = np.arange(lo, hi + 1, dtype=int)
    partner = k - idx
    grad[idx] = 2.0 * seq[partner]

    # step – we move *against* the gradient (to lower the max entry)
    step = -step_scale * grad
    cand = seq + step
    np.clip(cand, 0.0, 1000.0, out=cand)

    # renormalise to the standard target ℓ¹‑norm  sqrt(2·n)
    target = np.sqrt(2.0 * n)
    cand = normalise_to_target_sum(cand, target)
    return cand

def propose_candidate(seed: int = 12345, budget_s: float = 1100, **kwargs):
    """
    Build a Catalan‑type sequence and improve it with a lightweight
    stochastic optimisation loop (random block moves and gradient steps).
    The final sequence is returned as a plain list of Python floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Length selection:  if a previous construction is available use
    # its length, otherwise fall back to a moderately large value.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 0:
                raise ValueError
        except Exception:
            n = 8192          # a comfortable size for the time budget
    else:
        n = 8192

    # --------------------------------------------------------------
    # Baseline Catalan construction – the agreed asymptotically optimal
    # starting point.  It is normalised to the standard sum √(2·n).
    # --------------------------------------------------------------
    base = catalan_sequence(n)
    target_sum = np.sqrt(2.0 * n)
    base = normalise_to_target_sum(base, target_sum)

    try:
        best_score = float(evaluate_sequence(base.tolist()))
    except Exception:                     # safety net – never expected
        best_score = float('inf')
    best_seq = base.copy()
    print(f"[init] n={n}, initial evaluation={best_score:.9f}")

    # --------------------------------------------------------------
    # Stochastic improvement loop.
    # We keep trying two kinds of perturbations and accept any
    # change that improves the evaluation.  The loop respects the
    # total time budget, leaving a few seconds margin for a clean exit.
    # --------------------------------------------------------------
    stop_time = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    while time.monotonic() < stop_time:
        iteration += 1
        # 50 % block perturbation, 50 % gradient step
        if rng.random() < 0.5:
            cand = block_perturbation(best_seq, rng)
        else:
            # the step scale is adapted slowly with the iteration count
            scale = 0.03 * (0.95 ** (iteration // 200))
            cand = gradient_step(best_seq, rng, step_scale=scale)

        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            continue

        if np.isfinite(score) and score < best_score:
            best_seq = cand
            best_score = score
            print(f"[improve #{iteration}] new best = {best_score:.9f}")

        # occasional restart from the untouched Catalan baseline
        if iteration % 1500 == 0 and rng.random() < 0.15:
            cand = normalise_to_target_sum(
                block_perturbation(base, rng), target_sum)
            try:
                cand_score = float(evaluate_sequence(cand.tolist()))
            except Exception:
                continue
            if cand_score < best_score:
                best_seq = cand
                best_score = cand_score
                print(f"[restart #{iteration}] recovered better = {best_score:.9f}")

    print(f"[final] best evaluation={best_score:.9f} after {iteration} iterations")
    # Convert to plain Python list – required by the harness.
    return [float(x) for x in best_seq]
