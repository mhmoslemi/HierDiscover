def compute_grad(seq, active_idx):
    n = seq.shape[0]
    grad = np.zeros(n, dtype=np.float64)
    for k in active_idx:
        # compute contributions for i where j = k-i valid
        # i range max(0, k-n+1) <= i <= min(n-1, k)
        start = max(0, k - (n-1))
        end = min(k, n-1)
        # these i indices correspond to j = k - i
        # vectorized: i_range = np.arange(start, end+1); grad[i_range] += seq[k - i_range]
        i_range = np.arange(start, end+1)

def gradient_move(seq):
    n = len(seq)
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    active = np.flatnonzero(conv == max_b)
    grad = compute_grad(seq, active)
    # Use greedy transfer
    i_max = int(np.argmax(grad))
    i_min = int(np.argmin(grad))
    if i_max == i_min:
        return seq.copy()
    # compute delta = 0.05 * seq[i_max] (5% of mass)
    delta = min(seq[i_max] * 0.1, (seq[i_max] - 0))
    if delta <= 1e-12:
        return seq.copy()
    new_seq = seq.copy()
    new_seq[i_max] -= delta
    new_seq[i_min] += delta
    # Clip.
    np.clip(new_seq, 0.0, 1000.0, out=new_seq)
    return new_seq

    idx = np.arange(n)

The evaluation metric is  

\[
\text{score}(a)=\frac{2\,n\;\max\limits_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a\in\mathbb R_{\ge 0}^n .
\]

Because the score is **scale‑invariant** we may (and do) keep the sum of the
sequence equal to 1 during the whole optimisation – the ratio only depends
on the shape of the vector.

The search consists of three phases  

1. **Rich parametric sweep** – families such as exponential decay,
   power‑exponential, Gaussian, logistic, Hann, sine, polynomial and
   inverse‑power are sampled with many random parameters.  
   The best of those gives a strong initial point.

2. **Fast deterministic evaluation** – a local copy of the original
   `evaluate_sequence` is implemented with NumPy only, avoiding the
   Python‑level conversion overhead and allowing millions of evaluations
   inside the time budget.

3. **Stochastic refinement** – a simulated‑annealing loop that repeatedly
   mutates the current candidate with five complementary operators  

   * block‑wise multiplicative perturbation,  
   * element‑wise log‑normal jitter,  
   * simple value swap,  
   * a targeted *gradient move* that transfers mass from indices that
     contribute most to the current maximal convolution value to those that
     contribute least, and  
   * a small random transfer between two positions.

   The temperature (derived from the remaining time) controls the
   step‑size of the operators, and a Metropolis acceptance rule lets the
   algorithm escape shallow local minima.  Occasionally the current walk is
   re‑started from a fresh parametric candidate.

All moves keep the vector non‑negative, clip to the allowed `[0,1000]`
range and re‑normalise the sum to 1 before evaluation, preserving the
scale‑invariant score.

The routine prints progress information and returns the best sequence
found before the deadline (≈ 1000 s).  It works with the optional global
`height_sequence_1` if it is present, otherwise it falls back to a uniform
start of length 4806.

```python
"""
Propose a step‑function sequence with a low evaluation score.

The algorithm combines a dense parametric sweep (exponential, Gaussian,
logistic, Hann, sine, polynomial, inverse‑power families) with a
simulated‑annealing refinement that uses several mutation operators:
  * block‑wise multiplicative scaling,
  * per‑element log‑normal jitter,
  * random swaps,
  * a *gradient move* that shifts mass from indices that most increase the
    current maximum convolution entry to those that increase it least,
  * a tiny random transfer between two positions.

The score is scale‑invariant, therefore every candidate is re‑normalised
to unit ℓ₁‑norm before evaluation.  A pure NumPy implementation of the
evaluation function is used for speed.  The loop runs until the allocated
budget (≈ 1000 s) expires and returns the best sequence discovered.
"""

import math
import time
from typing import List, Tuple

import numpy as np

# ----------------------------------------------------------------------
# Fast local copy of `evaluate_sequence` (no Python list conversion)
# ----------------------------------------------------------------------
def _fast_evaluate(seq: np.ndarray) -> Tuple[float, np.ndarray]:
    """Return (score, convolution) for a non‑negative NumPy vector."""
    # Clip to the limits imposed by the original evaluator
    np.clip(seq, 0.0, 1000.0, out=seq)
    total = seq.sum()
    if total < 0.01:
        return float("inf"), np.empty(0, dtype=np.float64)

    conv = np.convolve(seq, seq)
    max_b = conv.max()
    n = seq.size
    score = 2.0 * n * max_b / (total * total)
    return float(score), conv


# ----------------------------------------------------------------------
# Utility helpers used by the stochastic refinements
# ----------------------------------------------------------------------
def _active_indices(conv: np.ndarray) -> np.ndarray:
    """Indices where the convolution attains its maximum."""
    max_val = conv.max()
    return np.flatnonzero(conv == max_val)


def _gradient(seq: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of `max(conv)` w.r.t. the sequence.
    For each active index k, d(conv_k)/d a_i = a_{k-i}.
    """
    n = seq.size
    grad = np.zeros_like(seq)
    for k in active:
        start = max(0, k - (n - 1))
        end = min(k, n - 1)
        i_range = np.arange(start, end + 1, dtype=np.int64)
        grad[i_range] += seq[k - i_range]
    return grad


def _block_perturb(seq: np.ndarray, rng: np.random.Generator,
                   sigma: float = 0.12) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = seq.size
    # block width between 0.5 % and 5 % of the whole vector
    w_min = max(1, int(n * 0.005))
    w_max = max(2, int(n * 0.05))
    width = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, n - width + 1))
    cand = seq.copy()
    factor = float(np.exp(rng.normal(0.0, sigma)))
    cand[start:start + width] *= factor
    # occasional mild symmetry – often helps the shape
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _jitter_move(seq: np.ndarray, rng: np.random.Generator,
                 sigma: float = 0.02) -> np.ndarray:
    """Element‑wise log‑normal jitter."""
    cand = seq * np.exp(rng.normal(0.0, sigma, size=seq.shape))
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _swap_move(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Swap two randomly chosen entries (preserves the ℓ₁‑norm)."""
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    cand = seq.copy()
    cand[i], cand[j] = cand[j], cand[i]
    return cand


def _transfer_move(seq: np.ndarray, rng: np.random.Generator,
                   frac: float = 0.05) -> np.ndarray:
    """Move a random fraction of mass from one position to another."""
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    if seq[i] <= 0.0:
        return seq.copy()
    delta = seq[i] * frac * rng.random()
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def _gradient_move(seq: np.ndarray, rng: np.random.Generator,
                   step_factor: float = 0.10) -> np.ndarray:
    """
    Transfer a small amount of mass from the index that contributes most
    to the current maximal convolution entry to the one that contributes
    least.
    """
    # compute active set once (needed for the gradient)
    conv = np.convolve(seq, seq)
    active = _active_indices(conv)
    grad = _gradient(seq, active)

    # all gradients equal → nothing to improve
    if np.allclose(grad, grad[0]):
        return seq.copy()

    i_max = int(np.argmax(grad))
    i_min = int(np.argmin(grad))

    delta = min(seq[i_max] * step_factor, seq[i_max] * 0.5)
    if delta <= 0.0:
        return seq.copy()

    cand = seq.copy()
    cand[i_max] -= delta
    cand[i_min] += delta
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

            frac = 0.07 * temperature + 0.01
            cand_seq = _transfer_move(parent_seq, rng, frac=frac)

        # normalise ℓ₁‑norm again (preserves the scale‑invariant score)
        tot = cand_seq.sum()
        if tot > 0:
            cand_seq = cand_seq / tot
        # evaluate the candidate
        cand_score, _ = _fast_evaluate(cand_seq)

        # update the global best if needed
        maybe_update(cand_seq, cand_score)

        # simulated‑annealing acceptance for the current walk
        if cand_score < parent_score:
            parent_seq, parent_score = cand_seq, cand_score
        else:
            # probability to accept a worse move goes to zero as time runs out
            prob = math.exp(-(cand_score - parent_score) / (0.01 + temperature))
            if rng.random() < prob:
                parent_seq, parent_score = cand_seq, cand_score

        # gentle drift of the walk towards the (ever improving) global best
        parent_seq = 0.9 * parent_seq + 0.1 * best_seq
        np.clip(parent_seq, 0.0, 1000.0, out=parent_seq)
        parent_seq = parent_seq / parent_seq.sum()  # keep unit sum

        # occasional full restart from a fresh parametric candidate
        if rng.random() < 0.005:
            restart = rng.choice(_parametric_candidates(rng, n, 30))
            restart = restart / restart.sum()
            parent_seq, parent_score = restart, _fast_evaluate(restart)[0]

    # ------------------------------------------------------------------
    # Return the best sequence discovered (still within the allowed range)
    # ------------------------------------------------------------------
    print(f"Search completed – best score {best_score:.6f}")
    final_seq = np.clip(best_seq, 0.0, 1000.0)
    return final_seq.tolist()
```
