Define:

def projection_simplex(v):
    """Project onto simplex {x >= 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]  # descending
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    # find rho = max i where condition holds
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

"""Propose a near‑optimal step‑function.

The algorithm proceeds in three phases:
1. **Initialisation** – a small grid over length, power‑law exponent `p` and
   offset `b` is scanned.  The best shape found (or the provided
   `height_sequence_1`) becomes the starting point.
2. **Targeted gradient descent** – repeatedly identify the shift `k` that
   attains the maximal self‑convolution, compute the gradient of that
   maximum w.r.t. the coefficients, take a short step in the opposite
   direction and project back onto the probability simplex.  The step size
   is randomly varied; occasional symmetry enforcement and block
   multiplicative perturbations are added to escape shallow local minima.
3. **Return the best sequence** seen before the time budget expires.
The search is completely self‑contained and uses only NumPy and the
`evaluate_sequence` function provided by the judge.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, sum x_i = 1}.
    """
    # works for any shape, but we only use 1‑d vectors
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    # Find the last index where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety: renormalise to exactly 1
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _make_power_sequence(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """
    Return a normalised power‑law sequence of length ``n``:
        a_i ∝ (i + 1 + offset)^{-p}
    The result is scaled so that the sum equals 1 (the evaluation is
    scale‑invariant, but a unit sum simplifies the projection step).
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -p, dtype=np.float64)
    total = seq.sum()
    if total <= 0.0:
        # fallback to uniform
        seq = np.ones(n, dtype=np.float64) / n
    else:
        seq /= total
    return seq


def _gradient_step(seq: np.ndarray, max_shift: int, step: float) -> np.ndarray:
    """
    Perform a single gradient step that tries to reduce the value of the
    convolution at ``max_shift``.  ``step`` is a positive scalar.  The
    output is projected onto the simplex.
    """
    n = len(seq)
    grad = np.zeros_like(seq)

    # contribution from a_i * a_{i+max_shift}
    if max_shift < n:
        grad[: n - max_shift] += seq[max_shift:]

    # contribution from a_{i-max_shift} * a_i
    if max_shift > 0:
        grad[max_shift:] += seq[: n - max_shift]

    # simple descent step (note: the gradient points *up*, we subtract)
    new_seq = seq - step * grad
    # enforce non‑negativity before the simplex projection
    new_seq = np.maximum(new_seq, 0.0)
    # a simplex projection restores the unit sum
    new_seq = _project_to_simplex(new_seq)
    return new_seq


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a sequence that minimises ``evaluate_sequence``.
    The routine respects the time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)  # safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 0) Load the previously discovered construction if it exists.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():

                sc = float(evaluate_sequence(cand.tolist()))
                best_seq = cand.copy()
                best_score = sc
                print(f"[init] loaded height_sequence_1 with score {best_score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1) Coarse grid search over length / exponent / offset
    # ------------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400]  # a few moderate sizes
    p_grid = np.linspace(0.55, 0.78, 46)      # step ≈0.005
    offsets = [0.0, 0.5, 1.0, 2.0, 4.0]      # simple offsets, all non‑negative

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        for p in p_grid:
            if time.monotonic() >= deadline:
                break
            for off in offsets:
                seq = _make_power_sequence(n, p, off)
                # Clip to the limits that evaluate_sequence will enforce.
                seq = np.clip(seq, 0.0, 1000.0)
                score = float(evaluate_sequence(seq.tolist()))
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"[grid] n={n}, p={p:.4f}, off={off:.1f} -> score={best_score:.6f}")
                if time.monotonic() >= deadline:
                    break

    # ------------------------------------------------------------------
    # 2) Local optimisation loop – a blend of gradient steps,
    #    symmetrical averaging and random block perturbations.
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # occasionally restart from a freshly sampled power‑law candidate
        if best_seq is None or rng.random() < 0.02:
            n = int(rng.integers(800, 2500))
            p = float(rng.uniform(0.55, 0.78))
            off = float(rng.exponential(1.0))
            seq = _make_power_sequence(n, p, off)
            seq = np.clip(seq, 0.0, 1000.0)
            score = float(evaluate_sequence(seq.tolist()))
            if score < best_score:
                best_score, best_seq = score, seq.copy()
                print(f"[restart] new best {best_score:.6f} (n={n})")
            continue

        # ------------------------------------------------------------------
        # a) Gradient descent directed at the current worst convolution index
        # ------------------------------------------------------------------
        cand = best_seq.copy()
        n = cand.size
        conv = np.convolve(cand, cand)
        max_idx = int(conv.argmax())
        # Random step size – scaled roughly to the magnitude of a single entry.
        step = float(rng.uniform(0.002, 0.02))
        cand_g = _gradient_step(cand, max_idx, step)

        # Occasionally enforce symmetry (helps for the flatness of the convolution)
        if rng.random() < 0.25:
            cand_g = 0.5 * (cand_g + cand_g[::-1])

        # Clip to the admissible range before evaluation
        cand_g = np.clip(cand_g, 0.0, 1000.0)

        score_g = float(evaluate_sequence(cand_g.tolist()))
        if score_g < best_score:
            best_score, best_seq = score_g, cand_g.copy()
            print(f"[grad] improved to {best_score:.6f} (n={n}, shift={max_idx})")
            continue   # we already improved – go to next iteration

        # ------------------------------------------------------------------
        # b) Random block multiplicative perturbation (helps to escape plateaus)
        # ------------------------------------------------------------------
        width = int(rng.integers(1, max(2, n // 15)))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width
        factor = float(math.exp(rng.normal(0.0, 0.07)))
        cand2 = best_seq.copy()
        cand2[start:stop] *= factor
        # Re‑normalise onto the simplex
        cand2 = _project_to_simplex(cand2)
        if rng.random() < 0.2:
            cand2 = 0.5 * (cand2 + cand2[::-1])
        cand2 = np.clip(cand2, 0.0, 1000.0)
        score2 = float(evaluate_sequence(cand2.tolist()))
        if score2 < best_score:
            best_score, best_seq = score2, cand2.copy()
            print(f"[block] improved to {best_score:.6f} (n={n})")

        # ------------------------------------------------------------------
        # c) Tiny random jitter on the whole vector (global exploration)
        # ------------------------------------------------------------------
        if rng.random() < 0.05:
            jitter = rng.normal(0.0, 0.003, size=n)
            cand3 = best_seq + jitter
            cand3 = np.maximum(cand3, 0.0)
            cand3 = _project_to_simplex(cand3)
            if rng.random() < 0.25:
                cand3 = 0.5 * (cand3 + cand3[::-1])
            cand3 = np.clip(cand3, 0.0, 1000.0)
            score3 = float(evaluate_sequence(cand3.tolist()))
            if score3 < best_score:
                best_score, best_seq = score3, cand3.copy()
                print(f"[jitter] improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Fallback – uniform distribution if nothing succeeded.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[fallback] uniform score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return plain Python floats as required by the outer harness.
    # ------------------------------------------------------------------
    return [float(x) for x in best_seq]

"""Propose a near‑optimal step‑function.

The algorithm works in three phases:

1. **Initial grid** – scan a modest set of lengths, power‑law exponents
   `p` and offsets `b` (a_i ∝ (i+1+b)^{-p}) and keep the best shape
   (or the provided `height_sequence_1`).

2. **Targeted descent** – repeatedly locate the shift `k` that yields the
   maximal self‑convolution, compute the gradient of that entry with respect
   to the coefficients, take a short step opposite to the gradient and
   project the result back onto the probability simplex.  Random symmetry
   and block‑multiplicative perturbations are added to avoid stagnation.

3. **Return** the best sequence seen before the allotted time expires.
   The routine respects the 1000 s budget and uses only NumPy and the
   already‑imported `evaluate_sequence` function.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x_i >= 0, sum_i x_i = 1}.
    """
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise (numerical safety)
    total = w.sum()
    if total == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= total
    return w


def _make_power_sequence(n: int, p: float, offset: float = 0.0) -> np.ndarray:
    """
    Normalised power‑law sequence of length ``n``:
        a_i ∝ (i + 1 + offset)^{-p}.
    The result is scaled so that the sum equals 1.
    """
    idx = np.arange(n, dtype=np.float64) + 1.0 + offset
    seq = np.power(idx, -p, dtype=np.float64)
    total = seq.sum()
    if total <= 0.0:
        seq = np.ones(n, dtype=np.float64) / n
    else:
        seq /= total
    return seq


def _gradient_step(seq: np.ndarray, k: int, step: float) -> np.ndarray:
    """
    One gradient step that tries to reduce the value of the convolution at
    index ``k``.  ``step`` is a positive scalar.  The result is projected onto
    the simplex.
    """
    n = len(seq)
    grad = np.zeros_like(seq)

    # contribution from a_i * a_{i+k}
    if k < n:
        grad[: n - k] += seq[k:]

    # contribution from a_{i-k} * a_i
    if k > 0:
        grad[k:] += seq[: n - k]

    new_seq = seq - step * grad
    new_seq = np.maximum(new_seq, 0.0)
    new_seq = _project_to_simplex(new_seq)
    return new_seq

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a coefficient vector that minimises ``evaluate_sequence``.
    The function respects the supplied time budget and returns the best
    sequence found (as a plain list of Python floats).
    """
    rng = np.random.default_rng(seed)
    # give a small safety margin for the surrounding harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 0) Use the previously discovered construction if it exists.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0:
                sc = float(evaluate_sequence(cand.tolist()))
                best_seq = cand.copy()
                best_score = sc
                print(f"[init] loaded height_sequence_1 → score {best_score:.6f}")
        except Exception:
            pass

    # --------------------------------------------------------------
    # 1) Coarse grid over length / exponent / offset
    # --------------------------------------------------------------
    lengths = [800, 1200, 1600, 2000, 2400]          # modest sizes, fast to eval
    p_grid = np.linspace(0.55, 0.78, 46)            # step ≈0.005
    offsets = [0.0, 0.5, 1.0, 2.0, 4.0]             # simple non‑negative shifts

    for n in lengths:

            break

                if sc < best_score:

                    best_seq = seq.copy()

    while time.monotonic() < deadline:

        # ----------------------------------------------------------
        # b) Random block multiplicative perturbation (global escape)
        # ----------------------------------------------------------
        width = int(rng.integers(1, max(2, n // 15)))
        start = int(rng.integers(0, n - width + 1))
        stop = start + width
        factor = float(math.exp(rng.normal(0.0, 0.07)))
        cand_b = best_seq.copy()
        cand_b[start:stop] *= factor
        cand_b = _project_to_simplex(cand_b)
        if rng.random() < 0.2:
            cand_b = 0.5 * (cand_b + cand_b[::-1])
        cand_b = np.clip(cand_b, 0.0, 1000.0)
        sc_b = float(evaluate_sequence(cand_b.tolist()))
        if sc_b < best_score:
            best_score, best_seq = sc_b, cand_b.copy()
            print(f"[block] improved → {best_score:.6f} (n={n})")
            continue

        # ----------------------------------------------------------
        # c) Tiny global jitter (adds stochasticity)
        # ----------------------------------------------------------
        if rng.random() < 0.05:
            jitter = rng.normal(0.0, 0.003, size=n)
            cand_j = best_seq + jitter
            cand_j = np.maximum(cand_j, 0.0)
            cand_j = _project_to_simplex(cand_j)
            if rng.random() < 0.25:
                cand_j = 0.5 * (cand_j + cand_j[::-1])
            cand_j = np.clip(cand_j, 0.0, 1000.0)
            sc_j = float(evaluate_sequence(cand_j.tolist()))
            if sc_j < best_score:
                best_score, best_seq = sc_j, cand_j.copy()
                print(f"[jitter] improved → {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Fallback – uniform distribution if nothing was found
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64) / 256.0
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print(f"[fallback] uniform → score {best_score:.6f}")

    # Return as plain Python floats as required by the surrounding harness
    return [float(x) for x in best_seq]
