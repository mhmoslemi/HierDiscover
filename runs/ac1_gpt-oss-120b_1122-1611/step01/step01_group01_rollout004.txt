def proj_simplex(v):
    # given vector and want projection onto {x >=0, sum=1}
    # from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

"""
A smoothed‐gradient search for the optimal non‑negative step‑function.

The score for a vector `a` (with sum(a)=1) is
    score = 2 * n * max(conv(a,a))

where `conv` is the linear self‑convolution.  The algorithm builds a
smooth approximation of the `max` using a soft‑max (log‑sum‑exp)
with a temperature β that grows from an initial small value to a
large one.  At each iteration we:

* compute the self‑convolution (FFT, O(n log n))
* form the soft‑max weights `w` over the convolution entries
* obtain the gradient of the smoothed objective:
      ∇ = 2 * correlation(w, a)   (valid mode, length n)
  the correlation is evaluated by a single FFT‑based convolution.
* take a projected gradient step onto the probability simplex
  (the set {a_i ≥ 0, sum a_i = 1}).
* occasionally apply a cheap random block perturbation to escape
  shallow local minima.

Both the step size η and the temperature β are scheduled linearly
with the elapsed time, starting aggressive and becoming more
conservative later on.  The best vector seen so far is retained and
returned at the end of the time budget.

The method is deterministic given a seed, runs with a single
process/thread, uses only NumPy and the standard library and respects
the required 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the next power of two >= x."""
    return 1 << (x - 1).bit_length()


def _fft_len_for_min_len(min_len: int) -> int:
    """Length of an FFT that can hold a linear convolution of size `min_len`."""
    return _next_pow2(min_len)


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum_i x_i = 1}
    Implementation of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # find the largest rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative perturbation of a random contiguous block.
    The factor is drawn from a log‑normal distribution.
    The resulting vector is re‑normalised onto the simplex.
    """
    n = a.size
    # block width between 1 and roughly n/10 (minimum 1)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    # project back onto simplex
    return _proj_simplex(out)


def _self_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a real vector using a real FFT.
    Returns the first (2*len(a)-1) entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the (valid) correlation of w with a:
        g_i = sum_j a_j * w_{i+j},  i = 0 … n-1   (n = len(a))

    This is equivalent to the convolution of rev(w) with a and taking the
    slice (len(w)-1) … (len(w)-1 - n + 1) in reverse order.
    """
    # reverse w (the `rev` in the derivation)
    w_rev = w[::-1]
    # FFTs with the same length
    fw = np.fft.rfft(w_rev, n=fft_len)
    fa = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(fw * fa, n=fft_len)
    # valid length = len(w_rev) + len(a) - 1 = 3*n - 2
    # index of the centre element (len(w_rev)-1) = 2*n - 2
    centre = len(w_rev) - 1
    # extract gradient entries (reversed order)
    idx = centre - np.arange(a.size)
    grad = 2.0 * conv_full[idx]   # factor 2 from ∂(a*a)/∂a
    return grad

        # 2. gradient of the smoothed objective (2 * n * log‑sum‑exp)
        # --------------------------------------------------------------
        grad = _correlation_fft(w, a, fft_len)              # length n

        # --------------------------------------------------------------
        # 3. projected gradient step onto the simplex
        # --------------------------------------------------------------
        a_new = a - eta * grad
        a_new = _proj_simplex(a_new)

        # --------------------------------------------------------------
        # 4. occasional stochastic exploration
        # --------------------------------------------------------------
        if rng.random() < 0.015:   # ~1.5 % chance per iteration
            a_new = _random_block_perturb(a_new, rng)

        a = a_new

        # --------------------------------------------------------------
        # 5. evaluate (exact) score and keep the best
        # --------------------------------------------------------------
        conv = _self_convolution_fft(a, fft_len)
        cur_score = 2.0 * n * np.max(conv)

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the
    # evaluation function).  Convert to plain Python floats.
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
