def _softmax(x, alpha):
    z = x * alpha
    max_z = np.max(z)
    exp_z = np.exp(z - max_z)  # stable
    sum_exp = np.exp(-max_z) * np.sum(exp_z)  # careful: Actually we shifted: exp_z = exp(z - max_z). So sum_exp = np.sum(exp_z)
    # Since we used shift, normalization = sum_exp
    return exp_z / sum_exp

target 1.5025 within the 1000‑second budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Next power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _fft_len_for_n(n: int) -> int:
    """FFT length sufficient for a linear convolution of two length‑n vectors."""
    return _next_pow2(2 * n - 1)


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a real, non‑negative vector `a`
    using an FFT of length `fft_len`.
    Returns the full convolution of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Soft‑max of the convolution values.
    Returns λ_k = exp(α·conv_k)/∑exp(α·conv_k) as a probability vector.
    """
    scaled = conv * alpha
    # numerical stability
    max_val = np.max(scaled)
    exp_vals = np.exp(scaled - max_val)
    sum_exp = np.sum(exp_vals)
    return exp_vals / sum_exp


def _gradient(a: np.ndarray, lam: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """
    Gradient of L = Σ λ_k·(a*a)_k  w.r.t. `a`.
    Uses the identity ∂L/∂a_i = 2·Σ_k λ_k·a_{k‑i}.
    Implements this as a convolution λ * a_rev.
    """
    a_rev = a[::-1]
    # convolution λ (length 2n‑1) with a_rev (length n)
    fl = np.fft.rfft(lam, n=fft_len_grad)
    fr = np.fft.rfft(a_rev, n=fft_len_grad)
    full = np.fft.irfft(fl * fr, n=fft_len_grad)[: len(lam) + len(a_rev) - 1]
    offset = a.size - 1
    grad = 2.0 * full[offset : offset + a.size]
    return grad


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative perturbation of a random contiguous block.
    The factor is log‑normally distributed; the vector is renormalised
    afterwards.  Guarantees strict positivity.
    """
    n = a.size
    width = int(rng.integers(1, max(2, n // 10)))
    start = int(rng.integers(0, n - width + 1))
    factor = float(np.exp(rng.normal(0.0, 0.15)))  # moderate change
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 1e-15, None)               # keep positivity
    out /= out.sum()
    return out


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (approximately) minimise
    the provided `evaluate_sequence` score.  The routine runs for roughly
    ``budget_s`` seconds (a small safety margin is kept) and uses a
    mirror‑descent optimiser with a smooth max‑convolution surrogate.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Initialise the candidate vector
    # --------------------------------------------------------------
    try:
        # `height_sequence_1` is expected to be a global list of floats.
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # fallback: uniform vector of moderate length
        base = np.ones(256, dtype=np.float64)

    # enforce strict positivity (required for the log‑style update)
    base = np.clip(base, 1e-12, None)
    base_sum = base.sum()
    a = base / base_sum                     # unit sum ⇒ scale‑invariant

    n = a.size
    fft_len_self = _fft_len_for_n(n)
    fft_len_grad = _next_pow2(3 * n - 2)    # length needed for λ * a_rev

    # keep the best vector seen so far
    best = a.copy()
    conv_best = _conv_self(best, fft_len_self)
    best_score = 2.0 * n * np.max(conv_best)

    # --------------------------------------------------------------
    # Optimisation loop
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 2.0)  # 2 s safety margin

    # hyper‑parameters
    eta0 = 0.8                     # initial step size (decays as 1/√t)
    alpha_start = 1.0              # start temperature for soft‑max
    alpha_end = 30.0               # final temperature
    prob_perturb = 0.02            # probability of a random block perturb
    stagnation_restart = 180.0     # seconds without improvement → jitter

    it = 0
    last_improve = start_time

    while time.monotonic() < deadline:
        # ----- evaluate current score -----
        conv = _conv_self(a, fft_len_self)           # linear self‑convolution
        cur_max = float(np.max(conv))
        cur_score = 2.0 * n * cur_max

        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")
            last_improve = time.monotonic()

        # ----- schedule for α (soft‑max temperature) -----
        elapsed = time.monotonic() - start_time
        total_time = deadline - start_time
        frac = min(1.0, elapsed / total_time)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # ----- compute soft‑max weights λ -----
        lam = _softmax_weights(conv, alpha)          # length 2·n‑1, sums to 1

        # ----- gradient of the smooth objective -----
        grad = _gradient(a, lam, fft_len_grad)       # length n

        # ----- exponentiated‑gradient (mirror descent) step -----
        eta = eta0 / np.sqrt(it + 1)                 # decreasing step‑size
        factor = np.exp(-eta * grad)                 # multiplicative update
        # Guard against extreme under‑/overflow
        factor = np.clip(factor, 1e-300, 1e300)
        a = a * factor
        a = np.maximum(a, 1e-15)                     # keep strict positivity
        a /= a.sum()                                 # project back onto simplex

        # ----- occasional stochastic exploration -----
        if rng.random() < prob_perturb:
            a = _random_block_perturb(a, rng)

        # ----- stagnation handling (jitter around the best so far) -----
        if time.monotonic() - last_improve > stagnation_restart:
            # inject multiplicative Gaussian noise around the current best
            noise = rng.normal(0.0, 0.07, size=n)
            a = best * np.exp(noise)
            a = np.maximum(a, 1e-15)
            a /= a.sum()
            last_improve = time.monotonic()  # reset timer

        it += 1

    # --------------------------------------------------------------
    # Return the best vector found (as plain Python floats)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)   # safety for the evaluator
    return [float(v) for v in best]
