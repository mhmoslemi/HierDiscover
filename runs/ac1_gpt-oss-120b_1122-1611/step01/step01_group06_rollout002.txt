    b = np.convolve(a, a)

def project_onto_simplex(v):
    # assumes v is np.ndarray
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Algorithm Overview
------------------

1. **Seed** – Load the globally‑available `height_sequence_1` (if it exists)
   and keep it as an initial candidate.

2. **Parametric sweep** – Search a 3‑parameter family of sequences  

      a_i ∝ (i+1)^(-p) · r^(i)          (p = power‑law exponent, r = exponential tail)

   over a few modest lengths (200‑800).  The best (p, r, n) triple is kept.
   The same (p, r) are then used to build a longer sequence (≈2000) which
   becomes the starting point for the next phase.

3. **Projected sub‑gradient descent** – The objective is  
   `max(conv(a, a))` (the scale‑invariant part of `evaluate_sequence`).  
   For the index `k*` where the convolution attains its maximum,
   the sub‑gradient is

       ∂b_k*/∂a_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, otherwise 0).

   At each iteration we move opposite to this sub‑gradient,
   then project back onto the probability simplex
   (non‑negative entries that sum to one).  A diminishing step size
   `η_t = η0 / sqrt(t+1)` is used.  The loop runs as long as time permits.

4. **Local random refinements** – After the deterministic phase we apply
   cheap random block‑multiplicative perturbations (the technique used in the
   reference solution) together with the usual renormalisation.  Each improvement
   is recorded.

The whole routine stops a few seconds before the given budget and returns the
best sequence found (as a plain list of Python floats).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (must be top‑level – no closures)
# ----------------------------------------------------------------------


def _make_power_exponential_sequence(n: int, p: float, r: float) -> np.ndarray:
    """
    Return a normalised sequence
        a_i ∝ (i+1)^(-p) * r**i   ,   i = 0 .. n‑1
    The sequence is scaled to sum to 1 (the objective is scale‑invariant).
    """
    idx = np.arange(n) + 1.0        # i+1 to avoid zero‑division
    seq = (idx ** (-p)) * (r ** (idx - 1.0))
    total = seq.sum()
    if total == 0.0:
        seq = np.ones(n, dtype=np.float64) / n
    else:
        seq /= total
    return seq


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive; return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]                     # last true index (1‑based)
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution_gradient(a: np.ndarray):
    """
    Compute the sub‑gradient of   max(conv(a, a))
    at the current point a.
    Returns (g, max_val, k_max) where
        g    – gradient vector (same shape as a),
        max_val – the maximal convolution entry,
        k_max   – index where the maximum is attained.
    """
    n = a.shape[0]
    b = np.convolve(a, a)                # full convolution, length 2n‑1
    k = int(np.argmax(b))                # index of the maximal entry
    max_val = b[k]

    # gradient: g_i = 2 * a_{k-i}  if 0 <= k-i < n else 0
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    # set g[lo:hi+1] = 2 * a[k - lo : k - hi - 1 : -1]
    idx = np.arange(lo, hi + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g, max_val, k


def _evaluate(seq: np.ndarray) -> float:
    """
    Thin wrapper around the provided `evaluate_sequence` that accepts a NumPy array.
    Returns a Python float (or np.inf on failure).
    """
    return float(evaluate_sequence(seq.tolist()))

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):

    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The function respects the time budget (seconds) and returns the best
    sequence found as a list of Python floats.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the outer wrapper
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")
    best_config = None          # (n, p, r)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    try:

        if seq0.ndim == 1 and seq0.size > 1:

            if score0 < best_score:
                best_seq = seq0.copy()
                best_score = score0
                print(f"[init] loaded height_sequence_1, score={score0:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # 1) Coarse parametric sweep on short lengths
    # --------------------------------------------------------------
    lengths_small = [200, 400, 800]
    p_grid = np.linspace(0.40, 0.80, 21)           # step ≈0.02
    r_grid = np.linspace(0.94, 0.999, 16)         # exponential tail

    for n in lengths_small:
        for p in p_grid:
            for r in r_grid:
                # time check – abort early if we are out of budget
                if time.monotonic() >= deadline:
                    break
                seq = _make_power_exponential_sequence(n, p, r)
                score = _evaluate(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    best_config = (n, p, r)
                    print(f"[grid] n={n} p={p:.3f} r={r:.3f} → score={score:.6f}")
            if time.monotonic() >= deadline:
                break
        if time.monotonic() >= deadline:
            break

    # --------------------------------------------------------------
    # 2) Upscale the best parametric shape to a longer sequence (if any)
    # --------------------------------------------------------------
    # Target length for the fine‑grained phase (≈2000 – the length that gave
    # the best known result).  If no good parameters were found we fallback
    # to a pure power‑law.
    target_len = 2000
    if best_config is not None:
        _, p_best, r_best = best_config
        cand = _make_power_exponential_sequence(target_len, p_best, r_best)
    else:
        # fallback to a plain power‑law (p≈0.6 is a good start)
        cand = _make_power_exponential_sequence(target_len, 0.60, 1.0)

    cand_score = _evaluate(cand)
    if cand_score < best_score:
        best_score = cand_score
        best_seq = cand.copy()
        print(f"[upscale] length={target_len} p={p_best:.3f} r={r_best:.3f} → score={cand_score:.6f}")

    # --------------------------------------------------------------
    # 3) Projected sub‑gradient descent on max‑convolution
    # --------------------------------------------------------------
    # We run this deterministic phase as long as we have time left.
    # Step size schedule: η_t = η0 / sqrt(t+1) with η0 tuned empirically.
    eta0 = 0.6
    max_iters = 5000          # hard cap; the loop will stop earlier by deadline

    seq = best_seq.copy()
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        # compute sub‑gradient of the max‑convolution entry
        g, _, _ = _max_convolution_gradient(seq)

        # diminishing step size
        eta = eta0 / math.sqrt(it + 1.0)

        # take an (additive) step opposite to the sub‑gradient
        new_seq = seq - eta * g

        # project back onto the simplex to keep non‑negative sum‑one vectors
        new_seq = _project_onto_simplex(new_seq)

        # safety clipping (values >1000 are impossible after projection)
        new_seq = np.clip(new_seq, 0.0, 1000.0)

        new_score = _evaluate(new_seq)
        if new_score < best_score:
            best_score = new_score
            best_seq = new_seq.copy()
            print(f"[subgrad] iter={it:4d} new best={new_score:.6f}")

        # keep the newest point for the next iteration (even if not improving)
        seq = new_seq

    # --------------------------------------------------------------
    # 4) Random local perturbations (same spirit as the reference code)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best_seq.copy()
        n = cand.size

        # ---- block multiplicative tweak ----
        start = rng.integers(0, n)
        # width up to 5 % of n (but at least 1)
        width = rng.integers(1, max(2, n // 20))
        stop = min(n, start + width)
        factor = math.exp(rng.normal(0.0, 0.06))   # log‑normal perturbation
        cand[start:stop] *= factor

        # ---- occasional smoothing (average with reversed copy) ----
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])

        # renormalise to sum‑one (project onto simplex is cheap)
        cand = _project_onto_simplex(cand)

        # evaluate
        score = _evaluate(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[improve] new best={score:.6f} (len={n})")

    # --------------------------------------------------------------
    # 5) Fallback – uniform sequence if everything else failed
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=np.float64) / 256.0
        best_score = _evaluate(best_seq)

    # Return a plain python list (required by the caller)
    return [float(x) for x in best_seq]
