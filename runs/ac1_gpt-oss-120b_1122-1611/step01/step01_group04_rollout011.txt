"""
Search for a step‑function minimizing ``evaluate_sequence``.

The method consists of three stages:

1. **Analytical family sweep** – we generate many candidates from
   simple parametric families (pure power‑law, power‑law with a sine
   envelope, and a short plateau followed by a power‑law tail).  All
   candidates are normalised to unit L¹ norm (the objective is scale‑
   invariant) and the best one is remembered.

2. **Smoothed‑max gradient descent** – the objective
   ``max(conv)`` is approximated by a soft‑max
   ``log Σ exp(α·conv)``.  Its gradient w.r.t. the coefficient vector
   can be written as a correlation of the current vector with the
   soft‑max weights.  Using FFTs we evaluate the convolution,
   the soft‑max weights and the gradient in O(n log n) time.  The
   update ``x ← x – η·∇`` is projected onto the non‑negative simplex
   (clip, then renorm).  The smoothing parameter ``α`` is increased
   gradually so that the optimisation first moves in a smooth landscape
   and later focuses on the true maximum.

3. **Local block perturbations** – after gradient descent we perform a
   large number of cheap multiplicative block tweaks (log‑normal
   factor on a randomly chosen contiguous block).  This helps to
   escape tiny local minima and gives a final polishing.

All steps respect the bounds enforced by ``evaluate_sequence`` and
run within the provided time budget (default 1000 s).  The function
returns a plain Python list of floats ready for the scoring harness.
"""

import time
import math
import numpy as np

# --------------------------------------------------------------
# Helper utilities
# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift) ** beta   (beta negative → decreasing)."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    # clipping is harmless – the shape stays far below the 1000‑limit.
    return np.clip(seq, 0.0, 1000.0)


def generate_power_sine(beta: float, shift: float,
                        gamma: float, n: int) -> np.ndarray:
    """a_i = (i+shift)^beta · sin(π·(i+0.5)/n)^gamma."""
    i = np.arange(n, dtype=np.float64) + shift
    power = i ** beta
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    envelope = np.sin(angles) ** gamma
    seq = power * envelope
    return np.clip(seq, 0.0, 1000.0)


def generate_hybrid(beta: float, shift: float,
                    plateau_frac: float, n: int) -> np.ndarray:
    """
    Simple hybrid: a short constant plateau followed by a power law.
    plateau_frac ∈ [0,1] is the fraction of indices belonging to the plateau.
    """
    p = int(plateau_frac * n)
    if p == 0:
        return generate_power(beta, shift, n)
    tail_len = n - p
    i_tail = np.arange(tail_len, dtype=np.float64) + shift
    tail = i_tail ** beta
    # Choose the plateau height so that the shape is roughly continuous.
    plateau_val = tail[0] if tail_len > 0 else 0.0
    seq = np.concatenate([np.full(p, plateau_val, dtype=np.float64), tail])
    return np.clip(seq, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray,
                            rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of a contiguous block."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5% and 8% of the length
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[st:en] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# --------------------------------------------------------------
# Core optimisation utilities (FFT‑based)
# --------------------------------------------------------------
def _conv_self_fft(x: np.ndarray, fft_len: int) -> np.ndarray:
    """Full convolution of x with itself, length 2·len(x)‑1, using FFT."""
    X = np.fft.rfft(x, n=fft_len)
    conv = np.fft.irfft(X * X, n=fft_len)[: 2 * x.size - 1]
    return conv


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max of the convolution with temperature 1/α."""
    mx = np.max(conv)
    # shift by max to avoid overflow
    exps = np.exp(alpha * (conv - mx))
    total = np.sum(exps)
    if total == 0.0:
        return np.full_like(conv, 1.0 / conv.size)
    return exps / total


def _gradient_fft(x: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of  Σ_i w_i·(x*x)_i  w.r.t. x.
    Implements 2·correlation(x, w) using FFT.
    """
    N = x.size
    M = w.size                     # = 2·N‑1
    X = np.fft.rfft(x, n=fft_len)
    # correlation needs the reversed kernel
    W_rev = np.fft.rfft(w[::-1], n=fft_len)
    corr = np.fft.irfft(X * W_rev, n=fft_len)[: N + M - 1]
    start = M - 1
    grad = 2.0 * corr[start : start - N : -1]      # length N
    return grad


def _score_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact objective value from a pre‑computed convolution."""
    max_b = np.max(conv)
    return float(2.0 * n * max_b / (sum_a * sum_a))

        conv = _conv_self_fft(cand, fft_len)
        score = 2.0 * n * np.max(conv)
        if score < best_score:
            best = cand
            best_score = score
            print(f"refine #{iter_refine} improved score → {score:.6f}")
        iter_refine += 1

    print(f"final best score = {best_score:.6f}")
    # Return a plain Python list (the scorer will re‑normalise anyway)
    return [float(v) for v in best]
