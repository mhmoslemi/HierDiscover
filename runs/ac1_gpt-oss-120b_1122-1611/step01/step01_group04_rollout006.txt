"""
Propose a high‑quality coefficient sequence for the “evaluate_sequence” objective.

Algorithm
---------

1. **Initial fallback** – a uniform vector (always feasible) and,
   if available, the previously found sequence ``height_sequence_1``.
2. **Random parametric search** – construct many candidate sequences
   by multiplying a few elementary building blocks:
      * power‑law  a_i = (i+shift)^β               (β < 0)
      * exponential a_i = exp(-λ·i)
      * sine‑hump  a_i = sin(π·(i+0.5)/n)^γ
   The three blocks are combined multiplicatively, parameters are drawn
   from generous ranges and the result is clipped to the [0,1000] interval.
   Each candidate is evaluated and the best one is remembered.
3. **Targeted block refinement** – after the parametric phase the best
   candidate is iteratively perturbed by scaling a random block up or
   down (log‑normal factor).  The block width is chosen proportionally
   to the sequence length, the modification is accepted only when it
   improves the objective.  This cheap local search removes residual
   peaks in the self‑convolution.
4. **Time management** – the whole routine runs for at most the
   supplied budget (default 1000 s).  Roughly 70 % of the time is spent
   on the parametric exploration, the remainder on the block‑wise
   refinement.  Progress is printed sporadically.

The returned list contains non‑negative floats (≤ 1000) and is the
best sequence discovered before the timeout.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all top‑level, no closures
# ----------------------------------------------------------------------
def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift) ** beta   (beta is negative)."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_exponential(lam: float, n: int) -> np.ndarray:
    """a_i = exp(-lam * i)."""
    i = np.arange(n, dtype=np.float64)
    seq = np.exp(-lam * i)
    return np.clip(seq, 0.0, 1000.0)


def generate_sine(gamma: float, n: int) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n) ** gamma   – symmetric hump."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    return np.clip(seq, 0.0, 1000.0)


def generate_triangle(n: int) -> np.ndarray:
    """Plain symmetric triangular shape – rises then falls linearly."""
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        seq = np.concatenate([half, half[::-1]])
    else:
        seq = np.concatenate([half, half[-2::-1]])
    return np.clip(seq, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative block perturbation (log‑normal factor) of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    # Block width: 0.5 % – 8 % of the sequence length
    w_min = max(1, n // 200)
    w_max = max(2, n // 12)
    w = int(rng.integers(w_min, w_max + 1))
    start = int(rng.integers(0, max(1, n - w + 1)))
    end = min(n, start + w)
    factor = float(np.exp(rng.normal(0.0, 0.25)))  # larger spread than before
    cand[start:end] *= factor
    # occasional symmetrisation (helps for roughly symmetric shapes)
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def evaluate(seq: np.ndarray) -> float:
    """Thin wrapper around the provided ``evaluate_sequence``."""
    return float(evaluate_sequence(seq.tolist()))

    print(f"final best score = {best_score:.6f}")
    # Convert to plain Python floats for the caller.
    return [float(x) for x in best_seq]
