"""
Propose a coefficient sequence that minimises the given evaluation function.

The code performs three stages:

1. **Coarse grid search** over a two‑parameter family
   a_i = (i + shift) ** beta   (beta < 0, shift > 0).
   For each (beta, shift) pair the exact score is evaluated.
   The grid spans beta∈[‑0.95,‑0.45] (step 0.025) and shift∈[0.5,20]
   (geometric steps).  The best pair is kept.

2. **Local stochastic refinement** of (beta, shift) around the current best.
   Small Gaussian perturbations are repeatedly tried while time permits.
   Any improvement updates the best parameters.

3. **Cheap multiplicative block perturbations** of the best numeric sequence.
   Random blocks are multiplied by a log‑normal factor and occasionally
   symmetrised.  These cheap moves allow the optimiser to escape tiny
   local minima.

All helper routines are defined at top level, no closures are used,
and the total run‑time is bounded by the supplied budget (default 1000 s).
The returned list contains plain Python floats and satisfies the
non‑negative, ≤ 1000 constraints required by `evaluate_sequence`.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """
    a_i = (i + shift) ** beta   (beta is negative).
    The result is clipped to the interval [0, 1000] required by
    `evaluate_sequence`.
    """
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Create a cheap O(n) non‑negative perturbation of ``base``.
    A random contiguous block (size ≈ 0.5‑5 % of n) is multiplied
    by a log‑normal factor.  Occasionally the sequence is symmetrised.
    """
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 5 % of the length
    w_min = max(1, n // 200)
    w_max = max(2, n // 20)
    w = int(rng.integers(w_min, w_max + 1))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = st + w
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # ≈10 % multiplicative noise
    cand[st:en] *= factor
    # occasional symmetrisation – often helpful for symmetric shapes
    if rng.random() < 0.33:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the test harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of heights that (hopefully) minimises the evaluation function.
    The method consists of a global grid search over a two‑parameter power‑law
    family, a local stochastic refinement of those parameters and a final phase
    of cheap block‑wise perturbations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 0.5)

    # --------------------------------------------------------------
    # Determine the working length.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        print(f"Using provided sequence length n = {n}")
        # start from the supplied sequence – it is already a valid candidate
        best = np.array(height_sequence_1, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
        print(f"initial provided candidate score = {best_score:.6f}")
    else:
        n = 6000                     # safe default size
        print(f"No external sequence – using default length n = {n}")
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
        print(f"initial uniform score = {best_score:.6f}")

    # placeholders for the parameters that generated the current best
    best_beta = None
    best_shift = None

    # --------------------------------------------------------------
    # 1) Coarse grid search over the (beta, shift) family.
    # --------------------------------------------------------------
    betas = np.arange(-0.95, -0.44, 0.025)          # -0.95 … -0.45 inclusive
    # geometric progression gives finer resolution for small shifts
    shifts = np.geomspace(0.5, 20.0, num=30)

    print(f"Starting grid search: {len(betas)} betas × {len(shifts)} shifts")
    for beta in betas:
        for shift in shifts:
            seq = generate_power(beta, shift, n)
            score = float(evaluate_sequence(seq.tolist()))
            if np.isfinite(score) and score < best_score:
                best = seq
                best_score = score
                best_beta = float(beta)
                best_shift = float(shift)
                print(f"grid best → beta={beta:.3f} shift={shift:.3f} score={score:.6f}")

    # --------------------------------------------------------------
    # 2) Local stochastic refinement of beta / shift.
    # --------------------------------------------------------------
    # If the grid never improved (unlikely) initialise with sensible defaults
    if best_beta is None:
        best_beta = -0.6
        best_shift = 2.0

    it = 0
    while time.monotonic() < deadline:
        # Gaussian proposals centred on the current best parameters
        beta_prop = float(np.clip(best_beta + rng.normal(0.0, 0.02), -0.99, -0.4))
        # shift is perturbed multiplicatively (log‑normal walk)
        shift_prop = float(np.clip(best_shift * np.exp(rng.normal(0.0, 0.10)),
                                   0.5, 200.0))
        seq = generate_power(beta_prop, shift_prop, n)
        score = float(evaluate_sequence(seq.tolist()))
        if np.isfinite(score) and score < best_score:
            best = seq
            best_score = score
            best_beta = beta_prop
            best_shift = shift_prop
            print(f"local refine #{it} → beta={beta_prop:.4f} shift={shift_prop:.4f} score={score:.6f}")
        it += 1

    # --------------------------------------------------------------
    # 3) Cheap multiplicative block perturbations around the best shape.
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"block refine #{iteration} improved score → {score:.6f}")
        iteration += 1

    print(f"final best score = {best_score:.6f}")
    # Return plain Python floats
    return [float(x) for x in best]
