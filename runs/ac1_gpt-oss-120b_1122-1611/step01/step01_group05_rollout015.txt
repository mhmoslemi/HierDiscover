"""
Propose a step‑function that improves the current best bound (≈1.57596).
The algorithm consists of three stages:

1.  Initialise with the analytically‑motivated “arcsine” shape
    a_i = 1/√(i+1) (the known extremal shape for minimizing the
    self‑convolution peak).

2.  Perform a focused stochastic local search.
    *   For the current best sequence we compute the convolution
        and locate the bucket k_max that attains the maximum.
    *   The contributors to this bucket are pairs (i, k_max‑i).
        One of the pair with the largest product is selected and a
        small amount of mass is transferred from it to a random
        position.  The move keeps the total sum (hence the scale)
        unchanged but can lower the peak.
    *   Occasionally a mild smoothing step (3‑point moving average)
        is tried.
    *   Every accepted move strictly reduces the evaluation score.

3.  Return the best sequence discovered as a plain ``list[float]``.
    The whole search stops a few seconds before the time limit so
    that the harness never kills the process.
"""
import time
import numpy as np

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a low‑evaluation step function.

    Returns
    -------
    list[float]
        The best height sequence found within the allotted time.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine a reasonable problem size.
    # Prefer the length of the supplied reference sequence if it exists.
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                n = int(arr.size)
            else:
                n = 2000
        except Exception:
            n = 2000
    else:
        n = 2000                     # a comfortable size for fast convolutions

    # ------------------------------------------------------------------
    # 1) INITIAL GUESS – the classic arcsine (sqrt) decay.
    # a_i = 1/√(i+1)  for i = 0 … n‑1
    idx = np.arange(1, n + 1, dtype=np.float64)
    best = 1.0 / np.sqrt(idx)
    best = np.clip(best, 0.0, 1000.0)    # safeguard (not really needed)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, start_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) LOCAL STOCHASTIC REFINEMENT
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)  # leave a margin
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # ----- compute the current convolution and the maximal bucket -----
        conv = np.convolve(best, best)               # full convolution, length 2n‑1
        k_max = int(conv.argmax())                  # bucket with the current peak
        max_val = conv[k_max]

        # ----- identify the pair (i, j) contributing most to that peak -----
        # i runs over the overlap range of the bucket
        start = max(0, k_max - (n - 1))
        end = min(n - 1, k_max)
        i_range = np.arange(start, end + 1, dtype=np.int64)
        j_range = k_max - i_range                   # partner indices, always in range
        contributions = best[i_range] * best[j_range]

        # Choose a heavily contributing index (biased by its product)
        if contributions.sum() == 0.0:
            i = rng.integers(start, end + 1)
        else:
            probs = contributions / contributions.sum()
            i = rng.choice(i_range, p=probs)

        # Choose a random destination index (different from i)
        j = rng.integers(0, n)
        while j == i:
            j = rng.integers(0, n)

        # ----- propose a tiny mass transfer from i to j -----------------
        # the transfer amount is a small fraction of the source height
        delta = rng.uniform(0.0, 0.1 * best[i])      # up to 10 % of a_i
        candidate = best.copy()
        candidate[i] -= delta
        candidate[j] += delta
        candidate = np.clip(candidate, 0.0, 1000.0)

        # ----- evaluate the candidate -----------------------------------
        score = float(evaluate_sequence(candidate.tolist()))
        if np.isfinite(score) and score < best_score:
            best, best_score = candidate, score
            print(f"[move] iter {iteration:5d} → score {best_score:.6f}")

        # ----- occasional smoothing move (helps to level peaks) -------
        if rng.random() < 0.08:                     # ~8 % of iterations
            kernel = np.array([0.25, 0.5, 0.25], dtype=np.float64)

            smooth = np.clip(smooth, 0.0, 1000.0)
            score_s = float(evaluate_sequence(smooth.tolist()))
            if np.isfinite(score_s) and score_s < best_score:
                best, best_score = smooth, score_s
                print(f"[smooth] iter {iteration:5d} → score {best_score:.6f}")

    # ------------------------------------------------------------------
    print(f"[done] best_score={best_score:.6f} (length={n})")
    return [float(v) for v in best]
