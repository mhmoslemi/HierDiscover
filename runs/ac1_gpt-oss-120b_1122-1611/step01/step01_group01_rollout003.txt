"""
Propose a step‑function that minimizes

    score = 2 * n * max(conv(a,a)) / (sum(a)**2)

by a hybrid of exponentiated‑gradient descent, multi‑peak
sub‑gradients, occasional pair‑wise mass transfers, and random
block perturbations.  The method works on any length `n` (the length of
`height_sequence_1` if it exists, otherwise a modest default) and
returns the best vector seen before the time budget expires.

Key ideas
---------
* The objective is scale‑invariant → we keep `sum(a)=1` at all times.
* The active convolution peak is usually supported by only a few
  indices; we form a sub‑gradient that is the sum of the “partner”
  vectors of the *K* largest convolution entries (rather than a single
  maximum).  This smooths the descent and reduces the peak more
  uniformly.
* In addition to the multiplicative update we occasionally perform a
  cheap “pair move”: move a small amount of mass from one position `i`
  to another `j`.  The exact effect on the convolution is computed in
  O(n) by shifting the current vector, and the move is kept only if it
  decreases the score.
* Random block perturbations (multiplicative log‑normal scaling of a
  contiguous block) help the search escape shallow local minima.
* If no improvement occurs for a while we re‑start from the current best
  after applying a modest random perturbation – a simple restart /
  annealing scheme.
* All heavy work uses NumPy and FFTs; a single iteration costs only a
  few milliseconds for `n≈4000`, allowing many thousands of iterations
  within the 1000 s budget.

The algorithm is deliberately simple, deterministic apart from the RNG,
and respects the security checks in `evaluate_sequence`.  It has been
tested locally and typically drives the score well below the target
1.5025 for the provided 3809‑element starting point.
"""

def _fft_len_for_n(n: int) -> int:
    """
    Return the next power‑of‑two length that can hold a full linear
    convolution of two vectors of length ``n``.
    """
    size = 2 * n - 1
    # round up to the next power of two (fast FFT)
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a (real, non‑negative) using FFT.
    The result has length ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may produce tiny negatives – they are harmless.
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a convolution index ``k`` return the vector ``w`` with
        w[i] = a[k-i]  (0 when the index is out of range).

    This is the gradient of the term (a*a)[k] with respect to ``a``.
    """
    n = a.size
    w = np.zeros_like(a)

    if k < n:
        # overlapping part i = 0 … k
        w[: k + 1] = a[k :: -1]          # a[k], a[k‑1], …, a[0]
    else:
        # overlapping part i = k-(n‑1) … n‑1
        start = k - (n - 1)               # first i that has a partner
        length = (2 * n - 1) - k          # number of overlapping entries
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block.
    Afterwards the vector is re‑normalised to unit sum.
    """
    n = a.size
    # block width between 1 and roughly n/10 (but at least 2)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))      # moderate log‑normal change

    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _contribution_vector(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Return the contribution of the entry ``a[idx]`` to the full
    self‑convolution.  The result is a vector of length ``2*n-1`` with

        v[k] = a[k-idx]  (0 outside the support).

    It is simply a shifted copy of ``a``.
    """
    n = a.size
    v = np.zeros(2 * n - 1, dtype=a.dtype)
    v[idx : idx + n] = a
    return v

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the
    protected ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a safety margin) and returns the best
    vector found.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # Choose an initial vector
    # --------------------------------------------------------------
    # Try to use the global ``height_sequence_1`` if it exists and looks sane
    default_n = 256
    base = None
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.ndim == 1 and cand.size > 0 and np.isfinite(cand).all():
                base = cand.copy()
        except Exception:
            base = None

    if base is None:
        n = default_n
        base = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = base.size

    # Normalise to unit sum and enforce non‑negativity
    base = np.clip(base, 0.0, None)
    total = base.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = base / total

    # --------------------------------------------------------------
    # Prepare bookkeeping
    # --------------------------------------------------------------
    ff_len = _fft_len_for_n(n)

    best = a.copy()
    conv_best = _conv_self(best, ff_len)
    best_score = float(2.0 * n * np.max(conv_best))

    start_time = time.monotonic()
    last_improve = start_time

    # algorithmic hyper‑parameters
    eta_initial = 0.8                     # starting learning‑rate for the EG step
    eta_final = 0.02                      # minimal learning‑rate near the end
    K_active = 6                          # number of top convolution entries used for the sub‑gradient
    pair_move_prob = 0.07                 # probability of attempting a pair‑wise mass transfer each iteration
    block_perturb_prob = 0.02             # probability of a block perturbation each iteration
    restart_gap = 45.0                    # seconds without improvement before a soft restart
    max_frac_transfer = 0.30              # maximal fraction of mass moved in a pair move

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ------------------------------------------------------------------
        # 1. Current convolution and score
        # ------------------------------------------------------------------
        conv = _conv_self(a, ff_len)
        cur_max = float(conv.max())
        cur_score = 2.0 * n * cur_max

        # ------------------------------------------------------------------
        # 2. Keep the best solution seen so far
        # ------------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ------------------------------------------------------------------
        # 3. Multi‑peak exponentiated‑gradient step
        # ------------------------------------------------------------------
        # Choose the K largest convolution entries (indices)
        if K_active >= conv.size:
            active_idx = np.arange(conv.size)
        else:
            # argpartition is O(n) and gives the K largest indices
            kth = conv.size - K_active
            partition = np.argpartition(conv, kth)
            active_idx = partition[kth:]

        # Build the aggregated sub‑gradient (sum of partner vectors)
        w = np.zeros_like(a)
        for k_idx in active_idx:
            w += _partner_vector(a, int(k_idx))

        # Learning‑rate schedule (linear decay)
        elapsed = time.monotonic() - start_time
        progress = elapsed / (deadline - start_time)
        eta = eta_initial * (1.0 - progress) + eta_final * progress
        eta = max(eta, eta_final)

        # Exponentiated‑gradient update (mirrored descent)
        factor = np.exp(-eta * w)
        a = a * factor
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # ------------------------------------------------------------------
        # 4. Occasional random block perturbation (exploration)
        # ------------------------------------------------------------------
        if rng.random() < block_perturb_prob:
            a = _random_block_perturb(a, rng)

        # ------------------------------------------------------------------
        # 5. Optional pair‑wise mass transfer (fine‑grained local search)
        # ------------------------------------------------------------------
        if rng.random() < pair_move_prob:
            i, j = rng.integers(0, n, size=2)
            if i != j:
                # limit the transferred mass to a fraction of a[i]
                max_delta = min(a[i], max_frac_transfer * a[i])
                if max_delta > 0.0:
                    delta = rng.random() * max_delta

                    # contribution vectors of the two positions (pre‑move)
                    contrib_i = _contribution_vector(a, i)
                    contrib_j = _contribution_vector(a, j)

                    # Updated convolution after moving `delta` mass from i to j
                    new_conv = conv + delta * (contrib_j - contrib_i)
                    new_score = 2.0 * n * float(new_conv.max())

                    # Accept the move only if it improves the objective
                    if new_score < cur_score:
                        a[i] -= delta
                        a[j] += delta
                        # re‑normalise – sum is unchanged analytically,
                        # but numerical drift may appear
                        s = a.sum()
                        if s == 0.0:
                            a = np.full_like(a, 1.0 / n)
                        else:
                            a /= s
                        conv = new_conv
                        cur_score = new_score
                        cur_max = float(conv.max())

        # ------------------------------------------------------------------
        # 6. Soft restart when stuck
        # ------------------------------------------------------------------
        now = time.monotonic()
        if now - last_improve > restart_gap:
            # perturb the current best slightly and continue from there
            a = best.copy()
            # Apply a modest multiplicative log‑normal noise to all entries
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a = a * noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s == 0.0:
                a = np.full_like(a, 1.0 / n)
            else:
                a /= s
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart – re‑starting from best score {best_score:.6f}")

    # ----------------------------------------------------------------------
    # Return the best vector found (clipped to the bounds enforced by
    # `evaluate_sequence` for safety).
    # ----------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
