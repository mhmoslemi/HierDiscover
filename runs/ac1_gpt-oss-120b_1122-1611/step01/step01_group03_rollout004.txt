"""
Propose a low‑score step‑function by a simulated‑annealing/gradient search.

The algorithm starts from the known “α‑sequence’’ (or from the global
height_sequence_1 if it exists) and first tries a few simple
“tail‑boost’’ perturbations that increase the L1‑norm while keeping the
self‑convolution almost unchanged.  Afterwards a stochastic search
combines three cheap move types (global Gaussian noise, mass‑transfer
between two positions, and block scaling) with an occasional cheap
gradient step that reduces the current maximal convolution entry.
A linear‑cooling temperature is used so that uphill moves are accepted
early on but the search becomes greedy later.  All operations are kept
on the simplex of non‑negative numbers; the score is evaluated by an
FFT‑based convolution, which makes each iteration fast even for
sequences of several thousand entries.

The function returns the raw (unnormalised) sequence – the evaluation
routine itself normalises by the L1‑norm, so any positive scaling of
the returned vector yields the same score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """
    First n coefficients a_i = C(2i,i)/4**i.
    Recurrence: a_{i+1} = a_i * (2*i+1) / (2*(i+1)).
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1) / (2.0 * i)
    return a


def conv_fft(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a using FFT (real‑to‑real).
    Returns the full convolution of length 2*len(a)-1.
    """
    n = a.shape[0]
    L = 2 * n - 1                     # required length of the linear conv.
    fa = np.fft.rfft(a, L)            # FFT of a padded to length L
    b = np.fft.irfft(fa * fa, L)      # element‑wise product back to time domain
    return b


def raw_score(a: np.ndarray) -> float:
    """
    Compute the objective 2*n*max(conv(a,a))/(sum(a)**2) without the
    extra clipping performed by evaluate_sequence.  This is much faster
    and is identical whenever the sequence stays inside the allowed
    range (0 … 1000) and sum>0.01.
    """
    n = a.shape[0]
    b = conv_fft(a)
    max_b = b.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a coefficient vector that minimises evaluate_sequence.
    The routine runs for roughly `budget_s` seconds and returns the best
    sequence found.  It works completely autonomously – no external
    files or network access are required.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine the length and the initial sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() and isinstance(
        height_sequence_1, (list, np.ndarray)
    ) and len(height_sequence_1) > 0:
        cur = np.array(height_sequence_1, dtype=np.float64)
        n = cur.shape[0]
    else:
        # a reasonably large size – still cheap with FFT based convolution
        n = 8000
        cur = generate_alpha_sequence(n)

    # ------------------------------------------------------------------
    # Small “tail‑boost’’ sweep: a simple parametric family that often
    # increases the L1‑norm while keeping the convolution peak almost
    # unchanged.  The best among these seeds becomes the start point.
    # ------------------------------------------------------------------
    best = cur.copy()
    best_score = raw_score(best)

    lin = np.linspace(0.0, 1.0, n, dtype=np.float64)   # increasing factor
    for lam in np.linspace(0.0, 1.0, 11):              # λ ∈ [0,1]
        cand = cur * (1.0 + lam * lin)                # boost the tail
        cand = np.clip(cand, 0.0, 1000.0)
        s = raw_score(cand)
        if s < best_score:
            best = cand.copy()
            best_score = s
            cur = cand.copy()
            print(f"[init] λ={lam:.3f} improved score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing / gradient descent loop
    # ------------------------------------------------------------------
    cur_score = raw_score(cur)

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # annealing schedule (linear cooling)
    T0 = max(0.01, (best_score - 1.0) * 0.5)   # initial temperature
    total_time = deadline - start_time

    # move‑size parameters – they are slowly cooled
    sigma_noise = 0.20
    block_sigma = 0.12
    max_transfer = 0.5          # up to 50 % of the source mass
    grad_prob = 0.12            # probability of picking a gradient move

    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # Choose a move type
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.35:                     # Gaussian multiplicative noise
            cand = cur * (1.0 + sigma_noise * rng.standard_normal(n))
        elif r < 0.65:                   # mass transfer between two positions
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = cur[i] * rng.uniform(0.0, max_transfer)
            cand = cur.copy()
            cand[i] = max(cand[i] - delta, 0.0)
            cand[j] += delta
        elif r < 0.90:                   # block scaling
            width = int(rng.integers(max(1, n // 200), max(2, n // 12)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, block_sigma)))
            cand = cur.copy()
            cand[start : start + width] *= factor
        else:                            # cheap gradient step (probability grad_prob)
            # compute convolution once
            b = conv_fft(cur)
            max_b = b.max()
            # pick a random index among the active (max) entries
            active = np.where(b >= max_b - 1e-12)[0]
            k = rng.choice(active)
            # sub‑gradient of the active entry (∂b_k/∂a_i = 2 a_{k-i})
            grad = np.zeros_like(cur)
            idx = np.arange(n)
            mask = (k - idx >= 0) & (k - idx < n)
            grad[mask] = 2.0 * cur[k - idx[mask]]
            # step size scaled so that the largest update is modest
            step = 0.03 / (np.max(grad) + 1e-12)
            cand = cur - step * grad

        # --------------------------------------------------------------
        # Project back into the feasible region (non‑negative, bounds)
        # --------------------------------------------------------------
        cand = np.clip(cand, 0.0, 1000.0)          # safety clipping
        if cand.sum() < 1e-8:                     # avoid degenerate vectors
            continue

        cand_score = raw_score(cand)

        # --------------------------------------------------------------
        # Simulated‑annealing acceptance
        # --------------------------------------------------------------
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        else:
            elapsed = time.monotonic() - start_time
            T = T0 * max(0.0, 1.0 - elapsed / total_time)   # linear cooling
            if T > 0 and rng.random() < np.exp(-(cand_score - cur_score) / T):
                cur = cand
                cur_score = cand_score

        # --------------------------------------------------------------
        # Record the best ever seen
        # --------------------------------------------------------------
        if cand_score < best_score:
            best = cand.copy()
            best_score = cand_score
            print(f"[{iteration:6d}] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # Cool the move‑size parameters a little every few thousand steps
        # --------------------------------------------------------------
        if iteration % 2500 == 0:
            sigma_noise *= 0.97
            block_sigma *= 0.97
            grad_prob = min(0.5, grad_prob * 1.02)

    # ------------------------------------------------------------------
    # Final report and return the best sequence (as a plain Python list)
    # ------------------------------------------------------------------
    print(f"[final] iterations={iteration}, best score={best_score:.6f}")
    # Ensure the final vector respects the limits of evaluate_sequence
    best = np.clip(best, 0.0, 1000.0).tolist()
    return best
