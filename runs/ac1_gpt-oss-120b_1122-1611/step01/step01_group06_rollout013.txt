def simplex_projection(v):
    """Project vector v onto the probability simplex (nonnegative, sum=1)."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.where(u + (1 - css) / (np.arange(1, n+1)) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = (css[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a near‑optimal step‑function.

The algorithm starts from a good analytical guess (a power‑law sequence)
and then performs a simple projected sub‑gradient descent that directly
minimises the quantity

    2·n·max(conv(a,a)) / (sum(a))²

subject to a ≥ 0 and sum(a)=1 (simplex).  For a given length n the
gradient of the objective is easy to obtain: if k* is an index where the
self‑convolution attains its maximum, then

    ∂/∂a_i  (max_k conv_k) = 2·a_{k*‑i}   (if 0≤k*‑i<n, else 0).

At each iteration we move a little against this gradient, project the
result back onto the probability simplex and keep the best sequence seen.
Occasional random “jumps’’ (block multiplicative perturbations) and a
simple restart strategy avoid getting stuck in poor local minima.

The method respects the 1000‑second budget, prints progress and finally
returns the best list of heights found.
"""

import time
import math
import numpy as np


def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n) + 1.0
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project a vector onto the probability simplex {x | x_i ≥ 0, Σ x_i = 1}.
    Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpinán (2013).
    """
    if np.allclose(v.sum(), 1) and np.all(v >= 0):
        return v
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    # Find the largest rho satisfying the KKT condition
    rhs = (css - 1) / np.arange(1, n + 1)
    rho = np.where(u - rhs > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (css[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine spends the allotted time searching and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # ----------------------------------------------------------------------
    # 1) Build a small set of reasonable starting points.
    # ----------------------------------------------------------------------
    n = 3000                     # a length that gives a good trade‑off
    start_candidates = []

    # uniform
    start_candidates.append(np.ones(n) / n)

    # power law around the known optimum (p≈0.5)
    for p in (0.48, 0.50, 0.52):
        start_candidates.append(_make_power_sequence(n, p))

    # a few Dirichlet draws (random points on the simplex)
    for _ in range(3):
        start_candidates.append(rng.dirichlet(np.ones(n)))

    # try to reuse the sequence supplied by the host, if it exists
    try:
        # The variable is expected to be a plain Python list of floats.
        global height_sequence_1
        seq0 = np.asarray(height_sequence_1, dtype=float)
        if seq0.size != n:
            # linear interpolation to the required size
            x_old = np.linspace(0.0, 1.0, seq0.size)
            x_new = np.linspace(0.0, 1.0, n)
            seq0 = np.interp(x_new, x_old, seq0)
        seq0 = seq0 / seq0.sum() if seq0.sum() > 0 else np.ones_like(seq0) / len(seq0)
        start_candidates.append(seq0)
    except Exception:
        pass  # the variable is simply not defined – ignore

    # ----------------------------------------------------------------------
    # 2) Evaluate the starters and keep the best.
    # ----------------------------------------------------------------------
    for cand in start_candidates:
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[init] new best_score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # 3) Main optimisation loop – projected sub‑gradient descent + noise.
    # ----------------------------------------------------------------------
    # Parameters for the descent
    step_factor = 0.25            # initial step size (multiplier of the gradient)
    min_step = 1e-4               # never go below this (relative) factor
    decay_interval = 5000         # how often we reduce the step
    decay_rate = 0.90
    no_improve_counter = 0
    stagnation_limit = 20000      # after this many non‑improving steps we restart

    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1

        # --- decay the step size regularly ---
        if iteration % decay_interval == 0:
            step_factor = max(step_factor * decay_rate, min_step)

        # Work on the current best point
        cand = best_seq.copy()
        n = cand.size

        # 1. compute the convolution and locate the worst offset
        conv = np.convolve(cand, cand)
        k_max = int(np.argmax(conv))

        # 2. build the sub‑gradient of max(conv) w.r.t. the vector a
        grad = np.zeros_like(cand)
        # indices i that participate in the k_max‑th convolution term:
        i_lo = max(0, k_max - (n - 1))
        i_hi = min(n - 1, k_max)
        idx = np.arange(i_lo, i_hi + 1, dtype=int)
        jdx = k_max - idx
        grad[idx] = 2.0 * cand[jdx]               # ∂/∂a_i max(conv) = 2·a_{k*‑i}

        # 3. gradient step (move *against* the gradient)
        cand -= step_factor * grad

        # 4. project back onto the simplex – guarantees feasibility
        cand = np.maximum(cand, 0.0)              # non‑negative orthant
        cand = _simplex_projection(cand)          # sum = 1

        # 5. occasional random perturbation to help escape flat plateaus
        if rng.random() < 0.015:
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 30))
            stop = min(n, start + width)
            factor = math.exp(rng.normal(0.0, 0.1))
            cand[start:stop] *= factor
            cand = np.clip(cand, 0.0, 1000.0)
            cand = _simplex_projection(cand)

        # ------------------------------------------------------------------
        # 6. evaluate the candidate
        # ------------------------------------------------------------------
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            no_improve_counter = 0
            print(f"[improve] iter={iteration:,}  new best_score={best_score:.6f}")
        else:
            no_improve_counter += 1

        # ------------------------------------------------------------------
        # 7. if we have been stuck for a long time, restart from a fresh point
        # ------------------------------------------------------------------
        if no_improve_counter > stagnation_limit:
            # draw a new random point on the simplex
            cand = rng.dirichlet(np.ones(n))
            cand = _simplex_projection(cand)
            cand_score = float(evaluate_sequence(cand.tolist()))
            if cand_score < best_score:
                best_score = cand_score
                best_seq = cand.copy()
                print(f"[restart] iter={iteration:,}  improved by restart: {best_score:.6f}")
            else:
                print(f"[restart] iter={iteration:,}  restart did not improve.")
            no_improve_counter = 0

    # ----------------------------------------------------------------------
    # 8) Fallback – should never be needed, but keep a safe guard.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=float) / 256.0

    # Return plain Python floats
    return [float(x) for x in best_seq]
