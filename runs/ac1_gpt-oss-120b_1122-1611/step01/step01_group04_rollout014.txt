"""
Propose a low‑score coefficient sequence.

The search combines three ideas:

1. **Piecewise power‑law with a long low‑valued tail**.
   The head follows a_i = (i+shift)^β (β<0) and after a cut point the
   sequence is filled with a small constant.  This shape makes the
   ℓ₁‑norm huge (thanks to the long tail) while keeping the maximal
   convolution small (determined by the first few head entries).

2. **Random exploration of many parameter families**:
   - pure power‑law
   - power‑law × sine envelope
   - the piecewise shape described above
   - a simple triangular shape
   Random draws of the parameters (β, shift, cut‑fraction, tail‑value,
   sine‑exponent, …) are evaluated; the best candidate is kept.

3. **Local refinement**:
   cheap multiplicative block perturbations (log‑normal factors) are
   applied repeatedly to the current best sequence.  Improvements are
   accepted immediately.

All steps respect the constraints of `evaluate_sequence` (non‑negative,
≤1000, no NaNs).  The routine stops after the given time budget (default
1000 s) and returns the best sequence found.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# Helper generators
# ----------------------------------------------------------------------
def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift)^beta (beta may be negative)."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_sine(gamma: float, n: int) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n)^γ – symmetric hump."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    return np.clip(seq, 0.0, 1000.0)


def generate_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape (1,…,n/2,…,1)."""
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        seq = np.concatenate([half, half[::-1]])
    else:
        seq = np.concatenate([half, half[-2::-1]])
    return np.clip(seq, 0.0, 1000.0)


def generate_piecewise(beta: float,
                       shift: float,
                       cut_frac: float,
                       tail_val: float,
                       n: int) -> np.ndarray:
    """
    Head = power law, Tail = constant.
    cut_frac ∈ (0,1)  – fraction of the sequence that belongs to the head.
    tail_val – value of the long low‑valued tail (must be ≥0).
    """
    cut = max(1, int(cut_frac * n))
    i = np.arange(n, dtype=np.float64) + shift
    head = i[:cut] ** beta
    tail = np.full(n - cut, tail_val, dtype=np.float64)
    seq = np.concatenate([head, tail])
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Local perturbation used in the refinement stage
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of ``base``."""
    cand = base.copy()
    n = cand.size
    # block width: 0.5–5 % of n
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    # log‑normal factor (sometimes <1, sometimes >1)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[st:en] *= factor
    # occasional symmetrisation (helps for symmetric shapes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a sequence of non‑negative heights minimising the
    provided ``evaluate_sequence`` function.  The algorithm runs a
    fast random search over several analytic families (including a
    piecewise power‑law with a long low tail) and then refines the
    best candidate with cheap block‑wise multiplicative tweaks.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Length selection
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 6000                     # a safe default length
    print(f"search length = {n}")

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Fallback: uniform sequence (always valid)
    # ------------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"initial uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Random sampling of analytic families
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        choice = rng.random()
        # ----- pure power‑law -----------------------------------------
        if choice < 0.25:
            beta = rng.uniform(-0.8, -0.2)
            shift = rng.uniform(0.5, 10.0)
            cand = generate_power(beta, shift, n)

        # ----- power‑law × sine envelope -------------------------------
        elif choice < 0.45:
            beta = rng.uniform(-0.8, -0.2)
            shift = rng.uniform(0.5, 10.0)
            gamma = rng.uniform(0.5, 2.5)
            base = generate_power(beta, shift, n)
            env = generate_sine(gamma, n)
            cand = np.clip(base * env, 0.0, 1000.0)

        # ----- piecewise power‑law + constant tail --------------------
        elif choice < 0.75:
            beta = rng.uniform(-0.9, -0.2)
            shift = rng.uniform(0.5, 8.0)
            cut_frac = rng.uniform(0.05, 0.85)          # head size
            # generate head first to obtain a sensible tail bound
            i = np.arange(n, dtype=np.float64) + shift
            head_vals = i ** beta
            cut = max(1, int(cut_frac * n))
            head_last = head_vals[cut - 1]
            # tail is a small fraction of the last head value
            tail_factor = rng.uniform(0.001, 0.45)
            tail_val = head_last * tail_factor
            cand = generate_piecewise(beta, shift, cut_frac, tail_val, n)

        # ----- triangular shape (baseline) -----------------------------
        elif choice < 0.85:
            cand = generate_triangle(n)

        # ----- cheap block perturbation of the current best ------------
        else:
            cand = make_block_perturbation(best, rng)

        # ----------------------------------------------------------------
        # Evaluate candidate
        # ----------------------------------------------------------------
        try:
            score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            # any unexpected error counts as invalid
            score = float('inf')

        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"iter {iteration:06d} new best → {score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 2) Local refinement with block‑wise multiplicative tweaks
    # ------------------------------------------------------------------
    refine_iter = 0
    while time.monotonic() < deadline:
        cand = make_block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"refine #{refine_iter} improved score → {score:.6f}")
        refine_iter += 1

    print(f"final best score = {best_score:.6f}")
    # Return a plain Python list as required by the harness.
    return [float(x) for x in best]
