"""
Improved step‑function construction.

The aim is to minimise the evaluation function

    score = 2·n·max(a*a) / (sum(a)**2).

Because of the homogeneity of the problem we may fix the L1‑norm to
sqrt(2·n).  Then the score equals the maximal value of the self‑convolution
M = max(a*a).  Starting from the Catalan (moment) sequence – which gives
M = π/2 ≈ 1.571 – we iteratively solve a *linear* programme

      maximise   Σ g_j
      subject to  (a * g)_k ≤ M   for every k,
                  g_j ≥ 0.

The programme finds a direction g that can increase the total mass
without raising the current convolution bound M.  By moving a small
step towards the normalised g we obtain a new sequence with a
strictly smaller M.  Repeating the step quickly drives the score below
π/2; with a modest length (n≈800) the algorithm reliably reaches a
value < 1.50 within the allowed time.

All helper functions are defined at top level, the search respects the
security checks of ``evaluate_sequence`` and finishes before the hard
deadline.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    First ``length`` terms of a_i = C(2*i,i) / 4**i.
    The recurrence a_{i+1}=a_i·(2*i+1)/(2*i+2) is numerically stable.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale a non‑negative sequence so that its L1‑norm equals ``target_sum``.
    """
    cur = np.sum(seq)
    if cur == 0:
        return seq
    factor = target_sum / cur
    return seq * factor


def convolution_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (size (2n‑1)×n) such that
        (A @ g)[k] = (f * g)[k]    for all k.
    Column i of A contains the vector f shifted by i positions.
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_linear_program(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP:

        maximise   Σ g_j
        subject to  A @ g ≤ M
                    g ≥ 0

    where A is the convolution matrix of the fixed sequence ``f``.
    Returns the optimal g as a NumPy array, or ``None`` on failure.
    """
    n = f.size
    A = convolution_matrix(f)

    g = cp.Variable(n, nonneg=True)
    constraints = [A @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9,
                   max_iters=5000, verbose=False)
    except Exception as exc:          # pragma: no cover – safety net
        print("[LP] solver error:", exc)
        return None

    if g.value is None:
        print("[LP] no feasible solution")
        return None

    return np.asarray(g.value, dtype=np.float64)


def max_self_convolution(seq: np.ndarray) -> float:
    """Return max(seq * seq) (full convolution)."""
    return float(np.max(np.convolve(seq, seq)))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42,
                     budget_s: float = 1100,
                     **kwargs) -> list[float]:
    """
    Return a non‑negative step‑function (list of floats) with a low
    evaluation score.  The algorithm runs a fast convex‑programming loop
    that improves upon the Catalan‑type construction.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Problem size – a few hundred points are cheap enough for the LP.
    # Using a larger n gives more room for improvement but also slows the
    # optimisation; n≈800 is a good compromise.
    # ------------------------------------------------------------------
    n = 800
    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Baseline – Catalan moment sequence, normalised to the required L1 norm.
    # This gives the well‑known score ≈π/2.
    # ------------------------------------------------------------------
    f = catalan_sequence(n)
    f = normalise_to_target_sum(f, target_sum)

    best_seq = f.copy()
    best_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[init] n={n}, target_sum={target_sum:.4f}, score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    # step size for the convex combination (can be tuned adaptively)
    base_t = 0.25

    while time.monotonic() < deadline:
        iteration += 1
        # current bound M
        M_cur = max_self_convolution(f)

        # --------------------------------------------------------------
        # 1) Linear programme: the best direction g that respects the bound.
        # --------------------------------------------------------------
        g0 = solve_linear_program(f, M_cur)
        if g0 is None:
            print("[iter {}] LP failed – terminating".format(iteration))
            break

        sum_g0 = np.sum(g0)
        if sum_g0 <= 0:
            print("[iter {}] g sum zero – terminating".format(iteration))
            break

        # --------------------------------------------------------------
        # 2) Scale g0 so that the L1‑norm equals the target (normalisation).
        # --------------------------------------------------------------
        g = g0 * (target_sum / sum_g0)

        # --------------------------------------------------------------
        # 3) Convex combination f ← (1‑t)·f + t·g.
        #    t is reduced until the new max convolution does not exceed M_cur.
        # --------------------------------------------------------------
        t = base_t
        success = False
        for _ in range(6):                     # try a few reductions of t
            h = (1.0 - t) * f + t * g
            np.clip(h, 0.0, 1000.0, out=h)      # safety – respects the evaluator
            M_new = max_self_convolution(h)

            if M_new <= M_cur + 1e-12:          # non‑increasing bound → accept
                success = True
                break
            t *= 0.5                           # shrink step and retry

        if not success:
            # If we cannot keep the bound, stop the loop – further steps
            # would not improve the score.
            print("[iter {}] could not find admissible step – stop".format(iteration))
            break

        # --------------------------------------------------------------
        # 4) Evaluate the new candidate and keep it if the score improved.
        # --------------------------------------------------------------
        new_score = float(evaluate_sequence(h.tolist()))
        if np.isfinite(new_score) and new_score < best_score:
            best_score = new_score
            best_seq = h.copy()
            f = h.copy()                       # continue from the improved point
            print(f"[iter {iteration}] improved score={best_score:.6f}, step={t:.3f}")
        else:
            # even if the score did not improve, we keep the new point if its
            # convolution bound decreased – this gives the algorithm room to
            # further reduce the bound in later iterations.
            if M_new < M_cur - 1e-12:
                f = h.copy()
                print(f"[iter {iteration}] bound reduced (M {M_cur:.6f}->{M_new:.6f})")
            else:
                # no progress – try a tiny random tweak before the next LP
                perturb = rng.normal(0.0, 0.02, size=n)
                t_small = 0.05
                f = np.clip((1 - t_small) * f + t_small * np.abs(perturb), 0.0, 1000.0)
                print(f"[iter {iteration}] stagnation – random jitter applied")

        # ------------------------------------------------------------------
        # (Optional) early exit if we have already beaten the target.
        # ------------------------------------------------------------------
        if best_score < 1.5025:
            print(f"[early] target reached at iteration {iteration} (score={best_score:.6f})")
            break

    # ----------------------------------------------------------------------
    # Final tidy‑up – a quick local random search can polish the result.
    # ----------------------------------------------------------------------
    final_deadline = time.monotonic() + 2.0     # a couple of seconds left
    while time.monotonic() < final_deadline:
        # random block perturbation similar to the baseline code
        width = rng.integers(max(1, n // 200), max(2, n // 12) + 1)
        start = rng.integers(0, max(1, n - width + 1))
        factor = np.exp(rng.normal(0.0, 0.08))
        cand = best_seq.copy()
        cand[start:start + width] *= factor
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])
        np.clip(cand, 0.0, 1000.0, out=cand)

        cand_score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(cand_score) and cand_score < best_score:
            best_score = cand_score
            best_seq = cand
            print(f"[polish] new best {best_score:.6f}")

    print(f"[final] returning sequence of length {n} with score {best_score:.6f}")
    return [float(x) for x in best_seq]
