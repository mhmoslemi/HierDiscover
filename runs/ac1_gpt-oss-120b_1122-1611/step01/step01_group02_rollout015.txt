def propose_candidate(seed=0, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    if 'height_sequence_1' in globals():
        n = int(np.asarray(height_sequence_1).size)
        if n <= 0:
            n = 256
    else:
        n = 256
    # Baseline Catalan
    a = catalan_sequence(n)
    a = normalize a to sum = 1
    best = a.copy()
    best_score = compute_score(best)
    cur = a.copy()
    cur_score = best_score
    # Precompute conv
    # Actually conv only used for gradient steps; compute when needed
    # Simulated annealing parameters
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    # Temperature schedule
    temp0 = 0.05
    temp_end = 1e-6
    # iteration count to update temperature
    it = 0
    while time.monotonic() < deadline:
        # Choose move type with probabilities
        move_type = rng.choice(['transfer','block','gradient'], p=[0.4,0.3,0.3])
        if move_type == 'transfer':
            # choose source with probability proportional to a
            src = rng.choice(n, p=cur/cur.sum())
            # choose target different (prefer later)
            tgt = rng.integers(0, n)
            if tgt == src:
                continue
            # compute epsilon as fraction of source
            eps = cur[src] * rng.uniform(0.0, 0.2)  # up to 20% of src
            if eps <=0:
                continue
            cand = cur.copy()
            cand[src] -= eps
            cand[tgt] += eps
            # ensure non-negative via clipping
            # may not need projection because sum unchanged
            # but we may keep tiny negative due to eps rounding, clip
            cand = np.clip(cand, 0.0, 1000.0)
            # renorm to sum=1 to avoid drift
            cand /= cand.sum()
        elif move_type == 'block':
            # block scaling similar to earlier but with renorm
            width = int(rng.integers(max(1, n // 200), max(2, n // 12) + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = np.exp(rng.normal(0.0, 0.08))
            cand = cur.copy()
            cand[start:start+width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            cand /= cand.sum()
        else:  # gradient
            # compute conv and active k
            conv = np.convolve(cur, cur)
            max_c = conv.max()
            active = np.where(conv >= max_c - 1e-12)[0]
            # randomly choose a k
            k = int(rng.choice(active))
            # gradient vector
            g = np.zeros_like(cur)
            # for indices <= k
            g[:k+1] = 2 * cur[k::-1]  # reversed slice of length k+1
            # step size schedule decreasing
            # compute step = eta * cur.mean() / (np.linalg.norm(g)+1e-12)
            eta = 0.02 * (1 - it / 100000)  # decays over iterations
            # ensure positive
            step = eta
            cand = cur - step * g
            # clip negative
            cand = np.clip(cand, 0.0, None)
            # renorm to sum = 1
            cand_sum = cand.sum()
            if cand_sum <= 0:
                continue
            cand /= cand_sum
        # compute score
        cand_score = compute_score(cand)
        # acceptance criteria
        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
        else:
            # compute temperature
            elapsed = time.monotonic() - start_time
            frac = elapsed / (deadline - start_time)
            temp = temp0 * (1 - frac) + temp_end * frac
            # acceptance probability
            if rng.random() < np.exp((cur_score - cand_score) / max(temp, 1e-12)):
                cur = cand
                cur_score = cand_score
        # update best
        if cur_score < best_score:
            best_seq = cur.copy()
            best_score = cur_score
            print(f"[improve] new best score {best_score:.6f} at iter {it}")
        it += 1
    # final check with evaluate_sequence
    # Convert to list
    final_list = [float(x) for x in best_seq]
    try:
        final_eval = float(evaluate_sequence(final_list))
        print(f"[final] best score {best_score:.6f}, evaluate_sequence output {final_eval:.6f}")
    except Exception:
        print("[final] error evaluating final sequence")
    return final_list

"""
Iterative LP‑based improvement of the Catalan step‑function.

The objective of ``evaluate_sequence`` is
    2·n·max_k (a*a)_k / (∑a)^2 .
Because the ratio is invariant under scaling we can fix the L1‑norm to
S = sqrt(2·n).  Then the score returned by ``evaluate_sequence`` equals the
sup‑norm M = max (a*a).  Starting from the Catalan construction (which
gives M = π/2 ≈ 1.571) we repeatedly solve a linear program

    maximise   ∑ g_j
    subject to   (a * g)_k ≤ M   for all k,
                 g_j ≥ 0 .

If a feasible g has a larger L1‑norm than a, we can blend a with a
renormalised copy of g; the blend keeps the total L1‑norm S while
strictly decreasing M.  Repeating the step drives M down towards the
theoretical lower bound 1.

The implementation works for any length n (default 256 or the length of
the previously found construction).  It uses CVXPY with the ECOS solver,
falls back to a simple stochastic local search if CVXPY is unavailable,
and respects the 1000 s time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    Return the first ``length`` terms of the Catalan–type sequence

        a_i = binom(2*i, i) / 4**i .

    The recurrence a_{i+1} = a_i * (2*i+1) / (2*i+2) is used for stability.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Scale ``seq`` so that its L1 norm equals ``target_sum``.
    The scaling factor is positive and therefore respects the non‑negative
    requirement of the problem.
    """
    cur = np.sum(seq)
    if cur == 0.0:
        return seq
    factor = target_sum / cur
    return seq * factor


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix B such that (a * g) = B @ g,
    i.e.   (a * g)[k] = Σ_i a_i·g_{k‑i} with out‑of‑range indices ignored.
    """
    n = a.size
    rows = np.arange(2 * n - 1)[:, None]        # shape (2n‑1, 1)
    cols = np.arange(n)[None, :]                # shape (1, n)
    diff = rows - cols                           # (2n‑1, n)
    mask = (diff >= 0) & (diff < n)
    # broadcast a over diff where mask=True
    B = np.where(mask, a[diff], 0.0)
    return B


def max_convolution(seq: np.ndarray) -> float:
    """
    Return max_k (seq * seq)[k]  (full convolution, non‑negative sequence).
    """
    conv = np.convolve(seq, seq)
    return float(conv.max())


def project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main function required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative floats that (hopefully) improves on the
    Catalan construction.  The algorithm proceeds as follows:

    1.  Build the Catalan sequence of length n and normalise it so that
        Σ a_i = sqrt(2·n).  With this normalisation the score equals
        the supremum norm M = max (a*a).

    2.  While time permits, solve a linear program to maximise the L1‑norm
        of a vector g subject to (a*g)_k ≤ M for every convolution index.
        If the optimal g has a larger L1‑norm than a, rescale it back to
        the target L1‑norm S and replace a by a convex combination
        a ← (1‑t)·a + t·g.  The mixing step strictly reduces M.

    3.  Keep the best sequence seen and finally return it as a plain Python
        list.  If the CVXPY library is not available the function falls back
        to a simple stochastic local search (the previous baseline).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length n to use.
    # --------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 0:
                raise ValueError
        except Exception:
            n = 256
    else:
        n = 256

    # Cap n to a reasonable size for the LP (avoid huge matrices).
    if n > 1024:
        n = 1024

    # --------------------------------------------------------------
    # Baseline Catalan sequence normalised to the standard L1 norm S.
    # --------------------------------------------------------------
    S = np.sqrt(2.0 * n)               # target L1‑norm
    base_seq = catalan_sequence(n)     # raw Catalan terms
    a = normalise_to_target_sum(base_seq, S)

    # Initial score (M = max convolution) – this equals the evaluation.
    try:
        cur_score = float(evaluate_sequence(a.tolist()))
    except Exception:      # safety – should never happen
        cur_score = float('inf')
    best_seq = a.copy()
    best_score = cur_score

    print(f"[init] n={n}, target L1={S:.6f}, initial score={cur_score:.6f}")

    # --------------------------------------------------------------
    # Try the LP‑based improvement loop.  If cvxpy is not available we
    # fall back to a cheap random walk (the original approach).
    # --------------------------------------------------------------
    try:
        import cvxpy as cp
        have_cvx = True
    except Exception:
        have_cvx = False
        print("[info] cvxpy not available – falling back to random perturbations.")

    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    t_mix = 0.5                      # mixing factor (convex combination weight)
    # Small epsilon to avoid division‑by‑zero in the projection step.
    eps = 1e-12

    while time.monotonic() < deadline:
        iteration += 1
        # ----------------------------------------------------------
        # Compute current supremum M = max (a*a)
        # ----------------------------------------------------------
        conv_aa = np.convolve(a, a)
        M = float(conv_aa.max())

        # ----------------------------------------------------------
        # Build the linear constraints (a * g)_k ≤ M
        # ----------------------------------------------------------
        if have_cvx:
            # Build matrix B once per iteration (size (2n‑1) × n)
            B = build_conv_matrix(a)

            g = cp.Variable(n, nonneg=True)
            constraints = [B @ g <= M]                # element‑wise ≤
            prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

            # Solve – ECOS is the default interior‑point method for QPs/LPs.
            try:
                prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
            except Exception:          # solver failure – abort LP loop
                have_cvx = False
                print("[warn] CVXPY solver failed – exiting LP loop.")
                break

            if prob.status not in ("optimal", "optimal_inaccurate"):
                # No useful improvement possible.
                print(f"[info] LP iteration {iteration}: status {prob.status}")
                break

            g_opt = g.value
            if g_opt is None:
                break

            sum_g = float(g_opt.sum())
            # ------------------------------------------------------
            # If the optimal LP does not increase the L1‑norm, stop.
            # ------------------------------------------------------
            if sum_g <= S + eps:
                # No increase – we have reached a stationary point.
                print(f"[info] LP iteration {iteration}: no L1 increase (sum_g={sum_g:.6f})")
                break

            # ------------------------------------------------------
            # Rescale g to have the same L1‑norm S as a.
            # ------------------------------------------------------
            scale = S / max(sum_g, eps)
            g_scaled = g_opt * scale

            # ------------------------------------------------------
            # Convex combination (mix) and enforce feasibility.
            # ------------------------------------------------------
            a_new = (1.0 - t_mix) * a + t_mix * g_scaled
            # Numerical noise may slightly violate the sum; re‑project.
            a_new = np.clip(a_new, 0.0, 1000.0)
            a_new_sum = float(a_new.sum())
            if a_new_sum == 0.0:
                a_new = normalise_to_target_sum(a_new, S)
            else:
                a_new = a_new * (S / a_new_sum)

            a = a_new
            # ------------------------------------------------------
            # Compute new score.
            # ------------------------------------------------------
            new_score = max_convolution(a)       # equals evaluate_sequence result
        else:
            # ------------------------------------------------------
            # Fallback: cheap stochastic block perturbation (as before).
            # ------------------------------------------------------
            # Pick a random block and apply a small log‑normal scaling.
            width = int(rng.integers(max(1, n // 200), max(2, n // 12) + 1))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.08)))
            cand = a.copy()
            cand[start:start + width] *= factor
            np.clip(cand, 0.0, 1000.0, out=cand)
            # Re‑normalise to the target L1‑norm.
            cand = normalise_to_target_sum(cand, S)
            new_score = max_convolution(cand)

            # Accept if improvement.
            if new_score < cur_score:
                a = cand
            else:
                # occasional uphill moves (simulated annealing style)
                temp = 0.05 * (1.0 - (iteration / 10000.0))
                if rng.random() < np.exp((cur_score - new_score) / max(temp, 1e-12)):
                    a = cand

        # ----------------------------------------------------------
        # Record best solution.
        # ----------------------------------------------------------
        if new_score < best_score:
            best_score = new_score
            best_seq = a.copy()
            print(f"[improve] iter {iteration:5d}, new best score = {best_score:.6f}")

        cur_score = new_score

        # (optional) slowly reduce the mixing factor to stabilise later.
        t_mix = max(0.1, t_mix * 0.9995)

    # --------------------------------------------------------------
    # Final evaluation and return.
    # --------------------------------------------------------------
    final_list = [float(x) for x in best_seq]
    try:
        final_val = float(evaluate_sequence(final_list))
        print(f"[final] best objective = {final_val:.6f} (iterations={iteration})")
    except Exception:
        print("[final] error during final evaluation – returning best raw sequence")
    return final_list
