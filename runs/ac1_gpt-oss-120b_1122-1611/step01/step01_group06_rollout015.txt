def project_simplex(v):
    # ensure non-negative and sum(v)=1
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(n) + 1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def propose_candidate(seed=0, budget_s=1000):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    best_seq = None
    best_score = np.inf

    # Phase 1: coarse search
    lengths = [800, 1200, 2000]
    p_vals = np.linspace(0.45, 0.55, 21)
    deltas = [0.0, 2.0, 5.0]  # shift parameter for power law
    for n in lengths:
        for p in p_vals:
            for delta in deltas:
                idx = np.arange(n) + delta
                seq = idx ** (-p)
                seq /= seq.sum()
                score = evaluate_sequence(seq.tolist())
                if score < best_score:
                    best_score, best_seq = score, seq.copy()
    # Phase 2: Simulated Annealing
    a = best_seq.copy()
    n = a.size
    current_score = best_score
    T0 = 0.1
    t_end = deadline - 5
    while time.monotonic() < t_end:
        # annealing temperature decays linearly
        fraction = (deadline - time.monotonic()) / (deadline - t_end)
        T = T0 * fraction
        # propose modification
        cand = a.copy()
        # choose two indices for mass transfer to keep sum near 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        delta = rng.normal(0, 0.02)  # change amount
        # transfer delta from i to j while keeping nonnegative
        if cand[i] + delta < 0:
            continue
        cand[i] += delta
        cand[j] -= delta
        # ensure nonnegatives
        if np.min(cand) < 0:
            continue
        # renormalize to sum 1
        cand /= cand.sum()
        cand = np.clip(cand, 0, 1000)
        new_score = evaluate_sequence(cand.tolist())
        if new_score < current_score or rng.random() < np.exp(-(new_score - current_score) / (T + 1e-12)):
            a = cand
            current_score = new_score
            if new_score < best_score:
                best_score, best_seq = new_score, cand.copy()
                print(f"[SA] new best {best_score:.6f}")
    # Phase 3: optional block refinement
    while time.monotonic() < deadline:
        cand = best_seq.copy()
        n = cand.size
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 30))
        stop = min(n, start + width)
        factor = np.exp(rng.normal(0, 0.05))
        cand[start:stop] *= factor
        cand /= cand.sum()
        cand = np.clip(cand, 0, 1000)
        score = evaluate_sequence(cand.tolist())

def _project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a near‑optimal step function.

The algorithm works in three stages:

1️⃣  Coarse grid search – try a few lengths and power‑law exponents (with an
    optional shift) and keep the best shape found.

2️⃣  Local optimisation – a fast hill‑climbing / simulated‑annealing loop.
    The current sequence is repeatedly perturbed by moving a small amount of
    mass between two random positions (or by a multiplicative block tweak).
    Each candidate is evaluated with the exact `evaluate_sequence` function.
    Only improving moves are kept (occasionally a worsening move is accepted
    with a tiny probability to escape plateaus).  The step size slowly
    shrinks so the search focuses on fine adjustments.

3️⃣  Final block polishing – if time permits a few extra block‑multiplicative
    perturbations are tried to squeeze out any remaining gain.

All sequences are kept non‑negative and are normalised to unit sum (the
objective is scale‑invariant, any other normalisation works as well).  The
best sequence found before the time budget expires is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _make_power_sequence(n: int, p: float, delta: float = 0.0) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+δ)^{-p}."""
    idx = np.arange(n, dtype=float) + delta + 1.0   # shift by δ (δ≥0)
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, sum x_i = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = u - cssv / (np.arange(n) + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _score_of(seq: np.ndarray) -> float:
    """Convenient wrapper that calls the external evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine spends the allotted time searching and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # Slight safety margin for the wrapper that calls us.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # 1) Coarse grid search over lengths, exponents and a small shift.
    # --------------------------------------------------------------
    lengths = [500, 1000, 1500, 2000]
    p_grid = np.linspace(0.45, 0.55, 21)            # focus around the theoretical optimum
    deltas = [0.0, 1.0, 3.0]                       # tiny shifts of the singularity

    for n in lengths:
        for delta in deltas:
            for p in p_grid:
                seq = _make_power_sequence(n, p, delta)
                score = _score_of(seq)
                if score < best_score:
                    best_score = score
                    best_seq = seq.copy()
                    print(f"[grid] n={n}, p={p:.3f}, δ={delta:.1f} → score={best_score:.6f}")

        # give a quick report for each length
        print(f"[grid] finished length={n}, best_score={best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Local optimisation – hill‑climbing + simulated‑annealing.
    # --------------------------------------------------------------
    # start from the best candidate found above
    a = best_seq.copy()
    n = a.shape[0]
    cur_score = best_score

    # parameters for the stochastic search
    step_factor = 0.2          # typical fraction of a[i] moved in a transfer
    block_sigma = 0.06         # std. dev. for block multiplicative tweaks
    temperature = 0.02        # annealing temperature for occasional uphill moves
    decay_rate = 0.9999        # slow cooling

    while time.monotonic() < deadline:
        # ----------------------------------------------------------
        # Decide which type of perturbation to apply
        # ----------------------------------------------------------
        if rng.random() < 0.7:
            # ----- mass‑transfer move (preserves total sum) -----
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            # transfer a small fraction of mass from i to j
            delta = rng.random() * step_factor * a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            # small numerical drift may appear – re‑normalise safely
            cand = _project_simplex(cand)
        else:
            # ----- block multiplicative perturbation (as in the starter) -----
            start = rng.integers(0, n)
            width = rng.integers(1, max(2, n // 30))
            stop = min(n, start + width)
            factor = math.exp(rng.normal(0.0, block_sigma))
            cand = a.copy()
            cand[start:stop] *= factor
            cand = _project_simplex(cand)

        # ----------------------------------------------------------
        # Evaluate the candidate
        # ----------------------------------------------------------
        new_score = _score_of(cand)

        # ----------------------------------------------------------
        # Acceptance rule (mostly greedy, occasional uphill moves)
        # ----------------------------------------------------------
        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # tiny probability to accept a worse move – helps to escape flat plateaus
            if rng.random() < math.exp(-(new_score - cur_score) / (temperature + 1e-12)):
                accept = True

        if accept:
            a = cand
            cur_score = new_score
            # keep the globally best sequence
            if cur_score < best_score:
                best_score = cur_score
                best_seq = a.copy()
                print(f"[improve] new best_score={best_score:.6f} (n={n})")

        # ----------------------------------------------------------
        # Update annealing / step parameters
        # ----------------------------------------------------------
        temperature *= decay_rate
        step_factor = max(step_factor * decay_rate, 1e-4)

    # --------------------------------------------------------------
    # 3) Final quick block polishing (optional, only if time left)
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        cand = best_seq.copy()
        # a modest random block tweak
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 20))
        stop = min(n, start + width)
        factor = math.exp(rng.normal(0.0, 0.04))
        cand[start:stop] *= factor
        cand = _project_simplex(cand)

        score = _score_of(cand)
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[polish] improved best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a simple uniform sequence.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.ones(256, dtype=float)
        best_seq /= best_seq.sum()

    # Convert to plain Python floats for the caller
    return [float(x) for x in best_seq]
