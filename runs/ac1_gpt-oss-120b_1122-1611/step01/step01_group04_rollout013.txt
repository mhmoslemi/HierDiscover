Propose a coefficient sequence that beats the previous 1.574374 bound.

Algorithm
---------
1. Determine the working length `n` (use the provided `height_sequence_1`
   if present, otherwise fall back to 6000).
2. Initialise the search with a family a_i = (i+shift)^β (β≈−0.5)
   – this family asymptotically attains the known lower bound π/2.
   A fine grid on β∈[‑0.55,‑0.45] and shift∈[0.5,5] is scanned using a
   fast FFT based evaluation of the score (the same formula that
   `evaluate_sequence` implements).
3. The best candidate from the grid search becomes the current solution.
4. Local optimisation:
   * Repeatedly compute the full auto‑convolution (FFT) and locate the
     index `k_max` where the convolution attains its maximum.
   * The contribution of each entry a_i to that maximum is
     w_i = a_i·a_{k_max‑i} (zero outside the support).
   * Pick a “high‑impact’’ index (`i_hi`) where w_i is large and a
     “low‑impact’’ index (`i_lo`) where w_i is small.
   * Transfer a small fraction (adaptively decreasing) of the mass from
     `i_hi` to `i_lo`.  The move is kept only if the score improves.
   * Occasionally perform a cheap random block‑wise multiplicative
     perturbation (as in the reference code) to escape plateaus.
   * All moves respect the required bounds (≥0, ≤1000) and the
     1000 s time budget.
5. After the time limit the best sequence found is verified with the
   official `evaluate_sequence` function and returned as plain Python
   floats.

The search is entirely self‑contained, uses only `numpy`, `time` and
`math`, and runs well within the allowed CPU and memory limits.

import numpy as np
import time

def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift)^beta , clipped to the [0,1000] range."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_sine(gamma: float, n: int) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n)^gamma   – symmetric hump."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    return np.clip(seq, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of a contiguous block."""
    cand = np.array(base, copy=True)
    n = cand.size
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[st:en] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def fft_conv_stats(a: np.ndarray, L: int) -> tuple[float, float, np.ndarray]:
    """
    Return (max_convolution, sum_of_a, full_convolution_array) using FFT.
    """
    f = np.fft.rfft(a, L)
    conv = np.fft.irfft(f * f, n=L)
    # the true linear convolution lives in the first 2*n-1 entries
    conv = conv[: 2 * a.size - 1]
    max_b = float(conv.max())
    sum_a = float(a.sum())
    return max_b, sum_a, conv


def fast_score(a: np.ndarray, L: int, n: int) -> float:
    """Score = 2 n·max(conv) / (sum(a)**2) – the same formula used by evaluate_sequence."""
    max_b, sum_a, _ = fft_conv_stats(a, L)
    if sum_a < 0.01:
        return math.inf
    return 2.0 * n * max_b / (sum_a ** 2)

            best = cand
            best_score = sc
            print(f"grid sine γ={gamma:.3f} → {sc:.6f}")

    # --------------------------------------------------------------
    # 3)  Local refinement – targeted mass transfer + occasional block tweaks
    # --------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score
    epsilon = 0.02          # initial fraction of mass to move
    it = 0
    accept_cnt = 0

    while time.monotonic() < deadline:
        # recompute convolution statistics for the current vector
        max_b, sum_a, conv = fft_conv_stats(cur, L)
        cur_score = 2.0 * n * max_b / (sum_a ** 2)

        # keep the global best
        if cur_score < best_score:
            best = cur.copy()
            best_score = cur_score
            print(f"[{it}] new global best → {best_score:.6f}")

        # ---- targeted move -------------------------------------------------
        k_max = int(conv.argmax())
        # contributions w_i = a_i * a_{k_max-i}
        start = max(0, k_max - (n - 1))
        end = min(n, k_max + 1)
        idx = np.arange(start, end)
        jdx = k_max - idx
        w = cur[idx] * cur[jdx]

        # pick a high‑impact index (largest w) and a low‑impact one
        hi = idx[w.argmax()]
        # choose a low‑impact index uniformly (could also use small w)
        lo = rng.integers(0, n)
        while lo == hi:
            lo = rng.integers(0, n)

        delta = cur[hi] * epsilon
        if delta < 1e-8:
            delta = epsilon  # fallback for very small entries
        if cur[hi] < delta:
            delta = cur[hi] * 0.5

        cand = cur.copy()
        cand[hi] = max(0.0, cand[hi] - delta)
        cand[lo] = min(1000.0, cand[lo] + delta)

        # evaluate quickly
        cand_score = fast_score(cand, L, n)

        if cand_score < cur_score:
            cur = cand
            cur_score = cand_score
            accept_cnt += 1
        else:
            # occasional random block perturbation to escape plateaus
            if rng.random() < 0.003:
                cand = make_block_perturbation(cur, rng)
                cand_score = fast_score(cand, L, n)
                if cand_score < cur_score:
                    cur = cand
                    cur_score = cand_score
                    accept_cnt += 1

        # decay epsilon slowly
        if it % 5000 == 0 and epsilon > 0.001:
            epsilon *= 0.95

        it += 1

    print(f"finished after {it} iterations, best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final verification with the official evaluator (just in case)
    # ------------------------------------------------------------------
    final_score = float(evaluate_sequence(best.tolist()))
    print(f"final official score = {final_score:.6f}")

    # Return plain Python floats
    return [float(x) for x in best]
