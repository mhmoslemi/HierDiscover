    deadline = time.monotonic() + max(0, budget_s - 2.0)

    # Initialize a: if height_sequence_1 exists, copy it; else uniform of moderate size (e.g., 2000)
    # Ensure positivity, renormalize sum=1.

    n = a.shape[0]

    # Precompute FFT length.
    fft_len = _fft_len_for_n(n)

    best = a.copy()
    best_score = compute_score(best)  # 2*n * max(conv)
    # Print initial.

    # Settings: schedule of step sizes and thresholds
    # We'll have phases: high eta (0.8) for first 30% time; mid (0.3) for next 40%; low (0.05) for final 30% maybe.

    total_time = deadline - start_time

    while time.monotonic() < deadline:
        # compute convolution
        conv = _conv_self(a, fft_len)
        max_val = conv.max()
        # compute active indices: conv >= max_val - delta where delta = max_val * 0.005 or maybe absolute difference
        delta = max_val * 0.003  # small threshold.
        active_idx = np.where(conv >= max_val - delta)[0]
        # compute aggregated partner vector
        w = np.zeros_like(a)
        for k in active_idx:
            w += _partner_vector(a, k)
        # optionally weight w by 1/len(active_idx) just mean
        if active_idx.size > 0:
            w = w / active_idx.size

        # adapt step size based on time:
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time
        if frac < 0.3:
            eta = 0.9
        elif frac < 0.7:
            eta = 0.3
        else:
            eta = 0.07

        factor = np.exp(-eta * w)
        a = a * factor

        # Normalise
        s = a.sum()
        if s == 0:
            a = np.full_like(a, 1.0/n)
        else:
            a /= s

        # occasional random move
        if rng.random() < 0.02: # 2% chance
            # Choose i with probability proportional to a*w (i.e., high contribution)
            probs_i = a * w
            if probs_i.sum() == 0:
                i = rng.integers(0, n)
            else:
                probs_i = probs_i / probs_i.sum()
                i = rng.choice(np.arange(n), p=probs_i)
            # Choose j with low contribution (use 1/(a*w+epsilon) maybe)
            probs_j = (a * (1.0/(w+1e-12)))  # maybe high a low w
            # but ensure not same
            probs_j = probs_j / probs_j.sum()
            j = rng.choice(np.arange(n), p=probs_j)
            # Move mass epsilon
            epsilon = min(0.03, a[i]) # move at most 3% of total sum maybe.
            a[i] -= epsilon
            a[j] += epsilon
            # Ensure non-negative
            a = np.clip(a, 0.0, None)
            # renormalize
            a /= a.sum()

        # Evaluate score
        cur_score = 2.0 * n * conv.max()
        # Could recompute conv after update (we already have conv) - need to recompute after modifications.
        # Actually conv holds for old a; we need conv after modify. Let's recompute:
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # Possibly random global scaling for exploration
        if rng.random() < 0.005:
            # apply random lognormal scaling factor to whole vector then renorm
            factor_all = np.exp(rng.normal(0.0, 0.2))
            a = a * factor_all
            a = np.clip(a, 0.0, None)
            a /= a.sum()
        # other operations?

    # after loop end, clip to bounds and convert to list.
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
A fast, multi‑phase exponentiated‑gradient search for a step‑function that
minimises

    score = 2 * n * max(conv(a, a)) / (sum(a)**2)

where ``conv`` is the linear self‑convolution.  The algorithm works with a
normalised vector (sum = 1) so that the score reduces to ``2·n·max(conv)``.
It improves on the reference implementation by

* aggregating the sub‑gradient over *all* indices whose convolution value
  is within a small fraction of the current maximum,
* using a three‑phase learning‑rate schedule,
* periodically performing a deterministic “mass‑swap’’ that moves probability
  from entries that contribute heavily to the active peak(s) toward entries
  that contribute little,
* enforcing symmetry (a_i = a_{n‑1‑i}) after each update – the optimal solution
  is expected to be symmetric,
* adding occasional random block perturbations to escape shallow minima.

The routine runs for approximately ``budget_s`` seconds (a safety margin of
two seconds is kept) and returns the best vector found.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """
    Return the smallest power‑of‑two length that can hold a full linear
    convolution of two vectors of length ``n``.
    """
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a (real, non‑negative) using the FFT.
    The result has length ``2*len(a)-1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a convolution index ``k`` (0‑based) return the vector ``w`` with
    ``w[i] = a[k‑i]`` (zero when the index is out of the support).
    It is the sub‑gradient of the active convolution entry (a*a)[k].
    """
    n = a.size
    w = np.zeros_like(a)

    if k < n:
        # overlap range i = 0 … k
        w[: k + 1] = a[k :: -1]          # a[k], a[k‑1], …, a[0]
    else:
        # overlap range i = k‑(n‑1) … n‑1
        start = k - (n - 1)              # first i that has a partner
        length = (2 * n - 1) - k         # number of overlapping entries
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises
    the provided ``evaluate_sequence`` score.  The routine runs for roughly
    ``budget_s`` seconds (leaving a small safety margin) and uses a
    multi‑phase exponentiated‑gradient descent with aggregated sub‑gradients,
    symmetry enforcement, and occasional stochastic moves.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise the candidate vector ``a``.
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
    except Exception:
        # fall back to a modestly sized uniform vector
        base = np.ones(3809, dtype=np.float64)

    # we keep the original length – the problem is scale‑invariant in n,
    # but the known baselines work with this size.
    n = base.size

    a = np.clip(base, 0.0, None)
    total = a.sum()
    if total <= 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / total                     # unit sum → scale‑invariant

    # enforce symmetry from the start (the optimal sequence is expected to be symmetric)
    a = (a + a[::-1]) * 0.5
    a = a / a.sum()

    # ------------------------------------------------------------------
    # Pre‑computations
    # ------------------------------------------------------------------
    fft_len = _fft_len_for_n(n)

    # fast score computation for normalised vectors: 2·n·max(conv)
    def _fast_score(vec):
        conv = _conv_self(vec, fft_len)
        return 2.0 * n * np.max(conv)

    best = a.copy()
    best_score = _fast_score(best)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Scheduling parameters
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 2.0)      # safety margin
    total_time = deadline - start

    # learning‑rate schedule (phase‑wise)
    eta_start = 0.9   # aggressive early updates
    eta_mid = 0.25
    eta_end = 0.03   # very fine adjustments at the end

    # active‑set tolerance schedule (fraction of the current maximum)
    # early iterations keep a *wide* active set, later we tighten it.
    tol_start = 0.20   # consider all entries ≥ 80 % of the peak
    tol_end   = 0.005  # finally focus on the very top of the peak

    # probabilities for stochastic moves
    prob_swap   = 0.04   # mass‑swap between high/low contributors
    prob_perturb = 0.015 # random block scaling

    # ------------------------------------------------------------------
    # Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # 1) current convolution and its peak
        conv = _conv_self(a, fft_len)
        max_val = float(conv.max())

        # 2) determine the active set (indices whose value is close to the maximum)
        elapsed = time.monotonic() - start
        frac = elapsed / total_time

        # linear interpolation of the tolerance
        tol = tol_start * (1.0 - frac) + tol_end * frac
        threshold = max_val * (1.0 - tol)
        active_idx = np.where(conv >= threshold)[0]

        # 3) aggregate the sub‑gradient over the whole active set
        w = np.zeros_like(a)
        for k in active_idx:
            # direct (slice‑based) partner computation – faster than a function call
            if k < n:
                w[: k + 1] += a[k :: -1]
            else:
                start_i = k - (n - 1)
                length = (2 * n - 1) - k
                rev = a[::-1]
                w[start_i:] += rev[:length]
        if active_idx.size > 0:
            w /= active_idx.size             # average contribution

        # 4) learning‑rate according to the time schedule
        if frac < 0.3:
            eta = eta_start * (1.0 - frac / 0.3) + eta_mid * (frac / 0.3)
        elif frac < 0.7:
            eta = eta_mid * (1.0 - (frac - 0.3) / 0.4) + eta_end * ((frac - 0.3) / 0.4)
        else:
            eta = eta_end

        # 5) multiplicative (exponentiated‑gradient) update
        a *= np.exp(-eta * w)

        # 6) enforce positivity and renormalise
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= a_sum

        # 7) symmetry projection (keeps the search in the symmetric sub‑space)
        a = (a + a[::-1]) * 0.5
        a /= a.sum()

        # ------------------------------------------------------------------
        # Stochastic exploration moves
        # ------------------------------------------------------------------
        # a) mass‑swap between a “high‑w’’ donor and a “low‑w’’ recipient
        if rng.random() < prob_swap:
            donor_weights = a * (w + 1e-12)
            if donor_weights.sum() > 0:
                donor_weights /= donor_weights.sum()
                i = rng.choice(np.arange(n), p=donor_weights)

                recip_weights = a * (1.0 / (w + 1e-12))
                if recip_weights.sum() > 0:
                    recip_weights /= recip_weights.sum()
                    j = rng.choice(np.arange(n), p=recip_weights)

                    if i != j:
                        # move a modest amount of probability mass
                        eps = min(0.01 * a[i], a[i] * 0.5)
                        a[i] -= eps
                        a[j] += eps
                        a = np.clip(a, 0.0, None)
                        a /= a.sum()
                        # keep symmetry after the swap
                        a = (a + a[::-1]) * 0.5
                        a /= a.sum()

        # b) random log‑normal scaling of a short contiguous block
        if rng.random() < prob_perturb:
            width = rng.integers(1, max(2, n // 15))
            start_block = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.12))
            a[start_block:start_block + width] *= factor
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            a = (a + a[::-1]) * 0.5
            a /= a.sum()

        # ------------------------------------------------------------------
        # Evaluate current candidate and keep the best
        # ------------------------------------------------------------------
        cur_score = _fast_score(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence encountered (as plain Python floats)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)   # safety for evaluate_sequence
    return [float(x) for x in best]
