def project_simplex(v):
    # Projects v onto the simplex {x >= 0, sum(x)=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def propose_candidate(seed=42, budget_s=1000, **kwargs):
    import time, numpy as np, math

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # choose length n based on a simple heuristic: use 2000 or 3000
    n = 2500

    # generate initial candidates
    best_seq = None
    best_score = float('inf')
    # power law candidates
    p_vals = [0.45,0.5,0.55,0.60,0.65,0.70]
    for p in p_vals:
        seq = _make_power_sequence(n, p)
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
    # uniform
    seq = np.ones(n) / n
    score = float(evaluate_sequence(seq.tolist()))
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
    # random simple
    seq = rng.random(n)
    seq /= seq.sum()
    score = float(evaluate_sequence(seq.tolist()))
    if score < best_score:
        best_score = score
        best_seq = seq.copy()
    print(f"Initial best_score={best_score:.6f}")

    # start gradient descent
    a = best_seq.copy()
    iteration = 0
    eta0 = 0.2  # may need scaling
    # also maintain momentum maybe
    momentum = np.zeros_like(a)
    mu = 0.9

    while time.monotonic() < deadline:
        # compute convolution via FFT
        b = _full_convolution(a)
        kmax = int(np.argmax(b))
        maxb = b[kmax]
        # compute gradient of b_kmax wrt a
        idx = kmax - np.arange(n)
        # valid positions where idx in [0,n-1]
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[idx[valid]]
        # step size
        eta = eta0 / np.sqrt(iteration + 1)
        # momentum update
        momentum = mu * momentum - eta * g
        a_new = a + momentum
        # project onto simplex (non-negative sum=1)
        a_new = _project_to_simplex(a_new)
        # evaluate
        score = float(evaluate_sequence(a_new.tolist()))
        if score < best_score - 1e-9:
            best_score = score
            best_seq = a_new.copy()
            print(f"[iter {iteration}] new best {best_score:.6f} (maxb={maxb:.6e})")
        a = a_new
        iteration += 1
        # occasional random perturb
        if iteration % 5000 == 0:
            # random small noise and renorm
            a = a + rng.normal(0.0, 0.01, size=n)
            a = _project_to_simplex(a)
    # after loop, fallback if not found any improvement
    if best_seq is None:
        best_seq = np.ones(256) / 256
    return [float(x) for x in best_seq]

def _full_convolution(a):
    n = a.shape[0]
    m = 1 << ((2*n-1).bit_length())  # next power of two >= 2n-1
    # real FFT
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return conv[:2*n-1]

def _project_to_simplex(v):
    """Project vector v onto probability simplex {x>=0, sum=1}."""
    # Reference: Duchi et al. 2008 algorithm
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all nonpositive
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a near‑optimal step‑function using projected gradient descent.

The algorithm:
  1. Initialise a set of candidate sequences (power‑law, uniform, random,
     optionally the provided `height_sequence_1`).  Keep the best one.
  2. Repeatedly reduce the maximal value of the self‑convolution.
     For the current sequence `a` we locate the index `k*` where
     `(a*a)[k*]` attains its maximum.
     The gradient of this entry w.r.t. `a` is
         ∂b_k*/∂a_i = 2·a_{k*-i}   (zero if the index is out of range).
     A small step against this gradient is taken and the result is projected
     back onto the probability simplex (non‑negative entries that sum to 1).
  3. The convolution is computed by FFT for speed (O(n log n) per iteration).
  4. The best sequence encountered is memorised; occasional random jitter
     helps escape plateaus.
  5. When the time budget expires the best sequence found is returned.
"""

import time
import math
import numpy as np


def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x ≥ 0, sum(x) = 1}."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a using a real FFT (fast for many iterations)."""
    n = a.shape[0]
    size = 2 * n - 1
    # next power of two for efficient FFT
    m = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, n=m)
    conv = np.fft.irfft(fa * fa, n=m)
    return conv[:size]


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimise
    `evaluate_sequence`.  The routine spends the allotted time searching and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the outer wrapper
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # -------------------------------------------------------------
    # 1) Choose a sequence length (2000 is a good trade‑off)
    # -------------------------------------------------------------
    n = 2000

    # -------------------------------------------------------------
    # 2) Gather a few initial candidates
    # -------------------------------------------------------------
    init_candidates = []

    # power‑law seeds around the theoretically good exponent
    for p in (0.45, 0.50, 0.55, 0.60, 0.65, 0.70):
        init_candidates.append(_make_power_sequence(n, p))

    # uniform seed
    init_candidates.append(np.full(n, 1.0 / n))

    # random Dirichlet‑like seed
    rand_vec = rng.random(n)
    init_candidates.append(rand_vec / rand_vec.sum())

    # try to use the provided reference construction if available
    try:
        # `height_sequence_1` is expected to be a plain Python list
        ref = np.asarray(height_sequence_1, dtype=np.float64)
        if ref.size < n:
            # pad with zeros
            ref = np.pad(ref, (0, n - ref.size), constant_values=0.0)
        elif ref.size > n:
            ref = ref[:n]
        # ensure it is a valid simplex point
        ref = np.clip(ref, 0.0, 1000.0)
        ref = _project_to_simplex(ref)
        init_candidates.append(ref)
    except NameError:
        pass  # no external reference present

    best_seq = None
    best_score = float("inf")

    # evaluate the few starters using the official evaluator (once each)
    for cand in init_candidates:
        sc = float(evaluate_sequence(cand.tolist()))
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
    print(f"[init] best_score={best_score:.6f}")

    # -------------------------------------------------------------
    # 3) Projected gradient descent on the max‑convolution objective
    # -------------------------------------------------------------
    a = best_seq.copy()
    # guarantee exact simplex normalisation
    a = _project_to_simplex(a)

    # pre‑compute the convolution for the starting point
    conv = _full_convolution(a)
    maxb = conv.max()
    best_score = 2.0 * n * maxb
    best_seq = a.copy()
    print(f"[start] score={best_score:.6f}")

    iteration = 0
    eta0 = 0.5               # initial step size (tuned empirically)
    no_improve = 0          # counter for stagnation detection

    while time.monotonic() < deadline:
        # index where the current convolution attains its maximum
        kmax = int(np.argmax(conv))

        # gradient of the selected entry
        idx = kmax - np.arange(n)
        valid = (idx >= 0) & (idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[idx[valid]]

        # decreasing step size
        eta = eta0 / math.sqrt(iteration + 1.0)

        # gradient step and simplex projection
        a_new = a - eta * g
        a_new = _project_to_simplex(a_new)

        # new convolution & objective
        conv_new = _full_convolution(a_new)
        maxb_new = conv_new.max()
        score_new = 2.0 * n * maxb_new

        # keep the best ever seen
        if score_new + 1e-12 < best_score:
            best_score = score_new
            best_seq = a_new.copy()
            no_improve = 0
            print(f"[iter {iteration:>6d}] new best score={best_score:.6f}")
        else:
            no_improve += 1

        # prepare for the next iteration
        a = a_new
        conv = conv_new
        iteration += 1

        # occasional random jitter to escape flat plateaus
        if iteration % 5000 == 0:
            jitter = rng.normal(0.0, 0.01, size=n)
            a = _project_to_simplex(a + jitter)
            conv = _full_convolution(a)

        # if we have been stuck for a long time, reset from the best so far
        if no_improve >= 20000:
            # small random perturbation around the current best
            a = best_seq + rng.normal(0.0, 0.005, size=n)
            a = _project_to_simplex(a)
            conv = _full_convolution(a)
            no_improve = 0

    # -------------------------------------------------------------
    # 4) Final verification (optional) and return
    # -------------------------------------------------------------
    final_score = float(evaluate_sequence(best_seq.tolist()))
    print(f"[final] best_score={best_score:.6f} (evaluate_sequence={final_score:.6f})")

    # ensure the output meets the evaluation function's safety limits
    result = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in result]
