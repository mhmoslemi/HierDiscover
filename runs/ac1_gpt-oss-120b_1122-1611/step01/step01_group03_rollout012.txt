    block_max_width = max(4, n // 20)

        iter_cnt += 1
        move_type = rng.random()
        cand = cur.copy()

"""Simple stochastic descent for minimizing sup norm of self‑convolution.

The algorithm starts from the optimal α‑sequence, then repeatedly mutates a
candidate distribution on the simplex (mass transfers from early to late
indices, multiplicative noise, block scaling and a gradient‑descent step
based on the current maximal convolution entry).  A simulated‑annealing
acceptance rule allows occasional uphill moves.  The best sequence found
before the time budget expires is returned.
"""

import time
import numpy as np

def generate_alpha_sequence(n: int) -> np.ndarray:
    """Generate the first n coefficients a_i = C(2i,i) / 4^i."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a * a using FFT (real valued output)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv

def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    conv = full_convolution(a)
    return float(conv.max())

def objective_score(a: np.ndarray) -> float:
    """Objective used for optimisation: 2·n·max(a*a) (a is already normalised)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """Search for a step‑function with a low self‑convolution supremum.

    The routine keeps a single current candidate and performs several kinds
    of stochastic mutations (mass transfer from the early part of the vector
    to its tail, multiplicative Gaussian noise, block scaling and a cheap
    projected gradient step).  A simple simulated‑annealing acceptance rule
    lets the search escape shallow local minima.  The best sequence seen
    before the time budget expires is returned.
    """
    rng = np.random.default_rng(seed)

    # ----------------------------------------------------------------------
    # Determine length of the construction
    # ----------------------------------------------------------------------
    global height_sequence_1
    if isinstance(height_sequence_1, (list, np.ndarray)) and len(height_sequence_1) > 0:
        n = int(len(height_sequence_1))
    else:
        n = 3000   # harmless fallback

    # ----------------------------------------------------------------------
    # Initialise from the optimal α‑sequence (scaled to ℓ₁ = 1)
    # ----------------------------------------------------------------------
    raw_alpha = generate_alpha_sequence(n)
    cur = raw_alpha / raw_alpha.sum()        # sum = 1
    best = cur.copy()
    best_score = objective_score(best)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ----------------------------------------------------------------------
    # Search parameters
    # ----------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    iteration = 0
    temperature = 0.05
    decay = 0.99996
    sigma = 0.22                       # magnitude of global multiplicative noise
    early_frac = 0.25
    late_frac  = 0.75
    early_end   = max(1, int(n * early_frac))
    late_start  = min(n - 1, int(n * late_frac))

    # block‑scaling widths (guarantee at least one element)
    min_block = max(2, n // 250)
    max_block = max(4, n // 20)

    # ----------------------------------------------------------------------
    # Main optimisation loop
    # ----------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1
        move = rng.random()
        cand = cur.copy()

"""Simple stochastic descent for minimizing the supremum of a self‑convolution.

The search starts from the optimal “α‑sequence” (a_i = C(2i,i)/4^i) which
has a flat self‑convolution on the first half of its support and gives a
score ≈ π/2 ≈ 1.57086.  The algorithm repeatedly mutates a current candidate
distribution on the probability simplex:

* transfer a fraction of mass from an early index to a tail index,
* apply global multiplicative Gaussian noise,
* scale a random contiguous block,
* perform a cheap projected gradient step that reduces the current maximal
  convolution entry (gradient is summed over *all* maximising indices).

A simulated‑annealing acceptance rule allows occasional uphill moves,
helping the search escape shallow local minima.  The best sequence seen
before the time budget expires is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """Return the first n coefficients a_i = C(2i,i) / 4^i (i = 0 .. n‑1)."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2 * i - 1) / (2.0 * i)
    return a


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a using FFT (real output)."""
    n = a.shape[0]
    size = 1 << ((2 * n - 1).bit_length())      # next power of two
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def max_convolution(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return float(full_convolution(a).max())


def objective_score(a: np.ndarray) -> float:
    """Score when a is already normalised to sum = 1: 2·n·max(a*a)."""
    n = a.shape[0]
    return 2.0 * n * max_convolution(a)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a step‑function with a low self‑convolution supremum.

    Returns the best sequence (list of non‑negative floats) found before
    the allocated time expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine the length to use
    # --------------------------------------------------------------
    global height_sequence_1
    if isinstance(height_sequence_1, (list, np.ndarray)) and len(height_sequence_1) > 0:
        n = int(len(height_sequence_1))
    else:
        n = 3000                     # fallback length

    # --------------------------------------------------------------
    # Initialise from the α‑sequence (ℓ₁‑normalised)
    # --------------------------------------------------------------
    raw_alpha = generate_alpha_sequence(n)
    cur = raw_alpha / raw_alpha.sum()        # sum = 1
    best = cur.copy()
    best_score = objective_score(best)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # --------------------------------------------------------------
    # Search parameters
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    it = 0
    temperature = 0.05                     # annealing temperature
    decay = 0.99996                         # exponential cooling
    sigma = 0.22                            # magnitude of global noise

    early_frac = 0.25
    late_frac = 0.75
    early_end = max(1, int(n * early_frac))
    late_start = min(n - 1, int(n * late_frac))

    min_block = max(2, n // 250)
    max_block = max(4, n // 20)

    cur_score = best_score

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        it += 1
        move = rng.random()
        cand = cur.copy()

        if move < 0.25:
            # ----- mass transfer from early part to tail (preserves ℓ₁) -----
            i = rng.integers(0, early_end)          # early index
            j = rng.integers(late_start, n)        # tail index
            if i != j:
                delta = cand[i] * rng.uniform(0.0, 0.4)   # up to 40 % of source mass
                cand[i] -= delta
                cand[j] += delta

        elif move < 0.5:

            cand *= (1.0 + sigma * rng.standard_normal(n))

        elif move < 0.75:

            width = int(rng.integers(min_block, max_block))

            step = 0.012 / (grad.max() + 1e-12)
            cand = cur - step * grad

        # --------------------------------------------------------------

        # --------------------------------------------------------------

                best_score = cand_score
                print(f"[{it:6d}] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # Parameter schedule
        # --------------------------------------------------------------
        temperature *= decay
        if it % 4000 == 0:
            sigma *= 0.96                     # slowly diminish the global noise

    print(f"[final] iterations={it}, best score={best_score:.6f}")
    # Return a plain Python list as required by the external evaluator
    return best.tolist()
