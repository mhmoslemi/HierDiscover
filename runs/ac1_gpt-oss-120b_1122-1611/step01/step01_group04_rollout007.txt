    max_b = conv.max()
    k = conv.argmax()

    total = a_new.sum()
    if total > 0:

    else:

"""
Propose a high‑quality step‑function.

The search works in three stages.
1)  A broad parametric sweep over monotone power‑law / power‑law‑with‑exponential
    families (including a sinusoidal envelope) quickly finds a decent seed.
2)  A fast gradient‑descent on the *active* convolution peak is applied.
    For a fixed ℓ1‑norm the objective is 2·ℓ·max(conv(a,a))/‖a‖₁²,
    so after normalising ‖a‖₁=1 the score is simply 2·ℓ·max(conv).
    The gradient of the active peak is easy to compute analytically,
    giving a cheap O(ℓ) update.
3)  Cheap block‑wise multiplicative perturbations (as in the reference code)
    are interleaved to escape shallow local minima.

All helpers are top‑level, the routine respects the 1000 s budget,
and the returned list contains only non‑negative floats.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Convolution utilities (FFT based, O(n log n))
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def convolve_self(a: np.ndarray) -> np.ndarray:
    """
    Full linear convolution a * a using real FFT.
    Returns a real array of length 2·len(a)‑1.
    """
    n = a.shape[0]
    size = _next_pow2(2 * n - 1)
    # rfft works on real data and is a little faster than fft
    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[: 2 * n - 1]
    return conv


def local_score(a: np.ndarray) -> float:
    """
    Fast replica of evaluate_sequence for a given vector a.
    The vector may be any non‑negative real numbers; the function
    applies the same clipping as evaluate_sequence.
    """
    # Clip to the allowed interval (the search never exceeds it,
    # but we mirror the official routine for safety).
    a = np.clip(a, 0.0, 1000.0)
    sum_a = float(a.sum())
    if sum_a < 1e-12:
        return float("inf")
    conv = convolve_self(a)
    max_b = float(conv.max())
    n = a.shape[0]
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------
def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift)^beta, beta may be negative."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_power_exp(beta: float, shift: float, lam: float, n: int) -> np.ndarray:
    """a_i = (i+shift)^beta * exp(-lam·i)."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = (i ** beta) * np.exp(-lam * (i - shift))
    return np.clip(seq, 0.0, 1000.0)


def generate_power_sine(beta: float,
                        shift: float,
                        lam: float,
                        amp: float,
                        freq: float,
                        n: int) -> np.ndarray:
    """Power law with optional exponential tail and a sinusoidal envelope."""
    i = np.arange(n, dtype=np.float64) + shift
    base = (i ** beta) * np.exp(-lam * (i - shift))
    envelope = 1.0 + amp * np.sin(2.0 * math.pi * freq * i / n)
    seq = base * envelope
    return np.clip(seq, 0.0, 1000.0)


def generate_sine(gamma: float, n: int) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n)^γ – symmetric hump."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    return np.clip(seq, 0.0, 1000.0)


def generate_triangle(n: int) -> np.ndarray:
    """Simple symmetric triangular shape."""
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        seq = np.concatenate([half, half[::-1]])
    else:
        seq = np.concatenate([half, half[-2::-1]])
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Small random perturbations (block multiplicative)
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) multiplicative perturbation of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width between 0.5% and ~8% of n
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[st:en] *= factor
    # occasional symmetrisation (helps for symmetric starters)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Gradient step for the active convolution peak
# ----------------------------------------------------------------------
def gradient_step(a: np.ndarray, lr: float) -> tuple[np.ndarray, float]:
    """
    Perform one gradient‑descent step aimed at reducing the currently
    maximal convolution entry.
    Returns (new_vector, new_score).
    """
    n = a.shape[0]
    conv = convolve_self(a)
    max_b = float(conv.max())
    k = int(conv.argmax())            # index of the active peak

    # Compute the sub‑gradient of max_b w.r.t. a.
    idx = k - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    # generic contribution (appears twice in the ordered sum)
    g[mask] = 2.0 * a[idx[mask]]
    # diagonal entry counted only once
    if k % 2 == 0:
        i_mid = k // 2
        if 0 <= i_mid < n:
            g[i_mid] = a[i_mid]     # replace the double count by a single one

    # Take a small step opposite to the gradient, re‑normalise.
    a_new = a - lr * g
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    else:
        a_new = a.copy()  # fallback – should never happen
    return a_new, local_score(a_new)

                last_improve = iter_cnt
                # slowly increase learning‑rate when we succeed
                lr = min(lr * 1.08, 0.5)
                print(f"[grad] #{iter_cnt} improved → {sc:.6f}")
            else:
                # no improvement – shrink step‑size
                lr = max(lr * 0.74, 1e-4)
                # keep current a (do not move)
        else:
            # Random block‑wise multiplicative tweak (helps escape plateaus)
            cand = make_block_perturbation(a, rng)
            # re‑normalise to ℓ1 = 1
            s = cand.sum()
            if s > 0:
                cand /= s
            else:
                cand = a.copy()
            sc = local_score(cand)
            if sc < best_score:
                best, a = cand.copy(), cand.copy()
                best_score = sc
                last_improve = iter_cnt
                print(f"[block] #{iter_cnt} improved → {sc:.6f}")

        iter_cnt += 1

        # occasional random restart if we stall for a long time
        if iter_cnt - last_improve > 3000 and rng.random() < 0.25:
            # jitter the current best a little and continue from there
            jitter = np.exp(rng.normal(0.0, 0.05, size=n))
            cand = best * jitter
            cand = np.clip(cand, 0.0, 1000.0)
            cand /= cand.sum()
            a = cand
            last_improve = iter_cnt
            print(f"[restart] #{iter_cnt} – injected jitter")

    print(f"[final] best score = {best_score:.6f}")
    # The official evaluator rescales anyway, we keep the ℓ1‑normalised vector.
    return [float(x) for x in best]
