import time
import numpy as np

def catalan_sequence(length):
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq

def normalise_to_target_sum(seq, target_sum):
    s = seq.sum()
    if s == 0:
        return seq
    return seq * (target_sum / s)

def block_perturbation(base, rng):
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand

def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Gradient‑based refinement of the Catalan construction.
    The algorithm keeps the L1 norm fixed (target_sum = sqrt(2*n))
    and repeatedly reduces the maximal entry of the self‑convolution
    using a sub‑gradient step followed by a small stochastic perturbation.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------
    # Determine the length of the candidate.
    # ------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 0:
                raise ValueError
        except Exception:
            n = 256
    else:
        n = 256

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------
    # Initialise with the Catalan (optimal for the constant‑convolution
    # problem) and normalise it.
    # ------------------------------------------------------------
    seq = catalan_sequence(n)
    seq = normalise_to_target_sum(seq, target_sum)

    # Record the best sequence seen so far.
    conv = np.convolve(seq, seq)
    best_val = float(conv.max())
    best_seq = seq.copy()
    print(f"[init] n={n}, initial max convolution={best_val:.6f}")

    # ------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # initial step size – relatively large, will decay.
    eta = 0.5
    # simple counter to decay eta slower
    it = 0
    while time.monotonic() < deadline:
        conv = np.convolve(seq, seq)
        max_idx = int(conv.argmax())
        max_val = float(conv[max_idx])

        # --- Sub‑gradient step (multiplicative update) -----------------
        # Gradient of the active quadratic (a*a)_k  is 2*a_{k-i}.
        lo = max(0, max_idx - (n - 1))
        hi = min(n - 1, max_idx)
        idx = np.arange(lo, hi + 1)
        grad = np.zeros_like(seq)
        grad[idx] = 2.0 * seq[max_idx - idx]

        # multiplicative update keeps positivity automatically
        seq = seq * np.exp(-eta * grad)

        # clip to allowed range and renormalise to the target L1 norm
        np.clip(seq, 0.0, 1000.0, out=seq)
        cur_sum = seq.sum()
        if cur_sum > 0:
            seq *= (target_sum / cur_sum)

        # evaluate after the deterministic step
        conv = np.convolve(seq, seq)
        max_val = float(conv.max())
        if max_val < best_val:
            best_val = max_val
            best_seq = seq.copy()
            print(f"[improve] new best max={best_val:.6f}")

        # --- Occasional random perturbation to avoid plateaus -----
        if rng.random() < 0.02:
            cand = block_perturbation(seq, rng)
            conv_cand = np.convolve(cand, cand)
            cand_val = float(conv_cand.max())
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                seq = cand  # keep the improved candidate
                print(f"[random] improved to {best_val:.6f}")

        # --- Step‑size decay --------------------------------------
        it += 1
        # exponential decay every few hundred iterations
        if it % 200 == 0:
            eta *= 0.95
            if eta < 1e-4:
                eta = 1e-4

    print(f"[final] best max convolution={best_val:.6f}")
    # Convert to a plain list as required by the harness.
    return [float(x) for x in best_seq]

"""
Gradient‑based refinement of the Catalan step‑function.

The evaluation for a sequence whose L1 norm is fixed to
√(2·n) is simply its maximal self‑convolution entry, because
    evaluate = 2·n·max(a*a) / (∑a)²  and (∑a)² = 2·n.
The algorithm therefore works directly on the quantity
max_k (a*a)[k].

A Catalan sequence gives a constant convolution of value
π/2 ≈ 1.5708.  By repeatedly taking a sub‑gradient of the
active (maximal) convolution entries, applying a tiny
multiplicative update, re‑normalising the L1 norm and
occasionally performing a random block perturbation,
the maximal convolution can be driven below the Catalan
bound.  The whole search runs until the time budget is
exhausted, keeping the best sequence found.
"""

import time
import numpy as np


def catalan_sequence(length: int) -> np.ndarray:
    """First ``length`` terms of a_i = C(2i,i) / 4**i (Catalan‑type)."""
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        # recurrence a_{i} = a_{i-1} * (2*i-1) / (2*i)
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``seq`` so that its L1 norm becomes ``target_sum``."""
    s = seq.sum()
    if s == 0:
        return seq
    return seq * (target_sum / s)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) perturbation: pick a random contiguous block
    and multiply it by a log‑normal factor; sometimes symmetrise.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:                     # occasional symmetrisation
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step‑function (list of non‑negative floats) with a
    maximal self‑convolution smaller than the Catalan construction.
    The optimisation runs until the time budget expires and keeps the
    best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Length of the candidate – reuse the previously discovered length
    # if it is available; otherwise fall back to a modest size.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            n = int(np.asarray(height_sequence_1).size)
            if n <= 0:
                raise ValueError
        except Exception:
            n = 256
    else:
        n = 256

    target_sum = np.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # Initialise with the Catalan sequence (optimal for a constant
    # convolution) and normalise it to the required L1 norm.
    # ------------------------------------------------------------------
    seq = catalan_sequence(n)
    seq = normalise_to_target_sum(seq, target_sum)

    conv = np.convolve(seq, seq)
    best_val = float(conv.max())
    best_seq = seq.copy()
    print(f"[init] n={n}, initial max convolution={best_val:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    eta = 0.5          # initial multiplicative step size
    it = 0
    while time.monotonic() < deadline:
        # ---- 1) compute full convolution and identify all active ks ----
        conv = np.convolve(seq, seq)
        max_val = float(conv.max())

        # a tolerance to catch all indices achieving the maximum
        tol = max_val * 1e-12
        active = (conv >= max_val - tol).astype(np.float64)   # length 2n‑1

        # ---- 2) sub‑gradient for the max (sum of gradients of all active ks)
        #    grad_i = 2 * Σ_{k active} a_{k-i}
        grad_full = np.convolve(active, seq[::-1])           # length 3n‑2
        grad = 2.0 * grad_full[n - 1: n - 1 + n]            # keep only n entries

        # ---- 3) multiplicative update (keeps positivity automatically) ----
        seq = seq * np.exp(-eta * grad)

        # ---- 4) enforce the problem‑specific bounds --------------------
        np.clip(seq, 0.0, 1000.0, out=seq)                  # security limits
        cur_sum = seq.sum()
        if cur_sum > 0:
            seq *= (target_sum / cur_sum)                  # renormalise L1 norm

        # ---- 5) evaluate ------------------------------------------------
        conv = np.convolve(seq, seq)
        max_val = float(conv.max())
        if max_val < best_val:
            best_val = max_val
            best_seq = seq.copy()
            print(f"[improve] iteration {it:5d} new best max={best_val:.6f}")

        # ---- 6) occasional random jump to escape plateaus ---------------
        if rng.random() < 0.02:   # about 2 % of iterations
            cand = block_perturbation(seq, rng)
            conv_cand = np.convolve(cand, cand)
            cand_val = float(conv_cand.max())
            if cand_val < best_val:
                best_val = cand_val
                best_seq = cand.copy()
                seq = cand
                print(f"[random] iteration {it:5d} improved to {best_val:.6f}")

        # ---- 7) step‑size decay -----------------------------------------
        it += 1
        if it % 200 == 0:
            eta *= 0.95
            if eta < 1e-4:
                eta = 1e-4

    print(f"[final] best max convolution={best_val:.6f} (length={n})")
    # The harness expects a plain Python list of floats.
    return [float(x) for x in best_seq]
