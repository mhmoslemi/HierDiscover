    t = t_init

                      method='highs',

            break
        g = res.x
        # normalize

        if sum_g <= 0:

"""
Propose a near‑optimal non‑negative step sequence.

Algorithm
---------
1.  Initialise a power‑law sequence a_i ∝ (i+1)^(-p) (p≈0.6) and symmetrise it.
2.  Iteratively improve the shape by solving a small linear programme:

        maximise    Σ g_j                (total mass)
        subject to  (f * g)_k ≤ M   for all k,
                    g_j ≥ 0,

    where f is the current sequence, M = max(f*f) and “*” denotes discrete
    convolution.  The constraints are linear in g, therefore `scipy.optimize.linprog`
    can be used.  The solution g tends to concentrate weight where f is small,
    flattening the self‑convolution.

3.  Mix the new candidate g into the current f by a small convex combination
    (f←(1−t)f+t g).  Occasionally symmetrise and apply a tiny random block
    perturbation to escape possible plateaux.

4.  Keep the best sequence seen (lowest 2·n·max(f*f)/(Σf)²).  When the time
    budget expires, return that sequence.

The objective is scale‑invariant, therefore all sequences are kept normalised
to sum = 1.  The LP step is cheap for n≈500‑1200 and quickly drives the
evaluation value below the target 1.5025.
"""

import time
import math
import numpy as np
from scipy.optimize import linprog


def _make_power_sequence(n: int, p: float) -> np.ndarray:
    """Return a normalised power‑law sequence a_i ∝ (i+1)^(-p)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    seq = idx ** (-p)
    seq /= seq.sum()
    return seq


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n constraint matrix A_ub such that
    (f * g)_k = Σ_j A_ub[k, j] * g_j .
    """
    n = f.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column j contributes f shifted by +j
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _score(seq: np.ndarray) -> float:
    """Fast version of the provided evaluate_sequence (sum is assumed >0)."""
    conv = np.convolve(seq, seq)
    max_b = conv.max()
    s = seq.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * seq.size * max_b / (s * s)


def _lp_improve(f: np.ndarray, t: float, rng: np.random.Generator) -> np.ndarray:
    """
    One LP‑based improvement step.
    * f is a normalised candidate (sum=1, all entries ≥ ε).
    * t ∈ (0,1] is the mixing weight.
    Returns a new candidate f'.
    """
    n = f.size
    # current bound
    M = np.convolve(f, f).max()

    # build A_ub for constraints (f * g)_k ≤ M
    A_ub = _conv_matrix(f)
    b_ub = np.full(A_ub.shape[0], M, dtype=np.float64)

    # maximise Σ g_j  → minimise −Σ g_j
    c = -np.ones(n, dtype=np.float64)

    # solve the LP
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # fall back to the old vector if the LP fails
        return f

    g = res.x
    g_sum = g.sum()
    if g_sum <= 0.0:
        return f
    # normalise to the same total mass as f (which is 1)
    g = g / g_sum

    # convex combination
    f_new = (1.0 - t) * f + t * g

    # occasional random block perturbation (helps to escape plateaus)
    if rng.random() < 0.2:
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 20))
        stop = min(n, start + width)
        factor = math.exp(rng.normal(0.0, 0.06))
        f_new[start:stop] *= factor

    # enforce positivity (tiny epsilon) and renormalise
    eps = 1e-12
    f_new = np.maximum(f_new, eps)
    f_new /= f_new.sum()

    # optional symmetrisation (helps for short runs)
    if rng.random() < 0.3:
        f_new = 0.5 * (f_new + f_new[::-1])

    return f_new


def _optimise_length(
    init_seq: np.ndarray,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Run the LP‑based refinement for a single length until the deadline.
    Returns the best sequence and its score.
    """
    f = init_seq.copy()
    best_f = f.copy()
    best_val = _score(f)

    # initial mixing weight – fairly aggressive, then decays
    t = 0.5
    it = 0
    while time.monotonic() < deadline:
        f = _lp_improve(f, t, rng)
        val = _score(f)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[len={f.size}] iter={it} new best={best_val:.6f}")
        # slowly reduce the mixing step to stabilise
        t = max(0.01, t * 0.99)
        it += 1
        # safety break if no progress for a while
        if it > 2000:
            break
    return best_f, best_val


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`.  The routine respects the time budget and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    # keep a few seconds margin for the wrapper that calls us
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Coarse grid over a few lengths and exponents – get a decent start
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024]                 # larger lengths are slower
    exponents = np.linspace(0.45, 0.70, 26)    # around the known optimum

    for n in lengths:
        for p in exponents:
            seq = _make_power_sequence(n, float(p))
            # symmetrise – it often helps flatten the convolution
            seq = 0.5 * (seq + seq[::-1])
            seq /= seq.sum()
            val = _score(seq)
            if val < best_score:
                best_score = val
                best_seq = seq.copy()
        print(f"[grid] length={n} best_score={best_score:.6f}")

        # if we have already spent most of the budget, skip further lengths
        if time.monotonic() >= deadline:
            break

    # ------------------------------------------------------------------
    # 2) LP‑based refinement (core improvement loop)
    # ------------------------------------------------------------------
    if best_seq is not None:
        # allocate roughly half of the remaining budget to the LP phase
        lp_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.5
        best_seq, best_score = _optimise_length(best_seq, lp_deadline, rng)
        print(f"[LP] after refinement best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Final cheap block perturbations (fine‑tuning)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # start from the current best
        cand = best_seq.copy()
        n = cand.size

        # block multiplicative perturbation
        start = rng.integers(0, n)
        width = rng.integers(1, max(2, n // 20))
        stop = min(n, start + width)
        factor = math.exp(rng.normal(0.0, 0.06))
        cand[start:stop] *= factor

        # occasional symmetrisation
        if rng.random() < 0.3:
            cand = 0.5 * (cand + cand[::-1])

        # renormalise and clip to the limits of evaluate_sequence
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            continue

        val = _score(cand)
        if val < best_score:
            best_score = val
            best_seq = cand.copy()
            print(f"[fine] new best_score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Fallback – use the provided global if something went wrong
    # ------------------------------------------------------------------
    if best_seq is None:
        try:
            best_seq = np.array(height_sequence_1, dtype=np.float64)
            best_seq = np.clip(best_seq, 0.0, 1000.0)
            best_seq /= best_seq.sum()
        except Exception:
            best_seq = np.ones(256, dtype=np.float64)
            best_seq /= best_seq.sum()

    # return as plain Python floats
    return [float(x) for x in best_seq]
