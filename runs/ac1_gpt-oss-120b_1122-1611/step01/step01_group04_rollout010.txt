def propose_candidate(seed=123, budget_s=1000, **kwargs):
    rng=np.random.default_rng(seed)
    deadline = time.monotonic() + max(0, budget_s - 5.0)
    # Determine n
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 6000

    # Prepare initial candidates
    candidates = []

    # Uniform base
    base_uni = np.ones(n)
    candidates.append(base_uni)

    # Power law coarse grid
    for beta in np.arange(-0.8,-0.2+1e-9,0.05):
        for shift in np.linspace(0.1,10,10):
            cand = generate_power(beta, shift, n)
            candidates.append(cand)

    # Sine humps various gamma
    for gamma in [0.5,1,2,3]:
        cand = generate_sine(gamma,n)
        candidates.append(cand)

    # Triangle
    candidates.append(generate_triangle(n))

    # Random (normal) small values
    random_vec = np.abs(rng.normal(scale=0.1, size=n))
    candidates.append(random_vec)

    # Evaluate all candidates, keep best
    best = None; best_score = np.inf
    for cand in candidates:
        score = evaluate_sequence(cand.tolist())
        if np.isfinite(score) and score < best_score:
            best = cand.copy()
            best_score = score
            print(f"initial candidate score {score:.6f}")

    # Ensure best is not None else fallback to ones
    if best is None:
        best = np.ones(n)
        best_score = evaluate_sequence(best.tolist())

    iteration = 0
    while time.monotonic() < deadline:
        # choose either block perturbation or gradient-like improvement with certain probability
        if rng.random() < 0.6:
            # block multiplicative perturbation
            cand = make_block_perturbation(best, rng)
        else:
            # gradient-inspired reduction of max conv

            best_score = score
            print(f"iter {iteration} improved score {score:.6f}")
        iteration += 1

    print(f"final best score {best_score:.6f} after {iteration} iterations")
    return [float(x) for x in best]

def gradient_step_reduce_max(base, rng):
    # Compute convolution
    conv = np.convolve(base, base)
    k_max = int(np.argmax(conv))
    n = base.shape[0]

    # Compute w: complement of base reversed shifted by k_max
    w = np.zeros_like(base)
    # For i from max(0, k_max-(n-1)) to min(n-1, k_max):
    i_start = max(0, k_max - (n-1))
    i_end = min(n-1, k_max)
    # For i in that range, j = k_max - i is valid index
    # Set w[i] = base[j]
    idx_i = np.arange(i_start, i_end+1)
    idx_j = k_max - idx_i
    w[idx_i] = base[idx_j]

    # Compute contribution per index = base * w
    contrib = base * w

    # Determine top indices
    top_k = max(1, int(0.05 * n)) # 5% top
    # Get sorted indices by contribution descending
    top_inds = np.argpartition(contrib, -top_k)[-top_k:]

    # Determine bottom indices (low contributions)
    bottom_k = max(1, int(0.05 * n))
    low_inds = np.argpartition(contrib, bottom_k)[:bottom_k]

    # Transfer mass from top to bottom
    # Compute total reduction amount: fraction eta times values at top
    eta = 0.1  # move up to 10%
    reduction = np.zeros_like(base)
    reduction[top_inds] = eta * base[top_inds]
    total_transfer = reduction.sum()

    # Distribute to bottom equally
    addition = np.zeros_like(base)
    if bottom_k > 0:
        addition[low_inds] = total_transfer / bottom_k

    cand = base.copy()
    cand -= reduction
    cand += addition

    # Clip to bounds
    cand = np.clip(cand, 0.0, 1000.0)

    return cand

"""
Propose a coefficient sequence that beats the previous best (≈1.574).
The search proceeds in three stages:

1.   A set of deterministic seeds (uniform, triangular, sine‑humps,
     several power‑law shapes) gives a solid initial bound.
2.   A fast random sweep over the two‑parameter power‑law family
     a_i = (i + shift)^beta (beta negative, shift > 0).  Small shifts
     keep a_0 modest and prevent the a_0² term from dominating the
     convolution; more negative beta improves the continuous theoretic
     bound 2·B(1‑p,1‑p)(1‑p)² (p = –beta).  The sweep is limited only by the
     remaining time budget.
3.   The best candidate is refined with cheap multiplicative block
     perturbations and a simple “gradient‑step’’ that reduces the entry
     contributing most to the current maximal convolution value.
All operations respect the limits required by evaluate_sequence
(non‑negative, ≤1000).  The routine stops a few seconds before the
global timeout so that the final answer is always returned.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions – all top level, no closures
# ----------------------------------------------------------------------
def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """a_i = (i + shift) ** beta  (beta can be negative)."""
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_sine(gamma: float, n: int) -> np.ndarray:
    """a_i = sin(π·(i+0.5)/n)^gamma – symmetric hump."""
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    return np.clip(seq, 0.0, 1000.0)


def generate_triangle(n: int) -> np.ndarray:
    """Symmetric triangular shape – rises then falls linearly."""
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        seq = np.concatenate([half, half[::-1]])
    else:
        seq = np.concatenate([half, half[-2::-1]])
    return np.clip(seq, 0.0, 1000.0)


def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative perturbation of a contiguous block."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width between 0.5 % and 5 % of the length
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))  # log‑normal tweak
    cand[st:en] *= factor
    # occasional symmetrisation – useful for symmetric shapes
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def gradient_step_reduce_max(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    One simple sub‑gradient step.
    Compute the index k where the convolution attains its maximum.
    Decrease the entries that directly contribute to this maximum
    (i.e. a_i where a_{k-i} is large) and re‑distribute the removed mass
    to indices that contribute little.
    """
    n = base.size
    conv = np.convolve(base, base)
    k_max = int(np.argmax(conv))

    # contribution vector w_i = a_{k_max-i} (zero outside the support)
    w = np.zeros_like(base)
    i_start = max(0, k_max - (n - 1))
    i_end = min(n - 1, k_max)
    idx_i = np.arange(i_start, i_end + 1)
    idx_j = k_max - idx_i
    w[idx_i] = base[idx_j]

    # contribution of each position to the maximal entry
    contrib = base * w

    # pick a small fraction of the most‑contributing indices
    top_k = max(1, int(0.04 * n))
    top_idx = np.argpartition(contrib, -top_k)[-top_k:]

    # pick a small fraction of the least‑contributing indices
    bot_k = max(1, int(0.04 * n))
    bot_idx = np.argpartition(contrib, bot_k)[:bot_k]

    # move a modest amount of mass from the top to the bottom
    eta = 0.08  # step size
    loss = eta * base[top_idx].sum()
    cand = np.array(base, copy=True)
    cand[top_idx] *= (1.0 - eta)
    cand[bot_idx] += loss / bot_k

    # keep everything in the admissible range
    return np.clip(cand, 0.0, 1000.0)

    deadline = time.monotonic() + max(0.0, float(budget_s) - 6.0)

    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():

    # ------------------------------------------------------------------
    candidates = []

    candidates.append(np.ones(n, dtype=np.float64))

    # 3) sine‑humps with a few exponents
    for gamma in (0.5, 1.0, 2.0, 3.0):

    # 4) a few hand‑picked power‑laws
    for beta, shift in [(-0.5, 1.0), (-0.55, 0.7), (-0.6, 0.5), (-0.45, 2.0)]:

            best_score = score

    # ------------------------------------------------------------------
    # Phase 2 – randomized power‑law sweep (beta, shift).
    # ------------------------------------------------------------------
    # ranges based on theoretical insight:
    #   beta in [-1.0, -0.2]   (more negative → steeper decay)
    #   shift in [0.05, 20]   (small enough to keep a0 moderate)
    beta_low, beta_high = -1.0, -0.2
    shift_low, shift_high = 0.05, 20.0

    # run as many random draws as time permits

        if np.isfinite(score) and score < best_score:

            best_score = score
            print(f"refine #{iter_no} improved → {score:.6f}")
        iter_no += 1

    print(f"final best score = {best_score:.6f} after {iter_no} refinements")
    # Convert to plain Python floats for the caller.
    return [float(x) for x in best]
