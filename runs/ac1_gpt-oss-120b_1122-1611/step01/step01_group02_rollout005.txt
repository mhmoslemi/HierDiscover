"""
Iterative LP‑based refinement of a Catalan–type step function.

The evaluation metric for a sequence `a` with fixed L1‑norm `S = sqrt(2·n)`
is simply `M = max(a * a)` because  

    eval = 2·n·M / S² = M.

The classic Catalan construction gives `M ≈ π/2 ≈ 1.571`.  To push the
value below the target 1.5025 we iteratively **increase** the total L1‑norm
while keeping the *cross‑convolution* `f * g` below the current peak `M`.
The linear program

    maximise   sum(g_j)
    subject to  f * g ≤ M      (component‑wise)
                g_j ≥ 0

produces a sequence `g` whose L1‑norm is larger than `S`.  After rescaling
`g` back to `S` we mix it with the current `f` (`f ← (1‑t)·f + t·g`) – the
mixing does not change the total L1‑norm but tends to lower the peak of the
self‑convolution.  Repeating the step yields a monotone decrease of `M`.

The implementation uses a sparse Toeplitz matrix for the linear constraints
and solves the LP with SciPy's HiGHS interior‑point solver.  If SciPy is
unavailable we fall back to a tiny random‑perturbation search (the code
used in the reference solution).

The algorithm stops when the time budget is exhausted or no further
improvement is observed; the best sequence found is returned as a plain
list of Python floats (all entries are non‑negative and ≤ 1000, satisfying
the security checks of `evaluate_sequence`).
"""

import time

# Helper utilities (top‑level as required)
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """First `length` terms a_i = C(2i,i) / 4**i.
    Recurrence a_{i+1} = a_i * (2*i+1) / (2*i+2) provides stable values.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def normalise_to_target_sum(seq: np.ndarray, target_sum: float) -> np.ndarray:
    """Scale ``seq`` so that its L1‑norm equals ``target_sum``."""
    cur = seq.sum()
    if cur == 0.0:
        return seq
    factor = target_sum / cur
    return seq * factor


def convolution_max(seq: np.ndarray) -> float:
    """Return max of the linear convolution of ``seq`` with itself."""
    conv = np.convolve(seq, seq)
    return float(conv.max())


# ----------------------------------------------------------------------
# Sparse convolution matrix (used in the linear program)
# ----------------------------------------------------------------------
def conv_matrix_sparse(f: np.ndarray):
    """
    Build a sparse matrix A (shape (2n‑1, n)) such that
        A @ g = f * g   (full linear convolution).
    Each column j contains the vector f shifted by j positions.
    """
    from scipy.sparse import diags

    n = f.size
    rows = 2 * n - 1
    data = []
    offsets = []

    # For offset i (i = row – col) the diagonal consists of `f[i]`
    # repeated for the first n positions, the remainder is zero.
    for i, coeff in enumerate(f):
        diag_len = rows - i                     # length of diagonal with this offset
        diag = np.empty(diag_len, dtype=np.float64)
        # first n entries get the coefficient, the tail (if any) stays zero
        diag[:n] = coeff
        if diag_len > n:
            diag[n:] = 0.0
        data.append(diag)
        offsets.append(i)

    return diags(data, offsets, shape=(rows, n), format="csr")


def solve_lp_max_sum(f: np.ndarray, M: float, target_sum: float):
    """
    Solve the linear program
        maximise   sum(g)
        subject to  f * g ≤ M   (component‑wise)
                    g ≥ 0
    and return the optimal `g`.  If SciPy is unavailable or the solver fails,
    ``None`` is returned.
    """
    try:
        from scipy.optimize import linprog
        from scipy.sparse import csr_matrix
    except Exception:                # SciPy not present
        return None

    n = f.size
    A = conv_matrix_sparse(f)        # (2n‑1) × n sparse matrix
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # maximise sum(g)  →  minimise –sum(g)
    c = -np.ones(n, dtype=np.float64)

    # variable bounds: 0 ≤ g_j ≤ 1000 (the same caps used in evaluate_sequence)
    bounds = [(0.0, 1000.0) for _ in range(n)]

    # HiGHS (default) works with sparse matrices
    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Main entry point required by the harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a step‑function (list of non‑negative floats) with a low
    evaluation score.  The routine starts from the Catalan construction,
    then repeatedly solves a small linear program to enlarge the L1‑norm
    under the current convolution peak, rescales the solution to keep the
    L1‑norm fixed, and mixes it back into the current sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a manageable length.  If a previous construction exists we
    # respect its size up to a hard cap; otherwise we default to 1800.
    # ------------------------------------------------------------------
    max_n = 1800                     # comfortably fits into memory for the LP
    if "height_sequence_1" in globals():
        try:
            n_raw = int(np.asarray(height_sequence_1).size)
            n = min(max_n, max(10, n_raw))
        except Exception:
            n = max_n
    else:
        n = max_n

    # ------------------------------------------------------------------
    # Baseline: Catalan sequence, normalised to L1 = sqrt(2·n)
    # ------------------------------------------------------------------
    base = catalan_sequence(n)
    target_sum = np.sqrt(2.0 * n)
    f = normalise_to_target_sum(base, target_sum)

    # Initial evaluation (for logging)
    try:
        best_score = float(evaluate_sequence(f.tolist()))
    except Exception:               # safety net – should never happen
        best_score = float("inf")
    best_seq = f.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Iterative LP refinement.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    # mixing parameter – start relatively aggressive and fade out
    t = 0.5
    it = 0
    while time.monotonic() < deadline:
        it += 1
        # Current peak of the self‑convolution
        M = convolution_max(f)

        # Solve the LP: maximise total mass while keeping f * g ≤ M
        g_raw = solve_lp_max_sum(f, M, target_sum)
        if g_raw is None:
            # LP not available or failed – fall back to cheap random tweaks
            width = max(1, n // 150)
            start = rng.integers(0, n - width + 1)
            cand = f.copy()
            cand[start:start + width] *= np.exp(rng.normal(0.0, 0.08))
            np.clip(cand, 0.0, 1000.0, out=cand)
            new_score = float(evaluate_sequence(cand.tolist()))
            if new_score < best_score:
                best_score = new_score
                best_seq = cand.copy()
                print(f"[rnd] improved to {best_score:.6f}")
            continue

        sum_g = float(g_raw.sum())
        if sum_g <= 0.0:
            break

        # Rescale to preserve the required L1‑norm (sqrt(2·n))
        factor = target_sum / sum_g
        g = g_raw * factor
        np.clip(g, 0.0, 1000.0, out=g)

        # Convex combination – keeps the L1‑norm unchanged
        f = (1.0 - t) * f + t * g

        # Numerical drift may move the sum away from the target – re‑normalise
        cur_sum = f.sum()
        if cur_sum <= 0.0:
            break
        f *= (target_sum / cur_sum)

        # Evaluate the candidate
        try:
            cur_score = float(evaluate_sequence(f.tolist()))
        except Exception:
            cur_score = float("inf")

        # Keep the best sequence seen so far
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[it {it:3d}] new best score = {best_score:.6f}")

        # Reduce the mixing step gradually (helps convergence)
        t = max(0.05, t * 0.92)

        # Early stop if we have already passed the target
        if best_score <= 1.5025:
            print("[stop] target reached")
            break

    print(f"[final] returning sequence length {n} with best score {best_score:.6f}")
    return [float(x) for x in best_seq]
