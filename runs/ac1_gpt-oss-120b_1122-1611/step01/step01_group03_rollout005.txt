def conv_full(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, n=N)
    fb = np.fft.rfft(b, n=N)
    fc = fa * fb
    c = np.fft.irfft(fc, n=N)[:n]
    return c

"""
Propose a low‑score step‑function using a smoothed max‑convolution
objective and exponentiated (mirror) gradient descent on the simplex.
The algorithm

* starts from the classic α‑sequence (or from the supplied
  ``height_sequence_1`` if it exists),
* repeatedly evaluates the self‑convolution,
* builds a soft‑max weighting of the convolution entries,
* computes the corresponding sub‑gradient of the smoothed max,
* performs a multiplicative (exponentiated‑gradient) update on the
  probability vector, re‑normalising to keep the L1‑norm equal to 1,
* occasionally adds cheap random perturbations (block scaling or
  Gaussian noise) and accepts uphill moves with a tiny Metropolis
  temperature that anneals to zero.

All calculations use FFT based convolutions (O(n log n)) so that a
large number of iterations fits into the 1000 s budget.  The best
sequence found during the run is returned as a plain Python list.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def generate_alpha_sequence(n: int) -> np.ndarray:
    """
    First n terms of a_i = C(2i,i) / 4^i .
    The recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1))  is used.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def conv_full_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """
    Full linear convolution (len(x)+len(y)-1) using real FFT.
    """
    n = len(x) + len(y) - 1
    N = 1 << (n - 1).bit_length()          # next power of two
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    fz = fx * fy
    z = np.fft.irfft(fz, n=N)
    return z[:n]


def conv_valid_fft(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Valid part of the convolution p * a.
    With len(p)=2n‑1 and len(a)=n, the valid region has length n and is
    obtained by slicing the full convolution from index n‑1 to 2n‑1.
    """
    full = conv_full_fft(p, a)
    n = len(a)
    start = n - 1
    end = start + n
    return full[start:end]


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs) -> list:
    """
    Search for a non‑negative sequence with a small value of
    ``evaluate_sequence``.  The routine runs until the supplied time budget
    (minus a 5 s safety margin) expires and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Determine problem size and initialise a starting sequence
    # --------------------------------------------------------------
    seed_seq = globals().get('height_sequence_1')
    if isinstance(seed_seq, (list, np.ndarray)) and len(seed_seq) > 0:
        n = int(len(seed_seq))
        a = np.asarray(seed_seq, dtype=np.float64).copy()
        a = np.maximum(a, 0.0)                # enforce non‑negativity
        s = a.sum()
        if s < 1e-12:                         # fall back if degenerate
            a = generate_alpha_sequence(n)
            a = a / a.sum()
        else:
            a = a / s
    else:
        n = 5000                               # a safe default size
        a = generate_alpha_sequence(n)
        a = a / a.sum()

    # maintain the best sequence seen so far
    best_seq = a.copy()
    b_full = conv_full_fft(a, a)
    best_score = 2.0 * n * b_full.max()
    print(f"[init] length={n}, initial score={best_score:.9f}")

    # --------------------------------------------------------------
    # Hyper‑parameters (tuned for the 1000 s budget)
    # --------------------------------------------------------------
    eta = 0.12                # step size for the exponentiated‑gradient update
    beta = 1.0                # smoothing parameter for the soft‑max (β·(b‑max))
    beta_mul = 1.001          # slow increase of β (sharper approximation of max)
    beta_max = 12.0
    temp = 1e-3               # Metropolis temperature for uphill moves
    temp_mul = 0.9995         # exponential cooling of the temperature
    momentum = 0.9            # exponential moving average for the gradient
    grad_avg = np.zeros_like(a)

    cur_seq = a.copy()
    cur_score = best_score

    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ----------------------------------------------------------
        # 1) current objective (full convolution) and bookkeeping
        # ----------------------------------------------------------
        b_full = conv_full_fft(cur_seq, cur_seq)
        max_b = b_full.max()
        cur_score = 2.0 * n * max_b

        if cur_score < best_score:
            best_score = cur_score
            best_seq = cur_seq.copy()
            print(f"[{it:6d}] new best = {best_score:.9f}")

        # ----------------------------------------------------------
        # 2) soft‑max weighting of the convolution entries
        # ----------------------------------------------------------
        # shift by max_b to keep the exponent ≤ 0 (prevents overflow)
        w = np.exp(beta * (b_full - max_b))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.ones_like(w) / w.size
        else:
            w = w / w_sum

        # ----------------------------------------------------------
        # 3) gradient of the smoothed max (valid convolution)
        # ----------------------------------------------------------
        grad = 2.0 * conv_valid_fft(w, cur_seq)          # shape (n,)

        # momentum smoothing
        grad_avg = momentum * grad_avg + (1.0 - momentum) * grad

        # ----------------------------------------------------------
        # 4) exponentiated‑gradient (mirror) update on the simplex
        # ----------------------------------------------------------
        log_seq = np.log(cur_seq + 1e-30)                # stay in log‑space
        log_new = log_seq - eta * grad_avg

        # numerical stability: subtract maximum before exponentiation
        log_new -= np.max(log_new)
        cand = np.exp(log_new)

        # enforce a tiny floor to avoid exact zeros (helps the next step)
        cand = np.maximum(cand, 1e-12)
        cand = cand / cand.sum()

        # ----------------------------------------------------------
        # 5) occasional random perturbations for exploration
        # ----------------------------------------------------------
        if rng.random() < 0.03:                         # Gaussian noise
            cand *= 1.0 + 0.10 * rng.standard_normal(n)
            cand = np.clip(cand, 0.0, None)

        if rng.random() < 0.02:                         # block scaling
            width = int(rng.integers(max(1, n // 300), max(2, n // 30)))
            start = int(rng.integers(0, max(1, n - width + 1)))
            factor = float(np.exp(rng.normal(0.0, 0.07)))
            cand[start:start + width] *= factor

        # re‑normalise after perturbations
        cand = np.clip(cand, 0.0, None)
        cand_sum = cand.sum()
        if cand_sum < 1e-12:
            continue
        cand = cand / cand_sum

        # ----------------------------------------------------------
        # 6) evaluate the proposal
        # ----------------------------------------------------------
        b_cand = conv_full_fft(cand, cand)
        cand_score = 2.0 * n * b_cand.max()

        # ----------------------------------------------------------
        # 7) accept/reject (Metropolis rule)
        # ----------------------------------------------------------
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # very small probability of accepting a worse move
            prob = np.exp(-(cand_score - cur_score) / max(temp, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = cand
            cur_score = cand_score

        # ----------------------------------------------------------
        # 8) schedule updates
        # ----------------------------------------------------------
        beta = min(beta_max, beta * beta_mul)           # sharpen soft‑max
        temp = max(1e-8, temp * temp_mul)               # cool Metropolis temp

        if it % 2000 == 0:
            eta *= 0.95                                 # slowly decay step

        if it % 5000 == 0:
            print(f"[{it}] current score={cur_score:.9f}, best={best_score:.9f}")

    print(f"[final] iterations={it}, best score={best_score:.9f}")
    return best_seq.tolist()
