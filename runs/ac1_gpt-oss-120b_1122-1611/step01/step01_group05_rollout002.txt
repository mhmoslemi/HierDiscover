"""
propose_candidate
=================

A three‑stage optimizer for the security evaluation function.

1. **Fast analytical sweep** – we explore families
   a[i] = (i+α)^(-p)  (power‑law decay) with a modest grid of
   (p,α,n).  The best candidate from this inexpensive scan is kept.

2. **Linear‑programming refinement** – for the current sequence `a`
   we compute M = max(a*a).  With `a` fixed, the convolution
   `a * x` is linear in the unknown `x`.  We therefore solve the LP

        maximise   sum(x)
        subject to a * x ≤ M   (elementwise)
                   x ≥ 0

   using ``scipy.optimize.linprog``.  The solution increases the total
   sum while keeping the peak convolution unchanged, thus decreasing
   the evaluation score.  The new vector is blended with the old one
   (α = 0.5) and the process repeats until no substantial gain or the
   time budget is exhausted.

3. **Light local random search** – a final short phase of random
   block perturbations, swaps and smoothing is performed to polish the
   result.

All steps respect the security checks of ``evaluate_sequence``:
non‑negative entries, clipping to ``[0,1000]`` and no I/O.
The routine stops a few seconds before the allotted 1000 s to avoid a
forced termination.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: generate a power‑law / shifted power‑law sequence
def make_powlaw(p: float, shift: float, n: int) -> np.ndarray:
    """Return a non‑negative vector a[i] = (i+1+shift)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64) + shift
    seq = 1.0 / np.power(idx, p)
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Helper: construct the convolution constraint matrix for a fixed a
def conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1) × n matrix A such that (A @ x)[k] = (a * x)[k].
    The matrix is dense but for n ≤ 2000 the memory consumption is modest.
    """
    n = a.size
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # Fill each row k with the relevant slice of a
    for k in range(rows):
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # a[lo:hi+1] contributes to row k
        A[k, lo : hi + 1] = a[lo : hi + 1]
    return A


# ----------------------------------------------------------------------
# Linear‑programming step: maximise sum(x) under a*x ≤ M
def lp_improve(a: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP that maximises the sum of a new vector x while keeping
    the convolution a*x below the current peak M.  Returns the optimal x
    or None if the solver fails.
    """
    n = a.size
    A = conv_matrix(a)                 # shape (2n‑1, n)
    ub = np.full(A.shape[0], M)        # upper bound for each convolution entry
    bounds = [(0.0, None) for _ in range(n)]  # x ≥ 0, no explicit upper bound
    c = -np.ones(n, dtype=np.float64)  # maximise sum(x)  → minimise -sum(x)

    # Use HiGHS via scipy (fast for dense LPs of this size)
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method="highs", options={"presolve": True})
    if res.success:
        # Clip to the limits enforced by evaluate_sequence
        return np.clip(res.x, 0.0, 1000.0)
    return None


# ----------------------------------------------------------------------
# Random block perturbation (same idea as in the reference code)
def block_perturb(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Apply a cheap O(n) multiplicative perturbation to a sub‑block."""
    cand = base.copy()
    n = cand.size
    # block width between 0.5 % and 8 % of the length
    min_w = max(1, n // 200)
    max_w = max(2, n // 12)
    width = int(rng.integers(min_w, max_w + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = min(n, start + width)
    cand[start:stop] *= np.exp(rng.normal(0.0, 0.12))
    # occasional mirroring – helps flatten the convolution peak
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 990, **kwargs):
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    the security evaluation function.  The procedure consists of an analytical
    sweep, an LP‑based refinement and a short stochastic polishing phase.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine a reasonable problem size.
    # If a reference sequence is available we reuse its length,
    # otherwise we start with a modest size.
    if "height_sequence_1" in globals():
        try:
            arr = np.asarray(height_sequence_1, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0 and np.isfinite(arr).all():
                base_len = int(arr.size)
            else:
                base_len = 1024
        except Exception:
            base_len = 1024
    else:
        base_len = 1024

    # Allow a few different lengths but stay below ~2000 (memory‑friendly)
    candidate_lengths = sorted(set([base_len,
                                   max(256, base_len // 2),
                                   min(2000, base_len * 2),
                                   512, 1024, 2048]))

    best = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) FAST PARAMETER SWEEP ------------------------------------------------
    start_time = time.monotonic()
    for n in candidate_lengths:
        # grid for p and shift – a little finer than the previous run
        for p in np.linspace(0.20, 0.80, 13):          # 0.20 … 0.80 (step ≈0.05)
            for shift in np.linspace(0.0, 8.0, 9):    # 0 … 8 (step 1)
                cand = make_powlaw(p, shift, n)
                # quick check – skip if sum is too tiny (evaluate would reject)
                if np.sum(cand) < 0.05:
                    continue
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best, best_score = cand, score
                    print(f"[sweep] n={n:4d} p={p:.3f} shift={shift:4.1f} → score {score:.6f}")

        # also try a simple sqrt‑decay family (α‑shifted)
        for shift in np.linspace(0.0, 12.0, 13):
            idx = np.arange(1, n + 1, dtype=np.float64) + shift
            cand = 1.0 / np.sqrt(idx)
            cand = np.clip(cand, 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = cand, score
                print(f"[sweep] sqrt n={n:4d} shift={shift:4.1f} → score {score:.6f}")

        # abort sweep early if we are close to the time limit
        if time.monotonic() - start_time > budget_s * 0.4:
            break

    # ------------------------------------------------------------------
    # If the global reference sequence exists we also try it as a seed
    if "height_sequence_1" in globals():
        try:
            cand = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best, best_score = cand, score
                print(f"[seed] provided sequence → score {score:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 2) LINEAR‑PROGRAMMING REFINEMENT ------------------------------------
    # This stage repeatedly maximises the sum while keeping the peak
    # convolution unchanged.  The improvement is dramatic for power‑law
    # shapes.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)  # keep a margin
    while time.monotonic() < deadline:
        # current convolution peak
        conv = np.convolve(best, best)
        M = float(np.max(conv))
        S_before = float(np.sum(best))

        # solve the LP
        new_vec = lp_improve(best, M)
        if new_vec is None:
            break   # solver failure – give up LP phase

        S_after = float(np.sum(new_vec))
        # if the sum did not increase appreciably stop the LP loop
        if S_after <= S_before * (1.0 + 1e-4):
            break

        # blend the old and the LP solution (t = 0.5 gives stability)
        blended = 0.5 * best + 0.5 * new_vec
        blended = np.clip(blended, 0.0, 1000.0)

        new_score = float(evaluate_sequence(blended.tolist()))
        if np.isfinite(new_score) and new_score < best_score:
            best, best_score = blended, new_score
            print(f"[lp] improved to {new_score:.6f} (sum {S_after:.3f}, M {M:.3f})")
        else:
            # No improvement – exit LP loop
            break

    # ------------------------------------------------------------------
    # 3) LIGHT RANDOM SEARCH (polishing) ---------------------------------
    # If time is still left we perform a few cheap stochastic moves.
    while time.monotonic() < deadline:
        move = rng.choice(["block", "swap", "smooth"])
        n = best.size
        if move == "block":
            cand = block_perturb(best, rng)
        elif move == "swap":
            cand = best.copy()
            i, j = rng.integers(0, n, size=2)
            if i != j:
                delta = rng.uniform(0.0, min(cand[i], 0.5 * cand[i] + 1e-12))
                cand[i] -= delta
                cand[j] += delta
            cand = np.clip(cand, 0.0, 1000.0)
        else:   # smooth
            cand = 0.5 * (best + best[::-1])
            jitter = np.exp(rng.normal(0.0, 0.04, size=n))
            cand = cand * jitter
            cand = np.clip(cand, 0.0, 1000.0)

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            print(f"[search] new best {sc:.6f} via {move}")

    # ------------------------------------------------------------------
    print(f"[done] best_score={best_score:.6f}  length={best.size}")
    return [float(x) for x in best]
