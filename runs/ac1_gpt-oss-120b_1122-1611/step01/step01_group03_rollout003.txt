import numpy as np, scipy.sparse as sp, scipy.optimize as opt

def solve_lp(f, M):
    n = len(f)
    # Build T: rows = 2n-1, columns = n
    # Use sp.diags for each diagonal offset d (0..n-1) with value f[d].
    diagonals = [np.full(n, f[d]) for d in range(n)]
    offsets = np.arange(n)  # 0..n-1
    T = sp.diags(diagonals, offsets, shape=(2*n-1, n), format='csr')
    # In addition, we need to impose also constraints for mirrored side? Actually f*g <= M for all k from 0..2n-2 created by T.
    # Have:
    c = -np.ones(n)  # maximize sum(g) -> minimize -sum(g)
    bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=T, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    return res.x
```

def solve_lp(f, M):
    n = len(f)
    # Build sparse constraint matrix T (2n-1 x n)
    # Each diagonal offset d (0..n-1) with constant f[d]
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    pos = 0
    for d in range(n):
        rows[pos:pos + n] = np.arange(d, d + n)
        cols[pos:pos + n] = np.arange(n)
        data[pos:pos + n] = f[d]
        pos += n
    T = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    c = -np.ones(n)
    bounds = [(0, None)] * n
    ub = np.full(2 * n - 1, M, dtype=np.float64)
    res = opt.linprog(c, A_ub=T, b_ub=ub, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None

"""
Propose a candidate step‑function using the linear‑programming refinement
described in the original paper.

The algorithm works with the following ideas:

1.  Keep the L¹‑norm fixed to   S = √(2·n) .
    With this normalisation the evaluation function becomes exactly the
    maximal self‑convolution value  M = max((f∗f)[k]).
2.  For a current sequence f compute   M = max(f∗f).
3.  Solve the linear programme
        maximise   Σ g_i
        subject to  (f∗g)[k] ≤ M   for all k,
                    g_i ≥ 0 .
    The constraints are linear in g because f is fixed.  The solution g
    adds as much L¹ mass as possible while never violating the current
    bound M.
4.  Rescale g so that Σ g_i = S (the same L¹‑norm as f) and form a convex
    combination   f←(1‑t)·f + t·g   (t≈0.1–0.3).  The theory guarantees that the
    new maximal convolution Mʹ ≤ M, i.e. the objective never grows.
5.  Repeat the LP‑step a few times (as long as the time budget allows).
    If an LP step fails to improve the score, a cheap random perturbation
    is tried as a fallback.

Because the LP matrix is a Toeplitz matrix built from f, we construct it
once per iteration using a sparse CSR representation (O(n²) non‑zeros,
still feasible for n≈3000).  The implementation falls back to the simple
evolutionary routine when the LP becomes too costly.

The returned list is a non‑negative sequence that (hopefully) has a lower
evaluation value than the previous best (target < 1.5025).
"""

def generate_alpha_sequence(n: int) -> np.ndarray:
    """
    Return the first n coefficients of the series
        a_i = C(2i,i) / 4^i .
    The recurrence a_{i+1} = a_i * (2*i+1) / (2*(i+1)) is used.
    """
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def build_toeplitz_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix T such that (T @ g) = f ∗ g.
    Only non‑zero entries are stored (lower triangular Toeplitz).
    """
    n = f.size
    # each diagonal offset d (0 … n‑1) holds the constant value f[d]
    # for offset d the diagonal has length n and starts at row d
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)

    pos = 0
    col_idx = np.arange(n, dtype=np.int32)          # 0 … n‑1
    for d in range(n):
        # rows for this diagonal are d … d+n‑1
        rows[pos:pos + n] = np.arange(d, d + n, dtype=np.int32)
        cols[pos:pos + n] = col_idx
        data[pos:pos + n] = f[d]
        pos += n

    T = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return T


def solve_lp_max_l1(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the LP
        maximise   Σ g_i
        subject to (f ∗ g)[k] ≤ M   for all k,
                    g_i ≥ 0 .
    Returns the optimal g (may contain very small negative entries due
    to numerical tolerance – they are clipped to zero afterwards).
    """
    n = f.size
    T = build_toeplitz_matrix(f)

    # objective: minimise -Σ g_i  (so maximise Σ g_i)
    c = -np.ones(n, dtype=np.float64)

    # upper bounds for all convolutions
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # variable bounds: g_i ≥ 0, no explicit upper bound
    bounds = [(0.0, None) for _ in range(n)]

    # use HiGHS – works with sparse matrices
    res = opt.linprog(c, A_ub=T, b_ub=ub, bounds=bounds,
                      method='highs', options={'presolve': True})
    if res.success:
        g = np.maximum(res.x, 0.0)           # clip tiny negatives
        return g
    else:
        return None


def evaluate(seq: np.ndarray) -> float:
    """Wrapper around the provided evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))

# Main entry point required by the competition harness

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a sequence with a low evaluation score.
    The method follows the LP‑based refinement of the “alpha’’ sequence.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Initialise the base sequence
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals() \
            and isinstance(height_sequence_1, (list, np.ndarray)) \
            and len(height_sequence_1) > 0:
        init = np.asarray(height_sequence_1, dtype=np.float64)
        n = init.size
    else:
        n = 3000                      # a reasonable default size
        init = generate_alpha_sequence(n)

    # Target L1 norm = √(2·n)  (the normalisation used in the paper)
    target_sum = np.sqrt(2.0 * n)
    init_sum = init.sum()
    if init_sum == 0.0:
        raise RuntimeError("Initial sequence has zero L1 sum.")
    f = init * (target_sum / init_sum)           # scale to target_sum
    f = np.clip(f, 0.0, None)                    # safety

    best_f = f.copy()
    best_score = evaluate(best_f)
    best_M = np.convolve(best_f, best_f).max()
    print(f"[init] n={n}, score={best_score:.6f}, max_conv={best_M:.6f}")

    # ------------------------------------------------------------------
    # Search loop – we try a handful of LP refinements within the time budget
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)   # reserve 5 s for cleanup
    iteration = 0
    # mixing step size – will be reduced if progress stalls
    t_mix = 0.25

    # keep a very cheap random‐perturbation routine as a fallback
    def random_perturb(x: np.ndarray) -> np.ndarray:
        cand = x * (1.0 + 0.12 * rng.standard_normal(x.size))
        cand = np.clip(cand, 0.0, None)
        cand = cand * (target_sum / cand.sum())
        return cand

    while time.monotonic() < deadline:
        iteration += 1
        # ------------------------------------------------------------------
        # 1) compute current bound M
        # ------------------------------------------------------------------
        M = np.convolve(f, f).max()

        # ------------------------------------------------------------------
        # 2) LP – maximise L1 while keeping convolution with current f ≤ M
        # ------------------------------------------------------------------
        g = solve_lp_max_l1(f, M)
        if g is None:
            print(f"[{iteration}] LP failed – falling back to random walk.")
            f_candidate = random_perturb(f)
        else:
            # rescale g to have the same L1 norm as f (target_sum)
            g = g * (target_sum / g.sum())

            # ------------------------------------------------------------------
            # 3) Convex combination (mixing).  Try a few step sizes.
            # ------------------------------------------------------------------
            improved = False
            for step in (t_mix, t_mix * 0.6, t_mix * 0.3):
                cand = (1.0 - step) * f + step * g
                cand = np.clip(cand, 0.0, None)
                # both f and g already have the exact target L1 sum,
                # the combination therefore also has it up to numerical noise.
                cand = cand * (target_sum / cand.sum())
                cand_score = evaluate(cand)
                if cand_score < best_score:
                    best_f = cand
                    best_score = cand_score
                    f = cand
                    best_M = np.convolve(best_f, best_f).max()
                    print(f"[{iteration}] new best score={best_score:.6f} (step={step:.3f})")
                    improved = True
                    # slowly shrink the mixing step when we make progress
                    t_mix = min(0.4, t_mix * 0.95)
                    break
            if not improved:
                # No improvement – try a tiny random perturbation to escape plateau
                f_candidate = random_perturb(f)
                cand_score = evaluate(f_candidate)
                if cand_score < best_score:
                    best_f = f_candidate
                    best_score = cand_score
                    f = f_candidate
                    best_M = np.convolve(best_f, best_f).max()
                    print(f"[{iteration}] random perturb improved score={best_score:.6f}")
                    # keep mixing step size unchanged
                else:
                    # If nothing works, decay the mixing step a little
                    t_mix = max(0.02, t_mix * 0.9)
        # ------------------------------------------------------------------
        # Stop early if we already passed the target threshold
        # ------------------------------------------------------------------
        if best_score < 1.5025:
            print(f"[target] reached score {best_score:.6f} at iter {iteration}")
            break

    print(f"[final] iterations={iteration}, best_score={best_score:.6f}")
    # The evaluation routine expects a list; return the best sequence we found.
    return best_f.tolist()
