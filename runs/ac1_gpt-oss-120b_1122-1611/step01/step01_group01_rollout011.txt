def propose_candidate(seed=42, budget_s=1100):
    import numpy as np, time

    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)

    # Get n:
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        n = base.size
    except Exception:
        n = 2048

    # Precompute FFT length
    fft_len = _fft_len_for_n(n)

    # Functions defined at top-level: _fft_len_for_n, _conv_self, _partner_vector, _random_block_perturb, _pairwise_swap_perturb

    # initial candidates: arcsine shape or uniform:
    def init_arcsine():
        xs = np.arange(n, dtype=np.float64) + 0.5
        # denominator sqrt(i * (n - i + 1))? transform:
        den = np.sqrt(xs * (n - xs + 1))
        a = 1.0 / den
        a = np.clip(a, 0, None)
        a_sum = a.sum()
        if a_sum == 0:
            a = np.full(n, 1.0/n)
        else:
            a /= a_sum
        return a

    def init_uniform():
        return np.full(n, 1.0/n, dtype=np.float64)

    a = init_arcsine()
    # optionally add small random noise:
    a += rng.normal(scale=0.001, size=n)
    a = np.clip(a, 0, None)
    a_sum = a.sum()
    a /= a_sum

    best = a.copy()
    best_score = 2.0 * n * np.max(_conv_self(best, fft_len))
    start_time = time.monotonic()
    elapsed = 0.0
    # parameters
    top_k = 5
    eta0 = 1.0
    # iteration counters for stagnation detection
    last_improve_time = start_time
    stagnation_limit = 30.0  # seconds without improvement -> restart
    iteration = 0

    while time.monotonic() < deadline:
        iteration += 1
        # compute convolution
        conv = _conv_self(a, fft_len)
        max_val = np.max(conv)
        # get top_k indices
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            # α: partition for performance
            part = np.argpartition(-conv, top_k-1)[:top_k]
            # optionally sort descending for stable order
            top_idx = part[np.argsort(-conv[part])]
        # compute gradient sum of partner vectors
        w = np.zeros_like(a)
        for k in top_idx:
            w += _partner_vector(a, int(k))
        # compute step size adaptively based on elapsed proportion
        t = (time.monotonic() - start_time) / (deadline - start_time)
        eta = eta0 * (1 - t)  # linear decay
        # apply exponentiated gradient
        factor = np.exp(-eta * w)
        a = a * factor
        # renormalize
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

                a[j] += delta

                a /= a.sum()
        # evaluate score
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * n * np.max(conv)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve_time = time.monotonic()
            print(f"[{last_improve_time - start_time:6.2f}s] new best score = {best_score:.6f}")

* Apply an exponentiated‑gradient step with a slowly decaying step size.
* Occasionally perform cheap stochastic perturbations (block scaling,
  pairwise mass swaps) to escape plateaus.
* Keep the best vector seen so far and restart from a fresh random point
  if no improvement occurs for a while.

The loop runs for the allotted time (≈ 1000 s) and finally returns the
best non‑negative step‑function as a plain Python list of floats.
"""

import time
import numpy as np
from scipy.sparse import diags
from scipy.optimize import linprog  # only imported – not used in the fast path


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _fft_len_for_n(n: int) -> int:
    """Return the next power‑of‑two length that can hold a full linear convolution."""
    size = 2 * n - 1
    return 1 << ((size - 1).bit_length())


def _conv_self(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a (real, non‑negative) using FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _partner_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    For a given convolution index k, return the vector w with
        w_i = a_{k-i}   (0 if the index is out of range).

    This is the (sub‑)gradient of (a*a)[k] with respect to a.
    """
    n = a.size
    w = np.zeros_like(a)

    if k < n:
        # i ranges from 0 … k
        w[: k + 1] = a[k :: -1]          # a[k], a[k-1], …, a[0]
    else:
        # i ranges from k-(n-1) … n-1
        start = k - (n - 1)
        length = (2 * n - 1) - k
        rev = a[::-1]
        w[start:] = rev[:length]
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _pairwise_swap_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Move a small random fraction of mass from one index to another."""
    n = a.size
    i = rng.integers(0, n)
    j = rng.integers(0, n)
    if i == j:
        return a
    # take up to 30 % of a[i] and give it to a[j]
    delta = rng.random() * 0.3 * a[i]
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = np.clip(out, 0.0, None)
    out /= out.sum()
    return out


def _init_arcsine(n: int) -> np.ndarray:
    """
    Initialise with a discrete version of the arcsine density:
        a_i ∝ 1 / sqrt( (i+½)·(n‑i‑½) )
    The shape has a very flat self‑convolution in the interior.
    """
    xs = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(xs * (n - xs + 1.0))
    a = 1.0 / den
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a_sum
    return a


def _init_uniform(n: int) -> np.ndarray:
    """Plain uniform start."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights that (empirically) minimises the
    provided `evaluate_sequence` score.  The routine runs for roughly
    `budget_s` seconds (leaving a tiny safety margin) and uses a hybrid
    exponentiated‑gradient / stochastic search as described in the module
    docstring.
    """
    rng = np.random.default_rng(seed)
    # leave a small safety margin for the harness
    deadline = time.monotonic() + max(0.0, float(budget_s) - 1.5)

    # ------------------------------------------------------------------
    # Determine problem size and initialise a candidate vector
    # ------------------------------------------------------------------
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.ndim != 1 or base.size == 0 or not np.isfinite(base).all():
            raise ValueError
        n = base.size
    except Exception:
        # fallback to a modest size if the global is missing
        n = 2048
        base = None

    a = _init_arcsine(n)
    # add a little random jitter to break perfect symmetry
    a += rng.normal(scale=0.001, size=n)
    a = np.clip(a, 0.0, None)
    a /= a.sum()

    # Pre‑computations for the fast convolution
    fft_len = _fft_len_for_n(n)

    # ------------------------------------------------------------------
    # Keep the best solution seen so far
    # ------------------------------------------------------------------
    conv_best = _conv_self(a, fft_len)
    best_score = 2.0 * n * np.max(conv_best)
    best = a.copy()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # optimisation parameters
    top_k = 10                     # number of active convolution peaks
    eta0 = 1.2                     # initial step‑size for the exponentiated update
    decay_start = 0.15             # after this fraction of the budget we start strong decay
    last_improve = time.monotonic()
    stagnation_limit = 30.0        # seconds without improvement → restart
    momentum = np.zeros_like(a)    # simple exponential smoothing of the gradient
    beta = 0.85                    # momentum coefficient
    iteration = 0

    start_time = time.monotonic()

    while time.monotonic() < deadline:
        iteration += 1

        # --------------------------------------------------------------
        # 1) compute the self‑convolution and collect the biggest peaks
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        max_val = np.max(conv)

        # indices of the `top_k` largest convolution entries
        if top_k >= conv.size:
            top_idx = np.arange(conv.size)
        else:
            part = np.argpartition(-conv, top_k - 1)[:top_k]
            top_idx = part[np.argsort(-conv[part])]      # sorted for reproducibility

        # --------------------------------------------------------------
        # 2) build a summed partner vector (sub‑gradient of the active set)
        # --------------------------------------------------------------
        w = np.zeros_like(a)
        for k in top_idx:
            w += _partner_vector(a, int(k))

        # apply momentum to smooth the update direction
        momentum = beta * momentum + (1.0 - beta) * w

        # --------------------------------------------------------------
        # 3) step‑size schedule (linear decay, steeper after `decay_start`)
        # --------------------------------------------------------------
        elapsed = time.monotonic() - start_time
        total = deadline - start_time
        frac = elapsed / total
        if frac < decay_start:
            eta = eta0 * (1.0 - frac / decay_start)   # gentle decay
        else:
            # after the early phase we switch to a much smaller step‑size
            eta = eta0 * 0.1 * (1.0 - (frac - decay_start) / (1.0 - decay_start))

        # --------------------------------------------------------------
        # 4) exponentiated‑gradient update, renormalise, protect against zeros
        # --------------------------------------------------------------
        factor = np.exp(-eta * momentum)
        a = a * factor
        a = np.clip(a, 1e-14, None)   # avoid exact zeros which hurt the next step
        a /= a.sum()

        # --------------------------------------------------------------
        # 5) occasional stochastic exploration (≈2 % of the iterations)
        # --------------------------------------------------------------
        r = rng.random()
        if r < 0.015:
            a = _random_block_perturb(a, rng)
        elif r < 0.025:
            a = _pairwise_swap_perturb(a, rng)

        # --------------------------------------------------------------
        # 6) evaluate the current score (fast version, same as evaluate_sequence)
        # --------------------------------------------------------------
        conv = _conv_self(a, fft_len)
        cur_score = 2.0 * n * np.max(conv)

        # --------------------------------------------------------------
        # 7) record improvement
        # --------------------------------------------------------------
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # 8) restart if we have been stuck for a while
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > stagnation_limit:
            # choose a fresh start: sometimes arcsine, sometimes random Dirichlet
            if rng.random() < 0.6:
                a = _init_arcsine(n)
            else:
                a = rng.gamma(shape=1.0, size=n)
                a /= a.sum()
            # a tiny jitter to avoid falling back onto the same point
            a += rng.normal(scale=0.001, size=n)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            momentum[:] = 0.0
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] restart (stagnation)")

    # ------------------------------------------------------------------
    # Return the best vector (clipped to satisfy evaluate_sequence's limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]
