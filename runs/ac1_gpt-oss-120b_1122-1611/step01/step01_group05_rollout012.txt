def conv_fft(a, b):
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    fft_a = np.fft.rfft(a, n=N)
    fft_b = np.fft.rfft(b, n=N)
    conv = np.fft.irfft(fft_a * fft_b, n=N)
    return conv[:n]

Implementation details for projection onto simplex:

def project_to_simplex(v):
    """Projects v onto the probability simplex (sum to 1, nonnegative)."""
    # Ensure np.array
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all nonpositive
        theta = np.max(v) - 1  # not needed
        w = np.maximum(v - theta, 0)
        return w / np.sum(w)  # fallback
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def conv_fft(x, y):
    n = len(x) + len(y) - 1
    L = 1 << ((n - 1).bit_length())

    Y = np.fft.rfft(y, n=L)
    conv = np.fft.irfft(X * Y, n=L)
    return conv[:n]

"""
Propose a step‑function (list of non‑negative heights) that minimises the
security‑evaluation function using a projected (soft‑max) gradient descent.
The algorithm works in three stages:

1. Initialise a probability vector from the provided ``height_sequence_1``
   (or a uniform vector) and normalise it to sum to 1.
2. Repeatedly
   * evaluate the full self‑convolution,
   * build a soft‑max weighting of the convolution coefficients that focusses
     on the current maximal entry,
   * compute the gradient of the weighted sum w.r.t. the vector (a cheap FFT
     convolution),
   * take a small step opposite to the gradient and project back onto the
     probability simplex,
   * occasionally perform random block‑multiplicative perturbations and a
     jitter step to escape plateaus.
   The best vector encountered is stored.
3. Return the best vector as a plain ``list[float]`` – the format expected by
   ``evaluate_sequence``.
"""

import time
import numpy as np

def _conv_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using FFT."""
    n = x.size + y.size - 1
    N = 1 << ((n - 1).bit_length())  # next power of two
    fx = np.fft.rfft(x, n=N)
    fy = np.fft.rfft(y, n=N)
    conv = np.fft.irfft(fx * fy, n=N)
    return conv[:n]

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum==1, v>=0)."""
    # implement the algorithm of Wang & Carreira‑Perpiñán (2013)
    if np.sum(v) == 1 and np.all(v >= 0):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _score(a: np.ndarray) -> float:
    """Compute the evaluation function for a non‑negative vector a."""
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    conv = _conv_fft(a, a)
    max_c = np.max(conv)
    return float(2.0 * a.size * max_c / (s * s))

    a_sum = a.sum()
    if a_sum < 1e-12:

    print(f"[init] length={n}, start_score={best_score:.6f}")

    gamma = 5.0          # soft‑max temperature
