"""
Propose a coefficient sequence that minimizes the evaluation function.

Algorithm
---------
1. Choose the working length `n` (use the length of the provided
   `height_sequence_1` if it exists, otherwise 6000).

2. Initialise the best sequence with a uniform vector (guaranteed to be
   valid).  Its score is 2.0.

3. Exhaustively search a two‑parameter family
       a_i = (i + shift) ** beta                     (beta < 0)
   on a fine grid of `beta` around the theoretical optimum –0.5 and of
   `shift` in a modest range.  For every candidate the score returned
   by `evaluate_sequence` is recorded.

4. Augment the power‑law family with a sinusoidal envelope
       a_i = (i + shift) ** beta * sin(π·(i+0.5)/n) ** gamma
   (the sine term forces the extreme entries towards zero and can
   decrease the large edge values).  A small grid of `gamma` values is
   examined.

5. Keep the best sequence found so far.

6. Refine the best sequence with cheap stochastic block perturbations:
   – pick a random contiguous block,
   – multiply the whole block by a log‑normal factor (σ≈0.06),
   – occasionally enforce symmetry,
   – clip to the bounds required by `evaluate_sequence`.
   If a perturbation improves the score, it is kept.

The whole search respects the 1000 s budget (the last few seconds are
reserved for a final printout).  The method is completely self‑contained
and uses only NumPy and the supplied `evaluate_sequence` function.
"""

import numpy as np
import time
import math

# ----------------------------------------------------------------------
# Helper functions (top‑level – no closures)
# ----------------------------------------------------------------------
def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    """Return a power‑law sequence a_i = (i + shift)^beta (beta negative)."""
    i = np.arange(n, dtype=np.float64)
    seq = (i + shift) ** beta
    # Numerical safety: enforce the limits required by evaluate_sequence.
    return np.clip(seq, 0.0, 1000.0)


def generate_power_sine(beta: float, shift: float, gamma: float, n: int) -> np.ndarray:
    """Power‑law multiplied by a sinusoidal envelope."""
    i = np.arange(n, dtype=np.float64)
    base = (i + shift) ** beta
    angles = np.pi * (i + 0.5) / n
    envelope = np.sin(angles) ** gamma
    seq = base * envelope
    return np.clip(seq, 0.0, 1000.0)


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Return a cheap O(n) non‑negative perturbation of ``base``."""
    cand = np.array(base, copy=True)
    n = cand.size
    # block width: from 0.5 % up to 5 % of n (inclusive)
    w_min = max(1, n // 200)
    w_max = max(2, n // 20)
    w = int(rng.integers(w_min, w_max + 1))
    st = int(rng.integers(0, n - w + 1))
    en = st + w
    # multiplicative log‑normal tweak of the whole block
    factor = float(np.exp(rng.normal(0.0, 0.06)))
    cand[st:en] *= factor
    # occasional symmetrisation – helpful for nearly‑symmetric shapes
    if rng.random() < 0.30:
        cand = 0.5 * (cand + cand[::-1])
    # enforce the bounds required by evaluate_sequence
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Main entry point required by the test harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises the
    provided evaluation function.  The routine consists of a fine
    grid‑search over a power‑law family (with an optional sinusoidal
    envelope) followed by cheap stochastic block refinements.
    """
    rng = np.random.default_rng(seed)
    # reserve a few seconds for the final printout
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # Determine the working length.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
    else:
        n = 6000                     # safe default size
    print(f"[+] working length = {n}")

    # ------------------------------------------------------------------
    # Start from a guaranteed‑valid fallback (uniform ones).
    # ------------------------------------------------------------------
    best = np.ones(n, dtype=np.float64)
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[+] initial uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Exhaustive grid over power‑law parameters.
    # ------------------------------------------------------------------
    betas = np.linspace(-0.65, -0.45, 21)          # step 0.01
    shifts = np.linspace(0.05, 5.0, 51)           # step ≈0.1
    for beta in betas:
        for shift in shifts:
            cand = generate_power(beta, shift, n)
            # trivial early‑exit if the sequence is all zeros (should not happen)
            score = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(score) and score < best_score:
                best = cand
                best_score = score
                print(f"[+] new best (power) β={beta:.3f} shift={shift:.2f} → {score:.6f}")

    # ------------------------------------------------------------------
    # 2) Power‑law with sinusoidal envelope.
    # ------------------------------------------------------------------
    gammas = [0.5, 1.0, 1.5, 2.0, 2.5]           # modest set for speed
    for beta in betas:
        for shift in shifts[::4]:                # thin the grid a bit
            for gamma in gammas:
                cand = generate_power_sine(beta, shift, gamma, n)
                score = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(score) and score < best_score:
                    best = cand
                    best_score = score
                    print(f"[+] new best (power·sine) β={beta:.3f} shift={shift:.2f} γ={gamma:.1f} → {score:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional: keep the previously supplied sequence if it exists,
    #    and see whether a quick perturbation improves it.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        cand = np.array(height_sequence_1, dtype=np.float64)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[+] starting from supplied sequence → {score:.6f}")

    # ------------------------------------------------------------------
    # 4) Stochastic block‑wise refinement.
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        cand = block_perturbation(best, rng)
        score = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(score) and score < best_score:
            best = cand
            best_score = score
            print(f"[+] refine #{it} improved score → {score:.6f}")
        it += 1

    print(f"[+] final best score = {best_score:.6f}")
    # Return plain Python floats for the caller.
    return [float(x) for x in best]
