def project_to_simplex(v):
   # Standard algorithm
   # Input vector of real numbers.
   # Output: nonnegative vector w with sum = 1.
   u = np.sort(v)[::-1]
   cssv = np.cumsum(u) - 1
   ind = np.arange(len(v)) + 1
   # find rho
   cond = u - cssv / ind > 0
   rho = np.where(cond)[0][-1]
   theta = cssv[rho] / (rho + 1)
   w = np.maximum(v - theta, 0)
   return w

def gradient_step(seq, eta):
    n = len(seq)
    conv = np.convolve(seq, seq)
    bmax = conv.max()
    k = conv.argmax()
    s = seq.sum()
    rev = np.zeros(n)
    start = max(0, k - (n-1))
    end = min(n-1, k)
    rev_start = start
    rev_end = end
    rev[rev_start: rev_end+1] = seq[k - np.arange(rev_start, rev_end+1)]
    direction = 2.0 * (s * rev - bmax)
    cand = seq - eta * direction
    # enforce non-negative
    cand = np.maximum(cand, 0.0)
    s2 = cand.sum()
    if s2>0:
        cand /= s2
    else:
        return seq
    return cand
```

"""
Propose a near‑optimal step‑function.

This implementation builds on the previous power‑law search but adds several
new ingredients:

*   Randomised families (power‑law, sqrt‑differences, exponential) with a
    tunable shift / decay parameter.
*   A simple projected gradient step that uses the analytic sub‑gradient of
    the objective (the maximum entry of the self‑convolution).  The step is
    tiny and the vector is re‑normalised to the probability simplex.
*   Three stochastic moves (block multiplicative, mass‑transfer and the
    gradient step) are applied to the current best sequence.  The algorithm
    keeps the best ever found and restarts from a fresh random candidate if
    it stalls.
*   The total runtime is limited by the supplied budget (default 1000 s).
    Progress is printed so the user can see the improvement over time.

The objective is scale‑invariant, therefore every candidate is normalised
to have a unit sum before it is evaluated.
"""

import time
import math
import numpy as np

def _make_power_seq(n: int, p: float, shift: float = 0.0) -> np.ndarray:
    """Power‑law a_i ∝ (i+1+shift)^{-p}, normalised to sum = 1."""
    idx = np.arange(1, n + 1, dtype=float) + shift
    seq = idx ** (-p)
    total = seq.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=float)
    seq /= total
    return seq


def _make_sqrtdiff_seq(n: int, shift: float = 0.0) -> np.ndarray:
    """
    a_i = √(i+1+shift) – √(i+shift)  (discrete derivative of √x),
    normalised to sum = 1.
    """
    x = np.arange(n, dtype=float) + shift
    seq = np.sqrt(x + 1.0) - np.sqrt(x)
    total = seq.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=float)
    seq /= total
    return seq


def _make_exponential_seq(n: int, lam: float) -> np.ndarray:
    """Exponential decay a_i ∝ exp(-lam·i), normalised to sum = 1."""
    idx = np.arange(n, dtype=float)
    seq = np.exp(-lam * idx)
    total = seq.sum()
    if total == 0.0:
        return np.full(n, 1.0 / n, dtype=float)
    seq /= total
    return seq


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical fallback – return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _gradient_step(seq: np.ndarray, rng: np.random.Generator,
                   eta: float = None) -> np.ndarray:
    """
    One projected gradient step that tries to reduce
    `2·n·max(conv(seq,seq))/sum(seq)^2`.

    The sub‑gradient of the objective w.r.t. a_i is
        2·(S·a_{k*‑i} – b_max)   where S = Σ a_i and k* is the index of the maximum
    entry of the convolution.  The step size η is drawn randomly if not supplied.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    b_max = conv.max()
    k = int(conv.argmax())
    S = seq.sum()

    # build the reversed segment that participates in the max entry
    rev = np.zeros_like(seq)
    lo = max(0, k - (n - 1))
    hi = min(k, n - 1)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        rev[idx] = seq[k - idx]

    # sub‑gradient (up to a positive constant factor that does not affect the step)
    direction = 2.0 * (S * rev - b_max)

    # pick a tiny step size; the scale is not critical because we re‑normalise anyway
    if eta is None:
        eta = rng.uniform(0.01, 0.15)

    cand = seq - eta * direction
    cand = np.clip(cand, 0.0, None)          # keep non‑negative
    cand = _project_to_simplex(cand)        # enforce Σ = 1
    return cand


def _block_perturb(seq: np.ndarray, rng: np.random.Generator,
                   sigma: float = 0.07) -> np.ndarray:
    """Multiply a randomly chosen contiguous block by a log‑normal factor."""
    n = seq.size
    start = rng.integers(0, n)
    max_width = max(2, n // 30)
    width = rng.integers(1, max_width + 1)
    stop = min(n, start + width)
    factor = math.exp(rng.normal(0.0, sigma))
    cand = seq.copy()
    cand[start:stop] *= factor
    cand = np.clip(cand, 0.0, None)
    cand = _project_to_simplex(cand)
    return cand


def _mass_transfer(seq: np.ndarray, rng: np.random.Generator,
                   max_fraction: float = 0.1) -> np.ndarray:
    """
    Transfer a small amount of mass from one index to another.
    This move keeps the total sum unchanged.
    """
    n = seq.size
    i, j = rng.choice(n, size=2, replace=False)
    # amount to move – never larger than max_fraction·a_i
    delta = rng.random() * max_fraction * seq[i]
    if delta <= 0.0:
        return seq
    cand = seq.copy()
    cand[i] -= delta
    cand[j] += delta
    cand = np.clip(cand, 0.0, None)
    cand = _project_to_simplex(cand)
    return cand


def _random_candidate(rng: np.random.Generator) -> np.ndarray:
    """
    Produce a fresh random candidate from one of the three families.
    The length, parameters and family are all drawn at random.
    """
    n = int(rng.integers(1500, 2501))   # keep the size modest for speed
    family = rng.choice(["power", "sqrtdiff", "exp"])
    if family == "power":
        p = rng.uniform(0.35, 0.80)
        shift = rng.uniform(0.0, 2.0)
        seq = _make_power_seq(n, p, shift)
    elif family == "sqrtdiff":
        shift = rng.uniform(0.0, 2.0)
        seq = _make_sqrtdiff_seq(n, shift)
    else:  # exponential family
        lam = rng.uniform(0.001, 0.02)
        seq = _make_exponential_seq(n, lam)
    seq = _project_to_simplex(seq)   # ensure a proper simplex point
    return seq


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimises
    `evaluate_sequence`. The routine spends the allotted time searching and
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    total_budget = float(budget_s)
    # safety margin – a few seconds before the outer timeout
    deadline = start_time + max(0.0, total_budget - 2.0)

    # we'll spend roughly the first 15 % of the budget on a wide random sweep
    exploration_deadline = start_time + 0.15 * total_budget

    best_seq = None
    best_score = float("inf")

    # ----------------------------------------------------------------------
    # 1) Broad random sweep over several families (no gradients yet)
    # ----------------------------------------------------------------------
    while time.monotonic() < exploration_deadline:
        seq = _random_candidate(rng)
        # Clip to the hard limits that `evaluate_sequence` enforces
        seq = np.clip(seq, 0.0, 1000.0)
        score = float(evaluate_sequence(seq.tolist()))
        if score < best_score:
            best_score = score
            best_seq = seq.copy()
            print(f"[explore] new best: {best_score:.6f} (n={len(seq)})")
        # Occasionally try a slightly refined grid around the known good exponent
        # (this gives deterministic progress but still stays within the time budget)
        if best_seq is not None and rng.random() < 0.02:
            n = best_seq.size
            for p in np.linspace(0.45, 0.55, 7):
                for shift in np.linspace(0.0, 1.0, 5):
                    cand = _make_power_seq(n, p, shift)
                    cand = np.clip(cand, 0.0, 1000.0)
                    score = float(evaluate_sequence(cand.tolist()))
                    if score < best_score:
                        best_score = score
                        best_seq = cand.copy()
                        print(f"[grid] improve: {best_score:.6f} (n={n}, p={p:.3f}, s={shift:.2f})")

    # If the random sweep failed to find anything (should not happen) fall back
    if best_seq is None:
        best_seq = np.ones(256, dtype=float)
        best_seq /= best_seq.sum()
        best_score = float(evaluate_sequence(best_seq.tolist()))
        print("[fallback] using uniform sequence")

    # ----------------------------------------------------------------------
    # 2) Local refinement – stochastic gradient + block moves
    # ----------------------------------------------------------------------
    cur_seq = best_seq.copy()
    cur_score = best_score
    no_improve = 0
    iter_count = 0

    while time.monotonic() < deadline:
        iter_count += 1
        move_type = rng.random()
        if move_type < 0.45:
            cand = _block_perturb(cur_seq, rng, sigma=0.07)
        elif move_type < 0.70:
            cand = _mass_transfer(cur_seq, rng, max_fraction=0.12)
        elif move_type < 0.95:
            # gradient step – step size drawn each time
            cand = _gradient_step(cur_seq, rng, eta=rng.uniform(0.015, 0.07))
        else:
            # full random restart (helps escape a possible plateau)
            cand = _random_candidate(rng)

        cand = np.clip(cand, 0.0, 1000.0)          # enforce limits of evaluate_sequence
        # re‑normalise – the projection already guarantees Σ=1 but clipping may disturb it
        if cand.sum() == 0.0:
            continue
        cand /= cand.sum()

        # evaluate
        score = float(evaluate_sequence(cand.tolist()))
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(f"[improve] iter {iter_count:06d}  new best {best_score:.6f} (n={len(cand)})")
            # keep the improving candidate as the current one
            cur_seq = cand.copy()
            cur_score = score
            no_improve = 0
        else:
            # if the candidate is better than the current but not the global best,
            # we still accept it (helps climbing out of shallow local minima)
            if score < cur_score:
                cur_seq = cand.copy()
                cur_score = score
                no_improve = 0
            else:
                no_improve += 1

        # occasional forced restart if we have been stuck for a long while
        if no_improve > 5000:
            cur_seq = _random_candidate(rng)
            cur_score = float(evaluate_sequence(cur_seq.tolist()))
            no_improve = 0
            print(f"[restart] iter {iter_count:06d}  reset with score {cur_score:.6f}")

    # ----------------------------------------------------------------------
    # 3) Return the best sequence found
    # ----------------------------------------------------------------------
    print(f"[final] best score after {iter_count} iterations: {best_score:.6f}")
    return [float(x) for x in best_seq]
