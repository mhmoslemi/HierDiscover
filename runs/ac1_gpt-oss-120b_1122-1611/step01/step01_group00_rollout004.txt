def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a non‑negative step‑function height sequence that minimises
    the evaluation function.  The algorithm uses a smoothed max‑convolution
    objective with an annealing temperature, gradient descent with momentum,
    and a final exact max‑gradient phase.  Symmetry of the sequence is
    enforced to reduce the peak of the auto‑convolution.
    """
    import time, numpy as np
    rng = np.random.default_rng(seed)
    # Determine length based on budget: if budget > 800 secs, use longer n.
    if budget_s > 800:
        n = 2000
    else:
        n = 1024
    # Initial uniform distribution on simplex
    a = np.full(n, 1.0/n, dtype=np.float64)
    # best tracking
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")
    # SSD: projection onto simplex function defined elsewhere (global)
    # Hyperparameters
    eta0 = 0.2     # base step size
    beta = 0.9     # momentum coefficient
    v = np.zeros_like(a)
    tau = 0.5
    tau_max = 20.0
    warmup_iters = 20000  # after this we switch to hard max gradient
    iteration = 0
    deadline = time.monotonic() + max(0.0, budget_s - 2.0)
    # precompute lengths for FFT
    N1 = 1 << ((2*n - 1).bit_length())
    N2 = 1 << ((3*n - 2).bit_length())
    # main loop
    while time.monotonic() < deadline:
        iteration += 1
        # FFT of a
        fft_a = np.fft.rfft(a, n=N1)
        conv = np.fft.irfft(fft_a * fft_a, n=N1)[:2*n-1]
        # compute gradient
        if iteration <= warmup_iters:
            # softmax weights
            max_conv = np.max(conv)
            shift = conv - max_conv  # avoid overflow
            exp_scaled = np.exp(tau * shift)
            w = exp_scaled / np.sum(exp_scaled)
            # compute gradient via convolution w with reversed a
            a_rev = a[::-1]

The goal is to find a non‑negative sequence `a` (step‑function heights) that minimises  

\[
E(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `*` denotes the discrete convolution.  
The evaluation is scale‑invariant, therefore we keep the ℓ‑1 norm of `a`
fixed to 1 (project onto the simplex).  

The optimiser works in two phases:

1. **Annealed smooth‑max phase** –  
   The maximum is approximated by a soft‑max  
   \(\displaystyle S_\tau(a)=\frac1\tau\log\sum_k e^{\tau (a*a)_k}\).  
   The gradient of `Sτ` is a smooth combination of many convolution
   entries, which yields a well‑behaved descent direction.
   The temperature `τ` starts small and is increased slowly to focus the
   optimisation on the true worst entry.

2. **Exact‑max phase** –  
   After a warm‑up period the algorithm switches to the exact sub‑gradient
   of the worst convolution entry (as used in the reference code).  

Both phases employ:

* **Projected gradient descent with momentum** – the update is projected
  back onto the probability simplex (non‑negative, unit sum).
* **Symmetrisation** – after each projection we enforce `a[i]=a[n‑1‑i]`,
  which is natural for an autocorrelation problem.
* **Occasional stochastic perturbations** – to escape plateaus.
* **FFT‑based convolutions** – the dominant O(n log n) operations keep the
  runtime low even for `n≈2000`.

The routine runs for the allotted time (default ≈ 1000 s), keeps the best
sequence found and returns it as a plain Python list of floats.

"""
Proposed optimiser for the step‑function evaluation problem.

The algorithm works in two stages:
  1. Annealed smooth‑max optimisation (soft‑max temperature τ is slowly raised).
  2. Exact max‑gradient descent (sub‑gradient of the worst convolution entry).

A probability‑simplex projection guarantees feasibility (a_i ≥ 0, Σa_i = 1).
After each projection the vector is symmetrised (a[i] = a[n‑1‑i]).
Momentum (β) and a diminishing step‑size η_t = η0 / sqrt(t) are used.
Random log‑normal perturbations are added occasionally to increase exploration.

All convolutions are performed with FFTs (O(n log n)) allowing a large
number of iterations within the 1000 s budget.  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if n == 0:
        return v
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive → uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Search for a sequence of non‑negative heights that minimises the
    evaluation function.  The method combines a soft‑max annealing phase
    with a final exact max‑gradient phase, using FFT‑based convolutions,
    momentum, simplex projection and occasional random perturbations.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose problem size (larger n gives more freedom but costs more per iter)
    # Adapt size to the time budget: a longer run can afford a larger n.
    if budget_s > 800:
        n = 2000                     # a comfortable size for long budgets
    else:
        n = 1024                     # fallback for shorter runs

    # Initial feasible point: uniform distribution (sum = 1)
    a = np.full(n, 1.0 / n, dtype=np.float64)

    # Track the best solution seen so far
    best = a.copy()
    best_score = float(evaluate_sequence(best.tolist()))
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # -------------------- hyper‑parameters --------------------------------
    eta0 = 0.2          # initial step size (will be divided by sqrt(iter))
    beta = 0.9          # momentum coefficient
    v = np.zeros_like(a)    # momentum buffer

    tau = 0.5           # starting temperature for soft‑max
    tau_max = 20.0      # maximum temperature
    tau_inc = 0.001     # linear increase per iteration (soft‑max phase)

    warmup_iters = 15000      # switch to exact max after this many iters
    eval_interval = 5         # evaluate only every few iterations (speed)

    # Pre‑compute FFT lengths (powers of two)
    N_conv = 1 << ((2 * n - 1).bit_length())        # for a*a
    N_corr = 1 << ((3 * n - 2).bit_length())        # for w * a_rev

    deadline = time.monotonic() + max(0.0, float(budget_s) - 2.0)
    iteration = 0

    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        iteration += 1

        # ---------- convolution a*a (FFT) ----------
        fft_a = np.fft.rfft(a, n=N_conv)
        conv_full = np.fft.irfft(fft_a * fft_a, n=N_conv)[:2 * n - 1]

        # ---------- gradient computation ----------
        if iteration <= warmup_iters:
            # ---- smooth max (soft‑max) gradient ----
            max_conv = np.max(conv_full)
            # shift to avoid overflow, then apply temperature τ
            shift = conv_full - max_conv
            w = np.exp(tau * shift)
            w_sum = np.sum(w)
            if w_sum == 0.0:                     # numeric safety
                w = np.full_like(w, 1.0 / w.size)
            else:
                w /= w_sum

            # correlation of w with a (linear) via FFT:
            a_rev = a[::-1]
            fft_w = np.fft.rfft(w, n=N_corr)
            fft_a_rev = np.fft.rfft(a_rev, n=N_corr)
            corr_full = np.fft.irfft(fft_w * fft_a_rev, n=N_corr)[:3 * n - 2]

            # gradient of the soft‑max objective:
            grad = 2.0 * corr_full[n - 1: n - 1 + n]

            # anneal temperature
            tau = min(tau_max, tau + tau_inc)

        else:
            # ---- exact max gradient (sub‑gradient) ----
            max_idx = int(np.argmax(conv_full))

            i_min = max(0, max_idx - (n - 1))
            i_max = min(n - 1, max_idx)

            grad = np.zeros_like(a)
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            grad[idx] = 2.0 * a[max_idx - idx]

        # ---------- momentum update ----------
        v = beta * v + grad
        eta = eta0 / np.sqrt(iteration)
        a = a - eta * v

        # ---------- feasibility (simplex) ----------
        a = _project_simplex(a)

        # ---------- enforce symmetry ----------
        a = (a + a[::-1]) / 2.0
        # (symmetrisation preserves the ℓ1‑norm)

        # ---------- occasional random shake ----------
        if rng.random() < 0.03:                     # ~3 % of iterations
            pos = rng.integers(0, n)
            factor = np.exp(rng.normal(0.0, 0.2))   # log‑normal jitter
            a[pos] *= factor
            a = _project_simplex(a)
            a = (a + a[::-1]) / 2.0

        # ---------- evaluate (periodic) ----------
        if iteration % eval_interval == 0:
            sum_a = a.sum()
            if sum_a < 1e-12:          # Protect against degenerate sum
                score = float("inf")
            else:
                conv_max = np.max(conv_full)
                score = 2.0 * n * conv_max / (sum_a * sum_a)

            if np.isfinite(score) and score < best_score:
                best = a.copy()
                best_score = float(score)
                print(f"[improve] iter={iteration}, score={best_score:.6f}")

                # early exit if we have already beaten the target
                if best_score <= 1.5025:
                    print("[target reached]")
                    break

        # ---------- occasional status report ----------
        if iteration % 50000 == 0:
            print(f"[status] iter={iteration}, current score≈{2.0 * n * np.max(conv_full) / (a.sum()**2):.6f}, best={best_score:.6f}")

    print(f"[final] best score={best_score:.6f} after {iteration} iterations")
    # Return the result as a plain Python list of floats.
    return [float(x) for x in best]
