"""
Propose a step‑function that (hopefully) beats the 1.57 bound.

The search consists of three phases:

1. **Parametric sweep** – generate a few hundred analytic families
   (power‑law, power‑law+exponential tail, “U‑shaped’’ sum of two
   power laws, sine‑humps, simple triangle).  Every candidate is
   evaluated with the provided ``evaluate_sequence`` and the best
   one is kept.

2. **Guided mixing** – starting from the current best, a
   *low‑frequency* partner is built from the inverse of the current
   shape (large values where the current entry is small).  The partner
   is scaled so that ``max(f * g)`` equals the current ``max(f*f)``.
   A convex combination ``(1‑t)f + t g`` (small ``t``) is then tested.
   The same idea with a “complement’’ partner (``max(f)-f``) is also
   tried.

3. **Random local perturbations** – cheap block‑wise multiplicative
   tweaks and per‑index log‑normal noise are applied while the time
   budget permits.  Any improvement replaces the current best.

All helpers are top‑level, there are no closures or lambdas, and the
code stays well below the 1000 s limit by using FFT‑based convolutions
for the mixing phase.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities (FFT based for speed)
# ----------------------------------------------------------------------
def convolution_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*b using FFT (real) as a 1‑D array."""
    n = a.size
    size = 2 * n                     # length >= 2n‑1, power‑of‑2 not required
    fa = np.fft.rfft(a, n=size)
    fb = np.fft.rfft(b, n=size)
    conv = np.fft.irfft(fa * fb, n=size)
    return conv[: 2 * n - 1]         # the true linear convolution length


def max_convolution_fft(x: np.ndarray) -> float:
    """Maximum entry of the autocorrelation of x (x*x)."""
    n = x.size
    # reuse convolution_fft for the autocorrelation
    conv = convolution_fft(x, x)
    return float(np.max(conv))


# ----------------------------------------------------------------------
# Analytic families
# ----------------------------------------------------------------------
def generate_power(beta: float, shift: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + shift
    seq = i ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_power_exp(beta: float, shift: float, gamma: float, n: int) -> np.ndarray:
    """a_i = (i+shift)^beta * exp(-gamma * i)"""
    i = np.arange(n, dtype=np.float64)
    seq = (i + shift) ** beta * np.exp(-gamma * i)
    return np.clip(seq, 0.0, 1000.0)


def generate_u_shape(beta: float, shift: float, n: int) -> np.ndarray:
    """Symmetric “U’’ shape: (i+shift)^beta + (n‑1‑i+shift)^beta."""
    i = np.arange(n, dtype=np.float64)
    seq = (i + shift) ** beta + (n - 1 - i + shift) ** beta
    return np.clip(seq, 0.0, 1000.0)


def generate_sine(gamma: float, n: int) -> np.ndarray:
    angles = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    seq = np.sin(angles) ** gamma
    return np.clip(seq, 0.0, 1000.0)


def generate_triangle(n: int) -> np.ndarray:
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        seq = np.concatenate([half, half[::-1]])
    else:
        seq = np.concatenate([half, half[-2::-1]])
    return np.clip(seq, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Random perturbations (compatible with the original script)
# ----------------------------------------------------------------------
def make_block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Cheap O(n) multiplicative block tweak."""
    cand = base.copy()
    n = cand.size
    # block width: 0.5 % – 5 % of n
    w = int(rng.integers(max(1, n // 200), max(2, n // 12)))
    st = int(rng.integers(0, max(1, n - w + 1)))
    en = min(n, st + w)
    factor = float(np.exp(rng.normal(0.0, 0.12)))   # log‑normal factor
    cand[st:en] *= factor
    # occasional symmetrisation (helps for symmetric shapes)
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    return np.clip(cand, 0.0, 1000.0)


def make_global_noise(base: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    """Multiplicative log‑normal noise applied to every entry."""
    factor = np.exp(rng.normal(0.0, sigma, size=base.shape))
    cand = base * factor
    return np.clip(cand, 0.0, 1000.0)


# ----------------------------------------------------------------------
# Guided mixing helpers
# ----------------------------------------------------------------------
def mix_with_inverse(base: np.ndarray, t: float, eps: float = 1e-12) -> np.ndarray:
    """
    Build a partner g proportional to 1/(base+eps).  Scale g so that
    max(base * g) == max(base * base) and return the convex combination.
    """
    n = base.size
    M = max_convolution_fft(base)

    inv = 1.0 / (base + eps)
    # avoid astronomically large values (they would be down‑scaled later anyway)
    inv = np.clip(inv, 0.0, 1e6)

    conv = convolution_fft(base, inv)
    cur_max = np.max(conv[: 2 * n - 1])
    # Guard against division by zero (should not happen)
    scale = M / cur_max if cur_max > 0 else 0.0

    g = inv * scale
    cand = (1.0 - t) * base + t * g
    return np.clip(cand, 0.0, 1000.0)


def mix_with_complement(base: np.ndarray, t: float) -> np.ndarray:
    """
    Partner built from (max(base) – base) (i.e. mass shifted to the low spots).
    """
    n = base.size
    M = max_convolution_fft(base)

    mx = np.max(base)
    comp = np.maximum(0.0, mx - base)

    conv = convolution_fft(base, comp)
    cur_max = np.max(conv[: 2 * n - 1])
    scale = M / cur_max if cur_max > 0 else 0.0

    g = comp * scale
    cand = (1.0 - t) * base + t * g
    return np.clip(cand, 0.0, 1000.0)

# Main entry point required by the test harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative heights that (hopefully) minimizes the
    evaluation function.  The routine:

      1. tries a handful of analytical families (power law, power law
         with exponential cut‑off, symmetric U‑shape, sine hump,
         triangle);
      2. keeps the best among them;
      3. repeatedly refines the incumbent using three cheap moves:
         * block‑wise multiplicative perturbation,
         * a convex combination with an “inverse’’ partner,
         * a convex combination with a “complement’’ partner.
      4. finally falls back to simple per‑index log‑normal noise
         when the time budget is almost exhausted.
    """
    rng = np.random.default_rng(seed)
    # Keep a small safety margin so we finish before the hard timeout.
    deadline = time.monotonic() + max(0.0, float(budget_s) - 4.0)

    # ------------------------------------------------------------------
    # Determine the working length.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        n = len(height_sequence_1)
        print(f"Using supplied sequence of length {n}")
    else:
        n = 6000                     # safe default when no external data
        print(f"No supplied sequence – using length {n}")

    # ------------------------------------------------------------------
    # Starting point: either the supplied sequence or a uniform vector.
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        best = np.array(height_sequence_1, dtype=np.float64).copy()
        best_score = float(evaluate_sequence(best.tolist()))
        print(f"Initial supplied score = {best_score:.6f}")
    else:
        best = np.ones(n, dtype=np.float64)
        best_score = float(evaluate_sequence(best.tolist()))
        print(f"Initial uniform score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1)  Systematic search over a few parametric families.
    # ------------------------------------------------------------------
    # Power‑law only
    betas = np.linspace(-0.8, -0.3, 21)          # 21 values
    shifts = np.linspace(0.1, 10.0, 19)          # 19 values
    for beta in betas:
        for shift in shifts:
            cand = generate_power(beta, shift, n)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best, best_score = cand, sc
                print(f"New best (power) β={beta:.3f} shift={shift:.2f} → {sc:.6f}")

    # Power‑law with exponential tail
    gammas = [0.0, 1e-4, 5e-4, 1e-3, 5e-3, 1e-2]   # exponential decay rates
    for beta in betas[::4]:       # coarser grid for speed
        for shift in shifts[::4]:
            for gamma in gammas[1:]:
                cand = generate_power_exp(beta, shift, gamma, n)
                sc = float(evaluate_sequence(cand.tolist()))
                if np.isfinite(sc) and sc < best_score:
                    best, best_score = cand, sc
                    print(f"New best (pow+exp) β={beta:.3f} shift={shift:.2f} γ={gamma:.5f} → {sc:.6f}")

    # Symmetric U‑shape (sum of two power laws)
    for beta in betas[::5]:
        for shift in shifts[::5]:
            cand = generate_u_shape(beta, shift, n)
            sc = float(evaluate_sequence(cand.tolist()))
            if np.isfinite(sc) and sc < best_score:
                best, best_score = cand, sc
                print(f"New best (U‑shape) β={beta:.3f} shift={shift:.2f} → {sc:.6f}")

    # Sine‑hump family
    for gamma in [0.5, 1.0, 2.0, 3.0]:
        cand = generate_sine(gamma, n)
        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            print(f"New best (sine) γ={gamma:.1f} → {sc:.6f}")

    # Simple triangle
    cand = generate_triangle(n)
    sc = float(evaluate_sequence(cand.tolist()))
    if np.isfinite(sc) and sc < best_score:
        best, best_score = cand, sc
        print(f"New best (triangle) → {sc:.6f}")

    # ------------------------------------------------------------------
    # 2)  Guided refinement phase (inverse and complement mixing + blocks).
    # ------------------------------------------------------------------
    iteration = 0
    # schedule for the random‑noise sigma (starts larger, decays)
    sigma_initial = 0.2
    while time.monotonic() < deadline:
        # Choose which move to apply.
        r = rng.random()
        if r < 0.35:
            # cheap block perturbation (same as previous baseline)
            cand = make_block_perturbation(best, rng)
            method = "block"
        elif r < 0.70:
            # mixing with the inverse‑weight partner
            t = rng.uniform(0.02, 0.14)   # modest convex‑combination weight
            cand = mix_with_inverse(best, t)
            method = "inverse_mix"
        else:
            # mixing with the complement partner
            t = rng.uniform(0.02, 0.14)
            cand = mix_with_complement(best, t)
            method = "comp_mix"

        # Occasionally (when time is nearly exhausted) fall back to pure per‑index noise
        if (deadline - time.monotonic()) < 5.0 and rng.random() < 0.5:
            remaining = deadline - time.monotonic()
            # sigma shrinks linearly with remaining time
            sigma = sigma_initial * (remaining / (budget_s - 4.0))
            sigma = max(0.01, sigma)
            cand = make_global_noise(best, rng, sigma)
            method = "global_noise"

        sc = float(evaluate_sequence(cand.tolist()))
        if np.isfinite(sc) and sc < best_score:
            best, best_score = cand, sc
            print(f"Iter {iteration:05d} ({method}) improved → {sc:.6f}")

        iteration += 1

    print(f"Search finished after {iteration} iterations – final best score = {best_score:.6f}")

    # Convert to plain Python floats for the caller.
    return [float(x) for x in best]
