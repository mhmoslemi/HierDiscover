def targeted_transfer(a, rng):

    n=len(a)

    b = np.convolve(a, a)

    k = int(np.argmax(b))

    # compute gradient contributions mapping as described:

    idx = np.arange(n)

    j = k - idx

    grad = np.zeros_like(a)

    mask = (j >= 0) & (j < n)

    grad[mask] = a[j[mask]]  # not multiplied by 2 for relative ranking

    # source is index with largest product a[i]*grad[i] maybe

    source = np.argmax(grad * a)  # may capture i with high impact.

    # intentionally avoid selecting target that is same as source

    # choose target among low a positions (maybe far from source)

    # random candidate for target: sample from low a distribution

    # sample weight mass to move:

    delta_frac = rng.uniform(0.05, 0.3)  # fraction of source mass to move

    delta = a[source] * delta_frac

    # choose target index j uniformly among indices where a is min (or far region)

    # prominence to indices > n//2

    candidates = np.where(a < np.percentile(a, 20))[0]  # low mass indices

    if len(candidates)==0:

         target = rng.integers(0, n)

    else:

         target = rng.choice(candidates)

    # ensure not same

    if target == source:

        target = (source + n//2) % n

    cand = a.copy()

    cand[source] -= delta

    cand[target] += delta

    # clip (source may become negative due to rounding): ensure non-negative

    cand = np.clip(cand, 0.0, None)

    # renorm:

    total = cand.sum()

    if total < 1e-12: return None

    cand /= total

    return cand

"""
Propose a low‑score step‑function using a combination of
* the analytic “α‑sequence” (C(2i,i)/4ⁱ) as a high‑quality seed,
* a simple projected exponential‑gradient descent that directly
  reduces the maximal self‑convolution entry,
* a cheap targeted mass‑transfer move that moves weight from the
  region contributing most to the current maximum to a low‑mass
  region,
* occasional random multiplicative noise for exploration.

The algorithm keeps the sequence non‑negative, enforces L¹‑norm = 1,
and updates a best solution whenever the objective improves.
It runs until the allotted time budget expires (≈ 1000 s) and
returns the best sequence discovered.
"""

import time
import math
import numpy as np


def generate_alpha_sequence(n: int) -> np.ndarray:
    """Alpha‑sequence a_i = C(2i,i)/4^i for i=0..n‑1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    return a


def random_noise(seq: np.ndarray, rng: np.random.Generator, sigma: float) -> np.ndarray:
    """Multiplicative Gaussian perturbation (global)."""
    return seq * (1.0 + sigma * rng.standard_normal(seq.size))


def targeted_transfer(seq: np.ndarray, rng: np.random.Generator) -> np.ndarray | None:
    """
    Move a fraction of mass from a position that most influences the current
    maximal convolution entry to a low‑mass position.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    k = int(np.argmax(conv))

    # gradient of the active entry w.r.t. each a_i (ignoring the factor 2)
    idx = np.arange(n)
    j = k - idx
    grad = np.zeros_like(seq)
    mask = (j >= 0) & (j < n)
    grad[mask] = seq[j[mask]]

    # source = index with large product a_i * grad_i (high influence)
    src = int(np.argmax(seq * grad))
    if seq[src] <= 0.0:
        return None

    # pick a target among the smallest values (helps to spread mass)
    low_thr = np.percentile(seq, 20)          # 20‑th percentile
    low_idx = np.where(seq <= low_thr)[0]
    if low_idx.size == 0:
        tgt = (src + n // 2) % n
    else:
        tgt = int(rng.choice(low_idx))
        if tgt == src:
            tgt = (src + n // 2) % n

    # move a modest fraction of source mass
    frac = rng.uniform(0.05, 0.30)            # 5‑30 % of source
    delta = seq[src] * frac
    cand = seq.copy()
    cand[src] -= delta
    cand[tgt] += delta
    return np.clip(cand, 0.0, None)


def exponential_update(seq: np.ndarray, rng: np.random.Generator,
                       it: int, base_step: float = 0.5) -> np.ndarray:
    """
    Projected exponential‑gradient descent on the objective
    f(a)=max_k (a*a)_k.  The update is:
        a ← a * exp(-η·g)   then renormalised,
    where g is the sum of gradients over all active (maximal) indices.
    """
    n = seq.size
    conv = np.convolve(seq, seq)
    max_val = conv.max()
    # treat every index attaining the maximal value as active
    eps = max(1e-12, max_val * 1e-12)
    active = (conv >= max_val - eps).astype(np.float64)

    # g_i = Σ_{k active} a_{k-i}
    # This is an (un‑reversed) convolution of the active mask with the reversed sequence.
    c = np.convolve(active, seq[::-1])
    g = c[n - 1:n - 1 + n]               # length‑n slice

    if g.max() == 0.0:
        return seq

    # decreasing step size (simple schedule)
    step = base_step / (1.0 + it * 1e-5)
    # bound the exponent to avoid under‑flow/overflow
    max_change = step * g.max()
    if max_change > 1.0:
        step = 1.0 / g.max()

    factor = np.exp(-step * g)
    cand = seq * factor
    return cand

    a = generate_alpha_sequence(n)

    best = a.copy()
