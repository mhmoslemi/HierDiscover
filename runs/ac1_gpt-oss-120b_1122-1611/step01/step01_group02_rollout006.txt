- Input list of floats (non-negative). They will be capped at max(0, x) and min(1000, x). So sequence final values are clipped to [0,1000].

- n = len(sequence)
- b_sequence = np.convolve(sequence, sequence) which results in length (2n - 1). So it's the discrete convolution (full). Then we compute max_b = max(b_sequence) (the sup-norm of convolution).

def solve_lp(f, M):
    n = len(f)
    mat = build_conv_matrix(f)   # shape (2n-1, n)
    # Convert to csr
    # Constraints: mat @ b <= M
    # Bound each variable b_i >= 0
    c = -np.ones(n)   # minimize -sum(b) => maximize sum(b)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=mat, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    return res.x
```

"""
Propose a step‑function sequence with a lower evaluation than the Catalan
construction (≈1.571).  The code starts from the Catalan sequence,
then repeatedly solves a linear programme

    maximise  Σ b_i
    subject to  b_i ≥ 0  and  (f * b)_k ≤ M   for all k,

where f is the current sequence and M = max(f*f).  The LP adds as much
mass as possible without increasing the self‑convolution supremum.
A short line‑search mixes the LP solution into f; the mixture is kept
whenever the evaluation value improves.  Occasionally cheap random
perturbations are tried as a secondary search.  The whole loop runs
until the time budget (≈1000 s) is exhausted, returning the best
sequence found.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


# ----------------------------------------------------------------------
# Helper functions – all defined at top level as required.
# ----------------------------------------------------------------------


def catalan_sequence(length: int) -> np.ndarray:
    """
    Return the first ``length`` terms of the sequence

        a_i = binom(2*i, i) / 4**i

    which satisfies (a * a)[k] = 1 (constant) for all k when the
    sequence is infinite.  The recurrence a_{i+1} = a_i * (2*i+1)/(2*i+2)
    is used for numerical stability.
    """
    seq = np.empty(length, dtype=np.float64)
    seq[0] = 1.0
    for i in range(1, length):
        seq[i] = seq[i - 1] * (2 * (i - 1) + 1) / (2 * i)
    return seq


def block_perturbation(base: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Cheap O(n) random perturbation: pick a contiguous block and
    multiply it by a log‑normal factor; occasionally symmetrise.
    """
    cand = base.copy()
    n = cand.size
    lo = max(1, n // 200)
    hi = max(2, n // 12)
    width = int(rng.integers(lo, hi + 1))
    start = int(rng.integers(0, max(1, n - width + 1)))
    stop = start + width
    factor = float(np.exp(rng.normal(0.0, 0.12)))
    cand[start:stop] *= factor
    if rng.random() < 0.35:
        cand = 0.5 * (cand + cand[::-1])
    np.clip(cand, 0.0, 1000.0, out=cand)
    return cand


def build_convolution_matrix(f: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the matrix A (shape (2*n-1, n)) such that
    conv = A @ x  is the discrete (full) convolution f * x.
    The (k, j)-entry of A equals f[k-j] if the index is in range,
    otherwise 0.
    """
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    for i in range(n):
        fi = f[i]
        if fi == 0.0:
            continue
        # for each column j we have row = i + j
        for j in range(n):
            rows.append(i + j)
            cols.append(j)
            data.append(fi)
    return scipy.sparse.csr_matrix((data, (rows, cols)),
                                   shape=(2 * n - 1, n))


def solve_lp_max_sum(f: np.ndarray, max_conv: float) -> np.ndarray:
    """
    Linear programme:
        maximise   Σ b_i
        subject to  b_i ≥ 0
                    (f * b)_k ≤ max_conv   for every k
    Returns the optimiser b as a dense ndarray.
    """
    n = f.shape[0]
    A = build_convolution_matrix(f)                     # (2n-1) × n
    b_ub = np.full(A.shape[0], max_conv, dtype=np.float64)

    # linprog minimises c·x, so we minimise -Σb_i to maximise Σb_i
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0, None)] * n
    # HiGHS (default) can handle sparse A_ub efficiently.
    res = scipy.optimize.linprog(c,
                                 A_ub=A,
                                 b_ub=b_ub,
                                 bounds=bounds,
                                 method='highs',
                                 options={"presolve": True})

    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")
    return res.x


def score_of(seq: np.ndarray) -> float:
    """Convenient wrapper around the provided evaluate_sequence."""
    return float(evaluate_sequence(seq.tolist()))


def try_mixture(f: np.ndarray, g: np.ndarray,
                current_score: float,
                max_trials: int = 6) -> tuple[np.ndarray, float]:
    """
    Test a few mixing coefficients t ∈ (0,1] and keep the best
    improvement (if any).  Returns (new_sequence, new_score).
    """
    best_seq = f
    best_score = current_score
    # A small set of mixing factors – dense enough to capture a decent step.
    t_candidates = np.linspace(0.1, 1.0, max_trials)
    for t in t_candidates:
        cand = (1.0 - t) * f + t * g
        sc = score_of(cand)
        if sc < best_score:
            best_seq = cand
            best_score = sc
    return best_seq, best_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 42, budget_s: float = 1100, **kwargs):
    """
    Return a list of non‑negative heights (a step function) that tries to
    minimise ``evaluate_sequence``.  The algorithm starts from a Catalan
    construction of moderate size (≈800 entries), then alternates between
    a linear‑programming expansion step and a short random local search,
    always keeping any improvement.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Choose a tractable length.  Use the length of the previously found
    # construction if it exists, but clamp it to a size that our LP can
    # solve comfortably (< 1200).
    # ------------------------------------------------------------------
    if 'height_sequence_1' in globals():
        try:
            proposed_n = int(np.asarray(height_sequence_1).size)
            n = min(proposed_n, 1200)
        except Exception:
            n = 800
    else:
        n = 800

    # ------------------------------------------------------------------
    # Baseline Catalan sequence (no extra normalisation – the evaluation
    # function is scale‑invariant, so any positive scaling works).
    # ------------------------------------------------------------------
    f = catalan_sequence(n).astype(np.float64)
    # Clip just in case – Catalan values are well below the limits.
    np.clip(f, 0.0, 1000.0, out=f)

    best_score = score_of(f)
    best_seq = f.copy()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    start_time = time.time()
    # main optimisation loop – run until the time budget (minus a safety margin)
    time_limit = start_time + max(0.0, float(budget_s) - 5.0)
    iteration = 0
    no_improve_counter = 0

    while time.time() < time_limit:
        # --------------------------------------------------------------
        # 1) Linear‑programming expansion step.
        # --------------------------------------------------------------
        # Compute the present supremum of f*f
        conv_f = np.convolve(f, f)
        M = float(conv_f.max())

        try:
            g = solve_lp_max_sum(f, M)
        except Exception as exc:                     # safety net
            print(f"[lp] failed at iter {iteration}: {exc}")
            break

        # --------------------------------------------------------------
        # 2) Mix the LP solution back into f.
        # --------------------------------------------------------------
        f_candidate, cand_score = try_mixture(f, g, best_score)

        # --------------------------------------------------------------
        # 3) If the mixture improved the score, accept it.
        # --------------------------------------------------------------
        if cand_score < best_score - 1e-12:
            f = f_candidate
            best_seq = f_candidate.copy()
            best_score = cand_score
            print(f"[iter {iteration:04d}] LP‑mix improved score={best_score:.6f}")
            no_improve_counter = 0
            iteration += 1
            continue

        # --------------------------------------------------------------
        # 4) Fallback: cheap random block perturbation.
        # --------------------------------------------------------------
        r_candidate = block_perturbation(f, rng)
        r_score = score_of(r_candidate)

        if r_score < best_score - 1e-12:
            f = r_candidate
            best_seq = r_candidate.copy()
            best_score = r_score
            print(f"[iter {iteration:04d}] random improve score={best_score:.6f}")
            no_improve_counter = 0
        else:
            no_improve_counter += 1

        # small stop‑condition: if we see no improvement for many rounds,
        # switch to pure random search for a short while.
        if no_improve_counter >= 30:
            # try a few pure random steps before giving up
            for _ in range(5):
                cand = block_perturbation(f, rng)
                sc = score_of(cand)
                if sc < best_score:
                    f = cand
                    best_seq = cand.copy()
                    best_score = sc
                    print(f"[rand‑burst] improved score={best_score:.6f}")
            no_improve_counter = 0

        iteration += 1

    print(f"[final] returning sequence of length {n} with score {best_score:.6f}")
    # Convert to plain Python list of floats for the harness.
    return [float(x) for x in best_seq]
