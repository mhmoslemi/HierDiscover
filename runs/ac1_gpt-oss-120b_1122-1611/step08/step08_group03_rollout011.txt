"""
Propose a non‑negative step function that (empirically) beats the
previous best bound 1.513206.  The algorithm consists of three parts
each respecting the simplex (∑a_i = 1) which makes the objective
scale‑invariant:

1. **Smooth‑max Adam** – a differentiable surrogate for the ‖f*f‖∞
   (soft‑max over the convolution).  Adam updates are projected onto the
   probability simplex.  The temperature and learning rate are annealed
   slowly so that the optimisation first sees a smooth landscape and
   finally concentrates on the true max.

2. **Exact‑max sub‑gradient descent** – when the temperature is tiny the
   surrogate becomes the true max.  Here we switch to a sub‑gradient of
   the exact max (∇_j max_k (a*a)[k] = 2·a_{k*‑j}) and continue a
   projected gradient descent with a decaying step‑size.

3. **Stochastic local refinement** – very small random mass‑transfer
   moves (the “mass‑move” hill‑climber) are performed until time runs
   out; any improvement is kept.

All phases share the same bookkeeping of the best solution seen so far.
The routine stops a few seconds before the total budget to guarantee
that a result is returned.

The implementation is fully vectorised, uses only NumPy and works
under the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a 1‑D vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Renormalise to eliminate numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full (linear) convolution of a with itself using FFT."""
    n = a.size
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv

# Core optimisation phases

def _smooth_max_adam(a: np.ndarray,
                     rng: np.random.Generator,
                     phase_deadline: float,
                     best_score: float,
                     best_a: np.ndarray):
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns updated (a, best_score, best_a) and the time when the
    phase stopped.
    """
    n = a.size
    # FFT length large enough for convolution + correlation
    fft_len = _next_pow2(3 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # Schedules
    TAU_START, TAU_END = 0.5, 5e-5        # temperature
    LR_START, LR_END = 0.3, 5e-4        # learning rate

    it = 0
    while time.monotonic() < phase_deadline:
        now = time.monotonic()
        # linear annealing factors
        frac = (now - (phase_deadline - (phase_deadline - time.monotonic()))) / (
            phase_deadline - (phase_deadline - (phase_deadline - time.monotonic())) + 1e-12
        )
        # Simpler: compute fraction of elapsed time w.r.t whole phase
        # (the above complicated expression reduces to 0 at start and 1 at end)
        frac = (now - (phase_deadline - (phase_deadline - time.monotonic()))) / (
            phase_deadline - (phase_deadline - time.monotonic()) + 1e-12
        )
        # Actually we can compute using a simple linear fraction:
        # but to avoid division by zero we recompute each loop:
        # (We will just recompute based on remaining budget.)
        # For simplicity we use a fixed schedule that does not depend on
        # exact timing – it is only a rough guide.
        # ----------------------------------------------------------

        # We'll just linearly interpolate based on elapsed time w.r.t full
        # deadline using a static start time stored outside.
        # (Since the schedule is not critical for correctness we keep it simple.)
        # ----------------------------------------------------------
        # compute fraction based on loop count (roughly proportional to time)
        frac = min(1.0, it / 3000.0)

        tau = TAU_START * (1.0 - frac) + TAU_END * frac
        lr = LR_START * (1.0 - frac) + LR_END * frac

        # ----- self‑convolution -----
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ----- soft‑max over the convolution -----
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)   # shift for stability
        w = exp_shifted / exp_shifted.sum()          # weight vector, sums to 1

        # ----- gradient of the smooth surrogate (correlation) -----
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projected gradient step -----
        a = a - step
        a = _project_onto_simplex(a)

        it += 1

        # ----- occasional exact evaluation -----
        if it % 400 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                elapsed = time.monotonic() - (phase_deadline - (phase_deadline - time.monotonic()))
                print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f}")

        # early exit if we already hit a very good bound
        if best_score <= 1.5025:
            break

    return a, best_score, best_a


def _exact_max_subgradient(a: np.ndarray,
                           rng: np.random.Generator,
                           phase_deadline: float,
                           best_score: float,
                           best_a: np.ndarray):
    """
    Projected sub‑gradient descent on the true max (a*a).
    """
    n = a.size
    it = 0
    while time.monotonic() < phase_deadline:
        conv = _conv_self(a)
        max_b = conv.max()
        # indices attaining the maximum (within machine epsilon)
        max_inds = np.where(conv >= max_b - 1e-14)[0]
        # pick one (or average) – we pick a random one among the ties
        k_star = rng.choice(max_inds)

        # gradient of max at k_star : g_j = 2 * a_{k_star - j} (if in range)
        idx = np.arange(n)
        valid = (k_star - idx >= 0) & (k_star - idx < n)
        g = np.zeros_like(a)
        g[valid] = 2.0 * a[k_star - idx[valid]]

        # step‑size schedule (linear decay)
        frac = (time.monotonic() - (phase_deadline - (phase_deadline - time.monotonic()))) / (
            phase_deadline - (phase_deadline - time.monotonic()) + 1e-12
        )
        # use a very mild schedule – 0.2 → 1e‑5
        lr = 0.2 * (1.0 - frac) + 1e-5 * frac

        a = a - lr * g
        a = _project_onto_simplex(a)

        # occasional exact evaluation
        if it % 250 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                elapsed = time.monotonic() - (phase_deadline - (phase_deadline - time.monotonic()))
                print(f"[{elapsed:6.2f}s] new best (sub‑grad) = {best_score:.6f}")

        if best_score <= 1.5025:
            break
        it += 1
    return a, best_score, best_a


def _stochastic_refinement(a: np.ndarray,
                           rng: np.random.Generator,
                           deadline: float,
                           best_score: float,
                           best_a: np.ndarray):
    """
    Tiny random mass‑transfer moves (the same idea as in the reference code).
    """
    n = a.size
    move_counter = 0
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # log‑uniform fraction in [0.001 , 0.05]
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[i] * frac_move

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        sc = evaluate_sequence(a_candidate.tolist())
        move_counter += 1

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_candidate.copy()
            a = a_candidate
            elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
            print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} (move #{move_counter})")
            if best_score <= 1.5025:
                break

        # jitter every few thousand moves to escape plateaus
        if move_counter % 4000 == 0:
            width = rng.integers(1, max(2, n // 50))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            a_j = a.copy()
            a_j[start_idx:start_idx + width] *= factor
            a = _project_onto_simplex(a_j)

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises

    The method consists of a smooth‑max Adam phase, an exact‑max
    sub‑gradient phase, and a stochastic local refinement.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a few seconds safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # Prefer the known construction if it exists, otherwise uniform.
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        a = a0.copy()
    else:
        n = 4096                       # power‑of‑two length – FFT‑friendly
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size

    # ------------------- bookkeeping -------------------
