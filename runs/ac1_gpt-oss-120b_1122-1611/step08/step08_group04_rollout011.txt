def _project_simplex_target(v, target):
    if target <= 0:
        raise ValueError
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, target / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid optimiser based on the Frank‑Wolfe (LP) scheme described in the
statement, plus a cheap sub‑gradient clean‑up.

* 1️⃣  Initialise a non‑negative vector `f` on the simplex
     (∑f = √(2·n)).  The length `n` is chosen to be a power of two,
     large enough to give a low theoretical bound (≈2·n/(2·n‑1)≈1).

* 2️⃣  Repeatedly:
        – compute the self‑convolution `c = f*f` (FFT) and the current
          peak `M = max(c)`;
        – solve the linear programme

                maximise   1ᵀ b
                subject to  f * b ≤ M   (convolution, all offsets)
                            b ≥ 0

          The LP finds a vector `b` with a larger ℓ¹‑norm while keeping
          its convolution with `f` below the present peak `M`.

        – scale `b` back to the normalisation (∑ = √(2·n)) obtaining `g`,
          and move a fraction `t` towards it:
                f ← (1‑t) f + t g .
          The step size `t` decays with the iteration count.

        The construction guarantees `max(f*f)` strictly decreases
        whenever `∑b > √(2·n)`.

* 3️⃣  After a few LP steps we run a few projected sub‑gradient steps
        on the exact max‑convolution to polish the solution.

* 4️⃣  Random restarts are performed when the search stalls.
        The best vector seen during the whole budget is returned
        (clipped to the limits of `evaluate_sequence`).

The whole routine respects the 1000 s budget, prints progress, and
returns a list of non‑negative heights that (empirically) improves on
the previous 1.513 … bound.
"""

import time
import numpy as np
from scipy.optimize import linprog

# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using an FFT (real‑valued)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of `evaluate_sequence` for a (no clipping needed)."""
    n = a.shape[0]
    conv = _conv_self_fft(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex_target(v: np.ndarray, target: float) -> np.ndarray:
    """Euclidean projection of v onto {x ≥ 0, sum(x)=target}."""
    if target <= 0:
        raise ValueError("target sum must be positive")
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), target):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target
    ind = np.arange(1, v.size + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform distribution
        return np.full_like(v, target / v.size)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  maximise 1ᵀ b
    subject to  f * b ≤ M  (linear convolution constraints)
                b ≥ 0
    Returns the optimal b or ``None`` on failure.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the convolution matrix A where A[k, j] = f[k-j]   (zero if out of range)
    A = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n, dtype=np.float64)          # maximise 1ᵀb  → minimise -1ᵀb
    b_ub = np.full(rows, M, dtype=np.float64)  # RHS of the inequality constraints
    bounds = [(0.0, None) for _ in range(n)]

    # `highs` is fast and robust for medium‑size problems.
    # We ask it to stop early if it exceeds a few seconds (the optimizer
    # timeout is not exposed directly, but the outer time‑budget will stop us).
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm combines the LP “Frank‑Wolfe” step
    from the reference method with a final sub‑gradient polishing phase.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start + total_budget

    # --------------------------------------------------------------
    # Choose a length n (power of two) and an initial distribution.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        n = max(2000, 1 << ((len(base) - 1).bit_length()))   # power of two ≥ len(base)
        if base.size != n:
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(new_x, old_x, base)
            a = np.clip(a, 0.0, None)
        else:
            a = base.copy()
    else:
        n = 2048
        a = np.full(n, 1.0, dtype=np.float64)

    # Normalise to the canonical sum √(2·n) – this makes the objective equal to max(conv)
    target_sum = np.sqrt(2.0 * n)
    eps_min = 1e-8
    a = np.maximum(a, eps_min)
    a = a * (target_sum / a.sum())

    best_a = a.copy()
    best_score = _objective_score(a)
    print(f"[{time.monotonic() - start:.2f}s] initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # Main search loop – several random restarts, each consisting of
    # an LP phase followed by a tiny sub‑gradient polishing.
    # --------------------------------------------------------------
    RESTART_WAIT = 120.0          # seconds without improvement → new random start
    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        # Restart if stuck for a while
        if now - last_improve > RESTART_WAIT:
            a = rng.random(n) + eps_min
            a = a * (target_sum / a.sum())
            last_improve = now
            print(f"[{now - start:.2f}s] restart with fresh random vector")

        # ---- Phase A – LP “Frank‑Wolfe” steps ---------------------------------
        max_lp_iters = 30
        for it in range(max_lp_iters):
            if time.monotonic() >= deadline:
                break

            conv = _conv_self_fft(a)
            M = conv.max()

            b_opt = _solve_lp_max_sum(a, M)
            if b_opt is None:
                # fallback: stop LP phase
                break

            sum_b = b_opt.sum()
            if sum_b <= target_sum * 1.01:   # not enough mass gain ⇒ stop early
                break

            # Scale b to the canonical sum.
            g = b_opt * (target_sum / sum_b)
            g = np.maximum(g, eps_min)      # keep strict positivity

            # Diminishing convex‑combination step size.
            t = 0.4 * (0.94 ** it)          # starts at 0.4, shrinks quickly
            a = (1.0 - t) * a + t * g

            # Evaluate and keep the best.
            cur_score = _objective_score(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start:.2f}s] LP improvement {best_score:.6f}")

            # Stop LP loop early if progress stalls.
            if it > 0 and cur_score >= best_score - 1e-12:
                # no improvement in this iteration
                continue

        # ---- Phase B – cheap sub‑gradient polishing -------------------------
        sub_step = 0.12
        for _ in range(40):
            if time.monotonic() >= deadline:
                break
            conv = _conv_self_fft(a)
            kmax = int(np.argmax(conv))

            # Gradient of the active offset (see analysis)
            idx = np.arange(n)
            j = kmax - idx
            mask = (j >= 0) & (j < n)
            grad = np.zeros_like(a)
            grad[mask] = 2.0 * a[j[mask]]

            # Projected update (onto the simplex of sum = target_sum)
            cand = _project_simplex_target(a - sub_step * grad, target_sum)
            cand_score = _objective_score(cand)

            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = cand.copy()
                a = cand
                sub_step *= 1.07
                last_improve = time.monotonic()
                print(f"[{last_improve - start:.2f}s] sub‑grad improvement {best_score:.6f}")
            else:
                sub_step *= 0.72

        # short jitter to escape tiny plateaus
        if rng.random() < 0.05:
            jitter = rng.normal(0.0, 0.001, size=n)
            a = _project_simplex_target(a + jitter, target_sum)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
