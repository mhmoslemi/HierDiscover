"""
Propose a step‑function (list of non‑negative heights) that minimises
``evaluate_sequence``.

Algorithm
---------
1. **Initialisation** – build a small portfolio of sensible seeds:
   uniform, triangular, sin‑squared, geometric and a random Dirichlet
   vector (lengths 1024, 2048 and 4096).  If the global variable
   ``height_sequence_1`` is present it is also added (padded / trimmed
   to the current length).  The best seed (by the exact objective
   ``2·n·max(conv)``) is taken as the starting point.

2. **Exponentiated‑gradient descent on the true max‑convolution** –  
   for the first ≈ 50 % of the time budget we iteratively move mass
   away from indices that contribute most to the current maximum
   (gradient of the max‑convolution) by a multiplicative update
   ``a_i ← a_i·exp(‑η·grad_i)`` and re‑normalise onto the simplex.
   The learning rate η is linearly annealed from a relatively large
   value (≈ 2 / n) down to a tiny one (≈ 1e‑4 / n).

3. **Smooth‑max Adam optimisation** – the remaining ≈ 35 % of the
   budget is spent on the classic soft‑max surrogate used in the
   reference solution, but with a slower temperature decay
   (τ∈[0.5,5e‑6]) and a finer learning‑rate schedule.  Gradients are
   obtained by FFT‑based correlation; Adam moments are updated and the
   iterate is projected onto the simplex.

4. **Fine‑grained stochastic local search** – the final ≈ 15 % of the
   budget performs tiny mass‑transfer moves (pick a source and a target,
   move a small fraction of the source mass) and accepts a move only
   if the exact score improves.  Periodically a tiny jitter is added to
   avoid stagnation.

5. **Optional pruning** – after the time expires we try a few
   aggressive pruning steps (keep only the largest‑mass entries,
   re‑normalise and evaluate) and keep the best among those.

All intermediate best scores are printed.  The routine respects the
5‑second safety margin required by the harness.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D vectors.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a simplex‑normalised vector."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def _max_conv_gradient(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Gradient of the (non‑smoothed) max‑convolution w.r.t. ``a``.
    For each index k where ``conv[k]`` attains the maximum we add the
    contribution 2·a_{k‑i} for all admissible i.
    """
    n = a.shape[0]
    max_val = conv.max()
    ks = np.where(conv == max_val)[0]
    grad = np.zeros_like(a)
    for k in ks:
        if k < n:
            # grad[0..k] += 2 * a[k], a[k‑1], ..., a[0]
            grad[: k + 1] += 2.0 * a[k :: -1]
        else:
            start = k - (n - 1)          # first i with a valid contribution
            idx = np.arange(start, n)    # i = start … n‑1
            grad[idx] += 2.0 * a[k - idx]
    return grad


def _softmax_weights(conv: np.ndarray, tau: float) -> np.ndarray:
    """Compute soft‑max weights w_i = exp(conv_i / τ) / Σ exp(...)."""
    shift = conv.max()
    exp_shifted = np.exp((conv - shift) / tau)
    w = exp_shifted / exp_shifted.sum()
    return w


def _smooth_max_adam(
    a: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best sequence and its exact score seen during the run.
    """
    n = a.shape[0]
    fft_len = _next_pow2(3 * n)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # annealing schedule
    TAU_START, TAU_END = 0.5, 5e-6
    LR_START, LR_END = 0.5, 5e-4

    phase_deadline = start_time + 0.35 * (deadline - start_time)   # ~35 % of total budget
    iteration = 0
    conv = np.convolve(a, a)   # initialise once
    while time.monotonic() < phase_deadline:
        now = time.monotonic()
        frac = (now - start_time) / (phase_deadline - start_time)

        tau = TAU_START * (1.0 - frac) + TAU_END * frac
        lr = LR_START * (1.0 - frac) + LR_END * frac

        # ----- smooth‑max weights -----
        w = _softmax_weights(conv, tau)

        # ----- gradient of smooth surrogate (FFT) -----
        a_fft = np.fft.rfft(a, n=fft_len)
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr : start_corr + n]

        # ----- Adam step -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        # recompute convolution for the next iteration
        conv = np.convolve(a, a)

        # occasional exact evaluation
        if iteration % 200 == 0:
            score = 2.0 * n * conv.max()
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a.copy()
                elapsed = now - start_time
                print(
                    f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f} "
                    f"(iter={iteration})"
                )
                if best_score <= 1.5025:
                    print("[early stop] target reached in Adam phase")
                    break

        iteration += 1
    return best_seq, best_score


def _local_refinement(
    a: np.ndarray,
    start_time: float,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """Tiny stochastic mass‑transfer moves (exact evaluation)."""
    n = a.size
    iteration = 0
    while time.monotonic() < deadline:
        # pick source and target
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (log‑uniform 0.1 % – 5 %)
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[i] * frac_move

        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta   # sum stays 1

        score = _objective(a_candidate)
        iteration += 1

        if score < best_score - 1e-12:
            best_score = score
            best_seq = a_candidate.copy()
            a = a_candidate
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} "
                f"(move #{iteration})"
            )
            if best_score <= 1.5025:
                print("[early stop] target reached in local phase")
                break

        # occasional jitter
        if iteration % 4000 == 0:
            width = rng.integers(1, max(2, n // 50))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            a_j = a.copy()
            a_j[start : start + width] *= factor
            a = _project_onto_simplex(a_j)

    return best_seq, best_score


def _prune_candidates(
    a: np.ndarray, rng: np.random.Generator, budget: float, start: float
) -> tuple[np.ndarray, float]:
    """
    Very cheap post‑processing: keep only a fraction of the largest entries,
    re‑normalise and keep the best (exact evaluation).  The function stops
    when the remaining time is below ``budget`` seconds.
    """
    best_seq = a.copy()
    best_score = _objective(best_seq)
    n = a.size
    # try several pruning ratios – from 90 % down to 30 %
    for frac in [0.9, 0.8, 0.7, 0.6, 0.5, 0.4, 0.3]:
        if time.monotonic() - start > budget:
            break
        k = max(2, int(frac * n))
        # indices of the k largest masses (preserve order)
        idx = np.argpartition(a, -k)[-k:]
        mask = np.zeros_like(a, dtype=bool)
        mask[idx] = True
        a2 = np.where(mask, a, 0.0)
        tot = a2.sum()
        if tot == 0.0:
            continue
        a2 /= tot
        sc = _objective(a2)
        if sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.2f}s] new best (prune {int(frac*100)}%) = {best_score:.6f}")
    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative step function that minimises ``evaluate_sequence``.
    The routine respects a 5‑second safety margin and prints progress.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Build a portfolio of initial seeds (different shapes, lengths)
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]           # a few manageable sizes
    init_vectors = []

    # helper to add a vector (clipped, normalised)
    def _add_vec(v: np.ndarray):
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s > 0.0:
            v = v / s
            init_vectors.append(v)
        else:
            # fallback to uniform
            init_vectors.append(np.full_like(v, 1.0 / v.size))

    # shapes for each length
    for n in lengths:
        # uniform
        _add_vec(np.full(n, 1.0 / n))

        # triangular (symmetric, peak in the centre)
        half = n // 2
        inc = np.arange(1, half + 1, dtype=np.float64)
        if n % 2 == 0:
            tri = np.concatenate([inc, inc[::-1]])
        else:
            tri = np.concatenate([inc, [half + 1], inc[::-1]])
        _add_vec(tri)

        # sin^2 shape (smooth hill)
        i = np.arange(n, dtype=np.float64)
        s = np.sin(np.pi * (i + 0.5) / n) ** 2
        _add_vec(s)

        # geometric decay
        r = 0.99
        g = (1.0 - r) * (r ** i)
        _add_vec(g)

        # random Dirichlet
        _add_vec(rng.dirichlet(np.ones(n)))

    # incorporate the global variable if present
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        for n in lengths:
            if base.size > n:
                seq = base[:n]
            elif base.size < n:
                seq = np.pad(base, (0, n - base.size), constant_values=0.0)
            else:
                seq = base.copy()
            _add_vec(seq)

    # ------------------------------------------------------------------
    # 1) Pick the best seed using the exact objective
    # ------------------------------------------------------------------
    best_seq = None
    best_score = np.inf
    for vec in init_vectors:
        sc = _objective(vec)
        if sc < best_score:
            best_score = sc
            best_seq = vec.copy()
    print(f"[0.00s] best seed score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 3) Phase 1 – exponentiated‑gradient descent on the true max
    # ------------------------------------------------------------------
    a = best_seq.copy()
    n = a.size
    phase1_deadline = start_time + 0.55 * (deadline - start_time)   # ≈ 55 %
    conv = np.convolve(a, a)      # initialise once

    # schedule for η: start relatively large, end very small
    eta_start = 2.0 / n          # roughly 2·max grad / n
    eta_end = 1e-4 / n

    while time.monotonic() < phase1_deadline:
        # gradient of current max‑convolution
        grad = _max_conv_gradient(a, conv)

        # anneal η
        now = time.monotonic()
        frac = (now - start_time) / (phase1_deadline - start_time)
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # multiplicative update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a /= s

        # recompute convolution for the next iteration
        conv = np.convolve(a, a)

        # evaluate exactly every few iterations
        if int(frac * 2000) % 200 == 0:          # roughly every ~0.05 % of phase
            score = 2.0 * n * conv.max()
            if score < best_score - 1e-12:
                best_score = score
                best_seq = a.copy()
                elapsed = now - start_time
                print(
                    f"[{elapsed:6.2f}s] new best (exp‑grad) = {best_score:.6f}"
                )
                if best_score <= 1.5025:
                    print("[early stop] target reached in exp‑grad phase")
                    break

    # ------------------------------------------------------------------
    # 4) Phase 2 – smooth‑max Adam optimisation (more fine‑grained)
    # ------------------------------------------------------------------
    best_seq, best_score = _smooth_max_adam(
        best_seq, start_time, deadline, best_seq, best_score
    )
    a = best_seq.copy()   # start local refinements from Adam's outcome

    # ------------------------------------------------------------------
    # 5) Phase 3 – stochastic local refinement (tiny mass transfers)
    # ------------------------------------------------------------------
    best_seq, best_score = _local_refinement(
        a, start_time, deadline, best_seq, best_score, rng
    )

    # ------------------------------------------------------------------
    # 6) Optional aggressive pruning before returning (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 2.0:   # give a tiny buffer
        best_seq, best_score = _prune_candidates(best_seq, rng, remaining, start_time)

    # ------------------------------------------------------------------
    # 7) Final sanity checks and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = _objective(best_seq)
    total_time = time.monotonic() - start_time
    print(f"Finished in {total_time:.2f}s – final score = {final_score:.6f}")

    # convert to plain Python list of floats
    return [float(v) for v in best_seq]
