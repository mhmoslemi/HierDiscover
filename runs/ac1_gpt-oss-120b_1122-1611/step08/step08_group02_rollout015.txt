def compute_counts(support, n):
    L = 2*n-1
    counts = np.zeros(L, dtype=np.int32)
    m = len(support)
    # self pairs
    for idx in support:
        counts[2*idx] += 1
    # cross pairs
    for i in range(m):
        vi = support[i]
        for j in range(i+1, m):
            s = vi + support[j]
            counts[s] += 2
    return counts
```

def attempt_move(support, support_set, counts, n, rng):
    m = len(support)
    # choose out
    out_idx = rng.choice(support)
    # choose new not in support
    while True:
        in_idx = rng.integers(0, n)
        if in_idx not in support_set:
            break
    # remove contributions
    counts_new = counts.copy()
    # self removal
    counts_new[2*out_idx] -= 1
    # cross removal: iterate over support except out_idx
    for j in support:
        if j == out_idx:
            continue
        counts_new[out_idx + j] -= 2
    # add contributions
    counts_new[2*in_idx] += 1
    for j in support:
        if j == out_idx:
            continue
        counts_new[in_idx + j] += 2
    # compute max
    new_max = int(counts_new.max())
    return in_idx, out_idx, counts_new, new_max
```

"""
Propose a step‑function with a low value of
    2·n·max(conv(a,a)) / (sum(a)^2)

Algorithm
---------
1. **Continuous optimisation (≈40 % of the budget)**
   – Initialise a probability vector `a` (sum = 1) on a moderate size
     (n = 1024, 2048 or 4096).
   – minimise a smooth surrogate of the L∞‑norm of the self‑convolution:
        Sβ(a) = (1/β)·log Σ_k exp(β·(a*a)_k) .
     The gradient of `Sβ` can be written using convolutions, so each
     iteration costs a single FFT.
   – β is increased slowly (β←β·1.02, capped at 30) and the step size
     is decreased (step←step·0.9995).  Whenever a better objective
     (the true max‑convolution) is found, the current vector is stored.

2. **Sparse‑support optimisation (≈60 % of the budget)**
   – Treat the sequence as a binary mask (value 1 on a support set S,
     0 elsewhere).  For a support `S` the objective reduces to

        score = 2·n·max_count / |S|²

     where `max_count` is the largest number of ordered pairs (i,j)∈S²
     with i+j equal to the same sum.
   – Initialise `S` randomly (size ≈ √(2·n)·factor) and keep the
     convolution counts in an integer array.
   – Repeatedly propose a “swap” – move one index out of the support
     and insert a new one not currently present.  The count array can
     be updated in O(|S|) time, the new score is evaluated, and the
     swap is kept iff it lowers the score.
   – If no improvement occurs for a few seconds a random restart is
     performed.  The best mask found over the whole run is kept.

3. The best sequence from the two phases is normalised (sum = 1) and
   returned.  All values are clipped to the range required by
   `evaluate_sequence`.

The search runs for the full time budget (default 1000 s) and prints
progress information.  The returned list can be fed directly to
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((int(x) - 1).bit_length())


def _conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a using an FFT."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (no clamping)."""
    n = a.size
    max_b = _conv_max(a)
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# 1) Continuous optimisation – gradient descent on a smooth max‑softening
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Soft‑max weights w_k ∝ exp(beta·conv_k) that sum to 1."""
    # shift by the maximum to avoid overflow
    shift = conv.max()
    w = np.exp(beta * (conv - shift))
    total = w.sum()
    if total == 0.0:
        # fallback – uniform weights
        return np.full_like(w, 1.0 / w.size)
    return w / total


def gradient_optimize(
    n: int,
    seed: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the smooth surrogate of the L∞ norm.
    Returns the best vector (sum = 1) and its true score.
    """
    rng = np.random.default_rng(seed)
    a = np.full(n, 1.0 / n, dtype=np.float64)   # start with a uniform density
    best_a = a.copy()
    best_score = compute_score(best_a)

    L = _next_pow2(2 * n - 1)

    beta = 1.0          # start with a very soft max
    step = 0.5          # initial learning rate

    last_print = time.monotonic()

    while time.monotonic() < deadline:
        # ----- compute convolution and true score -----------------------
        fa = np.fft.rfft(a, n=L)
        conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
        cur_score = 2.0 * n * conv.max()      # sum(a) == 1

        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            # print progress every couple of seconds
            now = time.monotonic()
            if now - last_print > 2.0:
                print(f"[{now:.2f}s] gradient new best (n={n}) → {best_score:.6f}")
                last_print = now

        # ----- compute gradient of the smooth max ----------------------
        w = _softmax_weights(conv, beta)               # shape (2n‑1,)
        fw = np.fft.rfft(w, n=L)
        grad_fft = fa * fw
        grad_full = np.fft.irfft(grad_fft, n=L)[: 2 * n - 1]
        # gradient component for a_i (ordered pairs) – slice centred at n‑1
        grad = 2.0 * grad_full[n - 1 : n - 1 + n]

        # ----- gradient step -------------------------------------------
        a_proposed = a - step * grad
        a_proposed = np.clip(a_proposed, 0.0, None)
        s = a_proposed.sum()
        if s == 0.0:
            a_proposed[:] = 1.0 / n
        else:
            a_proposed /= s

        # evaluate proposed point (cheap: reuse FFT)
        fa2 = np.fft.rfft(a_proposed, n=L)
        conv2 = np.fft.irfft(fa2 * fa2, n=L)[: 2 * n - 1]
        proposed_score = 2.0 * n * conv2.max()

        # accept if the true objective improves (or with a tiny random chance)
        if proposed_score <= cur_score or rng.random() < 0.02:
            a = a_proposed

        # ----- schedule updates -----------------------------------------
        beta = min(beta * 1.02, 30.0)          # make the surrogate sharper
        step = max(step * 0.9995, 1e-7)        # slowly shrink the step

        # occasional jitter to escape shallow plateaus
        if rng.random() < 0.01:
            a *= np.exp(rng.normal(0.0, 0.01, size=n))
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            a /= a_sum if a_sum > 0 else 1.0

    return best_a, best_score


# ----------------------------------------------------------------------
# 2) Sparse support optimisation – integer‑count based hill climbing
# ----------------------------------------------------------------------


def compute_counts(support: np.ndarray, n: int) -> np.ndarray:
    """
    For a binary mask (value 1 on `support`) compute the ordered‑pair
    count array C where C[k] = #{ (i,j)∈S² : i+j = k }.
    Length of the result is 2·n‑1.
    """
    L = 2 * n - 1
    counts = np.zeros(L, dtype=np.int32)
    m = support.size

    # self‑pairs (i,i)
    for i in support:
        counts[2 * i] += 1

    # distinct pairs (i,j) with i<j contribute two ordered pairs
    for ix in range(m):
        i = support[ix]
        for jy in range(ix + 1, m):
            j = support[jy]
            counts[i + j] += 2
    return counts


def improve_support(
    support: np.ndarray,
    n: int,
    deadline: float,
    rng: np.random.Generator,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing on a binary mask.
    Returns the best support found and its score (2·n·max_count/|S|²).
    """
    m = support.size
    support_set = set(support.tolist())
    counts = compute_counts(support, n)
    max_count = int(counts.max())
    best_score = 2.0 * n * max_count / (m * m)
    best_support = support.copy()
    best_counts = counts.copy()
    best_max = max_count

    last_improve = time.monotonic()
    stagnation_limit = 6.0   # seconds without improvement → restart

    # Convert to Python list for fast removal/insertion in the loop
    sup_list = list(support)

    while time.monotonic() < deadline:
        # ---- propose a swap -------------------------------------------------
        out_idx = rng.choice(sup_list)          # donor
        # ensure the new index is not already present
        in_idx = rng.integers(0, n)
        while in_idx in support_set:
            in_idx = rng.integers(0, n)

        # ---- compute the new count array (O(m)) ----------------------------
        new_counts = best_counts.copy()
        # remove contributions of out_idx
        new_counts[2 * out_idx] -= 1
        for j in sup_list:
            if j == out_idx:
                continue
            new_counts[out_idx + j] -= 2

        # add contributions of in_idx
        new_counts[2 * in_idx] += 1
        for j in sup_list:
            if j == out_idx:
                continue
            new_counts[in_idx + j] += 2

        new_max = int(new_counts.max())
        new_score = 2.0 * n * new_max / (m * m)

        # ---- accept if improving --------------------------------------------
        if new_score < best_score:
            # update bookkeeping
            best_score = new_score
            best_max = new_max
            best_counts = new_counts
            # modify the support list / set
            sup_list[sup_list.index(out_idx)] = in_idx
            support_set.remove(out_idx)
            support_set.add(in_idx)

            print(f"    [{time.monotonic():.2f}s] support improvement → {best_score:.6f}")
            last_improve = time.monotonic()
        # ---- stagnation handling --------------------------------------------
        if time.monotonic() - last_improve > stagnation_limit:
            # random restart
            sup_list = list(rng.choice(n, size=m, replace=False))
            support_set = set(sup_list)
            best_counts = compute_counts(np.array(sup_list, dtype=np.int32), n)
            best_max = int(best_counts.max())
            best_score = 2.0 * n * best_max / (m * m)
            print(f"    [{time.monotonic():.2f}s] random restart → {best_score:.6f}")
            last_improve = time.monotonic()

    return np.array(sup_list, dtype=np.int32), best_score


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a 5‑second safety margin for the outer wrapper
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Phase 1 – smooth‑max gradient descent on a dense vector
    # --------------------------------------------------------------
    for n in (1024, 2048, 4096):
        if time.monotonic() >= deadline:
            break
        phase_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.35
        a, sc = gradient_optimize(n, rng.integers(0, 2 ** 31 - 1), phase_deadline)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] gradient phase improvement (n={n}) → {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – sparse‑support hill climbing
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick a problem size (the longer the better, but we cannot be too slow)
        n = int(rng.integers(2000, 8001))

        # target support size around √(2n) with a random factor
        sqrt2n = math.sqrt(2 * n)
        factor = rng.uniform(0.8, 2.0)
        m = max(2, min(n, int(sqrt2n * factor)))

        # initialise a random support
        init_support = np.sort(rng.choice(n, size=m, replace=False)).astype(np.int32)

        # give this run a few seconds (or a fraction of the remaining budget)
        sub_deadline = time.monotonic() + min(30.0, max(5.0, (deadline - time.monotonic()) * 0.12))

        support, score = improve_support(init_support, n, sub_deadline, rng)

        if score < best_score:
            # build the full vector (value 1 on the support, 0 elsewhere)
            seq = np.zeros(n, dtype=np.float64)
            seq[support] = 1.0
            best_seq = seq.copy()
            best_score = score
            print(f"[{time.monotonic() - start_time:.2f}s] sparse phase new best (n={n}, |S|={m}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final safety fixes and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a simple uniform sequence of moderate size
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # protect against forbidden values (negative / >1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / best_seq.size
    else:
        best_seq /= total                     # normalise to sum = 1 (scaling is irrelevant)

    elapsed = time.monotonic() - start_time
    print(f"Finished search in {elapsed:.2f}s, final best score {best_score:.6f}")

    return [float(x) for x in best_seq]
