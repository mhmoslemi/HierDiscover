def simplex_projection(v):
    # returns projection of v onto simplex {x >=0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive => return uniform
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _subgradient_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    n = a.shape[0]
    grad = np.zeros_like(a)
    # indices k where i_max - k in [0,n-1]
    start = max(0, max_idx - (n - 1))
    end = min(max_idx, n - 1)
    # for k from start to end inclusive
    k = np.arange(start, end + 1)
    grad[k] = 2.0 * a[max_idx - k]
    return grad

        if a.ndim != 1 or a.size ==0:
            n=2048
            a = np.full(n, 1.0/n)
        else:
            n = a.size
            a = np.clip(a, 0, None)
            total = a.sum()
            if total>0:
                a /= total
            else:
                a = np.full(n, 1.0/n)
    else:
        n = 4096
        # arcsine shape initialization
        i = np.arange(n, dtype=np.float64)
        a = 1.0/np.sqrt((i+0.5)*(n - i - 0.5))
        a /= a.sum()
    # best tracking
    best = a.copy()

Propose a step‑function (non‑negative sequence) that minimises
``evaluate_sequence``.

Algorithm (sub‑gradient descent on the true max‑convolution):
-------------------------------------------------------------
1. Work on the simplex (∑a = 1).  Initialise with a “β‑distribution”
   (the discrete arcsine law) which is already known to give a
   relatively flat self‑convolution.  If a previous construction
   ``height_sequence_1`` exists we start from its normalised version.
2. Repeatedly:
   * Compute the linear self‑convolution via a single FFT.
   * Identify the index where the convolution attains its maximum.
   * The sub‑gradient of the objective 2·n·max(conv) w.r.t. ``a`` is
     easily obtained: for a given index `k` the derivative equals
     ``2·a[k−i]`` for all admissible `i`.  This is implemented by
     `_subgradient_max`.

     project back onto the probability simplex.

     of stagnation, restart from the current best solution with moderate
     noise.  This prevents the method from getting trapped in a local
     plateau.
3. The best sequence seen (exactly evaluated with ``evaluate_sequence``)
   is returned after the time budget (1000 s minus a 5 s safety margin).

The method is simple, deterministic apart from the random perturbations,
and usually beats the previous score 1.513206 within the allotted time.
"""

def _next_pow2(x: int) -> int:
    """Return the smallest power of two that is ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns the first ``2·len(a)-1`` entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are non‑positive – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry with respect to ``a``.
    ``max_idx`` is the index where ``a*a`` attains its maximum.
    The gradient is 2·a[max_idx‑k] for all admissible k, otherwise 0.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    # k such that i_max - k is a valid index
    start = max(0, max_idx - (n - 1))
    end = min(max_idx, n - 1)
    if start <= end:
        ks = np.arange(start, end + 1)
        grad[ks] = 2.0 * a[max_idx - ks]
    return grad

    # 5) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        # ----------- compute convolution and current score -----------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        cur_score = 2.0 * n * max_val          # sum(a)=1 ⇒ denominator = 1

        # ----------- update best solution --------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = time.monotonic()
            elapsed = last_improve - start
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------- sub‑gradient step -----------------------------------
        grad = _subgradient_max(a, max_idx)

        # Diminishing step size (≈ 1/√t)
        step = step0 / np.sqrt(iteration + 1.0)

        a = a - step * grad
        # projection onto the simplex keeps the sum exactly 1
        a = _project_simplex(a)

        # ----------- occasional jitter ------------------------------------
        if rng.random() < jitter_prob:
            jitter = np.exp(rng.normal(0.0, 0.2, size=n))
            a = a * jitter
            a = _project_simplex(a)

        # ----------- stagnation handling (soft restart) ------------------
        now = time.monotonic()
        if now - last_improve > stagnation_limit:
            # restart from the current best with moderate log‑normal noise
            noise = np.exp(rng.normal(0.0, restart_noise, size=n))
            a = best * noise
            a = _project_simplex(a)
            last_improve = now
            elapsed = now - start
            print(f"[{elapsed:6.2f}s] restart after stagnation, best {best_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
