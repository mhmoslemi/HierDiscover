def project_simplex(v):
    # v is 1D array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

    beta = 0.5

def target_hill_climb(a, deadline, rng):
    n = a.size
    best = a.copy()
    best_score = compute_score(best)
    last_improve = time.monotonic()
    stagnation = 0.0
    while time.monotonic() < deadline:
        conv = np.convolve(best, best)
        k_max = np.argmax(conv)
        # compute contribution weights for each i:
        # contribution = best[i] * best[k_max - i] if valid
        contrib = np.zeros_like(best)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = best[i] * best[j]
        if contrib.sum() == 0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(n, p=contrib / contrib.sum())
        # choose recipient where contribution low
        low = 1.0 - (best * (np.concatenate([best, np.zeros_like(best)])[:len(best)])) # not good
        # simpler: choose recipient from indices with small best
        inv = 1.0 - best
        inv[donor] = 0.0
        if inv.sum() > 0:
            recipient = rng.choice(n, p=inv / inv.sum())
        else:
            recipient = rng.integers(n)
        delta = rng.random() * 0.3 * best[donor]
        a_new = best.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta
        a_new /= a_new.sum()
        new_score = compute_score(a_new)
        if new_score < best_score:
            best = a_new
            best_score = new_score
            last_improve = time.monotonic()
        if time.monotonic() - last_improve > 5.0:
            # jitter
            a_new = best * np.exp(rng.normal(0.0, 0.05, size=n))
            a_new = project_simplex(a_new)
            new_score = compute_score(a_new)
            if new_score < best_score:
                best = a_new
                best_score = new_score
                last_improve = time.monotonic()
    return best, best_score

Propose a step‑function with a lower bound for ``evaluate_sequence``.

Algorithm
---------
1. **Exploration (≈30 % of the time)**
   – Try a modest set of lengths (256 … 8000) and a few classic families:
     uniform, several fixed Beta‑shapes, and many random Beta
     parameters (α,β∈[0.1,5]).
   – Also sample Dirichlet vectors with concentration < 1 for “spiky”
     starts.
   – Keep the best sequence seen.

2. **Gradient‑descent refinement**
   – Work on the current best length.
   – Minimise a smooth approximation of the objective,
     ``softmax_β(conv)`` with an increasing temperature β.
   – The gradient of the weighted sum is computed in O(n log n) by FFTs.
   – After each step the vector is projected onto the probability simplex.
   – β is slowly increased so that the smoothed objective converges to
     the true max‑convolution.

3. **Targeted stochastic hill‑climbing**
   – Starting from the gradient result, repeatedly move a small amount of
     mass from indices that contribute heavily to the current maximal
     convolution entry to indices that contribute little.
   – Occasional jitter and restarts avoid stagnation.

4. **Final normalisation**
   – Clip values to the limits imposed by ``evaluate_sequence`` and renormalise.

The mixture of a global search, a principled continuous optimisation,
and a final discrete hill‑climb is able to push the score well below the
previous bound (≈1.513) within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((int(x) - 1).bit_length())


def _max_self_convolution(a: np.ndarray) -> float:
    """Linear self‑convolution a*a using an FFT and return its maximum."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by ``evaluate_sequence`` (without its clamping)."""
    n = a.size
    max_b = _max_self_convolution(a)
    s = a.sum()
    if s < 0.01:                         # protection present in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def gen_beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) distribution on {0,…,n‑1}:
    a_i ∝ x_i^{α‑1}(1‑x_i)^{β‑1} with x_i = (i+½)/n.
    The result is normalised to sum = 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    raw = np.clip(raw, 0.0, None)
    s = raw.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / s


def project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σ x = 1}.  Implements the O(n) algorithm from
    Duchi et al., ICML 2008.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def self_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (full length 2·n‑1)."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def conv_rev_gradient(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Compute g_i = Σ_k a_k·w_{i+k}  (0≤i<n) using FFTs.
    This is the gradient of Σ w·(a*a) w.r.t. a.
    """
    a_rev = a[::-1]
    L = a_rev.size + w.size - 1
    L_fft = _next_pow2(L)
    Fa = np.fft.rfft(a_rev, n=L_fft)
    Fw = np.fft.rfft(w, n=L_fft)
    full = np.fft.irfft(Fa * Fw, n=L_fft)[:L]
    start = a.size - 1
    return full[start:start + a.size]


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------


def gd_optimize(start_a: np.ndarray,
                deadline: float,
                rng: np.random.Generator,
                start_time: float) -> tuple[np.ndarray, float]:
    """
    Gradient‑descent (with a smooth max) on the simplex.
    Returns the best vector seen and its true score.
    """
    a = start_a.copy()
    best_a = a.copy()
    best_score = compute_score(best_a)

    # parameters for the continuation on β
    beta = 0.5                     # start with a very smooth max
    max_beta = 250.0
    beta_inc = 1.02

    # step‑size control
    step_base = 0.1
    min_step = 1e-5

    # stagnation handling
    last_improve = time.monotonic()
    stagnation_limit = 12.0

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1) compute convolution and soft‑max weights
        # --------------------------------------------------------------
        conv_full = self_convolution(a)               # length 2n‑1
        max_conv = conv_full.max()
        # stable soft‑max
        shifted = conv_full - max_conv                 # ≤0
        expb = np.exp(beta * shifted)
        w = expb / expb.sum()                         # weights sum to 1

        # --------------------------------------------------------------
        # 2) gradient of the smoothed objective Σ w·(a*a)
        # --------------------------------------------------------------
        grad = 2.0 * conv_rev_gradient(a, w)          # length n

        # --------------------------------------------------------------
        # 3) simple backtracking line‑search on the true objective
        # --------------------------------------------------------------
        step = step_base
        improved = False
        while step > min_step:
            a_candidate = a - step * grad
            a_candidate = project_simplex(a_candidate)
            cand_score = compute_score(a_candidate)
            if cand_score < best_score - 1e-12:
                best_a = a_candidate.copy()
                best_score = cand_score
                a = a_candidate
                last_improve = time.monotonic()
                improved = True
                # slightly enlarge the step for the next iteration
                step_base = min(step_base * 1.2, 1.0)
                print(f"    [{time.monotonic() - start_time:.1f}s] gd → {best_score:.6f}")
                break
            step *= 0.5

        if not improved:
            # no improvement – apply a tiny random jitter
            a = a * np.exp(rng.normal(0.0, 0.02, size=a.shape))
            a = project_simplex(a)

        # --------------------------------------------------------------
        # 4) increase β (sharpen the max) and handle stagnation
        # --------------------------------------------------------------
        beta = min(beta * beta_inc, max_beta)

        if time.monotonic() - last_improve > stagnation_limit:
            # restart from the current best with a modest random perturbation
            a = best_a * np.exp(rng.normal(0.0, 0.08, size=a.shape))
            a = project_simplex(a)
            last_improve = time.monotonic()

    return best_a, best_score


def local_optimize(a: np.ndarray,
                   deadline: float,
                   rng: np.random.Generator,
                   start_time: float) -> tuple[np.ndarray, float]:
    """
    Targeted stochastic hill‑climbing.
    Moves mass from indices that heavily contribute to the current
    maximal convolution entry to indices that contribute little.
    """
    n = a.size
    best_a = a.copy()
    best_score = compute_score(best_a)

    last_improve = time.monotonic()
    stagnation_limit = 5.0

    while time.monotonic() < deadline:
        conv = np.convolve(best_a, best_a)          # length 2 n‑1
        k_max = int(conv.argmax())
        # contribution of each index to conv[k_max]
        contrib = np.zeros_like(best_a)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                contrib[i] = best_a[i] * best_a[j]
        if contrib.sum() == 0.0:
            donor = rng.integers(n)
        else:
            donor = rng.choice(n, p=contrib / contrib.sum())

        # Prefer a recipient with little contribution to the peak
        inv = 1.0 - best_a
        inv[donor] = 0.0
        if inv.sum() > 0.0:
            recipient = rng.choice(n, p=inv / inv.sum())
        else:
            recipient = rng.integers(n)

        # transfer a modest fraction of the donor's mass
        delta = rng.random() * 0.4 * best_a[donor]
        a_new = best_a.copy()
        a_new[donor] -= delta
        a_new[recipient] += delta
        a_new /= a_new.sum()

        new_score = compute_score(a_new)
        if new_score < best_score:
            best_a = a_new
            best_score = new_score
            last_improve = time.monotonic()
            print(f"    [{time.monotonic() - start_time:.1f}s] hill‑climb → {best_score:.6f}")

        # occasional jitter to escape plateaus
        if rng.random() < 0.012:
            a_jit = best_a * np.exp(rng.normal(0.0, 0.12, size=n))
            a_jit = project_simplex(a_jit)
            jit_score = compute_score(a_jit)
            if jit_score < best_score:
                best_a = a_jit
                best_score = jit_score
                last_improve = time.monotonic()
                print(f"    [{time.monotonic() - start_time:.1f}s] jitter → {best_score:.6f}")

        if time.monotonic() - last_improve > stagnation_limit:
            # random restart around the current best
            a_restart = best_a * np.exp(rng.normal(0.0, 0.1, size=n))
            a_restart = project_simplex(a_restart)
            restart_score = compute_score(a_restart)
            if restart_score < best_score:
                best_a = a_restart
                best_score = restart_score
                print(f"    [{time.monotonic() - start_time:.1f}s] restart → {best_score:.6f}")
            last_improve = time.monotonic()

    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (leave a small safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Initialise from a known construction if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / cand.size)
        sc = compute_score(cand)
        best_seq = cand.copy()
        best_score = sc
        print(f"[{0.00:.2f}s] start from known sequence → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 2) Exploration of families (≈30 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.30 * (deadline - start_time)

    n_candidates = [256, 512, 1024, 2048, 4096, 8000]
    special_params = [
        (2.0, 1.0),   # increasing ramp
        (1.0, 2.0),   # decreasing ramp
        (0.5, 0.5),   # arcsine (U‑shape)
        (0.3, 0.3),
        (0.2, 0.2),
        (1.5, 1.5),
    ]

    for n in n_candidates:
        if time.monotonic() >= explore_deadline:
            break

        # ----- uniform -----
        a_uni = np.full(n, 1.0 / n, dtype=np.float64)
        sc_uni = compute_score(a_uni)
        if sc_uni < best_score:
            best_seq = a_uni.copy()
            best_score = sc_uni
            print(f"[{time.monotonic() - start_time:.2f}s] new best (uniform n={n}) → {best_score:.6f}")

        # ----- fixed Beta families -----
        for α, β in special_params:
            a = gen_beta_seq(n, α, β)
            sc = compute_score(a)
            if sc < best_score:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.monotonic() - start_time:.2f}s] new best (β={α:.2f},{β:.2f} n={n}) → {best_score:.6f}")

        # ----- random Beta parameters -----
        for _ in range(20):
            if time.monotonic() >= explore_deadline:
                break
            α = np.exp(rng.uniform(np.log(0.1), np.log(5.0)))
            β = np.exp(rng.uniform(np.log(0.1), np.log(5.0)))
            a = gen_beta_seq(n, α, β)
            sc = compute_score(a)
            if sc < best_score:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.monotonic() - start_time:.2f}s] new best (rand β={α:.3f},{β:.3f} n={n}) → {best_score:.6f}")

        # ----- Dirichlet (low concentration) -----
        for _ in range(8):
            if time.monotonic() >= explore_deadline:
                break
            conc = rng.uniform(0.05, 0.8, size=n)          # low concentration → spiky starts
            a = rng.gamma(conc, 1.0)
            a /= a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_seq = a.copy()
                best_score = sc
                print(f"[{time.monotonic() - start_time:.2f}s] new best (dirichlet n={n}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Gradient‑descent refinement on the current best
    # ------------------------------------------------------------------
    if best_seq is not None:
        print(f"Starting gradient refinement (score {best_score:.6f}, length {best_seq.size})")
        refined_seq, refined_score = gd_optimize(best_seq, deadline, rng, start_time)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"Gradient refinement succeeded → new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Targeted stochastic hill‑climbing (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.monotonic() < deadline:
        print(f"Starting hill‑climbing (score {best_score:.6f}, length {best_seq.size})")
        polished_seq, polished_score = local_optimize(best_seq, deadline, rng, start_time)
        if polished_score < best_score:
            best_seq = polished_seq
            best_score = polished_score
            print(f"Hill‑climbing succeeded → new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Final safety fixes and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – simple uniform of moderate size
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce the clamps used inside evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / best_seq.size
    else:
        best_seq /= total

    elapsed = time.monotonic() - start_time
    print(f"Finished search in {elapsed:.2f}s, final best score {best_score:.6f}")

    return [float(x) for x in best_seq]
