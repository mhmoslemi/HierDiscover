"""
Hybrid optimizer for the self‑convolution problem.

The routine consists of four stages that run sequentially within the
time‑budget (a 5 s safety margin is kept).  All stages operate on the
simplex (non‑negative entries with unit sum) because the objective
2·n·max(conv) is scale‑invariant.

Stage 0 – Very cheap parametric sweep.  Power‑law and exponential families
          are examined; the best shape is kept as a warm start.

Stage 1 – Adam optimisation on a smooth log‑sum‑exp surrogate of the
          infinity‑norm of the self‑convolution.  The surrogate becomes
          sharper (β ramps from 5 to 800) so that the optimiser sees a
          close approximation of the true max.

Stage 2 – Projected sub‑gradient descent on the exact max‑convolution.
          The active offset is identified, its sub‑gradient is formed,
          and a Euclidean projection onto the simplex drives the iterate
          towards a flatter convolution.

Stage 3 – Simulated‑annealing refinement.  Gaussian jitter of decreasing
          amplitude is added, the perturbed vector is projected back onto
          the simplex and accepted with a Metropolis rule.  This helps
          escape the remaining plateaus and usually yields a small final
          improvement.

All helper functions are top‑level (no closures) and the code respects
the 1000 s wall‑clock limit.  The returned sequence is clipped to the
bounds imposed by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Soft‑max mapping logits → probability simplex (numerically stable)."""
    m = np.max(logits)
    e = np.exp(logits - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return e / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using an FFT of appropriate size."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (sum(a) is not forced)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, sum(x)=1}.
    """
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

    descent on the exact objective and a final simulated‑annealing polish.

    fft_len_conv = _next_pow2(2 * n - 1)

    while time.monotonic() < phase1_end:
        it += 1

        # logits → simplex
        a = _softmax_logits(x)

        # exact convolution (needed for the true objective)
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]

        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max                # denominator = 1 because sum(a)=1
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase1 new best {best_score:.6f}")

        # ----- smooth surrogate of max (log‑sum‑exp) -----
        beta = beta_end if it > beta_ramp_iters else (
            beta_start + (beta_end - beta_start) * it / beta_ramp_iters)
        scaled = beta * conv_full
        lse = logsumexp(scaled)                      # robust log‑sum‑exp
        w = np.exp(scaled - lse)                     # shape (2n‑1,), sums to 1

        # gradient of the surrogate w.r.t. a (correlation w ⋆ a)
        fft_len_corr = _next_pow2(3 * n - 2)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)

        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # chain rule → gradient w.r.t. logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to kick the optimiser out of plateaus
        if it % 500 == 0:
            x += rng.normal(0.0, 0.01, size=n)

        # hard restart if we have been stagnant for a while
        if time.monotonic() - last_improve > RESTART_WAIT:
            x = np.log(best_a + eps) + rng.normal(0.0, 0.2, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase1 restart")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    phase2_end = start + total_budget * 0.75
    a = best_a.copy()
    step = 0.2
    while time.monotonic() < phase2_end:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step *= 1.07                     # slightly enlarge step after success
            print(f"[{time.monotonic() - start:6.2f}s] subgrad improvement {best_score:.6f}")
        else:
            step *= 0.7                      # shrink after failure
            # occasional jitter to escape flat regions
            if step < 1e-8 or rng.random() < 0.001:
                jitter = rng.normal(0.0, 0.002, size=n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.01)

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing refinement (remaining budget)
    # ------------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = best_score
    sigma_start = 0.05
    sigma_end = 1e-6
    temp_start = 0.01
    temp_end = 1e-7

    while time.monotonic() < deadline:
        now = time.monotonic()
        # linear decay of sigma and temperature
        fraction = (deadline - now) / (deadline - phase2_end) if deadline > phase2_end else 0.0
        sigma = sigma_end + (sigma_start - sigma_end) * fraction
        temp = temp_end + (temp_start - temp_end) * fraction

        # propose a Gaussian‑perturbed candidate and re‑normalise onto the simplex
        proposal = cur_a + rng.normal(scale=sigma, size=n)
        proposal = np.maximum(proposal, 0.0)
        s = proposal.sum()
        if s > 0.0:
            proposal /= s
        else:
            proposal = cur_a.copy()

        val = _objective_score(proposal)

        # Metropolis acceptance
        if val < cur_score or rng.random() < np.exp((cur_score - val) / max(temp, 1e-12)):
            cur_a, cur_score = proposal, val
            if val < best_score:
                best_score = val
                best_a = cur_a.copy()
                print(f"[{now - start:6.2f}s] SA improvement {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
