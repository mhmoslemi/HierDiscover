"""
search for a step‑function sequence that minimizes ``evaluate_sequence``.
The algorithm performs a quick parametric sweep over several classic
window families (triangular, Hann, Gaussian, exponential, Kaiser,
DPSS), adds a handful of random Dirichlet draws, then refines the
best candidate with a mirror‑descent sub‑gradient method on the true
∞‑norm of the self‑convolution.  If time permits a final Adam/LSE
phase is run.  The routine respects the supplied time budget (with a
5‑second safety margin) and returns the best sequence found.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y = x - np.max(x)
    e = np.exp(y)
    s = e.sum()
    if s == 0:
        return np.full_like(x, 1.0 / x.size)
    return e / s

def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Linear self‑convolution a*a (FFT) and return its maximum entry."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())

def _conv_and_argmax(a: np.ndarray, fft_len: int):
    """Return (max value, arg‑max index) of the self‑convolution."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx

def _grad_max_conv(a: np.ndarray, k: int):
    """
    Sub‑gradient of the max‑convolution term for a given arg‑max index k.
    Returns a vector g where g_i = 2·a_{k-i} when the term exists,
    otherwise 0.
    """
    n = a.size
    g = np.zeros_like(a)
    start_i = max(0, k - n + 1)
    end_i = min(k, n - 1)  # inclusive
    # positions in g that are affected:
    g_start = k - end_i
    g_end = k - start_i + 1  # exclusive slice end
    g[g_start:g_end] = 2.0 * a[start_i:end_i + 1][::-1]
    return g

      * a quick sweep over several parametric families (triangular, Hann,
        Gaussian, exponential, Kaiser, DPSS) and a few random Dirichlet draws;
      * a mirror‑descent sub‑gradient refinement on the exact ∞‑norm

      * an optional Adam/LSE phase if time permits.

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

        a0 = np.asarray(init_seq, dtype=np.float64)
        n = a0.size
        s0 = a0.sum()
        if s0 > 0.0:
            a0 = a0 / s0
        else:

    else:

    # ------------------------------------------------------------------

        a = a * np.exp(-eta * g)

    # 8) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]

"""
search for a step‑function sequence that minimizes ``evaluate_sequence``.
The algorithm performs:

  1. a parametric sweep over classic window families (triangular, Hann,
     Gaussian, exponential, Kaiser, DPSS) and a handful of random
     Dirichlet draws;
  2. mirror‑descent sub‑gradient optimisation on the exact ∞‑norm of the
     self‑convolution (the true objective);
  3. an optional Adam/LSE phase if time permits.

The best sequence seen is returned.  All helper functions are defined at
module level (no closures) and the routine respects a 5‑second safety
margin before the supplied time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y = x - np.max(x)
    e = np.exp(y)
    s = e.sum()
    if s == 0.0:
        return np.full_like(x, 1.0 / x.size)
    return e / s


def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return the maximum entry of the linear self‑convolution a*a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def _conv_and_argmax(a: np.ndarray, fft_len: int):
    """Return (max value, index of the max) of a*a."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv.max()), idx


def _grad_max_conv(a: np.ndarray, k: int):
    """
    Sub‑gradient of the max‑convolution term for a given arg‑max index k.
    g_i = 2·a_{k-i} whenever the term exists, otherwise 0.
    """
    n = a.size
    g = np.zeros_like(a)
    start_i = max(0, k - n + 1)
    end_i = min(k, n - 1)          # inclusive
    # slice in g that receives the reversed contribution
    g_start = k - end_i
    g_end = k - start_i + 1       # exclusive
    g[g_start:g_end] = 2.0 * a[start_i:end_i + 1][::-1]
    return g


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimise
    ``evaluate_sequence`` within the allotted time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------- 0) time budget (5 s safety margin) -------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------- 1) initialise ------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)) and len(init_seq) > 0:
        a0 = np.asarray(init_seq, dtype=np.float64)
        n = a0.size
        s0 = a0.sum()
        if s0 > 0.0:
            a0 = a0 / s0
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2048
        a0 = np.full(n, 1.0 / n)

    # ------------------- 2) FFT length (used everywhere) ---------------------------
    fft_len = _next_pow2(2 * n - 1)

    # ------------------- 3) keep the best solution ---------------------------------
    best_a = a0.copy()
    best_score = _conv_max(best_a, fft_len) * 2.0 * n
    print(f"[init] n = {n}, best_score = {best_score:.6f}")

    # ------------------- 4) Stage‑1: parametric sweep -------------------------------
    cand_list = []

    x = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0

    # triangular (tent) window
    tri = 1.0 - np.abs(x - centre) / centre
    tri = np.clip(tri, 0.0, None)
    cand_list.append(("triangular", tri / tri.sum()))

    # Hann window (cosine‑square)
    hann = 0.5 * (1.0 - np.cos(2.0 * np.pi * x / (n - 1)))
    cand_list.append(("hann", hann / hann.sum()))

    # Gaussian family
    for sigma_factor in (0.08, 0.12, 0.16, 0.20, 0.30, 0.40):
        sigma = sigma_factor * n
        gauss = np.exp(-0.5 * ((x - centre) / sigma) ** 2)
        cand_list.append((f"gauss_{sigma_factor:.2f}", gauss / gauss.sum()))

    # Exponential (Laplace) family
    for tau_factor in (0.08, 0.12, 0.16, 0.20, 0.30):
        tau = tau_factor * n
        expw = np.exp(-np.abs(x - centre) / tau)
        cand_list.append((f"exp_{tau_factor:.2f}", expw / expw.sum()))

    # Kaiser windows (requires scipy.special)
    try:

        for beta in (0.0, 2.0, 5.0, 8.0, 12.0, 16.0):

    # DPSS (Slepian) windows – scipy.signal.windows

    except Exception:

    # a few random Dirichlet draws
    for i in range(12):

    for label, a in cand_list:

    max_sub_iters = 500_000          # will stop earlier if time runs out

    sub_iter = 0
    a = best_a.copy()

    while time.monotonic() < deadline and sub_iter < max_sub_iters:
        cur_max, max_idx = _conv_and_argmax(a, fft_len)
        g = _grad_max_conv(a, max_idx)

        total = a.sum()
        if total == 0.0:

            a = a / total

        # occasional exact check (every 1000 iterations)
        if sub_iter % 1000 == 0:

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                print(f"[subgrad] iter {sub_iter:6d} new best = {best_score:.6f}")

        sub_iter += 1

    # ------------------- 7) Stage‑3: optional Adam/LSE refinement ----------------
    if time.monotonic() < deadline:
        # initialise log‑weights from the current best
        x = np.log(best_a + 1e-12)
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-8
        t = 0

        start_adam = time.monotonic()
        total_adam = deadline - start_adam

        # schedules
        gamma_start, gamma_end = 0.5, 300.0
        lr_start, lr_end = 1.0, 0.02

        while time.monotonic() < deadline:
            now = time.monotonic()

            shift = conv - conv.max()
            w = np.exp(gamma * shift)
