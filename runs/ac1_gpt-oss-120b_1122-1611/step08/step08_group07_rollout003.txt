def project_simplex(v):
    # v: 1D array
    # returns projection onto simplex {x>=0, sum(x)=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1

    u = np.sort(v)[::-1]

"""
Multi‑restart projected sub‑gradient descent + simple local refinement.

The objective to minimise is

    score(a) = 2·n·max_k (a*a)_k               (a is normalised, sum(a)=1)

because evaluate_sequence returns 2·n·max(b) / (sum(a)^2) and the denominator
is invariant under scaling.  The algorithm works on a few sequence lengths,
starts from random points (or from the provided height_sequence_1), and
repeatedly:

* evaluates the self‑convolution using an FFT,
* identifies the index k* where the convolution attains its maximum,
* computes the sub‑gradient of the max‑convolution (g_i = 2·a_{k*-i}),
* takes a step opposite to the gradient, projects back onto the simplex,
* enforces symmetry a_i = a_{n-1-i} (the optimal shape is believed to be
  symmetric) and repeats.

When progress stalls, a small random jitter is applied.  After the gradient
phase, a cheap greedy “mass‑transfer’’ local search is performed to polish
the best solution found.  The whole search respects a 1000‑second budget and
keeps the globally best sequence across all restarts and lengths.

The returned list contains non‑negative heights ready for evaluate_sequence().
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, sum=1}."""
    # Duchi et al. (2008) – O(n log n) implementation.
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is all zeros or slightly negative – return uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_max_convolution(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of (a*a)_k with respect to a.
    For k = argmax(a*a) we have g_i = 2·a_{k‑i} for indices where the term
    exists, otherwise 0.
    """
    n = a.size
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def _random_jitter(a: np.ndarray, rng: np.random.Generator, scale: float = 0.02) -> np.ndarray:
    """Add a small zero‑mean Gaussian jitter and re‑project onto the simplex."""
    a_pert = a + scale * rng.normal(size=a.shape)
    a_pert = np.clip(a_pert, 0.0, None)
    return _project_simplex(a_pert)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the current best (~1.5132).  The routine
    runs for at most ``budget_s`` seconds (a few seconds safety margin is
    subtracted internally).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global time budget (5 s safety margin for the outer harness)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Configuration – sequence lengths to try and iteration caps
    # ------------------------------------------------------------------
    lengths = [512, 1024, 2048, 4096, 8192]          # powers of two for fast FFTs
    # we will drop lengths that are too large for the remaining budget
    max_restarts_per_len = 4
    max_gd_iters = 4000          # gradient‑descent iterations per restart
    max_local_iters = 500        # local‑refinement iterations per restart
    target_score = 1.5025        # stop early if we beat the target

    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # 2) Main loop over lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.monotonic() >= deadline:
            break
        # allocate FFT length once for this n
        fft_len_self = _next_pow2(2 * n - 1)

        # ----------------------------------------------------------------
        # 2a) Prepare a starting point – use the provided sequence if it
        #     matches the current length, otherwise fall back to Dirichlet.
        # ----------------------------------------------------------------
        # The global variable height_sequence_1 may contain a promising candidate.
        init_from_global = False
        init_seq = globals().get("height_sequence_1")
        if isinstance(init_seq, (list, tuple, np.ndarray)):
            a0 = np.asarray(init_seq, dtype=np.float64)
            if a0.ndim == 1 and a0.size == n:
                a0 = np.clip(a0, 0.0, None)
                if a0.sum() == 0.0:
                    a0 = np.full(n, 1.0 / n)
                else:
                    a0 = a0 / a0.sum()
                init_from_global = True

        # ----------------------------------------------------------------
        # 2b) Multiple restarts for this length
        # ----------------------------------------------------------------
        for restart in range(max_restarts_per_len):
            if time.monotonic() >= deadline:
                break

            # initialise a
            if init_from_global and restart == 0:
                a = a0.copy()
            else:
                a = rng.random(n)
                a = a / a.sum()

            # enforce symmetry from the start (a_i == a_{n-1-i})
            a = (a + a[::-1]) / 2.0
            a = _project_simplex(a)

            # sub‑gradient descent parameters
            lr = 0.4                     # initial learning rate
            lr_decay = 0.9998
            momentum = np.zeros_like(a)
            beta_mom = 0.9

            last_improve = time.monotonic()

            # ----------------------------------------------------------------
            # Phase I – projected (momentum) sub‑gradient descent on max‑conv.
            # ----------------------------------------------------------------
            for it in range(max_gd_iters):
                if time.monotonic() >= deadline:
                    break

                # 1) convolution & max index
                c = _self_conv(a, fft_len_self)
                k = int(np.argmax(c))
                max_val = c[k]
                cur_score = 2.0 * n * max_val

                # 2) Track the best overall solution
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_seq = a.copy()
                    last_improve = time.monotonic()
                    elapsed = last_improve - start_time
                    print(f"[{elapsed:6.2f}s] New global best = {best_score:.6f} (n={n}, restart={restart})")
                    if best_score < target_score:
                        # early exit – excellent solution found
                        deadline = time.monotonic()  # force termination after loop
                        break

                # 3) sub‑gradient of the max‑convolution
                g = _subgradient_max_convolution(a, k)

                # 4) momentum update, projected gradient step
                momentum = beta_mom * momentum + (1.0 - beta_mom) * g
                a = a - lr * momentum

                # 5) enforce symmetry and simplex constraints
                a = (a + a[::-1]) / 2.0
                a = _project_simplex(a)

                # 6) learning‑rate decay
                lr *= lr_decay

                # 7) occasional jitter if stagnant
                now = time.monotonic()
                if now - last_improve > 20.0:
                    a = _random_jitter(a, rng, scale=0.015)
                    a = (a + a[::-1]) / 2.0
                    a = _project_simplex(a)
                    last_improve = now

                # optional early break if learning rate becomes tiny
                if lr < 1e-6:
                    break

            # ----------------------------------------------------------------
            # Phase II – cheap greedy local refinement (mass‑transfer)
            # ----------------------------------------------------------------
            # Use the best a obtained in Phase I as the starting point for
            # the local search.
            a_local = a.copy()
            last_local_improve = time.monotonic()
            for loc_it in range(max_local_iters):
                if time.monotonic() >= deadline:
                    break

                c = _self_conv(a_local, fft_len_self)
                k = int(np.argmax(c))
                max_val = c[k]
                cur_score = 2.0 * n * max_val

                # contribution of each coordinate to the offending peak
                lo = max(0, k - (n - 1))
                hi = min(n - 1, k)
                idx = np.arange(lo, hi + 1)
                contrib = a_local[idx] * a_local[k - idx]
                i_max = int(idx[np.argmax(contrib)])

                # pick a random destination that is not i_max
                j_dest = rng.integers(0, n)
                while j_dest == i_max:
                    j_dest = rng.integers(0, n)

                # transfer a modest fraction of the mass
                delta = a_local[i_max] * rng.uniform(0.02, 0.12)
                if delta > a_local[i_max]:
                    delta = a_local[i_max]

                a_candidate = a_local.copy()
                a_candidate[i_max] -= delta
                a_candidate[j_dest] += delta
                a_candidate = _project_simplex(a_candidate)
                a_candidate = (a_candidate + a_candidate[::-1]) / 2.0
                a_candidate = _project_simplex(a_candidate)

                c_new = _self_conv(a_candidate, fft_len_self)
                new_score = 2.0 * n * np.max(c_new)

                # accept improving moves
                if new_score < cur_score - 1e-12:
                    a_local = a_candidate
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best_seq = a_local.copy()
                        elapsed = time.monotonic() - start_time
                        print(f"[{elapsed:6.2f}s] New global best (local) = {best_score:.6f} (n={n})")
                    last_local_improve = time.monotonic()
                else:
                    # tiny simulated‑annealing acceptance to escape plateaus
                    remaining = deadline - time.monotonic()
                    if remaining > 0:
                        T = 0.015 * (remaining / (deadline - start_time))
                        if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                            a_local = a_candidate

                # if we have been stuck for a while, jitter a little
                if time.monotonic() - last_local_improve > 25.0:
                    a_local = _random_jitter(a_local, rng, scale=0.01)
                    a_local = (a_local + a_local[::-1]) / 2.0
                    a_local = _project_simplex(a_local)
                    last_local_improve = time.monotonic()

            # end of this restart – if we already beat the target, break early
            if best_score < target_score:
                break

        # after all restarts for this length, stop early if target already reached
        if best_score < target_score:
            break

    # ------------------------------------------------------------------
    # 3) Fallback – if nothing was found (unlikely) use a uniform distribution
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback)

    # ------------------------------------------------------------------
    # 4) Clip to the limits imposed by evaluate_sequence and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
