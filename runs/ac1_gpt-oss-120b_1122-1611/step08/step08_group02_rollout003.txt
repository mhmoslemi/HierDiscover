def compute_M(a):
    return _max_self_convolution(a)

def lp_max_sum(f, M):
    Build A_ub as conv_matrix(f) (size (2n-1) x n)
    b_ub = np.full(2*n-1, M)
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None

def iterate(f, M, rng):
    g = lp_max_sum(f, M)
    if g is None:
        return None, M
    sum_f = f.sum()
    sum_g = g.sum()
    if sum_g == 0:
        return None, M
    scaling = sum_f / sum_g
    g_scaled = g * scaling
    # line search for best t
    best_t = None
    best_M = M
    for t in [0.05, 0.1, 0.2, 0.4, 0.8]:
        new_f = (1 - t) * f + t * g_scaled
        new_f = np.clip(new_f, 0, None)
        new_f_sum = new_f.sum()
        if new_f_sum == 0:
            continue
        new_f /= new_f_sum
        new_M = compute_M(new_f)
        if new_M < best_M:
            best_M = new_M
            best_t = t
    if best_t is not None:
        new_f = (1 - best_t) * f + best_t * g_scaled
        new_f = np.clip(new_f, 0, None)
        new_f /= new_f.sum()
        return new_f, best_M
    else:
        return None, M

def solve_lp_max_sum(f, M):
    n = len(f)
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for i in range(n):
        A[i:i+n, i] = f
    # bound vector
    b = np.full(2*n-1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if res.success:
        return res.x
    else:
        return None

"""
LP‑driven refinement of step‑functions.

The core idea follows the “alternating linear program” described in the
statement.  Starting from a reasonable seed (the previously discovered
construction or a Beta‑shaped distribution) we repeatedly

1. compute M = max(f∗f);
2. solve the linear program   max   Σ g_i
                               s.t.  f∗g ≤ M ,  g ≥ 0
   (the convolution is linear in g);
3. rescale the LP‑solution to have the same ℓ₁–norm as f;
4. replace f by a convex combination (1‑t)·f + t·ĝ with a modest step‑size
   t.  If the new self‑convolution peak is lower we keep the move,
   otherwise we try a smaller t or perform a tiny random perturbation.

The process is repeated until the time budget expires.  All heavy work
(convolution via FFT, LP construction) is kept in top‑level helpers so
that the code respects the “no closures” rule.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic numerical utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((int(x) - 1).bit_length())


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum value of the linear self‑convolution a∗a."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    # Remove tiny negatives caused by round‑off.
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (without its clamping).
    The vector is assumed to be non‑negative; its sum is used as denominator.
    """
    n = a.size
    max_b = _max_self_convolution(a)
    s = a.sum()
    if s < 0.01:                      # matches ``evaluate_sequence`` guard
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def gen_beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) distribution on {0,…,n‑1}:
        a_i ∝ x_i^{α‑1}(1‑x_i)^{β‑1},  x_i = (i+½)/n .
    Result is normalised to sum = 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    raw = np.clip(raw, 0.0, None)
    s = raw.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / s


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Linear‑programming helpers (conv(f,g) ≤ M)
# ----------------------------------------------------------------------


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2·n‑1, n)) such that
        A @ g = f ∗ g   (linear convolution, zero‑padded).
    """
    n = f.size
    m = 2 * n - 1
    A = np.zeros((m, n), dtype=np.float64)
    # For column i we need the slice f[0:n] placed at rows i … i+n‑1.
    for i in range(n):
        A[i:i + n, i] = f
    return A


def solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve
        max Σ g_i   s.t.   f ∗ g ≤ M , g ≥ 0 .
    Returns the optimal g (or ``None`` on failure).
    """
    n = f.size
    A = _conv_matrix(f)               # (2n‑1) × n
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # linprog minimises, so use –Σg_i
    bounds = [(0.0, None)] * n

    # ``highs`` is fast and robust for the moderate sizes we use (n ≤ 2000).
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The routine consists of:

        * a quick exploration of Beta‑shaped families (different lengths);
        * picking the three best candidates as seeds;
        * for each seed, applying the LP‑based refinement loop described
          above until the time budget ends.

    The best sequence encountered is returned.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (leave a small safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5‑s safety buffer

    best_seq: np.ndarray | None = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Gather an initial pool of candidates
    # ------------------------------------------------------------------
    pool: list[tuple[np.ndarray, str]] = []

    # a) start from the previously discovered construction, if it exists
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s == 0.0:
            cand = np.full_like(cand, 1.0 / cand.size)
        else:
            cand = cand / s
        pool.append((cand, "global_seed"))

    # b) explicit Beta shapes for several lengths
    lengths = [512, 1024, 2048, 3809, 8000]  # mixture of small & large
    special_params = [(2.0, 1.0), (1.0, 2.0),
                      (0.5, 0.5), (1.5, 1.5),
                      (0.8, 1.2), (1.2, 0.8)]

    for n in lengths:
        # uniform baseline (useful for sanity checks)
        uni = np.full(n, 1.0 / n, dtype=np.float64)
        pool.append((uni, f"uniform_n{n}"))

        for α, β in special_params:
            a = gen_beta_seq(n, α, β)
            pool.append((a, f"beta_{α:.2f}_{β:.2f}_n{n}"))

        # a few random β‑parameters
        for _ in range(8):
            log_a = rng.uniform(np.log(0.2), np.log(5.0))
            log_b = rng.uniform(np.log(0.2), np.log(5.0))
            α = np.exp(log_a)
            β = np.exp(log_b)
            a = gen_beta_seq(n, α, β)
            pool.append((a, f"randbeta_{α:.2f}_{β:.2f}_n{n}"))

    # ------------------------------------------------------------------
    # 2) Evaluate pool and keep the three most promising seeds
    # ------------------------------------------------------------------
    scored_pool = [(compute_score(v), v, tag) for v, tag in pool]
    scored_pool.sort(key=lambda x: x[0])
    seeds = [item[1] for item in scored_pool[:3]]
    print(f"[{time.monotonic() - start_time:.2f}s] selected {len(seeds)} seeds "
          f"with best score {scored_pool[0][0]:.6f}")

    # ------------------------------------------------------------------
    # 3) LP‑based refinement for each seed
    # ------------------------------------------------------------------
    for seed_idx, f in enumerate(seeds):
        if time.monotonic() >= deadline:
            break

        # Normalise (ℓ₁ = 1) – the objective is scale‑invariant.
        f = np.clip(f, 0.0, None)
        if f.sum() == 0.0:
            f[:] = 1.0 / f.size
        else:
            f = f / f.sum()

        n = f.size
        M = _max_self_convolution(f)
        cur_score = compute_score(f)
        if cur_score < best_score:
            best_score = cur_score
            best_seq = f.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] seed {seed_idx} initial "
                  f"score → {best_score:.6f}")

        # Limit LP to moderate sizes (memory concerns).  If n > 2000 we
        # temporarily down‑sample for the LP step, then project back.
        lp_n = n if n <= 2000 else 2000
        downsample = (lp_n != n)

        # Helper to obtain a working copy of the current vector at size lp_n
        def _project_to_lp(x: np.ndarray) -> np.ndarray:
            if not downsample:
                return x.copy()
            # simple averaging into blocks
            block = n // lp_n
            return x[:block * lp_n].reshape(lp_n, block).mean(axis=1)

        def _expand_from_lp(x_lp: np.ndarray) -> np.ndarray:
            if not downsample:
                return x_lp.copy()
            # repeat each entry to obtain the original length (rough)
            block = n // lp_n
            expanded = np.repeat(x_lp, block)
            # if there is a remainder, pad with the last value
            if expanded.size < n:
                expanded = np.concatenate([expanded,
                                           np.full(n - expanded.size, x_lp[-1])])
            return expanded

        # ------------------------------------------------------------------
        # Iterative refinement loop
        # ------------------------------------------------------------------
        iter_no = 0
        stagnation_start = time.monotonic()
        while time.monotonic() < deadline:
            iter_no += 1

            # --- 3.1) solve LP on the (possibly down‑sampled) vector ----------
            f_lp = _project_to_lp(f)
            M_lp = _max_self_convolution(f_lp)

            g_lp = solve_lp_max_sum(f_lp, M_lp)
            if g_lp is None:
                # LP failed – abort refinement of this seed.
                break

            sum_f_lp = f_lp.sum()
            sum_g_lp = g_lp.sum()
            if sum_g_lp == 0.0:
                break

            # rescale g so that it has the same ℓ₁‑norm as f
            g_lp_scaled = g_lp * (sum_f_lp / sum_g_lp)

            # --- 3.2) lift back to the original dimension --------------------
            g = _expand_from_lp(g_lp_scaled)

            # Clip and re‑normalise (tiny numerical drift may appear)
            g = np.clip(g, 0.0, None)
            g /= g.sum()

            # --- 3.3) line‑search for a suitable mixing factor ---------------
            improved = False
            # try a decreasing list of step‑sizes; the larger steps are tried first
            for t in (0.4, 0.25, 0.15, 0.08, 0.04):
                f_candidate = (1.0 - t) * f + t * g
                f_candidate = np.clip(f_candidate, 0.0, None)
                s = f_candidate.sum()
                if s == 0.0:
                    continue
                f_candidate /= s
                M_candidate = _max_self_convolution(f_candidate)

                if M_candidate < M * (1 - 1e-9):  # strict improvement
                    f = f_candidate
                    M = M_candidate
                    cur_score = compute_score(f)
                    improved = True
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = f.copy()
                        print(f"[{time.monotonic() - start_time:.2f}s] "
                              f"seed {seed_idx} iter {iter_no} → score {best_score:.6f}")
                    break  # accept the first improving step

            if not improved:
                # No improvement with deterministic steps – inject a tiny jitter.
                f = _random_block_perturb(f, rng)
                M = _max_self_convolution(f)
                cur_score = compute_score(f)
                # Update best if jitter happened to be lucky
                if cur_score < best_score:
                    best_score = cur_score
                    best_seq = f.copy()
                    print(f"[{time.monotonic() - start_time:.2f}s] "
                          f"seed {seed_idx} jitter → score {best_score:.6f}")

                if time.monotonic() - stagnation_start > 12.0:
                    break
            else:
                stagnation_start = time.monotonic()   # reset stagnation clock

            # safety guard – avoid runaway loops
            if iter_no >= 5000:
                break

    # ------------------------------------------------------------------
    # 4) Fallback in case nothing was discovered (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 5) Final sanitisation (clip, normalise) and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / best_seq.size
    else:
        best_seq = best_seq / total

    elapsed = time.monotonic() - start_time
    print(f"Finished search in {elapsed:.2f}s – final best score {best_score:.6f}")

    return [float(v) for v in best_seq]
