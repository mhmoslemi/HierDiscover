"""
Hybrid search for a low‑peak self‑convolution step function.

Algorithm
---------
1. Initialise a probability vector `a` from the global ``height_sequence_1`` if
   present, otherwise use a uniform distribution.  The vector is projected
   onto the simplex (non‑negative, unit sum) – the objective is scale‑invariant.

2. **Phase 1 – deterministic sub‑gradient refinement**  
   Repeatedly move a small step in the direction of the active (maximal)
   convolution entry and re‑project onto the simplex.  The step size is
   adapted based on success/failure and occasional jitter restarts are used
   after stagnation.

3. **Phase 2 – simulated annealing**  
   Starting from the best vector found in Phase 1, propose new candidates by
   adding isotropic Gaussian noise (scale slowly annealed to zero) and
   projecting back onto the simplex.  A Metropolis acceptance rule with a
   linearly decaying temperature allows occasional uphill moves, helping the
   search escape local minima.

4. The best sequence observed over the whole time budget is returned
   (clipped to the limits of ``evaluate_sequence``).

The implementation uses FFT‑based convolution for an exact objective
evaluation (identical to ``evaluate_sequence``) and respects the 1000 s
time budget with a small safety margin.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Exact linear self‑convolution a * a using a real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """
    Compute the exact value returned by ``evaluate_sequence`` for a.
    The vector ``a`` does *not* need to be normalised – the expression is
    homogeneous, but we keep the sum explicit for numerical stability.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    max_b = conv.max()
    sum_a = a.sum()
    # Guard against near‑zero sum (should never happen after projection)
    if sum_a < 1e-12:
        return np.inf
    return float(2.0 * n * max_b / (sum_a ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the O(n) algorithm from
    Duchi et al. (2008).
    """
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """
    Perform one projected sub‑gradient step on the active (max) convolution
    entry.  Returns the new probability vector.
    """
    n = a.shape[0]
    conv = _conv_self(a)
    kmax = int(np.argmax(conv))

    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)

    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]

    cand = _project_simplex(a - step * grad)
    return cand


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The optimiser first applies a sub‑gradient
    descent phase and then refines the result with simulated annealing.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Prepare the time budget (leave a 5 s safety margin for the harness)
    # --------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start + total_budget

    # --------------------------------------------------------------
    # Initialise a probability vector.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        a = base / base.sum()
        n = a.size
    else:
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # Keep the best solution seen so far
    best_a = a.copy()
    best_score = _objective_score(best_a)
    print(f"[{0.0:6.2f}s] initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – Projected sub‑gradient descent on the exact max‑convolution
    # --------------------------------------------------------------
    cur_a = best_a.copy()
    cur_score = best_score
    step = 0.2                     # initial step length
    last_improve = start
    phase1_end = start + total_budget * 0.15   # 15 % of time for phase 1

    while time.monotonic() < phase1_end:
        cand_a = _subgradient_step(cur_a, step)
        cand_score = _objective_score(cand_a)

        if cand_score < cur_score - 1e-12:
            cur_a, cur_score = cand_a, cand_score
            step = min(step * 1.2, 1.0)          # accelerate after a success
            if cand_score < best_score - 1e-12:
                best_a, best_score = cand_a.copy(), cand_score
                last_improve = time.monotonic()

        else:

            # hard restart after stagnation
            if step < 1e-6 or (time.monotonic() - last_improve) > 12.0:

                cur_score = _objective_score(cur_a)
                step = 0.2
                last_improve = time.monotonic()
                print(f"[{last_improve - start:6.2f}s] subgrad restart")

    # --------------------------------------------------------------
    # Phase 2 – Simulated annealing on the exact objective
    # --------------------------------------------------------------
    # Parameters for the annealing schedule
    temp_start = max(best_score * 0.12, 1e-4)   # initial temperature
    sigma_start = 0.025                        # initial Gaussian noise scale

    cur_a = best_a.copy()
    cur_score = best_score
    while time.monotonic() < deadline:
        now = time.monotonic()
        # Linear decay of temperature and noise amplitude
        remaining_frac = (deadline - now) / (deadline - start)
        temperature = max(temp_start * remaining_frac, 1e-9)
        sigma = sigma_start * remaining_frac

        cand = _project_simplex(cur_a + rng.normal(scale=sigma, size=n))
        cand_score = _objective_score(cand)

                print(f"[{now - start:6.2f}s] anneal new best {best_score:.6f}")

        if rng.random() < 0.001:

            cur_score = _objective_score(cur_a)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
