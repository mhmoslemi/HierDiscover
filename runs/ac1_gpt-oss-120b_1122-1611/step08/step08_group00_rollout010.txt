def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    import time, numpy as np
    rng = np.random.default_rng(seed)
    # Determine length n
    if 'height_sequence_1' in globals():
        n = len(globals()['height_sequence_1'])
        # bound: keep between 256 and 8192 maybe
        n = max(256, min(n, 8192))
    else:
        n = 2048   # default moderate size
    # compute fft_len = next power of two >= 2*n-1, maybe also used for other conv.
    conv_len = 2*n - 1
    fft_len = 1 << ((conv_len - 1).bit_length())   # next pow2

    def compute_J(a):
        # a is np.ndarray of shape (n,)
        # assume sum=1 (or else scale)
        A = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max())

    # initial candidates list
    candidates = []
    # add uniform
    candidates.append(np.full(n, 1.0/n))
    # exponential decays with different rates
    for r in [0.9, 0.95, 0.98]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)
    # linear ramp descending / ascending
    for direction in [1, -1]:
        v = np.arange(n, dtype=np.float64)
        if direction == -1:
            v = v[::-1]
        v = v.astype(np.float64)
        v += 1.0
        v /= v.sum()
        candidates.append(v)
    # random Dirichlet draws
    for _ in range(10):
        v = rng.dirichlet(np.ones(n))
        candidates.append(v)

    # include previous best if available
    if 'height_sequence_1' in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        if v.sum() < 1e-9:
            v[:] = 1.0/n
        else:
            v /= v.sum()
        candidates.append(v)

    # evaluate each candidate
    best_vec = None
    best_J = np.inf
    for cand in candidates:
        J = compute_J(cand)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()

    print(f"Initial best J = {best_J:.6f}")

    # Simulated annealing/hill climbing
    # Parameters
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # safety
    # schedule
    # Temperature start based on diff range
    T_start = 0.01 * best_J   # roughly 1% of J
    T_end = 1e-6
    # We'll decay linearly with time fraction
    # steps size for moves: relative to current vector magnitude
    # track iteration count
    it = 0
    cur_vec = best_vec.copy()
    cur_J = best_J

    # pre-define move functions
    while time.monotonic() < deadline:
        # compute fraction
        elapsed = time.monotonic() - start_time
        frac = elapsed / (deadline - start_time)
        # temperature schedule
        T = T_start * (1.0 - frac) + T_end * frac

        # propose move
        # choose move type: small gaussian perturb or transfer between random pair
        move_type = rng.choice(['pair_transfer', 'gaussian'], p=[0.7, 0.3])
        if move_type == 'pair_transfer':
            # choose donor and receiver indices distinct
            i = rng.integers(n)
            j = rng.integers(n-1)
            if j >= i:
                j += 1
            # amount to transfer
            # define epsilon as small proportion of donor's current mass
            donor_val = cur_vec[i]
            if donor_val <= 0:
                continue  # skip invalid move
            epsilon = donor_val * rng.uniform(0.0, 0.1)  # up to 10% of donor
            new_vec = cur_vec.copy()
            new_vec[i] -= epsilon
            new_vec[j] += epsilon
        else:
            # gaussian perturb all entries then reproject
            # apply log-space multiplicative perturb to all entries
            perturb = rng.normal(0.0, 0.02, size=n)  # small log-noise
            new_vec = cur_vec * np.exp(perturb)
            # ensure non-negative (always true)
            # project onto simplex
            s = new_vec.sum()
            if s == 0.0:
                new_vec[:] = 1.0/n
            else:
                new_vec /= s

        # evaluate J
        new_J = compute_J(new_vec)

        # acceptance
        if new_J < cur_J or rng.random() < np.exp(-(new_J - cur_J) / max(T, 1e-12)):
            cur_vec = new_vec
            cur_J = new_J
            # update best
            if new_J < best_J:
                best_J = new_J
                best_vec = new_vec.copy()
                print(f"[{elapsed:.2f}s] New best J = {best_J:.6f}")

        it += 1

    # final projection to simplex safety
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    best_vec /= best_vec.sum()  # ensure sum=1? not necessary but fine
    return [float(x) for x in best_vec]

"""
Simulated‑annealing optimiser for the “step‑function” problem.

The objective is   J(a) = 2·n·max(a*a)  / (∑a)² .
Because the problem is scale‑invariant we normalise ∑a = 1 and minimise

        J(a) = 2·n·max(conv(a,a))

by a plain stochastic search.  The algorithm

* builds a few diverse initial probability vectors (uniform, exponential,
  linear ramps, Dirichlet draws and, if present, the previously found
  `height_sequence_1`);
* keeps the best one and then runs a simulated‑annealing loop.
  The loop repeatedly proposes two kinds of moves:
      – transfer a random fraction of mass from one index to another,
      – a multiplicative log‑normal perturbation of the whole vector.
  Every proposal is projected back onto the probability simplex.
* Acceptance uses the Metropolis rule with a temperature that linearly
  cools from 5 % of the current best objective down to 1e‑8.
* If no improvement is seen for 30 s the current state is re‑initialised
  by adding a tiny Gaussian noise to the best vector (escape from plateaus).

All heavy work is done with an FFT‑based convolution (O(n log n)) and the
search runs for the whole time budget (default 1000 s, 5 s safety margin).
The routine returns the best sequence encountered as a plain list of
floats, ready for `evaluate_sequence`.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w ≥ 0, Σw = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Generate a small, diverse portfolio of probability vectors."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # geometric decays
    for r in [0.9, 0.95, 0.975, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear ramps (decreasing / increasing)
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)
    v = np.arange(1, n + 1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # a couple of Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # two‑spike (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    return cand


def _compute_J(a: np.ndarray, n: int, conv_len: int, fft_len: int) -> float:
    """
    Compute the true objective J = 2·n·max(conv(a,a)).
    The vector a is assumed to sum to 1 (probability simplex).
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises ``evaluate_sequence``.  The method is a simulated‑annealing
    optimiser that works directly on the true objective (no surrogate).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0. Problem size – respect the size of a previously found solution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            # keep the size in a reasonable range for FFT speed
            n = max(256, min(n, 8192))
        except Exception:
            n = 2048
    else:
        n = 2048                     # default length when nothing is given

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 1. Initialise a pool of candidate vectors (all on the simplex).
    # ------------------------------------------------------------------
    init_cands = _initial_candidates(n, rng)

    # optionally add the previously known construction
    if "height_sequence_1" in globals():
        v = np.asarray(height_sequence_1, dtype=np.float64)
        v = np.clip(v, 0.0, None)
        s = v.sum()
        if s == 0.0:
            v[:] = 1.0 / n
        else:
            v /= s
        init_cands.append(v)

    # evaluate the pool and keep the best
    best_vec = None
    best_J = np.inf
    for cand in init_cands:
        J = _compute_J(cand, n, conv_len, fft_len)
        if J < best_J:
            best_J = J
            best_vec = cand.copy()
    print(f"[0.00s] Initial best J = {best_J:.6f}")

    # ------------------------------------------------------------------
    # 2. Simulated‑annealing loop.
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # 5 s safety margin

    # temperature schedule (linear cooling)
    T_start = best_J * 0.05   # 5 % of the current best
    T_end = 1e-8

    # current state
    cur_vec = best_vec.copy()
    cur_J = best_J
    last_improve = start_time

    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        frac = (now - start_time) / (deadline - start_time)
        T = T_start * (1.0 - frac) + T_end * frac

        # ----- propose a move -------------------------------------------------
        move_type = rng.choice(["pair_small", "pair_large", "global_perturb"],
                               p=[0.5, 0.2, 0.3])
        if move_type.startswith("pair"):
            # pick donor and receiver (distinct)
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1

            donor_val = cur_vec[i]
            if donor_val == 0.0:
                continue
            # choose transfer fraction
            if move_type == "pair_small":
                frac_transfer = rng.uniform(0.0, 0.05)   # ≤ 5 %
            else:  # pair_large
                frac_transfer = rng.uniform(0.05, 0.5)   # up to 50 %
            epsilon = donor_val * frac_transfer
            new_vec = cur_vec.copy()
            new_vec[i] -= epsilon
            new_vec[j] += epsilon

        else:  # global_perturb – multiplicative log‑normal noise
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            new_vec = cur_vec * noise
            s = new_vec.sum()
            if s == 0.0:
                new_vec[:] = 1.0 / n
            else:
                new_vec /= s

        # projection back onto the simplex (protect against round‑off)
        new_vec = proj_simplex(new_vec)

        # ----- evaluate the proposal -----------------------------------------
        new_J = _compute_J(new_vec, n, conv_len, fft_len)

        # ----- Metropolis acceptance -----------------------------------------
        accept = False
        if new_J < cur_J:
            accept = True
        else:
            # probability of uphill move
            prob = np.exp(-(new_J - cur_J) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_vec = new_vec
            cur_J = new_J
            if new_J < best_J - 1e-12:
                best_J = new_J
                best_vec = new_vec.copy()
                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] New best J = {best_J:.6f}")

        # ----- stagnation handling -------------------------------------------
        if now - last_improve > 30.0:
            # add a small Gaussian jitter to the best solution and continue
            cur_vec = best_vec * np.exp(rng.normal(0.0, 0.04, size=n))
            cur_vec = proj_simplex(cur_vec)
            cur_J = _compute_J(cur_vec, n, conv_len, fft_len)
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_J:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 3. Return the best vector (clip to the limits imposed by evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    # tiny renormalisation – does not change the objective appreciably
    s = best_vec.sum()
    if s == 0.0:
        best_vec[:] = 1.0 / n
    else:
        best_vec /= s
    return [float(x) for x in best_vec]
