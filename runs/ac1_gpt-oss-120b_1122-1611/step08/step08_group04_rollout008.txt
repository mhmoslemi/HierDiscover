The evaluate_sequence does:

- Inputs: sequence list[float], checked for being list, non-empty, each element int or float (including bool but bool is rejected). If any NaN or inf, returns np.inf. Convert to float.

- It protects against negative numbers: replaces each element x by max(0, x). So we can feed negative numbers but they become zero.

- Protects against numbers that are too large: min(1000, x). So any values larger than 1000 become 1000.

- Then compute n = len(sequence). Compute convolution via np.convolve(sequence, sequence). This returns length 2n-1 array (full convolution). Then max_b = max(convolution) (max of all entries). sum_a = sum(sequence).

- If sum_a < 0.01, return np.inf.

- Return 2 * n * max_b / (sum_a^2). So the objective: 2*n*max(conv) / (sum(a)^2). This is the ratio.

def build_active_conv_matrix(f, active_offsets):
    n = len(f)
    rows = len(active_offsets)
    data = []
    row_ind = []
    col_ind = []
    for idx, k in enumerate(active_offsets):
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        # j range inclusive
        j_vals = np.arange(start_j, end_j + 1)
        i_vals = k - j_vals
        coeffs = f[i_vals]  # f_i for each j

        data.extend(coeffs.tolist())
        row_ind.extend([idx] * len(j_vals))
        col_ind.extend(j_vals.tolist())
    A_ub = scipy.sparse.csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))
    return A_ub

"""
Hybrid active‑set linear‑programming + sub‑gradient optimizer.

* Start from the best known sequence (``height_sequence_1``) or a uniform
  initialization.
* In Phase 1 we repeatedly solve a small linear program:
  maximize the ℓ₁‑norm of a candidate ``b`` under the active
  self‑convolution constraints ``f * b ≤ M`` where ``M`` is the current
  maximum of ``f*f``.  Only the offsets whose convolution value is within
  a tiny tolerance of ``M`` are kept – this yields a sparse constraint
  matrix and a cheap LP.
* The LP solution is scaled to the target ℓ₁‑norm and mixed with the
  current iterate.  Mixing factors are searched greedily.
* Phase 2 refines the obtained shape with a projected sub‑gradient step
  on the exact max‑convolution (the same update used in the reference
  code).  The step size is adapted and occasional jitter is added to
  escape plateaus.
* The whole routine respects the 1000 s budget and returns the best
  sequence found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full (linear) self‑convolution a*a using a real FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (no clipping)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _project_onto_sum(v: np.ndarray, total: float) -> np.ndarray:
    """Project onto the simplex of non‑negative vectors with given sum."""
    return _project_simplex(v) * total


def _build_active_conv_matrix(f: np.ndarray, offsets: np.ndarray) -> csr_matrix:
    """
    Build a (len(offsets) × n) sparse matrix A such that
    (A @ b)[k] = (f * b)[offsets[k]].
    """
    n = f.shape[0]
    rows = offsets.shape[0]
    data = []
    row_ind = []
    col_ind = []
    for row, k in enumerate(offsets):
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        j = np.arange(start_j, end_j + 1, dtype=int)
        i = k - j
        coeff = f[i]                # f_i for each j
        data.extend(coeff.tolist())
        row_ind.extend([row] * len(j))
        col_ind.extend(j.tolist())
    return csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,

    Search for a non‑negative step‑function (list of heights) that minimises

    linear‑programming updates, then refines the result with a projected
    sub‑gradient descent on the exact max‑convolution.
    """

    # ------------------------------------------------------------------

    col_ind = []
    for row, k in enumerate(offsets):
        start_j = max(0, k - (n - 1))
        end_j = min(n - 1, k)
        j = np.arange(start_j, end_j + 1, dtype=int)
        i = k - j
        coeff = f[i]                # f_i for each j
        data.extend(coeff.tolist())
        row_ind.extend([row] * len(j))
        col_ind.extend(j.tolist())
    return csr_matrix((data, (row_ind, col_ind)), shape=(rows, n))


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine first performs a few active‑set
    linear‑programming updates, then refines the result with a projected
    sub‑gradient descent on the exact max‑convolution.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start + total_budget

    # ------------------------------------------------------------------
    # Choose a length n and an initial distribution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        # cap the length to keep the LP affordable
        max_len = 3000
        if base.size > max_len:
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, max_len, endpoint=False)
            a = np.interp(new_x, old_x, base)
        else:
            a = base.copy()
    else:
        max_len = 2048
        a = np.full(max_len, 1.0 / max_len, dtype=np.float64)

    n = a.shape[0]

    # Normalise to the classical target ℓ₁‑norm √(2n)
    target_sum = np.sqrt(2.0 * n)
    a = a * (target_sum / np.sum(a))

    best_f = a.copy()
    best_score = _objective_score(a)
    print(f"initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – active‑set linear programming
    # ------------------------------------------------------------------
    # Use roughly 35 % of the budget for this phase.
    phase1_end = start + total_budget * 0.35
    max_active = 1200          # upper bound on the number of active constraints
    lp_iter = 0
    while time.monotonic() < phase1_end and lp_iter < 30:
        lp_iter += 1

        conv_ff = _conv_self(a)
        M = conv_ff.max()

        # --------------------------------------------------------------
        # Determine the active set of offsets (those close to the maximum).
        # --------------------------------------------------------------
        rel_tol = 0.99                         # keep offsets ≥ 99 % of the peak
        active_idx = np.where(conv_ff >= rel_tol * M)[0]
        if active_idx.size > max_active:
            idx = np.argpartition(conv_ff, -max_active)[-max_active:]
            active_idx = np.sort(idx)

        # --------------------------------------------------------------
        # Build the sparse constraint matrix A such that (A @ b) ≤ M.
        # --------------------------------------------------------------
        A_ub = _build_active_conv_matrix(a, active_idx)

        # --------------------------------------------------------------
        # Linear program:
        #   max   Σ b_i        ⇔   min   -Σ b_i
        #   s.t.  A_ub @ b ≤ M,   b_i ≥ 0
        # --------------------------------------------------------------
        c = -np.ones(n)                     # minimise –sum(b)
        b_ub = M * np.ones(active_idx.shape[0])
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c,
                          A_ub=A_ub,
                          b_ub=b_ub,
                          bounds=bounds,
                          method='highs',
                          options={'presolve': True})
        except Exception as exc:
            print(f"LP exception: {exc}")
            break

        if not res.success:
            print(f"LP iteration {lp_iter} failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            print("LP produced zero mass – aborting LP phase")
            break

        # --------------------------------------------------------------
        # Scale b to have the correct ℓ₁‑norm and create a candidate mixture.
        # --------------------------------------------------------------
        g = b * (target_sum / sum_b)

        # Simple line‑search over a few mixing coefficients.
        improved = False
        best_local_score = best_score
        best_candidate = None
        for mix in (0.15, 0.30, 0.45, 0.60):
            cand = (1.0 - mix) * a + mix * g
            cand_score = _objective_score(cand)
            if cand_score < best_local_score - 1e-12:
                best_local_score = cand_score
                best_candidate = cand.copy()
                improved = True

        if improved:
            a = best_candidate
            best_f = a.copy()
            best_score = best_local_score
            print(f"[{time.monotonic() - start:6.2f}s] LP improve {best_score:.6f}")
        else:
            # No improvement – inject a tiny jitter to shake the iterate.
            jitter = rng.normal(0.0, 0.001, size=n)
            a = a + jitter
            a = np.maximum(a, 0.0)
            a = a * (target_sum / np.sum(a))

    # ------------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    step = 0.1
    while time.monotonic() < deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # gradient of the active offset (conv[kmax])
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # tentative update and projection onto the simplex of sum = target_sum
        a_candidate = _project_onto_sum(a - step * grad, target_sum)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_f = a_candidate.copy()
            a = a_candidate
            step *= 1.05                # increase step after success
            print(f"[{time.monotonic() - start:6.2f}s] subgrad improve {best_score:.6f}")
        else:
            step *= 0.70                # decrease step after failure
            # occasional random jitter to escape flat regions
            if step < 1e-7 or rng.random() < 0.001:
                jitter = rng.normal(0.0, 0.002, size=n)
                a = a + jitter
                a = np.maximum(a, 0.0)
                a = a * (target_sum / np.sum(a))
                step = max(step, 0.01)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)
    return [float(v) for v in best_f]
