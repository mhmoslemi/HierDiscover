def _gradient_wrt_a_onehot(a, k):
    n = a.shape[0]
    g = np.zeros_like(a)
    shift = k - (n - 1)  # integer
    if shift >= 0:
        # g[shift:] = 2*a_rev[:n-shift]
        g[shift:] = 2*a[::-1][:n-shift]
    else:
        # shift < 0 -> g[:n+shift] = 2*a_rev[-shift:]
        g[:n+shift] = 2*a[::-1][-shift:]
    return g

"""
Improved search for a step‑function that minimises  

    2·n·max(conv(a,a)) / (sum(a)**2)

The optimiser works in three stages:

1. **Variance flattening** – minimise ∑(c‑µ)² where c = a*a.
   This drives the self‑convolution towards a constant shape.

2. **Smooth‑max refinement** – optimise a soft‑max (log‑sum‑exp) surrogate
   of the true max.  The temperature (γ) is annealed from 0.5 up to 200,
   which makes the surrogate increasingly tight.

3. **Exact max sub‑gradient** – once the convolution is almost flat we
   switch to the true max gradient (a single‑hot weight).  This gives a
   precise final push.

The variables are kept non‑negative and normalised by using a soft‑max
parameterisation `a = softmax(x)`.  Adam is used for the updates,
with a linear learning‑rate decay and occasional restarts/jitters when
the search stagnates.  The routine respects the global time budget
(`budget_s`, default 1000 s) and returns the best sequence seen.

The code is deliberately self‑contained (no closures, no file I/O) and
can be called repeatedly – the evaluation of the true objective
(`evaluate_sequence`) is performed only for logging, the optimiser works
with the fast FFT‑based surrogate.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level only (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    # keep only the linear part (length 2·n‑1)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Returns a vector of length n.
    """
    n = a.size
    # reverse a for the correlation step
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    # the relevant slice corresponds to indices [n‑1 … 2n‑2]
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _gradient_wrt_a_onehot(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the single convolution entry c_k = (a*a)_k.
    This is a cheap O(n) version of ``_gradient_wrt_a`` with a one‑hot
    weight vector.
    """
    n = a.size
    g = np.zeros_like(a)
    # shift = k - (n-1)  (how far the kernel is displaced from centre)
    shift = k - (n - 1)
    a_rev = a[::-1]

    if shift >= 0:
        # overlap region: a_rev[0 : n-shift] maps to g[shift : ]
        g[shift:] = 2.0 * a_rev[: n - shift]
    else:
        # shift < 0 : a_rev[-shift : ] maps to g[ : n+shift]
        g[: n + shift] = 2.0 * a_rev[-shift:]

    return g


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The function respects the time budget and returns
    the best sequence found when the time expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (keep a safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a length and initialise the log‑weights
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)          # start from log‑weights
        else:
            # fallback – pick a moderately large size
            n = 8000
            x = np.zeros(n, dtype=np.float64)
    else:
        # No previous construction – start from scratch
        n = 8000
        x = np.zeros(n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (used repeatedly)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)   # for a*a
    # for the correlation step (a_rev * w) where w has length 2n‑1
    fft_len_corr = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state (shared across all phases)
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Helper: convert logits → probability vector (sum = 1)
    # ------------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # ------------------------------------------------------------------
    # 5) Tracking the best solution seen so far
    # ------------------------------------------------------------------
    a = _a_from_x(x)
    c = _self_conv(a, fft_len_self)          # length 2·n‑1
    best = a.copy()
    best_score = 2.0 * n * np.max(c) / (a.sum() ** 2)
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 6) Phase schedule
    # ------------------------------------------------------------------
    # 0.0 – 0.45   → variance flattening
    # 0.45 – 0.80  → smooth‑max annealing
    # 0.80 – 1.00  → exact max sub‑gradient
    phase1_end = start_time + 0.45 * total_time
    phase2_end = start_time + 0.80 * total_time
    phase1 = True
    phase2 = False
    phase3 = False

    # ------------------------------------------------------------------
    # 7) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # phase transitions
        # --------------------------------------------------------------
        if phase1 and now >= phase1_end:
            phase1 = False
            phase2 = True
            # reset Adam for the new landscape
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] → switch to smooth‑max phase")
        if phase2 and now >= phase2_end:
            phase2 = False
            phase3 = True
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] → switch to exact‑max phase")

        # --------------------------------------------------------------
        # current a and its self‑convolution
        # --------------------------------------------------------------
        a = _a_from_x(x)                      # probability vector, sum = 1
        c = _self_conv(a, fft_len_self)       # length 2·n‑1
        cur_score = 2.0 * n * np.max(c) / (a.sum() ** 2)

        # --------------------------------------------------------------
        # record improvement
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # --------------------------------------------------------------
        # compute gradient according to the active phase
        # --------------------------------------------------------------
        if phase1:
            # ---- Phase 1 : variance (L2) flattening ----
            avg = 1.0 / (2 * n - 1)                 # target average of conv
            r = c - avg                             # residual vector, len 2n‑1
            g_a = 2.0 * _gradient_wrt_a(a, r)       # shape (n,)

        elif phase2:
            # ---- Phase 2 : smooth‑max (log‑sum‑exp) ----
            phase_elapsed = now - phase1_end
            phase_total = phase2_end - phase1_end
            frac = phase_elapsed / max(phase_total, 1e-12)

            # anneal γ from 0.5 up to 200
            gamma = 0.5 + (200.0 - 0.5) * frac

            # numerically stable weights
            c_shift = c - np.max(c)
            w = np.exp(gamma * c_shift)
            w /= w.sum()                            # now a probability vector
            g_a = _gradient_wrt_a(a, w)             # shape (n,)

        else:
            # ---- Phase 3 : exact max sub‑gradient ----
            k_max = int(np.argmax(c))               # index of the max entry
            g_a = _gradient_wrt_a_onehot(a, k_max)  # cheap O(n) gradient

        # --------------------------------------------------------------
        # softmax Jacobian (chain rule for x → a)
        # --------------------------------------------------------------
        avg_g = np.dot(a, g_a)          # scalar ⟨a,∂F/∂a⟩
        g_x = a * (g_a - avg_g)        # ∂F/∂x

        # --------------------------------------------------------------
        # Adam update
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # linear learning‑rate decay over the whole budget
        lr_start = 0.5
        lr_end = 0.02
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # --------------------------------------------------------------
        # stagnation handling – restart if no improvement for 30 s
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # re‑initialise around the best point with Gaussian noise
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # --------------------------------------------------------------
        # occasional global jitter to jump out of shallow minima
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 8) Return the best sequence (clip to limits enforced by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
