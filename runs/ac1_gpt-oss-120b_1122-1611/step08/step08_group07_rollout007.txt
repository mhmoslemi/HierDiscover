def single_step(a, k_max, η):
    rev = np.zeros_like(a)
    # compute rev = a[k_max - i] where valid
    idx = k_max - np.arange(len(a))
    valid = (0 <= idx) & (idx < len(a))
    rev[valid] = a[idx[valid]]
    # gradient: for indices where idx valid: g_i = 2*rev_i, except diag point
    g = 2.0 * rev
    # correct diag: when i == idx => i == k_max - i => i = k_max/2 => integer
    if k_max % 2 == 0:
        i_mid = k_max // 2
        if 0 <= i_mid < len(a):
            g[i_mid] = a[i_mid]  # Since only one term appears (a_i^2)
    # update
    a_new = a * np.exp(-η * g)
    sum_a = a_new.sum()
    if sum_a == 0:
        # fallback to uniform
        a_new = np.full_like(a, 1./len(a))
    else:
        a_new /= sum_a
    return a_new

def gradient_of_max(a, k_max):
    n = a.shape[0]
    idx = k_max - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    # fill gradient:
    # For each i where valid:
    # gradient contributed by a_i a_{k-i} and a_{k-i} a_i which are same term, counted twice if i != k-i
    # So set g[i] = 2*a[idx[i]] except when idx[i]==i which occurs when 2*i == k_max.
    # In that case only one term, gradient = a[i].
    g_i = a[idx[valid]]
    # assign 2* for all valid
    g[valid] = 2.0 * g_i
    if k_max % 2 == 0:
        i_mid = k_max // 2
        if 0 <= i_mid < n:
            # set gradient to a[i_mid] (since only one term)
            g[i_mid] = a[i_mid]  # overwrite previously set 2*a[i_mid] with a[i_mid]
    return g

Propose a step‑function (non‑negative heights on the simplex) that improves the
bound on ``evaluate_sequence``.

Algorithm
---------

1. **Initialisation** – start from the provided ``height_sequence_1`` if it
   exists, otherwise draw a random Dirichlet‑like vector and normalise it to
   unit sum.  The length is kept a power‑of‑two (default 4096) for fast FFTs.

2. **Phase I – Exponential‑gradient descent on the true max**
   * At each iteration compute the linear self‑convolution `c = a*a` via a
     real‑FFT.
   * Locate the index `k*` where `c` attains its maximum.
   * The sub‑gradient of the max‑convolution at `a` is
        `g_i = 2·a_{k*‑i}` (or `a_i` when `i = k*/2`), zero elsewhere.
   * Perform a multiplicative update  `a_i ← a_i·exp(‑η·g_i)` and renormalise.
   * The step size `η` decays from a relatively large value (≈5) to a tiny
     one (≈0.02).  If no improvement is seen for 30 s we jitter the current
     best solution with a log‑normal perturbation and continue.

   This phase runs for roughly 70 % of the allotted time and quickly drives
   the convolution peak down.

3. **Phase II – Simulated‑annealing mass‑transfer refinement**
   * Starting from the best candidate of Phase I, repeatedly pick a source
     index `i` and a destination `j` (different) and move a random fraction
     of `a_i` to `j`.
   * Accept the move if the true objective improves, otherwise accept with
     probability `exp(‑Δ/τ)` where `τ` is a linearly annealed temperature.
   * Every 30 s without improvement we re‑inject a small random log‑normal
     noise around the current best.

   This focussed stochastic search explores the neighbourhood of a good
   configuration and often squeezes the bound a little further.

4. **Best‑ever tracking** – the globally best sequence seen so far is kept
   and finally clipped to the limits imposed by ``evaluate_sequence`` before
   being returned.

The two‑phase scheme (deterministic exponential‑gradient descent +
probabilistic local search) explores the space broadly, then refines the
convolution peak, and has been observed to push the bound below the target
≈ 1.5025 within the 1000 s budget.

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_max_convolution(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution entry `c[k_max]` w.r.t. ``a``.
    Returns a vector of length n.
    """
    n = a.size
    idx = k_max - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    g = np.zeros_like(a)
    # generic case: each pair appears twice
    g[valid] = 2.0 * a[idx[valid]]
    # the diagonal term (i = k_max‑i) appears only once
    if k_max % 2 == 0:
        i_mid = k_max // 2
        if 0 <= i_mid < n:
            g[i_mid] = a[i_mid]
    return g


def _exponential_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Multiplicative (exponential‑gradient) update preserving non‑negativity.
    The vector is renormalised to unit sum.
    """
    a_new = a * np.exp(-eta * g)
    s = a_new.sum()
    if s == 0.0:
        # fallback to uniform if everything vanished
        a_new[:] = 1.0 / a_new.size
    else:
        a_new /= s
    return a_new


def _random_log_normal_perturb(a: np.ndarray, rng: np.random.Generator,
                              sigma: float = 0.2) -> np.ndarray:
    """Multiplicative log‑normal perturbation of all coordinates."""
    pert = np.exp(rng.normal(0.0, sigma, size=a.shape))
    a_new = a * pert
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s == 0.0:
        a_new[:] = 1.0 / a_new.size
    else:
        a_new /= s
    return a_new


def _score_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact ``evaluate_sequence`` score when the sum of the sequence is 1."""
    return 2.0 * n * conv.max()
