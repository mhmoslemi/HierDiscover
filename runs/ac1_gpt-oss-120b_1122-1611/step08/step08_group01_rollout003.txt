* Length n is a power‑of‑two (default 4096) – good for FFT speed.
* Initialise with an *arcsine* shape  
  a[i] ∝ 1/√((i+½)(n‑½‑i)), which is known to give a rather flat
  self‑convolution.
* Keep the exact self‑convolution `c = a*a` (linear convolution) in an
  array of length 2·n‑1.
* Repeatedly propose a transfer of a small amount `δ` of mass from a
  source index `i` to a destination `j`.  The effect on the whole
  convolution can be updated in O(n) using the closed‑form
  `c′ = c + 2·δ·(shift(j,a)‑shift(i,a)) + δ²·e_{2j}+δ²·e_{2i}‑2·δ²·e_{i+j}`.
* Accept the move if it improves the objective  
  `score = 2·n·max(c)`.  Otherwise accept with probability
  `exp((old‑score−new‑score)/T)` where the temperature `T` is slowly
  cooled.
* Occasionally perform a block‑wise log‑normal perturbation and a
  occasional restart around the best point seen so far.
* The best sequence (clipped to the limits used by ``evaluate_sequence``)
  is returned after the time budget expires.

The routine respects the 1000 s budget (a few seconds are kept for the
harness) and works with any non‑negative starting vector supplied in the
global ``height_sequence_1`` variable.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of a (length n) using an FFT of length fft_len.
    Result has length 2·n‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _update_convolution(
    c: np.ndarray,
    a: np.ndarray,
    i: int,
    j: int,
    delta: float,
    work: np.ndarray,
) -> None:
    """
    Apply the exact update to the convolution array ``c`` when the
    probability mass ``delta`` is moved from index ``i`` to index ``j``.
    ``work`` is a pre‑allocated temporary buffer (same shape as ``c``)
    that is cleared and reused to avoid allocations.
    The update follows
        c ← c + 2·δ·(shift(j,a) – shift(i,a))
            + δ²·e_{2j} + δ²·e_{2i} – 2·δ²·e_{i+j}
    where e_k is the unit vector with a 1 at position k.
    """
    work.fill(0.0)

    n = a.size
    # shift(j,a) contributes on slice [j, j+n)
    work[j : j + n] += 2.0 * delta * a
    # shift(i,a) contributes negatively
    work[i : i + n] -= 2.0 * delta * a

    delta2 = delta * delta
    # quadratic terms: affect at most three positions
    work[2 * j] += delta2
    work[2 * i] += delta2
    work[i + j] -= 2.0 * delta2

    # apply the update in place
    c += work


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor and renormalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time handling (keep a small safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a length – power of two for cheap FFTs.
    #    4096 is a good compromise between quality and speed.
    # ------------------------------------------------------------------
    n = 4096

    # ------------------------------------------------------------------
    # 2) Build an initial arcsine‑shaped probability vector.
    # ------------------------------------------------------------------
    i = np.arange(n, dtype=np.float64)
    denom = np.sqrt((i + 0.5) * (n - 0.5 - i))
    a = 1.0 / denom
    a /= a.sum()                       # now sum(a) == 1

    # ------------------------------------------------------------------
    # 3) If a previous construction exists, keep the better of the two.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        alt = np.asarray(init_seq, dtype=np.float64)
        if alt.ndim == 1 and alt.size > 0:
            alt = np.clip(alt, 0.0, None)
            s = alt.sum()
            if s > 0.0:
                alt /= s
            else:
                alt = np.full_like(alt, 1.0 / alt.size)
            # if lengths differ, pad/trim to n
            if alt.size != n:
                if alt.size < n:
                    pad = np.zeros(n, dtype=np.float64)
                    pad[: alt.size] = alt
                    alt = pad
                else:
                    alt = alt[:n]
            # evaluate both
            # fast evaluation: 2*n*max(conv)
            fft_len = _next_pow2(2 * n - 1)
            conv_a = _self_conv(a, fft_len)
            conv_alt = _self_conv(alt, fft_len)
            score_a = 2.0 * n * conv_a.max()
            score_alt = 2.0 * n * conv_alt.max()
            if score_alt < score_a:
                a = alt
                print(f"[init] using supplied height_sequence_1 with score {score_alt:.6f}")
            else:
                print(f"[init] using arcsine start with score {score_a:.6f}")

    # ------------------------------------------------------------------
    # 4) Pre‑compute FFT length for later full recomputation (rare)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 5) Initial convolution & score
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)          # length 2n‑1
    cur_score = 2.0 * n * c.max()
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 6) Simulated‑annealing parameters
    # ------------------------------------------------------------------
    # step‑size (max fraction of source weight that may be moved)
    step_start = 0.3
    step_end = 0.002

    # temperature schedule (scale of score differences we are willing to accept)
    T_start = 0.005 * cur_score      # about 0.5 % of the current score
    T_end = 1e-10

    # pre‑allocated buffer for convolution updates
    work = np.empty_like(c)

    # ------------------------------------------------------------------
    # 7) Main optimisation loop
    # ------------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = 1.0 - (elapsed / total_time)          # 1 → 0

        # annealing schedules (linear for simplicity)
        step_frac = step_end + (step_start - step_end) * frac
        T = T_end + (T_start - T_end) * frac

        # -------- propose a mass move ---------------------------------
        # pick a source with non‑zero weight
        i_src = rng.integers(0, n)
        while a[i_src] == 0.0:
            i_src = rng.integers(0, n)
        j_dst = rng.integers(0, n)
        if i_src == j_dst:
            continue

        # delta is a random fraction of the source weight
        delta = step_frac * rng.random() * a[i_src]
        if delta <= 0.0:
            continue

        # compute candidate convolution via fast incremental update
        #   (work array will hold the additive change)
        work.fill(0.0)
        # shift contribution of destination
        work[j_dst : j_dst + n] += 2.0 * delta * a
        # shift contribution of source (negative)
        work[i_src : i_src + n] -= 2.0 * delta * a
        # quadratic terms
        d2 = delta * delta
        work[2 * j_dst] += d2
        work[2 * i_src] += d2
        work[i_src + j_dst] -= 2.0 * d2

        cand_max = np.max(c + work)
        cand_score = 2.0 * n * cand_max

        # decide acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp((cur_score - cand_score) / max(T, 1e-16))
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the move permanently
            a[i_src] -= delta
            a[j_dst] += delta
            c += work
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # -------- occasional larger perturbations ----------------------
        if rng.random() < 0.004:   # ~0.4 % of iterations
            a_block = _random_block_perturb(a, rng)
            c_block = _self_conv(a_block, fft_len_self)
            block_score = 2.0 * n * c_block.max()
            if block_score < cur_score:
                a = a_block
                c = c_block
                cur_score = block_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] block‑perturb best = {best_score:.6f}")

        # -------- stagnation handling ---------------------------------
        if now - last_improve > 30.0:
            # restart locally around the best point seen so far
            noise = rng.normal(0.0, 0.35, size=n)
            a = np.exp(np.log(best + 1e-12) + noise)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            c = _self_conv(a, fft_len_self)
            cur_score = 2.0 * n * c.max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] local restart – score {cur_score:.6f}")

        iteration += 1

    # ------------------------------------------------------------------
    # 8) Return the best sequence (clipped to limits imposed by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
