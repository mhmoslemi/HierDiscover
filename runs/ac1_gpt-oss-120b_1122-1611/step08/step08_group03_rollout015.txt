search for step function minimizing evaluate_sequence.
Implements multi‑restart Adam with smooth‑max followed by a hard‑max
sub‑gradient refinement and a stochastic local search.  The algorithm
uses fast FFT based evaluation of the objective (identical to
evaluate_sequence) to explore many candidates within the time budget.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    ...

def _objective_fft(a: np.ndarray) -> float:
    # assume a sum ~1 (positive)
    n = a.size
    # compute convolution via FFT length power of two
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
    max_b = conv.max()
    # sum should be 1 after projection
    return float(2.0 * n * max_b)

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 5.0)   # safety margin

    # ---- parameters -------
    # length of the step function – power of two for fast FFTs
    n = 4096
    # number of random restarts
    n_restarts = 4

    best_score = np.inf
    best = None
    # ---- helper to project and normalise----
    def normalise_and_project(a):
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a[:] = 1.0 / a.size
        else:
            a /= a.sum()
        return _project_onto_simplex(a)

    # ---- evaluate initial candidates (uniform + random Dirichlet) ----
    init_candidates = []
    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))
    # optionally use previous construction if existing
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        if a0.size != n:
            # pad or truncate to length n
            if a0.size < n:
                pad = np.zeros(n - a0.size, dtype=np.float64)
                a0 = np.concatenate([a0, pad])
            else:
                a0 = a0[:n]
        a0 = normalise_and_project(a0)
        init_candidates.append(a0)

    # random Dirichlet seeds
    for _ in range(n_restarts - len(init_candidates)):
        x = rng.random(n)
        a = normalise_and_project(x)
        init_candidates.append(a)

    # keep best of the initial set
    for a in init_candidates:
        sc = _objective_fft(a)
        if sc < best_score:
            best_score = sc
            best = a.copy()
            print(f"[init] score = {best_score:.6f}")

    # ---------- Phase 1 : smooth‑max Adam ----------
    # Parameters for the Adam phase (common for all restarts)
    # we allocate roughly 60 % of the budget to this phase
    phase1_deadline = start + 0.60 * (deadline - start)

    # annealing schedule
    TAU_START, TAU_END = 0.6, 5e-07
    LR_START, LR_END = 0.4, 5e-05

    while time.monotonic() < phase1_deadline:
        now = time.monotonic()
        frac = (now - start) / (phase1_deadline - start)
        tau = TAU_START * (1.0 - frac) + TAU_END * frac
        lr = LR_START * (1.0 - frac) + LR_END * frac

        # compute self‑convolution via FFT (once per iteration)
        L = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(best, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]

        # soft‑max over conv
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)
        w = exp_shifted / exp_shifted.sum()

        # gradient of smooth surrogate (correlation)
        w_fft = np.fft.rfft(w, n=L)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=L)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # simple Adam step (no bias‑correction to keep code short)
        # we reuse a running first/second moment that is stored globally
        if not hasattr(propose_candidate, "_adam_state"):
            propose_candidate._adam_state = {}
        state = propose_candidate._adam_state
        if "m" not in state:
            state["m"] = np.zeros_like(best)
            state["v"] = np.zeros_like(best)
            state["beta1"] = 0.9
            state["beta2"] = 0.999
            state["beta1_pow"] = 1.0
            state["beta2_pow"] = 1.0
            state["eps"] = 1e-8

        # Adam update
        state["m"] = state["beta1"] * state["m"] + (1.0 - state["beta1"]) * grad
        state["v"] = state["beta2"] * state["v"] + (1.0 - state["beta2"]) * (grad * grad)
        state["beta1_pow"] *= state["beta1"]
        state["beta2_pow"] *= state["beta2"]
        m_hat = state["m"] / (1.0 - state["beta1_pow"])
        v_hat = state["v"] / (1.0 - state["beta2_pow"])
        step = lr * m_hat / (np.sqrt(v_hat) + state["eps"])

        # projected gradient step
        a_new = best - step
        a_new = _project_onto_simplex(a_new)

        # evaluate occasionally
        if int(now * 100) % 200 == 0:   # cheap periodic check
            sc = _objective_fft(a_new)
            if sc < best_score:
                best_score = sc
                best = a_new.copy()
                print(f"[{now-start:6.2f}s] Adam improve => {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached in Adam")
                    break
        # accept if improvement (otherwise keep old)
        if _objective_fft(a_new) <= best_score:
            best = a_new

    # --------- Phase 2 : hard‑max sub‑gradient descent ----------
    phase2_deadline = start + 0.85 * (deadline - start)
    step_size = 0.1   # initial step size for sub‑gradient moves
    while time.monotonic() < phase2_deadline:
        # compute exact convolution (FFT is fine)
        L = _next_pow2(2 * n - 1)
        a_fft = np.fft.rfft(best, n=L)
        conv = np.fft.irfft(a_fft * a_fft, n=L)[:2 * n - 1]
        max_val = conv.max()
        max_idxs = np.flatnonzero(conv == max_val)

        # build sub‑gradient: average over all maximal positions
        # gradient_i = sum_{k in max_idxs} a_{k-i} (if valid)
        grad = np.zeros_like(best)
        for k in max_idxs:
            # shift = k - i => i = k - j
            low = max(0, k - (n - 1))
            high = min(k, n - 1)
            # for i in [low, high]: contribution a_{k-i}
            # we can fill via slice
            # i runs from low..high, corresponding a_index = k - i
            grad[low:high + 1] += best[k - np.arange(low, high + 1)]

        # average gradient over multiple max positions
        grad /= len(max_idxs)

        # take a step opposite to gradient
        a_candidate = best - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate)

        sc = _objective_fft(a_candidate)
        if sc < best_score:
            best_score = sc
            best = a_candidate
            print(f"[{time.monotonic()-start:6.2f}s] subgrad improve => {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached in sub‑gradient")
                break
        else:
            # reduce step size modestly if no improvement
            step_size *= 0.95
            if step_size < 1e-7:
                step_size = 1e-7

    # --------- Phase 3 : stochastic local refinement ----------
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if best[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # pick a transfer amount (log‑uniform)
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = best[i] * frac_move
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand)
        sc = _objective_fft(cand)
        if sc < best_score:
            best_score = sc
            best = cand
            print(f"[{time.monotonic()-start:6.2f}s] local improve => {best_score:.6f}")
            if best_score <= 1.5025:
                break

    # final clipping & exact evaluation
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    print(f"Finished – final reported score = {final_score:.6f}")

"""
search for a non‑negative step function that minimises evaluate_sequence.

The algorithm works in three stages inside the allowed time budget:

* **Initialisation** – start from the provided `height_sequence_1` (if any)
  or from a uniform distribution.  The vector is projected onto the
  probability simplex (sum = 1) – the objective is scale‑invariant, so we
  can keep the sum fixed to one.

* **Smooth‑max Adam** – replace the hard max in the objective by a
  temperature‑controlled log‑sum‑exp, compute the exact gradient via the
  FFT and perform an Adam update while staying on the simplex.
  The temperature is annealed from 0.6 down to 5e‑7 and the learning
  rate from 0.4 down to 5e‑5.  The phase consumes ~60 % of the total
  budget.

* **Hard‑max sub‑gradient + stochastic refinement** – after the
  annealing phase we switch to the true max.  The sub‑gradient of
  `max(conv)` is a correlation of the current sequence with the
  indicator of the maximal shift(s); a projected gradient step is taken
  with a decreasing step size.  Finally a cheap stochastic mass‑transfer
  local search is performed until time runs out.

All heavy evaluation is performed with a fast FFT‑based surrogate that
produces exactly the same value returned by `evaluate_sequence` (the
function only clips values to [0,1000] and checks the sum, which we keep
strictly positive).  The best candidate found is returned; its true
score is printed using the original `evaluate_sequence` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


def _objective_fft(a: np.ndarray) -> float:
    """
    Fast evaluation of `2 * n * max(conv(a, a))` assuming a.sum()≈1.
    Uses an FFT of length a power of two.
    """
    n = a.shape[0]
    L = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(a_fft * a_fft, n=L)[: 2 * n - 1]
    max_b = conv.max()
    return float(2.0 * n * max_b)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves on
    the current best bound (≈1.5025).  The routine respects the time
    budget, prints progress, and finally returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Parameters and time‑budget handling
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)  # 5 s safety margin

    n = 4096                     # power‑of‑two length → fast FFT handling
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1) Initialise a feasible probability vector
    # --------------------------------------------------------------
    def _normalise(v: np.ndarray) -> np.ndarray:
        v = np.clip(v, 0.0, None)
        if v.sum() == 0.0:
            v[:] = 1.0 / v.size
        else:
            v /= v.sum()
        return _project_onto_simplex(v)

    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if a0.size != n:
            # pad or truncate to the required length
            if a0.size < n:
                a0 = np.concatenate([a0, np.zeros(n - a0.size, dtype=np.float64)])
            else:
                a0 = a0[:n]
        a = _normalise(a0.copy())
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    best = a.copy()
    best_score = _objective_fft(best)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 1 – smooth‑max Adam optimisation (≈60 % of the budget)
    # --------------------------------------------------------------
    phase1_deadline = start_time + 0.60 * (deadline - start_time)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0

    # Annealing schedules
    TAU_START, TAU_END = 0.60, 5e-07
    LR_START, LR_END = 0.40, 5e-05

    it = 0
    while time.monotonic() < phase1_deadline:
        now = time.monotonic()
        frac = (now - start_time) / (phase1_deadline - start_time)

        # linearly anneal temperature and learning rate
        tau = TAU_START * (1.0 - frac) + TAU_END * frac
        lr = LR_START * (1.0 - frac) + LR_END * frac

        # ----- self‑convolution via FFT -----
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ----- soft‑max weights (smooth surrogate for max) -----
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)    # shift avoids overflow
        w = exp_shifted / exp_shifted.sum()           # length 2n‑1, sums to 1

        # ----- gradient of the smooth surrogate (correlation) -----
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projected gradient step -----
        a = a - step
        a = _project_onto_simplex(a)

        it += 1
        # evaluate exactly every 250 iterations
        if it % 250 == 0:
            sc = _objective_fft(a)
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] Adam improve => {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached during Adam")
                    break

    # --------------------------------------------------------------
    # 3) Phase 2 – hard‑max sub‑gradient descent
    # --------------------------------------------------------------
    phase2_deadline = start_time + 0.85 * (deadline - start_time)
    a = best.copy()
    step_size = 0.15        # initial step size for the sub‑gradient step

    while time.monotonic() < phase2_deadline:
        # exact convolution (FFT is still cheap)
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        max_val = conv.max()
        max_pos = np.flatnonzero(conv == max_val)

        # build a sub‑gradient: average over all active positions
        grad = np.zeros_like(a)
        for k in max_pos:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            # contribution a_{k-i} for i∈[lo,hi]
            grad[lo:hi + 1] += a[k - np.arange(lo, hi + 1)]
        grad /= len(max_pos)               # average if several positions tie

        # projected gradient step (move opposite to gradient)
        cand = a - step_size * grad
        cand = _project_onto_simplex(cand)

        sc = _objective_fft(cand)
        if sc < best_score - 1e-12:
            best_score = sc
            best = cand.copy()
            a = cand
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] sub‑grad improve => {best_score:.6f}")
            if best_score <= 1.5025:
                print("[early stop] target reached during sub‑gradient")
                break
        else:
            # no improvement → shrink step size gently
            step_size *= 0.94
            # keep the current a unchanged for the next iteration
            if step_size < 1e-7:
                step_size = 1e-7

    # --------------------------------------------------------------
    # 4) Phase 3 – stochastic local refinement
    # --------------------------------------------------------------
    a = best.copy()
    it_local = 0
    while time.monotonic() < deadline:
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # fraction of source mass to transfer – log‑uniform between 0.1 % and 5 %
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
        delta = a[i] * frac_move

        cand = a.copy()
        cand[i] -= delta
        cand[j] += delta
        cand = _project_onto_simplex(cand)

        sc = _objective_fft(cand)
        it_local += 1
        if sc < best_score - 1e-12:
            best_score = sc
            best = cand.copy()
            a = cand
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] local improve => {best_score:.6f}")
            if best_score <= 1.5025:
                break

        # occasional jitter to escape tiny plateaus
        if it_local % 4000 == 0:
            width = rng.integers(1, max(2, n // 80))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            tmp = a.copy()
            tmp[start:start + width] *= factor
            a = _project_onto_simplex(tmp)

    # --------------------------------------------------------------
    # 5) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total = time.monotonic() - start_time
    print(f"Finished in {total:.2f}s – final reported score = {final_score:.6f}")
    return [float(x) for x in best]
