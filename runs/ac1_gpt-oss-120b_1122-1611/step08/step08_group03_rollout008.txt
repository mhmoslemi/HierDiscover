"""
Propose a non‑negative step‑function (list of heights) that minimises
``evaluate_sequence``.

Algorithm
---------
* **Initialisation** – start from the known construction if present,
  otherwise from a uniform distribution (sum normalised to 1).

* **Phase 1 – Smooth‑max Adam** – minimise a temperature‑controlled
  log‑sum‑exp surrogate for the convolution maximum.
  The surrogate gradient is obtained by FFT‑based cross‑correlation.
  Adam updates are projected onto the probability simplex.
  The temperature is annealed from 0.8 down to 5·10⁻⁶, the learning‑rate
  from 0.5 down to 1·10⁻⁴.  This phase lasts ~75 % of the time budget.

* **Phase 2 – Directed local refinement** – repeatedly identify the shift
  `k_max` where the convolution attains its current maximum.
  The derivative of the convolution at this shift is
  `w_i = a_{k_max-i}+a_{i-k_max}` (zero outside the array).
  Mass is moved from a source index with large `w_i·a_i` to a target
  with small `w_i`.  The move is kept only if the exact score improves.
  Random tiny moves and occasional jitter are mixed in to avoid stagnation.

* **Return** the best sequence seen (clipped to the evaluator limits).

The routine respects the 1000 s budget, prints progress and never
produces negative or NaN entries.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _grad_weights_at_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the convolution entry (a*a)[k] with respect to ``a``.
    For every i,  grad_i = a_{k-i} (if in range) + a_{i-k} (if in range).
    """
    n = a.size
    w = np.zeros(n, dtype=np.float64)

    # contribution a_{k-i}
    if k < n:
        # i = 0 .. k
        w[:k + 1] = a[k::-1]
    else:
        # k ≥ n
        start = k - (n - 1)          # first i that yields a valid index
        # i ranges from start .. n‑1
        # the corresponding a indices run backwards from n‑1 down to start
        w[start:] = a[n - 1 : start - 1 : -1]

    # contribution a_{i-k}
    if k < n:
        # i = k .. n‑1
        w[k:] += a[: n - k]
    # else k ≥ n → no contribution because i‑k < 0 for all i
    return w


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that
    minimises ``evaluate_sequence`` under the given time budget.
    The algorithm combines smooth‑max Adam optimisation with a
    directed local refinement that actively reduces the current peak
    of the self‑convolution.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Choose a problem size (power of two simplifies FFT handling)
    # --------------------------------------------------------------
    n = 4096                         # feel free to increase if budget permits
    fft_len = _next_pow2(3 * n)      # length for zero‑padded FFTs

    # --------------------------------------------------------------
    # 1) Initialise a feasible point (sum = 1, non‑negative)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)           # enforce non‑negativity
        if a0.size != n:                      # pad / truncate to the chosen length
            if a0.size < n:
                a = np.empty(n, dtype=np.float64)
                a[: a0.size] = a0
                a[a0.size :] = 0.0
            else:
                a = a0[:n].copy()
        else:
            a = a0.copy()
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a[:] = 1.0 / n

    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    best_time = time.monotonic()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Global optimisation – smooth‑max Adam
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin
    phase1_deadline = start_time + 0.75 * (deadline - start_time)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # annealing schedules (soft‑max temperature & learning‑rate)
    TAU_START, TAU_END = 0.80, 5e-06
    LR_START, LR_END = 0.50, 1e-04

    iteration = 0
    while time.monotonic() < phase1_deadline:
        # progress fraction across phase 1
        now = time.monotonic()
        frac = (now - start_time) / (phase1_deadline - start_time)

        # linearly anneal temperature and learning rate
        tau = TAU_START * (1.0 - frac) + TAU_END * frac
        lr = LR_START * (1.0 - frac) + LR_END * frac

        # ----- self‑convolution via FFT (full length 2n‑1) -----
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ----- smooth‑max weights (log‑sum‑exp surrogate) -----
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)       # shift to avoid overflow
        w = exp_shifted / exp_shifted.sum()              # sums to 1, length 2n‑1

        # ----- gradient of the surrogate (cross‑correlation) -----
        w_fft = np.fft.rfft(w, n=fft_len)
        corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
        start_corr = n - 1
        grad = (2.0 / tau) * corr_full[start_corr : start_corr + n]

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        a = _project_onto_simplex(a)

        iteration += 1

        # ----- exact evaluation every few hundred steps -----
        if iteration % 250 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                best_time = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best (Adam) = {best_score:.6f} (iter={iteration})")
                # early stop if we already beat the target
                if best_score <= 1.5025:
                    print("[early stop] target reached during Adam phase")
                    break

    # --------------------------------------------------------------
    # 3) Phase 2 – Directed stochastic local refinement
    # --------------------------------------------------------------
    a = best.copy()
    iteration_local = 0
    while time.monotonic() < deadline:
        # ----- compute full convolution once -----
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

        # ----- locate the dominant shift -----
        k_max = int(conv.argmax())
        max_val = conv[k_max]

        # ----- compute contribution weights for this shift -----
        w = _grad_weights_at_shift(a, k_max)

        # source: index with large a[i]·w[i] (large influence & mass)
        src_candidates = np.where(a > 0)[0]
        if src_candidates.size == 0:
            break
        src_metric = w[src_candidates] * a[src_candidates]
        src_idx = src_candidates[np.argmax(src_metric)]

        # target: index with smallest weight (little influence on the max)
        tgt_idx = int(np.argmin(w))

        # avoid degenerate source/target
        if src_idx == tgt_idx:
            # fall back to a random tiny move
            i = rng.integers(0, n)
            if a[i] <= 0.0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
            delta = a[i] * frac_move
            a_candidate = a.copy()
            a_candidate[i] -= delta
            a_candidate[j] += delta
        else:
            # move a moderate fraction of the source mass
            frac_move = rng.uniform(0.05, 0.25)   # 5‑25 % of source
            delta = a[src_idx] * frac_move
            a_candidate = a.copy()
            a_candidate[src_idx] -= delta
            a_candidate[tgt_idx] += delta

        sc = evaluate_sequence(a_candidate.tolist())
        iteration_local += 1

        if sc < best_score - 1e-12:
            best_score = sc
            best = a_candidate.copy()
            a = a_candidate                     # keep the improvement
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] new best (direct) = {best_score:.6f} "
                  f"(move #{iteration_local})")
            if best_score <= 1.5025:
                print("[early stop] target reached during local phase")
                break

        # occasional jitter to escape plateaus
        if iteration_local % 5000 == 0:
            width = rng.integers(1, max(2, n // 50))
            start_idx = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.02))
            a_j = a.copy()
            a_j[start_idx : start_idx + width] *= factor
            a = _project_onto_simplex(a_j)

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
