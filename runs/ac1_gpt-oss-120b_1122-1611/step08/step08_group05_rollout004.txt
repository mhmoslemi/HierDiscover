"""
Propose a step‑function (non‑negative heights) that minimizes the
`evaluate_sequence` objective.

Approach
--------
* **Phase 1 – stochastic exploration**  
  Starting from the best known construction (or a uniform start) we
  repeatedly apply a *block‑perturb* move: a random contiguous block is
  multiplied by a log‑normal factor, the vector is re‑normalised to the
  simplex and its exact score (2·n·max(conv)) is evaluated.
  Improvements are kept; after 30 s without progress the current candidate
  is re‑initialised from a Dirichlet distribution.

* **Phase 2 – refined Adam optimisation**  
  The best vector from phase 1 is refined with a smooth approximation of
  the max‑convolution (log‑sum‑exp).  We optimise the log‑softmax
  (mirrored‑gradient) using Adam while linearly raising the temperature
  parameter `γ` from 0.5 to 500.  Stagnation triggers a soft restart from
  the current best plus moderate Gaussian noise; occasional block‑jitter
  helps escaping shallow minima.  The exact score is monitored and the
  globally best vector is returned.

The routine respects a 1000 s budget (5 s safety margin) and works for any
available `height_sequence_1` (otherwise it starts from a uniform
distribution).  All helper functions are defined at module level and no
file‑ or network‑IO is performed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the LSE surrogate w.r.t. a.
    ``w`` is a probability vector (same length as the linear convolution of a with a).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform point
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to the probability simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 8))      # block size ≤ 12 % of n
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # --------------------- 0) time budget ------------------------------
    safety = 5.0
    deadline = time.monotonic() + max(0.0, float(budget_s) - safety)
    start = time.monotonic()

    # --------------------- 1) initialise -------------------------------
    # Use the provided construction if present, otherwise fall back to uniform.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
        else:
            n = 2048
            a0 = np.full(n, 1.0 / n)
    else:
        n = 2048
        a0 = np.full(n, 1.0 / n)

    # pre‑compute constants used throughout
    fft_len = _next_pow2(2 * n - 1)               # length for all FFT‑based convs

    # --------------------- 2) evaluate initial score --------------------
    a_best = a0.copy()
    cur_conv = _self_conv_fft(a_best, fft_len)
    best_score = 2.0 * n * cur_conv.max()
    best_time = time.monotonic()
    print(f"[{0.0:6.2f}s] initial score = {best_score:.6f}")

    # --------------------- 3) Phase 1 – stochastic search ---------------
    # allocate roughly 35 % of the remaining budget to phase 1
    total_budget = deadline - start
    phase2_min = 180.0                              # reserve at least 3 min for phase 2
    phase1_end = deadline - max(phase2_min, total_budget * 0.35)

    candidate = a_best.copy()
    cand_score = best_score
    last_improve = time.monotonic()

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start

        # propose a new point by block perturbation of the current candidate
        cand_new = _random_block_perturb(candidate, rng)
        # (re‑normalisation already performed inside the helper)
        conv_new = _self_conv_fft(cand_new, fft_len)
        cur_score = 2.0 * n * conv_new.max()

        # accept if better (strict improvement)
        if cur_score < cand_score - 1e-12:
            candidate = cand_new
            cand_score = cur_score

            # also update the global best if it beats it
            if cur_score < best_score - 1e-12:
                a_best = cand_new.copy()
                best_score = cur_score
                best_time = now
                last_improve = now
                print(f"[{elapsed:6.2f}s] new global best (phase 1) = {best_score:.6f}")

        # stagnation handling – restart from a fresh Dirichlet draw
        if now - last_improve > 30.0:
            candidate = rng.dirichlet(np.ones(n))
            cand_score = 2.0 * n * _self_conv_fft(candidate, fft_len).max()
            last_improve = now
            print(f"[{elapsed:6.2f}s] phase 1 restart (stagnation)")

    # --------------------- 4) Phase 2 – Adam on LSE surrogate ---------
    # Initialise Adam on the log‑softmax (mirrored) parametrisation.
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # start from the best vector found in phase 1
    x = np.log(a_best + 1e-12)               # log‑weights
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    # hyper‑parameter schedule
    gamma_start, gamma_end = 0.5, 500.0      # grow to a sharp surrogate
    lr_start, lr_end = 0.8, 0.03            # linear learning‑rate decay
    stagnation_timer = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = min(1.0, elapsed / total_budget)  # progress fraction in [0,1]

        # ---- current a on the simplex ---------------------------------
        a = _a_from_x(x)

        # ---- convolution (FFT) ----------------------------------------
        c = _self_conv_fft(a, fft_len)          # length 2·n‑1

        # ---- smooth‑max weights (log‑sum‑exp) -------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        c_shift = c - np.max(c)                 # improve numerical stability
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient wrt a --------------------------------------------
        g_a = _gradient_wrt_a(a, w)

        # ---- chain rule to log‑weights x (mirrored gradient) ----------
        avg = np.dot(a, g_a)                    # ⟨a,∇F⟩
        g_x = a * (g_a - avg)

        # ---- Adam update -----------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ---- exact score (true objective) -------------------------------
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            a_best = a.copy()
            best_score = cur_score
            best_time = now
            stagnation_timer = now
            print(f"[{elapsed:6.2f}s] new global best (phase 2) = {best_score:.6f}")

        # ---- stagnation handling (soft restart) ------------------------
        if now - stagnation_timer > 30.0:
            # Gaussian noise on log‑weights around the current best
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(a_best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            stagnation_timer = now
            print(f"[{elapsed:6.2f}s] phase 2 soft restart (stagnation)")

        # ---- occasional block‑jitter (global jump) ----------------------
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(a_best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the evaluator's bounds)
    # ------------------------------------------------------------------
    a_best = np.clip(a_best, 0.0, 1000.0)
    total_elapsed = time.monotonic() - start
    print(f"[{total_elapsed:6.2f}s] finished – best score = {best_score:.6f}")
    return [float(v) for v in a_best]
