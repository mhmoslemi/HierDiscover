def generate_sidon_greedy(n, rng):
    indices = list(range(n))
    rng.shuffle(indices)
    S = []
    sums = set()
    for cand in indices:
        # check if cand works
        conflict = False
        # Check self
        if cand + cand in sums:
            continue
        for s in S:
            if cand + s in sums:
                conflict = True
                break
        if conflict:
            continue
        # Accept
        for s in S:
            sums.add(cand + s)
        sums.add(cand + cand)
        S.append(cand)
    return S

def compute_max_conv_sparse(idx, w):
    # idx: array of ints, w: same length
    m = len(idx)
    conv = {}
    for i in range(m):
        wi = w[i]
        for j in range(i, m):
            wj = w[j]
            s = idx[i] + idx[j]
            val = wi * wj * (1.0 if i == j else 2.0)
            conv[s] = conv.get(s, 0.0) + val
    # compute max conv value
    max_val = max(conv.values()) if conv else 0.0
    return max_val

    sums=set()
    S=[]
    for cand in range(max_index+1):
        # check
        if cand + cand in sums:
            continue
        conflict=False
        for prev in S:
            if cand + prev in sums:
                conflict=True
                break
        if conflict:
            continue
        # accept
        for prev in S:
            sums.add(cand + prev)
        sums.add(cand + cand)
        S.append(cand)
    return S

Implementation of simplex projection:

def project_simplex(v):
    """Projection of vector v onto probability simplex (sum=1, nonneg)."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

import time, numpy as np

def _project_simplex(v):
    """Euclidean projection of v onto the probability simplex."""
    # v is a 1-D numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _next_pow2(x):
    return 1 << ((x - 1).bit_length())

def _self_conv_fft(a, fft_len):
    """Linear self‑convolution using FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[:2 * a.size - 1]
    return conv

def _subgradient_maxconv(a, i_max):
    """Sub‑gradient of max‑conv term at index i_max."""
    n = a.shape[0]
    g = np.zeros_like(a)
    # indices k where i_max - k is in [0, n-1]

"""
Sub‑gradient descent for the max‑convolution objective.

The evaluator is homogeneous:
    score = 2·n·max(conv) / (∑a)².
Consequently we may restrict the search to the simplex (∑a = 1) and
 minimise   F(a) = 2·n·max_i (a∗a)_i .
The function is non‑smooth because of the max.  At each iteration we
identify the index i* where the convolution attains its maximum,
compute a sub‑gradient of (a∗a)_{i*} (which is 2·a reflected and shifted
by i*), take a step along that direction and project back onto the
probability simplex.  The learning‑rate is linearly decayed.
If no improvement is observed for a few seconds we restart from the
best point with a modest log‑normal jitter; occasional global jitter
helps escape shallow plateaux.  The routine respects the 1000 s budget
and returns the best sequence discovered.

All helper functions are defined at module level (no closures) and the
code uses only NumPy and the standard library.
"""

import time

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a∗a using an FFT of fixed length.
    Returns the (2·len(a)‑1)‑long vector.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.  Implements the O(n log n) algorithm of
    Wang & Carreira‑Perpiñán (2013).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                     # all non‑positive → uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_maxconv(a: np.ndarray, i_max: int) -> np.ndarray:
    """
    Sub‑gradient of the active convolution term (a∗a)_{i_max}
    with respect to a.  For a ∈ ℝⁿ the gradient is
        g_k = 2·a_{i_max‑k}   if 0 ≤ i_max‑k < n,
        g_k = 0              otherwise.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    start = max(0, i_max - (n - 1))
    end = min(i_max + 1, n)                  # exclusive
    idx = i_max - np.arange(start, end)      # indices of a that appear
    g[start:end] = 2.0 * a[idx]
    return g

    pert = np.exp(rng.normal(0.0, scale, size=a.shape))
    b = a * pert
    b = np.clip(b, 0.0, None)
    s = b.sum()
    if s == 0.0:
        return np.full_like(a, 1.0 / a.size)
    return b / s
