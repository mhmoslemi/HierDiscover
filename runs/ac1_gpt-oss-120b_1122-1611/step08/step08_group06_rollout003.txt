"""
Propose a step‑function that (approximately) flattens its self‑convolution.
The optimizer combines two complementary objectives:

* **Variance‑flattening** – minimise the L² distance of the convolution
  from the constant value 1/(2·n‑1).  This spreads the mass of the
  auto‑correlation and reduces the overall peak.
* **Smooth‑max** – a differentiable surrogate of the true max using a
  log‑sum‑exp with an annealed temperature (γ).  The weight of this term
  grows while the L² weight decays, so the search first flattens the
  whole shape and then concentrates on the remaining spikes.

The optimisation runs on the soft‑max parametrisation
`a = softmax(x)`, guaranteeing non‑negativity and unit sum.
Adam is used for adaptive steps, with a simple learning‑rate decay.
If no improvement is seen for 30 s the algorithm restarts from the best
solution seen so far with a small random perturbation, and occasional
block‑jitter is applied to escape shallow minima.

The routine respects the global time budget (default 1000 s, 5 s safety
margin) and returns the best sequence found.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (must stay at top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to the vector ``a``.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# ----------------------------------------------------------------------

    else:

    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * c.max()          # sum(a)==1  → denominator = 1
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 6) Hyper‑parameter schedules
    # ------------------------------------------------------------------
    #   • λ (lambda) – weight of the variance term, linearly decreasing
    #   • γ (gamma) – temperature of the soft‑max, quadratically increasing
    #   • learning‑rate – linear decay from lr_start to lr_end
    lr_start = 0.8
    lr_end = 0.02
    gamma_start = 5.0
    gamma_end = 200.0

    # ------------------------------------------------------------------
    # 7) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # --------------------------------------------------------------
        # obtain current a and its self‑convolution
        # --------------------------------------------------------------
        a = _a_from_x(x)                     # always sums to 1
        c = _self_conv(a, fft_len_self)      # length 2·n‑1
        cur_score = 2.0 * n * c.max()        # exact objective (sum(a)=1)

        # --------------------------------------------------------------
        # record any improvement
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # schedule the two competing objectives
        # --------------------------------------------------------------
        # λ decays linearly from 1 → 0
        lam = max(0.0, 1.0 - elapsed / total_time)
        # γ grows quadratically (slow at start, fast at the end)
        phase_frac = elapsed / total_time
        gamma = gamma_start + (gamma_end - gamma_start) * (phase_frac ** 2)

        # --------------------------------------------------------------
        # 1) Variance‑flattening gradient
        # --------------------------------------------------------------
        avg = 1.0 / (2 * n - 1)               # average value of the convolution
        residual = c - avg
        grad_L2 = 2.0 * _gradient_wrt_a(a, residual)

        # --------------------------------------------------------------
        # 2) Smooth‑max gradient
        # --------------------------------------------------------------
        # stabilised log‑sum‑exp weights
        c_shift = c - np.max(c)               # max(c) becomes 0
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        if w_sum == 0.0:                      # numeric guard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_max = _gradient_wrt_a(a, w)

        # --------------------------------------------------------------
        # Blend the two gradients
        # --------------------------------------------------------------
        grad_a = lam * grad_L2 + (1.0 - lam) * grad_max

        # --------------------------------------------------------------
        # Chain‑rule to obtain gradient w.r.t. the log‑weights ``x``
        # --------------------------------------------------------------
        avg_g = np.dot(a, grad_a)            # ⟨a,∂F/∂a⟩
        grad_x = a * (grad_a - avg_g)        # Jacobian of the soft‑max

        # --------------------------------------------------------------
        # Adam step
        # --------------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # --------------------------------------------------------------
        # Stagnation handling – soft restart if no improvement for 30 s
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            # perturb the best solution a bit and continue from there
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # --------------------------------------------------------------
        # Occasional block‑jitter to escape shallow minima
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 8) Return the best sequence (clip to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
