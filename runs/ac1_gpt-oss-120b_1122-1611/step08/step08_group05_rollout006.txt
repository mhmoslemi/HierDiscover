def _local_move(a, rng, eps):
    n = len(a)
    i = rng.integers(n)
    j = rng.integers(n-1)
    if j >= i:
        j += 1
    # i source, j target
    # amount to move = min(eps, a[i])
    delta = min(eps, a[i])
    if delta == 0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    return out

def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, budget_s - 5.0)
    # initialize a from existing if present else random Dirichlet.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        a0 = np.clip(a0, 0, None)
        s = a0.sum()
        if s > 0:
            a0 /= s
        else:
            raise ValueError("init nonzero sum")
        n = a0.size
    else:
        n = 2048
        a0 = rng.dirichlet(np.ones(n))
    # best tracking
    best_a = a0.copy()
    best_score = evaluate_sequence(best_a.tolist())
    last_improve = time.monotonic()
    start = time.monotonic()
    total_time = deadline - start

    # set up parameters for LSE gradient
    x = np.log(best_a + 1e-12) # log-weights (log softmax)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9; beta2=0.999; eps=1e-8; t_step=0
    # precompute fft_len
    fft_len = _next_pow2(2*n-1)

    # Main loop
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total_time

        # determine method; simple schedule:
        # Use gradient descent for first 60% of time, then local search for rest.
        if elapsed < 0.6 * total_time:
            # gradient descent step
            a = _softmax(x)
            # compute conv via FFT for speed
            c = _self_conv_fft(a, fft_len)
            # compute gamma schedule
            gamma = 0.5 + (250.0 - 0.5) * frac  # go up to 250
            # compute weights
            c_centered = c - np.max(c)
            w = np.exp(gamma * c_centered)
            w /= w.sum()
            # gradient
            g_a = _gradient_wrt_a(a, w)
            # compute gradient w.r.t x
            avg = np.dot(a, g_a)
            g_x = a * (g_a - avg)
            # adam update
            t_step += 1
            m = beta1 * m + (1-beta1) * g_x
            v = beta2 * v + (1-beta2) * (g_x * g_x)
            m_hat = m / (1-beta1**t_step)
            v_hat = v / (1-beta2**t_step)
            lr = 0.8 * (1 - frac) + 0.02 * frac  # decay to small
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            x -= step
            # evaluate exact score periodically
            if now - last_improve > 10.0 or (t_step % 50 == 0):
                a_cur = _softmax(x)
                conv_cur = _self_conv_direct(a_cur)
                cur_score = 2.0 * n * conv_cur.max()
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a_cur.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] gradient new best score {best_score:.6f}")
        else:

            # pick random i/j, small epsilon
            a_cur = best.copy()  # start from best
            # try many moves in a batch

            cur_score = 2.0 * n * np.convolve(a_check, a_check).max()
            if cur_score < best_score:
                best_score = cur_score
                best_a = a_check.copy()
    return best_a, best_score

"""
Propose a low‑score step‑function using many short runs of
entropy‑regularised Adam on the log‑softmax of the coefficients.
Each run starts from a different random Dirichlet (or the known
construction) and performs a smooth‑max (log‑sum‑exp) surrogate
gradient descent with a slowly increasing temperature γ.
Restarts with log‑normal noise are triggered after a few seconds of
no improvement.  The best sequence seen across all runs is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities – must stay at module scope (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _self_conv_direct(a: np.ndarray) -> np.ndarray:
    """Exact linear convolution via NumPy (used for precise score checks)."""
    return np.convolve(a, a)


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the LSE surrogate w.r.t. a.
    ``w`` is a probability vector (same length as the linear self‑convolution).
    """
    n = a.size
    # correlation: rev(a) * w  (linear)
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _log_weights(a: np.ndarray) -> np.ndarray:
    """Convert a probability vector to log‑weights (used as optimiser state)."""
    return np.log(a + 1e-12)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to sum‑one.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Single optimisation run (gradient descent on the LSE surrogate)
# ----------------------------------------------------------------------


def _run_gradient_opt(
    a_start: np.ndarray,
    time_limit: float,
    rng: np.random.Generator,
    run_id: int = 0,
) -> tuple[np.ndarray, float]:
    """
    Perform a short Adam optimisation on the log‑softmax of ``a_start``
    (sum of elements forced to one).  Returns the best sequence found
    and its exact ``evaluate_sequence`` score.
    """
    n = a_start.size
    fft_len = _next_pow2(2 * n - 1)

    # Adam state
    x = _log_weights(a_start)          # log‑weights, a = softmax(x)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    step_counter = 0

    # schedules
    gamma_start = 0.5
    gamma_end = 500.0          # sharper surrogate than the baseline
    lr_start = 0.8
    lr_end = 0.02

    # bookkeeping
    start_time = time.monotonic()
    deadline = start_time + time_limit
    best_a = a_start.copy()
    best_score = 2.0 * n * _self_conv_direct(best_a).max()
    last_improve = start_time

    while time.monotonic() < deadline:
        # --- current iterate -------------------------------------------------
        a = _softmax(x)                     # simplex, sum = 1

        # --- self‑convolution (FFT, cheap) ----------------------------------
        c = _self_conv_fft(a, fft_len)      # length 2·n‑1

        # --- schedule parameters ---------------------------------------------
        elapsed = time.monotonic() - start_time
        frac = elapsed / time_limit
        gamma = gamma_start + (gamma_end - gamma_start) * frac

        # --- smooth‑max weights -----------------------------------------------
        c_shift = c - np.max(c)             # stabilise exp
        w = np.exp(gamma * c_shift)
        w /= w.sum()                        # probability vector

        # --- gradient w.r.t. a ------------------------------------------------
        g_a = _gradient_wrt_a(a, w)

        # --- chain‑rule to log‑weights x --------------------------------------
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # --- Adam update ------------------------------------------------------
        step_counter += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** step_counter)
        v_hat = v / (1.0 - beta2 ** step_counter)
        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # --- exact evaluation every few iterations ----------------------------
        if step_counter % 40 == 0:
            a_check = _softmax(x)
            cur_score = 2.0 * n * _self_conv_direct(a_check).max()
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a_check.copy()
                last_improve = time.monotonic()
                print(
                    f"[run {run_id:02d}]  t={elapsed:5.1f}s  new best score = {best_score:.6f}"
                )

        # --- stagnation handling (soft restart) -------------------------------
        if time.monotonic() - last_improve > 8.0:
            # add log‑normal noise around the current best and reset Adam.
            noise = rng.normal(0.0, 0.6, size=n)
            x = _log_weights(best_a) + noise
            m.fill(0.0)
            v.fill(0.0)
            step_counter = 0
            last_improve = time.monotonic()
            # occasional block jitter to shake the solution
            if rng.random() < 0.15:
                best_a = _random_block_perturb(best_a, rng)
                x = _log_weights(best_a)
                print(
                    f"[run {run_id:02d}]  jitter applied – score still {best_score:.6f}"
                )

    return best_a, best_score


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a sequence of non‑negative heights that minimises ``evaluate_sequence``.
    The routine runs many short Adam‑based optimisations from diverse
    random starts (including the provided ``height_sequence_1`` if present)
    and returns the best sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Keep track of the overall best solution
    # ------------------------------------------------------------------
    overall_best = None
    overall_score = float("inf")
    run_counter = 0

    # ------------------------------------------------------------------
    # 2) Main loop – keep launching short optimisation runs until time runs out
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        # reserve a small slice for the next run (min 8 s, max 40 s)
        run_time = min(40.0, max(8.0, 0.04 * time_left))

        # --- pick a starting point ---------------------------------------
        if run_counter == 0 and "height_sequence_1" in globals():
            # start from the previously best construction if it exists
            a0 = np.asarray(
                globals()["height_sequence_1"], dtype=np.float64
            )
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0.0:
                a0 /= s
            else:
                # fall back to a uniform Dirichlet
                a0 = None
        else:
            a0 = None

        if a0 is None:
            # random length in a sensible range, then Dirichlet draw
            n = int(rng.integers(1500, 4500))
            a0 = rng.dirichlet(np.ones(n))

        # ----------------------------------------------------------------
        # 3) Run a short Adam optimisation from this start
        # ----------------------------------------------------------------
        seq, sc = _run_gradient_opt(a0, run_time, rng, run_id=run_counter)

        # ----------------------------------------------------------------
        # 4) Update the global best if improved
        # ----------------------------------------------------------------
        if sc < overall_score - 1e-12:
            overall_score = sc
            overall_best = seq.copy()
            print(
                f"[global] run {run_counter:02d} improved overall best to {overall_score:.6f}"
            )

        run_counter += 1

    # ------------------------------------------------------------------
    # 5) Return the best sequence (safeguarded to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    if overall_best is None:
        # Extremely unlikely – fall back to a uniform distribution of length 2048
        overall_best = np.full(2048, 1.0 / 2048.0)

    overall_best = np.clip(overall_best, 0.0, 1000.0)
    return [float(v) for v in overall_best]
