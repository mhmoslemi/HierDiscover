def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 5.0   # safety margin

    best_seq = None
    best_score = np.inf

    # List of lengths to try, increasing.
    lengths = [1024, 2048, 4096, 8192, 16384]
    # Limit to max length that we can finish within budget.
    for n in lengths:
        # time left check.
        now = time.monotonic()
        if now >= deadline:
            break

        # Determine number of iterations allowed for this n.
        # Roughly allocate proportion ~ 2/n? Not needed.
        # We'll simply run a while loop until time
        # or reaching a fixed iteration budget per n.

        idx = np.arange(n, dtype=np.float64) + 1.0

        for a in init_candidates:
            # Ensure at least some time left.
            if time.monotonic() >= deadline:
                break

            a = a.copy()
            # Phase 1: Smooth‑max Adam
            # customised hyperparams per n: more iterations.
            best_local = a.copy()

            # Setup Adam parameters
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1, beta2 = 0.9, 0.999
            beta1_pow, beta2_pow = 1.0, 1.0

            # Determine dropout schedule: whole optimization runs for limited iterations.

            max_iter = 8000

            phase1_end_time = min(deadline, time.monotonic() + 0.4 * (deadline - start))

            while iteration < max_iter and time.monotonic() < phase1_end_time:
                # Compute temperature and learning rate varying linearly with iteration count.
                frac = iteration / max_iter
                tau = 0.4 * (1 - frac) + 5e-5 * frac      # from 0.4 down to 5e-5
                lr = 0.3 * (1 - frac) + 2e-4 * frac      # from 0.3 down to 2e-4

                fft_len = _next_pow2(3 * n)

                shift = conv.max()
                exp_shifted = np.exp((conv - shift) / tau)
                w = exp_shifted / exp_shifted.sum()
                # correlation gradient
                w_fft = np.fft.rfft(w, n=fft_len)
                corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
                start_corr = n - 1

                m = beta1 * m + (1 - beta1) * grad

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
        {x ≥ 0, sum(x) = 1}
    """
    if v.ndim != 1:                     # pragma: no cover
        raise ValueError("Only 1-D vectors are supported.")

    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):                # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))

    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)

    # re‑normalise to guard against tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w

            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full_like(a, 1.0 / n)

            # Quick evaluation of the start
            _maybe_update(a)

            # -----------------------------
            # Phase 1 – smooth‑max Adam
            # -----------------------------
            # FFT length big enough for convolutions of length 2*n‑1
            fft_len = _next_pow2(2 * n)

            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1, beta2 = 0.9, 0.999
            beta1_pow, beta2_pow = 1.0, 1.0

            # allocate roughly 30 % of the remaining budget to the Adam phase
            phase1_deadline = min(deadline,
                                  time.monotonic() + 0.30 * (deadline - start_time))

            it = 0
            while time.monotonic() < phase1_deadline:
                # linear annealing schedules
                frac = it / 20000.0                     # we plan for up to 20 k iters
                tau = 0.6 * (1.0 - frac) + 5e-5 * frac   # 0.6 → 5e‑5
                lr = 0.5 * (1.0 - frac) + 5e-4 * frac    # 0.5 → 5e‑4

                # ----- convolution via FFT -----
                a_fft = np.fft.rfft(a, n=fft_len)
                conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

                # ----- soft‑max surrogate for the max -----
                shift = conv.max()
                exp_shifted = np.exp((conv - shift) / tau)   # shift for numerical stability
                w = exp_shifted / exp_shifted.sum()          # weights sum to 1 (length 2n‑1)

                # ----- gradient of the smooth surrogate -----
                w_fft = np.fft.rfft(w, n=fft_len)
                corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
                start_corr = n - 1
                grad = (2.0 / tau) * corr_full[start_corr:start_corr + n]

                # ----- Adam update -----
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                beta1_pow *= beta1
                beta2_pow *= beta2
                m_hat = m / (1.0 - beta1_pow)
                v_hat = v / (1.0 - beta2_pow)
                step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)

                a = a - step
                a = _project_onto_simplex(a)

                # occasional exact evaluation
                if it % 500 == 0:
                    sc = evaluate_sequence(a.tolist())
                    _maybe_update(a, sc)
                    if sc <= 1.5025:                     # target reached – finish early
                        a = np.clip(a, 0.0, 1000.0)
                        return [float(x) for x in a.tolist()]

                it += 1

            # -----------------------------
            # Phase 2 – stochastic local refinement
            # -----------------------------
            a_local = a.copy()
            move_counter = 0
            while time.monotonic() < deadline:
                i = rng.integers(0, n)
                if a_local[i] <= 0.0:
                    continue
                j = rng.integers(0, n)
                if i == j:
                    continue

                # transfer a tiny fraction of mass from i to j
                frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.05)))
                delta = a_local[i] * frac_move

                cand = a_local.copy()
                cand[i] -= delta
                cand[j] += delta

                sc = evaluate_sequence(cand.tolist())
                move_counter += 1

                if sc < best_score - 1e-12:
                    _maybe_update(cand, sc)
                    a_local = cand
                    if sc <= 1.5025:
                        cand = np.clip(cand, 0.0, 1000.0)
                        return [float(x) for x in cand.tolist()]

                # occasional jitter to escape plateaus
                if move_counter % 4000 == 0:
                    width = rng.integers(1, max(2, n // 80))
                    start_idx = rng.integers(0, n - width + 1)
                    factor = np.exp(rng.normal(0.0, 0.015))
                    jitter = a_local.copy()
                    jitter[start_idx:start_idx + width] *= factor
                    a_local = _project_onto_simplex(jitter)

    # ------------------------------------------------------------------
    # Return the best sequence discovered (clip to evaluator limits)
    # ------------------------------------------------------------------
    if best_seq is None:                         # pragma: no cover
        # should never happen – fall back to a uniform vector
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    final_score = evaluate_sequence(best_seq.tolist())
    total = time.monotonic() - start_time
    print(f"Finished after {total:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best_seq]
```

This implementation expands the search space (multiple lengths and initialisations), uses a more aggressive annealing schedule, and adds a fast stochastic refinement stage.  It respects all constraints (non‑negative heights, sum normalisation via projection) and prints progress so that the harness can see intermediate improvements.
