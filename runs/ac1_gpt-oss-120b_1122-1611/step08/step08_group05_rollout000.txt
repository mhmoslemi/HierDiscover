"""
Propose candidate – Subgradient+Annealing optimizer with final hill‑climbing.

The objective is homogeneous, so we restrict ∑a = 1 (simplex) and
minimise  F(a)=2·n·max_i (a*a)_i .
We optimise the log‑softmax (log‑weights) `x` with Adam on the true
max (using a sharp log‑sum‑exp surrogate that becomes a one‑hot weight
vector).  After the main phase we perform a greedy local search that
moves a small amount of mass between two coordinates; this often
exploits patterns that a gradient step cannot see.

The routine respects the 1000 s budget (leaves a 5 s safety margin),
periodically prints progress, restarts from the current best when
stagnation is detected and finishes with a short hill‑climbing phase.
"""
import time
import numpy as np

# ------------------------------------------------------------
# Helper utilities (module level, no closures)
# ------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the weighted sum   Σ_i w_i (a*a)_i   w.r.t. a.
    ``w`` has length 2·n‑1 (same as the linear convolution).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _move_mass(a: np.ndarray, i: int, j: int, delta: float) -> np.ndarray:
    """
    Return a copy of `a` where `delta` is moved from coordinate `i` to `j`.
    Guarantees non‑negativity (clamps at zero if delta > a[i]).
    """
    if i == j:
        return a
    out = a.copy()
    d = min(delta, out[i])
    out[i] -= d
    out[j] += d
    return out


# ------------------------------------------------------------
# Main optimisation routine
# ------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.

    The algorithm consists of:
      1. Adam optimisation on the log‑softmax of the coefficients,
         using a log‑sum‑exp surrogate that becomes a hard max.
      2. A short stochastic hill‑climbing phase that moves tiny amounts
         of probability mass between coordinates to fine‑tune the max
         of the self‑convolution.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise from a known construction if available.
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim != 1 or a0.size == 0:
            a0 = None
    else:
        a0 = None

    if a0 is None:
        # fall back to a uniform start of a modest size
        n = 2048
        a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a0.size
        # enforce non‑negativity and normalise to the simplex
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0.0:
            a0[:] = 1.0 / n
        else:
            a0 /= s

    # Log‑weights (softmax parameterisation)
    x = np.log(a0 + 1e-12)          # small epsilon to avoid log(0)

    # ------------------------------------------------------------------
    # 2) FFT planning (same length for all convolutions)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Adam state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Helper to retrieve probabilities from log‑weights
    # ------------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # ------------------------------------------------------------------
    # 5) Track the best solution (exact evaluation)
    # ------------------------------------------------------------------
    a = _a_from_x(x)
    cur_conv = _self_conv_fft(a, fft_len_self)   # exact conv
    best_score = 2.0 * n * cur_conv.max()
    best = a.copy()
    last_improve = time.monotonic()
    start = time.monotonic()
    total_time = deadline - start

    # ------------------------------------------------------------------
    # 6) Hyper‑parameters (schedule)
    # ------------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 2000.0          # very sharp surrogate → true max
    lr_start = 1.0
    lr_end = 0.001

    # ------------------------------------------------------------------
    # 7) Main Adam optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total_time

        # ----- current iterate ------------------------------------------------
        a = _a_from_x(x)                # simplex (sum = 1)

        # ----- self‑convolution (FFT) -----------------------------------------
        c = _self_conv_fft(a, fft_len_self)   # length 2·n‑1
        c_max = c.max()
        c_shift = c - c_max                     # stabilise exponentials

        # ----- smooth‑max weights ---------------------------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        w /= w_sum                              # probability vector

        # ----- gradient wrt a --------------------------------------------------
        g_a = _gradient_wrt_a(a, w)               # length n

        # ----- chain rule to log‑weights x --------------------------------------
        avg = np.dot(a, g_a)                      # ⟨a,∇F⟩
        g_x = a * (g_a - avg)                     # keeps a on simplex

        # ----- Adam update ------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- exact evaluation -----------------------------------------------
        exact_score = 2.0 * n * c_max            # same as evaluate_sequence for sum=1
        if exact_score < best_score - 1e-12:
            best_score = exact_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best exact score = {best_score:.6f}")

        # ----- stagnation handling (soft restart) ------------------------------
        if now - last_improve > 30.0:
            # Re‑initialise log‑weights from the current best plus moderate noise
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # ----- occasional block‑jitter (global jump) ---------------------------
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 8) Local hill‑climbing phase (if time permits)
    # ------------------------------------------------------------------
    # Use the remaining time (if any) for a greedy search that moves a tiny
    # amount of mass between two coordinates.  This can shave a few 1e‑4
    # off the objective when the gradient phase has already converged.
    if time.monotonic() < deadline:
        a = best.copy()
        conv_a = _self_conv_fft(a, fft_len_self)
        best_score = 2.0 * n * conv_a.max()
        # Parameters for the hill‑climbing move
        movedelta = 0.005               # absolute amount of mass moved each step
        max_moves = 200000             # safety cap
        moves = 0
        while time.monotonic() < deadline and moves < max_moves:
            # pick source with non‑negligible mass to ensure a visible move
            src = rng.integers(n)
            if a[src] < movedelta:
                # try again a few times
                for _ in range(5):
                    src = rng.integers(n)
                    if a[src] >= movedelta:
                        break
                if a[src] < movedelta:
                    # all too small – pick a random source anyway
                    src = rng.integers(n)
            dst = rng.integers(n - 1)
            if dst >= src:
                dst += 1  # ensure dst != src

            a_new = _move_mass(a, src, dst, movedelta)

            # recompute convolution (FFT is cheap for one move)
            conv_new = _self_conv_fft(a_new, fft_len_self)
            score_new = 2.0 * n * conv_new.max()

            if score_new < best_score - 1e-12:
                # accept improvement
                best_score = score_new
                best = a_new.copy()
                a = a_new
                conv_a = conv_new
                elapsed = time.monotonic() - start
                print(f"[{elapsed:6.2f}s] hill‑climb improvement → {best_score:.6f}")
            # else reject (greedy)

            moves += 1

    # ------------------------------------------------------------------
    # 9) Return the best sequence (clipped to the bounds of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
