"""
Simulated‑Annealing + Projected‑Adam optimizer for the step‑function problem.

* Phase 1 – Global search:
  Simple Metropolis moves that transfer a small random amount of mass
  between two coordinates while staying on the probability simplex.
  The temperature is linearly cooled; occasional large Dirichlet jumps
  are performed to escape deep valleys.

* Phase 2 – Local refinement:
  Projected Adam on the *exact* max‑convolution sub‑gradient
  (no soft‑max smoothing).  The learning‑rate decays linearly,
  Adam moments accelerate convergence and the iterates are projected
  back onto the simplex after each step.

The routine respects the 1000 s budget (5 s safety margin) and returns
the best coefficient list found.  The initial population contains the
previous best construction (``height_sequence_1`` when available) as
well as a few hand‑crafted seeds (uniform, exponentially decaying,
two spikes, linear, Dirichlet draws).  All numbers are clipped to the
evaluation limits before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    non‑negative entries that sum to 1.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑dimensional")
    n = v.size
    # sort v in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _initial_candidates(n: int, rng: np.random.Generator):
    """Diverse ready‑made seeds, all normalised to sum = 1."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes (ends)
    v = np.zeros(n, dtype=np.float64)
    v[0] = v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # previous construction, if available
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.size != n:
            if prev.size < n:
                prev = np.concatenate([prev, np.zeros(n - prev.size)])
            else:
                prev = prev[:n]
        prev /= prev.sum()
        cand.append(prev)

    return cand


def _compute_J(a: np.ndarray, n: int, fft_len: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a⋆a) for a already on the simplex."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray, fft_len: int, conv_len: int):
    """Return the full self‑convolution and its maximum index/value."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    idx = int(conv.argmax())
    return conv, idx, float(conv.max())


def _subgradient_max(a: np.ndarray, idx: int) -> np.ndarray:
    """
    Exact sub‑gradient of max(a⋆a) at the (any) arg‑max position `idx`.
    For all i with 0 ≤ i < n and 0 ≤ idx‑i < n we have
        ∂b_idx/∂a_i = 2·a_{idx‑i},
    otherwise the derivative is zero.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    i_start = max(0, idx - (n - 1))
    i_end = min(idx, n - 1)  # inclusive
    i_range = np.arange(i_start, i_end + 1)
    j_range = idx - i_range
    g[i_range] = 2.0 * a[j_range]
    return g


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.maximum(out, 0.0)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method consists of a simulated‑annealing global
    exploration followed by a projected‑Adam refinement using the exact max‑
    convolution sub‑gradient.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        n = len(height_sequence_1)
        n = max(512, min(n, 4096))          # keep FFT cheap
    else:
        n = 4096
    # keep the original length (not forced to a power of two)
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)          # for a⋆a

    # ------------------------------------------------------------------
    # 2) Initialise a diverse pool of candidates and keep the best
    # ------------------------------------------------------------------
    candidates = _initial_candidates(n, rng)
    best_vec = None
    best_score = np.inf
    for cand in candidates:
        cand = proj_simplex(cand.astype(np.float64))
        sc = _compute_J(cand, n, fft_len, conv_len)
        if sc < best_score:
            best_score = sc
            best_vec = cand.copy()
    print(f"[{time.monotonic() - start_time:6.2f}s] Initial best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – Simulated Annealing (global search)
    # ------------------------------------------------------------------
    # allocate ~55 % of the remaining budget to SA
    time_now = time.monotonic()
    remaining = deadline - time_now
    sa_deadline = time_now + 0.55 * remaining

    a = best_vec.copy()               # current state
    cur_score = best_score
    last_improve = time_now
    it = 0

    # temperature schedule (linear cooling)
    T_start = 0.12
    T_end = 1e-4

    while time.monotonic() < sa_deadline:
        it += 1
        # linear cooling based on fraction of SA time used
        frac = (time.monotonic() - time_now) / (sa_deadline - time_now)
        T = T_start * (1.0 - frac) + T_end * frac

        # ----- propose a simple mass‑transfer move -----
        i = rng.integers(n)
        j = rng.integers(n - 1)
        if j >= i:
            j += 1                     # ensure i != j
        eps = min(a[i], rng.exponential(scale=0.01))
        if eps <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # already stays on the simplex, only non‑negativity may be violated
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s

        new_score = _compute_J(a_new, n, fft_len, conv_len)
        delta = new_score - cur_score

        accept = False
        if delta < 0.0:
            accept = True
        elif rng.random() < np.exp(-delta / max(T, 1e-12)):
            accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_vec = a.copy()
                last_improve = time.monotonic()
                print(f"[{last_improve - start_time:6.2f}s] SA new best J = {best_score:.6f}")

        # ----- occasional large Dirichlet jump (to escape deep valleys) -----
        if rng.random() < 0.008:
            p = 0.15  # mixing proportion
            dirich = rng.dirichlet(np.ones(n) * 0.3)
            a_trial = (1.0 - p) * a + p * dirich
            a_trial = proj_simplex(a_trial)
            trial_score = _compute_J(a_trial, n, fft_len, conv_len)
            if trial_score < cur_score:
                a = a_trial
                cur_score = trial_score
                if trial_score < best_score - 1e-12:
                    best_score = trial_score
                    best_vec = a.copy()
                    last_improve = time.monotonic()
                    print(f"[{last_improve - start_time:6.2f}s] SA Dirichlet jump improved J = {best_score:.6f}")

        # ----- stagnation restart -----
        if time.monotonic() - last_improve > 15.0:
            # re‑inject a tiny random perturbation around the current best
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            cur_score = _compute_J(a, n, fft_len, conv_len)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] SA restart after stagnation – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – Projected Adam using the exact max‑subgradient
    # ------------------------------------------------------------------
    # allocate remaining time to Adam refinement
    time_now = time.monotonic()
    remaining = deadline - time_now
    if remaining <= 0.0:
        # time exhausted, return current best
        best_vec = np.clip(best_vec, 0.0, 1000.0)
        return [float(x) for x in best_vec]

    a = best_vec.copy()
    m = np.zeros_like(a)         # Adam first moment
    v = np.zeros_like(a)         # Adam second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it_adam = 0
    last_improve = time_now

    # learning‑rate schedule (linear decay)
    lr_start, lr_end = 0.28, 0.001

    while time.monotonic() < deadline:
        it_adam += 1
        # current learning‑rate
        frac = (time.monotonic() - time_now) / remaining
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # convolution & argmax
        conv, idx_max, _ = _conv_and_max(a, fft_len, conv_len)

        # exact sub‑gradient of the max entry
        g = _subgradient_max(a, idx_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** it_adam)
        v_hat = v / (1.0 - beta2 ** it_adam)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        # project back onto the simplex
        a_new = proj_simplex(a_new)

        # evaluate
        new_score = _compute_J(a_new, n, fft_len, conv_len)
        if new_score < best_score - 1e-12:
            best_score = new_score
            best_vec = a_new.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Adam new best J = {best_score:.6f}")

        # accept the move (always keep the new point)
        a = a_new

        # occasional jitter to help escape plateaus
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng)

        # restart if no improvement for a while
        if time.monotonic() - last_improve > 20.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve = time.monotonic()
            print(f"[{last_improve - start_time:6.2f}s] Adam restart (stagnation) – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
