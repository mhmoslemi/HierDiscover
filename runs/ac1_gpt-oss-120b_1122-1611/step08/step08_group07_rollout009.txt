def project_simplex(v):
    # v is a np array.
    # algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u > (cssv - 1) / np.arange(1, len(v)+1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _project_simplex(v):
    """Project onto simplex {x >= 0, sum x = 1}."""
    # Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

        g_x = a * (g_a - avg)

        # evaluate actual

        if cur_score < best_score - 1e-12:

            best_a = a.copy()
            last_improve = now

            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now

        # occasional random block perturbation
        if rng.random() < 0.01:
            pert = _random_block_perturb(best_a, rng)
            x = np.log(pert + 1e-12)
            m.fill(0.0); v.fill(0.0); t = 0

    # Phase II: projected gradient descent on max
    a = best_a.copy()
    cur_score = compute_score(a)
    best_score = cur_score
    best_a = a.copy()
    last_improve = time.monotonic()
    # parameters for Phase II

    decay = 0.99
    min_step = 1e-5
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        # compute conv and find peak
        c = _self_conv(a, fft_len)
        p = int(np.argmax(c))
        max_val = c[p]
        # gradient g_i = a[p-i] if in range else 0
        g = np.zeros_like(a)
        # compute g by slicing
        # for i in 0..n-1 where 0 <= p-i < n:

            a_candidate = a - eta * g
            # project onto simplex
            a_candidate = _project_simplex(a_candidate)
            new_score = compute_score(a_candidate)
            if new_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = new_score
                improved = True
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] New best (grad) {best_score:.6f}")
                break
        if not improved:
            # reduce step size
            step_size *= decay
            if step_size < min_step:
                step_size = min_step
            # occasional random perturb to escape
            if rng.random() < 0.01:
                # small random shift in log-space
                idx = rng.integers(0, n)
                delta = rng.normal(0.0, 0.02)
                a_pert = a.copy()
                a_pert[idx] = max(a_pert[idx] + delta, 0.0)
                a_pert = _project_simplex(a_pert)
                a = a_pert
                cur_score = compute_score(a)
        # restart if no improvement for a while
        if now - last_improve > 30.0:
            a = best_a.copy()
            # jitter a few times
            for _ in range(3):
                a = _random_block_perturb(a, rng)
                a = _project_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Local restart (stagnation)")
    # final return
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]

"""
Improved search for a step‑function that minimises

    2 * n * max(conv(a,a)) / (sum(a)**2)

The algorithm has two stages.

Stage I – smooth‑max Adam optimisation
    * Parameters are kept in log‑space (soft‑max) so the simplex
      constraints (a_i ≥ 0, Σa_i = 1) are always satisfied.
    * The true max is replaced by a log‑sum‑exp surrogate with a
      linearly increasing temperature γ.  Adam updates the log‑weights.
    * Restarts with Gaussian noise are triggered after a period of
      stagnation; occasional block‑perturbations avoid getting stuck.

Stage II – projected sub‑gradient descent on the *actual* maximum
    * The current worst convolution index p = argmax (a*a) is found.
    * The gradient of the objective w.r.t. a at this index is
          g_i = a_{p-i}  (or 0 if the index is out of range).
    * To keep Σa_i = 1 we centre the gradient (g ← g – mean(g)).
    * A simple step a ← a – η g is taken and the result is projected back
      onto the probability simplex.
    * The step size η is auto‑adapted; if no improvement is seen it is
      reduced, and after a long stall the current best solution is jittered.

Both stages keep the globally best sequence seen so far and print progress.
The routine respects the 1000 s budget (5 s safety margin) and finally
returns a plain list of floats that can be fed directly to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    # keep only the linear part (length 2·n‑1)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate F_γ(a) = (1/γ)·log Σ_k exp(γ·c_k)
    with respect to ``a``.
    ``w`` are the soft‑max weights (they sum to 1 and have length 2·n‑1).
    Returns a vector of length n.
    """
    n = a.size
    # reverse a for the correlation that appears in the derivative
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    # the required slice corresponds to the indices n‑1 … 2·n‑2‑1 = n‑1 … n‑1+n‑1
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    Duchi et al., “Efficient Projections onto the l1‑Ball for Learning
    in High Dimensions”, 2008.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Degenerate case – return the uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to a probability vector.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _compute_score(a: np.ndarray, fft_len: int) -> float:
    """
    Fast evaluation of the exact objective (without the extra checks of
    ``evaluate_sequence``).  The sequence is assumed to be already normalised
    (Σa = 1) and non‑negative.
    """
    n = a.size
    c = _self_conv(a, fft_len)
    max_val = np.max(c)
    return float(2.0 * n * max_val)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the previous best (≈1.5132).
    The routine runs for at most ``budget_s`` seconds (a 5 s safety
    margin is subtracted internally).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a length – a power of two gives fast FFTs.
    #    Larger n allows a finer approximation of the continuous optimum.
    # ------------------------------------------------------------------
    n = 8192                     # 2¹³, still comfortably fast for FFTs

    # ------------------------------------------------------------------
    # 2) Initialise a probability vector.
    #    If a previous construction exists we reuse it (truncated / padded).
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64).ravel()
        if a0.size > n:
            a0 = a0[:n]
        else:
            a0 = np.pad(a0, (0, n - a0.size), constant_values=0.0)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a = a0 / s
        else:
            a = np.full(n, 1.0 / n)
    else:
        # Dirichlet(1) gives a random point on the simplex
        a = rng.dirichlet(np.ones(n))

    # ------------------------------------------------------------------
    # 3) Pre‑compute FFT length for the self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 4) Track the best solution seen so far
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = _compute_score(a, fft_len_self)
    start_time = time.monotonic()
    last_improve = start_time

    # ------------------------------------------------------------------
    # 5) ---------- Stage I – smooth‑max Adam optimisation ----------------
    # ------------------------------------------------------------------
    # 5.1 log‑weights (soft‑max guarantees the simplex)
    x = np.log(a + 1e-12)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # hyper‑parameters (tuned for a longer run)
    gamma_start = 0.5
    gamma_end = 500.0                # sharper surrogate at the end
    lr_start = 0.5
    lr_end = 0.02

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / (deadline - start_time)

        # stop early enough to leave room for Stage II
        if frac > 0.70:
            break

        # ---- current point on the simplex ----------------------------
        a = _softmax(x)

        # ---- linear convolution (exact) -------------------------------
        c = _self_conv(a, fft_len_self)

        # ---- smooth‑max weights ---------------------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        c_shift = c - np.max(c)               # for numeric stability
        w = np.exp(gamma * c_shift)
        w /= w.sum()

        # ---- gradient w.r.t. a (smooth surrogate) --------------------
        g_a = _gradient_wrt_a(a, w)

        # ---- chain rule to log‑weights (soft‑max Jacobian) -----------
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # ---- Adam step ------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ---- exact score (for tracking) -------------------------------
        cur_score = _compute_score(a, fft_len_self)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Adam new best score = {best_score:.6f}")

        # ---- stagnation restart ---------------------------------------
        if now - last_improve > 25.0:
            noise = rng.normal(0.0, 0.3, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (stagnation)")

        # ---- occasional global jitter ---------------------------------
        if rng.random() < 0.01:
            pert = _random_block_perturb(best_a, rng)
            x = np.log(pert + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 6) ---------- Stage II – projected sub‑gradient on the true max ---
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = _compute_score(a, fft_len_self)
    best_score = cur_score
    best_a = a.copy()
    last_improve = time.monotonic()

    # step‑size schedule for the sub‑gradient updates
    eta = 0.12          # initial step
    decay = 0.985
    min_eta = 1e-5

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # compute convolution and locate the offending index
        c = _self_conv(a, fft_len_self)
        p = int(np.argmax(c))          # index of the current maximum
        max_val = c[p]

        # gradient of max_{k} (a*a)_k w.r.t. a at the active index p
        g = np.zeros_like(a)
        lo = max(0, p - (n - 1))
        hi = min(p, n - 1)
        # for i ∈ [lo, hi] we have a_{p-i} inside the support
        idx_range = np.arange(lo, hi + 1)
        g[idx_range] = a[p - idx_range]

        # centre the gradient so that the update preserves the sum of a
        g -= g.mean()

        improved = False
        # try a few step‑size factors (0.5×, 1×, 2×) to obtain a decrease
        for factor in (1.0, 0.5, 2.0):
            a_candidate = a - eta * factor * g
            a_candidate = _project_simplex(a_candidate)
            new_score = _compute_score(a_candidate, fft_len_self)

            if new_score < cur_score - 1e-12:
                a = a_candidate
                cur_score = new_score
                improved = True
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best_a = a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] New best (grad) score = {best_score:.6f}")
                break

        if not improved:
            # no progress → shrink step size
            eta = max(min_eta, eta * decay)

            # occasional tiny random nudge to escape flat regions
            if rng.random() < 0.02:
                idx = rng.integers(0, n)
                delta = rng.normal(0.0, 0.02)
                a_pert = a.copy()
                a_pert[idx] = max(a_pert[idx] + delta, 0.0)
                a_pert = _project_simplex(a_pert)
                a = a_pert
                cur_score = _compute_score(a, fft_len_self)

        # restart/jitter if we have been stuck for a while
        if now - last_improve > 30.0:
            a = best_a.copy()
            for _ in range(3):
                a = _random_block_perturb(a, rng)
                a = _project_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Local restart (stagnation) – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 7) Return the best sequence found (clipped to the limits of
    #    ``evaluate_sequence``).  The list is ready for the scoring harness.
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
