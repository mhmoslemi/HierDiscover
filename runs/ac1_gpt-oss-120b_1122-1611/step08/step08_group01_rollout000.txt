```
def _simplex_proj(v):
    # projects v onto simplex (nonnegative, sum=1)
    # from scipy etc.
    # Implementation:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

```
def _targeted_descent(a, step_frac, fft_len_self):
    n = a.size
    conv = _self_conv(a, fft_len_self)
    c_max = conv.max()
    k_max = conv.argmax()
    # compute gradient g for c_k
    g = np.zeros_like(a)
    i_start = max(0, k_max - (n - 1))
    i_end = min(k_max, n - 1)
    i_range = np.arange(i_start, i_end + 1)
    j_range = k_max - i_range
    # accumulate contributions
    g[i_range] += a[j_range]
    g[j_range] += a[i_range]
    # note that when i_range == j_range (for even k_max), we added twice; actual gradient includes 2*a[i] so fine.
    # Now we want to move mass from high-g to low-g.
    src = np.argmax(g)
    dst = np.argmin(g)
    delta = step_frac * a[src]
    if delta <= 0: return a, c_max
    new_a = a.copy()
    new_a[src] -= delta
    new_a[dst] += delta
    # re-normalize (should keep sum constant because we moved)
    return new_a, conv # maybe conv not needed
```

The algorithm is a three‑phase optimiser that works directly on the simplex
(i.e. the sequence is always non‑negative and its entries sum to 1).

1. **Phase I – smoothed‑max Adam**  
   A smooth approximation of the `max` in the objective is built with a
   temperature γ that is slowly increased.  Adam updates are performed on the
   log‑softmax parameters so that the iterate always stays on the simplex.
   Stagnation is detected and the optimiser is re‑started with a small amount
   of noise.

2. **Phase II – targeted gradient descent on the true max**  
   The index `k*` where the convolution attains its maximum is found.
   The exact gradient of that convolution entry is cheap to compute
   (`O(n)`).  Mass is moved from the index with the largest gradient to the
   index with the smallest gradient (a projected‑gradient step).  The step
   size decays from 0.3 to 0.01, allowing coarse early moves and fine
   adjustments later.

3. **Phase III – occasional block perturbations & local restarts**  
   To escape possible traps a random *block* of consecutive entries is
   multiplied by a log‑normal factor, and a restart around the current best
   is performed if no improvement is seen for 30 s.

All phases respect a total wall‑clock budget of 1000 s (5 s are kept as a
safety margin).  The best sequence found so far is printed each time it
improves, and finally returned as a plain Python list.

```python
"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the bound on `evaluate_sequence`.  The optimiser combines

* a smoothed‑max Adam optimiser (Phase I),
* a targeted projected‑gradient descent that directly reduces the current
  maximal convolution entry (Phase II),
* occasional block‑perturbations and local restarts (Phase III).

The total wall‑clock budget is 1000 s (5 s safety margin).  The best
sequence discovered is returned as a list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a 1‑D real array using an FFT of length `fft_len`.
    Returns the first 2·len(a)‑1 entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate
        F(a) = Σ_k w_k (a*a)_k
    with respect to a.  `w` is a probability vector (Σ w = 1) over the
    2·n‑1 convolution positions.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max returning a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and re‑normalise.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _grad_of_conv_at_k(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact gradient of the k‑th convolution coefficient with respect to `a`.
    Returns a vector `g` where g[i] = a[k‑i] (if the partner index exists)
    plus the symmetric term.
    """
    n = a.size
    g = np.zeros_like(a)
    i_start = max(0, k - (n - 1))
    i_end = min(k, n - 1)
    i = np.arange(i_start, i_end + 1, dtype=np.int64)
    j = k - i
    # accumulate contributions
    g[i] += a[j]
    g[j] += a[i]
    return g


def _objective(a: np.ndarray, fft_len_self: int) -> float:
    """
    Exact evaluation of the objective (without calling `evaluate_sequence`).
    The sequence is assumed to be non‑negative; sum may be any positive number.
    """
    n = a.size
    conv = _self_conv(a, fft_len_self)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

# Main optimisation routine
# ----------------------------------------------------------------------

    Return a list of non‑negative heights (the best found) that minimises
    `evaluate_sequence` before the time budget elapses.

        if s > 0.0:

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

            v.fill(0.0)
            t = 0

    # 4) Phase II – targeted gradient descent on the true maximum
    # --------------------------------------------------------------
    cur = best.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    phase2_start = time.monotonic()
    phase2_total = deadline - phase2_start

    def _step_fraction(elapsed):
        """Linear decay from 0.30 to 0.01."""
        return 0.30 * (1.0 - elapsed / phase2_total) + 0.01

        elapsed = now - phase2_start

            cand = cur.copy()
            cand[src] -= delta
            cand[dst] += delta
            cand_score = _objective(cand, fft_len_self)

                cur_score = cand_score

            block_score = _objective(block, fft_len_self)

                cur = block
                cur_score = block_score
                if block_score < best_score - 1e-12:
                    best_score = block_score
                    best = block.copy()
```
