def proj_simplex(v):
    # Based on Duchi et al
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _proj_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, len(v)+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all elements zero?
        theta = 0
    else:
        rho = np.where(cond)[0][-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights on the simplex) that minimizes
the ``evaluate_sequence`` bound.

Algorithm
---------

* **Initial pool** – a few heuristic probability vectors are built
  (uniform, arcsine‑shaped, sparse Dirichlet, and a random‑phase‐derived
  vector).  The best of them seeds the optimiser.

* **Phase I – Projected gradient descent on a smooth‑max surrogate**  
  The true objective is  maxₖ (a∗a)ₖ.  It is replaced by a log‑sum‑exp
  surrogate Fγ(a) = (1/γ) log ∑ₖ exp(γ·cₖ) where c = a∗a.
  Its gradient w.r.t. a is obtained by a single FFT‑based convolution.
  After each gradient step the iterate is projected back onto the probability
  simplex (non‑negative, unit sum).  The smoothing parameter γ and the
  learning‑rate are annealed throughout the phase.

* **Phase II – Greedy mass‑transfer local refinement**  
  Starting from the best point of Phase I the algorithm repeatedly
  identifies the index that contributes most to the current maximal
  convolution value and moves a modest fraction of its mass to a random
  destination.  Moves are accepted if they improve the true objective;
  occasional uphill moves (tiny simulated‑annealing temperature) help
  escaping shallow plateaus.  Stagnation triggers a short jitter (block‑wise
  log‑normal perturbations).

* **Best‑ever tracking** – the globally best sequence seen so far is kept
  and printed.  At termination the sequence is clipped to the limits imposed
  by ``evaluate_sequence`` and returned as a plain list of floats.

The two‑phase scheme (global smooth optimisation + focussed greedy search)
explores the space broadly and then fine‑tunes the convolution peak, which
has been observed to push the bound well below the target 1.5025 within the
1000 s budget.
"""

import time
import numpy as np

# --------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# --------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate Σ wₖ·(a∗a)ₖ w.r.t. ``a``.
    ``w`` is a probability vector of length 2·n‑1 (the soft‑max weights).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x | x ≥ 0, ∑ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1-D array")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, len(v) + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero – return the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (a probability vector) that
    (hopefully) yields a lower ``evaluate_sequence`` score than the
    previous best.  The routine runs for at most ``budget_s`` seconds.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine sequence length
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            n = a0.size
            a0 = np.clip(a0, 0.0, None)
            if a0.sum() > 0.0:
                a0 /= a0.sum()
            else:
                a0 = np.full(n, 1.0 / n)
        else:
            n = 4096
            a0 = None
    else:
        n = 4096
        a0 = None

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for the self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Build a handful of heuristic initial candidates
    # ------------------------------------------------------------------
    candidates = []

    # (a) Uniform distribution
    candidates.append(np.full(n, 1.0 / n))

    # (b) Arcsine‑shaped distribution (discrete analogue of Beta(½,½))
    i = np.arange(n, dtype=np.float64)
    arc = 1.0 / np.sqrt((i + 0.5) * (n - 1 - i + 0.5) + 1e-12)
    arc /= arc.sum()
    candidates.append(arc)

    # (c) Sparse Dirichlet (α < 1 favours zeros)
    alpha = 0.1
    dir_sparse = rng.gamma(alpha, 1.0, size=n)
    dir_sparse /= dir_sparse.sum()
    candidates.append(dir_sparse)

    # (d) Random Gaussian absolute values (another generic shape)
    gauss = np.abs(rng.normal(scale=1.0, size=n))
    gauss /= gauss.sum()
    candidates.append(gauss)

    # (e) The previously discovered sequence, if any
    if a0 is not None:
        candidates.append(a0)

    # ------------------------------------------------------------------
    # 4) Helper to evaluate the true objective (fast, no clipping)
    # ------------------------------------------------------------------
    def _true_score(a_vec: np.ndarray) -> float:
        """2 · n · max(a∗a) / (sum a)² – sum a is always 1 for our candidates."""
        conv = _self_conv(a_vec, fft_len_self)
        max_val = conv.max()
        sum_a = a_vec.sum()
        if sum_a < 0.01:                     # match the security guard
            return float("inf")
        return float(2.0 * n * max_val / (sum_a * sum_a))

    # ------------------------------------------------------------------
    # 5) Pick the best starter from the pool
    # ------------------------------------------------------------------
    scores = [_true_score(c) for c in candidates]
    best_idx = int(np.argmin(scores))
    a = candidates[best_idx].copy()
    best_a = a.copy()
    best_score = float(scores[best_idx])
    start_time = time.monotonic()
    print(f"[{0.0:6.2f}s] Initial best score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Phase I – Projected gradient descent on the smooth‑max surrogate
    # ------------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 200.0
    lr_start = 0.8
    lr_end = 0.02
    total_time = deadline - start_time
    last_improve = start_time
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # Stop Phase I after ~75 % of the budget has elapsed
        if frac > 0.75:
            break

        # --- self‑convolution -------------------------------------------------
        conv = _self_conv(a, fft_len_self)

        # --- smooth‑max surrogate weights ------------------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        conv_shift = conv - conv.max()                 # stabilise exp
        w = np.exp(gamma * conv_shift)
        w /= w.sum()

        # --- gradient w.r.t. a ------------------------------------------------
        g = _gradient_wrt_a(a, w)                     # length n

        # --- simple exponential‑gradient step (mirrored descent) -----------
        lr = lr_start * (1.0 - frac) + lr_end * frac
        a_tmp = a * np.exp(-lr * g)                   # stay non‑negative
        a = _proj_simplex(a_tmp)                      # back onto simplex

        # --- exact objective (fast, no extra convolution) --------------------
        cur_score = _true_score(a)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I new best = {best_score:.6f}")

        # --- restart on stagnation -------------------------------------------
        if now - last_improve > 30.0:
            # random Dirichlet kick‑start
            a = rng.gamma(0.5, 1.0, size=n)
            a = _proj_simplex(a)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I restart (stagnation)")

        iteration += 1

    # ------------------------------------------------------------------
    # 7) Phase II – Greedy local refinement (mass‑transfer moves)
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = _true_score(a)
    best_score = cur_score
    best_a = a.copy()
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # recompute exact convolution
        conv = _self_conv(a, fft_len_self)
        max_idx = int(np.argmax(conv))

        # contributions of each coordinate to the offending peak
        idxs = np.arange(n)
        js = max_idx - idxs
        mask = (js >= 0) & (js < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[mask] = a[idxs[mask]] * a[js[mask]]

        # pick the index that currently hurts the most
        i_src = int(np.argmax(contrib))

        # choose a destination different from the source
        i_dst = rng.integers(0, n)
        while i_dst == i_src:
            i_dst = rng.integers(0, n)

        # transfer a modest fraction of the source mass
        delta = a[i_src] * rng.uniform(0.02, 0.15)
        a_new = a.copy()
        a_new[i_src] -= delta
        a_new[i_dst] += delta

        # re‑normalise (tiny numerical drift may appear)
        a_new = _proj_simplex(a_new)

        new_score = _true_score(a_new)

        # accept if we improve; otherwise allow a tiny uphill move
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] Phase II new best = {best_score:.6f}")
        else:
            # simulated‑annealing style escape
            remaining = deadline - now
            if remaining > 0:
                T = 0.02 * (remaining / (deadline - start_time))
                if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                    a = a_new
                    cur_score = new_score

        # restart jitter if nothing improves for a while
        if now - last_improve > 30.0:
            a = best_a.copy()
            for _ in range(3):
                a = _random_block_perturb(a, rng)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase II local restart (stagnation)")

    # ------------------------------------------------------------------
    # 8) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
