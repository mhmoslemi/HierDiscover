"""
Two‑phase optimizer for the step‑function problem (improved).

Phase 1 – Very cheap global exploration:
    * Enumerate a low‑dimensional family of “edge‑heavy’’ shapes
      (two blocks at the ends, optionally linearly tapered).
    * Supplement with a few random Dirichlet draws and cosine‑shapes.
    * Keep the best few candidates.

Phase 2 – Local refinement:
    * From each seed run a projected Adam optimiser on the smooth
      soft‑max surrogate of the max‑convolution.
    * Learning‑rate and temperature follow an exponential schedule.
    * Occasional jitter + restart on stagnation keep the search from
      being trapped.

All vectors are kept on the probability simplex (sum = 1);
the final sequence is obtained by scaling with √(2 n) before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _cosine_shape(n: int, r: float, power: float) -> np.ndarray:
    """Symmetric cosine shape: (1 + r*cos(theta))**power, normalised."""
    theta = np.pi * np.arange(n) / (n - 1)
    v = (1.0 + r * np.cos(theta)) ** power
    return v / v.sum()


def _edge_shape(n: int, L: int, gamma: float) -> np.ndarray:
    """
    Two edge blocks of length L with linear taper controlled by gamma.
    When gamma==0 the blocks are uniform; gamma==1 makes them drop to
    zero at the inner border.
    """
    if L == 0:
        return np.full(n, 1.0 / n)
    v = np.zeros(n, dtype=np.float64)
    # left block
    if L > 1:
        idx = np.arange(L)
        v[idx] = 1.0 - gamma * (idx / (L - 1))
    else:
        v[0] = 1.0
    # right block (mirrored)
    if L > 1:
        idx = np.arange(L)
        v[-L + idx] = 1.0 - gamma * (idx[::-1] / (L - 1))
    else:
        v[-1] = 1.0
    # clip negative values (should not happen for 0≤gamma≤1)
    v = np.clip(v, 0.0, None)
    if v.sum() == 0.0:
        return np.full(n, 1.0 / n)
    return v / v.sum()


def _softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Temperature‑controlled soft‑max weights (sum to 1)."""
    c_max = conv.max()
    shifted = conv - c_max          # improve numerical stability
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / w_sum


def _grad_softmax(a: np.ndarray, w: np.ndarray, fft_len_grad: int) -> np.ndarray:
    """Gradient of the soft‑max surrogate w.r.t. a (length‑n vector)."""
    # w_rev is reversed soft‑max weights, needed for convolution
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    # extracting the part that aligns with a (see original code)
    start_idx = w_rev.size - 1          # = 2n‑2
    return 2.0 * full[start_idx:start_idx + a.size]


def _objective(p: np.ndarray, fft_len_self: int, n: int) -> float:
    """
    Returns the true objective J = 2·n·max(p*p) for a probability vector p.
    """
    conv = _self_conv(p, fft_len_self)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  Two‑phase method: cheap global sweep over a few
    low‑dimensional families, followed by Adam‑based local refinement on the
    smooth‑max surrogate.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = min(max(n, 256), 4096)   # keep FFT cheap
        except Exception:
            n = 4096
    else:
        n = 4096

    fft_len_self = _next_pow2(2 * n - 1)
    fft_len_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Global cheap exploration (parameter sweep & random draws)
    # --------------------------------------------------------------
    best_candidates = []                 # list of (J, p) tuples
    def _push_candidate(p_vec: np.ndarray):
        """evaluate candidate and keep top‑5."""
        J = _objective(p_vec, fft_len_self, n)
        best_candidates.append((J, p_vec.copy()))
        best_candidates.sort(key=lambda x: x[0])
        if len(best_candidates) > 5:
            best_candidates.pop()   # keep only 5 best

    # a) uniform baseline
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _push_candidate(uniform)

    # b) systematic edge‑shape grid (L fraction & taper)
    for frac in np.linspace(0.18, 0.45, 14):          # block‑size fraction
        L = int(frac * n)
        if L == 0 or 2 * L > n:
            continue
        for gamma in np.linspace(0.0, 1.0, 11):
            p = _edge_shape(n, L, gamma)
            _push_candidate(p)

    # c) cosine‑family grid
    for r in np.linspace(0.0, 1.0, 13):
        for pw in np.linspace(0.3, 3.0, 9):
            p = _cosine_shape(n, r, pw)
            _push_candidate(p)

    # d) a handful of random Dirichlet draws (moderately concentrated)
    for _ in range(30):
        alpha = rng.uniform(0.05, 0.6)
        p = rng.dirichlet(np.full(n, alpha))
        _push_candidate(p)

    # d2) include the previous best sequence if present
    if "height_sequence_1" in globals():
        try:
            prev = np.asarray(height_sequence_1, dtype=np.float64)
            # normalise to probability simplex
            prev = np.clip(prev, 0.0, None)
            if prev.sum() == 0:
                prev = uniform
            else:
                prev = proj_simplex(prev / prev.sum())
            _push_candidate(prev)
        except Exception:
            pass

    # Keep the best candidate found so far
    best_candidates.sort(key=lambda x: x[0])
    best_J, best_p = best_candidates[0]

    # --------------------------------------------------------------
    # 3) Local refinement – projected Adam on soft‑max surrogate
    # --------------------------------------------------------------
    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # schedule parameters (exponential growth for temperature)
    lr_start, lr_end = 0.5, 0.001
    alpha_start, alpha_end = 5.0, 3000.0               # soft‑max temperature

    # function to run Adam from a given start point for a limited time
    def _adam_refine(p_start: np.ndarray, time_limit: float):
        nonlocal best_J, best_p
        a = p_start.copy()        # works on probability vector
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0
        last_improve = time.monotonic()
        start = time.monotonic()
        while time.monotonic() < time_limit:
            it += 1
            now = time.monotonic()
            frac = (now - start) / (time_limit - start)   # ∈[0,1]

            # learning‑rate schedule (exponential decay)
            lr = lr_start * (0.5 ** frac) + lr_end * (1 - 0.5 ** frac)

            # temperature schedule (slow start → fast increase)
            alpha = alpha_start * (1.0 - frac) + alpha_end * frac

            # --- convolution & soft‑max weights ---
            conv = _self_conv(a, fft_len_self)
            w = _softmax_weights(conv, alpha)

            # --- gradient of the smoothed surrogate ---
            grad = _grad_softmax(a, w, fft_len_grad)

            # --- Adam update & simplex projection ---
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = proj_simplex(a)

            # --- evaluate true objective (no smoothing) ---
            cur_J = _objective(a, fft_len_self, n)

            if cur_J < best_J - 1e-12:
                best_J = cur_J
                best_p = a.copy()
                last_improve = now
                print(f"[{now - start:.1f}s] Adam new best J = {best_J:.6f}")

            # --- jitter occasionally to escape plateaus ---
            if rng.random() < 0.02:
                # tiny log‑normal multiplicative noise on a random block
                block_width = rng.integers(1, max(2, n // 20))
                start_idx = rng.integers(0, n - block_width + 1)
                factor = np.exp(rng.normal(0.0, 0.07))
                a_pert = a.copy()
                a_pert[start_idx:start_idx + block_width] *= factor
                a_pert = proj_simplex(a_pert)
                a = a_pert

            # --- restart on stagnation ---
            if now - last_improve > 20.0:
                a = best_p.copy()
                # inject tiny unbiased noise
                noise = np.exp(rng.normal(0.0, 0.05, size=n))
                a = a * noise
                a = proj_simplex(a)
                last_improve = now
                print(f"[{now - start:.1f}s] restart (stagnation) – best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Run Adam from the top few seeds while respecting the budget
    # --------------------------------------------------------------
    # allocate roughly 80 % of remaining time to refinement
    remaining = deadline - time.monotonic()
    refine_budget = remaining * 0.80
    per_seed = refine_budget / max(1, len(best_candidates))

    for J0, p0 in best_candidates:
        if time.monotonic() >= deadline:
            break
        time_limit = min(per_seed, deadline - time.monotonic())
        _adam_refine(p0, time_limit)

    # --------------------------------------------------------------
    # 5) Return the best sequence found (scaled to satisfy the original
    #    evaluation’s sum‑normalisation √(2 n))
    # --------------------------------------------------------------
    scale = np.sqrt(2.0 * n)                     # required L1‑norm
    best_seq = best_p * scale
    # Clip to the limits enforced inside evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(x) for x in best_seq]
