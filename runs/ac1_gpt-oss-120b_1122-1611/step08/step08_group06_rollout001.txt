"""
propose_candidate – Adam‑based optimisation of a non‑negative step function.

The objective to minimise is

    2·n·max(conv(a,a)) / (sum(a)²)

Because the expression is scale‑invariant we may enforce sum(a)=1 and
simply minimise  max(conv(a,a)).  The algorithm works on the soft‑max
parameterisation  a = softmax(x)  so the feasibility constraints are
handled automatically.

Three stages are employed:

1. **Variance‑flattening** – minimise the L² deviation of the
   self‑convolution from its average value 1/(2·n‑1).  This drives the
   convolution towards a flat shape.

2. **Smoothed‑max phase** – minimise a smooth approximation of the
   maximum:
        Lγ(a) = (1/γ)·log Σ exp(γ·cₖ)   with  c = a∗a .
   The smoothing parameter γ is increased linearly from 0.5 up to 200,
   so the optimisation gradually concentrates on the true peak.

3. **Hard‑max phase** – after γ has become large the loss is replaced
   by the exact max (a one‑hot weight at the current maximal index).
   A few final Adam steps sharpen the remaining peak.

Adam’s learning‑rate decays linearly from 0.5 to 0.001.
If no improvement is observed for 15 s a stochastic restart from the
current best solution (Gaussian perturbation) is performed; occasional
random block jitter helps escaping shallow minima.

The routine respects a user supplied time‑budget (default 1000 s) and
returns the best sequence discovered before the deadline.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σₖ wₖ·(a*a)ₖ with respect to a.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    # block width between 1 and max(2, n//10)
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max, returns a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    metric.  Returns the best sequence discovered before the time budget
    expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # ------------------------------------------------------------------
    # 1) Choose a length and initialise log‑weights
    # ------------------------------------------------------------------
    # Target length – a bit larger than the previous construction
    # to give the optimiser more degrees of freedom.
    base_len = 8000
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        init_arr = np.asarray(init_seq, dtype=np.float64).ravel()
    else:
        init_arr = None

    n = base_len
    if init_arr is not None and init_arr.size > 0:
        # Pad / truncate the provided sequence to length n
        a0 = np.zeros(n, dtype=np.float64)
        take = min(init_arr.size, n)
        a0[:take] = np.clip(init_arr[:take], 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0[:] = 1.0 / n
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # Log‑weights for the soft‑max parametrisation
    x = np.log(a0 + 1e-12)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Adam state (shared across phases)
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # ------------------------------------------------------------------
    # 4) Best‑solution tracking
    # ------------------------------------------------------------------
    a = _softmax(x)
    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * c.max()          # sum(a) == 1 ⇒ denominator = 1
    last_improve = time.monotonic()

    # ------------------------------------------------------------------
    # 5) Phase schedule (fractions of the total run‑time)
    # ------------------------------------------------------------------
    total_time = deadline - start_time
    # 40 % variance, 45 % smoothed‑max, 15 % hard‑max
    t_var_end = start_time + 0.40 * total_time
    t_soft_end = start_time + 0.85 * total_time   # end of smoothed‑max
    # gamma will increase linearly from 0.5 to 200 during the smoothed phase
    gamma_start = 0.5
    gamma_end = 200.0

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ------------------------------------------------------------------
        # current a and its self‑convolution
        # ------------------------------------------------------------------
        a = _softmax(x)                      # probability vector, sum = 1
        c = _self_conv(a, fft_len_self)      # length 2·n‑1

        # ------------------------------------------------------------------
        # evaluate exact score (the quantity we finally care about)
        # ------------------------------------------------------------------
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # Choose loss / gradient according to the current phase
        # ------------------------------------------------------------------
        if now < t_var_end:
            # ------- Phase 1 – variance flattening -------
            avg = 1.0 / (2 * n - 1)                     # average convolution value
            r = c - avg                                 # residual vector (len 2n‑1)
            g_a = 2.0 * _gradient_wrt_a(a, r)           # gradient of Σ(r²)
        elif now < t_soft_end:
            # ------- Phase 2 – smooth approximation of max -------
            phase_frac = (now - t_var_end) / (t_soft_end - t_var_end)
            gamma = gamma_start + (gamma_end - gamma_start) * phase_frac
            # stable soft‑max on the convolution values
            c_shift = c - np.max(c)                     # avoid overflow
            w = np.exp(gamma * c_shift)
            w_sum = w.sum()
            if w_sum == 0.0:        # extreme under‑flow – fall back to uniform
                w = np.full_like(c, 1.0 / c.size)
            else:
                w /= w_sum
            g_a = _gradient_wrt_a(a, w)                 # gradient of smooth max
        else:
            # ------- Phase 3 – hard max (sub‑gradient) -------
            max_idx = int(np.argmax(c))
            w = np.zeros_like(c)
            w[max_idx] = 1.0
            g_a = _gradient_wrt_a(a, w)                 # gradient of exact max

        # ------------------------------------------------------------------
        # Chain rule: gradient w.r.t. the soft‑max logits x
        # ------------------------------------------------------------------
        avg_g = np.dot(a, g_a)                  # ⟨a, ∂F/∂a⟩
        g_x = a * (g_a - avg_g)                 # Jacobian of soft‑max

        # Adam update
        # ------------------------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)

        m_hat = m / (1 - beta1 ** t_iter)
        v_hat = v / (1 - beta2 ** t_iter)

        # Linear learning‑rate decay (0.5 → 0.001)
        lr_start = 0.5
        lr_end = 0.001
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ------------------------------------------------------------------
        # Stagnation handling – restart if no improvement for 15 s
        # ------------------------------------------------------------------
        if now - last_improve > 15.0:
            noise = rng.normal(0.0, 0.3, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # ------------------------------------------------------------------
        # Occasional global jitter to escape shallow minima
        # ------------------------------------------------------------------
        if rng.random() < 0.01:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0

    # ------------------------------------------------------------------
    # 7) Return the best sequence (clipped to evaluate_sequence limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best.tolist()]
