def propose_candidate(seed=12345, budget_s=1000.0):
    import time, numpy as np
    rng = np.random.default_rng(seed)

    # Determine n:
    if 'height_sequence_1' in globals():
        try:
            n = len(globals()['height_sequence_1'])
            n = min(max(n, 512), 4096)
        except:
            n = 4096
    else:
        n = 4096

    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)

    # Helper evaluate J
    def compute_J(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
        return float(2.0 * n * conv.max()), conv

    # starting candidates:
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # exponential decays
    for r in [0.9, 0.95, 0.97, 0.99]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        candidates.append(v)

    # random Dirichlet draws
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        candidates.append(v)

    # previous best if exists
    if 'height_sequence_1' in globals():
        prev = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        prev_sum = prev.sum()
        if prev_sum > 0:
            prev = proj_simplex(prev)
            candidates.append(prev)

    # maybe more random
    for _ in range(5):
        v = rng.random(n)
        v /= v.sum()
        candidates.append(v)

    best_score = float('inf')
    best_vec = None

    deadline = time.monotonic() + max(0.0, budget_s - 5.0)

    # loop through candidates, allocate time proportionally
    cand_idx = 0
    while time.monotonic() < deadline and cand_idx < len(candidates):
        a = candidates[cand_idx].copy()
        cand_idx += 1

        # local refinement
        # compute current score:
        cur_score, conv = compute_J(a)
        if cur_score < best_score:
            best_score = cur_score; best_vec = a.copy()
            print(f"Init candidate {cand_idx} J={cur_score:.6f}")

        # iteration loop
        it = 0
        last_improve = time.monotonic()
        start_local = time.monotonic()
        while time.monotonic() < deadline:
            it += 1
            # compute conv and max index
            fa = np.fft.rfft(a, n=fft_len)
            conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
            max_val = conv.max()
            k = int(conv.argmax())
            J = float(2.0 * n * max_val)

            # update best
            if J < best_score - 1e-12:
                best_score = J
                best_vec = a.copy()
                last_improve = time.monotonic()
                elapsed_total = time.monotonic() - start_local
                print(f"[{elapsed_total:6.2f}s] cand {cand_idx} best J={best_score:.6f}")

            # gradient for max k
            start_i = max(0, k - (n - 1))
            end_i = min(n - 1, k)
            idx = np.arange(start_i, end_i + 1)
            g = np.zeros_like(a)
            g[idx] = 2.0 * a[k - idx]

            # step size schedule:
            total_time = deadline - start_local

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec.tolist()]

"""
Propose a step‑function that minimises ``evaluate_sequence``.

Algorithm
---------
1. Determine the problem size ``n`` (use the length of the
   previously found sequence if available, otherwise fall back to 4096).
2. Build a small portfolio of diverse start vectors, all normalised to
   sum = 1 (probability simplex).  The portfolio contains the previous
   best solution, a uniform vector, a few exponential decays and several
   random Dirichlet draws.
3. For each start vector run a *projected sub‑gradient* optimiser that
   directly minimises the quantity

        J(a) = 2·n·max_k (a*a)_k ,

   i.e. the maximum entry of the self‑convolution.  Because the sum of
   the coefficients is fixed to 1, this is exactly the objective used
   by ``evaluate_sequence``.

   * At each iteration the convolution is obtained by an FFT (O(n log n)).
   * The index `k*` where the convolution attains its maximum is located.
   * The sub‑gradient of the max‑function at `k*` is
        g_i = 2·a_{k*‑i}   (if 0 ≤ k*‑i < n, else 0).
   * An additive update ``a ← a – η·g`` is performed and the result is
     projected back onto the simplex with the Euclidean projection.
   * A simple step‑size schedule η∈[0.5, 0.001] (linear decay) and a
     restart‑with‑noise mechanism (if no improvement for 20 s) are used
     to escape plateaus.
4. The best vector found over all starts is returned (clipped to the
   limits enforced by ``evaluate_sequence``).  The whole routine respects
   a 1000 s time budget (5 s safety margin).

The method is deliberately simple, fast (≈ 10 µs per FFT for n≈4000) and
does not rely on external optimisation libraries.  It improves on the
previous DE + Adam pipeline by optimising the true non‑smooth objective
with a sub‑gradient method that directly targets the current worst
convolution entry.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _gradient_at_max(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of ``max_k (a*a)_k`` at the maximizing index ``k``.
    Returns a vector ``g`` such that moving opposite to ``g`` reduces the
    current maximum entry.
    """
    n = a.size
    g = np.zeros_like(a)
    start_i = max(0, k - (n - 1))
    end_i = min(n - 1, k)
    # indices i for which a_{k-i} is defined
    idx = np.arange(start_i, end_i + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def _conv_max_fft(a: np.ndarray, fft_len: int, conv_len: int) -> tuple[float, int, np.ndarray]:
    """
    Compute the self‑convolution by FFT, return the maximum value,
    its index and the full convolution array (trimmed to length ``conv_len``).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    max_idx = int(conv.argmax())
    max_val = float(conv[max_idx])
    return max_val, max_idx, conv


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine builds several initial candidates,
    refines each of them with a projected sub‑gradient method that directly
    targets the maximum entry of the self‑convolution, and finally returns
    the best vector found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem dimension
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = min(max(n, 512), 4096)          # force 512 ≤ n ≤ 4096
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)               # length for a*a via FFT

    # ------------------------------------------------------------------
    # 2) Helper to evaluate the true objective (sum is forced to 1)
    # ------------------------------------------------------------------
    def _objective(a_vec: np.ndarray) -> float:
        """Return J = 2·n·max(conv) for a simplex‑normalised vector."""
        max_val, _, _ = _conv_max_fft(a_vec, fft_len, conv_len)
        return 2.0 * n * max_val

    # ------------------------------------------------------------------
    # 3) Build a small diverse portfolio of start vectors (all sum to 1)
    # ------------------------------------------------------------------
    start_candidates = []

    # uniform
    start_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few exponential decays
    for r in (0.99, 0.975, 0.95, 0.9, 0.85, 0.8):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        start_candidates.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    start_candidates.append(v)

    # Dirichlet draws (moderately concentrated)
    for _ in range(6):
        v = rng.dirichlet(np.ones(n) * 0.3)
        start_candidates.append(v.astype(np.float64))

    # previous best if present
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        if prev.sum() == 0.0:
            prev = np.full(n, 1.0 / n)
        else:
            prev = proj_simplex(prev)
        start_candidates.append(prev)

    # ------------------------------------------------------------------
    # 4) Global variables keeping track of the best solution found
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_vec = None

    # ------------------------------------------------------------------
    # 5) Main loop – refine each start vector until time runs out
    # ------------------------------------------------------------------
    cand_idx = 0
    while time.monotonic() < deadline and cand_idx < len(start_candidates):
        a = start_candidates[cand_idx].copy()
        cand_idx += 1

        # Ensure simplex feasibility
        a = proj_simplex(a)

        # Quick evaluation of the start point
        cur_score = _objective(a)
        if cur_score < best_score:
            best_score = cur_score
            best_vec = a.copy()
            print(f"[init {cand_idx}] J = {best_score:.6f}")

        # ----------------------------------------------------------------
        # 5.1) Projected sub‑gradient descent on the max‑convolution
        # ----------------------------------------------------------------
        it = 0
        last_improve = time.monotonic()
        start_local = time.monotonic()

        while time.monotonic() < deadline:
            it += 1

            # --- compute convolution and locate the maximal entry ----------
            max_val, k_max, _ = _conv_max_fft(a, fft_len, conv_len)
            J = 2.0 * n * max_val

            # --- record improvement ----------------------------------------
            if J < best_score - 1e-12:
                best_score = J
                best_vec = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_local
                print(f"[{elapsed:6.2f}s] cand {cand_idx} new best J = {best_score:.6f}")

            # --- sub‑gradient of the active max entry ----------------------
            g = _gradient_at_max(a, k_max)

            # --- step‑size schedule (linear decay) -------------------------
            total = deadline - start_local
            elapsed = time.monotonic() - start_local
            frac = elapsed / total
            eta = 0.5 * (1.0 - frac) + 0.001 * frac    # 0.5 → 0.001

            # --- additive update followed by projection onto simplex --------
            a = a - eta * g
            a = proj_simplex(a)

            # --- occasional random jitter to escape tiny plateaus ----------
            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)

            # --- restart if no improvement for a while --------------------
            if time.monotonic() - last_improve > 20.0:
                # small multiplicative noise, then renormalise
                noise = np.exp(rng.normal(0.0, 0.03, size=n))
                a = a * noise
                a = proj_simplex(a)
                last_improve = time.monotonic()
                elapsed = last_improve - start_local
                print(f"[{elapsed:6.2f}s] cand {cand_idx} restart (stagnation) – J = {best_score:.6f}")

        # end of refinement for this start vector

    # ------------------------------------------------------------------
    # 6) Return the best vector (clipped to the limits enforced by
    #    ``evaluate_sequence`` – values >1000 are truncated there anyway)
    # ------------------------------------------------------------------
    if best_vec is None:
        # fallback – return a uniform vector (should never happen)
        best_vec = np.full(n, 1.0 / n, dtype=np.float64)

    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec.tolist()]
