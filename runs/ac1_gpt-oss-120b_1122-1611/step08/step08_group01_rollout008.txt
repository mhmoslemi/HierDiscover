"""
Simulated‑Annealing search for a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

(the same quantity that ``evaluate_sequence`` returns).  The algorithm

* works on the simplex (the vector is kept normalised to sum = 1);
* stores the best vector seen so far;
* repeatedly proposes simple stochastic mutations:
    – move a small amount of mass between two random positions;
    – multiply a random contiguous block by a log‑normal factor;
    – a combination of both;
* accepts a candidate if it improves the exact objective or, with a
  temperature‑dependent probability, if it is only slightly worse.
  The temperature is exponentially cooled from a modest start to a tiny
  final value, allowing the search to escape shallow local minima early
  and to fine‑tune later.
* stops a few seconds before the allotted budget (default 1000 s) and
  returns the best sequence discovered.

All heavy work (self‑convolution and max‑extraction) is done with an
FFT of length a power of two, which makes a single evaluation O(n log n)
and fast enough for thousands of iterations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a 1‑D real array using an FFT of length
    ``fft_len``.  Returns an array of length ``2*len(a)‑1``.
    """
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    # keep only the useful part
    return conv[: 2 * a.size - 1]


def _move_mass(a: np.ndarray,
               rng: np.random.Generator,
               step_frac: float) -> np.ndarray:
    """
    Transfer a small fraction of mass from a random source index to a
    random destination index, then re‑normalise.
    """
    n = a.size
    src, dst = rng.integers(0, n, size=2)
    if src == dst or a[src] == 0.0:
        return a
    # amount to move – at most ``step_frac`` of the source weight
    delta = step_frac * rng.random() * a[src]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[src] -= delta
    out[dst] += delta
    # re‑normalise to avoid accumulation of rounding error
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _random_block_perturb(a: np.ndarray,
                          rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and
    re‑normalise.  Width is drawn from [1, max(2, n//10)].
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))  # modest jitter
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimise
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time handling – leave a safety margin for the harness
    # ------------------------------------------------------------------
    safety = 5.0
    deadline = time.monotonic() + max(0.0, float(budget_s) - safety)
    start_time = time.monotonic()
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1) Choose a working length ------------------------------------------------
    # Use the length of a previously discovered sequence if it looks
    # reasonable, otherwise fall back to a moderate size that still fits
    # into the time budget.
    # ------------------------------------------------------------------
    candidate_len = 4800                # a compromise between flexibility and speed
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 = a0 / s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            # adapt length if necessary
            if a0.size != candidate_len:
                if a0.size < candidate_len:
                    a0 = np.concatenate([a0, np.zeros(candidate_len - a0.size)])
                else:
                    a0 = a0[:candidate_len]
                a0 = a0 / a0.sum()
            a = a0
        else:
            a = np.full(candidate_len, 1.0 / candidate_len)
    else:
        a = np.full(candidate_len, 1.0 / candidate_len)

    n = a.size

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for the self‑convolution
    # ------------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Objective helper (exact – we use the same formula as evaluate_sequence)
    # ------------------------------------------------------------------
    def _objective(vec: np.ndarray) -> float:
        conv = _self_conv(vec, fft_len)
        return 2.0 * n * conv.max()

    cur_score = _objective(a)
    best = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    print(f"[{0.0:6.2f}s] start – n = {n}, initial score = {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing parameters
    # ------------------------------------------------------------------
    # temperature decays exponentially from a fraction of the initial
    # objective to a tiny final value.
    T_start = max(0.05, cur_score * 0.03)
    T_end = cur_score * 1e-4 + 1e-6

    def _temperature(frac: float) -> float:
        """Exponential cooling schedule."""
        return T_start * ((T_end / T_start) ** frac)

    # move‑size schedule – larger early, shrinking later
    def _step_fraction(frac: float) -> float:
        return 0.25 * (1.0 - frac) + 0.01

    # ------------------------------------------------------------------
    # 5) Main annealing loop
    # ------------------------------------------------------------------
    it = 0
    while time.monotonic() < deadline:
        elapsed = time.monotonic() - start_time
        frac = elapsed / total_time
        T = _temperature(frac)
        step_frac = _step_fraction(frac)

        # ---- propose a candidate -------------------------------------------------
        r = rng.random()
        if r < 0.55:
            cand = _move_mass(a, rng, step_frac)
        elif r < 0.85:
            cand = _random_block_perturb(a, rng)
        else:
            # a two‑stage move – block jitter followed by a small mass transfer
            cand = _random_block_perturb(a, rng)
            cand = _move_mass(cand, rng, step_frac * 0.5)

        cand_score = _objective(cand)

        # ---- acceptance test ----------------------------------------------------
        improve = cand_score < cur_score
        if improve or rng.random() < np.exp(-(cand_score - cur_score) / max(T, 1e-12)):
            a = cand
            cur_score = cand_score
            # record a new overall best
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = time.monotonic()
                print(f"[{elapsed:6.2f}s] new best = {best_score:.6f}")

        # ---- occasional diversification when stuck -------------------------------
        if time.monotonic() - last_improve > 30.0:
            # re‑initialise locally around the current best with a bit of noise
            noise = rng.normal(0.0, 0.4, size=n)
            perturbed = np.log(best + 1e-12) + noise
            a = np.exp(perturbed)
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective(a)
            last_improve = time.monotonic()
            print(f"[{elapsed:6.2f}s] diversification – score reset to {cur_score:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    print(f"Finished – best score = {best_score:.6f} after {it} iterations")
    return [float(v) for v in best]
