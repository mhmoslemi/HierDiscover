"""
Proposed optimizer for the “step‑function” evaluation problem.

Algorithm
---------
1.  Initialise a candidate on the probability simplex.
    • If a previous construction (``height_sequence_1``) exists we reuse it,
      otherwise a Dirichlet draw (≈ uniform) is used.
2.  Repeatedly perform a *projected gradient descent* on a smooth surrogate
    of the objective (soft‑max approx. of the max‑convolution).  The gradient
    is obtained analytically via FFT‑based convolution:
        ∇μ(a) = 2·(a * w_rev)  with w_i ∝ exp(α·(c_i – max(c))),
    where c = a∗a and α grows from a small value to a large one during the
    run, turning the smooth surrogate into the true max.
3.  Each descent step uses a simple back‑tracking line‑search (Armijo rule)
    and the iterate is projected back onto the simplex (Euclidean projection).
    The learning‑rate is adapted automatically: it is increased after a
    successful step and halved after a failure.
4.  If no improvement is seen for a while a tiny random *block perturbation*
    (log‑normal multiplicative noise) is applied to escape plateaus.
5.  The best vector found before the time budget expires is returned.

The method works directly on the full‑size vector (≈ 3800 entries) and
requires only a handful of FFTs per iteration, easily fitting into the
1000 s budget while typically improving the bound beyond the previous
DE + Adam result.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _initial_candidate(rng: np.random.Generator) -> np.ndarray:
    """
    Return an initial feasible vector (non‑negative, sum = 1).
    Preference is given to a previously discovered construction if it is
    present; otherwise a Dirichlet draw (≈ uniform) is used.
    """
    if "height_sequence_1" in globals():
        try:
            raw = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            # enforce a modest upper bound on length (the original code used ≤4096)
            max_len = 4096
            if raw.size > max_len:
                raw = raw[:max_len]
            elif raw.size < max_len // 2:      # allow a reasonable lower bound
                # pad with zeros to avoid too tiny a problem
                raw = np.concatenate([raw, np.zeros(max_len - raw.size)])
            # clip negatives (should not happen) and project onto simplex
            raw = np.clip(raw, 0.0, None)
            return proj_simplex(raw)
        except Exception:
            pass
    # fall back to a Dirichlet draw (α=0.5 gives some variability)
    n = 4096
    return proj_simplex(rng.dirichlet(np.full(n, 0.5)))


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine uses a projected gradient descent on a
    soft‑max surrogate of the max‑convolution objective, with an
    adaptive line‑search, temperature schedule and occasional stochastic
    perturbations to escape plateaus.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (keep a 5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem size & initial point
    # --------------------------------------------------------------
    a = _initial_candidate(rng)                     # on the simplex, sum = 1
    n = a.size

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)              # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)              # for a * w_rev
    start_idx = (fft_len_grad - conv_len) // 2       # = w_rev.size‑1 (2n‑2)

    # --------------------------------------------------------------
    # Helper: true objective J (no clipping, scaling‑invariant)
    # --------------------------------------------------------------
    def compute_J(vec: np.ndarray) -> float:
        fa = np.fft.rfft(vec, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        return float(2.0 * n * conv.max())

    best_a = a.copy()
    best_J = compute_J(a)
    print(f"[init] J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 2) Optimisation parameters
    # --------------------------------------------------------------
    # learning‑rate handling – we use a back‑tracking line‑search, thus an
    # initial guess for the step size is kept and adapted dynamically.
    step = 0.5
    step_min = 1e-8
    step_max = 2.0

    # soft‑max temperature (α) schedule – start smooth, finish sharp.
    alpha_start = 2.0
    alpha_end = 4000.0

    # stagnation handling
    last_improve = time.monotonic()
    stagnation_limit = 30.0          # seconds without improvement → random jitter

    # --------------------------------------------------------------
    # 3) Main optimisation loop (projected gradient descent)
    # --------------------------------------------------------------
    iteration = 0
    while time.monotonic() < deadline:
        iteration += 1
        now = time.monotonic()
        frac = (now - (deadline - budget_s + 5.0)) / (budget_s - 5.0)  # ∈[0,1]

        # ---- temperature schedule (soft‑max sharpness) ----
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        # ---- compute convolution and soft‑max weights ----
        fa_self = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa_self * fa_self, n=fft_len_self)[:conv_len]

        c_max = conv.max()
        # shifted for numerical stability
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the smooth surrogate ----
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        grad = 2.0 * full[start_idx:start_idx + n]   # shape (n,)

        # ---------------------------------------------------------
        # line‑search: try to find a step that improves the true J
        # ---------------------------------------------------------
        improved = False
        step_try = step
        while step_try >= step_min:
            a_candidate = proj_simplex(a - step_try * grad)
            J_candidate = compute_J(a_candidate)

            if J_candidate < best_J - 1e-12:          # strict improvement
                improved = True
                break

            # no improvement → shrink step
            step_try *= 0.5

        if improved:
            a = a_candidate
            best_a = a.copy()
            best_J = J_candidate
            last_improve = now
            # allow step to grow a little after a successful move
            step = min(step_try * 1.3, step_max)
            elapsed = now - (deadline - budget_s + 5.0)
            print(f"[{elapsed:6.2f}s] iter {iteration:5d} new best J = {best_J:.6f}")
        else:
            # shrink step size and maybe apply a jitter if we are stuck
            step = max(step * 0.5, step_min)

        # ---------------------------------------------------------
        # occasional stochastic perturbation to escape plateaus
        # ---------------------------------------------------------
        if now - last_improve > stagnation_limit:
            a = _random_block_perturb(best_a, rng)
            step = max(step, 0.1)          # give the jitter a chance to move
            last_improve = now
            elapsed = now - (deadline - budget_s + 5.0)
            print(f"[{elapsed:6.2f}s] jitter (stagnation) – J remains {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(x) for x in best_a]
