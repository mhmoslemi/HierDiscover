def _project_to_simplex(v, s):
    # Duchi projection onto simplex {x >= 0, sum x = s}
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all zero? return uniform
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the bound obtained by the previous Adam‑+‑hill‑climbing code.

Algorithm (high‑level)
----------------------
1. Initialise a probability vector `a` from the global `height_sequence_1`
   (if it exists) – otherwise start from the uniform distribution.
   The vector is scaled to the canonical sum  S = √(2·n)  that makes the
   evaluate_sequence() value equal to the maximal self‑convolution value.
2. **Phase I – projected sub‑gradient descent**  
   * The exact index `k*` where the convolution attains its maximum is
     identified.
   * For that index the sub‑gradient of the objective w.r.t. `a` is
       g_i = 2·a_{k*‑i}  (zero when the index is out of range).
   * A step  a ← a – η·g  is taken and the result is projected back onto
     the simplex {x ≥ 0, Σx = S}.  The step size η decays slowly (∝1/√t).
   * Whenever a strictly better objective is found it is stored as the
     current best; after a long period without improvement a random
     block‑perturbation is applied to escape possible plateaus.
3. **Phase II – targeted mass moves**  
   * Starting from the best solution of Phase I a simple “move‑mass”
     operator is repeatedly applied: a small fraction of the mass is taken
     from the index with the largest gradient entry and given to the
     index with the smallest gradient entry.
   * The move is kept only if it decreases the exact objective.
   * Occasionally a larger random block perturbation is performed to
     explore new basins.
4. The best sequence found (clipped to the limits enforced by
   evaluate_sequence) is returned.

Both phases respect the 1000 s budget (a 5 s safety margin is kept for the
harness).  No external files or network access are used, only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a 1‑D array using an FFT of length ``fft_len``.
    The returned array has length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the simplex
    {x | x_i ≥ 0, Σ_i x_i = s}.
    Implementation of the algorithm from Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero – return uniform distribution
        return np.full_like(v, s / n)
    rho = ind[cond][-1] - 1            # zero‑based index
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(a: np.ndarray, max_index: int) -> np.ndarray:
    """
    Sub‑gradient of  max_k (a*a)_k  at the current maximal index.
    For the convolution index `max_index` the gradient is
        g_i = 2·a_{max_index‑i}   (zero if the index is out of range).
    """
    n = a.size
    g = np.zeros_like(a)
    i_start = max(0, max_index - (n - 1))
    i_end = min(n - 1, max_index)
    if i_start <= i_end:                     # there is at least one overlap
        idx = np.arange(i_start, i_end + 1)
        g[idx] = 2.0 * a[max_index - idx]
    return g


def _objective(a: np.ndarray, fft_len: int, s: float) -> float:
    """
    Exact value returned by evaluate_sequence() for a vector ``a`` that
    already satisfies Σa = s.
    """
    conv = _conv_self_fft(a, fft_len)
    max_b = conv.max()
    n = a.size
    # Since Σa = s the denominator cancels and the expression reduces to max_b
    return float(max_b)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and re‑normalise
    onto the simplex of radius √(2·n).
    """
    n = a.size
    # block width is at most 10 % of the sequence length
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.3))   # modest log‑normal jitter
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    # projection onto the simplex (the target sum will be supplied later)
    return out


def _move_mass(a: np.ndarray, rng: np.random.Generator, frac: float) -> np.ndarray:
    """
    Transfer a small fraction (≤ ``frac``) of the mass from the index with the
    largest gradient entry to the one with the smallest.
    The resulting vector is projected onto the simplex of the required sum.
    """
    n = a.size
    # Randomly pick source and destination – the caller decides the amount
    src = rng.integers(0, n)
    dst = rng.integers(0, n)
    if src == dst or a[src] == 0.0:
        return a
    delta = frac * rng.random() * a[src]
    out = a.copy()
    out[src] -= delta
    out[dst] += delta
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights (the best found) that minimises
    ``evaluate_sequence`` before the time budget elapses.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (reserve a few seconds for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose sequence length and initialise ``a``
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a = np.asarray(init_seq, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            raise ValueError("height_sequence_1 must be a non‑empty 1‑D array")
        a = np.clip(a, 0.0, None)
    else:
        # fallback: a moderately long vector (≈ 3800)
        a = None

    if a is None:
        n = 3800
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = a.size

    # canonical sum S = √(2·n)  (the scaling used in the reference paper)
    S = np.sqrt(2.0 * n)
    current_sum = a.sum()
    if current_sum <= 0.0:
        a = np.full_like(a, S / n)
    else:
        a = a * (S / current_sum)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length (power of two)
    # ------------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Phase I – projected sub‑gradient descent
    # ------------------------------------------------------------------
    # Sub‑gradient step parameters
    step0 = S * 0.6                # initial step magnitude
    max_iter = 200000             # hard cap (the time budget will stop earlier)
    best = a.copy()
    best_score = _objective(best, fft_len, S)
    last_improve = time.monotonic()
    iteration = 0

    print(f"[0.00s] start Phase I – n={n}, initial score={best_score:.6f}")

    while time.monotonic() < deadline and iteration < max_iter:
        # current convolution and maximal index
        conv = _conv_self_fft(a, fft_len)
        max_val = conv.max()
        k_star = int(conv.argmax())

        # sub‑gradient at the current maximal index
        g = _subgradient(a, k_star)

        # decaying step size
        eta = step0 / (1.0 + iteration / 3000.0)

        # gradient step + projection onto the simplex Σa = S
        a_candidate = a - eta * g
        a_candidate = _project_to_simplex(a_candidate, S)

        cand_score = _objective(a_candidate, fft_len, S)

        # keep the better of the two (greedy acceptance)
        if cand_score < best_score - 1e-12:
            best = a_candidate.copy()
            best_score = cand_score
            last_improve = time.monotonic()
            elapsed = last_improve - (deadline - budget_s)
            print(f"[{elapsed:6.2f}s] new best (Phase I) = {best_score:.6f}")

        if cand_score < max_val - 1e-12:
            a = a_candidate
        else:
            # occasional random jitter to avoid getting stuck
            if rng.random() < 0.01:
                a = _random_block_perturb(a, rng)
                a = _project_to_simplex(a, S)

        # restart after a long stagnation period
        if time.monotonic() - last_improve > 25.0:
            a = _random_block_perturb(best, rng)
            a = _project_to_simplex(a, S)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - (deadline - budget_s):6.2f}s] restart (stagnation)")

        iteration += 1

    # ------------------------------------------------------------------
    # 4) Phase II – targeted local hill‑climbing
    # ------------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    phase2_start = time.monotonic()
    total_phase2 = deadline - phase2_start

    def _step_fraction(elapsed):
        """Linearly decay the move fraction from 0.25 to 0.02."""
        return 0.25 * (1.0 - elapsed / total_phase2) + 0.02

    print(f"[{time.monotonic() - (deadline - budget_s):6.2f}s] start Phase II")

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        step_frac = _step_fraction(elapsed)

        # propose a small mass transfer
        cand = _move_mass(a, rng, step_frac)
        cand = _project_to_simplex(cand, S)

        cand_score = _objective(cand, fft_len, S)

        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                last_improve = now
                print(
                    f"[{now - (deadline - budget_s):6.2f}s] new best (Phase II) = {best_score:.6f}"
                )
            # reset stagnation timer
            last_improve = now

        # every few iterations try a larger block perturbation
        if rng.random() < 0.015:
            a_block = _random_block_perturb(a, rng)
            a_block = _project_to_simplex(a_block, S)
            block_score = _objective(a_block, fft_len, S)
            if block_score < cur_score:
                a = a_block
                cur_score = block_score
                if cur_score < best_score - 1e-12:
                    best = a.copy()
                    best_score = cur_score
                    last_improve = now
                    print(
                        f"[{now - (deadline - budget_s):6.2f}s] new best (block) = {best_score:.6f}"
                    )

        # if nothing improves for a while, perform a local restart
        if now - last_improve > 30.0:
            a = _random_block_perturb(best, rng)
            a = _project_to_simplex(a, S)
            cur_score = _objective(a, fft_len, S)
            last_improve = now
            print(
                f"[{now - (deadline - budget_s):6.2f}s] local restart – score {cur_score:.6f}"
            )

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to the limits imposed by
    #    evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    print(f"Finished – best score = {best_score:.6f}")
    return [float(v) for v in best]
