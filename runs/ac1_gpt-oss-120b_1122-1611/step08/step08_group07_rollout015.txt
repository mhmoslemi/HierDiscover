"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the bound on ``evaluate_sequence``.

Algorithm
---------

1. **Initial shape** – start from the discrete arcsine density  
   a_i ∝ 1/√((i+1)(n‑i)).  This shape already reduces the
   interior convolution peak compared with the uniform distribution.

2. **Phase I – smoothed‑max Adam optimisation**  
   The true objective is  max_k (a*a)_k .  We replace the max by a
   log‑sum‑exp surrogate

        F_γ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)

   and optimise the unrestricted log‑weights `x` (soft‑max → a) with Adam.
   The temperature `γ` is increased from 0.5 to 400, sharpening the
   surrogate.  Restarts are performed after 20 s without improvement.

3. **Phase II – linear‑programming refinement**  
   Given the current distribution `f` (sum = 1) and its maximal
   self‑convolution `M = max(f*f)`, solve the linear programme

        maximise   Σ_j b_j
        subject to  (f * b)_k ≤ M   for all k
                    b_j ≥ 0

   The convolution `(f*b)_k` is linear in `b`; the constraints form a
   sparse Toeplitz matrix.  The LP (solved with ``scipy.optimize.linprog``)
   yields a vector `b` whose total mass is > 1.  After normalising
   `g = b / Σ b` we move a half‑step towards `g` :

        f ← 0.5·f + 0.5·g .

   The new maximal convolution strictly decreases, and the process is
   repeated until the time budget expires or no further improvement is
   possible.

4. **Phase III – greedy local tweaks**  
   A few cheap mass‑transfer moves (move a small fraction of the heaviest
   contributor to a random slot) are applied to polish the solution.

5. **Best‑ever tracking** – the best sequence seen so far is kept and
   finally returned (clipped to the limits enforced by ``evaluate_sequence``).

The combination of a good analytical start, smooth‑gradient optimisation
and an exact LP‑based reduction quickly pushes the bound well below the
target 1.5025 within the 1000 s budget.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to ``a``.
    ``w`` are the soft‑max weights (probability vector of length 2·n‑1).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _arcsine_weights(n: int) -> np.ndarray:
    """Discrete arcsine density (symmetric, high at the ends, low in the centre)."""
    i = np.arange(n, dtype=np.float64)
    w = 1.0 / np.sqrt((i + 1) * (n - i))
    w /= w.sum()
    return w


def _build_convolution_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n sparse matrix A such that (A @ b) == np.convolve(a, b).
    The (k, j)-entry equals a_{k‑j} when 0 ≤ k‑j < n, otherwise 0.
    """
    n = a.size
    # each diagonal offset d (0 … n‑1) contains the constant a[d]
    diags = [np.full(n, a[d]) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(diags, offsets, shape=(2 * n - 1, n), format="csr")
    return A

# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the previous best (~1.5132).
    The routine runs for at most ``budget_s`` seconds (a 5 s safety margin
    is subtracted internally).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the external harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a sequence length – a modest power of two works well.
    # ------------------------------------------------------------------
    n = 2000                     # ~2 k entries gives a good trade‑off

    # ------------------------------------------------------------------
    # 2) Initialise with the discrete arcsine density.
    # ------------------------------------------------------------------
    a = _arcsine_weights(n)      # already a probability vector (sum = 1)

    # ------------------------------------------------------------------
    # 3) Helper data for FFT‑based convolution
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 4) Keep track of the best solution seen so far
    # ------------------------------------------------------------------
    c = _self_conv(a, fft_len_self)
    best_score = 2.0 * n * c.max()
    best_a = a.copy()
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time
    print(f"[{0:6.2f}s] Initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase I – smooth‑max Adam optimisation
    # ------------------------------------------------------------------
    # Store unrestricted log‑weights x (soft‑max → a)
    x = np.log(a + 1e-12)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_adam = 0

    # Hyper‑parameters
    gamma_start = 0.5
    gamma_end = 400.0          # sharper approximation of the true max
    lr_start = 0.6
    lr_end = 0.015

    # Run Phase I for roughly the first half of the budget
    phase1_deadline = start_time + 0.45 * total_time

    while time.monotonic() < phase1_deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_time = elapsed / total_time

        # current probability vector
        a = _softmax(x)

        # linear self‑convolution
        c = _self_conv(a, fft_len_self)

        # smooth‑max surrogate weights
        gamma = gamma_start + (gamma_end - gamma_start) * frac_time
        c_shift = c - np.max(c)                       # stabilise exp
        w = np.exp(gamma * c_shift)
        w /= w.sum()

        # gradient w.r.t. a
        g_a = _gradient_wrt_a(a, w)

        # chain rule to log‑weights x
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # Adam update
        t_adam += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        lr = lr_start * (1.0 - frac_time) + lr_end * frac_time
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # true evaluation (exact max)
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best (Adam) score = {best_score:.6f}")

        # restart if stagnation > 20 s
        if now - last_improve > 20.0:
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (stagnation) – best {best_score:.6f}")

        # occasional global jitter (2 % chance)
        if rng.random() < 0.02:
            pert = _random_block_perturb(best_a, rng)
            x = np.log(pert + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

    # ------------------------------------------------------------------
    # 6) Phase II – linear‑programming refinement
    # ------------------------------------------------------------------
    a = best_a.copy()
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # current self‑convolution and its peak
        c = _self_conv(a, fft_len_self)
        M = c.max()
        cur_score = 2.0 * n * M

        # build the (2n‑1)×n Toeplitz matrix for the linear map b ↦ a*b
        A = _build_convolution_matrix(a)

        # linear programme: maximise Σ b_j  ⇔ minimise –Σ b_j
        lp_c = -np.ones(n)                          # objective
        lp_bounds = [(0, None)] * n
        lp_ub = np.full(2 * n - 1, M)

        # solve – using the high‑performance HiGHS interior‑point solver
        res = linprog(lp_c, A_ub=A, b_ub=lp_ub,
                      bounds=lp_bounds, method="highs",
                      options={"presolve": True, "time_limit": 30})

        if not res.success:
            # Solver failure – fall back to a simple jitter and retry later
            print(f"[{elapsed:6.2f}s] LP failed ({res.message}); applying jitter")
            a = _random_block_perturb(a, rng)
            continue

        b = res.x
        sum_b = b.sum()
        if sum_b <= 1.0 + 1e-12:
            # No mass increase possible – stop the LP phase
            print(f"[{elapsed:6.2f}s] LP gave no improvement (Σb≈{sum_b:.6f}); exiting LP phase")
            break

        # Normalise to a probability vector
        g = b / sum_b

        # Mix a fraction towards the LP solution (step size 0.5)
        a_candidate = 0.5 * a + 0.5 * g

        # Evaluate the candidate
        c_new = _self_conv(a_candidate, fft_len_self)
        new_score = 2.0 * n * c_new.max()

        if new_score < best_score - 1e-12:
            best_score = new_score
            best_a = a_candidate.copy()
            a = a_candidate
            last_improve = now
            print(f"[{elapsed:6.2f}s] LP step improved score = {best_score:.6f}")
        else:
            # If the full step does not improve, try a smaller move
            a_candidate = 0.8 * a + 0.2 * g
            c_new = _self_conv(a_candidate, fft_len_self)
            new_score = 2.0 * n * c_new.max()
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_candidate.copy()
                a = a_candidate
                last_improve = now
                print(f"[{elapsed:6.2f}s] Smaller LP step improved score = {best_score:.6f}")
            else:
                # No improvement – apply a tiny random jitter and continue
                a = _random_block_perturb(a, rng)

        # stop the LP loop if we are very close to the deadline
        if time.monotonic() > deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 7) Phase III – greedy local refinement (tiny mass transfers)
    # ------------------------------------------------------------------
    a = best_a.copy()
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        c = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{elapsed:6.2f}s] New best (local) score = {best_score:.6f}")

        # locate the most offending peak
        max_idx = int(np.argmax(c))

        # contribution of each coordinate to that peak
        idxs = np.arange(n)
        js = max_idx - idxs
        valid = (js >= 0) & (js < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = a[idxs[valid]] * a[js[valid]]

        i_max = int(np.argmax(contrib))

        # pick a random destination far from i_max
        j_dest = rng.integers(0, n)
        while j_dest == i_max:
            j_dest = rng.integers(0, n)

        # move a modest fraction of the mass
        delta = a[i_max] * rng.uniform(0.02, 0.12)
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[j_dest] += delta

        # evaluate new candidate
        c_new = _self_conv(a_new, fft_len_self)
        new_score = 2.0 * n * c_new.max()

        # accept if better (or with a tiny simulated‑annealing probability)
        if new_score < cur_score - 1e-12:
            a = a_new
        else:
            # tiny temperature to escape plateaus
            T = 0.01 * (deadline - time.monotonic()) / (deadline - start_time)
            if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                a = a_new

        # occasional jitter if stuck
        if now - last_improve > 30.0:
            a = best_a.copy()
            for _ in range(2):
                a = _random_block_perturb(a, rng)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Local restart (stagnation) – best {best_score:.6f}")

    # ------------------------------------------------------------------
    # 8) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
