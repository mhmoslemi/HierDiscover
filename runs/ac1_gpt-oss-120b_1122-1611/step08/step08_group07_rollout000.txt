def _project_simplex(v):
    # returns v projected onto simplex {x>=0, sum=1}
    # algorithm: sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a step‑function (non‑negative heights on the simplex) that improves
the bound on ``evaluate_sequence``.

Algorithm
---------

We combine two complementary phases:

* **Phase I – Smooth‑max Adam**  
  The exact objective is ``max_k (a*a)[k]``.  We replace the max by a
  log‑sum‑exp surrogate and optimise the logits of a soft‑max with Adam.
  This quickly discovers a good “shape’’ for the distribution.

* **Phase II – Mirror‑descent + targeted mass transfer**  
  From the best point of Phase I we switch to the true max.  The sub‑gradient
  of the max‑convolution at the current peak is known analytically:
  ``∂c_max/∂a_i = 2·a_{k*‑i}``.  Using a KL‑mirror‑descent step
  (multiply by ``exp(-η·grad)`` and renormalise) preserves non‑negativity and
  the simplex constraint.  Occasionally we perform an explicit “mass‑transfer’’
  move: relocate a fraction of the mass from the index that contributes most
  to the peak to an index with very small contribution.  Simple restarts and
  jitter keep the search from stalling.

The routine respects the 1000 s budget (a 5 s safety margin), works for any
sequence length (default 8192, a power of two for fast FFTs) and returns a list
of floats ready for ``evaluate_sequence``.  It prints progress so you can see
improvements during the run.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. ``a``.
    ``w`` are the soft‑max weights (probability vector of length 2·n‑1).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    # the central part of the convolution of a_rev with w corresponds to
    # the required gradient; see derivation in the doc‑string.
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()

# Main optimisation routine
# ----------------------------------------------------------------------

    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the previous best (~1.5132).
    The routine runs for at most ``budget_s`` seconds (a few seconds safety
    margin are subtracted internally).
    """

    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # 1) Determine sequence length and initialise a
    # ------------------------------------------------------------------
    # If a previous construction is available, resample it; otherwise start
    # from a uniform distribution with a tiny log‑normal perturbation.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        seq0 = np.asarray(init_seq, dtype=np.float64)
        if seq0.ndim == 1 and seq0.size > 0:
            old_n = seq0.size
        else:
            seq0 = None
            old_n = None
    else:
        seq0 = None
        old_n = None

    min_len = 4096
    if old_n is not None:
        min_len = max(min_len, old_n)
    n = _next_pow2(min_len)
    # Cap the length to keep the run fast; 16384 is still comfortable.
    if n > 16384:

    if seq0 is not None:

        x_old = np.arange(seq0.size, dtype=np.float64)

        m_hat = m / (1 - beta1 ** t_adam)
        v_hat = v / (1 - beta2 ** t_adam)

        lr = 0.6 * (1.0 - frac) + 0.02 * frac   # linear decay of learning rate
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # true objective (no surrogate)
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_cur.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I – new best score = {best_score:.6f}")

        # restart if stagnation lasts >20 s
        if now - last_improve > 20.0:
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I – restart due to stagnation")

        # occasional global jitter (helps escape shallow basins)
        if rng.random() < 0.01:
            pert = best_a * np.exp(rng.normal(0.0, 0.15, size=n))
            pert = pert / pert.sum()
            x = np.log(pert + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

    # ------------------------------------------------------------------
    # 4) Phase II – Mirror‑descent + targeted mass‑transfer
    # ------------------------------------------------------------------
    a = best_a.copy()
    best_score = 2.0 * n * _self_conv(a, fft_len_self).max()
    best_a = a.copy()
    last_improve = time.monotonic()
    it_sub = 0
    eta0 = 0.5                # initial step size for mirror descent

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # current convolution and peak index
        c = _self_conv(a, fft_len_self)
        max_idx = int(np.argmax(c))

        # ---- sub‑gradient of the exact max ----
        idxs = np.arange(n)
        js = max_idx - idxs
        mask = (js >= 0) & (js < n)
        g = np.zeros_like(a)
        g[mask] = 2.0 * a[js[mask]]

        # ---- KL‑mirror‑descent update ----
        it_sub += 1
        eta = eta0 / math.sqrt(it_sub)
        a = a * np.exp(-eta * g)
        s = a.sum()
        a = a / s if s != 0.0 else np.full(n, 1.0 / n)

        # ---- evaluate ----
        cur_score = 2.0 * n * _self_conv(a, fft_len_self).max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase II – mirror descent best = {best_score:.6f}")

        # ---- occasional targeted mass‑transfer (≈10 % of iterations) ----
        if rng.random() < 0.1:
            # contribution of each coordinate to the offending peak
            contrib = np.zeros_like(a)
            contrib[mask] = a[idxs[mask]] * a[js[mask]]
            i_max = int(np.argmax(contrib))

            # pick a destination among indices with low contribution (bottom 20 %)
            thresh = np.percentile(contrib, 20)
            low_candidates = np.where((contrib <= thresh) & (np.arange(n) != i_max))[0]
            if low_candidates.size == 0:
                j_dest = rng.integers(0, n)
                while j_dest == i_max:
                    j_dest = rng.integers(0, n)
            else:
                j_dest = rng.choice(low_candidates)

            delta = a[i_max] * rng.uniform(0.02, 0.12)   # move a modest fraction
            a_new = a.copy()
            a_new[i_max] -= delta
            a_new[j_dest] += delta
            # (sum unchanged, non‑negativity preserved by construction)

            cand_score = 2.0 * n * _self_conv(a_new, fft_len_self).max()
            if cand_score < cur_score - 1e-12:
                a = a_new
                cur_score = cand_score
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best_a = a.copy()
                    last_improve = now
                    print(f"[{elapsed:6.2f}s] Phase II – mass‑transfer improved to {best_score:.6f}")

        # ---- stagnation handling (restart with jitter) ----
        if now - last_improve > 30.0:
            a = best_a * np.exp(rng.normal(0.0, 0.2, size=n))
            a = a / a.sum()
            it_sub = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase II – jitter restart due to stall")

    # ------------------------------------------------------------------
    # 5) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
