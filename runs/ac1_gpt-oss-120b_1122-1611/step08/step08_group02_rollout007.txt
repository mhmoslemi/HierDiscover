"""
Propose a step‑function (non‑negative heights) that improves the known bound.
The algorithm consists of three stages:

1. **Broad exploration** (≈30 % of the time)
   – systematic families: uniform, several classical signal‑processing windows
     (hann, hamming, blackman, kaiser, chebyshev, tukey),
     sinusoidal / power / exponential shapes,
   – a wide Beta‑distribution family (α,β sampled log‑uniformly from
     0.01 to 20) and a few random Dirichlet vectors.
   All candidates are normalised to sum = 1 and scored with the exact
   objective (2·n·max (conv(a,a))).

2. **Refinement** (≈ 60 % of the time)
   Simulated‑annealing / hill‑climbing on the simplex:
   repeatedly move a random fraction of mass from a “donor’’ index to a
   “recipient’’ index, accepting moves that improve the score or that are
   accepted probabilistically according to a linearly cooling temperature.
   Occasionally a block‑wise log‑normal jitter is applied to escape plateaus.

3. **Finalisation**
   The best vector seen is re‑normalised (clipped to the limits enforced by
   ``evaluate_sequence``) and returned as a plain Python list.

The whole routine respects the 1000 s budget (leaving a 5 s safety margin)
and never uses closures or filesystem I/O.
"""

import time
import numpy as np
from scipy import signal  # signal processing windows (hann, kaiser, …)


# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((int(x) - 1).bit_length())


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution a*a (FFT implementation)."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    # numerical noise may produce tiny negatives – clamp
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence (without its clamping).
    The vector is assumed to sum to 1; the returned value is
    2·n·max(conv(a,a)).
    """
    n = a.size
    max_b = _max_self_convolution(a)
    if max_b < 0:        # should never happen
        max_b = 0.0
    return 2.0 * n * max_b


# ----------------------------------------------------------------------
# Parametric families of candidate sequences
# ----------------------------------------------------------------------


def gen_beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}, normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n   # centre of each cell
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    raw = np.clip(raw, 0.0, None)
    s = raw.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / s


def gen_exponential_seq(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    raw = np.exp(-lam * i)
    s = raw.sum()
    return raw / s


def gen_power_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum = 1."""
    i = np.arange(1, n + 1, dtype=np.float64)
    raw = np.power(i, -p)
    s = raw.sum()
    return raw / s


def gen_sine_seq(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+½)/n), normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    raw = np.sin(np.pi * (i + 0.5) / n)
    raw = np.clip(raw, 0.0, None)
    return raw / raw.sum()


def gen_window_seq(n: int, kind: str, param: float | None = None) -> np.ndarray:
    """Generate a classic window (hann, hamming, blackman, kaiser, chebwin, tukey)."""
    if kind == "hann":
        w = np.hanning(n)
    elif kind == "hamming":
        w = np.hamming(n)
    elif kind == "blackman":
        w = np.blackman(n)
    elif kind == "kaiser":
        w = np.kaiser(n, param if param is not None else 0.0)
    elif kind == "chebwin":
        # ``param`` is the sidelobe attenuation in dB
        w = signal.chebwin(n, at=param if param is not None else 60.0)
    elif kind == "tukey":
        w = signal.tukey(n, alpha=param if param is not None else 0.5)
    else:
        raise ValueError(f"Unsupported window kind: {kind}")
    # Ensure non‑negativity (all these windows are ≥0) and normalise.
    w = np.clip(w, 0.0, None)
    s = w.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return w / s


# ----------------------------------------------------------------------
# Local optimisation – simulated annealing / hill‑climbing
# ----------------------------------------------------------------------


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def simulated_annealing(a0: np.ndarray,
                        deadline: float,
                        rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing optimiser on the simplex.
    Returns the best vector found and its score.
    """
    n = a0.size
    best = a0.copy()
    best_score = compute_score(best)

    cur = best.copy()
    cur_score = best_score

    start_time = time.monotonic()
    total_time = deadline - start_time
    # Initial temperature – roughly the current excess over the theoretical lower bound (≈1)
    T0 = max(best_score - 1.0, 0.1)

    while time.monotonic() < deadline:
        # --------------------------------------------------------------
        # 1) propose a move (transfer some mass from donor → recipient)
        # --------------------------------------------------------------
        donor = rng.choice(n, p=cur)                 # more mass ⇒ more likely to donate
        if cur[donor] < 1e-12:
            continue

        recipient = rng.integers(n)
        # keep at least a tiny amount at the donor to stay on the simplex
        max_delta = cur[donor]
        # magnitude of the move grows early, shrinks later
        elapsed = time.monotonic() - start_time
        frac = 1.0 - (elapsed / total_time)          # 1 → 0 as time passes
        delta = rng.random() * max_delta * (0.1 + 0.9 * frac)

        new_vec = cur.copy()
        new_vec[donor] -= delta
        new_vec[recipient] += delta
        # renormalise to the simplex (exact sum = 1)
        new_vec /= new_vec.sum()

        new_score = compute_score(new_vec)

        # --------------------------------------------------------------
        # 2) acceptance test (Metropolis)
        # --------------------------------------------------------------
        if new_score < cur_score:
            accept = True
        else:
            # linearly cooled temperature
            T = T0 * frac
            accept = rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12))

        if accept:
            cur = new_vec
            cur_score = new_score
            if new_score < best_score:
                best = new_vec.copy()
                best_score = new_score
                print(f"[{time.monotonic():.2f}s] SA improvement → {best_score:.6f}")

        # --------------------------------------------------------------
        # 3) occasional jitter to escape deep plateaus
        # --------------------------------------------------------------
        if rng.random() < 0.02:
            jitter = _random_block_perturb(cur, rng)
            jitter_score = compute_score(jitter)
            if jitter_score < best_score:
                best = jitter.copy()
                best_score = jitter_score
                cur = jitter.copy()
                cur_score = jitter_score
                print(f"[{time.monotonic():.2f}s] jitter improvement → {best_score:.6f}")

    return best, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (leave a small safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # If a previous construction exists, initialise from it
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / cand.size)
        sc = compute_score(cand)
        best_seq = cand.copy()
        best_score = sc
        print(f"[0.00s] start from known sequence → score {best_score:.6f}")

    # --------------------------------------------------------------
    # 0) Baseline – uniform for a few different lengths (quick)
    # --------------------------------------------------------------
    length_pool = [256, 512, 1024, 2048, 4096, 8000]
    for n in length_pool:
        a_uni = np.full(n, 1.0 / n, dtype=np.float64)
        sc = compute_score(a_uni)
        if sc < best_score:
            best_score = sc
            best_seq = a_uni.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] uniform n={n} → {best_score:.6f}")

    # --------------------------------------------------------------
    # 1) Exploration phase (≈30 % of the total budget)
    # --------------------------------------------------------------
    explore_deadline = start_time + 0.30 * (deadline - start_time)

    # families to explore
    window_kinds = [
        ("hann", None),
        ("hamming", None),
        ("blackman", None),
        ("kaiser", 5.0),      # a few representative β values will be tried later
        ("chebwin", 80.0),    # representative attenuation
        ("tukey", 0.5),
    ]

    for n in length_pool:
        if time.monotonic() >= explore_deadline:
            break

        # ---- classic windows -------------------------------------------------
        for kind, param in window_kinds:
            # a few extra parameters for flexible families
            param_set = [param] if param is not None else [None]
            if kind == "kaiser":
                param_set = [0.0, 2.0, 5.0, 8.0, 12.0]
            if kind == "chebwin":
                param_set = [30.0, 60.0, 90.0, 120.0, 150.0]
            if kind == "tukey":
                param_set = [0.1, 0.5, 0.9]

            for p in param_set:
                a = gen_window_seq(n, kind, p)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_seq = a.copy()
                    print(f"[{time.monotonic() - start_time:.2f}s] win {kind} p={p} n={n} → {best_score:.6f}")

        # ---- sinusoidal / power / exponential -------------------------------
        a = gen_sine_seq(n)
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] sine n={n} → {best_score:.6f}")

        for lam in (0.001, 0.01, 0.1, 0.5, 1.0, 2.0):
            a = gen_exponential_seq(n, lam)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] exp λ={lam:.3f} n={n} → {best_score:.6f}")

        for p in (0.0, 0.5, 1.0, 1.5, 2.0, 3.0):
            a = gen_power_seq(n, p)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] power p={p:.1f} n={n} → {best_score:.6f}")

        # ---- Beta‑shaped distributions --------------------------------------
        special_beta = [(2.0, 1.0), (1.0, 2.0), (0.5, 0.5), (1.5, 1.5), (0.8, 1.2), (1.2, 0.8)]
        for α, β in special_beta:
            a = gen_beta_seq(n, α, β)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] β={α:.2f},{β:.2f} n={n} → {best_score:.6f}")

        # Random Beta draws – broaden the α,β range dramatically
        for _ in range(15):
            if time.monotonic() >= explore_deadline:
                break
            loga = rng.uniform(np.log(0.01), np.log(20.0))
            logb = rng.uniform(np.log(0.01), np.log(20.0))
            α = np.exp(loga)
            β = np.exp(logb)
            a = gen_beta_seq(n, α, β)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] rand β={α:.3f},{β:.3f} n={n} → {best_score:.6f}")

        # Random Dirichlet vectors (sparse / dense regimes)
        for _ in range(10):
            if time.monotonic() >= explore_deadline:
                break
            # concentration ≈ gamma distributed to get a range of sparsities
            conc = rng.gamma(shape=2.0, scale=1.0)  # mean 2, heavy tail
            a = rng.dirichlet(np.full(n, conc))
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_seq = a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] Dirichlet conc={conc:.2f} n={n} → {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Local refinement (simulated annealing) on the best candidate
    # --------------------------------------------------------------
    if best_seq is not None:
        print(f"Refinement start – score {best_score:.6f}, length {best_seq.size}")
        refined_seq, refined_score = simulated_annealing(best_seq, deadline, rng)
        if refined_score < best_score:
            best_seq = refined_seq
            best_score = refined_score
            print(f"Refinement succeeded → new best score {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Final safety fixes
    # --------------------------------------------------------------
    if best_seq is None:
        # fallback – moderate uniform sequence
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce the clamping performed inside ``evaluate_sequence``
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / best_seq.size
    else:
        best_seq /= total

    elapsed = time.monotonic() - start_time
    print(f"Finished search in {elapsed:.2f}s, final best score {best_score:.6f}")

    return [float(x) for x in best_seq]
