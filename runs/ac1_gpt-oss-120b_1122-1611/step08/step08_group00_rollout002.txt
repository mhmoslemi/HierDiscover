def _subgradient_step(a, lr, eps_rel, conv_len, fft_len_self, fft_len_grad):
    # a is 1d array sum=1
    # compute conv self
    Fa_self = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(Fa_self * Fa_self, n=fft_len_self)[:conv_len]
    M = conv.max()
    # determine active indices
    # Use absolute tolerance.
    tol = max(eps_rel, 1e-12)
    active = np.where(conv >= M - tol)[0]
    if active.size == 0:
        # fallback use max index
        active = np.array([np.argmax(conv)], dtype=int)
    # build weight vector
    w = np.zeros(conv_len, dtype=np.float64)
    w[active] = 1.0
    # Could weight equally; optional normalizing not needed
    w_rev = w[::-1]
    Fa_grad = np.fft.rfft(a, n=fft_len_grad)

    conv_rev = np.fft.irfft(Fa_grad * Fw, n=fft_len_grad)[:(n+conv_len-1)]
    L = conv_len
    grad = 2.0 * conv_rev[L-1 - np.arange(len(a))]

  second stage.

* Local stage – a deterministic projected sub‑gradient method that works on
  the true non‑smooth objective `max(a*a)`.  At every iteration the active
  set of convolution indices that attain (or are within a tiny tolerance of)
  the current maximum is collected; the gradient of the average active
  constraint is computed by an FFT‑based convolution and a simple step
  (with a decaying learning‑rate) is taken.  After each step the vector is
  projected onto the probability simplex (∑aᵢ = 1, aᵢ ≥ 0).  Random block
  perturbations and a stagnation‑restart scheme keep the search from getting
  stuck.

The routine respects the 1000 s budget (a 5 s safety margin) and returns the
best sequence found (clipped to the limits imposed by `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities (top‑level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {w | w_i ≥ 0, Σ w_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("proj_simplex expects a 1‑D array")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all elements are zero – return uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """
    Produce a small, diverse portfolio of starting vectors (all already on the
    simplex Σaᵢ = 1).  The collection contains analytic shapes and random
    Dirichlet draws.
    """
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8, 0.75]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # arcsine‑like shape (heavy on the ends)
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    v = 1.0 / np.sqrt(x * (1.0 - x))
    v = np.clip(v, 0.0, None)
    v /= v.sum()
    cand.append(v)

    # two spikes at the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    v /= v.sum()
    cand.append(v)

    # half heavy, half zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    v /= v.sum()
    cand.append(v)

    # a few moderate Dirichlet draws
    for _ in range(8):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    # include the previous best if its length matches n
    if "height_sequence_1" in globals():
        try:
            seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if seq.shape[0] == n:
                seq = np.clip(seq, 0.0, None)
                seq = proj_simplex(seq)
                cand.append(seq)
        except Exception:
            pass

    return cand


def _self_conv_fft(a: np.ndarray, fft_len: int, conv_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns the first ``conv_len`` entries (i.e. the linear convolution).
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return conv


def _subgradient_step(
    a: np.ndarray,
    lr: float,
    eps_abs: float,
    conv_len: int,
    fft_len_self: int,
    fft_len_grad: int,
) -> tuple[np.ndarray, float]:
    """
    One projected sub‑gradient step for the objective
        J(a) = 2·n·max(a*a).

    Parameters
    ----------
    a : np.ndarray
        Current iterate (already on the simplex, sum == 1).
    lr : float
        Learning‑rate (absolute step size after normalisation).
    eps_abs : float
        Absolute tolerance for the active set (entries within
        M‑eps_abs are regarded as active).
    conv_len, fft_len_self, fft_len_grad : int
        Pre‑computed lengths for the FFTs.

    Returns
    -------
    a_new : np.ndarray
        Updated iterate (projected onto the simplex).
    M : float
        The maximal convolution value after the forward pass.
    """
    # ----- forward self‑convolution -----
    Fa_self = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(Fa_self * Fa_self, n=fft_len_self)[:conv_len]
    M = conv.max()

    # ----- active set -----
    # Use a modest absolute tolerance; also accept the exact arg‑max.
    active = np.where(conv >= M - eps_abs)[0]
    if active.size == 0:  # fallback – should never happen
        active = np.array([np.argmax(conv)], dtype=int)

    # Build a weight vector that averages the active constraints.
    w = np.zeros(conv_len, dtype=np.float64)
    w[active] = 1.0 / active.size
    w_rev = w[::-1]

    # ----- gradient of the averaged active constraint -----
    Fa_grad = np.fft.rfft(a, n=fft_len_grad)
    Fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(Fa_grad * Fw, n=fft_len_grad)
    # ``full`` length = n + conv_len – 1 = 3·n – 2
    L = conv_len
    grad = 2.0 * full[L - 1 - np.arange(a.size)]

    # Normalise the step size so that ``lr`` behaves like an absolute step.
    scale = np.mean(np.abs(grad)) + 1e-12
    step = lr * grad / scale
    a_new = a - step
    a_new = proj_simplex(np.clip(a_new, 0.0, None))
    return a_new, M


def _objective_J(a: np.ndarray, n: int, conv_len: int, fft_len_self: int) -> float:
    """
    Compute the exact objective  J = 2·n·max(a*a)  for a vector on the simplex.
    """
    conv = _self_conv_fft(a, fft_len_self, conv_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345, budget_s: float = 1000.0, **kwargs
) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method consists of a cheap global screening
    followed by a focused projected sub‑gradient descent.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (keep a 5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Choose problem size
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = max(512, min(8192, n0))          # keep FFTs cheap
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Global screening – generate candidates and perform a few
    #    sub‑gradient burn‑in steps on each.
    # --------------------------------------------------------------
    cand_pool = _initial_candidates(n, rng)
    best_vec = None
    best_score = np.inf

    # at most 60 s for the cheap screening stage
    screening_deadline = start_time + min(60.0, (deadline - start_time) * 0.15)

    for base in cand_pool:
        a = proj_simplex(base)
        # a very short burn‑in (10–30 iterations)
        burn_in = 15
        for it in range(burn_in):
            # decreasing learning‑rate (0.2 → 0.005)
            frac = it / burn_in
            lr = 0.20 * (1.0 - frac) + 0.005 * frac
            # tolerance shrinks with iterations
            eps = max(1e-8, 1e-6 * (1.0 - frac))
            a, _ = _subgradient_step(
                a, lr, eps, conv_len, fft_len_self, fft_len_grad
            )
        score = _objective_J(a, n, conv_len, fft_len_self)
        if score < best_score:
            best_score = score
            best_vec = a.copy()
        if time.monotonic() >= screening_deadline:
            break

    print(f"Initial screening best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Local refinement – long–run projected sub‑gradient descent.
    # --------------------------------------------------------------
    a = best_vec.copy()
    it = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        elapsed = now - start_time
        total = deadline - start_time
        frac = elapsed / total  # ∈[0,1]

        # learning‑rate schedule: 0.3 → 0.001
        lr = 0.30 * (1.0 - frac) + 0.001 * frac
        # tolerance for the active set (becomes stricter over time)
        eps = max(1e-12, 1e-6 * (1.0 - frac))

        a, M = _subgradient_step(
            a, lr, eps, conv_len, fft_len_self, fft_len_grad
        )
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best J = {best_score:.6f}")

        # occasional random perturbation to escape tiny plateaus
        if rng.random() < 0.015:
            a = _random_block_perturb(a, rng)

        # restart if no improvement for ~30 s
        if now - last_improve > 30.0:
            a = best_vec.copy()
            a = _random_block_perturb(a, rng)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart after stagnation – best J = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence found (clipped to the limits enforced
    #    by ``evaluate_sequence``).
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
