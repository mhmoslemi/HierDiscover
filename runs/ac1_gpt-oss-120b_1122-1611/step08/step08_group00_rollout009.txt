def propose_candidate(seed=12345, budget_s=1000.0, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # Determine n
    if 'height_sequence_1' in globals():
        try:
            n = len(globals()['height_sequence_1'])
            n = max(min(n, 5000), 1024)  # ensure not too big
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2*n-1
    fft_len = _next_pow2(conv_len)

    def compute_J(a):
        fa = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(fa*fa, n=fft_len)[:conv_len]
        return 2.0 * n * conv.max()

    # init a from best known if available else uniform
    if 'height_sequence_1' in globals():
        init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        if init.size != n:
            # resample to target n using interpolation
            init = np.interp(np.linspace(0, init.size-1, n), np.arange(init.size), init)
        a = proj_simplex(init)
    else:
        a = np.full(n, 1.0/n, dtype=np.float64)

    best = a.copy()
    best_J = compute_J(best)
    print(f"Initial J = {best_J:.6f}")

    # SA parameters
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    T0 = 0.1  # initial temperature for accepting worse solutions
    T_end = 1e-4

    # schedule: exponential decay on sigma_epsilon perhaps
    total_time = deadline - start_time
    iteration = 0
    accept = 0

    while time.monotonic() < deadline:
        # compute fraction of time passed
        t_now = time.monotonic()
        frac = (t_now - start_time) / total_time
        # temperature
        T = T0 * (T_end/T0) ** frac

        # decide move type: small transfer or block perturb
        if rng.random() < 0.8:
            # transfer small mass between two indices
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # choose amount delta
            # relative to donor's mass, with scaling controlled by temperature
            delta_max = min(a[i], 0.2)  # can't move more than 20%
            # sample delta, maybe log-uniform
            delta = rng.random() * delta_max * (0.5 + 0.5*frac)  # smaller as time passes
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # ensure nonnegative
            # sum unchanged
            a_new = proj_simplex(a_new)   # reproject to maintain simplex (ensures sum=1)
        else:
            # block perturb
            a_new = _random_block_perturb(a, rng)

        # compute objective
        J_new = compute_J(a_new)

        # Acceptance
        if J_new < best_J:
            best_J = J_new
            best = a_new.copy()
            a = a_new.copy()
            accept += 1
            print(f"[{t_now-start_time:6.2f}s] New best J={best_J:.6f}")
        else:
            # accept with probability exp(-(J_new - best_J)/T)
            # if T is very small, probability tiny
            if rng.random() < np.exp(-(J_new - best_J) / max(T, 1e-12)):
                a = a_new.copy()
                accept += 1

        iteration += 1
        # occasional restart: if no improvement in certain time, use best and perturb
        if (t_now - start_time) - (frac * total_time) < 0:  # nonsense; ignoring

        # we may include periodic heavy perturbation
        if iteration % 10000 == 0:
            # reinitialize random start maybe around best
            a = best.copy()
            # small random noise
            a = a * np.exp(rng.normal(0.0, 0.02, size=n))
            a = proj_simplex(a)

    # after finish, return best clipped to [0,1000]
    best = np.clip(best, 0.0, 1000.0)
    return [float(x) for x in best]

"""
Propose a step‑function sequence that (approximately) minimizes
``evaluate_sequence``.  The routine combines three cheap optimisation
phases that respect a 1000 s time‑budget:

  1. A tiny Differential‑Evolution (DE) population explores the simplex.
  2. A projected sub‑gradient descent works on the *hard* max‑convolution
     (the true objective) – this often escapes the plateau where a
     soft‑max surrogate can get stuck.
  3. A projected Adam optimiser refines the best vector using a
     temperature‑controlled soft‑max surrogate (as in the reference code).

All helpers are top‑level (no closures) and the algorithm stays within
the allowed computational budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Top‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by simplex projection."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A small portfolio of diverse starting points, all normalised to sum = 1."""
    cand: list[np.ndarray] = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # exponential decays
    for r in [0.99, 0.975, 0.95, 0.9, 0.85, 0.8]:
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # two spikes on the ends
    v = np.zeros(n, dtype=np.float64)
    v[0] = 0.5
    v[-1] = 0.5
    cand.append(v)

    # half‑heavy, half‑zero
    half = n // 2
    v = np.zeros(n, dtype=np.float64)
    v[:half] = 2.0 / n
    cand.append(v)

    # a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v.astype(np.float64))

    return cand


def compute_J(a: np.ndarray, n: int,
              fft_len_self: int, conv_len: int) -> float:
    """Exact objective J = 2·n·max(a∗a) (sum(a)=1 is assumed)."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return float(2.0 * n * conv.max())


def self_convolution(a: np.ndarray,
                     fft_len_self: int, conv_len: int) -> np.ndarray:
    """Linear self‑convolution a∗a using an FFT of given length."""
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    return conv


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) minimising
    ``evaluate_sequence``.  The method runs a short DE global search,
    a projected sub‑gradient descent on the true max‑convolution, and a
    final Adam refinement with a soft‑max surrogate.  It respects the
    time budget (5 s safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # ------------------------------------------------------------------
    now = time.monotonic()
    deadline = now + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = len(globals()["height_sequence_1"])
            n = max(1024, min(5000, n))          # keep n moderate
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for a ⋆ w_rev (Adam)

    # ------------------------------------------------------------------
    # 2) Helper to evaluate the true objective
    # ------------------------------------------------------------------
    # (compute_J defined above)

    # ------------------------------------------------------------------
    # 3) Build initial population (DE) and evaluate
    # ------------------------------------------------------------------
    total_time = deadline - now
    # allocate 15 % of the remaining budget to DE
    de_deadline = now + total_time * 0.15

    popsize = 30
    base_cands = _initial_candidates(n, rng)

    # include the previously known construction if present
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size != n:
            # simple linear interpolation to required length
            x_old = np.linspace(0, 1, prev.size)
            x_new = np.linspace(0, 1, n)
            prev = np.interp(x_new, x_old, prev)
        base_cands.append(proj_simplex(prev))

    # fill up to popsize with Dirichlet draws if needed
    while len(base_cands) < popsize:
        base_cands.append(rng.dirichlet(np.ones(n)))

    # pick a random subset of size popsize
    indices = rng.choice(len(base_cands), size=popsize, replace=False)
    population = [proj_simplex(base_cands[i].astype(np.float64)) for i in indices]
    scores = [compute_J(v, n, fft_len_self, conv_len) for v in population]

    best_idx = int(np.argmin(scores))
    best_vec = population[best_idx].copy()
    best_score = scores[best_idx]
    print(f"[{0.0:6.2f}s] DE start – best J = {best_score:.6f}")

    # DE parameters
    F = 0.8
    CR = 0.9

    while time.monotonic() < de_deadline:
        for i in range(popsize):
            # three distinct donors
            donors = rng.choice([j for j in range(popsize) if j != i],
                                size=3, replace=False)
            r1, r2, r3 = donors
            mutant = population[r1] + F * (population[r2] - population[r3])
            mutant = np.clip(mutant, 0.0, None)
            s = mutant.sum()
            if s == 0.0:
                mutant[:] = 1.0 / n
            else:
                mutant /= s

            # binomial crossover
            cross_mask = rng.random(n) < CR
            if not np.any(cross_mask):
                cross_mask[rng.integers(n)] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = proj_simplex(trial)

            trial_score = compute_J(trial, n, fft_len_self, conv_len)
            if trial_score < scores[i]:
                population[i] = trial
                scores[i] = trial_score
                if trial_score < best_score:
                    best_score = trial_score
                    best_vec = trial.copy()
                    elapsed = time.monotonic() - now
                    print(f"[{elapsed:6.2f}s] DE new best J = {best_score:.6f}")

        # safety break
        if time.monotonic() >= de_deadline:
            break

    # ------------------------------------------------------------------
    # 4) Phase 2 – Projected sub‑gradient descent on the *hard* max
    # ------------------------------------------------------------------
    sg_deadline = now + total_time * 0.75   # up to 75 % of total time
    a = best_vec.copy()
    last_improve = time.monotonic()
    it = 0
    # step‑size schedule (linear decay)
    eta_start, eta_end = 0.5, 1e-4

    while time.monotonic() < sg_deadline:
        it += 1
        # current convolution and the position of the maximum
        conv = self_convolution(a, fft_len_self, conv_len)
        k_max = int(np.argmax(conv))

        # sub‑gradient of the maximal entry (hard max)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[idx[mask]]

        # learning‑rate schedule based on elapsed fraction
        elapsed = time.monotonic() - now
        frac = (elapsed / (sg_deadline - now))
        eta = eta_start * (1.0 - frac) + eta_end * frac

        # projected additive update
        a_new = a - eta * grad
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s == 0.0:
            a_new[:] = 1.0 / n
        else:
            a_new /= s

        cur_score = compute_J(a_new, n, fft_len_self, conv_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a_new.copy()
            a = a_new
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - now:6.2f}s] Sub‑grad new best J = {best_score:.6f}")
        else:
            a = a_new

        # occasional random block jitter
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart if stagnation > 15 s
        if time.monotonic() - last_improve > 15.0:
            a = best_vec.copy()
            # tiny multiplicative noise before restart
            a = a * np.exp(rng.normal(0.0, 0.04, size=n))
            a = proj_simplex(a)
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - now:6.2f}s] sub‑grad restart – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase 3 – Projected Adam refinement (soft‑max surrogate)
    # ------------------------------------------------------------------
    adam_deadline = deadline
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    it = 0
    last_improve = time.monotonic()
    start_adam = time.monotonic()
    total_adam = adam_deadline - start_adam

    while time.monotonic() < adam_deadline:
        it += 1
        now_iter = time.monotonic()
        frac = (now_iter - start_adam) / total_adam   # ∈[0,1]

        # learning‑rate schedule (exponential decay)
        lr_start, lr_end = 0.25, 0.001
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # temperature schedule for soft‑max
        alpha_start, alpha_end = 2.0, 3000.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ----- compute self‑convolution -----
        conv = self_convolution(a, fft_len_self, conv_len)

        # ----- soft‑max over convolution entries -----
        c_max = conv.max()
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth surrogate -----
        w_rev = w[::-1]
        fa = np.fft.rfft(a, n=fft_len_grad)
        fw = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa * fw, n=fft_len_grad)
        start_idx = w_rev.size - 1                      # = 2·n − 2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----- Adam update + projection onto simplex -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # ----- evaluate true objective -----
        cur_score = compute_J(a, n, fft_len_self, conv_len)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = now_iter
            print(f"[{now_iter - now:6.2f}s] Adam new best J = {best_score:.6f}")

        # occasional jitter
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart after 20 s of stagnation
        if now_iter - last_improve > 20.0:
            a = best_vec.copy()
            a = a * np.exp(rng.normal(0.0, 0.06, size=n))
            a = proj_simplex(a)
            last_improve = now_iter
            print(f"[{now_iter - now:6.2f}s] Adam restart – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
