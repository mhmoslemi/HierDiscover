def _move_mass(a, rng, step_frac):

    i, j = rng.integers(0, n, size=2)

    # renorm
    s = out.sum()
    if s == 0.0:

    else:

The evaluation function  

\[
E(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2}
\]

is *scale‑invariant*, therefore we can restrict ourselves to probability vectors  
\(\sum_i a_i = 1,\; a_i\ge 0\).  In this case  

\[
E(a)=2\,n\;\max_k (a*a)_k .
\]

Hence the task is to minimise the infinity norm of the self‑convolution while staying on the simplex.

The previous code used a smooth‑max surrogate together with Adam.  In the
new approach we **optimise the true max directly** by a *projected sub‑gradient /
multiplicative‑mirror descent* method.  The algorithm consists of three phases

1. **Initialisation** – start from the best known construction (`height_sequence_1`)
   or, if unavailable, from the uniform distribution.
2. **Exact max‑gradient descent** – at each iteration the index `k*` where the
   convolution attains its maximum is identified.
   The sub‑gradient of the corresponding quadratic form is
   \[
   \frac{\partial}{\partial a_i}(a*a)_{k*}=2\,a_{k*-i}\quad (0\le i\le n-1),
   \]
   (outside the range the term is zero).  
   A *mirror‑descent* (multiplicative) update  
   \[
   a_i \leftarrow a_i\,
          \exp\!\bigl(-\eta_t\, g_i\bigr)
   \quad\text{and normalise to sum 1}
   \]
   keeps the iterate on the simplex and directly reduces the maximal
   convolution value.  The step size is slowly decreased
   \(\eta_t = \eta_0/\sqrt{t}\).

3. **Fine hill‑climbing** – when the time budget is almost exhausted we
   perform cheap local moves (transfer a small fraction of mass between two
   positions) and occasional block perturbations, always accepting only
   improvements.

Stagnation (`> 30 s` without improvement) triggers a *soft restart* around the
current best solution by adding Gaussian noise in log‑space.  All operations are
vectorised; the convolution is performed with an FFT of length a power of two,
so each iteration costs only `O(n log n)`.

The routine respects the 1000 s budget (a few seconds are left for the

* works directly on the true max (no log‑sum‑exp surrogate)
* multiplicative mirror‑descent keeps the iterate on the simplex
* occasional random restarts and a final hill‑climbing phase
* uses FFT‑based convolution for speed (O(n log n) per iteration)
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution using a pre‑allocated FFT length.
    Result length is 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _subgradient_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Exact sub‑gradient of the maximal convolution entry:
        g_i = ∂/∂a_i (a*a)_k  = 2 * a_{k-i}   (if 0 ≤ k-i < n, else 0)
    """
    n = a.size
    g = np.zeros_like(a)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1, dtype=int)
    g[idx] = 2.0 * a[k - idx]
    return g


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Project vector onto the probability simplex:
    clip negatives to zero and renormalise.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / v.size)
    return v / s


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor,
    then re‑normalise to a probability vector.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    return _simplex_projection(out)


def _move_mass(a: np.ndarray, rng: np.random.Generator, step_frac: float) -> np.ndarray:
    """
    Transfer a small fraction of mass from a random source index to a
    random destination index and renormalise.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = step_frac * rng.random() * a[i]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    return _simplex_projection(out)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best bound.  The routine uses:

    * exact max‑gradient descent with multiplicative (mirror‑descent) updates
    * stochastic restarts when stagnation is detected
    * a short hill‑climbing phase for fine tuning
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget handling – keep a small safety margin for the harness
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose sequence length and initialise a probability vector
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 = a0 / s
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        n = a0.size
    else:
        # fallback: moderately long uniform start
        n = 3840
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for convolution
    # ------------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Best‑solution tracking
    # ------------------------------------------------------------------
    conv = _self_conv(a0, fft_len)
    best = a0.copy()
    best_score = 2.0 * n * conv.max()
    last_improve = start_time
    print(f"[{0:6.2f}s] initial score = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase I – exact max‑gradient mirror descent
    # ------------------------------------------------------------------
    a = best.copy()
    t_iter = 0
    eta0 = 1.0            # initial step‑size factor
    stagnation_limit = 30.0   # seconds without improvement → restart

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        # --- compute convolution and locate the maximal entry
        conv = _self_conv(a, fft_len)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]
        cur_score = 2.0 * n * max_val

        # --- record improvement
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best (grad) = {best_score:.6f}")

        # --- sub‑gradient of the active max entry
        g = _subgradient_max_conv(a, max_idx)

        # --- step‑size schedule (diminishing)
        t_iter += 1
        eta = eta0 / np.sqrt(t_iter)

        # --- multiplicative mirror‑descent update (keeps sum = 1)
        a = a * np.exp(-eta * g)
        a = _simplex_projection(a)

        # --- stagnation handling
        if now - last_improve > stagnation_limit:
            # soft restart around the current best (log‑space Gaussian noise)
            noise = rng.normal(0.0, 0.4, size=n)
            a = np.exp(np.log(best + 1e-12) + noise)
            a = _simplex_projection(a)
            t_iter = 0
            last_improve = now
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] restart (stagnation) – score ≈ {2.0*n*_self_conv(a, fft_len).max():.6f}")

        # --- occasional larger diversion
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

    # ------------------------------------------------------------------
    # 5) Phase II – fine hill‑climbing (greedy mass moves)
    # ------------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    phase2_start = time.monotonic()
    phase2_total = deadline - phase2_start

    def _step_fraction(elapsed):
        """Linear decay of the move‑size fraction."""
        return 0.3 * (1.0 - elapsed / phase2_total) + 0.01

    while True:
        now = time.monotonic()
        if now >= deadline:
            break

        elapsed = now - phase2_start
        step_frac = _step_fraction(elapsed)

        # small random transfer of mass
        cand = _move_mass(a, rng, step_frac)
        cand_score = 2.0 * n * _self_conv(cand, fft_len).max()

        if cand_score < cur_score - 1e-12:
            a = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = a.copy()
                print(f"[{now - start_time:6.2f}s] new best (hill‑climb) = {best_score:.6f}")

        # occasional block‑perturbation for larger jumps
        if rng.random() < 0.015:
            a_block = _random_block_perturb(a, rng)
            block_score = 2.0 * n * _self_conv(a_block, fft_len).max()
            if block_score < cur_score:
                a = a_block
                cur_score = block_score
                if block_score < best_score - 1e-12:
                    best_score = block_score
                    best = a.copy()
                    print(f"[{now - start_time:6.2f}s] new best (block) = {best_score:.6f}")

        # stagnation safety – tiny random restart
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.5, size=n)
            a = np.exp(np.log(best + 1e-12) + noise)
            a = _simplex_projection(a)
            cur_score = 2.0 * n * _self_conv(a, fft_len).max()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] local restart (hill‑climb) – score {cur_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return the best sequence found (clipped to evaluation limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
