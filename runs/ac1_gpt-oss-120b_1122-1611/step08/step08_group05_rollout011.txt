"""
Exponential‑decay step function generator

The convolution of a geometric (exponential) sequence a_i = C·exp(-δ·i/n)
has a closed form and its maximum can be driven to arbitrarily small
values by an appropriate choice of the decay parameter δ.
We therefore search a small set of δ values, build the corresponding
sequence (length taken from the previously discovered best construction
or a default size), evaluate the exact score with ``evaluate_sequence``
and keep the best one.

The routine uses only top‑level helpers (no closures) and respects the
time budget – the whole search needs a few milliseconds even for
n≈4000.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – builds the exponential candidate for a given decay δ and length n
# ----------------------------------------------------------------------
def _build_exponential_seq(delta: float, n: int) -> np.ndarray | None:
    """Return a non‑negative vector of length n with sum≈1.

    a_i = C·exp(-δ·i/n),  where  C = (1‑r)/(1‑r**n)  and  r = exp(-δ/n).
    The vector is re‑normalised to guard against round‑off.
    """
    # r = exp(-δ/n)
    r = np.exp(-delta / n)
    # denominator can become zero only for pathological δ; guard against it
    denom = 1.0 - r**n
    if denom == 0.0:
        return None
    C = (1.0 - r) / denom
    i = np.arange(n, dtype=np.float64)
    seq = C * np.exp(-delta * i / n)          # = C·r**i
    # re‑normalise (theoretical sum is 1, but floating error may accumulate)
    total = seq.sum()
    if total == 0.0:
        return None
    seq /= total
    return seq
