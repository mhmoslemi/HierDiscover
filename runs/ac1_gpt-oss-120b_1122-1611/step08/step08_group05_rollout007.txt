"""
Advanced step‑function optimizer

The algorithm works in two phases inside the 1000 s budget.

Phase 1 – smooth‑max Adam optimisation:
    * Optimise log‑weights `x` on the simplex via a log‑sum‑exp (LSE) surrogate
      of the max‑convolution.
    * The surrogate sharpens from γ≈0.5 to γ≈2000 while the learning‑rate
      decays, allowing a smooth trajectory followed by a near‑exact gradient.
    * Stagnation triggers a soft restart with moderate log‑normal noise;
      occasional block‑jitter helps escape shallow plateaux.

Phase 2 – permutation / simulated‑annealing refinement:
    * The multiset of values obtained in Phase 1 is kept fixed while their
      ordering is altered.
    * Random swaps (and occasional block perturbations) are accepted if they
      improve the exact objective or with a temperature‑scaled probability.
    * The temperature cools linearly to zero, turning the process into
      greedy hill‑climbing near the end.

Both phases use FFT‑based self‑convolution for speed
(`_self_conv_fft`).  The best sequence found is clipped to the limits enforced
by `evaluate_sequence` and returned as a plain ``list[float]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must stay at module level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns the full convolution array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the LSE surrogate (or a one‑hot weight) w.r.t. ``a``.
    ``w`` has length 2·len(a)‑1 (same as a*a).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(a_rev.size + w.size - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]


def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """
    Return max_i (a*a)_i using FFT for speed.
    """
    conv = _self_conv_fft(a, fft_len)
    return float(conv.max())


def _score_from_a(a: np.ndarray, fft_len: int) -> float:
    """
    Exact objective value for a (sum(a) is assumed to be 1).
    score = 2·n·max(a*a) .
    """
    n = a.size
    return 2.0 * n * _conv_max(a, fft_len)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Optimise a non‑negative step‑function under the evaluation metric.
    Returns the best list of heights found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # time budget (5 s safety margin)
    # --------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    total_time = deadline - start

    # --------------------------------------------------------------
    # initialise a on the simplex (use previous construction if present)
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            a0 = a0 / s0 if s0 > 0 else np.full_like(a0, 1.0 / a0.size)
            n = a0.size
        else:
            n = 3809  # fallback length
            a0 = np.full(n, 1.0 / n)
    else:
        n = 3809
        a0 = np.full(n, 1.0 / n)

    # log‑weights (Adam works on the unrestricted space)
    x = np.log(a0 + 1e-12)

    # pre‑compute FFT length for the whole run
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # Adam state
    # --------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_step = 0

    # --------------------------------------------------------------
    # bookkeeping
    # --------------------------------------------------------------
    a = _softmax(x)
    best = a.copy()
    best_score = _score_from_a(best, fft_len)
    last_improve = start

    # schedule: gradient phase occupies roughly 60 % of the budget
    grad_phase_end = start + 0.6 * total_time

    # hyper‑parameters for the Adam phase
    gamma_start = 0.5
    gamma_end = 2000.0          # very sharp surrogate
    lr_start = 1.0
    lr_end = 0.02

    # --------------------------------------------------------------
    # Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac_total = elapsed / total_time

        # ----------------------------------------------------------
        # Phase 1 – smooth‑max Adam optimisation
        # ----------------------------------------------------------
        if now < grad_phase_end:
            # simplex point
            a = _softmax(x)

            # self‑convolution (FFT, cheap)
            c = _self_conv_fft(a, fft_len)                     # length 2·n‑1

            # schedule for the surrogate sharpness
            gamma = gamma_start + (gamma_end - gamma_start) * frac_total

            # for the last 20 % of the Adam phase use a true sub‑gradient
            if frac_total > 0.8:
                # one‑hot weight at the current max index
                w = np.zeros_like(c)
                w[np.argmax(c)] = 1.0
            else:
                # smooth LSE weights
                c_shift = c - np.max(c)          # stabilise exponentials
                w = np.exp(gamma * c_shift)
                w_sum = w.sum()
                if w_sum > 0.0:
                    w /= w_sum
                else:
                    w = np.full_like(w, 1.0 / w.size)

            # gradient w.r.t. a
            g_a = _gradient_wrt_a(a, w)

            # chain rule to log‑weights x (ensures sum stays 1)
            avg = np.dot(a, g_a)
            g_x = a * (g_a - avg)

            # Adam update
            t_step += 1
            m = beta1 * m + (1.0 - beta1) * g_x
            v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
            m_hat = m / (1.0 - beta1 ** t_step)
            v_hat = v / (1.0 - beta2 ** t_step)

            # linear learning‑rate decay
            lr = lr_start * (1.0 - frac_total) + lr_end * frac_total
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            x -= step

            # exact evaluation and bookkeeping
            cur_score = 2.0 * n * np.max(c)    # equivalent to full score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] Adam new best = {best_score:.6f}")

            # stagnation handling – soft restart
            if now - last_improve > 20.0:
                noise = rng.normal(0.0, 0.8, size=n)
                x = np.log(best + 1e-12) + noise
                m.fill(0.0)
                v.fill(0.0)
                t_step = 0
                last_improve = now
                print(f"[{elapsed:6.2f}s] Adam restart (stagnation) – best {best_score:.6f}")

            # occasional block‑jitter
            if rng.random() < 0.015:
                a_jit = _random_block_perturb(best, rng)
                x = np.log(a_jit + 1e-12)
                m.fill(0.0)
                v.fill(0.0)
                t_step = 0

        # ----------------------------------------------------------
        # Phase 2 – permutation / simulated‑annealing refinement
        # ----------------------------------------------------------
        else:
            # initialise local variables on first entry to Phase 2
            if 'cur_a' not in locals():
                cur_a = best.copy()
                cur_score = best_score

            # temperature schedule (cooling linear)
            perm_elapsed = now - grad_phase_end
            perm_total = deadline - grad_phase_end
            frac_perm = perm_elapsed / perm_total if perm_total > 0 else 1.0
            temperature = 0.10 * (1.0 - frac_perm) + 0.001 * frac_perm

            # propose a random swap
            i, j = rng.integers(0, n, size=2)
            if i != j:
                cur_a[i], cur_a[j] = cur_a[j], cur_a[i]          # swap
                new_score = _score_from_a(cur_a, fft_len)

                # accept if better or with Metropolis probability
                if new_score < cur_score or rng.random() < np.exp(-(new_score - cur_score) / (temperature + 1e-16)):
                    cur_score = new_score
                    if new_score < best_score - 1e-12:
                        best_score = new_score
                        best = cur_a.copy()
                        print(f"[{now - start:6.2f}s] perm swap improved → {best_score:.6f}")
                else:
                    # revert swap
                    cur_a[i], cur_a[j] = cur_a[j], cur_a[i]

            # occasional block‑jitter in the permutation phase
            if rng.random() < 0.003:
                cur_a = _random_block_perturb(cur_a, rng)
                cur_score = _score_from_a(cur_a, fft_len)

    # --------------------------------------------------------------
    # Return the best sequence (clipped to evaluate_sequence limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
