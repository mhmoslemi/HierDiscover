"""
Hybrid optimiser with extra fine‑tuning.

Phase 1 – Adam optimisation on a soft‑max approximation of the
           max‑convolution (high temperature schedule, occasional jitter,
           hard restarts if no improvement).

Phase 2 – Projected sub‑gradient descent on the exact max‑convolution
           (simple Euclidean projection onto the probability simplex,
           adaptive step‑size).

Phase 3 – Light‑weight simulated‑annealing / random local moves for the
           remaining budget (mass transfers between two coordinates,
           always staying on the simplex).

The routine works with any existing `height_sequence_1` (by up‑sampling it)
or falls back to a uniform initial distribution.
All helper functions are top‑level; no closures are used.
"""

import time
import numpy as np
from scipy.special import logsumexp


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (logits → simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using an FFT (real‑valued)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of ``evaluate_sequence`` for a (sum(a) is not forced)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to escape plateaus
        if it % 500 == 0:
            x += rng.normal(0.0, 0.02, size=n)

        # hard restart if we have been stagnant for a while
        if time.monotonic() - last_improve > RESTART_WAIT:
            x = np.log(best_a + eps) + rng.normal(0.0, 0.15, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase1 hard restart")

    # -----------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact max‑convolution
    # -----------------------------------------------------------------
    a = best_a.copy()
    step = 0.15
    last_improve_sub = time.monotonic()

    while time.monotonic() < deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # gradient of the active shift (see analysis)
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # tentative Euclidean projection onto the simplex
        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step *= 1.07
            last_improve_sub = time.monotonic()
            print(f"[{last_improve_sub - start:6.2f}s] subgrad improvement {best_score:.6f}")
        else:
            step *= 0.68
            # occasional jitter when step becomes tiny or at random
            if step < 1e-9 or rng.random() < 0.002:
                jitter = rng.normal(0.0, 0.0015, size=n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.015)

        # reserve a few seconds for the final random‑local‑search phase
        if deadline - time.monotonic() < 15.0:
            break

    # -----------------------------------------------------------------
    # Phase 3 – Simple random local moves (simulated‑annealing style)
    # -----------------------------------------------------------------
    while time.monotonic() < deadline:
        # pick two distinct indices
        i = rng.integers(0, n)
        j = rng.integers(0, n - 1)
        if j >= i:
            j += 1
        if a[i] <= 1e-14:
            continue

        # transfer a fraction of mass from i to j
        delta = rng.uniform(0.0, min(0.5 * a[i], 0.01))
        if delta <= 0.0:
            continue
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _objective_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best_a = a_new.copy()
            a = a_new
            # give a tiny log feedback about progress
            if rng.random() < 0.1:
                print(f"[{time.monotonic() - start:6.2f}s] random move improvement {best_score:.6f}")

    # -----------------------------------------------------------------
    # Return the best sequence (clipped to evaluator limits)
    # -----------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
