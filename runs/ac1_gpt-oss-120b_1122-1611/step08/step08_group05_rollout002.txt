def _project_simplex(v):
    """Project vector v onto simplex {x >=0, sum x = 1}."""
    # Using algorithm from Wang & Carreira-Perpinan (2013)
    n = v.shape[0]
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all negative, return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Arcsine‑based step‑function search
---------------------------------
The self‑convolution of a continuous arcsine density on [0,1] is exactly
uniform on [0,2].  Discretising this density therefore yields a
distribution whose discrete convolution is almost flat, i.e. the maximum
probability of a sum is close to the theoretical lower bound
≈ 1/(2 n‑1).  The score 2·n·max(conv) / (∑a)² then approaches 1,
substantially better than the previous ≈ 1.51.

The algorithm proceeds in three phases:

1. **Parameter sweep** – generate families a_i(α)= (x_i·(1−x_i))^−α
   with x_i=(i+½)/n (α≈½ is the arcsine law).  A cheap grid search over
   α yields a good starting point.

2. **Local refinement** – a projected sub‑gradient method on the exact
   max‑convolution (one‑hot surrogate).  The gradient is a simple
   shifted copy of the current vector; after each step the iterate is
   projected onto the probability simplex.

3. **Stagnation handling** – if no improvement is seen for 30 s a
   moderate Gaussian jitter is added and the optimisation restarts.

The routine respects the 1000 s time budget (leaving a 5 s safety
margin) and returns the best sequence found.  All helper functions are
defined at module level as required.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must be top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    # Numerical noise may produce tiny negatives – clamp to zero.
    conv[conv < 0] = 0.0
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _arcsine_initial(n: int, alpha: float) -> np.ndarray:
    """
    Return a normalised discretisation of the arcsine law with exponent α:
        a_i ∝ (x_i·(1−x_i))^{−α},  x_i = (i+½)/n .
    """
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid division by zero at the exact ends
    eps = 1e-12
    w = np.power(x * (1.0 - x) + eps, -alpha)
    a = w / w.sum()
    return a

    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose problem size -------------------------------------------------
    # Use the length of a previously found construction if provided,
    # otherwise fall back to a moderate size (≈3000).
    # ------------------------------------------------------------------
    prev = globals().get("height_sequence_1")
    if isinstance(prev, (list, tuple, np.ndarray)) and len(prev) > 0:
        n = int(len(prev))
    else:
        n = 3000

    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 2) Coarse sweep over the exponent α (the arcsine law corresponds to α=0.5)
    # ------------------------------------------------------------------
    best_score = float("inf")
    best_a = None
    best_alpha = None

    # grid roughly covering the plausible region
    alphas = np.linspace(0.40, 0.60, 9)   # 0.40,0.425,…,0.60
    for alpha in alphas:
        if time.monotonic() >= deadline:
            break
        a = _arcsine_initial(n, alpha)
        # evaluate – this is the exact objective (evaluate_sequence does all clipping)
        score = evaluate_sequence(a.tolist())
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            best_alpha = alpha
            print(f"[grid] α={alpha:.3f} → new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Refine the exponent α by random local search (few dozen samples)
    # ------------------------------------------------------------------
    if best_alpha is not None:
        alpha = best_alpha
        while time.monotonic() < deadline:
            # propose a nearby α
            cand_alpha = alpha + rng.normal(0.0, 0.02)
            cand_alpha = float(np.clip(cand_alpha, 0.30, 0.70))
            a = _arcsine_initial(n, cand_alpha)
            score = evaluate_sequence(a.tolist())
            if np.isfinite(score) and score < best_score:
                best_score = score
                best_a = a.copy()
                alpha = cand_alpha
                print(f"[local α] α={cand_alpha:.4f} → better score {best_score:.6f}")
                # keep exploring around the new best
            # stop after a modest number of attempts if time permits
            if time.monotonic() > deadline - 30:   # leave time for next phase
                break

    # If nothing succeeded (should not happen), fall back to uniform.
    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_score = evaluate_sequence(best_a.tolist())
        print(f"[fallback] uniform → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()                # current iterate (always Σa = 1)
    best_iter = a.copy()
    last_improve = time.monotonic()
    it = 0
    eta0 = 0.6                       # initial step‑size coefficient

    while time.monotonic() < deadline:
        # compute self‑convolution via FFT (fast, enough for gradient)
        conv = _self_conv_fft(a, fft_len)
        i_star = int(conv.argmax())               # index attaining the max
        max_val = conv[i_star]

        # occasional exact evaluation (every 50 iterations) for logging
        if it % 50 == 0:
            exact_score = 2.0 * n * max_val
            if exact_score < best_score - 1e-12:
                best_score = exact_score
                best_iter = a.copy()
                last_improve = time.monotonic()
                print(f"[{it:5d}] exact improvement → score {best_score:.6f}")

        # ----------- compute sub‑gradient (one‑hot surrogate) -------------
        g = np.zeros_like(a)
        lo = max(0, i_star - (n - 1))
        hi = min(i_star, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        # contribution: g_k = 2·a_{i*−k} for all valid k
        g[idx] = 2.0 * a[i_star - idx]

        # ----------- step‑size (diminishing) ----------------------------
        step = eta0 / np.sqrt(it + 1.0)
        a_new = a - step * g
        # ----------- projection onto simplex -----------------------------
        a = _project_simplex(a_new)

        it += 1

        # ----------- stagnation handling ---------------------------------
        now = time.monotonic()
        if now - last_improve > 30.0:
            # Add moderate Gaussian jitter to escape a flat region
            jitter = np.exp(rng.normal(0.0, 0.15, size=n))
            a = _project_simplex(best_iter * jitter)
            last_improve = now
            print(f"[{it:5d}] restart (stagnation) – score {best_score:.6f}")

        # ----------- safety break if we are very close to the bound ------
        # The theoretical lower bound is 1.0; stop early if we are already
        # < 1.02 to save time.
        if best_score < 1.02:
            print(f"[{it:5d}] early termination – low bound reached {best_score:.6f}")
            break

    # ------------------------------------------------------------------
    # 5) Final check with the official evaluator (ensures clipping etc.)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_iter.tolist())
    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score
        best_iter = best_iter.copy()
    print(f"Final best score reported by `evaluate_sequence`: {best_score:.6f}")

    # ------------------------------------------------------------------
    # 6) Return best vector (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_iter = np.clip(best_iter, 0.0, 1000.0)
    return [float(v) for v in best_iter.tolist()]
