```
def perturb_move(a, rng, step_scale=0.02):
    # propose new a
    # choose random source and dest indices.
    source = rng.integers(0, n)
    dest = rng.integers(0, n)
    if source == dest:
        return a
    delta = step_scale * a[source] * rng.random() # random fraction up to step_scale
    if delta <= 0:
        return a
    a_new = a.copy()
    a_new[source] -= delta
    a_new[dest] += delta
    # Ensure nonneg
    if a_new[source] < 0:
        a_new[dest] += a_new[source]  # subtract leftover
        a_new[source] = 0.0
    # Renorm sum =1
    s = a_new.sum()
    a_new /= s
    return a_new
```

```python
"""
Hybrid parametric + gradient+local-search for minimizing max-self‑convolution.
The algorithm:
1. Scan a few low‑dimensional families (exponential, symmetric exponential,
   beta PDFs, Gaussian, raised‑cosine) to get a good seed.
2. Refine the seed with an Adam optimiser on a smooth log‑sum‑exp surrogate
   of the max‑convolution, slowly increasing the sharpness parameter.
3. Finish with a short simulated‑annealing / block‑jitter hill‑climbing
   phase for fine‑grained improvement.
All steps respect the 1000‑second budget and operate on the simplex
(∑a = 1).  The returned list contains non‑negative heights suitable for
`evaluate_sequence`.
"""
import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution using a real FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]

def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the LSE surrogate w.r.t. a.
    ``w`` is a probability vector of length 2·n‑1.
    Returns a vector of length n.
    """
    n = a.size
    # correlation of reversed a with w
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    # extract the n entries that correspond to the gradient
    start = n - 1
    return 2.0 * corr[start:start + n]

def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()

def fast_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a non‑negative
    vector ``a`` (the sum is not assumed to be 1).
    Computed via FFT for speed.
    """
    n = a.size
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # linear convolution via FFT
    fft_len = _next_pow2(2 * n - 1)
    conv = _self_conv_fft(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)

# Families of parametric sequences
# ----------------------------------------------------------------------

def _exp_sequence(r: float, n: int) -> np.ndarray:
    """
    a_i ∝ r**i, 0 < r < 1, normalised to sum 1.
    """
    # use np.geomspace for stability when r is close to 1
    i = np.arange(n, dtype=np.float64)
    a = np.exp(i * np.log(r))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _sym_exp_sequence(r: float, n: int) -> np.ndarray:
    """
    a_i ∝ r**|i - (n‑1)/2|, normalised.
    Symmetric exponential that peaks at the centre.
    """
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(np.abs(i - centre) * np.log(r))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Beta‑pdf shaped sequence on the grid {0,…,n‑1}."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.power(x, alpha) * np.power(1.0 - x, beta)
    # protect against all‑zero (when alpha,beta large)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a

def _gaussian_sequence(sigma: float, n: int) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the array."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _hann_sequence(p: float, n: int) -> np.ndarray:
    """Raised‑cosine (Hann) window raised to power p."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    if p != 1.0:
        a = np.power(a, p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))  # up to 10 % of n
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

                     budget_s: float = 1000.0,

    surrogate of the max‑convolution and a final simulated‑annealing

    # ------------------------------------------------------------------

    # ------------------------------------------------------------------

    lr_end      = 0.02

        conv = _self_conv_fft(a, fft_len_self)   # length 2·n‑1
```

"""
Hybrid parametric + gradient + local‑search minimiser for the max‑self‑convolution
objective used by ``evaluate_sequence``.

The routine works in three phases (all respecting the 1000 s budget):

1. **Parametric sweep** – evaluates a handful of low‑dimensional families
   (exponential, symmetric exponential, Beta PDFs, Gaussians and raised‑cosine
   windows) on the simplex (∑a = 1) and keeps the best candidate.

2. **Adam optimiser on a smooth LSE surrogate** – the non‑smooth max
   ``max_i (a∗a)_i`` is approximated by a log‑sum‑exp with a sharpening factor
   γ that is increased exponentially.  The optimisation is performed on the
   log‑softmax (log‑weights) to enforce positivity and unit‑sum automatically.
   Adam with a linearly decaying learning‑rate is used, together with occasional
   sparsification, soft restarts and block‑jitter to escape plateaus.

3. **Hill‑climbing / simulated‑annealing** – starting from the best vector of
   phase 2, we repeatedly apply a random log‑normal block perturbation and
   accept improvements (or occasional uphill moves) under a slowly cooling
   temperature schedule.  This fine‑tunes the solution.

All helper utilities are defined at module level (no closures) and the
function returns a list of non‑negative heights ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution via a real FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the LSE surrogate w.r.t. the probability vector ``a``.
    ``w`` must be a probability vector of length ``2·n‑1`` (the same length as
    the linear convolution of ``a`` with itself).  Returns a vector of length
    ``n``.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    return 2.0 * corr[start:start + n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def fast_score(a: np.ndarray) -> float:
    """
    Exact value of ``evaluate_sequence`` for a non‑negative vector ``a``.
    Computed via FFT for speed (the same formula as the evaluator).
    """
    n = a.size
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 1e-12:
        return np.inf
    fft_len = _next_pow2(2 * n - 1)
    conv = _self_conv_fft(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Parametric families for quick seeding
# ----------------------------------------------------------------------


def _exp_sequence(r: float, n: int) -> np.ndarray:
    """a_i ∝ r**i, 0 < r < 1, normalised to sum 1."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(i * np.log(r))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _sym_exp_sequence(r: float, n: int) -> np.ndarray:
    """a_i ∝ r**|i‑(n‑1)/2|, normalised."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(np.abs(i - centre) * np.log(r))
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _beta_sequence(alpha: float, beta: float, n: int) -> np.ndarray:
    """Beta‑pdf shape on the discrete grid {0,…,n‑1}."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = np.power(x, alpha) * np.power(1.0 - x, beta)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _gaussian_sequence(sigma: float, n: int) -> np.ndarray:
    """Discrete Gaussian centred in the middle of the array."""
    centre = (n - 1) / 2.0
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _hann_sequence(p: float, n: int) -> np.ndarray:
    """Raised‑cosine (Hann) window raised to power p, then normalised."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    if p != 1.0:
        a = np.power(a, p)
    s = a.sum()
    if s == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))           # up to 10 % of the vector
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence`` under a 1000 s time budget.  The strategy combines
    a cheap parametric sweep, Adam optimisation on a smooth surrogate of the
    max‑convolution and a final simulated‑annealing hill‑climb.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine problem size (use the provided construction if any)
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim != 1 or a0.size == 0:
            n = 3809
            a0 = np.full(n, 1.0 / n)
        else:
            n = a0.size
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            a0 = a0 / s if s > 0 else np.full_like(a0, 1.0 / n)
    else:
        n = 3809                      # length that has proved useful historically
        a0 = np.full(n, 1.0 / n)      # uniform start

    # ------------------------------------------------------------------
    # 2) Keep track of the best candidate seen so far
    # ------------------------------------------------------------------
    best_seq = a0.copy()
    best_score = fast_score(best_seq)
    print(f"[init] length={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – cheap parametric sweep
    # ------------------------------------------------------------------
    # a) pure exponential decay
    for r in np.linspace(0.999, 0.99997, 25):
        cand = _exp_sequence(r, n)
        sc = fast_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"[sweep] exp r={r:.6f} → score={sc:.6f}")

    # b) symmetric exponential (peaked in the centre)
    for r in np.linspace(0.995, 0.9999, 20):
        cand = _sym_exp_sequence(r, n)
        sc = fast_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"[sweep] sym‑exp r={r:.6f} → score={sc:.6f}")

    # c) Beta‑pdf family
    alphas = np.linspace(0.2, 5.0, 12)
    betas = np.linspace(0.2, 5.0, 12)
    for a_ in alphas:
        for b_ in betas:
            cand = _beta_sequence(a_, b_, n)
            sc = fast_score(cand)
            if sc < best_score:
                best_score, best_seq = sc, cand
                print(f"[sweep] beta a={a_:.2f} b={b_:.2f} → score={sc:.6f}")

    # d) Gaussian widths
    for sigma in np.linspace(0.10 * n, 0.35 * n, 12):
        cand = _gaussian_sequence(sigma, n)
        sc = fast_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"[sweep] gauss σ={sigma:.1f} → score={sc:.6f}")

    # e) Raised‑cosine (Hann) powers
    for p in np.linspace(0.5, 3.0, 12):
        cand = _hann_sequence(p, n)
        sc = fast_score(cand)
        if sc < best_score:
            best_score, best_seq = sc, cand
            print(f"[sweep] hann p={p:.2f} → score={sc:.6f}")

    # ------------------------------------------------------------------
    # 4) Phase 2 – Adam optimisation on a smooth max surrogate
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(best_seq + eps)          # log‑weights (softmax gives the simplex)

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    t = 0

    # FFT length used throughout the optimisation
    fft_len_self = _next_pow2(2 * n - 1)

    # schedule parameters
    start = time.monotonic()
    total = deadline - start
    gamma_start = 0.5
    gamma_end = 250.0
    lr_start = 0.8
    lr_end = 0.02

    last_improve = start

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total                     # ∈[0,1]

        # current probability vector (holds the simplex)
        a = _softmax(x)

        # cheap self‑convolution (FFT)
        conv = _self_conv_fft(a, fft_len_self)     # length 2·n‑1

        # log‑sum‑exp surrogate (smooth max)
        gamma = gamma_start * (gamma_end / gamma_start) ** frac   # exponential rise
        shift = conv - conv.max()                  # stabilise exponentials
        w = np.exp(gamma * shift)
        w_sum = w.sum()
        w = w / w_sum if w_sum != 0.0 else np.full_like(w, 1.0 / w.size)

        # gradient w.r.t. a
        g_a = _gradient_wrt_a(a, w)

        # chain rule to the log‑weights (keeps the simplex)
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        x -= step

        # occasional sparsification (helps escape flat plateaus)
        if (t % 200) == 0:
            thr = 1e-6 + 1e-6 * frac
            low = a < thr
            if low.any():
                a[low] = 0.0
                s = a.sum()
                if s > 0.0:
                    a /= s
                    x = np.log(a + eps)

        # exact (non‑surrogate) objective – using the fact that Σa = 1
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score, best_seq = cur_score, a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Adam improvement → score={best_score:.6f}")

        # stagnation → soft restart
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best_seq + eps) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Adam restart (stagnation) – score={best_score:.6f}")

        # occasional global jitter (block perturbation)
        if rng.random() < 0.015:
            a_jit = _random_block_perturb(best_seq, rng)
            x = np.log(a_jit + eps)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] Adam block‑jitter – score={best_score:.6f}")

        # safety break if we are too close to the deadline
        if now + 0.5 > deadline:
            break

    # ------------------------------------------------------------------
    # 5) Phase 3 – Hill‑climb / simulated annealing for fine tuning
    # ------------------------------------------------------------------
    a = best_seq.copy()
    cur_score = best_score
    temp_start = 1e-3
    temp_end = 1e-6

    while time.monotonic() < deadline:
        now = time.monotonic()
        remaining = deadline - now
        # linear temperature decay
        frac = 1.0 - remaining / total
        T = temp_start * (1.0 - frac) + temp_end * frac

        # propose a random block perturbation
        a_prop = _random_block_perturb(a, rng)
        prop_score = fast_score(a_prop)

        # Metropolis acceptance
        if prop_score < cur_score or rng.random() < np.exp(-(prop_score - cur_score) / max(T, 1e-12)):
            a = a_prop
            cur_score = prop_score
            if cur_score < best_score - 1e-12:
                best_score, best_seq = cur_score, a.copy()
                print(f"[{now - start:6.2f}s] hill‑climb improvement → score={best_score:.6f}")

        # occasional larger reshuffle when idle
        if remaining < total * 0.1 and rng.random() < 0.05:
            a = _beta_sequence(rng.uniform(0.2, 5.0), rng.uniform(0.2, 5.0), n)
            cur_score = fast_score(a)

    # ------------------------------------------------------------------
    # 6) Return the best sequence (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    return [float(v) for v in best_seq]
