"""
Propose a non‑negative step‑function (list of heights) that minimizes
``evaluate_sequence``.

Algorithm – three phases within the given time budget:

1. **Random symmetric Dirichlet sampling** (global exploration).  
   Draw many symmetric probability vectors from Dirichlet distributions
   with a wide range of concentration parameters and keep the best.

2. **Projected Adam on a smooth surrogate of the max‑convolution**.  
   Starting from the best sampled vector we minimise the
   temperature‑controlled soft‑max of the self‑convolution.
   The gradient of the smooth surrogate is computed analytically via FFT.
   After every Adam step the vector is symmetrised and projected onto the
   probability simplex.

3. **(1 + 1) Evolution Strategy (ES) with step‑size adaptation**.  
   A single‑parent ES explores the neighbourhood of the current best
   solution.  The offspring is obtained by Gaussian mutation, symmetrised,
   projected onto the simplex and accepted if it improves the true
   objective.  The mutation step‑size follows the classic 1/5‑success rule.

All phases respect a global deadline (budget s – 5 s safety margin).  The
best vector encountered is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):            # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_J(a: np.ndarray) -> float:
    """
    Exact objective J = 2·n·max(a*a) for a vector that already sums to 1.
    """
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[:conv_len]
    return float(2.0 * n * conv.max())


def grad_smooth(a: np.ndarray,
                alpha: float,
                fft_len_self: int,
                fft_len_grad: int) -> tuple[np.ndarray, float]:
    """
    Gradient of the soft‑max smoothed max‑convolution.
    Returns (gradient, current max of a*a).
    """
    n = a.shape[0]
    conv_len = 2 * n - 1

    # ----- self‑convolution -----
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

    # ----- soft‑max weights -----
    c_max = conv.max()
    shifted = conv - c_max                     # for numerical stability
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum

    # ----- gradient of the smooth surrogate -----
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start_idx = w_rev.size - 1                 # = 2n‑2
    grad = 2.0 * full[start_idx:start_idx + n]

    return grad, c_max


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _symmetric_dirichlet(n: int,
                         rng: np.random.Generator,
                         concentration: float) -> np.ndarray:
    """
    Produce a symmetric probability vector of length ``n`` by drawing a
    Dirichlet vector of length ceil(n/2) and mirroring it.
    """
    m = (n + 1) // 2                      # size of the half‑vector
    half = rng.dirichlet(np.full(m, concentration))
    if n % 2 == 0:                        # even length → perfect mirror
        full = np.concatenate([half, half[::-1]])
    else:                                 # odd length → middle element not duplicated
        full = np.concatenate([half, half[-2::-1]])
    # Numerical drift may break the sum‑to‑one constraint
    return proj_simplex(full)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method consists of three phases:
    random symmetric Dirichlet sampling, projected Adam optimisation on a
    soft‑max surrogate, and a (1 + 1) Evolution Strategy with 1/5‑success
    step‑size adaptation.  The best vector found within the time budget is
    returned as a plain Python list.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (keep a safety margin)
    # ------------------------------------------------------------------
    safe_margin = 5.0
    deadline = time.monotonic() + max(0.0, float(budget_s) - safe_margin)

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            n = min(max(n0, 512), 4096)          # enforce 512 ≤ n ≤ 4096
        except Exception:
            n = 4096
    else:
        n = 4096

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for a * w_rev

    # ------------------------------------------------------------------
    # 2) Helper: keep the best candidate seen so far
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float('inf')

    def _update_best(candidate: np.ndarray, score: float, label: str = ""):
        nonlocal best_vec, best_score
        if score < best_score - 1e-12:
            best_score = score
            best_vec = candidate.copy()
            now = time.monotonic()
            elapsed = now - start_time
            print(f"[{elapsed:6.2f}s] new best {label}J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 0 – Random symmetric Dirichlet sampling (≈15 % of time)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    total_time = deadline - start_time
    phase0_deadline = start_time + 0.15 * total_time

    # deterministic baselines (uniform, linear)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _update_best(uniform, compute_J(uniform), label="uniform ")

    linear = np.arange(n, 0, -1, dtype=np.float64)
    linear /= linear.sum()
    _update_best(linear, compute_J(linear), label="linear  ")

    # random draws
    while time.monotonic() < phase0_deadline:
        # draw concentration log‑uniformly from [0.05, 5.0]
        log_c = rng.uniform(np.log(0.05), np.log(5.0))
        concentration = np.exp(log_c)
        cand = _symmetric_dirichlet(n, rng, concentration)
        score = compute_J(cand)
        _update_best(cand, score, label="sample  ")

    # ------------------------------------------------------------------
    # 4) Phase 1 – Projected Adam on the soft‑max surrogate (≈60 % of time)
    # ------------------------------------------------------------------
    phase1_deadline = phase0_deadline + 0.60 * total_time
    a = best_vec.copy()                     # start from the best sample

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    # scheduling parameters
    lr_start, lr_end = 0.25, 0.001
    alpha_start, alpha_end = 2.0, 8000.0   # temperature (soft‑max sharpness)

    it = 0
    last_improve = time.monotonic()
    while time.monotonic() < phase1_deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / total_time          # ∈[0,1]

        # learning‑rate schedule (linear decay)
        lr = lr_start * (1.0 - frac) + lr_end * frac

        # temperature schedule (linear increase)
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ----- gradient of the smooth surrogate -----
        grad, conv_max = grad_smooth(a, alpha, fft_len_self, fft_len_grad)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection + symmetrisation -----
        a = (a + a[::-1]) / 2.0          # enforce exact symmetry
        a = proj_simplex(a)

        # ----- exact objective -----
        cur_score = 2.0 * n * conv_max
        _update_best(a, cur_score, label="Adam   ")

        # ----- occasional jitter to leave plateaus -----
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)
            a = (a + a[::-1]) / 2.0
            a = proj_simplex(a)

        # ----- restart on stagnation (no improvement > 20 s) -----
        if now - last_improve > 20.0:
            a = best_vec.copy()
            # inject a tiny amount of multiplicative noise
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = (a + a[::-1]) / 2.0
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart (stagnation) – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Phase 2 – (1 + 1) ES with 1/5‑success step‑size adaptation (≈25 % of time)
    # ------------------------------------------------------------------
    phase2_deadline = deadline
    cur = best_vec.copy()
    cur_score = compute_J(cur)

    sigma = 0.15                       # initial mutation strength
    success_cnt = 0
    trial_cnt = 0
    window = 30                       # adaptation window

    while time.monotonic() < phase2_deadline:
        trial_cnt += 1
        offspring = cur + sigma * rng.normal(size=n)
        offspring = proj_simplex(offspring)
        offspring = (offspring + offspring[::-1]) / 2.0
        offspring = proj_simplex(offspring)  # second projection after symmetrisation

        off_score = compute_J(offspring)

        if off_score < cur_score:
            cur = offspring
            cur_score = off_score
            success_cnt += 1

            _update_best(cur, cur_score, label="(1+1)ES")

        # adaptation every ``window`` trials
        if trial_cnt % window == 0:
            success_rate = success_cnt / window
            if success_rate > 0.20:          # target 1/5 success
                sigma *= 1.5
            else:
                sigma *= 0.82
            success_cnt = 0

        # occasional re‑initialisation if ES makes no progress for a while
        if time.monotonic() - last_improve > 30.0:
            cur = best_vec.copy()
            cur_score = compute_J(cur)
            sigma = 0.15
            last_improve = time.monotonic()
            print(f"[{time.monotonic() - start_time:6.2f}s] ES restart – sigma reset")

    # ------------------------------------------------------------------
    # 6) Return the best vector discovered (clipped to admissible limits)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)   # safety – evaluate_sequence caps anyway
    return [float(x) for x in best_vec]
