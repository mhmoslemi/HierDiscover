def _simplex_projection(v):
    # ensures v >=0 and sum=1
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function that minimises

    2·n·max(conv(a,a)) / (sum(a)²)

using a two‑phase Adam optimiser on a simplex‑parametrised vector.
Phase 1 (variance) drives the self‑convolution towards a flat shape,
Phase 2 (max‑subgradient) reduces the actual peak.
A large number of steps (≈20000) makes the theoretical lower bound
≈2 n/(2 n‑1) ≈ 1 achievable.  The optimiser includes
stagnation restarts, occasional jitter and a simple symmetry
enforcement to speed up convergence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – always kept at top level (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _init_gaussian(n: int) -> np.ndarray:
    """Return a symmetric Gaussian‑shaped probability vector of length n."""
    i = np.arange(n, dtype=np.float64)
    sigma = n / 6.0
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total > 0:
        a /= total
    else:
        a = np.full_like(a, 1.0 / n)
    return a


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Enforce mirror symmetry and renormalise to sum‑1."""
    v = (v + v[::-1]) * 0.5
    s = v.sum()
    if s > 0:
        v /= s
    else:
        v = np.full_like(v, 1.0 / v.size)
    return v

# Main optimisation routine
# ----------------------------------------------------------------------

    """
    Search for a non‑negative step‑function (list of heights) that gives a
    low value of `evaluate_sequence`.  The function respects the given
    time budget and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    total_time = deadline - start_time

    # ------------------------------------------------------------------
    # 1) Choose a length (large enough to make the theoretical bound ≈1)
    # ------------------------------------------------------------------
    # Heuristic: use a larger n when the budget allows more iterations.
    if total_time > 400.0:
        n = 20000
    elif total_time > 200.0:
        n = 12000
    else:
        n = 8000

    # ------------------------------------------------------------------
    # 2) Initialise a (probability vector) – use a symmetric Gaussian
    # ------------------------------------------------------------------
    a0 = _init_gaussian(n)
    # If a previous construction `height_sequence_1` exists, embed it.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        seq = np.asarray(init_seq, dtype=np.float64).ravel()
        if seq.size > 0:
            # centre‑align the old sequence within the new length
            embed = np.zeros(n, dtype=np.float64)
            start = max(0, (n - seq.size) // 2)
            end = start + min(seq.size, n)
            embed[start:end] = seq[: end - start]
            embed = np.clip(embed, 0.0, None)
            s = embed.sum()
            if s > 0:
                embed /= s
                a0 = embed
    # Parameterise by logits to keep the simplex constraint automatic.
    x = np.log(a0 + 1e-12)                     # log‑weights

    # ------------------------------------------------------------------
    # 3) FFT lengths (re‑used each iteration)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)       # for a*a
    # ------------------------------------------------------------------
    # 4) Adam optimiser state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_iter = 0

    # ------------------------------------------------------------------
    # 5) Tracking the best solution seen so far
    # ------------------------------------------------------------------
    a = _softmax(x)
    a = _symmetrize(a)
    c = _self_conv(a, fft_len_self)
    sum_a = a.sum()
    best = a.copy()
    best_score = 2.0 * n * c.max() / (sum_a ** 2)
    last_improve = start_time

    # ------------------------------------------------------------------
    # 6) Main optimisation loop
    # ------------------------------------------------------------------
    # Phase schedule:
    #   – first 50 % of the budget: pure variance minimisation
    #   – remaining time: gradually mix in max‑subgradient (linear fade‑out)
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # --------------------------------------------------------------
        # Obtain current a (softmax → symmetric → renormalise)
        # --------------------------------------------------------------
        a = _softmax(x)
        a = _symmetrize(a)          # symmetry helps flatten the convolution
        sum_a = a.sum()
        if sum_a == 0.0:
            a = np.full_like(a, 1.0 / n)
            sum_a = 1.0

        # --------------------------------------------------------------
        # Self‑convolution and score
        # --------------------------------------------------------------
        c = _self_conv(a, fft_len_self)               # length 2·n‑1
        cur_score = 2.0 * n * c.max() / (sum_a ** 2)

        # --------------------------------------------------------------
        # Record improvement
        # --------------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # --------------------------------------------------------------
        # Determine blending weight α between variance (α=1) and max (α=0)
        # --------------------------------------------------------------
        if elapsed < 0.5 * total_time:
            alpha = 1.0
        else:
            # linear decay from 1 → 0 over the second half of the budget
            alpha = max(0.0, 1.0 - (elapsed - 0.5 * total_time) / (0.5 * total_time))

        # --------------------------------------------------------------
        # Gradient of variance term: J₁ = Σ (c - avg)²
        # --------------------------------------------------------------
        if alpha > 0.0:
            avg = 1.0 / (2 * n - 1)                     # because sum(a)=1
            r = c - avg
            g_var = 2.0 * _gradient_wrt_a(a, r)         # shape (n,)
        else:
            g_var = np.zeros_like(a)

        # --------------------------------------------------------------
        # Gradient of max term: J₂ = max(c) (sub‑gradient)
        # --------------------------------------------------------------
        if alpha < 1.0:
            max_val = c.max()
            # treat all indices within a tiny tolerance as maximisers
            idxs = np.where(np.abs(c - max_val) <= 1e-12)[0]
            w = np.zeros_like(c)
            if idxs.size > 0:
                w[idxs] = 1.0 / idxs.size                 # convex combination of sub‑gradients
            else:
                w[np.argmax(c)] = 1.0
            g_max = 2.0 * _gradient_wrt_a(a, w)           # shape (n,)
        else:
            g_max = np.zeros_like(a)

        # --------------------------------------------------------------
        # Blended gradient w.r.t. a
        # --------------------------------------------------------------
        g_a = alpha * g_var + (1.0 - alpha) * g_max

        # --------------------------------------------------------------
        # Chain rule to obtain gradient w.r.t. logits x (softmax Jacobian)
        # --------------------------------------------------------------
        avg_g = np.dot(a, g_a)               # ⟨a,∂F/∂a⟩
        g_x = a * (g_a - avg_g)              # ∂F/∂x

        # --------------------------------------------------------------
        # Adam update
        # --------------------------------------------------------------
        t_iter += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t_iter)
        v_hat = v / (1.0 - beta2 ** t_iter)

        # learning‑rate schedule (linear decay)
        lr = 0.5 * (1.0 - elapsed / total_time) + 0.02 * (elapsed / total_time)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        x -= step
        # keep logits in a reasonable numerical range
        x = np.clip(x, -30.0, 30.0)

        # --------------------------------------------------------------
        # Stagnation handling – restart around the best solution
        # --------------------------------------------------------------
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.4, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

        # --------------------------------------------------------------
        # Occasional global jitter to escape shallow minima
        # --------------------------------------------------------------
        if rng.random() < 0.015:
            width = rng.integers(1, max(2, n // 30))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.25))
            a_jit = best.copy()
            a_jit[start:start + width] *= factor
            a_jit = np.clip(a_jit, 0.0, None)
            s = a_jit.sum()
            if s > 0:
                a_jit /= s
            else:
                a_jit = np.full_like(a_jit, 1.0 / n)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t_iter = 0

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
