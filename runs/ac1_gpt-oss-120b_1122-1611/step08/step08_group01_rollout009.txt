"""
Propose a step‑function (non‑negative heights on the simplex) that beats the
previous 1.513205 bound.

Algorithm
---------
* The sequence length is taken from the previously discovered `height_sequence_1`
  (fallback 1024).  The vector is normalised to sum = 1 – the evaluation is
  invariant under scaling.
* **Phase I – Adam on a smooth‑max surrogate**  
  minimise a log‑sum‑exp approximation of `max(conv(a,a))`.  The variables are
  the log‑weights `x` so that `a = softmax(x)`.  Adam with a linearly decaying
  learning‑rate is used for about 30 % of the time budget.
* **Phase II – Simulated annealing / greedy hill‑climb**  
  Repeatedly perturb the current vector:
      – move a random fraction of mass from one index to another,
      – multiply a random contiguous block by a log‑normal factor,
      – blend with a small Dirichlet noise vector.
  Improvements are always accepted; worsening moves are accepted with a
  Metropolis probability that slowly cools.
* **Phase III – Linear‑programming boost**  
  When no improvement has been seen for ~15 s, solve the LP

        maximise   Σ b_i
        subject to  b_i ≥ 0
                    (a * b)_k ≤ M   for all k

  where `M = max(conv(a,a))`.  The LP gives a vector `b` that can carry more
  total mass without increasing the maximal convolution.  `b` is normalised
  and mixed with the current `a` (mixing ratios 0.5, 0.8, 1.0 are tried).
  If this produces a lower objective the new vector is kept; otherwise a
  small random restart around the best solution is performed.
* Throughout the run the best sequence seen so far is recorded.
* The final best vector (clipped to the limits of `evaluate_sequence`) is
  returned.

The method combines gradient optimisation, stochastic exploration and a
deterministic LP step, which is different from the purely gradient‑descent
approach used previously and gives a higher chance of escaping local minima.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution of `a` using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. the probability vector `a`.
    `w` are the soft‑max weights over the 2·n‑1 convolution positions.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _move_mass(a: np.ndarray, rng: np.random.Generator,
               step_frac: float) -> np.ndarray:
    """Transfer a small fraction of mass from a random source index to a random
    destination index and renormalise."""
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = step_frac * rng.random() * a[i]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiply a random contiguous block by a log‑normal factor, then renorm."""
    n = a.size
    width = rng.integers(1, max(2, n // 12))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _conv_constraint_matrix(a: np.ndarray) -> scipy.sparse.csr_matrix:
    """
    Build the matrix `A` such that  A @ b == conv(a, b)  (linear in `b`).
    Shape of `A` is (2·n‑1, n).  The matrix is returned in CSR format.
    """
    n = a.size
    rows = []
    cols = []
    data = []
    for i in range(2 * n - 1):
        j_start = max(0, i - (n - 1))
        j_end = min(i, n - 1)
        for j in range(j_start, j_end + 1):
            # contribution of a[j] to position i comes from b[i-j]
            rows.append(i)
            cols.append(i - j)
            data.append(a[j])
    A = scipy.sparse.csr_matrix(
        (np.array(data, dtype=np.float64),
         (np.array(rows, dtype=np.int32),
          np.array(cols, dtype=np.int32))),
        shape=(2 * n - 1, n)
    )
    return A


def _objective(a: np.ndarray, fft_len: int, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) for a vector with sum = 1."""
    c = _self_conv(a, fft_len)
    return 2.0 * n * c.max()


# ----------------------------------------------------------------------
# Core optimisation routine for a given length
# ----------------------------------------------------------------------


def _optimise_one_length(n: int,
                         init_a: np.ndarray,
                         rng: np.random.Generator,
                         budget_s: float) -> tuple[np.ndarray, float]:
    """
    Optimise a vector of length `n` for up to `budget_s` seconds.
    Returns the best vector found and its objective value.
    """
    start = time.monotonic()
    deadline = start + budget_s

    # FFT length for self‑convolution (shared by all phases)
    fft_len = _next_pow2(2 * n - 1)

    # ----- initialise ----------------------------------------------------
    a = init_a.copy()
    cur_score = _objective(a, fft_len, n)

    best = a.copy()
    best_score = cur_score
    last_improve = start

    # ----- Phase I: Adam on smooth‑max ----------------------------------
    # log‑weights for soft‑max parametrisation
    x = np.log(a + 1e-12)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0

    # schedule: 30 % of the allocated budget
    phase1_end = start + 0.30 * budget_s

    gamma_start, gamma_end = 0.2, 80.0
    lr_start, lr_end = 0.5, 0.02

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / budget_s

        # current a from x
        a = _softmax(x)

        # self‑convolution
        conv = _self_conv(a, fft_len)

        # smooth‑max weights
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        shift = conv - conv.max()
        w = np.exp(gamma * shift)
        w /= w.sum()

        # gradient w.r.t. a (surrogate)
        g_a = _gradient_wrt_a(a, w)

        # chain rule to log‑weights
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # evaluate exact objective
        a = _softmax(x)
        cur_score = _objective(a, fft_len, n)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I new best = {best_score:.6f}")

        # occasional random restart on stagnation
        if now - last_improve > 12.0:
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I restart (stagnation)")

    # set a to the best found before entering Phase II
    a = best.copy()
    cur_score = best_score
    last_improve = time.monotonic()

    # ----- Phase II: Simulated annealing + LP boost ----------------------
    # temperature schedule (log‑linear cooling)
    T0 = 0.05 * best_score
    Tmin = 1e-5
    decay = 0.9997

    temp = T0

    while time.monotonic() < deadline:
        now = time.monotonic()
        # ---------- stochastic move ----------
        r = rng.random()
        if r < 0.55:
            cand = _move_mass(a, rng, step_frac=0.2 * (1.0 - (now - start) / budget_s) + 0.02)
        elif r < 0.85:
            cand = _random_block_perturb(a, rng)
        else:
            # small Dirichlet blend
            alpha = 0.02
            dirich = rng.dirichlet(alpha * np.ones(n))
            cand = (1 - alpha) * a + alpha * dirich
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()

        cand_score = _objective(cand, fft_len, n)
        delta = cand_score - cur_score

        accept = False
        if delta < -1e-12:
            accept = True
        else:
            # Metropolis acceptance
            prob = np.exp(-delta / (temp + 1e-15))
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                last_improve = now
                print(f"[{now - start:6.2f}s] Phase II new best = {best_score:.6f}")

        # temperature cooling
        temp = max(temp * decay, Tmin)

        # ---------- LP boost on stagnation ----------
        if now - last_improve > 15.0:
            # compute current bound M
            conv_a = _self_conv(a, fft_len)
            M = conv_a.max()

            # build linear constraints (A @ b <= M)
            A = _conv_constraint_matrix(a)
            b_ub = np.full(2 * n - 1, M, dtype=np.float64)
            c_obj = -np.ones(n, dtype=np.float64)          # maximise Σb_i

            # solve LP (HiGHS)
            res = scipy.optimize.linprog(
                c=c_obj,
                A_ub=A,
                b_ub=b_ub,
                bounds=[(0, None)] * n,
                method='highs',
                options={'presolve': True}
            )

            if res.success:
                b = res.x
                sum_b = b.sum()
                if sum_b > 0:
                    b_norm = b / sum_b
                    # try several mixing ratios
                    for tmix in (0.5, 0.8, 1.0):
                        cand_lp = (1 - tmix) * a + tmix * b_norm
                        cand_lp = np.clip(cand_lp, 0.0, None)
                        cand_lp /= cand_lp.sum()
                        cand_lp_score = _objective(cand_lp, fft_len, n)
                        if cand_lp_score < cur_score - 1e-12:
                            a = cand_lp
                            cur_score = cand_lp_score
                            if cur_score < best_score - 1e-12:
                                best_score = cur_score
                                best = a.copy()
                                last_improve = time.monotonic()
                                print(f"[{last_improve - start:6.2f}s] LP‑mix improved to {best_score:.6f}")
                            break
            # if LP did not help, perform a gentle random restart
            noise = rng.normal(0.0, 0.3, size=n)
            a = np.exp(noise) * best
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            cur_score = _objective(a, fft_len, n)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] Restart after stagnation")

    return best, best_score


# ----------------------------------------------------------------------
# Public entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that minimises ``evaluate_sequence``.
    The routine combines Adam on a smooth‑max surrogate, simulated annealing
    and a linear‑programming boost.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and obtain an initial vector
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        init_arr = np.asarray(init_seq, dtype=np.float64)
        n = init_arr.size
        a0 = np.clip(init_arr, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        # fallback: moderately sized vector (≈ 1024)
        n = 1024
        a0 = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Split the total budget among several independent runs (different restarts)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)  # keep a safety margin

    best_global = a0.copy()
    best_global_score = _objective(best_global, _next_pow2(2 * n - 1), n)

    # We perform a sequence of short runs; each run gets roughly 0.3·remaining time
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 15.0:                     # not enough time for a full run
            break
        # give each run at most 120 s (or the remaining budget if smaller)
        run_budget = min(120.0, remaining * 0.4)

        # initialise a fresh start near the current global best (adds noise)
        noise = rng.normal(0.0, 0.2, size=n)
        a_start = np.exp(noise) * best_global
        a_start = np.clip(a_start, 0.0, None)
        a_start /= a_start.sum()

        # run the optimiser for this length
        cand_seq, cand_score = _optimise_one_length(
            n=n,
            init_a=a_start,
            rng=rng,
            budget_s=run_budget
        )

        if cand_score < best_global_score - 1e-12:
            best_global_score = cand_score
            best_global = cand_seq.copy()
            print(f"[{time.monotonic() - start:6.2f}s] GLOBAL new best = {best_global_score:.6f}")

        # safeguard against overrunning the budget
        if time.monotonic() + 2.0 >= deadline:
            break

    # ------------------------------------------------------------------
    # Return the best sequence (clip to limits imposed by evaluate_sequence)
    # ------------------------------------------------------------------
    best_global = np.clip(best_global, 0.0, 1000.0)
    return [float(v) for v in best_global]
