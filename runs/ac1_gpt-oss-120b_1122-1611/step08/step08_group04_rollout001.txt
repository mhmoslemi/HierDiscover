Hybrid multi‑scale optimiser for the self‑convolution minimisation problem.

The routine works in several stages:

1.  Choose a modest length (≈1024) and initialise the step heights.
    We try a few simple parametric families (powers of the distance

    (non‑negative, unit sum) with a Euclidean projection.

3.  A short stochastic local‑search phase (random pairwise mass swaps)
    is performed until the time budget is exhausted.

The whole search respects the 1000 s time limit (a 5 s safety margin is
kept).  The best sequence found is clipped to the limits imposed by
`evaluate_sequence` and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (real‑valued)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (sum(a) is not forced)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    `evaluate_sequence`.  The method works on several length scales,
    combines a max‑convolution sub‑gradient with a variance‑flattening
    gradient, and finishes with a short stochastic local search.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start + total_budget

    # --------------------------------------------------------------
    # Length schedule – start small, double while we have time
    # --------------------------------------------------------------
    length_schedule = [1024, 2048, 4096, 8192]

    best_a = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Helper to build an initial candidate for a given length `n`
    # --------------------------------------------------------------
    def _initial_candidate(n: int) -> np.ndarray:
        # start from a previously known construction if available
        if "height_sequence_1" in globals():
            base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            base = np.clip(base, 0.0, None)
            if base.sum() == 0.0:
                base[:] = 1.0 / base.size
            if base.size != n:
                old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
                new_x = np.linspace(0.0, 1.0, n, endpoint=False)
                a0 = np.interp(new_x, old_x, base)
                a0 = np.clip(a0, 0.0, None)
                a0 = a0 / a0.sum()
            else:
                a0 = base.copy()
                a0 = a0 / a0.sum()
        elif best_a is not None:
            # upscale previous best
            old_n = best_a.shape[0]
            old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a0 = np.interp(new_x, old_x, best_a)
            a0 = np.clip(a0, 0.0, None)
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n, dtype=np.float64)

        return a0

    # --------------------------------------------------------------
    # Main loop over increasing lengths
    # --------------------------------------------------------------
    for stage, n in enumerate(length_schedule):
        now = time.monotonic()
        time_left = deadline - now
        if time_left < 20.0:               # not enough time for another stage
            break

        # Allocate about 70 % of the remaining time to this stage,
        # leaving the rest for later (or final random search).
        stage_budget = time_left * 0.7
        stage_deadline = now + stage_budget

        # -------------------- initialise candidate --------------------
        a = _initial_candidate(n)

        # ------------------ quick parametric sweep --------------------
        # we test a small family a_i ∝ ((i+1)*(n-i))^p for several p
        base_idx = np.arange(n)
        weight_base = (base_idx + 1) * (n - base_idx)          # >0 everywhere
        for p in np.linspace(-0.6, 0.3, 13):
            w = weight_base.astype(np.float64) ** p
            w = np.clip(w, 0.0, None)
            if w.sum() == 0.0:
                continue
            a_p = w / w.sum()
            sc = _objective_score(a_p)
            if sc < best_score:
                best_score = sc
                best_a = a_p.copy()
                a = a_p.copy()
                print(f"[{time.monotonic() - start:6.2f}s] length {n} param p={p:+.3f} new best {best_score:.6f}")

        # also try a tiny random jitter around the current a
        for _ in range(3):
            jitter = rng.normal(0.0, 0.001, size=n)
            a_j = _project_simplex(a + jitter)
            sc = _objective_score(a_j)
            if sc < best_score:
                best_score = sc
                best_a = a_j.copy()
                a = a_j.copy()
                print(f"[{time.monotonic() - start:6.2f}s] length {n} jitter init best {best_score:.6f}")

        # --------------------- gradient descent --------------------
        step = 0.05               # initial step size on the simplex
        lambda_var = 0.25         # weight for the variance‑flattening term
        it = 0
        while time.monotonic() < stage_deadline:
            it += 1

            # self‑convolution
            conv = _conv_self(a)
            kmax = int(np.argmax(conv))

            # ---- sub‑gradient of the max‑convolution (active offset) ----
            idx = np.arange(n)
            j = kmax - idx
            mask = (j >= 0) & (j < n)
            grad_max = np.zeros_like(a)
            grad_max[mask] = 2.0 * a[j[mask]]

            # ---- variance‑flattening gradient (correlation with deviation) ----
            # average of the convolution (constant because sum(a)=1)
            avg = (np.sum(a) ** 2) / (2 * n - 1)          # = 1/(2n-1) after normalisation
            e = conv - avg                                 # deviation from average
            # correlation e ⋆ a  (equivalently convolution of e with a reversed)
            fft_len_corr = _next_pow2(3 * n - 2)
            e_fft = np.fft.rfft(e, n=fft_len_corr)
            a_rev_fft = np.fft.rfft(a[::-1], n=fft_len_corr)
            corr_full = np.fft.irfft(e_fft * a_rev_fft, n=fft_len_corr)
            # the central slice corresponds to shifts 0..n‑1
            grad_var = 2.0 * corr_full[n - 1:n - 1 + n]

            # combined direction
            grad = grad_max + lambda_var * grad_var

            # projected gradient step on the probability simplex
            a_new = _project_simplex(a - step * grad)

            # evaluate
            sc = _objective_score(a_new)

            # ----------- adaptive step / λ handling ----------
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = a_new.copy()
                a = a_new
                step *= 1.04            # modest step‑size increase on success
                lambda_var *= 0.97      # slowly reduce the variance weight
                # occasional larger jitter after a success
                if rng.random() < 0.02:
                    a = _project_simplex(a + rng.normal(0.0, 0.001, size=n))
                # print occasional progress
                if it % 250 == 0:
                    print(f"[{time.monotonic() - start:6.2f}s] n={n} it={it} best={best_score:.6f} step={step:.5f} λ={lambda_var:.4f}")
            else:
                step *= 0.70            # shrink step on failure
                # occasionally inject a tiny random perturbation to escape flat regions
                if step < 1e-7 or rng.random() < 0.003:
                    a = _project_simplex(a + rng.normal(0.0, 0.0015, size=n))
                    step = max(step, 0.01)

            # very small step → stop early
            if step < 5e-8:
                break

        # after the refinement keep the best for the next resolution
        if best_a is not None and best_a.shape[0] == n:
            a = best_a.copy()
        else:
            a = _project_simplex(a)    # ensure we are on the simplex

        # --------------------------------------------------------------
        # Brief stochastic local search on the current length (if time)
        # --------------------------------------------------------------
        ls_deadline = deadline - 1.0   # keep ~1 s buffer for returning
        while time.monotonic() < ls_deadline:
            # pick two distinct indices and transfer a small random amount
            i, j = rng.choice(n, size=2, replace=False)
            eps = rng.random() * 1e-4
            if best_a[i] < eps:
                continue
            cand = best_a.copy()
            cand[i] -= eps
            cand[j] += eps
            # sum is preserved, only need to check non‑negativity
            sc = _objective_score(cand)
            if sc < best_score:
                best_score = sc
                best_a = cand
                # optionally enlarge step for next gradient phase (not used here)
        # end of stage -------------------------------------------------

    # ------------------------------------------------------------------
    # Return the best sequence (clip to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    if best_a is None:
        # fallback – uniform vector of length 256 (should never happen)
        best_a = np.full(256, 1.0 / 256, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
