"""
Hybrid optimizer for the self‑convolution problem.

1. **Down‑scale** the problem to a moderate size (≈1500) to explore fast.
2. **Phase 1 – Adam** on a soft‑max surrogate of the max‑convolution
   (smooth approximation of the ∞‑norm).  A slow temperature ramp and
   occasional jitter quickly locate a good basin.
3. **Phase 2 – Projected sub‑gradient descent** on the exact max‑convolution.
   The step‑size is adapted after each success/failure and the iterate is
   always projected onto the probability simplex.
4. **Phase 3 – Conditional‑gradient (LP) step**:
   For the current iterate `a` we solve the linear programme

       maximise   1ᵀ g
       subject to  a * g ≤ M·1 ,   g ≥ 0

   where `M = max(a*a)`.  The solution `g` can have a total mass
   `Σg > 1`; after normalising (`ĝ = g / Σg`) the bound on the
   convolution is reduced to `M / Σg`.  A small line‑search mixes
   `a` and `ĝ` and keeps the best candidate.
5. **Upscale** the best small‑size solution to a larger dimension
   (≈4000) and repeat phases 2‑3 to exploit the extra degrees of freedom.

The algorithm respects the 1000 s time budget, prints occasional progress
messages and finally returns the best sequence (clipped to the limits of
`evaluate_sequence`).  All helper functions are defined at top level.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Real‑valued self‑convolution a * a using FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value returned by `evaluate_sequence` when sum(a)=1."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x ≥ 0, Σx = 1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """One projected sub‑gradient step on max‑convolution."""
    n = a.shape[0]
    conv = _conv_self(a)
    kmax = int(np.argmax(conv))
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    return _project_simplex(a - step * grad)


# ----------------------------------------------------------------------
# Linear‑programming step (conditional gradient)
# ----------------------------------------------------------------------


def _lp_max_sum(a: np.ndarray, M: float) -> tuple[np.ndarray, float] | tuple[None, None]:
    """
    Solve   maximise Σ g_j   subject to   conv(a,g) ≤ M,   g ≥ 0.
    Returns (g, Σg) or (None,None) on failure.
    The constraint matrix is the Toeplitz matrix of `a`.
    """
    n = a.shape[0]
    # Build the sparse constraint matrix A (size (2n‑1) × n)
    # Row indices are i+j for i,j∈[0,n)
    i = np.arange(n)[:, None]
    j = np.arange(n)[None, :]
    rows = (i + j).ravel()
    cols = np.repeat(np.arange(n), n)
    data = np.tile(a, n)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # linprog solves a minimisation problem → minimise −Σg
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 30},
    )
    if not res.success:
        return None, None
    g = res.x
    return g, float(g.sum())


# ----------------------------------------------------------------------
# Helper to up‑sample a distribution
# ----------------------------------------------------------------------


def _upscale(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation to a new length, followed by simplex projection."""
    old_n = a.shape[0]
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    a_new = np.interp(new_x, old_x, a)
    a_new = np.maximum(a_new, 0.0)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start + total_budget

    # ------------------------------------------------------------------
    # Choose a moderate size for the exploratory phases
    # ------------------------------------------------------------------
    small_n = 1500

    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if base.size != small_n:
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, small_n, endpoint=False)
            base = np.interp(new_x, old_x, base)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / small_n
        a = base / base.sum()
    else:
        a = np.full(small_n, 1.0 / small_n, dtype=np.float64)

    best_a = a.copy()
    best_score = _objective_score(a)
    print(f"[{time.monotonic() - start:6.2f}s] initial small‑size score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smooth surrogate of max(conv)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(a + eps)          # logits → simplex via soft‑max
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    lr = 0.35
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    beta_start = 5.0
    beta_end = 1500.0
    beta_ramp_iters = 4000

    RESTART_WAIT = 30.0
    last_improve = start

    fft_len_conv = _next_pow2(2 * small_n - 1)

    phase1_end = start + total_budget * 0.35
    it = 0
    while time.monotonic() < phase1_end:
        it += 1

        a = _softmax_logits(x)                     # current simplex point
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * small_n - 1]

        cur_max = conv_full.max()
        cur_score = 2.0 * small_n * cur_max          # denominator is 1 (Σa = 1)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(
                f"[{last_improve - start:6.2f}s] phase1 new best {best_score:.6f}"
            )

        # temperature schedule
        beta = (
            beta_end
            if it > beta_ramp_iters
            else beta_start + (beta_end - beta_start) * it / beta_ramp_iters
        )

        scaled = beta * conv_full
        lse = logsumexp(scaled)
        w = np.exp(scaled - lse)                     # soft‑max over conv entries

        # gradient of surrogate w.r.t. a (correlation w ⋆ a)
        fft_len_corr = _next_pow2(3 * small_n - 2)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)

        grad_a = 2.0 * corr_full[small_n - 1 : small_n - 1 + small_n]

        # chain rule to logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter
        if it % 500 == 0:
            x += rng.normal(0.0, 0.018, size=small_n)

        # hard‑restart if stagnant
        if time.monotonic() - last_improve > RESTART_WAIT:
            x = np.log(best_a + eps) + rng.normal(0.0, 0.13, size=small_n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase1 restart")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    step = 0.2
    subgrad_end = start + total_budget * 0.65
    while time.monotonic() < subgrad_end:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        idx = np.arange(small_n)
        j = kmax - idx
        mask = (j >= 0) & (j < small_n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        a_cand = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_cand)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_cand.copy()
            a = a_cand
            step *= 1.07                      # reward success
            print(
                f"[{time.monotonic() - start:6.2f}s] subgrad improvement {best_score:.6f}"
            )
        else:
            step *= 0.68                       # penalise failure
            # occasional jitter to escape plateaus
            if step < 5e-5 or rng.random() < 0.001:
                jitter = rng.normal(0.0, 0.0025, size=small_n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.015)

    # ------------------------------------------------------------------
    # Phase 3 – Conditional‑gradient (LP) refinement + final sub‑grad
    # ------------------------------------------------------------------
    a = best_a.copy()
    lp_calls = 0
    max_lp_calls = 7                       # keep LP workload modest
    final_end = deadline

    # pre‑compute indices for the LP matrix (size independent of a)
    i = np.arange(small_n)[:, None]
    j = np.arange(small_n)[None, :]
    lp_rows = (i + j).ravel()
    lp_cols = np.repeat(np.arange(small_n), small_n)

    while time.monotonic() < final_end:
        # --------------------------------------------------------------
        # Occasionally try a linear‑programming step
        # --------------------------------------------------------------
        try_lp = (lp_calls < max_lp_calls) and (rng.random() < 0.25)
        if try_lp:
            M = _conv_self(a).max()
            # Build constraint matrix with current a
            data = np.tile(a, small_n)
            A_ub = sp.csr_matrix(
                (data, (lp_rows, lp_cols)), shape=(2 * small_n - 1, small_n)
            )
            b_ub = np.full(2 * small_n - 1, M, dtype=np.float64)
            c = -np.ones(small_n, dtype=np.float64)

            res = linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=[(0, None)] * small_n,
                method="highs",
                options={"presolve": True, "time_limit": 20},
            )
            lp_calls += 1
            if res.success:
                g = res.x
                sum_g = float(g.sum())
                if sum_g > 1.0001:
                    g_hat = g / sum_g
                    # line‑search over a few mixing coefficients
                    best_mix_score = best_score
                    best_mix = a.copy()
                    for t in (0.3, 0.5, 0.7, 0.9, 1.0):
                        cand = (1.0 - t) * a + t * g_hat
                        cand = _project_simplex(cand)
                        sc = _objective_score(cand)
                        if sc < best_mix_score:
                            best_mix_score = sc
                            best_mix = cand
                    if best_mix_score < best_score - 1e-12:
                        best_score = best_mix_score
                        best_a = best_mix.copy()
                        a = best_mix.copy()
                        print(
                            f"[{time.monotonic() - start:6.2f}s] LP‑mix improvement {best_score:.6f}"
                        )
                        continue  # skip a sub‑gradient step this round
        # --------------------------------------------------------------
        # One sub‑gradient step (fallback / complement)
        # --------------------------------------------------------------
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))
        idx = np.arange(small_n)
        j = kmax - idx
        mask = (j >= 0) & (j < small_n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        a_cand = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_cand)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_cand.copy()
            a = a_cand
            step *= 1.06
            print(
                f"[{time.monotonic() - start:6.2f}s] subgrad (post‑LP) improvement {best_score:.6f}"
            )
        else:
            step *= 0.65
            if step < 4e-5 or rng.random() < 0.001:
                jitter = rng.normal(0.0, 0.002, size=small_n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.012)

    # ------------------------------------------------------------------
    # Upscale to a larger size and finish with a short sub‑gradient pass
    # ------------------------------------------------------------------
    large_n = max(3800, small_n * 2)                # a few thousand entries
    a_large = _upscale(best_a, large_n)
    best_score = _objective_score(a_large)
    a = a_large.copy()
    step = 0.15
    upscale_end = deadline
    while time.monotonic() < upscale_end:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))
        idx = np.arange(large_n)
        j = kmax - idx
        mask = (j >= 0) & (j < large_n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        a_cand = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_cand)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_cand.copy()
            a = a_cand
            step *= 1.04
        else:
            step *= 0.66
            if step < 3e-5 or rng.random() < 0.001:
                jitter = rng.normal(0.0, 0.0015, size=large_n)
                a = _project_simplex(a + jitter)
                step = max(step, 0.01)

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
