Implement projection onto simplex:

def simplex_projection(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

That's O(n log n). Acceptable.

def propose_candidate(seed=12345,budget_s=1000.0,**kwargs):
    rng = np.random.default_rng(seed)
    # Determine n
    init_seq = globals().get('height_sequence_1')
    if init_seq and isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        n = a0.shape[0]
    else:
        n = 2048
    # Normalize to sum=1
    if init_seq:
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s == 0:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = a0 / s
    else:
        a0 = np.full(n, 1.0 / n)
    # initialize log-weights
    x = np.log(a0 + 1e-12)
    # Precompute FFT length
    fft_len_self = _next_pow2(2 * n - 1)
    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # track best
    best_a = a0.copy()
    best_score = compute_score(best_a)
    last_improve = time.monotonic()
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)
    total_time = deadline - start
    # parameters
    phase1_end = start + 0.7 * total_time   # 70% time
    # Adam phase
    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total_time
        # compute a
        a = _softmax(x)
        # Conv via FFT
        c = _self_conv_fft(a, fft_len_self)
        # smooth max weight
        gamma = 0.2 + (200.0 - 0.2) * frac
        c_shift = c - np.max(c)
        w = np.exp(gamma * c_shift)
        w /= w.sum()
        # gradient wrt a
        grad_a = _gradient_wrt_a(a, w)
        # chain to x
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)
        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = 0.5 * (1.0 - frac) + 0.01 * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step
        # Evaluate exact
        cur_score = compute_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best exact score = {best_score:.6f}")
        # Stagnation handling

            # soft restart with noise
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best {best_score:.6f}")

            jittered = _random_block_perturb(best_a, rng)
            x = np.log(jittered + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
    # Phase 2: subgradient descent on exact max
    a = best_a.copy()
    step0 = 0.3
    iteration = 0

    while time.monotonic() < deadline:

        elapsed = now - start

        max_val = c.max()
        max_idx = np.flatnonzero(c == max_val)

            high = min(n - 1, i)
            j = np.arange(low, high + 1, dtype=int)

        # step size
        iteration += 1
        step = step0 / np.sqrt(iteration)

        # project onto simplex
        a_new = simplex_projection(a_new)
        # evaluate

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            last_improve2 = now
            print(f"[{elapsed:6.2f}s] subgrad new best = {best_score:.6f}")

        a = a_new
        # occasional random perturbation if stagnation

            a = rng.dirichlet(np.ones(n))

            print(f"[{elapsed:6.2f}s] subgrad restart")
    # Return best
    best = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best]

"""
Propose a high‑quality step‑function sequence.

Algorithm
---------
The search works on the probability simplex (∑a = 1).  It consists of
two phases inside the allocated time budget (≈ 1000 s).

* Phase 1 – smooth‐max Adam optimisation.
  We minimise a log‑sum‑exp surrogate of the max self‑convolution.
  The smoothing parameter γ is raised slowly from 0.2 to 200.
  Adam with a linearly decaying learning rate updates the log‑weights
  (log‑softmax of the coefficients).  Stagnation triggers a soft restart
  with Gaussian noise; occasional block‑jitter moves mass in a random
  contiguous block.

* Phase 2 – direct sub‑gradient descent on the true objective.
  After Phase 1 the surrogate is sharp and we switch to minimising
  the exact quantity 2·n·max(a*a) (the denominator is fixed to 1).
  The sub‑gradient of the max‑convolution is obtained analytically,
  and each step is projected onto the simplex (Euclidean projection).
  A simple 1/√t step‑size schedule together with restarts when no
  improvement is seen for 10 s completes the optimisation.

The best sequence encountered (clipped to the limits enforced by
`evaluate_sequence`) is returned.  The routine respects the 1000 s
budget with a 5 s safety margin and prints occasional progress
messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (must be defined at module level)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate w.r.t. ``a``.
    ``w`` is a probability vector on the convolution domain (length 2·n‑1).
    """
    n = a.size
    # correlation a_rev (*) w   (linear)
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    # Extract the range that corresponds to Σ_j a_j w_{i+j}
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to the unit simplex.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of ``v`` onto the probability simplex
    {x ≥ 0, Σ x = 1}.  Implements the algorithm of
    Duchi et al., 2008.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only defined for 1‑D vectors")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # Numerical safety – fall back to uniform simplex point
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def compute_score(a: np.ndarray, n: int) -> float:
    """
    Exact objective for a probability vector ``a`` of length ``n``:
        score = 2·n·max(a*a)
    (the denominator (∑a)² = 1 because we keep the vector on the simplex).
    """
    fft_len = _next_pow2(2 * n - 1)
    conv = _self_conv_fft(a, fft_len)
    return float(2.0 * n * np.max(conv))


def _max_conv_and_active(a: np.ndarray, n: int):
    """
    Return the maximal convolution value and the list of indices where it occurs.
    """
    fft_len = _next_pow2(2 * n - 1)
    conv = _self_conv_fft(a, fft_len)
    max_val = np.max(conv)
    active_idx = np.flatnonzero(conv == max_val)
    return max_val, active_idx, conv


def _subgradient_maxconv(a: np.ndarray, active_idx: np.ndarray, n: int):
    """
    Sub‑gradient of max_i (a*a)_i evaluated at a point ``a``.
    The gradient for a single active index i is
        2 * shift_i(a)  (linear shift of a)
    and we average over all active indices.
    """
    grad = np.zeros_like(a)
    for i in active_idx:
        low = max(0, i - (n - 1))
        high = min(n - 1, i)
        j = np.arange(low, high + 1, dtype=int)
        grad[j] += 2.0 * a[i - j]
    grad /= max(1, len(active_idx))
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises
    ``evaluate_sequence``.  The routine runs for up to ``budget_s`` seconds
    (with a 5 s safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Determine problem size and initialise a feasible point on the simplex
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64).ravel()
        n = a0.shape[0]
        a0 = np.clip(a0, 0.0, None)
        s = a0.sum()
        if s > 0.0:
            a0 /= s
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        # fallback: uniform vector of moderate size
        n = 2048
        a0 = np.full(n, 1.0 / n)

    # Log‑weights for Adam (optimise in the log‑softmax domain)
    x = np.log(a0 + 1e-12)

    # Pre‑compute FFT length for convolutions (used throughout)
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Adam optimiser state (Phase 1)
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # Book‑keeping
    # ------------------------------------------------------------------
    best_a = a0.copy()
    best_score = compute_score(best_a, n)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)   # safety margin
    total_time = deadline - start
    last_improve = start

    # Split budget: 70 % for Phase 1, remainder for Phase 2
    phase1_end = start + 0.70 * total_time

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the log‑sum‑exp surrogate
    # ------------------------------------------------------------------
    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total_time

        # Current iterate on the simplex
        a = _softmax(x)

        # Self‑convolution via FFT (cheap)
        c = _self_conv_fft(a, fft_len_self)

        # Smooth‑max weights (log‑sum‑exp with temperature γ)
        gamma = 0.2 + (200.0 - 0.2) * frac
        c_shift = c - np.max(c)               # for numerical stability
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        w /= w_sum

        # Gradient of surrogate w.r.t. a
        grad_a = _gradient_wrt_a(a, w)

        # Chain rule to log‑weights x (keeps a on simplex)
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update (linear learning‑rate decay)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        lr = 0.5 * (1.0 - frac) + 0.01 * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # Exact evaluation (true objective) for bookkeeping
        cur_score = compute_score(a, n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase 1 new best = {best_score:.6f}")

        # Stagnation handling (soft restart)
        if now - last_improve > 15.0:
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase 1 restart (stagnation) – best {best_score:.6f}")

        # Occasional block‑jitter to escape shallow plateaus
        if rng.random() < 0.05:
            jitter = _random_block_perturb(best_a, rng)
            x = np.log(jitter + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # Phase 2 – direct sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    step0 = 0.3               # initial step‑size (will decay as 1/√t)
    iteration = 0
    last_improve2 = time.monotonic()
    TARGET = 1.5025           # optional early‑stop target

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start

        # Compute max‑convolution and active set
        max_val, active_idx, _ = _max_conv_and_active(a, n)

        # Sub‑gradient of the true objective (max‑conv)
        grad = _subgradient_maxconv(a, active_idx, n)

        # Step‑size schedule (1/√t)
        iteration += 1
        step = step0 / np.sqrt(iteration)

        # Gradient step + projection onto simplex
        a_next = a - step * grad
        a_next = _simplex_projection(a_next)

        # Evaluate
        cur_score = compute_score(a_next, n)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_next.copy()
            last_improve2 = now
            print(f"[{elapsed:6.2f}s] Phase 2 new best = {best_score:.6f}")

        # Early exit if target already reached
        if best_score <= TARGET:
            print(f"[{elapsed:6.2f}s] Target score reached; terminating early.")
            break

        # Stagnation → random Dirichlet restart
        if now - last_improve2 > 10.0:
            a_next = rng.dirichlet(np.ones(n))
            iteration = 0
            last_improve2 = now
            print(f"[{elapsed:6.2f}s] Phase 2 random restart")

        a = a_next

    # ------------------------------------------------------------------
    # Return the best sequence (respect the bounds of evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
