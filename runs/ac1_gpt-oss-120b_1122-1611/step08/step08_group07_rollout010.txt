propose_candidate()

A two‑phase optimizer for the convolution‑max bound.

Phase I – smooth‑max Adam on the simplex
    * Parameters are stored as unconstrained log‑weights `x`.
    * The objective `max_k (a*a)_k` is approximated by a log‑sum‑exp
      surrogate with a γ that is increased from 0.5 up to 800.
    * Adam with a linearly decaying learning‑rate (0.8 → 0.01) drives the

      Gaussian noise (soft‑restart).

Phase II – aggressive greedy refinement
    * Repeatedly compute the current self‑convolution, locate the peak
      index `k*` and the per‑coordinate contribution `a_i·a_{k*‑i}`.
    * Transfer a modest fraction of the mass of the *worst* contributor
      to a location that minimises the new peak (edges or a random low‑

      a full random shuffle or a deterministic sort (descending/ascending)
      – any improvement is kept.
    * Restarts (re‑jittering the best solution) happen after 30 s of
      stagnation.

Both phases keep track of the globally best sequence and print progress.
The routine respects the 5‑second safety margin required by the harness
and finally returns the best sequence (clipped to the limits imposed by

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    # Real‑to‑real FFT is sufficient because the input is real.
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed‑max surrogate w.r.t. ``a``.
    ``w`` are the soft‑max weights (probability vector of length 2·n‑1).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit L1‑norm.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 15))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.12))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _score_of(a: np.ndarray) -> float:
    """
    Exact evaluation of the objective used by ``evaluate_sequence``.
    ``a`` is assumed to be a non‑negative 1‑D array.
    """
    n = a.size
    if n == 0:
        return np.inf
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _self_conv(a, _next_pow2(2 * n - 1))
    max_b = conv.max()
    return 2.0 * n * max_b / (s * s)


def _mass_transfer_move(a: np.ndarray, rng: np.random.Generator,
                       fft_len: int) -> tuple[np.ndarray, float]:
    """
    Propose a move that transfers a fraction of the mass from the index
    that contributes most to the current peak to a location that is likely
    to lower the peak.
    Returns the new sequence and its score.
    """
    n = a.size
    conv = _self_conv(a, fft_len)
    peak_idx = int(np.argmax(conv))

    # contribution of each element to the peak (product a[i] * a[peak-i])
    idx = np.arange(n)
    partner = peak_idx - idx
    valid = (partner >= 0) & (partner < n)
    prod = np.zeros_like(a)
    prod[valid] = a[idx[valid]] * a[partner[valid]]

    i_max = int(np.argmax(prod))

    # pick a few candidate destinations (edges and random low‑contrib slots)
    cand = [0, n - 1]
    # add three random low‑contrib slots
    low_contrib = np.argsort(prod)[:5]
    for _ in range(3):
        cand.append(int(rng.choice(low_contrib)))
    # add a random index (fallback)
    cand.append(int(rng.integers(0, n)))

    best_seq = a
    best_score = _score_of(a)

    # transfer a modest amount of mass
    delta_frac = rng.uniform(0.03, 0.12)  # 3‑12 % of the bucket
    delta = a[i_max] * delta_frac
    if delta <= 0.0:
        return a, best_score

    for j in cand:
        if j == i_max:
            continue
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[j] += delta
        # keep non‑negativity
        if a_new[i_max] < 0.0:
            continue
        score = _score_of(a_new)
        if score < best_score:
            best_score = score
            best_seq = a_new

    return best_seq, best_score


def _sorted_move(a: np.ndarray, descending: bool = True) -> np.ndarray:
    """
    Return a copy of ``a`` sorted either descending or ascending.
    """
    if descending:
        return np.sort(a)[::-1]
    else:
        return np.sort(a)

        frac = elapsed / total_budget

        # --- current iterate ------------------------------------------------
        a = _softmax(x)                     # probability vector, sum = 1

        # --- self‑convolution ------------------------------------------------
        c = _self_conv(a, fft_len_self)     # length 2·n‑1

        # --- smooth‑max weight computation -----------------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        c_shift = c - np.max(c)             # for numerical stability
        w = np.exp(gamma * c_shift)
        w /= w.sum()

        # --- gradient wrt a ---------------------------------------------------
        g_a = _gradient_wrt_a(a, w)         # length n

        # --- chain rule to log‑weights x -------------------------------------
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # --- Adam step --------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # --- exact evaluation (true objective) --------------------------------
        cur_score = _score_of(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:7.2f}s] Phase I – new best score = {best_score:.6f}")

        # --- restart on stagnation --------------------------------------------
        if now - last_improve > 15.0:
            # jitter around the current best
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:7.2f}s] Phase I – restart due to stagnation (best {best_score:.6f})")

        # occasional global jitter (helps escape flat plateaus)
        if rng.random() < 0.015:
            pert = _random_block_perturb(best_a, rng)
            x = np.log(pert + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # --------------------------------------------------------------
    # 7) Phase II – focused greedy refinement (remaining budget)
    # --------------------------------------------------------------
    a = best_a.copy()
    cur_score = _score_of(a)
    best_score = cur_score
    best_a = a.copy()
    last_improve = time.monotonic()
    iteration = 0

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        remaining = deadline - now
        iteration += 1

        # 7.1 – propose a mass‑transfer move (most effective)
        a_new, new_score = _mass_transfer_move(a, rng, fft_len_self)

        # 7.2 – occasional block‑perturbation (exploration)
        if rng.random() < 0.02:
            a_block = _random_block_perturb(a, rng)
            block_score = _score_of(a_block)
            if block_score < new_score:
                a_new, new_score = a_block, block_score

        # 7.3 – try a deterministic sort (descending or ascending)
        if rng.random() < 0.01:
            a_sorted = _sorted_move(a, descending=True)
            sort_score = _score_of(a_sorted)
            if sort_score < new_score:
                a_new, new_score = a_sorted, sort_score

        # 7.4 – acceptance test (strict improvement or tempered uphill)
        accept = False
        if new_score < cur_score - 1e-12:
            accept = True
        else:
            # simulated‑annealing temperature, linearly cooling to 0
            T = max(0.005, remaining / total_budget * 0.04)
            if rng.random() < np.exp(-(new_score - cur_score) / T):
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:7.2f}s] Phase II – new best score = {best_score:.6f}")

        # restart jitter if stuck for a while
        if now - last_improve > 30.0:
            a = best_a.copy()
            # apply a few random perturbations to shake out of plateaus
            for _ in range(2):
                a = _random_block_perturb(a, rng)
            cur_score = _score_of(a)
            last_improve = now
            print(f"[{elapsed:7.2f}s] Phase II – jitter after stagnation (best {best_score:.6f})")

        # early exit if target already achieved
        if best_score < 1.5025:
            print(f"[{elapsed:7.2f}s] Target reached – terminating early.")
            break

    # --------------------------------------------------------------
    # 8) Return the best sequence found (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
