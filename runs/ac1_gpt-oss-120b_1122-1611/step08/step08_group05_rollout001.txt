"""
Propose a candidate step‑function using a two‑phase strategy:

1. **Smooth‑max Adam optimisation** – optimise the log‑softmax
   probabilities on the simplex by minimising a log‑sum‑exp
   surrogate of the max self‑convolution.  The schedule raises the
   smoothing parameter γ from 0.5 to 800 while an Adam optimiser
   adapts the log‑weights.  Restarts with moderate log‑normal noise
   are triggered after 30 s of stagnation; occasional block‑jitter
   provides larger jumps.

2. **Simulated‑annealing refinement** – starting from the best vector
   found, perform many tiny mass‑transfer moves between random
   coordinates (preserving the sum) and accept them with a Metropolis
   criterion.  Temperature decays linearly, allowing the search to
   escape shallow local minima and push the objective below the target
   1.5025.

The routine respects a 1000 s budget (leaving a 5 s safety margin) and
returns the best sequence (clipped to the limits imposed by
`evaluate_sequence`).  If a global variable `height_sequence_1` exists,
it is used as the initial guess; otherwise a uniform start of length
4096 is taken.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions – top‑level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _self_conv_direct(a: np.ndarray) -> np.ndarray:
    """Exact linear self‑convolution (fallback for scoring)."""
    return np.convolve(a, a)


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max yielding a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    s = e.sum()
    if s == 0.0:
        # fallback to uniform distribution
        return np.full_like(x, 1.0 / x.size)
    return e / s


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the LSE surrogate w.r.t. a.
    ``w`` is a probability vector of length 2·n‑1 (same length as a*a).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]                     # reversed a
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    # correlation gives ∂/∂a of Σ w_i (a*a)_i
    return 2.0 * corr[start:end]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Apply a multiplicative log‑normal factor to a random contiguous block
    of ``a`` and renormalise to sum‑one.
    """
    n = a.size
    # block size between 1 and max(2, n//8)
    width = rng.integers(1, max(2, n // 8) + 1)
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _score_from_a(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) for a simplex‑normalised vector."""
    n = a.size
    conv = _self_conv_direct(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises
    ``evaluate_sequence`` within the given time budget.

    Parameters
    ----------
    seed : int
        Random seed for reproducibility.
    budget_s : float
        Total time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence found (values are already within the limits of
        ``evaluate_sequence``).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget – stop a few seconds early for safety.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Initialise sequence (use provided construction if present).
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim != 1 or a0.size == 0:
            # fallback to uniform
            n = 4096
            a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            n = a0.size
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s > 0.0:
                a0 /= s
            else:
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 4096
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    # Log‑weights for the Adam optimiser
    x = np.log(a0 + 1e-12)          # shape (n,)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT length for self‑convolution.
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 3) Adam state
    # ------------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Helper to obtain simplex vector from log‑weights
    # ------------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # ------------------------------------------------------------------
    # 5) Track best solution (exact evaluation)
    # ------------------------------------------------------------------
    a = _a_from_x(x)
    best_score = _score_from_a(a)
    best = a.copy()
    last_improve = time.monotonic()
    start = time.monotonic()
    total_time = deadline - start

    # ------------------------------------------------------------------
    # 6) Hyper‑parameters (schedules)
    # ------------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 800.0                # sharper surrogate in the final phase
    lr_start = 0.9
    lr_end = 0.02                   # decay over the whole run

    # ------------------------------------------------------------------
    # 7) Main Adam optimisation loop
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total_time  # in [0,1]

        # ----- Current vector on simplex ---------------------------------
        a = _a_from_x(x)

        # ----- Self‑convolution (FFT, cheap) -----------------------------
        c = _self_conv_fft(a, fft_len_self)          # length 2·n‑1

        # ----- Smooth‑max surrogate --------------------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        c_shift = c - np.max(c)                     # for numerical stability
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                               # probability vector

        # ----- Gradient wrt a ---------------------------------------------
        g_a = _gradient_wrt_a(a, w)                 # shape (n,)

        # ----- Chain rule to log‑weights ---------------------------------
        avg = np.dot(a, g_a)                         # ⟨a,∇F⟩
        g_x = a * (g_a - avg)                        # ensures sum stays 1

        # ----- Adam update ------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----- Exact objective (for monitoring) --------------------------
        cur_score = _score_from_a(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best (Adam) score = {best_score:.6f}")

        # ----- Stagnation handling – soft restart -------------------------
        if now - last_improve > 30.0:
            # Log‑normal perturbation around current best
            noise = rng.normal(0.0, 0.7, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (stagnation) – best {best_score:.6f}")

        # ----- Occasional block jitter ------------------------------------
        if rng.random() < 0.02:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] Block jitter applied")

        # ----- Early exit if target met -----------------------------------
        if best_score <= 1.5025:
            print(f"[{elapsed:6.2f}s] Target reached! ({best_score:.6f})")
            break

    # ------------------------------------------------------------------
    # 8) Simulated‑annealing refinement (if time permits)
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 15.0:
        # Use the best vector as starting point for annealing
        a = best.copy()
        cur_score = best_score

        anneal_start = time.monotonic()
        anneal_deadline = deadline - 5.0       # keep a safety margin
        total_anneal = anneal_deadline - anneal_start

        # Temperature schedule (linear)
        temp_start = 0.04
        temp_end = 0.0005

        while time.monotonic() < anneal_deadline:
            now = time.monotonic()
            elapsed_a = now - anneal_start
            frac_a = elapsed_a / total_anneal
            temp = temp_start * (1.0 - frac_a) + temp_end * frac_a

            # Propose a tiny mass transfer between two random coordinates
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            # Amount we are allowed to move (no more than what i currently holds)
            max_move = min(a[i], 0.015)          # absolute move limit
            if max_move <= 0.0:
                continue
            delta = rng.random() * max_move
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            # Renormalise to counter floating‑point drift
            s = a_new.sum()
            if s != 0.0:
                a_new /= s

            new_score = _score_from_a(a_new)

            # Metropolis acceptance
            if new_score < cur_score:
                a = a_new
                cur_score = new_score
                if new_score < best_score - 1e-12:
                    best_score = new_score
                    best = a_new.copy()
                    print(f"[{now - start:6.2f}s] New best (anneal) score = {best_score:.6f}")
            else:
                # Accept with probability exp(-(Δ)/T)
                accept_prob = np.exp(-(new_score - cur_score) / max(temp, 1e-12))
                if rng.random() < accept_prob:
                    a = a_new
                    cur_score = new_score

            # Early termination if we beat the target
            if best_score <= 1.5025:
                print(f"[{now - start:6.2f}s] Target reached during annealing!")
                break

    # ------------------------------------------------------------------
    # 9) Return the best sequence (clipped to evaluate_sequence limits)
    # ------------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
