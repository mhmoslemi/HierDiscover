    # choose length n

    # progress
    start = time.monotonic()
    deadline = start + budget_s - 5

    # parameters
    iter = 0
    max_iters = 1000000
    T0 = 0.1
    while time.monotonic() < deadline and iter < max_iters:
        # compute conv
        c = _self_conv(a, _next_pow2(2*n-1))
        max_idx = int(np.argmax(c))
        max_val = c[max_idx]
        cur_score = 2 * n * max_val
        # compute contributions
        idx = np.arange(n)
        js = max_idx - idx
        mask = (js >= 0) & (js < n)
        contrib = np.zeros(n)
        contrib[mask] = a[idx[mask]] * a[js[mask]]
        i_max = int(np.argmax(contrib))
        j_partner = max_idx - i_max

        # decide mass transfer
        # fraction based on iteration/power
        frac = 0.1 if iter < 20000 else 0.05 if iter < 50000 else 0.02
        delta_i = a[i_max] * rng.uniform(frac/2, frac)
        # ensure not too large
        if delta_i > a[i_max]:
            delta_i = a[i_max] * frac
        # destination
        dest_i = rng.integers(0, n)
        while dest_i == i_max:
            dest_i = rng.integers(0, n)

        a_new = a.copy()
        a_new[i_max] -= delta_i
        a_new[dest_i] += delta_i

        # If partner not same as i_max, also move from partner
        if j_partner != i_max and j_partner >= 0 and j_partner < n:
            # Choose delta_j similar
            delta_j = a[j_partner] * rng.uniform(frac/2, frac)
            dest_j = rng.integers(0, n)
            while dest_j == j_partner or dest_j == dest_i:
                dest_j = rng.integers(0, n)
            a_new[j_partner] -= delta_j
            a_new[dest_j] += delta_j

        # clip small negative due to rounding: enforce >=0
        a_new = np.clip(a_new, 0.0, None)
        # renormalize to keep sum=1 (should be same but due residual rounding)
        s = a_new.sum()
        if s == 0:
            a_new = np.full(n, 1.0/n)
        else:
            a_new /= s

        # evaluate
        c_new = _self_conv(a_new, _next_pow2(2*n-1))
        max_val_new = c_new[max_idx]  # Actually max may be at different idx; need recompute max.
        score_new = 2 * n * c_new.max()

        # acceptance
        if score_new < cur_score - 1e-12:
            a = a_new
            cur_score = score_new
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_a = a_new.copy()
                print(f"{time.monotonic()-start:.2f}s new best {best_score}")

                cur_score = score_new

"""
Simple simulated‑annealing optimiser for the step‑function bound.

The search works directly on the simplex (the coefficients are kept
non‑negative and renormalised after every move).  At each iteration we
identify the shift `k*` where the self‑convolution `a*a` is maximal.
The pair of indices `(i , k*‑i)` that contributes most to this peak is
found and a small fraction of its mass is transferred to random
positions – this directly attacks the dominant term of the objective.
The move is accepted if it improves the true bound
`2·n·max(a*a)`; otherwise it may be accepted with a temperature‑dependent
probability (simulated annealing).  The temperature decays linearly,
and a global “jitter” restart is triggered after 30 s without progress.

The routine uses an FFT‑based convolution for speed and respects the
`evaluate_sequence` limits (non‑negative, ≤ 1000).  It runs for at most
`budget_s` seconds (a 5 s safety margin is reserved for the harness) and
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence``.  The algorithm is a simulated‑annealing
    optimiser that repeatedly attacks the current convolution peak.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Initialise the sequence (use the previously found construction
    #    if it exists, otherwise start from the uniform distribution)
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a = np.asarray(init_seq, dtype=np.float64)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            n = a.size if a.ndim == 1 and a.size > 0 else 1024
            a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        n = 2048                     # power of two → fast FFTs
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size                     # final length

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT length for the self‑convolution
    # --------------------------------------------------------------
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 3) Helper to evaluate the true bound quickly
    # --------------------------------------------------------------
    def _score(vec: np.ndarray) -> float:
        """Return 2·n·max( a*a ) for a probability vector ``vec``."""
        conv = _self_conv(vec, fft_len)
        return 2.0 * n * np.max(conv)

    cur_score = _score(a)
    best_a = a.copy()
    best_score = cur_score
    start_time = time.monotonic()
    last_improve = start_time
    print(f"[0.00s] initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4) Simulated‑annealing parameters
    # --------------------------------------------------------------
    T_start = 0.20                # initial temperature
    T_end = 1e-5                  # final temperature
    max_iters = 2_000_000        # hard iteration cap (won’t be reached)
    it = 0

    # --------------------------------------------------------------
    # 5) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline and it < max_iters:
        it += 1

        # linear temperature schedule
        T = T_start * (1.0 - it / max_iters) + T_end * (it / max_iters)

        # ------------------------------------------------------------------
        # 5a) Choose a move: global jitter (10 % of iterations) or a
        #      targeted mass‑transfer move based on the current worst shift.
        # ------------------------------------------------------------------
        if rng.random() < 0.10:
            # ----- global log‑normal perturbation -----
            factor = np.exp(rng.normal(0.0, 0.12, size=n))
            a_new = a * factor
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            a_new /= s if s > 0.0 else 1.0
        else:
            # ----- targeted move -------------------------------------------------
            conv = _self_conv(a, fft_len)
            max_idx = int(np.argmax(conv))

            # contribution of each coordinate to the offending peak
            idxs = np.arange(n)
            js = max_idx - idxs
            valid = (js >= 0) & (js < n)
            contrib = np.zeros(n, dtype=np.float64)
            contrib[valid] = a[idxs[valid]] * a[js[valid]]

            i_max = int(np.argmax(contrib))
            j_partner = max_idx - i_max

            # fraction of mass to move – larger early, smaller later
            frac = 0.15 if it < max_iters // 2 else 0.05
            delta_i = a[i_max] * rng.uniform(frac * 0.5, frac)
            dest_i = rng.integers(0, n)
            while dest_i == i_max:
                dest_i = rng.integers(0, n)

            a_new = a.copy()
            a_new[i_max] -= delta_i
            a_new[dest_i] += delta_i

            # also touch the partner index if it is different
            if j_partner != i_max and 0 <= j_partner < n:
                delta_j = a[j_partner] * rng.uniform(frac * 0.5, frac)
                dest_j = rng.integers(0, n)
                while dest_j == j_partner or dest_j == dest_i:
                    dest_j = rng.integers(0, n)
                a_new[j_partner] -= delta_j
                a_new[dest_j] += delta_j

            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            a_new /= s if s > 0.0 else 1.0

        # ------------------------------------------------------------------
        # 5b) Evaluate the new candidate
        # ------------------------------------------------------------------
        new_score = _score(a_new)

        # ------------------------------------------------------------------
        # 5c) Acceptance test (always accept improvements)
        # ------------------------------------------------------------------
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a_new.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] NEW BEST = {best_score:.6f}")
        else:
            # uphill move with probability exp(-Δ/T)
            if T > 0.0 and rng.random() < np.exp(-(new_score - cur_score) / T):
                a = a_new
                cur_score = new_score

        # ------------------------------------------------------------------
        # 5d) Stagnation handling – small jitter after 30 s without progress
        # ------------------------------------------------------------------
        if time.monotonic() - last_improve > 30.0:
            a = best_a.copy()
            a = _random_block_perturb(a, rng)
            cur_score = _score(a)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Restart after stagnation (score={cur_score:.6f})")

    # --------------------------------------------------------------
    # 6) Return the best sequence seen (clipped to the limits of the
    #    ``evaluate_sequence`` function)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
