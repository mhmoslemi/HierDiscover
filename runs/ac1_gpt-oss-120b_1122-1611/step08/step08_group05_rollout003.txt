"""
propose_candidate – multi‑scale Adam/LSE optimiser

The evaluator is homogeneous:   score = 2·n·max(conv) / (∑a)².
Hence we restrict the search to the simplex (∑a = 1) and minimise

    F(a) = 2·n·max_i (a*a)_i .

Direct optimisation of the non‑smooth max is difficult, therefore we use a
log‑sum‑exp (LSE) smooth approximation controlled by a temperature γ.
For a fixed length n we run a projected Adam optimiser on the log‑soft‑max
parameters x (so that a = softmax(x)).  γ is raised linearly from a small
value (≈0.2) to a large one (≈800) while the learning‑rate decays.
Stagnation triggers a full random restart; occasional “block‑jitter”
perturbations help to escape shallow basins.

Instead of a single run we perform a *multi‑scale* schedule:
starting from n=256 we double the size repeatedly (256, 512, …, up to the
largest n that fits into the remaining time budget).  Each stage receives a
fraction of the total budget; the best vector over all stages is kept.
All calculations are performed with FFT‑based linear self‑convolution
( O(n log n) ) and the analytic gradient of the LSE surrogate.
The final sequence is clipped to the limits enforced by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – must be defined at module level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _self_conv_direct(a: np.ndarray) -> np.ndarray:
    """Direct convolution – only for tiny vectors (fallback)."""
    return np.convolve(a, a)


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the LSE surrogate w.r.t. a.
    ``w`` is a probability vector of length 2·n‑1 (same length as a*a).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    corr = np.fft.irfft(fa * fw, n=N)              # length n+len(w)-1
    start = n - 1
    end = start + n
    return 2.0 * corr[start:end]                  # ∂/∂a of Σ w_i (a*a)_i


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 8))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

# Core optimiser for a fixed length
# ----------------------------------------------------------------------


def _optimise_fixed_n(n: int,
                      time_budget: float,
                      rng: np.random.Generator,
                      init_a: np.ndarray | None = None) -> tuple[np.ndarray, float]:
    """
    Run the Adam/LSE optimiser for a given length ``n``.
    Returns the best vector (sum=1) and its exact score.
    The optimiser works on the log‑softmax parameters ``x``.
    """
    # ----- initialise ----------------------------------------------------
    if init_a is not None:
        a0 = np.asarray(init_a, dtype=np.float64)
        if a0.ndim != 1 or a0.size != n:
            a0 = np.full(n, 1.0 / n)
        else:
            a0 = np.clip(a0, 0.0, None)
            s = a0.sum()
            if s == 0.0:
                a0 = np.full(n, 1.0 / n)
            else:
                a0 /= s
    else:
        a0 = np.full(n, 1.0 / n)

    x = np.log(a0 + 1e-12)                     # log‑weights (unconstrained)

    m = np.zeros_like(x)                      # Adam first moment
    v = np.zeros_like(x)                      # Adam second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # pre‑compute FFT length for self‑convolution
    fft_len_self = _next_pow2(2 * n - 1)

    # track best solution
    a = _softmax(x)
    conv = _self_conv_direct(a)               # exact (tiny overhead)
    best_score = 2.0 * n * conv.max()
    best_a = a.copy()
    last_improve = time.monotonic()
    start = time.monotonic()
    deadline = start + time_budget

    # hyper‑parameters
    gamma_start = 0.2
    gamma_end = 800.0                         # very sharp surrogate
    lr_start = 0.6
    lr_end = 0.02
    stagnation_seconds = 15.0
    jitter_prob = 0.02

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / time_budget

        # -------- current iterate ----------------------------------------
        a = _softmax(x)                       # stays on simplex

        # -------- self‑convolution (FFT, cheap) -------------------------
        c = _self_conv_fft(a, fft_len_self)   # length 2·n‑1

        # -------- surrogate weights (smooth max) -------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac
        c_shift = c - np.max(c)               # stabilise exponentials
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        if w_sum == 0.0:                      # numerical safeguard
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum                         # probability vector

        # -------- gradient w.r.t. a --------------------------------------
        g_a = _gradient_wrt_a(a, w)           # length n

        # -------- chain‑rule to log‑weights x (projected onto simplex) --
        avg = np.dot(a, g_a)                  # ⟨a,∇F⟩
        g_x = a * (g_a - avg)                 # keeps sum(a)=1

        # -------- Adam update --------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # -------- exact evaluation (true objective) ----------------------
        cur_max = np.max(c)                   # same as exact conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            # minimal printing – only when progress is made
            print(f"[{elapsed:6.2f}s] n={n} new best score = {best_score:.6f}")

        # -------- stagnation handling (full random restart) -------------
        if now - last_improve > stagnation_seconds:
            # random Gaussian noise on the log‑weights
            x = rng.normal(0.0, 0.6, size=n)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] n={n} restart (stagnation) – best {best_score:.6f}")

        # -------- occasional block‑jitter (global jump) ------------------
        if rng.random() < jitter_prob:
            a_jit = _random_block_perturb(best_a, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # final exact evaluation (safety)
    a = best_a
    conv = _self_conv_direct(a)
    best_score = 2.0 * n * conv.max()
    return best_a, best_score

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The method runs a multi‑scale Adam optimiser on several vector lengths,
    keeping the best result seen before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Global time budget – keep a 5 s safety margin.
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Starting point – use already‑found construction if present.
    # ------------------------------------------------------------------
    init_global = None
    if "height_sequence_1" in globals():
        tmp = globals()["height_sequence_1"]
        if isinstance(tmp, (list, tuple, np.ndarray)):
            arr = np.asarray(tmp, dtype=np.float64)
            if arr.ndim == 1 and arr.size > 0:
                init_global = np.clip(arr, 0.0, None)
                s = init_global.sum()
                if s > 0.0:
                    init_global /= s

    # ------------------------------------------------------------------
    # 2) Multi‑scale schedule.  We start from a modest size and keep
    #    doubling while time permits.
    # ------------------------------------------------------------------
    # Minimal size – too small makes the problem trivial and gives poor
    # scores.  256 is a comfortable lower bound.
    n = 256
    best_overall_a = None
    best_overall_score = float("inf")
    total_time = deadline - time.monotonic()
    # Rough allocation: each stage receives a fraction of the remaining time
    # (larger stages get a larger share).
    while time.monotonic() < deadline and n <= 16384:
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            break                     # not enough time for another stage

        # Allocate at most 30 % of the remaining budget to the current stage,
        # but at least 12 seconds – this guarantees a few optimisation steps.
        stage_budget = min(remaining * 0.30, max(12.0, remaining - 30.0))
        print(f"=== Stage n={n} – allocated {stage_budget:.1f}s ===")

        # Initialise this stage – either from the global construction resized,
        # or from a uniform start.
        if init_global is not None and init_global.size >= n:
            # simply take the first n entries and renormalise
            init_a = init_global[:n].copy()
            if init_a.sum() == 0.0:
                init_a = None
        else:
            init_a = None

        # Run the optimiser for this length.
        try:
            cand_a, cand_score = _optimise_fixed_n(n,
                                                   stage_budget,
                                                   rng,
                                                   init_a=init_a)
        except Exception as e:   # defensive – should never happen
            print(f"Stage n={n} crashed: {e}")
            cand_a = None
            cand_score = float("inf")

        # Update global best if improved.
        if cand_score < best_overall_score:
            best_overall_score = cand_score
            best_overall_a = cand_a.copy()
            print(f"*** New global best – n={n} score={best_overall_score:.6f}")

        # Prepare for next size – double the length.  The current best can be
        # interpolated to provide a warm start for the next stage.
        if best_overall_a is not None:
            # linear interpolation to the new size
            old_n = best_overall_a.size
            x_old = np.arange(old_n) / (old_n - 1) if old_n > 1 else np.zeros(old_n)
            x_new = np.arange(2 * old_n) / (2 * old_n - 1)
            init_global = np.interp(x_new, x_old, best_overall_a)
            init_global = np.clip(init_global, 0.0, None)
            s = init_global.sum()
            if s > 0.0:
                init_global /= s
            else:
                init_global = None
        else:
            init_global = None

        n *= 2

    # ------------------------------------------------------------------
    # 3) Final safety check – if nothing was found, fall back to uniform.
    # ------------------------------------------------------------------
    if best_overall_a is None:
        n_fallback = 256
        best_overall_a = np.full(n_fallback, 1.0 / n_fallback)

    # ------------------------------------------------------------------
    # 4) Clip to the bounds enforced by ``evaluate_sequence`` and return.
    # ------------------------------------------------------------------
    best_overall_a = np.clip(best_overall_a, 0.0, 1000.0)
    return [float(v) for v in best_overall_a]
