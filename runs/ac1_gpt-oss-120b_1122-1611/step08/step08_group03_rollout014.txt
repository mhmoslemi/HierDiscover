"""
Enhanced stochastic‑annealing search for a step‑function minimizing
evaluate_sequence.

The algorithm differs from the previous implementation in three ways:

1. **Multi‑scale restarts** – we try several problem sizes (different
   lengths of the step function) and repeatedly re‑initialise the
   candidate from a fresh Dirichlet draw.  This avoids getting stuck in
   the basin of a single local minimum.

2. **Simulated‑annealing acceptance** – instead of only accepting
   improving moves we sometimes accept worsening moves with a
   temperature‑controlled probability.  The temperature is annealed
   slowly from ~1.0 to 1e‑5, allowing large exploratory jumps early on.

3. **FFT‑based exact scoring** – every candidate is evaluated with an
   exact convolution performed via the FFT, which is both fast (O(n log n))
   and numerically stable.  The best candidate seen across all restarts
   and all lengths is returned.

All updates keep the vector on the probability simplex (non‑negative,
unit L1‑norm) by a Euclidean projection.  The routine respects the
global 1000 s budget (5 s safety margin) and prints progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x (used for FFT padding)."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x ≥ 0, sum(x) = 1}
    Implements the classic O(n log n) algorithm based on sorting.
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D vectors.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid tiny numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _compute_score(a: np.ndarray, fft_len: int) -> float:
    """
    Exact evaluation of the objective for a given (already clipped) vector.
    Uses an FFT to obtain the full (linear) convolution a*a.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)
    conv = conv[: 2 * a.size - 1]               # keep only the linear part
    max_b = float(conv.max())
    sum_a = float(a.sum())
    if sum_a < 0.01:
        return float('inf')
    return 2.0 * a.size * max_b / (sum_a * sum_a)

# Main search routine
# ----------------------------------------------------------------------

    Search for a non‑negative height vector that (approximately) minimises
    evaluate_sequence.  The routine runs for at most ``budget_s`` seconds
    (minus a small safety margin) and returns the best vector it found.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Book‑keeping for the best solution seen so far
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float('inf')
    print(f"[0.00s] search started – budget = {budget_s:.0f}s")

    # ------------------------------------------------------------------
    # 2) Lengths to explore
    # ------------------------------------------------------------------
    # If a previous construction exists we also try its length.
    lengths = []

    # add a few powers of two – these give fast FFTs.
    lengths.extend([512, 1024, 2048, 4096, 6144, 8192])

    # ------------------------------------------------------------------
    # 3) Main multi‑restart loop
    # ------------------------------------------------------------------
    for n in lengths:

        # ------------------------------------------------------------------

            # keep the best globally

                elapsed = time.monotonic() - start_time

    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in best_vec]
