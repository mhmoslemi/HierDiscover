def targeted_optimize(a, deadline, rng):
    n = a.size
    best = a.copy()
    best_score = compute_score(best)
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        # Compute convolution
        conv = _max_self_convolution_and_full(a) # returns full conv array
        max_val = conv.max()
        # find indices of top contributions
        # Determine 'critical shifts' where conv >= 0.95 * max_val
        crit_mask = conv >= max_val * 0.95
        crit_shifts = np.nonzero(crit_mask)[0]  # indices
        # Compute risk per index
        risk = np.zeros(n)
        for k in crit_shifts:
            # contributions: i in valid range
            low = max(0, k - (n-1))
            high = min(n-1, k)
            # for i in low..high, contribution product = a[i] * a[k-i]
            # derivative w.r.t a[i] is a[k-i]
            # accumulate risk[i] += a[k-i]
            # can compute vectorized using slice operations
            i_range = np.arange(low, high+1)
            partner = k - i_range
            risk[i_range] += a[partner]
        # risk normalized maybe
        # Choose donor with probability proportional to risk * a (since larger a_i also matters)
        prob = risk * a
        if prob.sum() == 0:
            # fallback random
            donor = rng.integers(n)
        else:
            prob = prob / prob.sum()
            donor = rng.choice(n, p=prob)
        # Choose recipient maybe with low risk
        inv_prob = (risk.max() - risk) * (1 - a)  # heuristic
        if inv_prob.sum() == 0:
            recipient = rng.integers(n)
        else:
            inv_prob = inv_prob / inv_prob.sum()
            recipient = rng.choice(n, p=inv_prob)
        # Transfer small delta
        delta = rng.random() * 0.3 * best[donor]  # 0-30% of donor
        new = best.copy()
        new[donor] -= delta
        new[recipient] += delta
        # Ensure no negatives
        if new[donor] < 0:
            continue
        new_sum = new.sum()
        if new_sum == 0: # shouldn't happen
            continue
        new_score = compute_score(new)
        if new_score < best_score:
            best = new
            best_score = new_score
            last_improve = time.monotonic()
            # print improvement
            print(f"    [{time.monotonic():.2f}s] targeted improvement → {best_score:.6f}")
        # restart if stagnated
        if time.monotonic() - last_improve > 5.0:
            # random jitter
            noise = np.exp(rng.normal(0.0, 0.1, size=n))
            new = best * noise
            new = np.clip(new, 0.0, None)
            new /= new.sum()

            last_improve = time.monotonic()

def lp_refine(a, deadline, rng):
    n = a.size
    # Keep current best and maybe perform few iterations
    # compute conv_f_f and M
    conv = np.convolve(a, a)
    M = conv.max()
    # Build matrix A
    # use dense matrix for now
    m = 2*n - 1
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    # define constraints
    ub = np.full(m, M, dtype=np.float64)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    # solve LP
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        return a, compute_score(a)
    g = res.x
    sum_f = a.sum()
    sum_g = g.sum()
    if sum_g == 0:
        return a, compute_score(a)
    scale = sum_f / sum_g
    g_scaled = g * scale
    # mix with t=0.5 maybe adapt
    t = 0.5
    new = (1 - t) * a + t * g_scaled
    # normalize to sum_f (should already)
    # compute new score
    new_score = compute_score(new)
    return new, new_score

"""
Hybrid optimiser for the quadratic‐convolution objective.

The routine builds several candidate step‑functions (Beta‑shaped ramps,
random Beta laws and the optional global `height_sequence_1`).  For each
candidate it first runs a *targeted* hill‑climber that directly reduces
the contribution to the worst convolution shift (the shift where the
self‑convolution attains its maximum).  Occasionally the current
distribution is refined by a small linear‑programming step: given the
current bound M = max(conv(f,f)) we maximise ∑g under the linear
constraints conv(f,g) ≤ M, then mix the optimiser g back into f.  The
LP uses a dense Toeplitz matrix; this is only applied to moderate
lengths (≤ 1500) so that memory stays reasonable.

The search runs until the supplied time budget expires (default 1000 s)
and returns the best sequence it has encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy linear programming (used for the LP refinement step)
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Utility functions (no closures, no lambdas)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((int(x) - 1).bit_length())


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum of the full self‑convolution a*a computed via FFT."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(conv.max())


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Full convolution a*a (length 2·n‑1) via FFT."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def compute_score(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (without its clamping)."""
    n = a.size
    max_b = _max_self_convolution(a)
    s = a.sum()
    if s < 0.01:                      # protection present in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_b / (s * s)


def gen_beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discrete Beta(α,β) distribution on {0,…,n‑1}:
    a_i ∝ x_i^{α‑1}(1‑x_i)^{β‑1} with x_i = (i+½)/n.
    The result is normalised to sum = 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    raw = np.clip(raw, 0.0, None)
    s = raw.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / s


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


# ----------------------------------------------------------------------
# Targeted hill‑climber (focuses on the worst convolution shift)
# ----------------------------------------------------------------------


def _risk_vector(a: np.ndarray, conv: np.ndarray, max_idx: int) -> np.ndarray:
    """
    For a given shift `max_idx` (the location of the current maximum
    of the self‑convolution) compute a simple risk measure:
    risk[i] = a[k‑i] whenever a[i] participates in that shift,
    otherwise 0.
    """
    n = a.size
    low = max(0, max_idx - (n - 1))
    high = min(n - 1, max_idx)
    risk = np.zeros(n, dtype=np.float64)
    if low <= high:
        i_vals = np.arange(low, high + 1)
        partner = max_idx - i_vals
        risk[i_vals] = a[partner]
    return risk


def targeted_optimize(a: np.ndarray,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Stochastic hill‑climber that preferentially moves mass away from
    indices that contribute heavily to the current worst convolution
    shift.  Stops when ``deadline`` is reached.
    Returns the best sequence seen and its score.
    """
    n = a.size
    best = a.copy()
    best_score = compute_score(best)

    last_improve = time.monotonic()
    stagnation_limit = 8.0          # seconds without improvement before a jitter

    while time.monotonic() < deadline:
        conv = _full_convolution(best)
        max_idx = int(conv.argmax())
        max_val = conv[max_idx]

        # risk vector describes how much each coefficient participates in the worst shift
        risk = _risk_vector(best, conv, max_idx)

        # donor: large *risk* (i.e. large coefficient that also appears in the worst shift)
        donor_weights = risk * best
        if donor_weights.sum() > 0.0 and rng.random() < 0.9:
            donor = rng.choice(n, p=donor_weights / donor_weights.sum())
        else:
            donor = rng.choice(n, p=best / best.sum())

        # recipient: low risk (prefer indices that hardly appear in the worst shift)
        inv_risk = risk.max() - risk + 1e-12
        inv_risk[donor] = 0.0               # never pick the donor as recipient
        if inv_risk.sum() > 0.0:
            recipient = rng.choice(n, p=inv_risk / inv_risk.sum())
        else:
            recipient = rng.integers(n)

        # transfer a modest fraction of the donor's mass
        delta = rng.random() * 0.4 * best[donor]   # up to 40 % of donor mass
        if delta <= 0.0:
            continue
        new_seq = best.copy()
        new_seq[donor] -= delta
        new_seq[recipient] += delta
        # numerical safety: renormalise (exactly preserves sum)
        new_seq = np.clip(new_seq, 0.0, None)
        new_seq /= new_seq.sum()

        new_score = compute_score(new_seq)
        if new_score < best_score:
            best = new_seq
            best_score = new_score
            last_improve = time.monotonic()
            print(f"    [{time.monotonic():.2f}s] targeted improvement → {best_score:.6f}")

        # occasional jitter to escape flat plateaus
        if rng.random() < 0.01:
            jitter = _random_block_perturb(best, rng)
            jitter_score = compute_score(jitter)
            if jitter_score < best_score:
                best = jitter
                best_score = jitter_score
                last_improve = time.monotonic()
                print(f"    [{time.monotonic():.2f}s] jitter improvement → {best_score:.6f}")

        # restart (small random noise) if stagnating
        if time.monotonic() - last_improve > stagnation_limit:
            noise = np.exp(rng.normal(0.0, 0.08, size=n))
            restart = best * noise
            restart = np.clip(restart, 0.0, None)
            restart /= restart.sum()
            restart_score = compute_score(restart)
            if restart_score < best_score:
                best = restart
                best_score = restart_score
                print(f"    [{time.monotonic():.2f}s] restart improvement → {best_score:.6f}")
            last_improve = time.monotonic()

    return best, best_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (only for moderate lengths)
# ----------------------------------------------------------------------


def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Construct the dense matrix A (size (2n‑1) × n) such that
    (A @ g)[k] = sum_i f[i] * g[k‑i] = (f * g)[k].
    """
    n = f.size
    m = 2 * n - 1
    A = np.empty((m, n), dtype=np.float64)
    # column j contains f shifted by j positions
    for j in range(n):
        A[j:j + n, j] = f
    return A


def lp_refine(f: np.ndarray,
              deadline: float,
              rng: np.random.Generator,
              mix: float = 0.5) -> tuple[np.ndarray, float]:
    """
    One LP‑based refinement step:
    * maximise Σ g under the linear constraints conv(f,g) ≤ M,
      where M = max(conv(f,f)).
    * rescale g to have the same ℓ₁‑norm as f,
    * mix the two vectors (weight `mix`) to obtain a new candidate.
    Returns the mixed vector and its score.  If linear programming
    fails or is unavailable, the original vector is returned unchanged.
    """
    if not SCIPY_AVAILABLE:
        return f, compute_score(f)

    n = f.size
    # compute the current bound M
    conv_ff = _full_convolution(f)
    M = float(conv_ff.max())

    # build constraint matrix A (dense)
    A = _build_conv_matrix(f)
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # objective: maximise Σ g  <=> minimise -Σ g
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # solve the LP (HiGHS is fast for moderate sizes)
    res = linprog(c,
                  A_ub=A,
                  b_ub=ub,
                  bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': max(0.0, deadline - time.monotonic())})

    if not res.success:
        return f, compute_score(f)

    g = res.x
    sum_f = f.sum()
    sum_g = g.sum()
    if sum_g == 0.0:
        return f, compute_score(f)

    # scale g so that it has the same ℓ₁‑norm as f
    scale = sum_f / sum_g
    g_scaled = g * scale

    # convex mixture
    new = (1.0 - mix) * f + mix * g_scaled
    new /= new.sum()                     # protect against tiny numerical drift
    new_score = compute_score(new)
    return new, new_score


def iterative_lp_optimize(f: np.ndarray,
                          deadline: float,
                          rng: np.random.Generator,
                          max_iter: int = 5) -> tuple[np.ndarray, float]:
    """
    Apply several LP refinement steps sequentially (each with a modest
    mixing factor).  Stops earlier if the deadline is reached.
    """
    cur = f.copy()
    cur_score = compute_score(cur)
    for _ in range(max_iter):
        if time.monotonic() >= deadline:
            break
        # a relatively small mixing factor encourages smooth progress
        cur, new_score = lp_refine(cur, deadline, rng, mix=0.3)
        if new_score + 1e-9 < cur_score:
            cur_score = new_score
            print(f"    [{time.monotonic():.2f}s] LP refinement → {cur_score:.6f}")
        else:
            # No improvement – stop early
            break
    return cur, cur_score


# ----------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step function with a low value of
    ``evaluate_sequence``.  The routine respects the supplied time budget,
    performs a quick exploration of Beta‑shaped families, then refines the
    best candidate with the targeted hill‑climber and, for moderate
    lengths, with a short sequence of LP‑based updates.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) time‑budget handling (leave a tiny safety margin)
    # ------------------------------------------------------------------
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # 1) Gather initial candidates (global state, ramps, random betas)
    # ------------------------------------------------------------------
    init_candidates = []

    # a) existing global construction, if present
    if "height_sequence_1" in globals():
        seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        seq = np.clip(seq, 1e-12, None)
        seq /= seq.sum()
        init_candidates.append(seq)

    # b) a few deterministic Beta ramps
    for n in (256, 512, 1024, 2048, 3809):
        for α, β in ((2.0, 1.0), (1.0, 2.0)):
            seq = gen_beta_seq(n, α, β)
            init_candidates.append(seq)

    # c) random Beta families (limited number to keep time low)
    for _ in range(12):
        n = rng.integers(300, 2500)
        log_a = rng.uniform(np.log(0.2), np.log(5.0))
        log_b = rng.uniform(np.log(0.2), np.log(5.0))
        α = np.exp(log_a)
        β = np.exp(log_b)
        seq = gen_beta_seq(n, α, β)
        init_candidates.append(seq)

    # ------------------------------------------------------------------
    # 2) Evaluate all initial candidates, keep the best one
    # ------------------------------------------------------------------
    for cand in init_candidates:
        cand = np.clip(cand, 1e-12, None)
        cand /= cand.sum()
        sc = compute_score(cand)
        if sc < best_score:
            best_score = sc
            best_seq = cand.copy()
            print(f"[{time.monotonic() - start:.2f}s] initial candidate (n={cand.size}) → {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Main optimisation loop (targeted hill‑climbing + occasional LP)
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 2.0:               # not enough time for another pass
            break

        # --------------------------------------------------------------
        # 3a) Targeted hill‑climbing for a short burst (≈5 s)
        # --------------------------------------------------------------
        iter_deadline = min(deadline, now + min(5.0, remaining * 0.4))
        new_seq, new_score = targeted_optimize(best_seq, iter_deadline, rng)
        if new_score < best_score - 1e-9:
            best_seq, best_score = new_seq, new_score
            print(f"[{time.monotonic() - start:.2f}s] new global best (targeted) → {best_score:.6f}")
            continue        # give the improved point a chance to be polished

        # --------------------------------------------------------------
        # 3b) LP refinement (only if length is modest and we have time)
        # --------------------------------------------------------------
        if SCIPY_AVAILABLE and best_seq.size <= 1500 and remaining > 10.0:
            lp_deadline = min(deadline, now + min(8.0, remaining * 0.3))
            refined_seq, refined_score = iterative_lp_optimize(best_seq, lp_deadline, rng, max_iter=3)
            if refined_score < best_score - 1e-9:
                best_seq, best_score = refined_seq, refined_score
                print(f"[{time.monotonic() - start:.2f}s] new global best (LP) → {best_score:.6f}")
                continue

        # --------------------------------------------------------------
        # 3c) If still no progress, perform a cheap random restart
        # --------------------------------------------------------------
        if time.monotonic() - last_improve > 30.0 or rng.random() < 0.15:
            # generate a fresh random Beta candidate
            n = rng.integers(400, 2200)
            α = np.exp(rng.uniform(np.log(0.2), np.log(5.0)))
            β = np.exp(rng.uniform(np.log(0.2), np.log(5.0)))
            cand = gen_beta_seq(n, α, β)
            cand = np.clip(cand, 1e-12, None)
            cand /= cand.sum()
            cand_score = compute_score(cand)
            if cand_score < best_score:
                best_seq, best_score = cand, cand_score
                print(f"[{time.monotonic() - start:.2f}s] restart candidate (n={n}) → {best_score:.6f}")
            last_improve = time.monotonic()

    # ------------------------------------------------------------------
    # 4) Final sanitisation and return
    # ------------------------------------------------------------------
    if best_seq is None:                 # very unlikely, fallback to uniform
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / best_seq.size
    else:
        best_seq /= total

    elapsed = time.monotonic() - start
    print(f"Finished search in {elapsed:.2f}s, final best score {best_score:.6f}")
    return [float(v) for v in best_seq]
