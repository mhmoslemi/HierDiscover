def _optimize_subgrad_adam(a_init, phase_deadline, rng):
    a = a_init.copy()
    n = a.size
    fft_len = _next_pow2(3*n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1=0.9; beta2=0.999; eps=1e-8
    beta1_pow=beta1; beta2_pow=beta2
    start = time.monotonic()
    best_a = a.copy()
    best_sc = evaluate_sequence(a.tolist())
    # config
    LR_START=0.3; LR_END=0.001
    eval_interval=500
    while time.monotonic() < phase_deadline:
        # compute conv and k
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]
        k = np.argmax(conv)
        # compute gradient
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = a[idx[mask]]
        grad = 2.0*grad
        # scheduling
        now = time.monotonic()
        frac = (now - start) / (phase_deadline - start)
        lr = LR_START * (1-frac) + LR_END * frac
        # Adam
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)
        v_hat = v / (1 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = a - step
        # project onto simplex
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = np.full_like(a, 1.0/n)
        # evaluate occasionally
        if total_iters % eval_interval == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a.copy()
        total_iters += 1
    # final evaluate

"""
Propose a non‑negative step‑function that minimizes ``evaluate_sequence``.

Algorithm
---------
1. **Multiple restarts** – each restart picks a random length ``n`` (power of two
   for cheap FFTs) and a random Dirichlet‑drawn initial vector.
2. **Adam on the true max‑convolution** – at every iteration the exact
   convolution ``a*a`` (via FFT) is computed, the index ``k`` of the current
   maximum is identified and the sub‑gradient
   ``∇ = 2·shift_rev(a, k)`` (the reversed shifted copy of ``a``) is fed to an
   Adam optimiser.  After the step a cheap Euclidean simplex projection
   (clip + renorm) keeps the vector feasible.
   The learning rate is linearly annealed from ``0.3`` to ``0.001``.
   The best exact score seen during the phase is recorded.
3. **Stochastic local refinement** – starting from the best vector of the
   Adam phase a large number of tiny mass‑transfer moves (move a random
   fraction of weight from one index to another) are accepted only when they
   improve the exact objective.  Every few thousand moves a small “jitter”
   (random multiplicative factor on a short sub‑segment) is injected to
   escape possible plateaus.
4. The overall best vector from all restarts is returned (clipped to the
   limits of ``evaluate_sequence``).  The routine respects the given time
   budget (≈1000 s) and prints intermediate scores.

The method differs from the previous solution by targeting the *exact*
maximum of the convolution for the gradient (instead of a smooth‑max) and
by performing many independent restarts with diverse initialisations.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= s
    return w


# ----------------------------------------------------------------------
# Core optimisation primitives
# ----------------------------------------------------------------------


def _adam_subgrad_max(
    a0: np.ndarray,
    run_deadline: float,
    rng: np.random.Generator,
    eval_interval: int = 300,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation where the gradient is the exact sub‑gradient
    of the ``max`` of the self‑convolution.
    Returns the best vector encountered and its exact score.
    """
    a = a0.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # learning‑rate schedule
    LR_START, LR_END = 0.30, 0.001

    # bookkeeping
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    it = 0
    start = time.monotonic()

    while time.monotonic() < run_deadline:
        # --- convolution and max index ---
        a_fft = np.fft.rfft(a, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
        k = int(np.argmax(conv))

        # --- sub‑gradient of the current max ---
        idx = k - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad = np.zeros_like(a)
        grad[mask] = a[idx[mask]]
        grad = 2.0 * grad

        # --- Adam step ---
        it += 1
        now = time.monotonic()
        frac = (now - start) / (run_deadline - start)
        lr = LR_START * (1.0 - frac) + LR_END * frac

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = a - step
        # cheap simplex projection (clip + renorm)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full_like(a, 1.0 / n)

        # --- periodical exact evaluation ---
        if it % eval_interval == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                elapsed = now - start
                print(
                    f"[{elapsed:6.2f}s]  Adam phase new best = {best_score:.6f} (iter {it})"
                )
                # early exit flag is handled by the caller
    # final check
    sc = evaluate_sequence(a.tolist())
    if sc < best_score:
        best_score = sc
        best = a.copy()
    return best, best_score


def _local_refinement(
    a0: np.ndarray,
    run_deadline: float,
    rng: np.random.Generator,
    global_best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Tiny stochastic mass‑transfer moves followed by occasional jitter.
    Works on the simplex (mass is conserved, non‑negative).
    """
    a = a0.copy()
    n = a.size
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    move_counter = 0

    while time.monotonic() < run_deadline:
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue

        # move a random fraction (log‑uniform) of the source mass
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.03)))
        delta = a[i] * frac_move
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta  # total sum stays 1

        sc = evaluate_sequence(a_candidate.tolist())
        move_counter += 1

        if sc < best_score - 1e-12:
            best_score = sc
            best = a_candidate.copy()
            a = a_candidate
            elapsed = time.monotonic() - (run_deadline - (run_deadline - time.monotonic()))
            print(
                f"[{elapsed:6.2f}s]  Local move #{move_counter} improved to {best_score:.6f}"
            )
            if best_score <= 1.5025:
                print("[early stop] target reached during local refinement")
                break

        # periodic jitter to avoid stagnation
        if move_counter % 4000 == 0:
            width = rng.integers(1, max(2, n // 40))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            a_j = a.copy()
            a_j[start : start + width] *= factor
            a_j = np.clip(a_j, 0.0, None)
            s = a_j.sum()
            if s > 0.0:
                a_j /= s
            else:
                a_j = np.full_like(a_j, 1.0 / n)
            a = a_j

    # final check
    sc = evaluate_sequence(best.tolist())
    if sc < best_score:
        best_score = sc
        best = best.copy()
    return best, best_score


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises ``evaluate_sequence``.
    The routine uses several random restarts, Adam optimisation on the true
    max‑convolution sub‑gradient and a final stochastic local refinement.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # keep a safety margin for the harness
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # bookkeeping
    # ------------------------------------------------------------------
    global_best_score = np.inf
    global_best = None

    # possible lengths (powers of two simplify the FFT handling)
    length_choices = [2048, 3072, 4096, 5120, 6144, 7168, 8192]

    restart_counter = 0

    while time.monotonic() < deadline:
        # stop if we already have the target
        if global_best_score <= 1.5025:
            break

        # --------------------------------------------------------------
        # 1) pick a random length and initialise a start vector
        # --------------------------------------------------------------
        n = int(rng.choice(length_choices))
        # draw a Dirichlet vector; concentration is random to diversify shapes
        conc = rng.uniform(0.05, 0.5)
        a0 = rng.dirichlet(np.full(n, conc))
        a0 = a0 / a0.sum()  # ensure exact simplex

        # occasionally seed from the historic best if available
        if "height_sequence_1" in globals() and rng.random() < 0.15:
            hist = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if hist.size != n:
                # resample/interpolate to the required length
                x_old = np.linspace(0, 1, hist.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                hist = np.interp(x_new, x_old, hist)
            a0 = np.clip(hist, 0.0, None)
            a0 = a0 / a0.sum()

        restart_counter += 1
        print(f"[{time.monotonic() - start_time:6.2f}s] Restart #{restart_counter}, n={n}")

        # --------------------------------------------------------------
        # 2) Adam on the exact max‑convolution sub‑gradient
        # --------------------------------------------------------------
        phase1_deadline = min(deadline, time.monotonic() + 0.55 * (deadline - start_time))
        a_opt, sc_opt = _adam_subgrad_max(a0, phase1_deadline, rng)

        # --------------------------------------------------------------
        # 3) Stochastic local refinement
        # --------------------------------------------------------------
        phase2_deadline = deadline
        a_opt, sc_opt = _local_refinement(a_opt, phase2_deadline, rng, global_best_score)

        # --------------------------------------------------------------
        # 4) Update global best
        # --------------------------------------------------------------
        if sc_opt < global_best_score - 1e-12:
            global_best_score = sc_opt
            global_best = a_opt.copy()
            elapsed = time.monotonic() - start_time
            print(
                f"[{elapsed:6.2f}s] *** New global best = {global_best_score:.6f} (restart #{restart_counter})"
            )
            if global_best_score <= 1.5025:
                print("[early stop] target reached")
                break

    # ------------------------------------------------------------------
    # 5) Fallback in the unlikely case that we found nothing
    # ------------------------------------------------------------------
    if global_best is None:
        # uniform 1024‑vector as a safe default
        n = 1024
        global_best = np.full(n, 1.0 / n, dtype=np.float64)

    # final clipping to the limits enforced by ``evaluate_sequence``
    global_best = np.clip(global_best, 0.0, 1000.0)
    final_score = evaluate_sequence(global_best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")

    return [float(x) for x in global_best]
