def linear_program_step(a, M):
    # Solve maximize sum(b) st. conv(a,b) <= M.
    # Equivalent to minimize -1^T b.
    n = len(a)
    # Build constraint matrix using csr.
    rows = []
    cols = []
    data = []
    # For each offset i in 0..2n-2
    for i in range(2*n-1):
        # j ranges max(0,i-(n-1)) to min(n-1,i)
        j_start = max(0, i - (n-1))
        j_end = min(n-1, i)
        # for each j, coefficient is a[i-j]
        for j in range(j_start, j_end+1):
            rows.append(i)
            cols.append(j)
            data.append(a[i-j])  # note a index i-j
    # Then A_ub = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    # Solve linear program
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    # if fail, fallback to something else
    b = res.x
    return b, res.success

"""
Hybrid optimizer for the self‑convolution problem.

Stage 1 – stochastic mirror descent / sub‑gradient descent on the exact
          max‑convolution.  Gradient = 2·a shifted by the currently active
          offset.  Two complementary updates are used:
          * multiplicative‑weights (mirror) update,
          * projected Euclidean sub‑gradient update.

Stage 2 – linear‑programming “mass‑inflation” step (in the spirit of the
          reference algorithm).  For the current vector `a` we solve

              max   1ᵀ b
              s.t.  a * b ≤ M   (component‑wise)
                    b ≥ 0

          where `M = max(a*a)`.  The optimal `b` carries more ℓ₁‑mass than
          `a` without increasing the infinity‑norm of the convolution.
          After normalising `b` to a probability vector we convex‑mix it
          into `a`.  Repeating this alternating loop drives the peak
          convolution down further.

Both stages respect the simplex (non‑negative, unit sum) and are
re‑started with a small jitter if no improvement is observed for a
while.  The routine runs for the allocated budget (≈ 1000 s) and returns
the best step‑function seen.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Linear‑algebra helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Real‑valued self‑convolution using FFT (full length 2·n‑1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (no clipping, sum not forced)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


# ----------------------------------------------------------------------
# Simplex utilities
# ----------------------------------------------------------------------


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    n = v.size
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_step(a: np.ndarray, step: float) -> np.ndarray:
    """Projected sub‑gradient step for max‑convolution."""
    n = a.shape[0]
    conv = _conv_self(a)
    kmax = int(np.argmax(conv))
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    a_new = _project_simplex(a - step * grad)
    return a_new


def _mirror_step(a: np.ndarray, eta: float) -> np.ndarray:
    """Multiplicative‑weights (mirror) update for the active offset."""
    n = a.shape[0]
    conv = _conv_self(a)
    kmax = int(np.argmax(conv))
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    # exponentiated update, then re‑normalise to the simplex
    a = a * np.exp(-eta * grad)
    total = a.sum()
    if total == 0.0:
        a = np.full_like(a, 1.0 / n)
    else:
        a = a / total
    return a


# ----------------------------------------------------------------------
# Linear‑programming mass‑inflation step
# ----------------------------------------------------------------------


def _solve_lp_max_sum(a: np.ndarray, M: float):
    """Solve  max 1ᵀb  s.t.   a*b ≤ M ,  b ≥ 0  (exact LP, returns b)."""
    n = a.shape[0]

    # Build the constraint matrix A (size (2n‑1) × n) where
    # A[i, j] = a[i‑j]  if 0 ≤ i‑j < n else 0.
    # This is a lower‑triangular Toeplitz matrix; we construct it via ``diags``.
    offsets = np.arange(n)                     # 0 … n‑1
    # each diagonal consists of the constant a[d] repeated n times
    diags_data = [np.full(n, a[d], dtype=np.float64) for d in range(n)]
    A = sp.diags(diags_data, offsets, shape=(2 * n - 1, n), format="csr")

    # RHS of the inequality constraints
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear‑programming: maximise sum(b)  → minimise  –1ᵀb
    c = -np.ones(n, dtype=np.float64)

    # Use the HiGHS interior‑point solver (fast for moderate sizes)
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
        if res.success:
            return res.x, True
    except Exception as e:                     # pragma: no cover – safety net
        print("LP solve error:", e)

    # fall‑back: return a zero vector (no improvement possible)
    return np.zeros_like(a), False


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm mixes stochastic mirror descent,
    projected sub‑gradient descent and occasional linear‑programming mass‑inflation.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # Time‑budget handling (≈ 5 s safety margin)
    # --------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start + total_budget

    # --------------------------------------------------------------
    # Choose an initial sequence.
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a = np.clip(a, 0.0, None)
        if a.sum() == 0.0:
            a[:] = 1.0 / a.size
        a = a / a.sum()                     # normalise to a probability vector
    else:
        # fallback: medium‑size uniform vector
        n = 3000
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.shape[0]

    best_a = a.copy()
    best_score = _objective_score(a)
    print(f"[0.00s] initial score {best_score:.6f}")

    # --------------------------------------------------------------
    # Hyper‑parameters
    # --------------------------------------------------------------
    mirror_eta = 0.45          # exponentiated‑gradient steps size
    subgrad_step = 0.12        # projected sub‑gradient steps size
    mix_alpha = 0.27           # convex‑mixing weight of LP‑generated vector
    lp_every = 30              # try an LP step every *lp_every* iterations
    RESTART_WAIT = 45.0        # seconds without improvement → jitter restart
    last_improve = start

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # ---- stochastic update -------------------------------------------------
        if rng.random() < 0.68:
            a = _mirror_step(a, mirror_eta)
        else:
            a = _subgradient_step(a, subgrad_step)

        # maintain feasibility (simplex)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0.0:
            a = np.full_like(a, 1.0 / n)
        else:
            a = a / s

        # ---- evaluate -----------------------------------------------------------
        cur_score = _objective_score(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve-start:6.2f}s] new best {best_score:.6f}")

        # ---- occasional linear‑programming mass‑inflation -----------------------
        if it % lp_every == 0:
            conv = _conv_self(a)
            M = conv.max()
            b_opt, ok = _solve_lp_max_sum(a, M)
            if ok:
                B = b_opt.sum()
                if B > 1e-12:
                    g = b_opt / B                       # normalise to simplex
                    # convex mix (multiple tiny mixes to stabilise)
                    a = (1.0 - mix_alpha) * a + mix_alpha * g
                    a = np.clip(a, 0.0, None)
                    a = a / a.sum()
                    # re‑evaluate after mixing
                    cur_score = _objective_score(a)
                    if cur_score < best_score - 1e-12:
                        best_score = cur_score
                        best_a = a.copy()
                        last_improve = time.monotonic()
                        print(f"[{last_improve-start:6.2f}s] LP mix improvement {best_score:.6f}")

        # ---- stagnation handling ------------------------------------------------
        if time.monotonic() - last_improve > RESTART_WAIT:
            # jitter around the best vector seen so far
            jitter = rng.normal(0.0, 0.011, size=n)
            a = best_a + jitter
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
            # reset step‑sizes (helps escape)
            mirror_eta = 0.45
            subgrad_step = 0.12
            last_improve = time.monotonic()
            print(f"[{last_improve-start:6.2f}s] restart with jitter")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    # little sanity check – convert to plain Python list of floats
    return [float(v) for v in best_a]
