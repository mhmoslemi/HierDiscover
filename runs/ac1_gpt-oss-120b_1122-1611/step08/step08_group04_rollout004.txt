"""
Hybrid optimiser for the self‑convolution minimisation problem.

The optimiser works in three stages:

1️⃣ **Smooth surrogate (Adam)**
   A soft‑max approximation of the infinity‑norm of the self‑convolution is
   minimised with Adam.  The temperature (β) is ramped up to a large value
   (≈2000) so that the surrogate tightly follows the true max.  This stage
   quickly finds a good basin.

2️⃣ **Exact sub‑gradient descent with momentum**
   After the surrogate has converged the exact max‑convolution is used.
   A projected sub‑gradient step (with Nesterov‑style momentum) is taken
   on the simplex.  The step size decays as 1/√t and is adapted
   multiplicatively on success/failure.

3️⃣ **Random local jitter (simulated‑annealing style)**
   The remaining time is spent applying tiny random mass transfers between
   coordinates and accepting any improvement.  This helps escape tiny plateaus
   that the deterministic steps cannot see.

All three phases keep the best sequence observed and finally return it
(clipped to the limits required by `evaluate_sequence`).  The routine
honours the 1000 s time‑budget (5 s safety margin).

"""

import time
import numpy as np
from scipy.special import logsumexp
import scipy.sparse as sp

# ----------------------------------------------------------------------
# Low‑level helpers (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a (real‑valued) via FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value returned by `evaluate_sequence` for a (sum(a) is not forced)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    # Fast path
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient(a: np.ndarray) -> tuple[np.ndarray, int]:
    """Return sub‑gradient of max​(a*a) and the active offset."""
    n = a.shape[0]
    conv = _conv_self(a)
    kmax = int(np.argmax(conv))
    idx = np.arange(n)
    j = kmax - idx
    mask = (j >= 0) & (j < n)
    grad = np.zeros_like(a)
    grad[mask] = 2.0 * a[j[mask]]
    return grad, kmax


def _build_conv_constraint_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the sparse matrix A (shape (2n‑1, n)) such that
    (A @ b) == a * b  (linear convolution, zero‑padding outside).
    """
    n = a.shape[0]
    # Indices for the COO representation
    i = np.repeat(np.arange(n, dtype=np.int32), n)          # row = i + j
    j = np.tile(np.arange(n, dtype=np.int32), n)           # column = j
    rows = i + j
    data = np.repeat(a, n)                                 # coefficient = a[i]
    A = sp.coo_matrix((data, (rows, j)), shape=(2 * n - 1, n))
    return A.tocsr()


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(
    seed: int = 12345,
    budget_s: float = 1000.0,
    **kwargs,
) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method runs three phases described in the
    module docstring and returns the best sequence found before the time
    budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start + total_budget

    # ------------------------------------------------------------------
    # Initialise a – try to reuse the best sequence discovered so far.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        base = base / base.sum()
        n = max(3000, base.size)
        if base.size != n:
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(new_x, old_x, base)
            a = np.clip(a, 0.0, None)
            a = a / a.sum()
        else:
            a = base.copy()
    else:
        n = 4096                     # a power of two is convenient for the FFT
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # always keep a on the simplex
    a = _project_simplex(a)

    best_a = a.copy()
    best_score = _objective_score(a)
    print(f"[{0.0:.2f}s] initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smooth surrogate of the max‑convolution
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(a + eps)                 # logits → simplex via soft‑max
    m = np.zeros_like(x)                # Adam first moment
    v = np.zeros_like(x)                # Adam second moment

    # Adam hyper‑parameters
    lr0 = 0.45
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # Soft‑max temperature schedule
    beta_start = 30.0
    beta_end = 2000.0
    ramp_iters = 8000                    # enough iterations for a deep ramp

    phase1_end = start + total_budget * 0.45   # ~45 % of the budget
    it = 0
    last_improve = start

    # FFT length for convolutions (shared across iterations)
    fft_len_conv = _next_pow2(2 * n - 1)

    while time.monotonic() < phase1_end:
        it += 1
        # logits → probability vector (sum = 1)
        a = _softmax_logits(x)

        # linear self‑convolution a * a via FFT
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]

        cur_max = conv_full.max()
        cur_score = 2.0 * n * cur_max          # denominator = 1 (sum == 1)

        # keep the best seen so far
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase‑1 new best {best_score:.6f}")

        # temperature for the smooth max
        beta = beta_end if it > ramp_iters else beta_start + (beta_end - beta_start) * it / ramp_iters

        # soft‑max over the convolution (smooth surrogate of max)
        scaled = beta * conv_full
        lse = logsumexp(scaled)                     # log‑sum‑exp
        w = np.exp(scaled - lse)                    # shape (2n‑1,), sums to 1

        # gradient of the surrogate w.r.t. a (correlation w⋆a)
        fft_len_corr = _next_pow2(3 * n - 2)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)

        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # chain rule → gradient w.r.t. logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam step (learning‑rate decays slowly)
        lr = lr0 / (1.0 + it * 1e-4)

        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # occasional jitter to escape flat regions
        if it % 500 == 0:
            x += rng.normal(0.0, 0.02, size=n)

        # hard restart after prolonged stagnation
        if time.monotonic() - last_improve > 35.0:
            # re‑initialise logits around the best point found so far
            x = np.log(best_a + eps) + rng.normal(0.0, 0.15, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - start:6.2f}s] phase‑1 restart")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    momentum = np.zeros_like(a)
    mu = 0.9                      # momentum factor
    step0 = 0.3
    iter2 = 0
    last_improve_sub = time.monotonic()
    phase2_end = start + total_budget * 0.90   # leave ~10 % for the final jitter phase

    while time.monotonic() < phase2_end:
        iter2 += 1
        grad, kmax = _subgradient(a)

        # step size decays as 1/√t and is scaled by the norm of the gradient
        step = step0 / np.sqrt(iter2) / (np.linalg.norm(grad) + 1e-12)

        # Nesterov‑style momentum
        momentum = mu * momentum + grad
        a_candidate = _project_simplex(a - step * momentum)

        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            last_improve_sub = time.monotonic()
            print(f"[{last_improve_sub - start:6.2f}s] subgrad improvement {best_score:.6f}")
            # slowly increase step0 after a success to accelerate
            step0 *= 1.02
        else:
            # on failure shrink step0 and momentum
            step0 *= 0.75
            momentum *= 0.7

        # occasional small random kick to break ties
        if iter2 % 700 == 0:
            jitter = rng.normal(0.0, 0.0015, size=n)
            a = _project_simplex(a + jitter)

        # hard restart if nothing improved for a while
        if time.monotonic() - last_improve_sub > 30.0:
            a = _project_simplex(best_a + rng.normal(0.0, 0.005, size=n))
            momentum.fill(0.0)
            step0 = 0.3
            last_improve_sub = time.monotonic()
            print(f"[{last_improve_sub - start:6.2f}s] subgrad restart")

    # ------------------------------------------------------------------
    # Phase 3 – Tiny random mass swaps (simulated‑annealing flavour)
    # ------------------------------------------------------------------
    a = best_a.copy()
    while time.monotonic() < deadline:
        # pick two distinct indices
        i, j = rng.choice(n, size=2, replace=False)
        # move a small fraction of mass from the larger to the smaller entry
        if a[i] > a[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        delta = rng.uniform(0.0, min(0.02, a[src]))   # at most 2 % of src mass
        if delta <= 0.0:
            continue
        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[dst] += delta
        # projection is not needed – the operation keeps the sum unchanged
        cand_score = _objective_score(a_candidate)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            print(f"[{time.monotonic() - start:6.2f}s] random‑swap improvement {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
```
