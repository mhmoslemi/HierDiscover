Search for a non‑negative step function that minimises
``evaluate_sequence`` (i.e. 2·n·max(a*a)/sum(a)^2).
The algorithm combines many short random restarts with an
Adam optimiser that uses a sub‑gradient of the max‑convolution
(soft‑max weight limited to the *active* indices only).  After each
restart a cheap stochastic local refinement is applied.  The
objective is evaluated with a fast FFT‑based routine, so we can
check the score on every iteration and keep the best solution
found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # renormalise to avoid tiny drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence for valid inputs)
# ----------------------------------------------------------------------


def _fast_score(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Compute 2·n·max(conv(a,a)) / (sum(a)**2) using FFT.
    Assumes a contains only finite non‑negative numbers.
    """
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    max_b = np.max(conv)
    s = a.sum()
    if s < 0.01:          # matches the guard in evaluate_sequence
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Find a list of non‑negative heights (step function) with a low
    evaluate_sequence score.  The routine runs for up to ``budget_s``
    seconds (5 s safety margin) and returns the best sequence found.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Problem size
    # --------------------------------------------------------------
    # Larger n gives more freedom; 8192 works comfortably within the budget.
    n = 8192
    fft_len = _next_pow2(3 * n)          # >= 3n for safe linear convolution

    # --------------------------------------------------------------
    # 1) Initialise best solution (use previous construction if any)
    # --------------------------------------------------------------
    best = np.full(n, 1.0 / n, dtype=np.float64)   # uniform fallback
    if "height_sequence_1" in globals():
        h = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        h = np.clip(h, 0.0, None)
        # resample / interpolate to the current length n
        if h.size != n:
            old_x = np.linspace(0.0, 1.0, h.size)
            new_x = np.linspace(0.0, 1.0, n)
            h = np.interp(new_x, old_x, h)
        if h.sum() > 0.0:
            h = h / h.sum()
        best = h.copy()

    best_score = _fast_score(best, n, fft_len)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 2) Hyper‑parameters (tuned for speed & quality)
    # --------------------------------------------------------------
    # Adam parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr_start, lr_end = 0.20, 0.001          # linear annealing per restart

    # Overall loop: many short random restarts
    restart_cnt = 0
    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:          # not enough time for a full restart
            break

        # ----------------------------------------------------------
        # 2a) Build a fresh starting point
        # ----------------------------------------------------------
        # Occasionally start from the best seen so far plus a small jitter.
        if restart_cnt % 5 == 0 and restart_cnt > 0:
            a = best.copy()
            jitter = rng.normal(0.0, 0.02, size=n)        # small Gaussian noise
            a = a + jitter
            a = np.clip(a, 0.0, None)
            a = _project_onto_simplex(a)
        else:
            # random sparse seed (5‑30 non‑zero entries) mixed with a tiny uniform background
            k_nonzero = rng.integers(5, 31)
            idx = rng.choice(n, size=k_nonzero, replace=False)
            vals = rng.random(k_nonzero)
            vals /= vals.sum()
            a = np.zeros(n, dtype=np.float64)
            a[idx] = vals
            a = 0.7 * a + 0.3 * (1.0 / n)                 # keep a small uniform mass
            a = _project_onto_simplex(a)

        # ----------------------------------------------------------
        # 2b) Adam optimisation with *sub‑gradient* of the max
        # ----------------------------------------------------------
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1_pow, beta2_pow = beta1, beta2

        # allocate a modest number of iterations – the cost is cheap
        max_iter = int(min(2000, remaining * 800))   # ~800 iters per second
        if max_iter < 200:
            max_iter = 200

        for it in range(max_iter):
            # FFT of current vector
            a_fft = np.fft.rfft(a, n=fft_len)

            # linear self‑convolution (real part)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]

            # indices attaining the maximal value
            max_val = conv.max()
            mask = (conv == max_val)
            w = mask.astype(float) / mask.sum()        # uniform over active set

            # gradient = 2 * correlation(a, w)
            w_fft = np.fft.rfft(w, n=fft_len)
            corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
            grad = 2.0 * corr_full[n - 1:n - 1 + n]

            # Adam updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            beta1_pow *= beta1
            beta2_pow *= beta2
            m_hat = m / (1.0 - beta1_pow)
            v_hat = v / (1.0 - beta2_pow)

            # linear learning‑rate schedule
            lr = lr_start * (1.0 - it / max_iter) + lr_end * (it / max_iter)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)

            a -= step
            a = _project_onto_simplex(a)

            # fast exact score (no extra convolution)
            cur_score = 2.0 * n * max_val / (a.sum() ** 2)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (restart {restart_cnt}) = {best_score:.6f}")

            # early termination if target already reached
            if best_score <= 1.5025:
                print("[early stop] target reached during Adam phase")
                deadline = time.monotonic()  # break outer loops
                break

        # ----------------------------------------------------------
        # 2c) Very cheap stochastic local refinement (tiny moves)
        # ----------------------------------------------------------
        # Use the best vector found so far as the seed for the local phase.
        a_loc = best.copy()
        moves = int(min(20000, (deadline - time.monotonic()) * 5000))
        for mv in range(moves):
            i = rng.integers(0, n)
            if a_loc[i] <= 0.0:
                continue
            j = rng.integers(0, n)
            if i == j:
                continue
            # fraction moved: log‑uniform between 0.1 % and 10 %
            frac = np.exp(rng.uniform(np.log(0.001), np.log(0.10)))
            delta = a_loc[i] * frac
            cand = a_loc.copy()
            cand[i] -= delta
            cand[j] += delta
            cand = _project_onto_simplex(cand)   # keep feasibility (very cheap)

            # fast score
            a_fft = np.fft.rfft(cand, n=fft_len)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
            sc = 2.0 * n * conv.max() / (cand.sum() ** 2)

            if sc < best_score - 1e-12:
                best_score = sc
                best = cand.copy()
                a_loc = cand
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (local {restart_cnt}) = {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached during local phase")
                    deadline = time.monotonic()
                    break

        restart_cnt += 1

    # --------------------------------------------------------------
    # 3) Final directed refinement (gradient‑guided mass transfer)
    # --------------------------------------------------------------
    if time.monotonic() < deadline:
        a_cur = best.copy()
        # simple loop – move a small percentage from the highest‑gradient
        # index to the lowest‑gradient index.
        while time.monotonic() < deadline:
            a_fft = np.fft.rfft(a_cur, n=fft_len)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
            max_val = conv.max()
            mask = (conv == max_val)
            w = mask.astype(float) / mask.sum()
            w_fft = np.fft.rfft(w, n=fft_len)
            corr_full = np.fft.irfft(np.conj(a_fft) * w_fft, n=fft_len)
            grad = 2.0 * corr_full[n - 1:n - 1 + n]

            i_max = int(np.argmax(grad))
            i_min = int(np.argmin(grad))

            if a_cur[i_max] <= 0.0:
                # fallback: pick any positive index
                pos = np.where(a_cur > 0.0)[0]
                if pos.size == 0:
                    break
                i_max = int(rng.choice(pos))

            delta = a_cur[i_max] * 0.03      # move 3 % of the donor mass
            a_cur[i_max] -= delta
            a_cur[i_min] += delta
            a_cur = _project_onto_simplex(a_cur)

            # evaluate
            a_fft = np.fft.rfft(a_cur, n=fft_len)
            conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
            sc = 2.0 * n * conv.max() / (a_cur.sum() ** 2)

            if sc < best_score - 1e-12:
                best_score = sc
                best = a_cur.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] new best (directed) = {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached during directed phase")
                    break

    # --------------------------------------------------------------
    # 4) Return the best sequence (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
