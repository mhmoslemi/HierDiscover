def _evaluate_J(a):
    sum_a = a.sum()
    # safety: if sum too small
    if sum_a < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    max_b = conv.max()

"""
Iterative LP‑based refinement for the step‑function problem.

The algorithm works with a modest sequence length (n≈200‑500) and
repeatedly solves a linear program:

    maximise   Σ g_j
    subject to f∗g  ≤  M          (element‑wise)
               g_j ≥ 0 , g_j ≤ 1000

where f is the current sequence, M = max(f∗f) and “∗’’ denotes the linear
convolution.  The LP finds a distribution g that can carry more ℓ₁‑mass
while respecting the same convolution bound.  After scaling g to the
prescribed ℓ₁‑norm ( √(2n) ) we either replace f by g (if it improves the
objective) or mix the two vectors.  The objective J = 2·n·max(f∗f)/‖f‖₁²
is invariant under scaling, so we keep the ℓ₁‑norm fixed for numerical
stability.

Several initialisations (uniform, exponential decays, Dirichlet draws,
sparse random supports) are tried for each problem size; the best
candidate over all runs is returned.  Random block perturbations and a
simple annealing schedule help to escape plateaus.  The whole process
honours the 1000 s time budget.
"""

import time
import numpy as np
import cvxpy as cp


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("input must be 1‑D")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _evaluate_J(a: np.ndarray) -> float:
    """Exact objective J = 2·n·max(a*a) / (∑a)²."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = np.convolve(a, a)
    return 2.0 * a.size * conv.max() / (s * s)


def _initial_f_candidates(n: int, rng: np.random.Generator,
                         sum_target: float) -> list[np.ndarray]:
    """Diverse starting points, all with ℓ₁‑norm = sum_target."""
    cand = []

    # uniform
    cand.append(np.full(n, sum_target / n, dtype=np.float64))

    # exponential decays
    for r in (0.98, 0.95, 0.90, 0.85, 0.80):
        v = r ** np.arange(n, dtype=np.float64)
        v = v / v.sum() * sum_target
        cand.append(v)

    # Dirichlet draws (moderately concentrated)
    for _ in range(5):
        v = rng.dirichlet(np.ones(n) * 0.3)
        cand.append(v * sum_target)

    # sparse random support (size ≈ √(2n))
    k = max(2, int(np.sqrt(2.0 * n)))
    for _ in range(3):
        inds = rng.choice(n, size=k, replace=False)
        v = np.zeros(n, dtype=np.float64)
        v[inds] = sum_target / k
        cand.append(v)

    return cand


def _solve_lp(f: np.ndarray, M: float, rng: np.random.Generator) -> np.ndarray | None:
    """
    Solve   maximise Σ g_j
            s.t.   (f ∗ g)_k ≤ M   for all k
                  0 ≤ g_j ≤ 1000
    Returns the optimal g (or None if the problem is infeasible).
    """
    n = f.size
    rows = 2 * n - 1

    # Build the convolution matrix C where (C @ g)_k = (f∗g)_k
    C = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f

    g = cp.Variable(n, nonneg=True)
    constraints = [
        C @ g <= M * np.ones(rows),
        g <= 1000.0
    ]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000)
    except Exception:                       # fallback if ECOS is unavailable
        prob.solve(solver=cp.OSQP, warm_start=True, max_iter=5000)

    if g.value is None:
        return None
    return np.maximum(g.value, 0.0)


def _lp_iterative_opt(f0: np.ndarray,
                      sum_target: float,
                      deadline: float,
                      rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Starting from f0, repeatedly solve the LP described above and
    replace/mix the current sequence.  Stops when the time budget expires.
    Returns the best sequence encountered and its objective value.
    """
    f = f0.copy()
    best_f = f.copy()
    best_J = _evaluate_J(f)

    # mixing coefficient – slowly reduced to favour new proposals
    t_initial = 0.35
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # current convolution bound
        M = np.convolve(f, f).max()

        g_opt = _solve_lp(f, M, rng)
        if g_opt is None:
            break

        sum_g = g_opt.sum()
        if sum_g <= 0.0:
            break

        # scale g so that ℓ₁‑norm = sum_target
        scale = sum_target / sum_g
        g = g_opt * scale

        J_g = _evaluate_J(g)

        # Accept an improvement immediately
        if J_g < best_J - 1e-12:
            f = g
            best_f = g.copy()
            best_J = J_g
            # shrink the mixing factor after a successful move
            t = max(0.05, t_initial * (0.97 ** it))
            continue

        # otherwise perform a convex combination
        t = max(0.05, t_initial * (0.97 ** it))
        f = (1.0 - t) * f + t * g
        f = f * (sum_target / f.sum())   # renormalise to keep ℓ₁ constant

        # occasional stochastic perturbation to shake off plateaus
        if rng.random() < 0.06:
            width = rng.integers(1, max(2, f.size // 12))
            start = rng.integers(0, f.size - width + 1)
            factor = np.exp(rng.normal(0.0, 0.18))
            f[start:start + width] *= factor
            f = np.clip(f, 0.0, None)
            f = f * (sum_target / f.sum())

        J_f = _evaluate_J(f)
        if J_f < best_J - 1e-12:
            best_f = f.copy()
            best_J = J_f

        # early exit if we have already beaten the target
        if best_J <= 1.5025:
            break

    return best_f, best_J


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine tries several problem sizes,
    generates a portfolio of initial guesses and improves each of them by the
    LP‑based scheme described above.  The best vector found before the
    time budget expires is returned.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    best_vec = None
    best_score = float('inf')

    # problem sizes to explore (small enough for fast LP solves)
    n_candidates = [200, 300, 400, 500]

    # optionally incorporate the existing construction (if present) as a
    # candidate size – we cap it to keep the LP cheap.
    if "height_sequence_1" in globals():
        existing_len = len(height_sequence_1)
        if 100 <= existing_len <= 800:
            n_candidates.append(existing_len)

    n_candidates = sorted(set(n_candidates))

    for n in n_candidates:
        if time.monotonic() >= deadline:
            break

        sum_target = np.sqrt(2.0 * n)      # fixing ℓ₁‑norm makes J = max(f∗f)

        # generate a small pool of starting points
        start_pool = _initial_f_candidates(n, rng, sum_target)

        # quick evaluation of the pool – keep the single best as a seed
        for cand in start_pool:
            Jc = _evaluate_J(cand)
            if Jc < best_score:
                best_score = Jc
                best_vec = cand.copy()
                print(f"[{time.monotonic():.2f}s] n={n} pool best J={best_score:.6f}")

        # allocate a slice of the remaining budget to this size
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            break
        per_n_budget = remaining * 0.85 / len(n_candidates)
        n_deadline = time.monotonic() + per_n_budget

        # run the LP‑based optimisation from each seed (limited by time)
        for seed_vec in start_pool:
            if time.monotonic() >= n_deadline:
                break
            improved_vec, improved_J = _lp_iterative_opt(seed_vec, sum_target, n_deadline, rng)
            if improved_J < best_score:
                best_score = improved_J
                best_vec = improved_vec.copy()
                print(f"[{time.monotonic():.2f}s] n={n} new global best J={best_score:.6f}")

            # stop early if we are already under the target bound
            if best_score <= 1.5025:
                break

        if best_score <= 1.5025:
            break

    # Fallback – if nothing was produced (highly unlikely) return a simple uniform vector
    if best_vec is None:
        n_fallback = 150
        best_vec = np.full(n_fallback, np.sqrt(2.0 * n_fallback) / n_fallback)

    # Clip to the numeric limits enforced by ``evaluate_sequence`` (0 ≤ x ≤ 1000)
    best_vec = np.clip(best_vec, 0.0, 1000.0)

    return [float(x) for x in best_vec]
