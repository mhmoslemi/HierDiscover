def propose_candidate(seed=12345, budget_s=1000.0, **kwargs):
    """
    Multi‑phase optimizer for the self‑convolution max‑norm problem.
    • Phase 1 – Adam on a temperature‑scaled softmax surrogate of the max convolution.
    • Phase 2 – Projected gradient descent on the L2 deviation of the convolution from its average.
    • Phase 3 – Projected sub‑gradient descent on the exact maximum.
    Random restarts, jitter and simple learning‑rate schedules are used.
    The best sequence found before the deadline is returned.
    """
    import time, numpy as np
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + max(0.0, float(budget_s) - 5.0)

    # Determine length n:
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        n = max(3000, base.size)
        if base.size != n:
            # linear interpolation to size n
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(new_x, old_x, base)
        else:
            a = base.copy()
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a_sum
    else:
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = a.copy()
    best_score = _objective_score(best_a)

    # Phase 1: Adam on softmax surrogate.
    # We'll allocate roughly 50% of remaining budget.
    phase1_deadline = start + (deadline - start) * 0.5

def _adam_softmax(a0, rng, deadline, best_a, best_score):
    a = a0.copy()
    n = a.shape[0]
    x = np.log(a + 1e-12)   # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.3
    beta_start = 5.0
    beta_end = 5000.0
    beta_ramp_iters = 5000
    it = 0
    # precomputations: fft_len for conv, wmx
    fft_len_conv = _next_pow2(2 * n - 1)
    while time.monotonic() < deadline:
        it += 1
        # softmax logits to probabilities
        a = _softmax_logits(x)
        # compute convolution via FFT
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[:2*n-1]
        cur_max = conv.max()
        cur_score = 2.0 * n * cur_max
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.monotonic() - start:.2f}s] phase1 new best {best_score:.6f}")
        # softmax temperature
        beta = beta_end if it > beta_ramp_iters else beta_start + (beta_end - beta_start) * it / beta_ramp_iters
        # softmax over conv values
        scaled = beta * conv
        lse = logsumexp(scaled)
        w = np.exp(scaled - lse)   # shape 2n-1
        # gradient of surrogate = 2 * correlation of w with a

        fft_len_corr = _next_pow2(3 * n - 2)

        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)

        grad_a = 2.0 * corr_full[start_idx:start_idx + n]
        # gradient w.r.t logits (softmax derivative)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + eps)

        if it % 500 == 0:

def _l2_flatten(a0, rng, deadline, best_a, best_score):
    a = a0.copy()
    n = a.shape[0]
    avg = 1.0 / (2 * n - 1)
    step = 0.5
    # precompute FFT length for conv of length (2n-1 + n -1) = 3n-2
    fft_len = _next_pow2(3 * n - 2)
    while time.monotonic() < deadline:
        conv = _conv_self(a)  # via FFT
        diff = conv - avg
        # compute gradient = 2 * conv(diff, a) slice
        diff_fft = np.fft.rfft(diff, n=fft_len)
        a_fft = np.fft.rfft(a, n=fft_len)
        corr_full = np.fft.irfft(diff_fft * np.conj(a_fft), n=fft_len)
        # slice
        grad = 2.0 * corr_full[n-1:n-1+n]
        # take a step
        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)
        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step *= 1.2
            print(f"[{time.monotonic() - start:.2f}s] L2 improvement {best_score:.6f}")
        else:
            step *= 0.7
            if step < 1e-6:

"""
Multi‑phase optimiser for the self‑convolution max‑norm problem.

The optimiser works on three successive phases:

1. Adam on a temperature‑scaled soft‑max surrogate of the max self‑convolution.
   A gradually increasing “β” turns the soft‑max into an increasingly sharp
   approximation of the infinity norm.  The parameters are tuned for a very
   smooth descent and occasional jitter to escape plateaus.

2. Projected gradient descent on the L2 deviation of the convolution from its
   uniform average.  Minimising this deviation directly flattens the
   convolution, which is the key to lowering the max‑value and therefore the
   evaluation score.

3. Projected sub‑gradient descent on the exact max‑convolution for fine
   polishing.  The step size is adapted based on success / failure and a tiny
   amount of random jitter is injected near stagnation.

Random restarts, simple learning‑rate schedules and the use of the global
`height_sequence_1` (if present) give a good chance of improving the current
best bound (≈1.513206) within the 1000 s budget.  The best sequence found is
returned (clipped to the limits enforced by `evaluate_sequence`).
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (real‑valued)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a probability vector (sum = 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())   # denominator = 1 because sum(a)=1


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Phase 1 – Adam on a soft‑max surrogate
# ----------------------------------------------------------------------


def _adam_softmax_phase(a0: np.ndarray,
                        rng: np.random.Generator,
                        deadline: float,
                        start_time: float,
                        best_a: np.ndarray,
                        best_score: float) -> tuple[np.ndarray, float]:
    """Adam optimiser on the soft‑max approximation of max(conv)."""
    a = a0.copy()
    n = a.shape[0]

    # logits (soft‑max parametrisation)
    eps = 1e-12
    x = np.log(a + eps)

    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # Adam hyper‑parameters
    lr = 0.35
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # temperature schedule
    beta_start = 5.0
    beta_end = 20000.0               # very sharp surrogate at the end
    beta_ramp_iters = 8000

    it = 0
    fft_len_conv = _next_pow2(2 * n - 1)

    while time.monotonic() < deadline:
        it += 1

        # probability vector
        a = _softmax_logits(x)

        # self‑convolution via FFT
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]

        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] phase‑1 new best {best_score:.6f}")

        # temperature (β) – slowly increases to “hard” max
        if it <= beta_ramp_iters:
            beta = beta_start + (beta_end - beta_start) * it / beta_ramp_iters
        else:
            beta = beta_end

        # smooth max via log‑sum‑exp
        scaled = beta * conv
        lse = logsumexp(scaled)                 # stable log‑sum‑exp
        w = np.exp(scaled - lse)                # soft‑max weights, sum to 1

        # gradient of the surrogate: 2 * correlation(w, a)
        fft_len_corr = _next_pow2(3 * n - 2)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)
        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # chain rule through the soft‑max parametrisation
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # jitter to shake out of plateaus
        if it % 600 == 0:
            x += rng.normal(0.0, 0.012, size=n)

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 2 – Projected gradient descent on L2 deviation of conv
# ----------------------------------------------------------------------


def _l2_flatten_phase(a0: np.ndarray,
                      rng: np.random.Generator,
                      deadline: float,
                      start_time: float,
                      best_a: np.ndarray,
                      best_score: float) -> tuple[np.ndarray, float]:
    """Gradient descent on the squared deviation of the convolution from uniform."""
    a = a0.copy()
    n = a.shape[0]
    avg = 1.0 / (2 * n - 1)          # target constant value of the convolution
    step = 0.5
    fft_len = _next_pow2(3 * n - 2)

    while time.monotonic() < deadline:
        conv = _conv_self(a)
        diff = conv - avg               # deviation from the average

        # gradient = 2 * (diff * a)  (correlation)
        diff_fft = np.fft.rfft(diff, n=fft_len)
        a_fft = np.fft.rfft(a, n=fft_len)
        corr_full = np.fft.irfft(diff_fft * np.conj(a_fft), n=fft_len)
        grad = 2.0 * corr_full[n - 1:n - 1 + n]

        # perform a projected step
        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step *= 1.25                # enlarge step after a success
            print(f"[{time.monotonic() - start_time:6.2f}s] L2‑phase new best {best_score:.6f}")
        else:
            step *= 0.70                # shrink step after a failure
            if step < 1e-5:
                step = 0.02
            # occasional jitter when stuck
            if rng.random() < 0.001:
                a = _project_simplex(a + rng.normal(0.0, 0.001, size=n))

    return best_a, best_score


# ----------------------------------------------------------------------
# Phase 3 – Projected sub‑gradient descent on exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_phase(a0: np.ndarray,
                       rng: np.random.Generator,
                       deadline: float,
                       start_time: float,
                       best_a: np.ndarray,
                       best_score: float) -> tuple[np.ndarray, float]:
    """Simple projected sub‑gradient method on the exact max‑convolution."""
    a = a0.copy()
    n = a.shape[0]
    step = 0.09

    while time.monotonic() < deadline:
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # gradient of the active offset (single active index)
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        a_candidate = _project_simplex(a - step * grad)
        cand_score = _objective_score(a_candidate)

        if cand_score < best_score - 1e-12:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step *= 1.07
            print(f"[{time.monotonic() - start_time:6.2f}s] sub‑grad new best {best_score:.6f}")
        else:
            step *= 0.68
            if step < 1e-7 or rng.random() < 0.001:
                # small random perturbation to escape flat regions
                a = _project_simplex(a + rng.normal(0.0, 0.002, size=n))
                step = max(step, 0.01)

    return best_a, best_score

# Main entry point required by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method runs three optimisation phases
    (Adam on a smooth surrogate, L2‑flattening, and exact sub‑gradient) and
    returns the best sequence discovered before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # Time handling (reserve a small safety margin for the harness)
    # ------------------------------------------------------------------
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)
    deadline = start_time + total_budget

    # ------------------------------------------------------------------
    # Choose a working length and an initial distribution.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        if base.sum() == 0.0:
            base[:] = 1.0 / base.size
        n = max(3000, base.size)          # work with a reasonably large vector
        if base.size != n:
            # linear interpolation to the new size
            old_x = np.linspace(0.0, 1.0, base.size, endpoint=False)
            new_x = np.linspace(0.0, 1.0, n, endpoint=False)
            a = np.interp(new_x, old_x, base)
            a = np.clip(a, 0.0, None)
        else:
            a = base.copy()
        a_sum = a.sum()
        if a_sum == 0.0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a_sum
    else:
        n = 3840
        a = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = a.copy()
    best_score = _objective_score(best_a)
    print(f"initial score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on the soft‑max surrogate
    # ------------------------------------------------------------------
    phase1_deadline = start_time + total_budget * 0.48   # 48 % of time
    best_a, best_score = _adam_softmax_phase(best_a, rng,
                                             phase1_deadline,
                                             start_time,
                                             best_a,
                                             best_score)

    # ------------------------------------------------------------------
    # Phase 2 – L2‑flattening (gradient descent on deviation from uniform)
    # ------------------------------------------------------------------
    phase2_deadline = start_time + total_budget * 0.85   # up to 85 % of time
    best_a, best_score = _l2_flatten_phase(best_a, rng,
                                           phase2_deadline,
                                           start_time,
                                           best_a,
                                           best_score)

    # ------------------------------------------------------------------
    # Phase 3 – Exact sub‑gradient descent for final polishing
    # ------------------------------------------------------------------
    best_a, best_score = _subgradient_phase(best_a, rng,
                                            deadline,
                                            start_time,
                                            best_a,
                                            best_score)

    # ------------------------------------------------------------------
    # Return the best sequence (respecting the evaluator’s clipping limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"final best score {best_score:.6f} (length {best_a.size})")
    return [float(v) for v in best_a]
