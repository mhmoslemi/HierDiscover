assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the step‑function heights) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `*` denotes the linear convolution.  
Because both the numerator and denominator scale quadratically, the problem is
equivalent to minimising the largest entry of the self‑convolution of a **probability
vector** (`∑a = 1`).  

The implementation below follows a two‑phase scheme that improves on the
previous code:

1. **Smoothed‑max Adam optimisation (Phase I)**
   * Parameters are stored as log‑weights `x` (`a = softmax(x)`).
   * The non‑smooth `max` is replaced by a temperature‑controlled
     log‑sum‑exp (`soft‑max`).  
     The temperature `γ` grows **exponentially** from 0.2 to 200,
     giving a very tight approximation in the later stage.
   * Adam with decaying learning‑rate updates `x`.  
   * After each update the vector is **symmetrised** (`a = (a + a[::-1])/2`);
     symmetry is natural for the convolution objective and halves the
     effective search space.
   * Stagnation triggers a soft‑restart (Gaussian noise around the best
     point) and occasional block‑perturbations.

2. **Simulated‑annealing hill climbing (Phase II)**
   * Starts from the best vector of Phase I.
   * Randomly moves a small fraction of mass from one index to another
     (`_move_mass`).  The candidate is symmetrised and re‑normalised.
   * Acceptance follows a temperature schedule (annealing from `0.1` down to
     `1e‑4`); uphill moves are occasionally accepted, allowing the
     algorithm to escape local minima.
   * Every few iterations a larger **block‑perturbation** is tried.
   * If the search stagnates for >30 s a local restart around the current
     best is performed.

3. **Multi‑start** – the whole optimiser is run several times (different
   random seeds, slightly different lengths) until the 1000 s budget is
   exhausted.  The best solution among all runs is returned.

All heavy work (convolutions, gradients) uses FFTs for speed.  Helper
functions are kept at module level (no closures) to satisfy the
execution restrictions.

```python
"""
Propose a step‑function (non‑negative heights on the simplex) that
improves the bound on `evaluate_sequence`.

The algorithm consists of
  • Phase I – Adam optimisation of a smooth surrogate of the max‑convolution.
    Symmetry is enforced at every step and the temperature controlling the
    log‑sum‑exp grows exponentially from 0.2 to 200.
  • Phase II – Simulated‑annealing hill climbing with mass‑move proposals,
    occasional block perturbations and restarts on stagnation.
  • Several independent runs (different seeds / lengths) are performed
    within the total time budget; the best sequence seen is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (must stay at top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution of a 1‑D real array using an FFT of length ``fft_len``.
    Returns the first ``2*len(a)-1`` entries.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient ∂/∂a  of the smooth‑max surrogate  f(a)=∑_k w_k (a*a)_k .
    ``a`` – probability vector (length n).
    ``w`` – weight vector over the 2·n‑1 convolution positions (sums to 1).
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(a_rev.size + w.size - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    # relevant slice corresponds to indices n‑1 … 2·n‑2
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max – returns a probability vector."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force symmetry v_i = v_{n-1-i} and re‑normalise to sum 1."""
    s = (v + v[::-1]) * 0.5
    total = s.sum()
    if total == 0.0:
        s[:] = 1.0 / s.size
    else:
        s /= total
    return s


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor,
    clip to non‑negative, symmetrise and renormalise.
    """
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out = _symmetrize(out)
    return out


def _move_mass(a: np.ndarray, rng: np.random.Generator, step_frac: float) -> np.ndarray:
    """
    Transfer a small amount of probability mass from a random source index
    to a random destination index.  The resulting vector is symmetrised and
    renormalised.
    """
    n = a.size
    i, j = rng.integers(0, n, size=2)
    if i == j or a[i] == 0.0:
        return a
    delta = step_frac * rng.random() * a[i]
    if delta <= 0.0:
        return a
    out = a.copy()
    out[i] -= delta
    out[j] += delta
    out = _symmetrize(out)
    return out


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact objective used by `evaluate_sequence` (sum(a) is 1)."""
    conv = _self_conv(a, fft_len)
    return 2.0 * a.size * conv.max()


# ----------------------------------------------------------------------
# Core optimisation for a single run (independent of the outer budget)
# ----------------------------------------------------------------------


def _run_optimization(seed: int, n: int, run_budget: float) -> tuple[np.ndarray, float]:
    """
    Perform Phase I + Phase II optimisation for a fixed length ``n``.
    Returns the best vector found and its objective value.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + run_budget

    # FFT length for self‑convolution (same for every call)
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initial point – Dirichlet(1) → uniform‑like but randomised
    # ------------------------------------------------------------------
    a0 = rng.dirichlet(np.ones(n).astype(np.float64))
    a0 = _symmetrize(a0)                     # enforce symmetry
    x = np.log(a0 + 1e-12)                    # log‑weights for soft‑max

    # Adam state
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    # tracking best solution of this run
    a = _softmax(x)
    a = _symmetrize(a)
    cur_score = _objective(a, fft_len)
    best_a = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start = time.monotonic()
    total_time = deadline - start

    # ------------------------------------------------------------------
    # Phase I – smoothed‑max Adam optimisation
    # ------------------------------------------------------------------
    phase1_end = start + 0.6 * total_time          # 60 % of the run budget
    gamma_start, gamma_end = 0.2, 200.0            # exponential schedule
    lr_start, lr_end = 0.5, 0.02

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start
        frac = elapsed / total_time

        # current probability vector
        a = _softmax(x)
        a = _symmetrize(a)

        # self‑convolution
        c = _self_conv(a, fft_len)

        # smooth‑max weights (log‑sum‑exp surrogate)
        gamma = gamma_start * (gamma_end / gamma_start) ** frac
        c_shift = c - np.max(c)                   # stability
        w = np.exp(gamma * c_shift)
        w /= w.sum()

        # gradient of the surrogate
        g_a = _gradient_wrt_a(a, w)

        # chain rule to obtain gradient w.r.t. x (soft‑max Jacobian)
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # exact objective – keep the best ever seen
        cur_score = _objective(a, fft_len)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I new best = {best_score:.6f}")

        # restart on stagnation (≈15 s without improvement)
        if now - last_improve > 15.0:
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Phase I restart (stagnation) – best {best_score:.6f}")

        # occasional global jitter
        if rng.random() < 0.015:
            jitter = _random_block_perturb(best_a, rng)
            x = np.log(jitter + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # Phase II – simulated‑annealing hill climbing
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    phase2_start = time.monotonic()
    phase2_total = deadline - phase2_start

    # temperature schedule
    def _temperature(elapsed):
        return 0.1 * (1.0 - elapsed / phase2_total) + 1e-4

    # linear decay of the fraction of mass moved
    def _step_frac(elapsed):
        return 0.25 * (1.0 - elapsed / phase2_total) + 0.01

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        T = _temperature(elapsed)
        step_frac = _step_frac(elapsed)

        # propose a simple mass‑move
        cand = _move_mass(a, rng, step_frac)

        cand_score = _objective(cand, fft_len)

        # Metropolis acceptance
        accept = False
        if cand_score < cur_score:
            accept = True
        else:
            prob = np.exp(-(cand_score - cur_score) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a = cand
            cur_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best_a = a.copy()
                last_improve = now
                print(f"[{now - start:6.2f}s] Phase II new best = {best_score:.6f}")

        # occasional larger block perturbation
        if rng.random() < 0.02:
            block = _random_block_perturb(a, rng)
            block_score = _objective(block, fft_len)
            if block_score < cur_score:
                a = block
                cur_score = block_score
                if block_score < best_score - 1e-12:
                    best_score = block_score
                    best_a = a.copy()
                    last_improve = now
                    print(f"[{now - start:6.2f}s] block‑perturb new best = {best_score:.6f}")

        # restart after long stagnation
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.4, size=n)
            x = np.log(best_a + 1e-12) + noise
            a = _softmax(x)
            a = _symmetrize(a)
            cur_score = _objective(a, fft_len)
            last_improve = now
            print(f"[{now - start:6.2f}s] local restart (stagnation) – score {cur_score:.6f}")

    return best_a, best_score


# ----------------------------------------------------------------------
# Public entry point – the function that will be called by the harness
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    Several independent runs (different lengths and seeds) are performed
    within the time budget; the best sequence found is returned.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)   # safety margin

    # global best across all runs
    global_best_a = None
    global_best_score = float("inf")

    # we will perform a few runs with varying lengths (2000‑5000)
    possible_lengths = [2400, 3000, 3600, 3840, 4200, 4600, 5000]

    # each run gets a slice of the remaining budget
    while time.monotonic() < deadline:
        time_left = deadline - time.monotonic()
        if time_left < 80.0:          # not enough time for a full run
            break

        # pick a random length (prefer powers of two for speed)
        n = int(rng.choice(possible_lengths))

        # allocate about half of the remaining time to this run
        run_budget = min(0.5 * time_left, 400.0)

        run_seed = rng.integers(0, 2**31 - 1)
        best_a, best_score = _run_optimization(run_seed, n, run_budget)

        if best_score < global_best_score:
            global_best_score = best_score
            global_best_a = best_a.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] *** GLOBAL NEW BEST (n={n}) = {global_best_score:.6f}")

    # fallback – uniform distribution if nothing was found (should never happen)
    if global_best_a is None:
        n = 3840
        global_best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by `evaluate_sequence`
    global_best_a = np.clip(global_best_a, 0.0, 1000.0)

    return [float(v) for v in global_best_a]
```
