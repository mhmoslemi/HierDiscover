def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    # Choose n
    init = globals().get('height_sequence_1')
    if init:
        n = len(init)
        a0 = np.clip(np.asarray(init, dtype=np.float64), 0, None)
    else:
        n = 4000  # default
        # random Dirichlet
        a0 = rng.random(n)
    a0 = np.clip(a0, 0, None)
    if a0.sum() > 0:
        a0 = a0 / a0.sum()
    else:
        a0 = np.full(n, 1.0/n)
    x = np.log(a0 + 1e-12)

    # Precompute FFT lengths
    fft_len = _next_pow2(2 * n - 1)

    # Adam states
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1=0.9
    beta2=0.999
    eps=1e-8
    t=0

    best = a0.copy()
    best_score = 2 * n * _self_conv(best, fft_len).max()
    last_improve = time.monotonic()
    start = time.monotonic()
    deadline = start + max(0.0, budget_s - 6.0)   # leave margin 6 secs

    phase1_end = start + 0.7 * (deadline - start)
    phase2_end = start + 0.9 * (deadline - start)

    while time.monotonic() < deadline:
        now = time.monotonic()
        a = _softmax(x)
        c = _self_conv(a, fft_len)
        cur_score = 2 * n * c.max()

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{now-start:6.2f}s] best={best_score:.6f}")

        # Determine loss gradient
        if now < phase1_end:
            avg = 1.0 / (2 * n - 1)
            r = c - avg
            g_a = 2.0 * _gradient_wrt_a(a, r)   # variance gradient
        elif now < phase2_end:
            # mixture: 0.5 var, 0.5 max
            avg = 1.0 / (2 * n - 1)
            r = c - avg
            g_var = 2.0 * _gradient_wrt_a(a, r)
            idx = np.argmax(c)
            w = np.zeros_like(c)
            w[idx] = 1.0
            g_max = _gradient_wrt_a(a, w)
            # combine with weight based on phase progress
            mix = (now - phase1_end) / (phase2_end - phase1_end)   # 0->1
            g_a = (1 - mix) * g_var + mix * g_max
        else:
            idx = np.argmax(c)
            w = np.zeros_like(c); w[idx] = 1.0
            g_a = _gradient_wrt_a(a, w)

        # chain rule to x
        avg_g = np.dot(a, g_a)
        g_x = a * (g_a - avg_g)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * g_x
        v = beta2 * v + (1 - beta2) * (g_x * g_x)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        # learning rate schedule: start larger, decays
        total = deadline - start
        elapsed = now - start
        lr_start = 0.3
        lr_end = 0.01
        lr = lr_start * (1 - elapsed/total) + lr_end * (elapsed/total)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # Stagnation restart
        if now - last_improve > 25.0:
            # new random start around best
            noise = rng.normal(0.0, 0.3, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0); v.fill(0.0); t = 0
            last_improve = now
            print(f"[{now-start:6.2f}s] restart due to stagnation")
            continue

        # occasional jitter
        if rng.random() < 0.01:
            # random block perturb of current a
            a_jit = _random_block_perturb(a, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0); v.fill(0.0); t = 0

    # final local search using random block perturbation (simulate annealing)
    local_start = time.monotonic()
    a_cur = best.copy()
    score_cur = best_score
    while time.monotonic() < deadline:
        a_new = _random_block_perturb(a_cur, rng)
        c_new = _self_conv(a_new, fft_len)
        score_new = 2 * n * c_new.max()
        if score_new < score_cur:
            a_cur = a_new
            score_cur = score_new
            best = a_new.copy()
            best_score = score_new
            print(f"[{time.monotonic()-start:6.2f}s] local improvement {best_score:.6f}")
        # limit loops
        if time.monotonic() - local_start > 5.0:
            break

    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]

"""
propose_candidate
=================

A three‑phase optimiser that seeks a non‑negative “step function’’ (a
probability vector `a` with sum = 1) minimising

        2·n·max(conv(a,a)) / (sum(a)²)

The algorithm works inside the given time budget (≈ 1000 s) and
returns the best sequence seen.

Phases
------
1. **Variance flattening** – minimise Σ (conv – avg)², i.e. drive the
   self‑convolution towards a flat constant.  This is the most
   important phase; we devote ~70 % of the budget to it.

2. **Mixed optimisation** – blend the variance gradient with a direct
   sub‑gradient of the current maximal convolution entry.  The weight
   of the “max” part increases linearly, guiding the optimisation from
   a smooth landscape to the true objective.

3. **Pure max‑gradient + local jitter** – finish with a pure
   sub‑gradient of the maximal entry (the non‑smooth part of the loss)
   and occasional random block perturbations to escape shallow minima.

Implementation notes
--------------------
* Parameters are stored as logits `x` and transformed with a soft‑max,
  guaranteeing non‑negativity and unit sum.
* Gradients are obtained analytically via FFT‑based convolution, which
  is O(n log n) and cheap for n ≈ 4000 – 8000.
* Adam with a linear learning‑rate decay drives the updates; stagnation
  triggers a soft restart around the current best solution.
* The routine works for any length; if a previous construction is
  available via the global `height_sequence_1` it is used as a seed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must stay at top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k w.r.t. a.
    Returns a vector of length n (the length of ``a``).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max; output sums to one."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random block, then re‑normalise."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.2))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises `evaluate_sequence`.  Returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5‑second safety margin for the harness)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Choose a length and initialise log‑weights
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64).reshape(-1)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            if s0 > 0.0:
                a0 /= s0
            else:
                a0 = np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)
        else:
            n = 5000
            x = np.zeros(n, dtype=np.float64)
    else:
        # default length; a longer vector gives a tighter theoretical bound
        n = 6000
        x = np.zeros(n, dtype=np.float64)

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (used repeatedly)
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)          # for a*a

    # --------------------------------------------------------------
    # 3) Adam optimiser state (shared by all phases)
    # --------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # --------------------------------------------------------------
    # 4) Tracking the best solution seen so far
    # --------------------------------------------------------------
    a = _softmax(x)
    c = _self_conv(a, fft_len_self)
    best_a = a.copy()
    best_score = 2.0 * n * c.max()               # sum(a)=1 ⇒ denominator = 1
    last_improve = start_time

    # --------------------------------------------------------------
    # 5) Phase schedule (times are absolute moments)
    # --------------------------------------------------------------
    total_budget = deadline - start_time
    phase1_end = start_time + 0.70 * total_budget      # variance flattening
    phase2_end = start_time + 0.90 * total_budget      # mixed phase
    # phase3 runs until the deadline

    # --------------------------------------------------------------
    # 6) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----------------------------------------------------------
        # Current a and its self‑convolution
        # ----------------------------------------------------------
        a = _softmax(x)
        c = _self_conv(a, fft_len_self)

        # exact objective value (the quantity we finally care about)
        cur_score = 2.0 * n * c.max()

        # ----------------------------------------------------------
        # Record improvement
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # Choose gradient according to the current phase
        # ----------------------------------------------------------
        if now < phase1_end:
            # ------- Phase 1 : variance flattening -------
            avg = 1.0 / (2 * n - 1)
            r = c - avg                                 # residual (length 2n‑1)
            g_a = 2.0 * _gradient_wrt_a(a, r)           # d/da Σ (c−avg)²
        elif now < phase2_end:
            # ------- Phase 2 : mix variance & max -------
            # linear interpolation between the two objectives
            mix = (now - phase1_end) / (phase2_end - phase1_end)   # 0 → 1
            # variance part
            avg = 1.0 / (2 * n - 1)
            r = c - avg
            g_var = 2.0 * _gradient_wrt_a(a, r)

            # max‑entry part (sub‑gradient of max)
            idx_max = np.argmax(c)
            w = np.zeros_like(c)
            w[idx_max] = 1.0
            g_max = _gradient_wrt_a(a, w)

            # blend – we also normalise each component to keep scales comparable
            # (optional – here we simply linearly combine)
            g_a = (1.0 - mix) * g_var + mix * g_max
        else:
            # ------- Phase 3 : pure max‑gradient -------
            idx_max = np.argmax(c)
            w = np.zeros_like(c)
            w[idx_max] = 1.0
            g_a = _gradient_wrt_a(a, w)

        # ----------------------------------------------------------
        # Chain rule: from a‑gradient to x‑gradient (soft‑max Jacobian)
        # ----------------------------------------------------------
        avg_g = np.dot(a, g_a)            # ⟨a,∂F/∂a⟩
        g_x = a * (g_a - avg_g)           # ∂F/∂x

        # ----------------------------------------------------------
        # Adam step with a linear learning‑rate decay
        # ----------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (smooth decay)
        lr_start = 0.40
        lr_end = 0.02
        lr = lr_start * (1.0 - elapsed / total_budget) + lr_end * (elapsed / total_budget)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----------------------------------------------------------
        # Stagnation handling – soft restart if no improvement for 25 s
        # ----------------------------------------------------------
        if now - last_improve > 25.0:
            # re‑initialise around the current best with Gaussian noise
            noise = rng.normal(0.0, 0.5, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # ----------------------------------------------------------
        # Occasional global jitter to escape shallow minima
        # ----------------------------------------------------------
        if rng.random() < 0.015:
            a_jit = _random_block_perturb(best_a, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # ------------------------------------------------------------------
    # 7) Final short local‑search phase (block jitter)
    # ------------------------------------------------------------------
    # Give the remaining few seconds to a tiny hill‑climbing loop.
    while time.monotonic() < deadline:
        a_candidate = _random_block_perturb(best_a, rng)
        c_cand = _self_conv(a_candidate, fft_len_self)
        cand_score = 2.0 * n * c_cand.max()
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            print(f"[{time.monotonic() - start_time:6.2f}s] local improvement = {best_score:.6f}")
        # break if only a couple of seconds remain
        if deadline - time.monotonic() < 1.0:
            break

    # ------------------------------------------------------------------
    # 8) Return the best sequence (clip to the limits used by evaluate_sequence)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
