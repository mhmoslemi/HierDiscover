def _simplex_projection(v):
    """Projects vector v onto probability simplex (non-negative sum=1). Works for any shape."""
    # see Duchi et al. 2008
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / np.arange(1, n+1) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Projected sub‑gradient optimisation of the max‑convolution objective.

The evaluation function is  

    E(a) = 2·n·max_k (a * a)_k / (sum a)^2 ,

with a_i ≥ 0.  After normalising the vector to a probability simplex
(sum = 1) the denominator disappears and we have to minimise  

    F(a) = 2·n·max_k (a * a)_k .

For a fixed index k* that attains the maximum the (sub‑)gradient of
max_k (a*a)_k is very simple:

    ∂/∂a_i  (a*a)_k* = 2·a_{k*‑i}   if   0 ≤ k*‑i < n,
                       0         otherwise.

We therefore iterate:

* compute the full linear self‑convolution by FFT,
* locate the index k* where it is maximal,
* build the sub‑gradient,
* take a small step and project back onto the simplex
  (the projection guarantees non‑negativity and sum = 1).

To avoid getting stuck we inject occasional random block
perturbations, perform simulated‑annealing style uphill moves,
and restart from the current best after a period of stagnation.
The whole routine is limited to the supplied time budget
(including a 5 s safety margin) and finally returns the best
sequence seen so far as a plain list of floats.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Low‑level utilities (must be top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    # keep only the linear (non‑circular) part
    return conv[: 2 * a.size - 1]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0 , Σ x_i = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """
    Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex.
    """
    n = a.size
    # block width: between 1 and max(2, n//20)
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.14))  # modest log‑normal spread
    out = a.copy()
    out[start:start + width] *= factor
    out = _simplex_projection(out)
    return out


def _gradient_of_max_conv(a: np.ndarray, max_idx: int) -> np.ndarray:
    """
    Sub‑gradient of the objective max_k (a*a)_k at the active index
    ``max_idx``.  Returned vector has length n.
    """
    n = a.size
    grad = np.zeros_like(a)
    # indices i for which 0 ≤ max_idx‑i < n
    i_start = max(0, max_idx - (n - 1))
    i_end = min(n - 1, max_idx)
    if i_start <= i_end:
        idx = np.arange(i_start, i_end + 1, dtype=int)
        grad[idx] = 2.0 * a[max_idx - idx]
    return grad


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) gives a lower
    ``evaluate_sequence`` value than the previous best (~1.5132).
    The routine runs for at most ``budget_s`` seconds (a 5 s safety

    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Determine length and initialise a (probability vector)
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):

        if a.ndim != 1 or a.size == 0:

            s = a.sum()
            if s > 0.0:

            else:

    else:
        # default: a Dirichlet (Gamma) sample of size 4096
        n = 4096

    conv = _self_conv(a, fft_len)
