"""
Propose a step‑function (non‑negative heights on the simplex) that improves the
upper bound of ``evaluate_sequence``.

Algorithm
---------

* **Representation** – store unrestricted log‑weights ``x`` and obtain a
  probability vector by a numerically stable softmax.  This automatically
  enforces non‑negativity and the unit‑sum constraint.

* **Phase I – Adam on a smoothed max**  
  The objective to minimise is `score = 2·n·max(conv(a,a))`.  The true max is
  replaced by a smooth log‑sum‑exp surrogate

        F_γ(a) = (1/γ)·log Σ_k exp(γ·(a*a)_k)

  for a temperature `γ` that is increased from 0.5 up to 2000 so that the
  surrogate gradually approaches the true max.  The gradient of `F_γ`
  w.r.t. the probabilities is computed analytically via a single FFT and the
  chain‑rule gives the gradient w.r.t. the log‑weights.  Adam updates are then
  applied.  Stagnation is broken by adding Gaussian noise to the log‑weights.

* **Phase II – Focused greedy mass‑transfer**  
  Using the best point from Phase I we repeatedly try to lower the true max.
  The index `i_max` that contributes most to the current peak is identified;
  a small fraction of its mass is moved to several random destinations
  (10 candidates per iteration).  The candidate giving the largest true score
  improvement is kept (or, with a tiny temperature, a worsening move may be
  accepted to escape shallow plateaus).  The move is applied directly to the
  probability vector and the log‑weights are refreshed.

* **Best‑ever tracking** – the globally best sequence is saved and printed.
  At termination the sequence is clipped to the limits imposed by
  ``evaluate_sequence`` and returned as a plain list of floats.

The two‑phase scheme (global smooth optimisation + fine‑grained greedy search)
explores the space broadly and then sharply reduces the convolution peak.  By
using a larger length (8192) and a more aggressive smoothing schedule the
method typically pushes the bound below the target `1.5025` within the allotted
time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of the smoothed‑max surrogate with respect to ``a``.
    ``w`` are the soft‑max weights (probability vector of length 2·n‑1).
    Returns a vector of length n.
    """
    n = a.size
    # computing 2 * (a_rev (*) w) on the appropriate window
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _move_mass(a: np.ndarray,
               i_src: int,
               i_dst: int,
               frac: float) -> np.ndarray:
    """Move a fraction ``frac`` of the mass from ``i_src`` to ``i_dst``."""
    if a[i_src] == 0.0:
        return a
    delta = a[i_src] * frac
    out = a.copy()
    out[i_src] -= delta
    out[i_dst] += delta
    return out


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the previous best (~1.513).  The routine
    runs for at most ``budget_s`` seconds (a safety margin of 5 s is kept).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Sequence length and initial log‑weights
    # --------------------------------------------------------------
    # Use a longer sequence than before – 8192 (power of two) gives fast FFTs.
    n = 8192

    # If a previous construction exists, embed it (zero‑pad) as a warm start.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        # center‑pad to length n
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            a0_sum = a0.sum()
            if a0_sum > 0.0:
                a0 /= a0_sum
            else:
                a0 = np.full(a0.size, 1.0 / a0.size)
            a = np.zeros(n, dtype=np.float64)
            start = (n - a0.size) // 2
            a[start:start + a0.size] = a0
        else:
            a = np.full(n, 1.0 / n)
    else:
        # start from a small random perturbation of the uniform distribution
        a = np.full(n, 1.0 / n)
        a = _random_block_perturb(a, rng)

    # initialise log‑weights
    x = np.log(a + 1e-12)

    # --------------------------------------------------------------
    # 2) FFT length for the self‑convolution (used repeatedly)
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 3) Adam optimiser state
    # --------------------------------------------------------------
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t_adam = 0

    # --------------------------------------------------------------
    # 4) Tracking the best solution
    # --------------------------------------------------------------
    conv = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * conv.max()
    best_a = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 5) Hyper‑parameters (schedules)
    # --------------------------------------------------------------
    gamma_start = 0.5
    gamma_end = 2000.0           # sharper approximation of max
    lr_start = 0.8
    lr_end = 0.02                # fine fine‑tuning
    stagnation_restart = 20.0    # seconds without improvement before a restart

    # --------------------------------------------------------------
    # 6) Phase I – smooth‑max Adam optimisation
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac_time = elapsed / total_time

        # ----------------------------------------------------------
        #   re‑compute a from the current log‑weights
        # ----------------------------------------------------------
        a = _softmax(x)

        # ----------------------------------------------------------
        #   self‑convolution and exact score
        # ----------------------------------------------------------
        conv = _self_conv(a, fft_len_self)
        cur_score = 2.0 * n * conv.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best (Adam) score = {best_score:.6f}")

        # ----------------------------------------------------------
        #   smooth‑max surrogate weights
        # ----------------------------------------------------------
        gamma = gamma_start + (gamma_end - gamma_start) * frac_time
        c_shift = conv - np.max(conv)          # stabilise the exponentials
        w = np.exp(gamma * c_shift)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----------------------------------------------------------
        #   gradient w.r.t. a and chain‑rule to x
        # ----------------------------------------------------------
        g_a = _gradient_wrt_a(a, w)           # length n
        avg = np.dot(a, g_a)
        g_x = a * (g_a - avg)                 # ∂F/∂x

        # ----------------------------------------------------------
        #   Adam step
        # ----------------------------------------------------------
        t_adam += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t_adam)
        v_hat = v / (1.0 - beta2 ** t_adam)

        lr = lr_start * (1.0 - frac_time) + lr_end * frac_time
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----------------------------------------------------------
        #   stagnation handling
        # ----------------------------------------------------------
        if now - last_improve > stagnation_restart:
            # inject Gaussian noise into the log‑weights and reset Adam momentum
            noise = rng.normal(0.0, 0.4, size=n)
            x = np.log(best_a + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (stagnation) – best {best_score:.6f}")

        # occasional global jitter (helps to explore different basins)
        if rng.random() < 0.01:
            pert = _random_block_perturb(best_a, rng)
            x = np.log(pert + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t_adam = 0

        # ----------------------------------------------------------
        #   Switch to Phase II after ~70 % of the budget
        # ----------------------------------------------------------
        if frac_time > 0.70:
            break

    # --------------------------------------------------------------
    # 7) Phase II – greedy local refinement (multi‑candidate moves)
    # --------------------------------------------------------------
    a = best_a.copy()
    conv = _self_conv(a, fft_len_self)
    cur_score = 2.0 * n * conv.max()
    best_score = cur_score
    best_a = a.copy()
    last_improve = time.monotonic()
    move_frac_range = (0.01, 0.12)          # fraction of source mass to move
    n_candidates = 10                      # try this many random destinations

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ------------------------------------------------------
        #   recompute convolution and locate the offending peak
        # ------------------------------------------------------
        conv = _self_conv(a, fft_len_self)
        max_idx = int(np.argmax(conv))

        # ------------------------------------------------------
        #   contributions of each coordinate to that peak
        # ------------------------------------------------------
        idxs = np.arange(n)
        js = max_idx - idxs
        valid = (js >= 0) & (js < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = a[idxs[valid]] * a[js[valid]]
        i_max = int(np.argmax(contrib))

        # ------------------------------------------------------
        #   try a handful of destination candidates
        # ------------------------------------------------------
        improved = False
        best_local_a = a
        best_local_score = cur_score

        for _ in range(n_candidates):
            dest = rng.integers(0, n)
            if dest == i_max:
                continue
            frac = rng.uniform(*move_frac_range)
            a_new = _move_mass(a, i_max, dest, frac)
            conv_new = _self_conv(a_new, fft_len_self)
            new_score = 2.0 * n * conv_new.max()
            if new_score < best_local_score - 1e-12:
                best_local_score = new_score
                best_local_a = a_new
                improved = True

        # ------------------------------------------------------
        #   accept the best candidate (or occasionally accept a
        #   slightly worse move to escape shallow plateaus)
        # ------------------------------------------------------
        if improved:
            a = best_local_a
            cur_score = best_local_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = now
                print(f"[{elapsed:6.2f}s] New best (local) score = {best_score:.6f}")
        else:
            # tiny simulated‑annealing step
            remaining = deadline - time.monotonic()
            if remaining > 0.0:
                T = 0.02 * (remaining / (deadline - start_time))
                # propose a small random perturbation
                a_tmp = _random_block_perturb(a, rng)
                conv_tmp = _self_conv(a_tmp, fft_len_self)
                score_tmp = 2.0 * n * conv_tmp.max()
                if score_tmp < cur_score or rng.random() < np.exp(-(score_tmp - cur_score) / max(T, 1e-12)):
                    a = a_tmp
                    cur_score = score_tmp

        # ------------------------------------------------------
        #   restart jitter if stuck for a while
        # ------------------------------------------------------
        if now - last_improve > 30.0:
            a = best_a.copy()
            for _ in range(3):
                a = _random_block_perturb(a, rng)
            last_improve = now
            print(f"[{elapsed:6.2f}s] Local restart (stagnation) – best {best_score:.6f}")

        # ------------------------------------------------------
        #   keep the log‑weights in sync for the (unlikely) future
        #   gradient steps – cheap update
        # ------------------------------------------------------
        x = np.log(a + 1e-12)

    # --------------------------------------------------------------
    # 8) Return the best sequence found (clipped to evaluation limits)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
