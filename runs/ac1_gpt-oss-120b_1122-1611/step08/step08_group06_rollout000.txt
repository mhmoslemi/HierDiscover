"""
Propose a step‑function that minimizes

    2·n·max(conv(a,a)) / (sum(a)^2)

using a two‑phase Adam optimiser:

* Phase 1 (≈60 % of the time) – minimise the L² deviation of the
  self‑convolution from the constant value 1/(2·n‑1).  This “flattens”
  the spectrum and provides a good starting point.

* Phase 2 (the remaining time) – minimise a smooth approximation of the
  max using a log‑sum‑exp surrogate with a γ that grows from 0.5 up to
  200.  The surrogate becomes arbitrarily close to the true max, but
  remains differentiable.

The optimisation works on the log‑softmax parametrisation
`a = softmax(x)`, which guarantees non‑negativity and a unit sum.
The initial vector is a *U‑shaped* Beta(α,β) distribution (α,β≈0.5)
that already yields a relatively flat autocorrelation.  The routine
periodically restarts from the best point seen so far (plus Gaussian
noise) if no improvement occurs for 30 s, and occasionally injects a
random block‑wise jitter to escape shallow minima.

All heavy work (self‑convolution, gradient) is performed with FFTs of
pre‑computed length, which makes the method fast enough to explore
thousands of candidates within the 1000 s budget.

The function returns the best sequence discovered (clipped to the
interval used by ``evaluate_sequence``).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top‑level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k with respect to a.
    Returned vector has length len(a).
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    # indices that correspond to the original (non‑padded) a
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _beta_initial(n: int, alpha: float, beta: float, rng: np.random.Generator) -> np.ndarray:
    """U‑shaped Beta distribution on a discrete grid, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64)
    x = (i + 0.5) / n
    # avoid the singularities when alpha/beta < 1
    eps = 1e-12
    a = np.power(np.clip(x, eps, 1.0 - eps), alpha - 1.0) * \
        np.power(np.clip(1.0 - x, eps, 1.0 - eps), beta - 1.0)
    a = np.clip(a, eps, None)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that gives a
    low value of ``evaluate_sequence``.  The routine uses a two‑phase
    Adam optimiser (variance flattening → smooth‑max) with
    periodic restarts and random block jittters.  It returns the best
    sequence found before the time budget expires.
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Choose a length and initialise log‑weights
    # ------------------------------------------------------------------
    # Try to reuse the best known construction if it exists.
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64)
        if a0.ndim == 1 and a0.size > 0:
            a0 = np.clip(a0, 0.0, None)
            s0 = a0.sum()
            a0 = a0 / s0 if s0 > 0 else np.full_like(a0, 1.0 / a0.size)
            n = a0.size
            x = np.log(a0 + 1e-12)
        else:

            n = 6000
            a0 = _beta_initial(n, 0.5, 0.5, rng)

        n = 6000
        # randomise the shape a little each run

    t = 0

    best_score = 2.0 * n * c.max() / (a.sum() ** 2)   # a.sum() == 1 by construction
    last_improve = time.monotonic()
    start_time = time.monotonic()

        cur_score = 2.0 * n * c.max() / (a.sum() ** 2)

        if cur_score < best_score - 1e-12:

            best = a.copy()

            gamma_start = 0.5

            c_shift = c - np.max(c)                 # max becomes zero

            w_sum = w.sum()

                w[np.argmax(c)] = 1.0
            else:

        avg_g = np.dot(a, g_a)          # ⟨a, ∂F/∂a⟩

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr_start = 0.6
        lr_end   = 0.04
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ---- stagnation handling – restart if nothing improved for 30 s ---
        if now - last_improve > 30.0:

            m.fill(0.0)

            last_improve = now
            print(f"[{elapsed:6.2f}s] *** stagnation restart *** – best score {best_score:.6f}")

            m.fill(0.0)
            v.fill(0.0)
            t = 0
            print(f"[{elapsed:6.2f}s] block jitter applied")

    # ----------------------------------------------------------------------
