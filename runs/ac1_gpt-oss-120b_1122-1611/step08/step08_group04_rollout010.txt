Pseudo:

```python
def _grad_logsumexp_surrogate(a, beta):
    n = a.size
    conv = _conv_self(a)  # full conv length 2n-1
    # compute softmax weights:
    scaled = beta * conv
    lse = logsumexp(scaled)
    w = np.exp(scaled - lse) # shape 2n-1, sum to 1
    # correlation w * a:
    fft_len = _next_pow2(3*n-2)
    w_fft = np.fft.rfft(w, n=fft_len)
    a_fft = np.fft.rfft(a, n=fft_len)
    corr_full = np.fft.irfft(w_fft * np.conj(a_fft), n=fft_len)
    start = n-1
    grad = 2.0 * corr_full[start:start+n]
    return grad
```

```
def hill_climb(a, time_deadline, step_init=0.01, min_step=1e-6, patience=100):
    best_score = _objective_score(a)
    best_a = a.copy()
    step = step_init
    no_improve = 0
    rng = np.random.default_rng()
    while time.monotonic() < time_deadline:
        # propose random move
        i = rng.integers(0, n)
        j = rng.integers(0, n-1)
        if j >= i: j += 1
        eps = step * a[i]
        if eps < 1e-12:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        # ensure non-negative, but we ensure eps <= a[i]
        cand_score = _objective_score(a_candidate)
        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step = min(step * 1.05, 0.5) # maybe increase a bit
            no_improve = 0
        else:
            step = max(step * 0.9, min_step)
            no_improve += 1
            if no_improve > patience:
                # random jitter to escape
                a = a + rng.normal(0, step*0.5, size=n)
                a = np.clip(a, 0, None)
                a = a / a.sum()
                no_improve = 0
    return best_a
```

```
def upsample(seq, new_n):
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(new_x, old_x, seq)
    up = np.clip(up, 0, None)
    up = up / up.sum()
    return up
```

def _adam_optimize(a0, n, deadline, beta_max, lr_initial, beta_start=5.0,
                   beta_ramp_iters=5000, jitter_std=0.01, jitter_every=2000,
                   restart_wait=30.0):
    rng = np.random.default_rng()
    eps = 1e-12
    x = np.log(a0 + eps)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    best_a = _softmax_logits(x)
    best_score = _objective_score(best_a)
    last_improve = time.monotonic()
    it = 0
    while time.monotonic() < deadline:
        it += 1
        a = _softmax_logits(x)
        # compute full conv
        a_fft = np.fft.rfft(a, n=_next_pow2(2 * n - 1))
        conv_full = np.fft.irfft(a_fft * a_fft, n=_next_pow2(2 * n - 1))[:2 * n - 1]
        cur_score = _objective_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            # print progress
        # compute gradient via surrogate
        beta = beta_max if it > beta_ramp_iters else beta_start + (beta_max - beta_start) * it / beta_ramp_iters
        # compute softmax weights
        scaled = beta * conv_full
        max_scaled = np.max(scaled)
        exp_shift = np.exp(scaled - max_scaled)
        w = exp_shift / exp_shift.sum()
        # correlation: w * a
        fft_len = _next_pow2(3 * n - 2)
        w_fft = np.fft.rfft(w, n=fft_len)
        a_fft_corr = np.fft.rfft(a, n=fft_len)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len)
        start = n - 1
        grad_a = 2.0 * corr_full[start:start + n]
        # gradient wrt logits
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)
        # Adam update
        m = 0.9 * m + 0.1 * grad_x
        v = 0.999 * v + 0.001 * (grad_x * grad_x)
        m_hat = m / (1 - 0.9 ** it)
        v_hat = v / (1 - 0.999 ** it)
        # learning rate schedule (linear decay)
        lr = lr_initial * max(0.1, 1 - it / (beta_ramp_iters * 5))
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        # jitter
        if it % jitter_every == 0:
            x += rng.normal(0.0, jitter_std, size=n)
        # restart if no improvement for a while
        if time.monotonic() - last_improve > restart_wait:
            x = np.log(best_a + eps) + rng.normal(0.0, jitter_std * 5, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
    return best_a, best_score

def _hill_climb(a_start, deadline):
    rng = np.random.default_rng()
    a = a_start.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    step = 0.02
    no_improve = 0
    while time.monotonic() < deadline:
        n = a.shape[0]
        i = rng.integers(n)
        if a[i] <= 1e-12:
            continue
        j = rng.integers(n-1)
        if j >= i:
            j += 1
        eps = step * a[i]
        if eps < 1e-12:
            continue
        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps
        # re-normalize? sum unchanged, but due to floating precision, might not sum exactly 1.
        candidate_score = _objective_score(a_candidate)
        if candidate_score < best_score:
            best_score = candidate_score
            best_a = a_candidate.copy()
            a = a_candidate
            step = min(step * 1.05, 0.5)
            no_improve = 0
        else:
            step = max(step * 0.9, 1e-6)
            no_improve += 1
            if no_improve > 2000:
                # jitter whole vector

                a = a + jitter
                a = np.clip(a, 0.0, None)
                a = a / a.sum()
                no_improve = 0
    return best_a, best_score

"""
Hybrid multi‑scale optimizer for the self‑convolution bound.

* Phase 0  – start from an existing construction (if any) or a uniform
  distribution, interpolate it to a very small size (256) and run a fast
  Adam optimizer on a smooth log‑sum‑exp surrogate of the max‑convolution.

* Phase 1‑3 – gradually increase the sequence length (1024 → 4096 → 8192)
  re‑interpolating the current best solution each time.  In every phase we
  continue the Adam optimisation, slowly raising the inverse‑temperature
  β to a large value (up to 4000) so that the surrogate becomes an accurate
  approximation of the true ∞‑norm.  Adam is used on the log‑softmax
  parametrisation (logits → simplex) with bias‑corrected moments and occasional
  jitter/restarts to avoid stagnation.

* Final phase – a targeted hill‑climbing / coordinate‑descent routine that
  repeatedly moves a tiny amount of probability mass from the index that
  contributes most to the current maximal convolution entry to the index that
  contributes the least.  The step size is adapted and, if needed, a small
  random jitter is added.

All helpers are top‑level, no closures are used and the routine respects the
global time budget (≈ 1000 s).  The returned list is already clipped to the
[0, 1000] limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Low‑level utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Soft‑max mapping from logits to a probability vector (sum = 1)."""
    m = np.max(logits)
    e = np.exp(logits - m)
    s = e.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return e / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution a * a using an FFT."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    af = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(af * af, n=fft_len)[: 2 * n - 1]
    return conv


def _objective_score(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)²) for a (no clipping)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max() / (np.sum(a) ** 2))


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a normalized sequence to length ``new_n``."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 1.0, new_n, endpoint=False)
    up = np.interp(new_x, old_x, seq)
    up = np.clip(up, 0.0, None)
    up_sum = up.sum()
    if up_sum == 0.0:
        up[:] = 1.0 / new_n
    else:
        up /= up_sum
    return up


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    beta_max: float,
    lr_initial: float,
    beta_start: float = 5.0,
    beta_ramp_iters: int = 5000,
    jitter_std: float = 0.012,
    jitter_every: int = 2000,
    restart_wait: float = 30.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector (on the simplex) discovered before ``deadline``.
    """
    rng = np.random.default_rng()
    eps = 1e-12
    x = np.log(a0 + eps)                     # logits
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # pre‑compute FFT lengths once
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n - 2)

    best_a = _softmax_logits(x)
    best_score = _objective_score(best_a)
    last_improve = time.monotonic()
    it = 0

    while time.monotonic() < deadline:
        it += 1

        a = _softmax_logits(x)                # probability vector

        # ----- exact self‑convolution (needed for β‑schedule) -----
        af = np.fft.rfft(a, n=fft_len_conv)
        conv = np.fft.irfft(af * af, n=fft_len_conv)[: 2 * n - 1]

        # ----- keep the best exact score -----
        cur_score = _objective_score(a)
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - (deadline - (deadline - start)):.2f}s] "
                  f" n={n} new best {best_score:.6f}")

        # ----- β‑schedule (smooth → max) -----
        if it <= beta_ramp_iters:
            beta = beta_start + (beta_max - beta_start) * it / beta_ramp_iters
        else:
            beta = beta_max

        # ----- soft‑max weights w ∝ exp(β·conv) -----
        scaled = beta * conv
        max_scaled = np.max(scaled)
        exp_shift = np.exp(scaled - max_scaled)          # shift for stability
        w = exp_shift / np.sum(exp_shift)                # shape (2n‑1,)

        # ----- gradient of the surrogate w.r.t. a -----
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)
        start = n - 1
        grad_a = 2.0 * corr_full[start : start + n]      # ∂/∂a

        # ----- chain rule to logits -----
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # ----- Adam update (bias‑corrected) -----
        beta1, beta2 = 0.9, 0.999
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # simple linear decay of the learning rate
        lr = lr_initial * max(0.1, 1.0 - it / (beta_ramp_iters * 5))
        x -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- occasional jitter to escape plateaus -----
        if it % jitter_every == 0:
            x += rng.normal(0.0, jitter_std, size=n)

        # ----- hard restart if stagnation -----
        if time.monotonic() - last_improve > restart_wait:
            x = np.log(best_a + eps) + rng.normal(0.0, jitter_std * 3, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.monotonic()
            print(f"[{last_improve - (deadline - (deadline - start)):.2f}s] "
                  f" n={n} restart (no improvement)")

    return best_a, best_score


def _hill_climb(
    a_start: np.ndarray,
    deadline: float,
    init_step: float = 0.02,
    min_step: float = 1e-6,
) -> tuple[np.ndarray, float]:
    """
    Targeted coordinate descent on the exact max‑convolution.
    Moves probability from the index that contributes most to the current
    maximal convolution entry to the index that contributes the least.
    """
    rng = np.random.default_rng()
    a = a_start.copy()
    best_a = a.copy()
    best_score = _objective_score(a)
    step = init_step
    stagnation = 0

    while time.monotonic() < deadline:
        n = a.shape[0]
        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # contribution of each a_i to conv[kmax]
        idx = np.arange(n)
        partner = kmax - idx
        mask = (partner >= 0) & (partner < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[partner[mask]]

        i_high = int(np.argmax(contrib))
        i_low = int(np.argmin(contrib))

        if i_high == i_low:
            # fallback to a random pair
            i = rng.integers(n)
            j = rng.integers(n - 1)
            if j >= i:
                j += 1
        else:
            i, j = i_high, i_low

        epsilon = step * a[i]
        if epsilon < min_step:
            epsilon = min_step

        # ensure we never make a[i] negative
        epsilon = min(epsilon, a[i])

        a_candidate = a.copy()
        a_candidate[i] -= epsilon
        a_candidate[j] += epsilon

        cand_score = _objective_score(a_candidate)

        if cand_score < best_score:
            best_score = cand_score
            best_a = a_candidate.copy()
            a = a_candidate
            step = min(step * 1.05, 0.5)
            stagnation = 0
        else:
            step = max(step * 0.85, min_step)
            stagnation += 1
            if stagnation > 3000:
                # add a small random jitter to the whole vector
                jitter = rng.normal(0.0, step * 0.5, size=n)
                a = a + jitter
                a = np.clip(a, 0.0, None)
                total = a.sum()
                if total == 0.0:
                    a[:] = 1.0 / n
                else:
                    a /= total
                stagnation = 0

    return best_a, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000.0, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine runs a multi‑scale Adam optimiser followed by a fine‑grained
    hill‑climbing stage, respecting the given time budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)   # small safety margin
    deadline = start + total_budget

    # ------------------------------------------------------------------
    # Initialise from a previously discovered construction if available.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        base = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        base = np.clip(base, 0.0, None)
        s = base.sum()
        if s == 0.0:
            base[:] = 1.0 / base.size
        else:
            base /= s
    else:
        base = None

    def start_sequence(n: int) -> np.ndarray:
        """Return a normalised starting vector of length n."""
        if base is not None:
            return _interpolate_seq(base, n)
        else:
            return np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Multi‑scale Adam optimisation (256 → 1024 → 4096 → 8192)
    # ------------------------------------------------------------------
    # (length, time‑fraction of total budget, initial LR, β‑max)
    phase_cfg = [
        (256, 0.07, 0.5, 400.0),
        (1024, 0.18, 0.4, 800.0),
        (4096, 0.60, 0.25, 2000.0),
        (8192, 0.05, 0.15, 4000.0),
    ]

    best_a = None
    best_score = np.inf
    cumulative = 0.0

    for n, frac, lr_init, beta_max in phase_cfg:
        cumulative += frac
        phase_deadline = start + total_budget * cumulative
        if time.monotonic() >= deadline:
            break
        if time.monotonic() >= phase_deadline:
            continue

        # initialise (interpolate the current best if it exists)
        if best_a is None:
            a0 = start_sequence(n)
        else:
            a0 = _interpolate_seq(best_a, n)

        a_opt, score_opt = _adam_optimize(
            a0,
            n,
            phase_deadline,
            beta_max=beta_max,
            lr_initial=lr_init,
            beta_start=5.0,
            beta_ramp_iters=8000,
            jitter_std=0.012,
            jitter_every=2500,
            restart_wait=30.0,
        )

        if score_opt < best_score:
            best_score = score_opt
            best_a = a_opt.copy()
            print(f"[{time.monotonic() - start:.2f}s] Phase n={n} improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Final fine‑grained local optimisation (targeted hill‑climb)
    # ------------------------------------------------------------------
    if best_a is None:
        # Fallback – very unlikely, but guarantees a valid return value.
        best_a = start_sequence(256)

    final_deadline = deadline
    a_refined, score_refined = _hill_climb(best_a, final_deadline, init_step=0.02)

    if score_refined < best_score:
        best_score = score_refined
        best_a = a_refined
        print(f"[{time.monotonic() - start:.2f}s] Final hill‑climb improved to {best_score:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (clipped to evaluator limits)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
