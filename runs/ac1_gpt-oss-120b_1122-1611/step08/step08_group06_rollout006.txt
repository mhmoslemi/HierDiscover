"""
Hybrid variance‑max optimizer for the step‑function problem.

The algorithm works directly on a probability vector `a` (non‑negative,
sum = 1) and minimizes
        2·n·max(conv(a,a))
by a time‑scheduled combination of two convex surrogates:

* Phase A – minimise the L²‑distance of the self‑convolution from its
  flat average 1/(2·n‑1).  This flattens the whole spectrum.
* Phase B – gradually shift the objective towards the true ℓ∞‑norm.
  The weight λ(t) grows from 0 to 1, and the gradient is a convex
  combination

        g = (1‑λ)·∇ Var(conv) + λ·∇ max(conv)

  where the second term is the sub‑gradient of the maximum (implemented
  as a weighted sum over all maximal indices).  Adam (with a simple
  linear learning‑rate decay) updates the log‑weights `x` of a soft‑max
  parametrisation `a = softmax(x)`, which automatically keeps the
  simplex constraints.

Stagnation is detected (no improvement for 30 s) and a stochastic
restart centred on the current best solution is performed.  Occasionally
a global “block jitter’’ perturbs a random contiguous block to escape
shallow minima.  The routine respects the given time budget (default
1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (must stay at top level – no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)[: 2 * a.size - 1]
    return conv


def _gradient_wrt_a(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Gradient of Σ_k w_k·(a*a)_k  w.r.t. a.
    Returns a vector of length n.
    """
    n = a.size
    a_rev = a[::-1]
    N = _next_pow2(len(a_rev) + len(w) - 1)
    fa = np.fft.rfft(a_rev, n=N)
    fw = np.fft.rfft(w, n=N)
    conv = np.fft.irfft(fa * fw, n=N)
    start = n - 1
    end = start + n
    return 2.0 * conv[start:end]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (outputs a probability vector)."""
    y = x - np.max(x)
    e = np.exp(y)
    return e / e.sum()


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by re‑normalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.3))          # stronger jitter than before
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out

    # 4) Helper converting log‑weights to probabilities
    # --------------------------------------------------------------
    def _a_from_x(vec: np.ndarray) -> np.ndarray:
        return _softmax(vec)

    # --------------------------------------------------------------
    # 5) Tracking the best solution seen so far
    # --------------------------------------------------------------
    a = _a_from_x(x)
    c = _self_conv(a, fft_len_self)
    best = a.copy()
    best_score = 2.0 * n * c.max()          # sum(a)=1 ⇒ denominator =1
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_time = deadline - start_time

    # --------------------------------------------------------------
    # 6) Main optimisation loop
    # --------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        # ----------------------------------------------------------
        # current a and its convolution
        # ----------------------------------------------------------
        a = _a_from_x(x)                     # remains on simplex
        c = _self_conv(a, fft_len_self)      # length 2·n‑1

        # ----------------------------------------------------------
        # exact objective (the quantity we finally care about)
        # ----------------------------------------------------------
        cur_score = 2.0 * n * c.max()

        # ----------------------------------------------------------
        # record improvement
        # ----------------------------------------------------------
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] new best score = {best_score:.6f}")

        # ----------------------------------------------------------
        # scheduling: λ grows from 0 to 1 (quadratically) – controls
        # the trade‑off between variance flattening and max reduction.
        # ----------------------------------------------------------
        lambda_max = (elapsed / total_time) ** 2
        lambda_max = min(1.0, max(0.0, lambda_max))

        # ----------------------------------------------------------
        # 1) variance‑flattening gradient (L2 distance to flat average)
        # ----------------------------------------------------------
        avg = 1.0 / (2 * n - 1)               # target value of each conv entry
        r = c - avg                           # residual (length 2n‑1)
        g_var = 2.0 * _gradient_wrt_a(a, r)   # ∇ Σ (c‑avg)²

        # ----------------------------------------------------------
        # 2) max‑gradient (sub‑gradient of ℓ∞‑norm)
        # ----------------------------------------------------------
        max_val = c.max()
        # all indices achieving the maximum (within machine epsilon)
        max_idxs = np.where(np.abs(c - max_val) < 1e-14)[0]
        w = np.zeros_like(c)
        w[max_idxs] = 1.0 / max_idxs.size       # uniform weight over the arg‑max set
        g_max = _gradient_wrt_a(a, w)            # ∇ max(c)

        # ----------------------------------------------------------
        # 3) convex combination of the two gradients
        # ----------------------------------------------------------
        g_a = (1.0 - lambda_max) * g_var + lambda_max * g_max

        # ----------------------------------------------------------
        # 4) chain rule to obtain gradient w.r.t. log‑weights x
        #    (soft‑max Jacobian)
        # ----------------------------------------------------------
        avg_g = np.dot(a, g_a)               # ⟨a,∂F/∂a⟩
        g_x = a * (g_a - avg_g)              # ∂F/∂x

        # ----------------------------------------------------------
        # 5) Adam step
        # ----------------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g_x
        v = beta2 * v + (1.0 - beta2) * (g_x * g_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # learning‑rate schedule (linear decay)
        lr_start = 0.6
        lr_end = 0.02
        lr = lr_start * (1.0 - elapsed / total_time) + lr_end * (elapsed / total_time)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # ----------------------------------------------------------
        # 6) stagnation handling – restart if no improvement for 30 s
        # ----------------------------------------------------------
        if now - last_improve > 30.0:
            # stochastic restart centred on the best solution seen so far
            noise = rng.normal(0.0, 0.6, size=n)
            x = np.log(best + 1e-12) + noise
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] restart (stagnation) – best score {best_score:.6f}")

        # ----------------------------------------------------------
        # 7) occasional global jitter to escape shallow minima
        # ----------------------------------------------------------
        if rng.random() < 0.03:
            a_jit = _random_block_perturb(best, rng)
            x = np.log(a_jit + 1e-12)
            m.fill(0.0)
            v.fill(0.0)
            t = 0

    # --------------------------------------------------------------
    # 8) Return the best sequence (clip to the limits used by evaluate_sequence)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    return [float(v) for v in best]
