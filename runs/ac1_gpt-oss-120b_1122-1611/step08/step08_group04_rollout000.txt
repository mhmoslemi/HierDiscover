"""
Hybrid optimiser for the self‑convolution bound.

* Phase 1 – Adam on a soft‑max (log‑sum‑exp) surrogate of the infinity‑norm.
  A slow β‑ramp (5 → 2000) and a modest learning‑rate schedule give a very
  accurate gradient of the max‑convolution while keeping the iterates on the
  probability simplex (soft‑max of logits).

* Phase 2 – Projected sub‑gradient descent on the exact max‑convolution.
  A back‑tracking line‑search guarantees monotone improvement; the step size
  is enlarged after a success and shrunk after a failure.  Random jitter
  prevents stagnation.

* Phase 3 – Simulated‑annealing‑style random walk: Gaussian perturbations
  followed by simplex projection, with a temperature schedule that accepts
  occasional uphill moves.  This refines the solution beyond what pure
  sub‑gradient descent attains.

The routine works with any starting sequence supplied in the global variable
`height_sequence_1`.  If that variable is absent a collection of simple
initialisers (uniform, half‑sine, parabola) is used.  The search is bounded
by the supplied ``budget_s`` (default 1000 s) and returns the best sequence
found (clipped to the limits enforced by ``evaluate_sequence``).
"""

import time
import numpy as np
from scipy.special import logsumexp

# ----------------------------------------------------------------------
# Helper utilities – top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _softmax_logits(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (logits → probability simplex)."""
    m = np.max(logits)
    exp = np.exp(logits - m)
    s = exp.sum()
    if s == 0.0:
        return np.full_like(logits, 1.0 / logits.size)
    return exp / s


def _conv_self(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (real‑valued)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    a_fft = np.fft.rfft(a, n=fft_len)
    conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * n - 1]
    return conv_full


def _objective_score(a: np.ndarray) -> float:
    """Exact value of evaluate_sequence for a (sum(a) is assumed to be 1)."""
    n = a.shape[0]
    conv = _conv_self(a)
    return float(2.0 * n * conv.max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x ≥ 0, sum(x)=1}."""
    n = v.shape[0]
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _build_parabola(n: int) -> np.ndarray:
    """Discrete parabola a_i = i * (n - i - 1), normalized to unit sum."""
    i = np.arange(n, dtype=np.float64)
    a = i * (n - i - 1)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _build_half_sine(n: int) -> np.ndarray:
    """Discrete half‑sine a_i ∝ sin(π (i+0.5)/n), normalized."""
    theta = np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.sin(theta)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _interpolate_to_length(src: np.ndarray, n: int) -> np.ndarray:
    """Linear interpolation of src to length n, then normalise to sum 1."""
    if src.size == n:
        a = src.copy()
    else:
        old_x = np.linspace(0.0, 1.0, src.size, endpoint=False)
        new_x = np.linspace(0.0, 1.0, n, endpoint=False)
        a = np.interp(new_x, old_x, src)
    a = np.clip(a, 0.0, None)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

        candidates.append(_interpolate_to_length(base, n))

    # (2) Uniform distribution
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # (3) Half‑sine shape
    candidates.append(_build_half_sine(n))

    # (4) Parabolic shape (zero at boundaries)
    candidates.append(_build_parabola(n))

    # (5) Random Dirichlet draws (few of them)
    for _ in range(3):
        rand = rng.random(n)
        rand /= rand.sum()
        candidates.append(rand)

    # Select the best among the crude initialisers as the starting point
    best_a = None
    best_score = np.inf
    for cand in candidates:
        score = _objective_score(cand)
        if score < best_score:
            best_score = score
            best_a = cand.copy()
    print(f"[0.00s] initial best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on a smooth surrogate of max(conv)
    # ------------------------------------------------------------------
    eps = 1e-12
    x = np.log(best_a + eps)                 # logits → simplex via soft‑max
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    # Adam hyper‑parameters
    lr = 0.45
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    # β‑schedule (smooth‑max temperature)
    beta_start = 5.0
    beta_end = 2000.0
    beta_ramp_iters = 8000

    # Pre‑compute FFT lengths for efficiency
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n - 2)

    it = 0
    last_improve_time = start_time

    while time.monotonic() < deadline_phase1:
        it += 1

        # Current probability vector
        a = _softmax_logits(x)

        # Exact self‑convolution (needed for the surrogate)
        a_fft = np.fft.rfft(a, n=fft_len_conv)
        conv_full = np.fft.irfft(a_fft * a_fft, n=fft_len_conv)[: 2 * n - 1]

        cur_score = 2.0 * n * conv_full.max()    # denominator = 1 (sum = 1)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            last_improve_time = time.monotonic()
            elapsed = last_improve_time - start_time
            print(f"[{elapsed:6.2f}s] phase1 new best {best_score:.6f}")

        # β at this iteration (linear ramp)
        beta = beta_end if it > beta_ramp_iters else \
               beta_start + (beta_end - beta_start) * it / beta_ramp_iters

        # Soft‑max surrogate of the max (log‑sum‑exp)
        scaled = beta * conv_full
        lse = logsumexp(scaled)
        w = np.exp(scaled - lse)                     # weights, sum to 1

        # Gradient of surrogate wrt a:
        # grad_a = 2 * (w ⋆ a) , where ⋆ is the correlation (reverse‑convolution)
        w_fft = np.fft.rfft(w, n=fft_len_corr)
        a_fft_corr = np.fft.rfft(a, n=fft_len_corr)
        corr_full = np.fft.irfft(w_fft * np.conj(a_fft_corr), n=fft_len_corr)
        start_idx = n - 1
        grad_a = 2.0 * corr_full[start_idx:start_idx + n]

        # Chain rule for logits (soft‑max Jacobian)
        dot = np.dot(grad_a, a)
        grad_x = a * (grad_a - dot)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        x -= lr * m_hat / (np.sqrt(v_hat) + adam_eps)

        # Periodic jitter to shake the iteration out of plateaus
        if it % 500 == 0:
            x += rng.normal(0.0, 0.02, size=n)

        # Hard restart if no improvement for a while
        if time.monotonic() - last_improve_time > 45.0:
            x = np.log(best_a + eps) + rng.normal(0.0, 0.12, size=n)
            m.fill(0.0)
            v.fill(0.0)
            last_improve_time = time.monotonic()
            print(f"[{last_improve_time - start_time:6.2f}s] phase1 restart")

    # ------------------------------------------------------------------
    # Phase 2 – Projected sub‑gradient descent on the exact max‑convolution
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = _objective_score(a)
    step = 0.8
    sub_it = 0

    while time.monotonic() < deadline_phase2:
        sub_it += 1

        conv = _conv_self(a)
        kmax = int(np.argmax(conv))

        # Sub‑gradient of the active offset
        idx = np.arange(n)
        j = kmax - idx
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # Back‑tracking line search
        step_try = step
        improved = False
        for ls in range(12):
            a_candidate = _project_simplex(a - step_try * grad)
            cand_score = _objective_score(a_candidate)
            if cand_score < cur_score - 1e-13:
                a = a_candidate
                cur_score = cand_score
                improved = True
                break
            step_try *= 0.5

        if improved:
            # keep the improvement globally if it beats the best so far
            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_a = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] subgrad improvement {best_score:.6f}")
            step = max(step_try, step * 1.05)   # slightly enlarge after success
        else:
            step *= 0.7                          # shrink after failure

        # Occasional jitter to escape flat regions
        if sub_it % 200 == 0 or step < 1e-7:
            jitter = rng.normal(0.0, 0.003, size=n)
            a = _project_simplex(a + jitter)
            cur_score = _objective_score(a)

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing refinement (Gaussian perturbations)
    # ------------------------------------------------------------------
    a_sa = best_a.copy()
    cur_score_sa = best_score
    temp = 0.02
    temp_decay = 0.9995
    sigma = 0.006

    while time.monotonic() < deadline_phase3:
        # Gradually shrink the perturbation magnitude
        sigma *= 0.9999
        proposal = _project_simplex(a_sa + rng.normal(0.0, sigma, size=n))
        prop_score = _objective_score(proposal)

        if prop_score < cur_score_sa - 1e-13:
            a_sa = proposal
            cur_score_sa = prop_score
            if prop_score < best_score - 1e-13:
                best_score = prop_score
                best_a = proposal.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] SA improvement {best_score:.6f}")
        else:
            # Metropolis acceptance
            delta = prop_score - cur_score_sa
            if rng.random() < np.exp(-delta / temp):
                a_sa = proposal
                cur_score_sa = prop_score
        temp *= temp_decay

    # ------------------------------------------------------------------
    # Return the best sequence (clipped to the limits enforced by the evaluator)
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    return [float(v) for v in best_a]
