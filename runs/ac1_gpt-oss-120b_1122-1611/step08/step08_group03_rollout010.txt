def propose_candidate(seed=12345, budget_s=1000.0):
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - 5.0)

    # Determine length n (maybe from global known best)

    if "height_sequence_1" in globals():
        base_seq = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        n = base_seq.size
    else:
        n = 2048

    # generate initial candidates
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0/n))
    # beta shapes
    alphas = [0.4, 0.5, 0.6, 0.8, 1.0]  # 1.0 is uniform, not needed
    for a in alphas:
        x = np.arange(n) / (n-1)
        eps = 1e-12
        x = np.clip(x, eps, 1-eps)
        p = (x**(a-1)) * ((1-x)**(a-1))
        p = p / p.sum()
        candidates.append(p)

    # random Dirichlet draws
    for _ in range(3):
        p = rng.dirichlet(np.ones(n) * 0.5)  # concentration less than 1 yields spiky
        candidates.append(p)

    # if height_sequence_1 exists, add it (already uniform maybe)
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0:
            a0 = a0 / s0
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        candidates.append(a0)

    # Evaluate initial candidates and pick best
    best = None
    best_score = np.inf
    for cand in candidates:
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best = cand.copy()
    print(f"[0.00s] initial best score = {best_score:.6f}")

    # Phase 1: Smooth max Adam
    # Setup Adam for best sequence
    # For each step until time, iterate.
    # We'll incorporate customizing schedule maybe more aggressive.

    n = best.size
    fft_len = _next_pow2(3*n)

    # Adam parameters
    m = np.zeros_like(best)
    v = np.zeros_like(best)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    beta1_pow = beta1
    beta2_pow = beta2

    # We aim to use full f.g. TAU_START=0.5, TAU_END=1e-6, LR_START=0.3, LR_END=1e-3

    phase1_deadline = start_time + 0.40 * (deadline - start_time)

    iteration = 0

    while time.monotonic() < phase1_deadline:
        now = time.monotonic()
        frac = (now - start_time) / (phase1_deadline - start_time)  # 0-1
        tau = 0.5 * (1 - frac) + 1e-6 * frac
        lr = 0.3 * (1 - frac) + 1e-3 * frac
        # compute conv via FFT
        a_fft = np.fft.rfft(best, n=fft_len)
        conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[:2*n-1]

        # compute softmax weights
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)  # safe
        w = exp_shifted / exp_shifted.sum()
        # compute gradient = 2 * conv_rev_w
        a_rev = best[::-1]
        a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
        w_fft = np.fft.rfft(w, n=fft_len)
        corr = np.fft.irfft(a_rev_fft * w_fft, n=fft_len)

        beta2_pow *= beta2
        m_hat = m / (1 - beta1_pow)

"""
Propose a non‑negative step function that (hopefully) beats the previous
score 1.513206.  

The search consists of three stages:

1️⃣ **Exploratory seeding** – try a handful of analytically‑motivated
   shapes (uniform, symmetric Beta‑U‑shapes, harmonic edge‑heavy
   profiles and a few Dirichlet draws).  The best of these is taken as the
   starting point.

2️⃣ **Two‑phase gradient optimisation**  
   • *Phase A* – minimise a smooth approximation of the max‑convolution
     using Adam, but with the **exact** gradient of the log‑sum‑exp
     surrogate (the earlier code mistakenly multiplied by 1/τ).  
   • *Phase B* – switch to the true sub‑gradient of the hard max (only the
     current maximal convolution entry contributes).  Adam is reused
     with a decaying learning‑rate.

   Both phases keep the iterate on the probability simplex via Euclidean
   projection.

3️⃣ **Stochastic local refinement** – mass‑transfer moves between random
   indices are accepted if they improve the exact score, otherwise they
   may be accepted with a Metropolis‑type probability (simulated annealing)
   to escape shallow local minima.  Periodic jitter on short blocks
   prevents stagnation.

The routine respects the time‑budget (default 1000 s, 5 s safety margin)
and prints progress.  The returned list is clipped to the limits enforced
by ``evaluate_sequence`` (0 ≤ x ≤ 1000).  The implementation uses only
NumPy, no external I/O, and all helpers are top‑level functions.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # re‑normalise to avoid numerical drift
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Self‑convolution via FFT (real output, length 2·n−1)."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    return conv


def _grad_smooth_max(a: np.ndarray, w: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of the log‑sum‑exp surrogate (τ·log Σ exp(conv/τ))
    w.r.t. a.  The formula is 2·(a_rev * w) restricted to the original
    length n.  ``w`` must be a probability vector (sum=1).
    """
    a_rev = a[::-1]
    a_rev_fft = np.fft.rfft(a_rev, n=fft_len)
    w_fft = np.fft.rfft(w, n=fft_len)
    corr_full = np.fft.irfft(a_rev_fft * w_fft, n=fft_len)
    # slice [n‑1 : n‑1+n] contains the required n components
    grad = 2.0 * corr_full[a.size - 1 : a.size - 1 + a.size]
    return grad


def _grad_max_convolution(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the hard max ‖a*a‖∞ at the index k_max.
    """
    n = a.size
    grad = np.zeros_like(a)
    # grad[i] = 2·a[k_max‑i] where the index is in range
    i0 = max(0, k_max - (n - 1))
    i1 = min(n - 1, k_max)
    idx = np.arange(i0, i1 + 1)
    grad[idx] = 2.0 * a[k_max - idx]
    return grad

# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence`` under the given time budget.  Returns the best
    sequence found as a plain Python list of floats.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    total_budget = max(0.0, float(budget_s) - 5.0)   # safety margin
    deadline = start_time + total_budget

    # --------------------------------------------------------------
    # 1) Choose a working length n
    # --------------------------------------------------------------
    # If a previous construction is available, keep its size;
    # otherwise use a convenient power of two.
    if "height_sequence_1" in globals():
        n = len(globals()["height_sequence_1"])
    else:
        n = 4096  # power of two simplifies FFT padding

    # --------------------------------------------------------------
    # 2) Generate a small pool of initial candidates
    # --------------------------------------------------------------
    init_candidates = []

    # uniform
    init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # symmetric Beta‑U‑shapes  (α in (0,1) gives edge heavy density)
    alphas = np.linspace(0.25, 0.90, 13)  # 13 values give a reasonable sweep
    x = np.arange(n, dtype=np.float64) / (n - 1)
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)          # avoid 0 or 1 singularities
    for a in alphas:
        p = (x ** (a - 1.0)) * ((1.0 - x) ** (a - 1.0))
        p = p / p.sum()
        init_candidates.append(p)

    # harmonic edge‑heavy families  a_i ∝ 1/(i+1)^p + 1/(n‑i)^p
    for pwr in (0.4, 0.6, 0.9, 1.2):
        idx = np.arange(1, n + 1, dtype=np.float64)
        w = (1.0 / idx ** pwr) + (1.0 / idx[::-1] ** pwr)
        w = w / w.sum()
        init_candidates.append(w)

    # a few random Dirichlet draws (concentration < 1 ⇒ spikier)
    for _ in range(5):
        w = rng.dirichlet(np.full(n, 0.5, dtype=np.float64))
        init_candidates.append(w)

    # include the previous best if it exists
    if "height_sequence_1" in globals():
        prev = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        prev = np.clip(prev, 0.0, None)
        s = prev.sum()
        if s > 0.0:
            prev = prev / s
        else:
            prev = np.full_like(prev, 1.0 / prev.size)
        init_candidates.append(prev)

    # --------------------------------------------------------------
    # 3) Quick scan of the pool (stop early if budget is almost gone)
    # --------------------------------------------------------------
    best = init_candidates[0].copy()
    best_score = float("inf")
    for cand in init_candidates:
        sc = evaluate_sequence(cand.tolist())
        if sc < best_score:
            best_score = sc
            best = cand.copy()
        # bail out of the scan if we have spent ≈10 % of the budget
        if time.monotonic() > start_time + 0.10 * total_budget:
            break
    print(f"[0.00s] initial best score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Phase A – smooth‑max Adam optimisation (≈35 % of the total budget)
    # --------------------------------------------------------------
    phase_a_end = start_time + 0.35 * total_budget
    a = best.copy()
    n = a.size
    fft_len = _next_pow2(3 * n)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    beta1_pow, beta2_pow = beta1, beta2

    # annealing schedules
    TAU_START, TAU_END = 0.5, 1e-6
    LR_START, LR_END = 0.25, 5e-4

    it = 0
    while time.monotonic() < phase_a_end:
        now = time.monotonic()
        progress = (now - start_time) / (phase_a_end - start_time)

        # linearly anneal temperature and learning rate
        tau = TAU_START * (1.0 - progress) + TAU_END * progress
        lr = LR_START * (1.0 - progress) + LR_END * progress

        # self‑convolution (FFT)
        conv = _conv_fft(a, fft_len)

        # soft‑max weights (log‑sum‑exp surrogate)
        shift = conv.max()
        exp_shifted = np.exp((conv - shift) / tau)
        w = exp_shifted / exp_shifted.sum()

        # exact gradient of the surrogate (no 1/τ factor!)
        grad = _grad_smooth_max(a, w, fft_len)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_onto_simplex(a)

        it += 1
        if it % 250 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best (smooth‑max) = {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached in Phase A")
                    break

    # --------------------------------------------------------------
    # 5) Phase B – exact‑max Adam optimisation (≈25 % of the total budget)
    # --------------------------------------------------------------
    phase_b_end = start_time + 0.60 * total_budget
    a = best.copy()                # start from best of Phase A
    m[:] = 0.0
    v[:] = 0.0
    beta1_pow, beta2_pow = beta1, beta2
    it = 0

    while time.monotonic() < phase_b_end:
        now = time.monotonic()
        progress = (now - phase_a_end) / (phase_b_end - phase_a_end)
        lr = 0.15 * (1.0 - progress) + 5e-4 * progress   # a bit more aggressive

        k_max = int(np.argmax(conv))

        # sub‑gradient of the hard max
        grad = _grad_max_convolution(a, k_max)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        a = a - step
        a = _project_onto_simplex(a)

        it += 1
        if it % 200 == 0:
            sc = evaluate_sequence(a.tolist())
            if sc < best_score - 1e-12:
                best_score = sc
                best = a.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best (exact‑max) = {best_score:.6f}")
                if best_score <= 1.5025:
                    print("[early stop] target reached in Phase B")
                    break

    # --------------------------------------------------------------
    # 6) Phase C – stochastic local refinement with simulated annealing
    #    (remaining budget)
    # --------------------------------------------------------------
    a = best.copy()
    current_score = evaluate_sequence(a.tolist())
    move_counter = 0
    # annealing temperature schedule for the Metropolis criterion
    TEMP_START, TEMP_END = 0.12, 1e-5

    while time.monotonic() < deadline:
        now = time.monotonic()
        # linear temperature decay over the local‑search window
        frac = (now - phase_b_end) / (deadline - phase_b_end)
        temperature = TEMP_START * (1.0 - frac) + TEMP_END * frac

        # ----- propose a tiny mass transfer -----
        i = rng.integers(0, n)
        if a[i] <= 0.0:
            continue
        j = rng.integers(0, n)
        if i == j:
            continue
        # move a random fraction (log‑uniform) of the source mass
        frac_move = np.exp(rng.uniform(np.log(0.001), np.log(0.10)))  # up to 10 %
        delta = a[i] * frac_move
        a_candidate = a.copy()
        a_candidate[i] -= delta
        a_candidate[j] += delta

        cand_score = evaluate_sequence(a_candidate.tolist())
        move_counter += 1

        # ----- acceptance test -----
        accept = False
        if cand_score < current_score:
            accept = True
        else:
            # Metropolis acceptance (allows escaping shallow minima)
            prob = np.exp(-(cand_score - current_score) / max(temperature, 1e-12))
            accept = rng.random() < prob

        if accept:
            a = a_candidate
            current_score = cand_score
            if cand_score < best_score - 1e-12:
                best_score = cand_score
                best = a_candidate.copy()
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] new best (local) = {best_score:.6f} "
                      f"(move #{move_counter})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in Phase C")
                    break

        # occasional block‑wise jitter to diversify the search space
        if move_counter % 4000 == 0:
            width = rng.integers(1, max(2, n // 50))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.015))
            a_j = a.copy()
            a_j[start:start + width] *= factor
            a = _project_onto_simplex(a_j)
            # keep current_score up‑to‑date
            current_score = evaluate_sequence(a.tolist())

    # --------------------------------------------------------------
    # 7) Return the best sequence found (apply evaluator’s caps)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
