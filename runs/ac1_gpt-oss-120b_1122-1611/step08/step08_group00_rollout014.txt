    # Set n

        a = np.full(n, 1.0/n)
    else:
        a = base / base_sum

    # FFT lengths
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)
    fft_len_grad = _next_pow2(3 * n - 2)

    # Helper function to compute max conv and J
    def compute_max_and_J(vec):
        fa = np.fft.rfft(vec, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        max_b = conv.max()
        return max_b, float(2 * n * max_b)

    # Keep best
    max_b, best_J = compute_max_and_J(a)
    best_vec = a.copy()
    print(f"Initial J={best_J}")

    # Hyperparameters for subgradient descent
    beta1, beta2 = 0.9, 0.999  # for Adam if used
    # We'll use simple Polyak step size and projection

    # Main loop
    it = 0
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        # compute convolution
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        max_b = conv.max()
        cur_J = float(2 * n * max_b)
        # if better
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = time.monotonic()
            print(f"[{last_improve - (deadline - (budget_s - 5)):.2f}s] new best J={best_J}")

        # compute active mask (tight constraints)
        eps = 1e-12
        active = np.abs(conv - max_b) <= eps
        # optionally extend active set for smoothing (like select top few)
        # create weight array w of size conv_len
        # For simplicity we use binary weights
        w = active.astype(float)

"""
Proposed sequence optimizer.

The algorithm works in two phases:

1️⃣  Gradient‑descent phase (Adam on the simplex):
   – The variables are kept on the probability simplex by Euclidean projection.
   – A temperature‑controlled soft‑max approximates the nonsmooth max
     of the self‑convolution.  The gradient of the smoothed objective is
     obtained analytically via a second FFT.
   – Learning‑rate and soft‑max temperature are annealed (lr: 0.5→0.001,
     α: 10→5000).  Periodic block perturbations and a restart‑after‑stagnation
     scheme keep the search from getting stuck.

2️⃣  Permutation‑local‑search phase:
   – The best vector from phase 1 is taken as a starting point.
   – Random swaps of two positions are proposed; the new max‑convolution
     is recomputed with an FFT and the swap is kept if it improves the
     objective (or accepted with a tiny probability to escape plateaus).
   – This cheap combinatorial refinement often reduces the peak of the
     convolution a little more than gradient steps alone.

The whole routine respects the 1000 s budget (5 s safety margin) and
returns the best sequence it has found.  All numbers are non‑negative,
sum‑normalised (so the evaluation ratio equals `2·n·max(b)`), and finally
clipped to the limits imposed by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – top level, no closures
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]                     # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    out_sum = out.sum()
    if out_sum == 0.0:
        out[:] = 1.0 / n
    else:
        out /= out_sum
    return out


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The method first refines an existing good
    construction (if provided via the global ``height_sequence_1``) with a
    projected‑Adam optimiser that smooths the max‑convolution, then performs
    a fast permutation‑swap local search.
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 1) Problem dimensions
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            base = np.asarray(height_sequence_1, dtype=np.float64)
            base = np.clip(base, 0.0, None)          # protect against negatives
            n = base.size
        except Exception:
            base = None
            n = 4096
    else:
        base = None
        n = 4096                     # default length if nothing is provided

    # enforce a modest but reasonably large size (avoid >5000 for speed)
    n = max(512, min(n, 4096))

    # ------------------------------------------------------------------
    # Prepare FFT lengths – we reuse them throughout the run
    # ------------------------------------------------------------------
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # a * w_rev

    # ------------------------------------------------------------------
    # Helper to compute the true objective J = 2·n·max(a*a)
    # (the denominator sum(a)^2 cancels out after normalisation)
    # ------------------------------------------------------------------
    def compute_J(vec: np.ndarray) -> float:
        fa = np.fft.rfft(vec, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
        return float(2.0 * n * conv.max())

    # ------------------------------------------------------------------
    # Initialise a feasible point (sum = 1, non‑negative)
    # ------------------------------------------------------------------
    if base is not None and base.size == n:
        a = base.copy()
        s = a.sum()
        if s < 1e-12:
            a = np.full(n, 1.0 / n)
        else:
            a = a / s
        print(f"[init] Using provided sequence (len={n})")
    else:
        a = np.full(n, 1.0 / n)
        print(f"[init] No good seed – starting from uniform (len={n})")

    best_vec = a.copy()
    best_J = compute_J(best_vec)
    print(f"[init] J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 2) Phase 1 – Projected Adam on the smoothed max‑convolution
    # --------------------------------------------------------------
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    start_time = time.monotonic()
    # reserve ~30 s at the end for the permutation phase
    perm_time = 30.0
    adam_deadline = deadline - perm_time

    last_improve = start_time
    it = 0
    while time.monotonic() < adam_deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_time) / (adam_deadline - start_time)   # ∈[0,1]

        # ----------------------------------------------------------
        # learning‑rate & temperature schedules
        # ----------------------------------------------------------
        lr_start, lr_end = 0.5, 0.001
        lr = lr_start * (1.0 - frac) + lr_end * frac

        alpha_start, alpha_end = 10.0, 5000.0
        alpha = alpha_start * (1.0 - frac) + alpha_end * frac

        # ----------------------------------------------------------
        # self‑convolution (full) and soft‑max weights
        # ----------------------------------------------------------
        fa = np.fft.rfft(a, n=fft_len_self)
        conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]

        c_max = conv.max()
        # soft‑max on the convolution values, numerically stable
        shifted = conv - c_max
        w = np.exp(alpha * shifted)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----------------------------------------------------------
        # gradient of the smoothed objective
        # ----------------------------------------------------------
        w_rev = w[::-1]
        fa_a = np.fft.rfft(a, n=fft_len_grad)
        fa_w = np.fft.rfft(w_rev, n=fft_len_grad)
        full = np.fft.irfft(fa_a * fa_w, n=fft_len_grad)
        start_idx = w_rev.size - 1                # = 2n‑2
        grad = 2.0 * full[start_idx:start_idx + n]

        # ----------------------------------------------------------
        # Adam update + projection onto simplex
        # ----------------------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # ----------------------------------------------------------
        # evaluate current objective
        # ----------------------------------------------------------
        cur_J = compute_J(a)
        if cur_J < best_J - 1e-12:
            best_J = cur_J
            best_vec = a.copy()
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Adam new best J = {best_J:.6f}")

        # occasional jitter to help escape plateaus
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart if no improvement for >25 s
        if now - last_improve > 25.0:
            a = best_vec.copy()
            # tiny multiplicative noise before restarting
            noise = np.exp(rng.normal(0.0, 0.07, size=n))
            a = a * noise
            a = proj_simplex(a)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] restart (stagnation) – best J = {best_J:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 2 – Permutation local search (swap moves)
    # --------------------------------------------------------------
    # Allocate the remaining budget for swaps
    swap_deadline = deadline
    cur_vec = best_vec.copy()
    cur_J = best_J
    t_swap_start = time.monotonic()
    attempts = 0
    accepts = 0

    # Fixed tiny temperature for occasional uphill moves
    temp = 1e-4

    while time.monotonic() < swap_deadline:
        attempts += 1
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # propose a swap
        cand = cur_vec.copy()
        cand[i], cand[j] = cand[j], cand[i]

        cand_J = compute_J(cand)

        if cand_J < cur_J - 1e-12:
            cur_vec = cand
            cur_J = cand_J
            accepts += 1
            if cand_J < best_J - 1e-12:
                best_J = cand_J
                best_vec = cand.copy()
                print(f"[{time.monotonic() - start_time:6.2f}s] swap new best J = {best_J:.6f}")
        else:
            # very small probability of uphill acceptance
            if rng.random() < np.exp(-(cand_J - cur_J) / temp):
                cur_vec = cand
                cur_J = cand_J
                accepts += 1

        # occasional heavy perturbation to escape deep local minima
        if attempts % 20000 == 0:
            cur_vec = _random_block_perturb(cur_vec, rng)

    print(f"Swap phase finished – total attempts: {attempts}, accepts: {accepts}")
    print(f"Best J found = {best_J:.6f}")

    # --------------------------------------------------------------
    # 4) Return the best sequence (clip to evaluation limits)
    # --------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
