"""
Propose a step‑function that minimises
    J(a) = 2·n·max(conv(a,a)) / (sum(a))²
where conv is the linear self‑convolution.
The algorithm consists of three stages:

1️⃣  **Diverse warm‑starts** – a portfolio of analytic shapes
    (uniform, exponential, linear, beta‑family) together with a few
    Dirichlet draws and the previous best solution (if available).

2️⃣  **Simulated‑annealing global search** – a cheap pairwise mass‑transfer
    move that respects the simplex, accepted with a temperature schedule.
    This quickly discovers non‑trivial structures that lower the peak of the
    convolution.

3️⃣  **Projected Adam refinement** – a smooth surrogate of the
    max‑operator (`softmax`) is built from the current convolution.
    Gradients are obtained analytically via FFTs and Adam with simplex
    projection is used to polish the solution.

All helper routines are top‑level (no closures) and the routine respects a
1000 s time budget (5 s safety margin).  The returned list is clipped to the
bounds enforced by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Linear‑algebra / geometry helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.size - 1]
    return conv


def compute_J(a: np.ndarray, fft_len: int) -> float:
    """Exact objective J (sum(a) is assumed to be 1)."""
    conv = _self_conv(a, fft_len)
    return float(2.0 * a.size * conv.max())


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Log‑normal multiplicative perturbation of a random contiguous block."""
    n = a.size
    width = rng.integers(1, max(2, n // 20))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    total = out.sum()
    if total == 0.0:
        out[:] = 1.0 / n
    else:
        out /= total
    return out


def _initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """A diverse set of starting points (all already normalised)."""
    cand = []

    # 1 – uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # 2 – exponential decays
    for r in (0.995, 0.98, 0.96, 0.94, 0.9):
        v = r ** np.arange(n, dtype=np.float64)
        v /= v.sum()
        cand.append(v)

    # 3 – linear decreasing
    v = np.arange(n, 0, -1, dtype=np.float64)
    v /= v.sum()
    cand.append(v)

    # 4 – beta‑family (shape α on [0,1])
    xs = (np.arange(n, dtype=np.float64) + 0.5) / n
    for α in (0.5, 0.8, 1.2, 1.5, 2.0):
        v = (xs ** (α - 1)) * ((1.0 - xs) ** (α - 1))
        v /= v.sum()
        cand.append(v)

    # 5 – a few Dirichlet draws (moderately concentrated)
    for _ in range(5):
        cand.append(rng.dirichlet(np.ones(n) * 0.3))

    # 6 – previous best if it exists
    if "height_sequence_1" in globals():
        prev = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, None)
        cand.append(proj_simplex(prev))

    return cand


# ----------------------------------------------------------------------
# Smoothed max surrogate (soft‑max) utilities
# ----------------------------------------------------------------------


def softmax_weights(conv: np.ndarray, alpha: float) -> np.ndarray:
    """Soft‑max weights that sum to 1, stabilised with the usual shift."""
    c_max = conv.max()
    shifted = conv - c_max
    w = np.exp(alpha * shifted)
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.size)
    return w / s


def softmax_gradient(a: np.ndarray,
                     fft_len_self: int,
                     fft_len_grad: int,
                     alpha: float) -> np.ndarray:
    """
    Gradient of the smoothed objective
        f(a) = (2·n/α)·log Σ exp(α·conv)
    with respect to a; a is assumed to lie on the simplex.
    """
    conv = _self_conv(a, fft_len_self)
    w = softmax_weights(conv, alpha)          # shape 2·n‑1
    w_rev = w[::-1]
    # convolution a * w_rev
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1                     # = 2·n‑2
    grad = 2.0 * full[start:start + a.size]    # factor 2 comes from d(conv)/da
    return grad


# ----------------------------------------------------------------------
# Main optimisation entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """Search for a non‑negative step‑function that minimises ``evaluate_sequence``."""
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time budget (5 s safety margin)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Problem size
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        try:
            n = int(len(globals()["height_sequence_1"]))
            n = max(512, min(n, 4000))            # keep FFT cheap
        except Exception:
            n = 2000
    else:
        n = 2000                                     # default

    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(conv_len + n - 1)       # for a * weights

    # ------------------------------------------------------------------
    # 2) Warm‑start portfolio
    # ------------------------------------------------------------------
    candidates = _initial_candidates(n, rng)

    best_vec = None
    best_score = float("inf")
    for cand in candidates:
        score = compute_J(cand, fft_len_self)
        if score < best_score:
            best_score = score
            best_vec = cand.copy()
    print(f"[init] J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 3) Phase 1 – Simulated annealing on the simplex
    # ------------------------------------------------------------------
    # allocate roughly 30 % of the remaining budget
    remaining = deadline - time.monotonic()
    phase1_end = time.monotonic() + remaining * 0.30

    cur = best_vec.copy()
    cur_score = best_score
    start_sa = time.monotonic()
    # initial temperature proportional to the current objective
    init_temp = max(cur_score * 0.2, 1e-6)

    while time.monotonic() < phase1_end:
        # propose a cheap pairwise mass‑transfer move
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        factor = np.exp(rng.normal(0.0, 0.12))
        new_i = cur[i] * factor
        delta = new_i - cur[i]
        new_j = cur[j] - delta
        if new_j < 0.0:
            continue
        proposal = cur.copy()
        proposal[i] = new_i
        proposal[j] = new_j
        proposal = proj_simplex(proposal)
        prop_score = compute_J(proposal, fft_len_self)

        # linear temperature decay
        elapsed = time.monotonic() - start_sa
        T = max(init_temp * (1.0 - elapsed / (phase1_end - start_sa)), 1e-12)

        if prop_score < cur_score or rng.random() < math.exp(-(prop_score - cur_score) / T):
            cur = proposal
            cur_score = prop_score
            if prop_score < best_score:
                best_score = prop_score
                best_vec = cur.copy()
                now = time.monotonic()
                print(f"[{now - (deadline - budget_s):6.2f}s] SA new best J = {best_score:.6f}")

        # occasional random restart to shake out of deep valleys
        if rng.random() < 0.001:
            cur = rng.dirichlet(np.ones(n))
            cur_score = compute_J(cur, fft_len_self)

    # ------------------------------------------------------------------
    # 4) Phase 2 – Projected Adam on a soft‑max surrogate
    # ------------------------------------------------------------------
    # give the remaining ~70 % of the budget to this phase
    a = best_vec.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    it = 0
    start_adam = time.monotonic()
    last_improve = start_adam

    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        frac = (now - start_adam) / (deadline - start_adam)  # ∈[0,1]

        # learning‑rate decay (aggressive start, gentle end)
        lr = 0.35 * (1.0 - frac) + 0.005 * frac

        # temperature schedule for the soft‑max (slow → sharp)
        alpha = 1.5 + 2500.0 * frac

        # gradient of the smoothed objective
        grad = softmax_gradient(a, fft_len_self, fft_len_grad, alpha)

        # Adam update + simplex projection
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = proj_simplex(a)

        # evaluate true objective
        cur_score = compute_J(a, fft_len_self)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_vec = a.copy()
            last_improve = now
            print(f"[{now - (deadline - budget_s):6.2f}s] Adam new best J = {best_score:.6f}")

        # occasional block perturbation to avoid plateaus
        if rng.random() < 0.01:
            a = _random_block_perturb(a, rng)

        # restart if stagnation > 20 s
        if now - last_improve > 20.0:
            a = best_vec.copy()
            noise = np.exp(rng.normal(0.0, 0.09, size=n))
            a = proj_simplex(a * noise)
            last_improve = now
            print(f"[{now - (deadline - budget_s):6.2f}s] restart – best J = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 5) Return the best candidate (clipped to the limits of evaluate_sequence)
    # ------------------------------------------------------------------
    best_vec = np.clip(best_vec, 0.0, 1000.0)
    return [float(x) for x in best_vec]
