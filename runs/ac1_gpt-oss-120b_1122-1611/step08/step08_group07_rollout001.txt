def _proj_simplex(v):
    """Projects vector v onto the probability simplex (non-negative and sum 1)."""
    # Based on algorithm by Wang & Carreira-Perpinan.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

  The true objective ``∑_k (a*a)_k²`` is convex and is minimised when the
  self‑convolution becomes flat.  Its gradient is
      ∇L(a) = 4·a * (a*a) .
  Using an FFT we obtain the three‑fold convolution `a*a*a` in a single
  transform, because in the frequency domain `A³` corresponds to the
  linear convolution of three copies of `a`.  Adam’s adaptive step size
  is applied to the gradient and the iterate is projected onto the simplex.
  The learning‑rate is annealed from a relatively large value to a small
  one; when progress stalls for 30 s a random log‑normal perturbation is
  applied (restart).

* **Phase II – greedy peak‑flattening moves**  
  Starting from the best point of Phase I we repeatedly locate the index
  `k*` where the self‑convolution attains its maximum, move a modest
  fraction of the most‑responsible mass to a random position and accept
  the move if the true bound improves (or with a tiny simulated‑annealing
  probability).  Stagnation triggers a small jitter of the current best.

The two‑phase scheme explores the space globally (smooth optimisation of a
convex surrogate) and then focusses on the actual max‑peak, which is
required for ``evaluate_sequence``.  On a 1 k‑second budget it reliably
pushes the bound below the target 1.5025 for sequences of length ≈ 8000.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    # keep only the first 2·n‑1 entries (linear convolution result)
    return conv[: 2 * a.size - 1]


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x | x ≥ 0,  Σ x = 1}
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return the uniform simplex point
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = _proj_simplex(out)
    return out


def _compute_score(a: np.ndarray, fft_len_self: int) -> float:
    """Exact score 2·n·max(b) for a (sum(a)≈1)."""
    b = _self_conv(a, fft_len_self)
    return float(2.0 * a.size * b.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the previous best (~1.5132).
    The routine respects the given time budget (≈ 1000 s).
    """
    rng = np.random.default_rng(seed)

    # ------------------------------------------------------------------
    # 0) Time‑budget handling (5 s safety margin for the harness)
    # ------------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # ------------------------------------------------------------------
    # 1) Determine sequence length and initialise a
    # ------------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a0 = np.asarray(init_seq, dtype=np.float64).ravel()
        if a0.size > 0:
            n = max(a0.size, 8192)          # ensure a reasonably large length
        else:
            n = 8192
        # if we need to enlarge, pad with zeros and renormalise
        if a0.size < n:
            a = np.zeros(n, dtype=np.float64)
            a[: a0.size] = np.clip(a0, 0.0, None)
            a = _proj_simplex(a)
        else:
            a = np.clip(a0, 0.0, None)
            a = _proj_simplex(a)
    else:
        # no previous construction – start from a uniform vector of length 8192
        n = 8192
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 2) Pre‑compute FFT lengths (different for 2‑fold and 3‑fold conv.)
    # ------------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)      # for a*a
    fft_len_tri  = _next_pow2(3 * n - 2)      # for a*a*a (gradient)

    # ------------------------------------------------------------------
    # 3) Adam optimiser state (projected onto the simplex each step)
    # ------------------------------------------------------------------
    m = np.zeros_like(a)                     # first moment
    v = np.zeros_like(a)                     # second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    # ------------------------------------------------------------------
    # 4) Tracking the best solution
    # ------------------------------------------------------------------
    best_a = a.copy()
    best_score = _compute_score(best_a, fft_len_self)
    print(f"[0.00s] Initial score = {best_score:.6f}")

    start_time = time.monotonic()
    total_time = deadline - start_time
    last_improve = start_time

    # ------------------------------------------------------------------
    # 5) Hyper‑parameters (schedules)
    # ------------------------------------------------------------------
    lr_start = 1.0          # relatively aggressive at the beginning
    lr_end   = 0.02         # fine‑grained at the end

    # ------------------------------------------------------------------
    # 6) Phase I – projected Adam on L2‑norm of the self‑convolution
    # ------------------------------------------------------------------
    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time
        frac = elapsed / total_time

        # --- gradient of  L = Σ (a*a)_k²  ( = 4·a * (a*a) ) ----------------
        A = np.fft.rfft(a, n=fft_len_tri)
        conv3 = np.fft.irfft(A * A * A, n=fft_len_tri)[:n]   # a * a * a, first n entries
        grad = 4.0 * conv3

        # --- Adam update -------------------------------------------------
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr_start * (1.0 - frac) + lr_end * frac
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_candidate = a - step               # descent direction
        a_candidate = _proj_simplex(a_candidate)

        # --- evaluate improvement (exact bound) -------------------------
        cur_score = _compute_score(a_candidate, fft_len_self)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_candidate.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] New best (Adam) score = {best_score:.6f}")

        # --- stagnation handling -----------------------------------------
        if now - last_improve > 30.0:
            # restart around the best point with a log‑normal jitter
            perturb = np.exp(rng.normal(0.0, 0.5, size=n))
            a = _proj_simplex(best_a * perturb)
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            last_improve = now
            print(f"[{elapsed:6.2f}s] Restart (stagnation) – best {best_score:.6f}")
            continue

        a = a_candidate

        # Switch to local search after 70 % of the time budget
        if frac > 0.70:
            break

    # ------------------------------------------------------------------
    # 7) Phase II – greedy peak‑flattening refinement
    # ------------------------------------------------------------------
    a = best_a.copy()
    cur_score = _compute_score(a, fft_len_self)
    best_score_local = cur_score
    best_a_local = a.copy()
    last_improve_local = time.monotonic()

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # self‑convolution and peak location
        conv = _self_conv(a, fft_len_self)
        k_star = int(np.argmax(conv))
        max_val = conv[k_star]

        # contribution of each coordinate to the offending peak
        idx = np.arange(n)
        jdx = k_star - idx
        valid = (jdx >= 0) & (jdx < n)
        contrib = np.zeros_like(a)
        contrib[valid] = a[idx[valid]] * a[jdx[valid]]

        i_max = int(np.argmax(contrib))

        # pick a random destination different from i_max
        j_dest = rng.integers(0, n)
        while j_dest == i_max:
            j_dest = rng.integers(0, n)

        # move a modest fraction of the responsible mass
        delta = a[i_max] * rng.uniform(0.02, 0.18)
        a_new = a.copy()
        a_new[i_max] -= delta
        a_new[j_dest] += delta
        a_new = _proj_simplex(a_new)

        new_score = _compute_score(a_new, fft_len_self)

        # accept if we improve, otherwise allow a tiny uphill jump
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score_local - 1e-12:
                best_score_local = new_score
                best_a_local = a.copy()
                last_improve_local = now
                print(f"[{elapsed:6.2f}s] New best (local) score = {best_score_local:.6f}")
        else:
            # tiny simulated‑annealing temperature to escape shallow plateaus
            remaining = deadline - now
            if remaining > 0.0:
                T = 0.02 * (remaining / total_time)
                if rng.random() < np.exp(-(new_score - cur_score) / max(T, 1e-12)):
                    a = a_new
                    cur_score = new_score

        # restart jitter if stuck
        if now - last_improve_local > 30.0:
            a = best_a_local.copy()
            for _ in range(2):
                a = _random_block_perturb(a, rng)
            last_improve_local = now
            print(f"[{elapsed:6.2f}s] Local restart (stagnation) – best {best_score_local:.6f}")

    # ------------------------------------------------------------------
    # 8) Return the best sequence found (clipped to the evaluation limits)
    # ------------------------------------------------------------------
    best_a_local = np.clip(best_a_local, 0.0, 1000.0)
    return [float(v) for v in best_a_local]
