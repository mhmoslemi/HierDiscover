def compute_gradient_max(a, kmax):
    n = a.shape[0]
    grad = np.zeros_like(a)
    if kmax < n:
        grad[:kmax+1] = 2 * a[kmax::-1]
    else:
        start = kmax - (n - 1)
        grad[start:] = 2 * a[n-1:start-1:-1]
    return grad

"""
Hybrid optimizer for the step‑function problem.

Algorithm
---------
1. Initialise the sequence (use the supplied `height_sequence_1` if present,
   otherwise a uniform distribution).  All vectors are normalised to sum = 1.

2. **Phase 1 – ADAM on the exact max‑convolution**.  The hard max in the
   objective is handled directly: after each FFT‑based self‑convolution we
   locate the index `k_max` attaining the maximum and compute the sub‑gradient
   `g_i = 2·a_{k_max−i}`.  An ADAM update (with a linearly annealed learning
   rate) is applied and the result is projected onto the probability simplex.
   This rapidly reduces the max‑convolution while keeping the iterates feasible.

3. **Phase 2 – gradient‑guided stochastic refinement**.  Using the current
   `k_max` we build a source set (indices with large gradient, i.e. large
   contribution to the peak) and a target set (indices with small gradient).
   Tiny mass transfers (`δ`) from a source to a target are proposed;
   the exact objective (computed by an FFT) is evaluated and the move is kept
   iff it improves the best score.  Periodic “jitter’’ operations (multiplicative
   noise on a random block) help escape plateaus.

4. The best sequence seen is returned (clipped to the evaluator limits).

The whole procedure respects the 1000 s budget and prints progress.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {x ≥ 0, sum(x) = 1}.
    """
    if v.ndim != 1:
        raise ValueError("Only 1‑D vectors are supported.")
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    w_sum = w.sum()
    if w_sum == 0.0:
        w[:] = 1.0 / len(v)
    else:
        w /= w_sum
    return w


def _conv_max_and_index(a: np.ndarray, fft_len: int) -> tuple[float, int, np.ndarray]:
    """Return (max value, arg‑max index, full convolution) of a with itself."""
    a_fft = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(a_fft * a_fft, n=fft_len)[: 2 * a.size - 1]
    idx = int(conv.argmax())
    return float(conv[idx]), idx, conv


def _gradient_of_max(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the max‑convolution w.r.t. a.
    For each i the derivative of conv[k_max] w.r.t. a_i equals
        2 * a_{k_max-i}   (if the index is in range, else 0).
    """
    n = a.size
    g = np.zeros_like(a)
    if k_max < n:
        # i = 0 … k_max
        g[: k_max + 1] = 2.0 * a[k_max::-1]
    else:
        # k_max ≥ n
        start = k_max - (n - 1)          # first i that contributes
        g[start:] = 2.0 * a[n - 1 : start - 1 : -1]
    return g


def _objective_from_max(max_val: float, n: int) -> float:
    """Exact objective (2·n·max_convolution)."""
    return 2.0 * n * max_val


# ----------------------------------------------------------------------
# Main search routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list[float]:
    """Search for a step‑function (list of heights) that minimises
    ``evaluate_sequence`` under the given time budget."""
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Initialise a feasible sequence (sum = 1)
    # --------------------------------------------------------------
    if "height_sequence_1" in globals():
        a0 = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        a0 = np.clip(a0, 0.0, None)
        s0 = a0.sum()
        if s0 > 0.0:
            a0 /= s0
        else:
            a0 = np.full_like(a0, 1.0 / a0.size)
        a = a0.copy()
    else:
        # default: power‑of‑two length simplifies FFT handling
        n = 4096
        a = np.full(n, 1.0 / n, dtype=np.float64)

    n = a.size
    fft_len = _next_pow2(3 * n)

    # --------------------------------------------------------------
    # 1) Time‑budget handling (a few seconds safety margin)
    # --------------------------------------------------------------
    start_time = time.monotonic()
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping for the best solution seen so far
    # --------------------------------------------------------------
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())
    best_time = time.monotonic()
    print(f"[0.00s] initial score = {best_score:.6f}")

    # --------------------------------------------------------------
    # 3) Phase 1 – ADAM on the exact max‑convolution
    # --------------------------------------------------------------
    # ADAM state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    beta1_pow, beta2_pow = 1.0, 1.0

    # Learning‑rate schedule (linear annealing)
    LR_START, LR_END = 0.20, 5e-04   # a bit more conservative than the previous code

    # Allocate roughly 40 % of the budget to this phase
    phase1_end = start_time + 0.40 * (deadline - start_time)

    iter_adam = 0
    while time.monotonic() < phase1_end:
        # convolution & max‑index
        max_val, k_max, _ = _conv_max_and_index(a, fft_len)

        # sub‑gradient of the exact max
        grad = _gradient_of_max(a, k_max)

        # ADAM bookkeeping
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        beta1_pow *= beta1
        beta2_pow *= beta2
        m_hat = m / (1.0 - beta1_pow)
        v_hat = v / (1.0 - beta2_pow)

        # learning‑rate annealing
        frac = (time.monotonic() - start_time) / (phase1_end - start_time)
        lr = LR_START * (1.0 - frac) + LR_END * frac

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        # projected additive update
        a = a - step
        a = _project_onto_simplex(a)

        iter_adam += 1

        # occasional exact evaluation (every few hundred steps)
        if iter_adam % 200 == 0:
            current_score = _objective_from_max(max_val, n)
            if current_score < best_score - 1e-12:
                best_score = current_score
                best = a.copy()
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Phase 1 new best = {best_score:.6f} (iter={iter_adam})")
                if best_score <= 1.5025:
                    print("[early stop] target reached in Phase 1")
                    break

    # --------------------------------------------------------------
    # 4) Phase 2 – gradient‑guided stochastic refinement
    # --------------------------------------------------------------
    a = best.copy()
    iter_local = 0
    # parameters controlling the source/target sampling
    top_ratio = 0.08          # proportion of indices considered “high‑gradient’’
    low_ratio = 0.08          # proportion considered “low‑gradient’’
    min_move_frac = 1e-3      # minimum fraction of source mass to move
    max_move_frac = 2e-2      # maximum fraction

    while time.monotonic() < deadline:
        # current convolution
        max_val, k_max, conv = _conv_max_and_index(a, fft_len)
        cur_obj = _objective_from_max(max_val, n)

        # gradient for the current peak
        g = _gradient_of_max(a, k_max)

        # define source / target sets
        sorted_idx = np.argsort(g)                     # ascending
        low_k = max(1, int(low_ratio * n))
        high_k = max(1, int(top_ratio * n))
        tgt_set = sorted_idx[:low_k]
        src_set = sorted_idx[-high_k:]

        # pick source and target
        src = rng.choice(src_set)
        if a[src] <= 0.0:
            # nothing to move – skip this iteration
            continue
        tgt = rng.choice(tgt_set)
        if src == tgt:
            continue

        # propose a tiny mass transfer
        frac_move = rng.uniform(min_move_frac, max_move_frac)
        delta = a[src] * frac_move

        a_candidate = a.copy()
        a_candidate[src] -= delta
        a_candidate[tgt] += delta   # preserves sum, stays non‑negative

        # evaluate candidate (exact objective)
        cand_max, _, _ = _conv_max_and_index(a_candidate, fft_len)
        cand_obj = _objective_from_max(cand_max, n)

        iter_local += 1

        if cand_obj < best_score - 1e-12:
            best_score = cand_obj
            best = a_candidate.copy()
            a = a_candidate
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Phase 2 new best = {best_score:.6f} (move #{iter_local})")
            if best_score <= 1.5025:
                print("[early stop] target reached in Phase 2")
                break

        # occasional jitter to shake the system out of plateaus
        if iter_local % 5000 == 0:
            width = rng.integers(2, max(3, n // 100))
            start = rng.integers(0, n - width + 1)
            factor = np.exp(rng.normal(0.0, 0.012))
            a_jitter = a.copy()
            a_jitter[start:start + width] *= factor
            a_jitter = _project_onto_simplex(a_jitter)
            # accept jitter only if it improves
            jitter_max, _, _ = _conv_max_and_index(a_jitter, fft_len)
            jitter_obj = _objective_from_max(jitter_max, n)
            if jitter_obj < best_score - 1e-12:
                best_score = jitter_obj
                best = a_jitter.copy()
                a = a_jitter
                elapsed = time.monotonic() - start_time
                print(f"[{elapsed:6.2f}s] Phase 2 jitter improvement = {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence discovered (clip to evaluator limits)
    # --------------------------------------------------------------
    best = np.clip(best, 0.0, 1000.0)
    final_score = evaluate_sequence(best.tolist())
    total_elapsed = time.monotonic() - start_time
    print(f"Finished in {total_elapsed:.2f}s – final score = {final_score:.6f}")
    return [float(x) for x in best]
