                print(f"[{time.monotonic()-start:.2f}s] new best β={α},{β} n={n} score {sc:.6f}")
            top_candidates.append((sc, a.copy()))
        # exponential (decreasing)
        for r in [0.2, 0.4, 0.6, 0.8, 0.9]:
            a = gen_exponential(n, r, reverse=False)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc; best_seq = a.copy()
                print(f"[{time.monotonic()-start:.2f}s] new best exp r={r} n={n} score {sc:.6f}")
            top_candidates.append((sc, a.copy()))
            # reversed
            a = gen_exponential(n, r, reverse=True)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc; best_seq = a.copy()
                print(f"[{time.monotonic()-start:.2f}s] new best rev exp r={r} n={n} score {sc:.6f}")
            top_candidates.append((sc, a.copy()))
        # triangular
        a_tri = gen_triangular(n)
        sc_tri = compute_score(a_tri)
        if sc_tri < best_score:
            best_score = sc_tri; best_seq = a_tri.copy()
            print(f"[{time.monotonic()-start:.2f}s] new best triangular n={n} score {sc_tri:.6f}")
        top_candidates.append((sc_tri, a_tri.copy()))
        # quadratic
        a_quad = gen_quadratic(n)
        sc_quad = compute_score(a_quad)
        if sc_quad < best_score:
            best_score = sc_quad; best_seq = a_quad.copy()
            print(f"[{time.monotonic()-start:.2f}s] new best quadratic n={n} score {sc_quad:.6f}")
        top_candidates.append((sc_quad, a_quad.copy()))
        # mixed exponential random draws
        for _ in range(10):
            if time.monotonic() >= explore_deadline:
                break
            r = rng.uniform(0.2, 0.9)
            lam = rng.random()
            a = gen_mixed_exponential(n, r, lam)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc; best_seq = a.copy()
                print(f"[{time.monotonic()-start:.2f}s] new best mix r={r:.3f} lam={lam:.3f} n={n} score {sc:.6f}")
            top_candidates.append((sc, a.copy()))

    # keep only top K candidates for refinement.
    K = 5
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:K]

    # 3. Refinement stage: targeted gradient descent on each candidate.
    for idx, (init_score, init_seq) in enumerate(top_candidates):
        if time.monotonic() >= deadline:
            break
        # allocate a slice of remaining time
        remaining = deadline - time.monotonic()
        # each gets at most remaining/(K-idx) to avoid starvation
        sub_deadline = time.monotonic() + remaining / (len(top_candidates)-idx)
        print(f"Refinement #{idx+1}/{len(top_candidates)} start (score {init_score:.6f})")
        refined_seq, refined_score = gradient_descent(init_seq, sub_deadline, rng)
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined_seq.copy()
            print(f"--> improvement from refinement #{idx+1}: new best {best_score:.6f}")

    # final safety normalization
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / best_seq.size
    else:
        best_seq /= total
    elapsed = time.monotonic() - start
    print(f"Finished after {elapsed:.2f}s, best score {best_score:.6f}, length {best_seq.size}")
    return [float(v) for v in best_seq]

        elapsed = time.monotonic() - start

"""
Simple yet aggressive optimizer for the self‑convolution bound.

The algorithm works in two stages:

1. **Exploration** – a modest sweep over several parametric families
   (Beta, exponential, triangular, quadratic and mixtures).  The best
   few candidates (by the exact objective `compute_score`) are kept.

2. **Refinement** – each kept candidate is fed to a targeted gradient‑
   descent: the current maximal convolution entry is identified,
   its sub‑gradient (a shifted copy of the vector) is used for a
   multiplicative mirror‑descent step, the vector is renormalised onto
   the simplex and the process repeats until the time budget expires.
   Occasional jitter and stagnation restarts help to escape plateaus.

All operations respect the non‑negativity and sum‑normalisation required
by `evaluate_sequence`.  The routine stops a few seconds before the
budget to guarantee a return value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to ``x``."""
    return 1 << ((int(x) - 1).bit_length())


def _max_self_convolution(a: np.ndarray) -> float:
    """Maximum of the linear self‑convolution a*a using FFT."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return float(conv.max())


def compute_score(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` (without its clamping).
    The vector is assumed to be non‑negative; the function works for any
    sum – the denominator is handled internally.
    """
    n = a.size
    max_b = _max_self_convolution(a)
    s = a.sum()
    if s < 0.01:                     # matches evaluate_sequence protection
        return float("inf")
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# Parametric families
# ----------------------------------------------------------------------


def gen_beta_seq(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) on {0,…,n‑1}; normalised to sum = 1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    eps = 1e-12
    x = np.clip(x, eps, 1.0 - eps)
    raw = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    raw = np.clip(raw, 0.0, None)
    s = raw.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return raw / s


def gen_exponential(n: int, r: float, reverse: bool = False) -> np.ndarray:
    """Geometric shape a_i ∝ r**i (or reversed) and normalised."""
    if r <= 0.0 or r >= 1.0:
        r = 0.5
    i = np.arange(n, dtype=np.float64)
    if reverse:
        vals = np.power(r, n - 1 - i)
    else:
        vals = np.power(r, i)
    s = vals.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return vals / s


def gen_triangular(n: int) -> np.ndarray:
    """Symmetric triangle a_i ∝ min(i+1, n‑i)."""
    a = np.minimum(np.arange(1, n + 1), np.arange(n, 0, -1)).astype(np.float64)
    return a / a.sum()


def gen_quadratic(n: int) -> np.ndarray:
    """Parabolic shape a_i ∝ (i+1)*(n‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1.0) * (n - i)
    return a / a.sum()


def gen_mixed_exponential(n: int, r: float, lam: float) -> np.ndarray:
    """Mixture λ·geom‑decrease + (1‑λ)·geom‑increase, normalised."""
    i = np.arange(n, dtype=np.float64)
    dec = np.power(r, i)
    inc = np.power(r, n - 1 - i)
    a = lam * dec + (1.0 - lam) * inc
    s = a.sum()
    if s == 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return a / s


# ----------------------------------------------------------------------
# Convolution utilities used by the gradient descent
# ----------------------------------------------------------------------


def compute_full_conv(a: np.ndarray) -> np.ndarray:
    """Full linear self‑convolution (length 2 n‑1)."""
    n = a.size
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, n=L)
    conv = np.fft.irfft(fa * fa, n=L)[: 2 * n - 1]
    return conv


def compute_gradient_maxconv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    For the index k_max we have   ∂(a*a)_k/∂a_i = 2·a_{k-i}  (if in range).
    """
    n = a.size
    grad = np.zeros_like(a)
    lo = max(0, k_max - (n - 1))
    hi = min(n - 1, k_max)
    idx = np.arange(lo, hi + 1)
    partner = k_max - idx
    grad[idx] = 2.0 * a[partner]
    return grad


# ----------------------------------------------------------------------
# Gradient‑descent refinement
# ----------------------------------------------------------------------


def gradient_descent(a_init: np.ndarray,
                    deadline: float,
                    rng: np.random.Generator) -> tuple[np.ndarray, float]:
    """
    Targeted mirror‑descent on the L∞‑norm of the self‑convolution.
    Returns the best vector seen and its score.
    """
    a = a_init.copy()
    n = a.size
    best = a.copy()
    best_score = compute_score(best)

    start = time.monotonic()
    last_improve = start
    eta0 = 0.5               # initial step size (multiplicative)

    while time.monotonic() < deadline:
        conv = compute_full_conv(a)
        k_max = int(conv.argmax())
        # gradient of the active convolution entry
        grad = compute_gradient_maxconv(a, k_max)

        # linear schedule for the step size
        elapsed = time.monotonic() - start
        total = deadline - start
        eta = eta0 * max(0.0, 1.0 - elapsed / total)
        if eta < 1e-4:
            eta = 1e-4

        # multiplicative mirror‑descent step and re‑normalisation
        a_new = a * np.exp(-eta * grad)
        s = a_new.sum()
        if s == 0.0:
            a_new = np.full_like(a, 1.0 / n)
        else:
            a_new /= s

        new_score = compute_score(a_new)
        if new_score < best_score:
            best_score = new_score
            best = a_new.copy()
            last_improve = time.monotonic()
            print(f"    [{time.monotonic() - start:.2f}s] gradient improvement → {best_score:.6f}")

        # occasional jitter to escape shallow plateaus
        if rng.random() < 0.02:
            jitter = np.exp(rng.normal(0.0, 0.07, size=n))
            a_jit = a_new * jitter
            a_jit /= a_jit.sum()
            jitter_score = compute_score(a_jit)
            if jitter_score < best_score:
                best_score = jitter_score
                best = a_jit.copy()
                last_improve = time.monotonic()
                print(f"    [{time.monotonic() - start:.2f}s] jitter improvement → {best_score:.6f}")
                a_new = a_jit

        # restart after prolonged stagnation
        if time.monotonic() - last_improve > 5.0:
            noise = np.exp(rng.normal(0.0, 0.12, size=n))
            a_restart = best * noise
            a_restart /= a_restart.sum()
            restart_score = compute_score(a_restart)
            if restart_score < best_score:
                best_score = restart_score
                best = a_restart.copy()
                print(f"    [{time.monotonic() - start:.2f}s] stagnation restart → {best_score:.6f}")
            a_new = a_restart
            last_improve = time.monotonic()

        a = a_new

        # early exit if we already beat the target bound
        if best_score < 1.5025:
            break

    return best, best_score

# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (approximately) minimises
    ``evaluate_sequence`` within the given time budget.
    """
    rng = np.random.default_rng(seed)

    start_time = time.monotonic()
    # leave a small safety margin
    deadline = start_time + max(0.0, float(budget_s) - 5.0)

    best_seq = None
    best_score = float("inf")
    top_candidates: list[tuple[float, np.ndarray]] = []

    # ------------------------------------------------------------------
    # 0) Seed from a previously discovered construction, if available
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        cand = np.clip(cand, 0.0, None)
        s = cand.sum()
        if s > 0.0:
            cand /= s
        else:
            cand = np.full_like(cand, 1.0 / cand.size)
        sc = compute_score(cand)
        best_seq = cand.copy()
        best_score = sc
        top_candidates.append((sc, cand.copy()))
        print(f"[0.00s] start from known sequence → score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 1) Exploration of a variety of parametric families (≈30 % budget)
    # ------------------------------------------------------------------
    explore_deadline = start_time + 0.30 * (deadline - start_time)

    n_choices = [512, 1024, 2048, 4096, 8192]

    # special Beta parameters known to be promising
    special_beta = [
        (2.0, 1.0),   # linear ramp (inc)
        (1.0, 2.0),   # linear ramp (dec)
        (0.5, 0.5),   # arcsine U‑shape
        (1.5, 1.5),   # mild U‑shape
        (0.8, 1.2),
        (1.2, 0.8),
    ]

    for n in n_choices:
        if time.monotonic() >= explore_deadline:
            break

        # Uniform baseline
        a_uni = np.full(n, 1.0 / n, dtype=np.float64)
        sc_uni = compute_score(a_uni)
        if sc_uni < best_score:
            best_score, best_seq = sc_uni, a_uni.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] new best (uniform, n={n}) → {best_score:.6f}")
        top_candidates.append((sc_uni, a_uni.copy()))

        # Beta families (special + random)
        for α, β in special_beta:
            a = gen_beta_seq(n, α, β)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] new best (β={α:.2f},{β:.2f}, n={n}) → {best_score:.6f}")
            top_candidates.append((sc, a.copy()))

        # Random Beta draws
        for _ in range(12):
            if time.monotonic() >= explore_deadline:
                break
            log_a = rng.uniform(np.log(0.2), np.log(5.0))
            log_b = rng.uniform(np.log(0.2), np.log(5.0))
            α, β = np.exp(log_a), np.exp(log_b)
            a = gen_beta_seq(n, α, β)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] new best (rand β={α:.3f},{β:.3f}, n={n}) → {best_score:.6f}")
            top_candidates.append((sc, a.copy()))

        # Exponential families
        for r in (0.2, 0.4, 0.6, 0.8, 0.9):
            a = gen_exponential(n, r, reverse=False)
            sc = compute_score(a)
            if sc < best_score:
                best_score, best_seq = sc, a.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] new best (exp r={r:.2f}, n={n}) → {best_score:.6f}")
            top_candidates.append((sc, a.copy()))

            a_rev = gen_exponential(n, r, reverse=True)
            sc = compute_score(a_rev)
            if sc < best_score:
                best_score, best_seq = sc, a_rev.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] new best (rev‑exp r={r:.2f}, n={n}) → {best_score:.6f}")
            top_candidates.append((sc, a_rev.copy()))

        # Triangular shape
        a_tri = gen_triangular(n)
        sc_tri = compute_score(a_tri)
        if sc_tri < best_score:
            best_score, best_seq = sc_tri, a_tri.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] new best (triangular, n={n}) → {best_score:.6f}")
        top_candidates.append((sc_tri, a_tri.copy()))

        # Quadratic shape
        a_quad = gen_quadratic(n)
        sc_quad = compute_score(a_quad)
        if sc_quad < best_score:
            best_score, best_seq = sc_quad, a_quad.copy()
            print(f"[{time.monotonic() - start_time:.2f}s] new best (quadratic, n={n}) → {best_score:.6f}")
        top_candidates.append((sc_quad, a_quad.copy()))

        # Mixed exponential families (randomised)
        for _ in range(6):
            if time.monotonic() >= explore_deadline:
                break
            r = rng.uniform(0.2, 0.9)
            lam = rng.random()
            a_mix = gen_mixed_exponential(n, r, lam)
            sc = compute_score(a_mix)
            if sc < best_score:
                best_score, best_seq = sc, a_mix.copy()
                print(f"[{time.monotonic() - start_time:.2f}s] new best (mix r={r:.3f}, λ={lam:.3f}, n={n}) → {best_score:.6f}")
            top_candidates.append((sc, a_mix.copy()))

    # ------------------------------------------------------------------
    # 2) Keep only a few top candidates for the refinement phase
    # ------------------------------------------------------------------
    K = 5
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:K]

    # ------------------------------------------------------------------
    # 3) Gradient‑descent refinement of the selected candidates
    # ------------------------------------------------------------------
    for idx, (init_score, init_seq) in enumerate(top_candidates):
        if time.monotonic() >= deadline:
            break
        # split the remaining budget roughly equally among the pending runs
        remaining_total = deadline - time.monotonic()
        runs_left = len(top_candidates) - idx
        sub_deadline = time.monotonic() + remaining_total / runs_left
        print(f"Refinement #{idx + 1}/{len(top_candidates)} (init score {init_score:.6f})")
        refined_seq, refined_score = gradient_descent(init_seq, sub_deadline, rng)
        if refined_score < best_score:
            best_score, best_seq = refined_score, refined_seq.copy()
            print(f"   improvement → new best score {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4) Final safety normalisation and return
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – a modest uniform sequence
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total == 0.0:
        best_seq[:] = 1.0 / best_seq.size
    else:
        best_seq /= total

    elapsed = time.monotonic() - start_time
    print(f"Finished after {elapsed:.2f}s – best score {best_score:.6f}, length {best_seq.size}")

    # Return a plain Python list as required by the evaluation harness
    return [float(v) for v in best_seq]
