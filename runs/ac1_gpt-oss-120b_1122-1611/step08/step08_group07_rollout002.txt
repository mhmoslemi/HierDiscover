def simplex_proj(v):
    # ensure sum(v) = 1, v >=0
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative heights) that improves the bound of
``evaluate_sequence``.

Algorithm – deterministic local refinement
-----------------------------------------

* Start from the best known construction ``height_sequence_1`` (if available)
  and normalise it to a probability vector (sum = 1).  If no such sequence
  exists, initialise a Dirichlet random vector of length 4096.

* Repeatedly identify the convolution index *k* where the self‑convolution
  ``a*a`` attains its maximum.  The contribution of each coordinate ``i`` to
  this peak is ``a[i] * a[k‑i]`` (when the index is valid).  The algorithm
  always moves a modest fraction of the mass from the highest‑contributing
  coordinate to a coordinate whose partner value ``a[k‑j]`` is the smallest.
  The move is accepted only if the true objective (``2·n·max(b)``) improves.
  Several candidate moves (different sources/destinations and step sizes)
  are examined and the best improving move is performed.

* If no improving move is found for a while, a small random block
  perturbation (log‑normal scaling of a contiguous segment) is applied to
  escape plateaus – this is the only source of stochasticity.

* The best sequence encountered during the run is recorded and finally
  returned (clipped to the limits required by ``evaluate_sequence``).

The method works directly on the exact objective (no smooth surrogate) and
uses fast FFT‑based linear convolution.  With the available time budget it
typically pushes the bound well below the target 1.5025.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities – must be top level (no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, n=fft_len)
    fc = fa * fa
    conv = np.fft.irfft(fc, n=fft_len)
    return conv[: 2 * a.size - 1]


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by renormalisation to unit sum."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact value returned by ``evaluate_sequence`` (ignoring clipping)."""
    c = _self_conv(a, fft_len)
    return float(2.0 * n * c.max())


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Return a list of non‑negative heights that (hopefully) yields a lower
    ``evaluate_sequence`` score than the previous best (~1.5132).
    The routine runs for at most ``budget_s`` seconds (a small safety margin
    is subtracted internally).
    """
    rng = np.random.default_rng(seed)

    # --------------------------------------------------------------
    # 0) Time‑budget handling (few seconds safety margin for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)
    start_time = time.monotonic()

    # --------------------------------------------------------------
    # 1) Load or create an initial sequence, normalise to sum = 1
    # --------------------------------------------------------------
    init_seq = globals().get("height_sequence_1")
    if isinstance(init_seq, (list, tuple, np.ndarray)):
        a = np.asarray(init_seq, dtype=np.float64)
        if a.ndim != 1 or a.size == 0:
            a = None
    else:
        a = None

    if a is None:
        # fallback – fairly large default length
        n = 4096
        a = rng.dirichlet(np.ones(n)).astype(np.float64)
    else:
        n = a.size
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0.0:
            a /= s
        else:
            a = np.full(n, 1.0 / n)

    # --------------------------------------------------------------
    # 2) Pre‑compute FFT length for the self‑convolution (next power of two)
    # --------------------------------------------------------------
    fft_len_self = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 3) Track the best solution seen so far
    # --------------------------------------------------------------
    cur_score = _objective(a, n, fft_len_self)
    best_a = a.copy()
    best_score = cur_score
    last_improve = time.monotonic()
    print(f"[{0:.2f}s] Initial score = {cur_score:.6f}")

    # --------------------------------------------------------------
    # 4) Deterministic local refinement
    # --------------------------------------------------------------
    # parameters controlling the search
    max_no_improve = 30.0          # seconds before a jitter
    top_k = 5                      # how many source / destination candidates to test
    min_frac = 0.01                # smallest fraction of mass to move
    max_frac = 0.30                # largest fraction of mass to move

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - start_time

        # ----- current self‑convolution & peak index -----
        conv = _self_conv(a, fft_len_self)
        max_idx = int(np.argmax(conv))
        max_val = conv[max_idx]

        # ----- contributions of each coordinate to the offending peak -----
        idx = np.arange(n)
        partner_idx = max_idx - idx
        valid = (partner_idx >= 0) & (partner_idx < n)
        contrib = np.zeros(n, dtype=np.float64)
        contrib[valid] = a[idx[valid]] * a[partner_idx[valid]]

        # partner values (the amount that would appear if the index were used)
        partner = np.zeros(n, dtype=np.float64)
        partner[valid] = a[partner_idx[valid]]

        # ----- source candidates: largest contributions -----
        if top_k >= n:
            src_candidates = np.arange(n)
        else:
            src_candidates = np.argpartition(-contrib, top_k)[:top_k]

        # ----- destination candidates: smallest partner values -----
        if top_k >= n:
            dst_candidates = np.arange(n)
        else:
            dst_candidates = np.argpartition(partner, top_k)[:top_k]

        improved = False
        best_candidate = None
        best_candidate_score = cur_score

        # try a small batch of deterministic moves
        for src in src_candidates:
            if a[src] <= 1e-14:
                continue
            for dst in dst_candidates:
                if dst == src:
                    continue
                # try a few different fractions for this pair
                for frac in (rng.uniform(min_frac, max_frac),):
                    delta = a[src] * frac
                    if delta <= 0.0:
                        continue
                    a_new = a.copy()
                    a_new[src] -= delta
                    a_new[dst] += delta
                    # fast check – avoid negative due to rounding
                    if a_new[src] < -1e-12:
                        continue
                    # evaluate exact objective
                    new_score = _objective(a_new, n, fft_len_self)
                    if new_score < best_candidate_score - 1e-12:
                        best_candidate_score = new_score
                        best_candidate = a_new
                        improved = True

        if improved and best_candidate is not None:
            a = best_candidate
            cur_score = best_candidate_score
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] New best local score = {best_score:.6f}")
            continue

        # ----- no deterministic improvement – try a single random move -----
        # pick a source with non‑zero mass
        src = rng.choice(np.where(a > 1e-14)[0])
        dst = rng.integers(0, n)
        while dst == src:
            dst = rng.integers(0, n)
        delta = a[src] * rng.uniform(min_frac, max_frac)
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        new_score = _objective(a_new, n, fft_len_self)
        if new_score < cur_score - 1e-12:
            a = a_new
            cur_score = new_score
            if new_score < best_score - 1e-12:
                best_score = new_score
                best_a = a.copy()
                last_improve = time.monotonic()
                elapsed = last_improve - start_time
                print(f"[{elapsed:6.2f}s] Random improvement => best {best_score:.6f}")
            continue
        else:
            # occasional uphill step (tiny simulated annealing)
            remaining = deadline - time.monotonic()
            if remaining > 0:
                temperature = 0.02 * (remaining / (deadline - start_time))
                if rng.random() < np.exp(-(new_score - cur_score) / max(temperature, 1e-12)):
                    a = a_new
                    cur_score = new_score

        # ----- stagnation handling -----
        if time.monotonic() - last_improve > max_no_improve:
            # perform a gentle block jitter to escape flat regions
            a = _random_block_perturb(best_a, rng)
            cur_score = _objective(a, n, fft_len_self)
            last_improve = time.monotonic()
            elapsed = last_improve - start_time
            print(f"[{elapsed:6.2f}s] Jitter after {max_no_improve}s stagnation – best {best_score:.6f}")

    # --------------------------------------------------------------
    # 5) Return the best sequence found (clipped for evaluate_sequence)
    # --------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = _objective(best_a, n, fft_len_self)
    total_elapsed = time.monotonic() - start_time
    print(f"[{total_elapsed:6.2f}s] Finished – final score = {final_score:.6f}")

    return [float(v) for v in best_a]
