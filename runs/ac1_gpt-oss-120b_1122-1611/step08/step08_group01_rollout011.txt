    M = c.max()
    # consider values >= M - eps
    eps = 1e-12
    idx = np.where(c >= M - eps)[0]
    return M, idx

def _build_A_active(a, active_idx):
    n = a.size
    rows = active_idx.size
    if rows == 0:
        return np.empty((0, n))
    A = np.zeros((rows, n), dtype=np.float64)
    for r, k in enumerate(active_idx):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        A[r, lo:hi+1] = a[lo:hi+1]
    return A

def _solve_lp_for_g(a, M, active_idx):
    A = _build_A_active(a, active_idx)
    if A.shape[0] == 0:
        # no constraints – any g is allowed; return uniform
        n = a.size
        return np.full(n, 1.0 / n)
    b = np.full(A.shape[0], M)
    # maximize sum(g) => minimize -sum(g)
    c = -np.ones(a.shape, dtype=np.float64)
    # bounds >=0
    bounds = [(0, None)] * a.size
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback to uniform
        n = a.size
        return np.full(n, 1.0 / n)
    g = res.x
    # avoid all zeros
    if np.all(g == 0):

        return np.full(n, 1.0 / n)
    # normalize to unit sum
    s = g.sum()
    if s <= 0:
        n = a.size
        return np.full(n, 1.0 / n)
    return g / s

"""
Propose a step‑function (non‑negative heights on the simplex) that beats the
previous bound.

Algorithm
---------
1. Initialise a probability vector `a` (sum = 1) – either from the global
   `height_sequence_1` or from a uniform start.

2. **LP‑based improvement** (≈ 70 % of the time budget)  
   * Compute the self‑convolution `c = a*a` and its maximum `M`.  
   * Build an *active set* consisting of the indices where `c` attains `M`.  
   * Solve a tiny linear programme  

        maximise   Σ g_i  
        subject to (a * g)_k ≤ M  for every k in the active set,  g_i ≥ 0  

     The LP is tiny because the active set usually contains only a few
     indices.  The solution `g` is normalised to unit sum.

   * Update the current distribution by a convex combination  
        a ← (1‑t)·a + t·g` with a step size `t` that linearly decays from
        0.30 to 0.02 throughout the LP phase.

   * Occasionally jitter `a` (random block scaling) and restart from the
     best point if stagnation lasts longer than 30 s.

3. **Hill‑climbing refinement** (remaining budget)  
   * Repeatedly move a small fraction of mass from a random source index
     to a random destination index.  Keep the move only if the exact
     `evaluate_sequence` value improves.  
   * Every few iterations perform a larger block‑wise perturbation.  
   * If no improvement is seen for > 40 s, re‑initialise around the best
     point with Gaussian noise.

4. Trim trailing near‑zero entries, clip to the limits imposed by
   `evaluate_sequence`, and return the best sequence found.

The LP is solved with SciPy’s HiGHS implementation (`scipy.optimize.linprog`);
the active‑set formulation keeps the LP tiny and fast, allowing thousands of
iterations inside the 1000 s budget.

import time
import numpy as np
from scipy.optimize import linprog

def _clip_sequence(arr: np.ndarray) -> np.ndarray:
    """Clip values to the range accepted by ``evaluate_sequence``."""
    return np.clip(arr, 0.0, 1000.0)


def _convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution of a 1‑D array."""
    return np.convolve(a, a)


def _active_indices(conv: np.ndarray, tol: float = 1e-12) -> np.ndarray:
    """Return the indices where ``conv`` attains its maximum (up to ``tol``)."""
    m = conv.max()
    return np.where(conv >= m - tol)[0]


def _build_active_matrix(a: np.ndarray, act: np.ndarray) -> np.ndarray:
    """
    Build the matrix *A* for the active constraints:
        (a * g)[k] = a_slice @ g   for each k in ``act``.
    Row ``r`` of *A* contains the slice ``a[lo:hi+1]`` placed at the
    appropriate columns, the rest being zero.
    """
    n = a.size
    rows = act.size
    if rows == 0:
        return np.empty((0, n), dtype=np.float64)

    A = np.zeros((rows, n), dtype=np.float64)
    for r, k in enumerate(act):
        lo = max(0, k - (n - 1))
        hi = min(n - 1, k)
        A[r, lo : hi + 1] = a[lo : hi + 1]
    return A


def _solve_lp_direction(a: np.ndarray, M: float, act: np.ndarray,
                       rng: np.random.Generator) -> np.ndarray:
    """
    Solve the LP:

        maximise   Σ g_i
        subject to A @ g ≤ M·1 (active rows only),   g ≥ 0

    where ``A`` encodes the convolution constraints for the active indices.
    Returns a *unit‑sum* vector ``g`` (or the uniform vector if the LP fails).
    """
    A = _build_active_matrix(a, act)
    if A.shape[0] == 0:
        # no constraints – any distribution works → uniform
        n = a.size
        return np.full(n, 1.0 / n, dtype=np.float64)

    b = np.full(A.shape[0], M, dtype=np.float64)
    c = -np.ones(a.shape, dtype=np.float64)          # maximise Σg  ⇔ minimise –Σg
    bounds = [(0.0, None)] * a.size

    # HiGHS is very fast for the tiny problems we create.
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success or np.all(res.x == 0):
        # fallback – uniform distribution
        n = a.size
        return np.full(n, 1.0 / n, dtype=np.float64)

    g = res.x
    s = g.sum()
    if s <= 0.0:
        n = a.size
        return np.full(n, 1.0 / n, dtype=np.float64)
    return g / s


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator,
                         width_frac: float = 0.025,
                         sigma: float = 0.15) -> np.ndarray:
    """
    Multiply a random contiguous block by a log‑normal factor and re‑normalise.
    """
    n = a.size
    width = max(1, int(width_frac * n))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, sigma))
    out = a.copy()
    out[start : start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s > 0.0:
        out /= s
    else:
        out[:] = 1.0 / n
    return out

            a = np.full_like(a, 1.0 / a.size, dtype=np.float64)
    else:
        # fallback: a moderate‑size uniform vector
        n0 = 1024
        a = np.full(n0, 1.0 / n0, dtype=np.float64)

    n = a.size

    # --------------------------------------------------------------
    # 1) Time‑budget handling (reserve a few seconds for the harness)
    # --------------------------------------------------------------
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)

    # --------------------------------------------------------------
    # 2) Book‑keeping of the best solution
    # --------------------------------------------------------------
    conv = _convolution(a)
    M = conv.max()
    best_score = 2.0 * n * M
    best = a.copy()
    last_improve = time.monotonic()
    start_time = time.monotonic()
    total_budget = deadline - start_time

    # --------------------------------------------------------------
    # 3) Helper for the convex‑combination step size (decreases linearly)
    # --------------------------------------------------------------
    def _step_size(elapsed: float) -> float:
        """Linear decay from 0.30 → 0.02 over the whole LP phase."""
        frac = elapsed / total_budget
        return 0.30 * (1.0 - frac) + 0.02

    # --------------------------------------------------------------
    # 4) Phase I – LP‑based updates (≈ 70 % of the budget)
    # --------------------------------------------------------------
    phase1_end = start_time + 0.70 * total_budget

    while time.monotonic() < phase1_end:
        now = time.monotonic()
        elapsed = now - start_time

        # current convolution and active indices
        conv = _convolution(a)
        M = conv.max()
        active = _active_indices(conv)

        # LP direction
        g = _solve_lp_direction(a, M, active, rng)

        # convex combination
        t = _step_size(elapsed)
        a = (1.0 - t) * a + t * g
        # re‑normalise (numerical drift)
        s = a.sum()
        if s > 0.0:
            a /= s

        # evaluate
        conv = _convolution(a)
        score = 2.0 * n * conv.max()
        if score < best_score - 1e-12:
            best_score = score
            best = a.copy()
            last_improve = now
            print(f"[{elapsed:6.2f}s] LP‑phase new best = {best_score:.6f}")

        # occasional stochastic jitter to diversify the search space
        if rng.random() < 0.03:
            a = _random_block_perturb(a, rng, width_frac=1 / max(2, n // 12), sigma=0.15)

        # restart if we have been idle for a while
        if now - last_improve > 30.0:
            noise = rng.normal(0.0, 0.4, size=n)
            a = np.clip(best + noise * best, 0.0, None)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] LP‑restart (stagnation)")

    # --------------------------------------------------------------
    # 5) Phase II – simple hill‑climbing / simulated annealing
    # --------------------------------------------------------------
    a = best.copy()
    cur_score = best_score
    last_improve = time.monotonic()
    phase2_start = time.monotonic()
    phase2_budget = deadline - phase2_start

    def _step_fraction(elapsed: float) -> float:
        """Decay from 0.25 → 0.01 during the hill‑climb."""
        return 0.25 * (1.0 - elapsed / phase2_budget) + 0.01

    while time.monotonic() < deadline:
        now = time.monotonic()
        elapsed = now - phase2_start
        sf = _step_fraction(elapsed)

        # ---- small mass transfer -------------------------------------------------
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i != j and a[i] > 0.0:
            delta = sf * rng.random() * a[i]
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            s = cand.sum()
            if s > 0.0:
                cand /= s
            else:
                cand = np.full(n, 1.0 / n, dtype=np.float64)

            cand_score = 2.0 * n * np.convolve(cand, cand).max()
            if cand_score < cur_score - 1e-12:
                a = cand
                cur_score = cand_score
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best = a.copy()
                    print(f"[{now - start_time:6.2f}s] Hill‑climb new best = {best_score:.6f}")

        # ---- occasional larger block perturbation ---------------------------------
        if rng.random() < 0.02:
            a = _random_block_perturb(a, rng, width_frac=1 / max(2, n // 10), sigma=0.20)

        # ---- stagnation handling --------------------------------------------------
        if now - last_improve > 40.0:
            a = np.clip(best + rng.normal(0.0, 0.3, size=n), 0.0, None)
            s = a.sum()
            if s > 0.0:
                a /= s
            else:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            last_improve = now
            print(f"[{now - start_time:6.2f}s] Hill‑climb restart")

    # --------------------------------------------------------------
    # 6) Trim trailing near‑zero entries and final clipping
    # --------------------------------------------------------------
    eps_trim = 1e-12
    nz = np.where(best > eps_trim)[0]
    if nz.size == 0:
        final = np.array([1.0], dtype=np.float64)
    else:
        final = best[: nz[-1] + 1]

    final = _clip_sequence(final)

    # Return a plain Python list as required by the harness
    return [float(v) for v in final]
