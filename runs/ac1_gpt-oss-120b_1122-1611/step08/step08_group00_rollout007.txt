def compute_smooth_gradient(x, alpha):
    a = expand_bins(x)
    # compute conv
    fa = np.fft.rfft(a, n=fft_len_self)
    conv = np.fft.irfft(fa * fa, n=fft_len_self)[:conv_len]
    res = conv - conv.max()
    w = np.exp(alpha * res)
    w_sum = w.sum()
    if w_sum == 0: w = np.full_like(w, 1.0/w.size)
    else: w /= w_sum
    w_rev = w[::-1]
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    fa = np.fft.rfft(a, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1
    grad_a = 2.0 * full[start:start + n]
    # aggregate into bins:
    grad_x = np.add.reduceat(grad_a, np.arange(0, n, bin_size)) / bin_size  # average per bin
    return grad_x, a
```

def compute_grad_softmax(a, alpha, fft_len_self, fft_len_grad):

    conv = linear_self_conv(a, fft_len_self)

"""
Multi‑scale projected Adam optimizer for the step‑function attack.

The search works on a hierarchy of resolutions (bin sizes).  At a coarse
scale we optimise only a few “bin‑weights’’ (the vector is projected onto
the probability simplex).  After a promising coarse shape is found the
solution is up‑sampled and refined on a finer grid, finally reaching the
full resolution.  Each optimisation phase uses a smooth soft‑max
approximation of the max‑convolution and a projected Adam update.  Random
restarts, occasional jitter and a stagnation‑restart strategy keep the
search from getting trapped.  The routine respects a 1000 s budget and
returns the best non‑negative sequence it discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (all top‑level, no closures)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _self_conv(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear self‑convolution a*a using an FFT of length ``fft_len``.
    Returns the full (2*len(a)-1) result.
    """
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.size - 1]


def _expand_bins(x: np.ndarray, bin_size: int, n: int) -> np.ndarray:
    """Repeat each bin value ``bin_size`` times and truncate to length ``n``."""
    a = np.repeat(x, bin_size)[:n]
    return a.astype(np.float64)


def _random_block_perturb(a: np.ndarray, rng: np.random.Generator) -> np.ndarray:
    """Multiplicative log‑normal perturbation of a random contiguous block,
    followed by a projection onto the simplex."""
    n = a.size
    width = rng.integers(1, max(2, n // 10))
    start = rng.integers(0, n - width + 1)
    factor = np.exp(rng.normal(0.0, 0.15))
    out = a.copy()
    out[start:start + width] *= factor
    out = np.clip(out, 0.0, None)
    s = out.sum()
    if s == 0.0:
        out[:] = 1.0 / n
    else:
        out /= s
    return out


def compute_J(a: np.ndarray) -> float:
    """
    Exact value of the evaluation objective (ignoring the clipping that
    ``evaluate_sequence`` performs, which does not affect feasible vectors).
    The vector is assumed to have non‑negative entries and a non‑zero sum.
    """
    n = a.size
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _self_conv(a, fft_len)
    max_b = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


def compute_grad(a: np.ndarray,
                 alpha: float,
                 fft_len_self: int,
                 fft_len_grad: int) -> np.ndarray:
    """
    Gradient of the soft‑max surrogate  φ(a)=log∑exp(α·(a*a−max)) .
    Returns ∇φ, a vector of the same length as ``a``.
    """
    # self‑convolution
    conv = _self_conv(a, fft_len_self)
    cmax = conv.max()
    shifted = conv - cmax                     # stabilisation
    w = np.exp(alpha * shifted)
    w_sum = w.sum()
    if w_sum == 0.0:
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    # convolution a * w_rev
    w_rev = w[::-1]
    fa = np.fft.rfft(a, n=fft_len_grad)
    fw = np.fft.rfft(w_rev, n=fft_len_grad)
    full = np.fft.irfft(fa * fw, n=fft_len_grad)
    start = w_rev.size - 1                     # = 2n‑2
    grad = 2.0 * full[start:start + a.size]
    return grad


# ----------------------------------------------------------------------
# Core optimisation routine for a given length
# ----------------------------------------------------------------------


def _optimize_length(n: int,
                     rng: np.random.Generator,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Multi‑scale optimisation for a fixed length ``n``.
    Returns the best sequence (as a full‑resolution ndarray) and its J value.
    """
    # FFT lengths that stay constant for this ``n``
    conv_len = 2 * n - 1
    fft_len_self = _next_pow2(conv_len)               # for a*a
    fft_len_grad = _next_pow2(3 * n - 2)               # for a * w_rev

    # ------------------------------------------------------------------
    # 0)  initialise a feasible starting point (uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_J = compute_J(best_a)

    # incorporate a known construction, if any, provided it has the same length
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
            if cand.shape[0] == n:
                cand = np.clip(cand, 0.0, None)
                cand_sum = cand.sum()
                if cand_sum > 0.0:
                    cand = cand / cand_sum
                    Jcand = compute_J(cand)
                    if Jcand < best_J:
                        best_J, best_a = Jcand, cand.copy()
                        print(f"Initial known sequence gives J={best_J:.6f}")
        except Exception:
            pass

    # ------------------------------------------------------------------
    # 1)  Multi‑scale hierarchy (coarse → fine)
    # ------------------------------------------------------------------
    # bin sizes must divide ``n``; we start with a very coarse grid.
    bin_sizes = [64, 32, 16, 8, 4, 2, 1]   # 64⋅64 = 4096 for the default n
    prev_x = None
    prev_bin = None

    start_time = time.monotonic()

    for bin_sz in bin_sizes:
        if time.monotonic() >= deadline:

        B = n // bin_sz                      # number of bins at this scale

        # ------------------------------------------------------------------
        #   a) generate a diverse pool of starters on this scale
        # ------------------------------------------------------------------
        pool = []

        for r in (0.99, 0.975, 0.95, 0.9, 0.85):

            pool.append(v)

        v /= v.sum()
        pool.append(v)

        # two spikes at the ends

        v[0] = 0.5
        v[-1] = 0.5
        pool.append(v)

        # Dirichlet draws with a few concentration values
        for conc in (0.1, 0.3, 0.7, 2.0):
            for _ in range(3):
                pool.append(rng.dirichlet(np.ones(B) * conc))

        # up‑sample the best solution from the previous (coarser) level
        if prev_x is not None:
            factor = prev_bin // bin_sz
            up = np.repeat(prev_x, factor)[:B]
            up = proj_simplex(up)
            pool.append(up)

        # ------------------------------------------------------------------
        #   b) short Adam runs for each starter (global exploration)
        # ------------------------------------------------------------------
        best_local_x = None
        best_local_J = float("inf")
        for init in pool:
            if time.monotonic() >= deadline:
                break
            x = proj_simplex(init.copy())
            m = np.zeros_like(x)
            v_ = np.zeros_like(x)

            # a modest number of iterations – enough to move off the
            # initial point but cheap enough to try many starters.
            short_iters = 80
            for it in range(short_iters):
                if time.monotonic() >= deadline:
                    break
                # schedule for the soft‑max temperature (smooth → sharp)
                frac = (it + 1) / short_iters
                alpha = 5.0 + 195.0 * (frac ** 2)

                a_full = _expand_bins(x, bin_sz, n)
                grad_full = compute_grad(a_full, alpha, fft_len_self, fft_len_grad)

                # aggregate the gradient back to the bin level (average per bin)
                grad_bin = np.add.reduceat(grad_full,
                                           np.arange(0, n, bin_sz)) / bin_sz

                # Adam step
                beta1, beta2, eps = 0.9, 0.999, 1e-8
                lr = 0.20 * (1 - frac) + 0.002 * frac
                m = beta1 * m + (1 - beta1) * grad_bin
                v_ = beta2 * v_ + (1 - beta2) * (grad_bin * grad_bin)
                m_hat = m / (1 - beta1 ** (it + 1))
                v_hat = v_ / (1 - beta2 ** (it + 1))
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
                x = proj_simplex(x)

            # evaluate the fully expanded vector
            a_candidate = _expand_bins(x, bin_sz, n)
            Jcand = compute_J(a_candidate)
            if Jcand < best_local_J:
                best_local_J, best_local_x = Jcand, x.copy()

        # ------------------------------------------------------------------
        #   c) longer refinement on the best candidate of this scale
        # ------------------------------------------------------------------
        if best_local_x is None:
            continue
        x = best_local_x.copy()
        m = np.zeros_like(x)
        v_ = np.zeros_like(x)

        long_iters = 400
        for it in range(long_iters):
            if time.monotonic() >= deadline:
                break
            frac = (it + 1) / long_iters

                                       np.arange(0, n, bin_sz)) / bin_sz

            m = beta1 * m + (1 - beta1) * grad_bin
            v_ = beta2 * v_ + (1 - beta2) * (grad_bin * grad_bin)

            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            x = proj_simplex(x)

        # evaluate after the long run
        a_refined = _expand_bins(x, bin_sz, n)
        Jref = compute_J(a_refined)
        if Jref < best_J:

    # ------------------------------------------------------------------

    m = np.zeros_like(a)
    v_ = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    it = 0
    last_improve = time.monotonic()
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()

        lr = 0.25 * (1 - frac) + 0.001 * frac
        # temperature schedule (smooth → sharp)
        alpha = 2.0 * (1 - frac) + 2000.0 * frac

        # gradient of the smooth surrogate
        grad = compute_grad(a, alpha, fft_len_self, fft_len_grad)

        m = beta1 * m + (1 - beta1) * grad
        v_ = beta2 * v_ + (1 - beta2) * (grad * grad)

        v_hat = v_ / (1 - beta2 ** it)

        # evaluate the true objective every few iterations

            if Jcur < best_J - 1e-12:

                last_improve = now
                elapsed = now - start_time
                print(f"[{elapsed:6.2f}s] Full‑res new best J = {best_J:.6f}")

            # restart if no progress for a while
            if now - last_improve > 30.0:
                a = best_a.copy()
                noise = np.exp(rng.normal(0.0, 0.07, size=n))
                a = a * noise
                a = proj_simplex(a)
                last_improve = now
                elapsed = now - start_time

# ----------------------------------------------------------------------


def propose_candidate(seed: int = 12345,
                     budget_s: float = 1000.0,
                     **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The algorithm runs a coarse‑to‑fine projected‑Adam
    optimisation for several problem sizes and returns the best sequence found
    before the time budget expires.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(0.0, float(budget_s) - 5.0)   # safety margin

    # ------------------------------------------------------------------
    # Choose a handful of lengths to explore (the evaluation scales linearly
    # with the length, so very large n is costly).  1024, 2048 and 4096 give a
    # good trade‑off while still covering the previous construction length.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]
    if "height_sequence_1" in globals():
        try:
            n0 = len(globals()["height_sequence_1"])
            if n0 not in lengths:
                lengths.append(n0)
        except Exception:
            pass

    best_overall_J = float("inf")
    best_overall_seq = None
    start_time = time.monotonic()

    for n in lengths:
        if time.monotonic() >= deadline:
            break
        print(f"--- Optimising length {n} ---")
        seq, J = _optimize_length(n, rng, deadline)
        if J < best_overall_J:
            best_overall_J, best_overall_seq = J, seq.copy()
            elapsed = time.monotonic() - start_time
            print(f"[{elapsed:6.2f}s] Overall new best J={best_overall_J:.6f} (n={n})")

    # Clip to the limits enforced by ``evaluate_sequence`` (just a safety net)
    if best_overall_seq is None:
        # fall back to a trivial uniform sequence (should never happen)
        best_overall_seq = np.full(512, 1.0 / 512, dtype=np.float64)
    best_overall_seq = np.clip(best_overall_seq, 0.0, 1000.0)

    return [float(x) for x in best_overall_seq]
